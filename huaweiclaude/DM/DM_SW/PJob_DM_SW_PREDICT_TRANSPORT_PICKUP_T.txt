with SALES_FORECAST_FACT_DATA as(  -- 销售预测的实际数据
	SELECT t1.stat_month ,
		   t1.operation_center_code,
		   t1.material_num,
		   sum(t1.actual_billed_qty) as actual_billed_qty
	FROM DM_SW.DM_OTD_SALES_FORECAST_FACT_T t1
	where t1.def_channnel = '经销'
	and t1.operation_center_name is not null
	and t1.material_num is not null
	and t1.stat_month <= substring('${PERIOD_ID_D}',1,7)
	GROUP BY t1.stat_month ,
		     t1.operation_center_code,
		     t1.material_num
),
SALES_FORECAST_DIM_DATA as(   -- 预测数据发散，为了用函数算近36个月的数据,从预测近37个月开发散
  select SUBSTRING(day_dt,1,7) as stat_month
  from DM.dm_dim_date_d T
  where day_dt between add_months('${PERIOD_ID_D}',-12) and '${PERIOD_ID_D}' 
  group by SUBSTRING(day_dt,1,7)
),
SALES_FORECAST_FACT_DATA_01 as( -- 预测实际明细数据
	SELECT t1.stat_month,
	       t2.operation_center_code,
	       t2.material_num,
	       nvl(t3.actual_billed_qty,0) as actual_billed_qty
	FROM SALES_FORECAST_DIM_DATA t1
	left join(
		select distinct 
		       operation_center_code,
		       material_num
		from SALES_FORECAST_FACT_DATA
	) t2
	on 1 = 1
	left join SALES_FORECAST_FACT_DATA t3
	on t1.stat_month = t3.stat_month
	and nvl(t2.operation_center_code,'ISNULL') = nvl(t3.operation_center_code,'ISNULL')
	and nvl(t2.material_num,'ISNULL') = nvl(t3.material_num,'ISNULL')
),

near12_avg_pickup_goods_data as(  -- 近12个月月均提货数&近3个月的累计提货数
	select t1.stat_month,
	       t1.operation_center_code,
	       t1.material_num,
	       t1.actual_billed_qty,
	       ceil(
	             (
	               sum(nvl(t1.actual_billed_qty,0)) over(PARTITION by t1.operation_center_code,t1.material_num 
			 	   order by t1.stat_month asc rows between 11 PRECEDING AND CURRENT ROW) -
			 	   max(t1.actual_billed_qty) over(PARTITION by t1.operation_center_code,t1.material_num 
				   order by t1.stat_month asc rows between 11 PRECEDING AND CURRENT ROW) - 
				   min(t1.actual_billed_qty) over(PARTITION by t1.operation_center_code,t1.material_num 
				   order by t1.stat_month asc rows between 11 PRECEDING AND CURRENT ROW)
				 )/10
			   ) as near12_avg_pickup_goods_qty, -- 近12个月月均提货数
		   SUM(t1.actual_billed_qty) over(PARTITION by t1.operation_center_code,t1.material_num 
	 	       order by t1.stat_month asc rows between 2 PRECEDING AND CURRENT ROW) as near3_lj_pickup_goods_qty-- 近3个月的累计提货数
	from SALES_FORECAST_FACT_DATA_01 T1
