delete from dm.dm_sw_business_center_achievement_daily_t where calday=to_char(to_date('${PERIOD_ID_D}'),'yyyyMMdd');

with region_title as (--前端固定显示
    select distinct node_desc5 region
    from dm.dm_rpt_sales_group_t
    where node_name4 = 'H03230613'
      and node_name5 in ('H05230627', 'H05230628', 'H05230629', 'H05230630','H05250102')
),
     business_type_title as (--前端固定显示
         select '直营KA' business_type, 1 seq
         union all
         select '工程经销商' business_type, 2 seq
--         union all
--         select '新签约平台' business_type, 3 seq
         union all
         select '运营中心' business_type, 3 seq
     ),
     title as (
         select *
         from region_title,
              business_type_title
         union all
         select '1+N', '1+N' business_type, 998 seq
         union all
         select '未分类', '未分类', 999
     ),
     /*战略工程归属大区 as (--写死用户
         select emp_num, '卫浴华东大区' new_region
         from dm.dm_wf_emp_info_t
         where emp_num in ('10003663', '15000664', '15001181')
         union all
         select emp_num, '卫浴华南大区' new_region
         from dm.dm_wf_emp_info_t
         where emp_num in ('15001327')
         union all
         select emp_num, '卫浴华西大区' new_region
         from dm.dm_wf_emp_info_t
         where emp_num in ('15001586')
--         union all 
--         select emp_num, '卫浴东南大区' new_region
--         from dm.dm_wf_emp_info_t
--         where emp_num in ('15002023')
         
     )*/
     ,dataresult as (
         select region,
                business_type,
                sum(achievement_day)             achievement_day,
                sum(achievement_month)           achievement_month,
                sum(achievement_year)            achievement_year,
                sum(achievement_last_year_month) achievement_last_year_month,
                sum(achievement_last_year)       achievement_last_year,
                sum(no_tax_month)                no_tax_month,
                sum(no_tax_year)                 no_tax_year,
                sum(sales_cost_month)            sales_cost_month,
                sum(sales_cost_year)             sales_cost_year
         from dm.dm_sw_business_center_achievement_daily_F
         where calday = to_char(to_date('${PERIOD_ID_D}'),'yyyyMMdd')
         group by region, business_type
         having sum(achievement_year) != 0
             or sum(achievement_last_year) != 0
         order by region,
                  business_type
                  
                  ),
     target as (
         select *
         from upload.upload_sw_commercial_target_t
         where 1 = 1
           and stat_month = to_char(to_date('${PERIOD_ID_D}'), 'yyyy-MM')
     )
 insert into dm.dm_sw_business_center_achievement_daily_t
select coalesce(a.region, '未分类')        region,
       coalesce(a.business_type, '未分类') business_type,
       b.achievement_day,
       b.achievement_month,
       b.achievement_year,
       b.achievement_last_year_month,
       b.achievement_last_year,
       b.no_tax_month,
       b.no_tax_year,
       b.sales_cost_month,
       b.sales_cost_year,
       to_char(to_date('${PERIOD_ID_D}'),'yyyyMMdd') calday,
       current_timestamp()
from title a
         right join dataresult b
                    on coalesce(a.region, '未分类') = coalesce(b.region, '未分类') and
                       coalesce(a.business_type, '未分类') = coalesce(b.business_type, '未分类')
 order by case when strpos(coalesce(a.region, '未分类'), '大区') > 0 then 1 else 2 end, a.seq,
         sum(coalesce(b.achievement_month, 0)) over (partition by a.region) desc
