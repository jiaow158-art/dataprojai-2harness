delete from DM_SW.DM_SO_TRANSIT_INVENTORY_ALL_T 
where stat_month >=  CASE WHEN substring('${PERIOD_ID_D}',9,2) IN ('01','02','03') THEN 
                               SUBSTRING(ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3),1,7) -- 传昨天，所在月份1号的前3个月
                           ELSE
                    		   SUBSTRING('${PERIOD_ID_D}',1,7) -- 传昨天，当月
                       END;
INSERT INTO DM_SW.DM_SO_TRANSIT_INVENTORY_ALL_T
select t1.stat_month,
	   t1.cust_code,
	   t1.cust_short_name,
	   t1.sap_delivery_num,
	   t1.material_num,
	   t2.material_name,
	   t1.pickup_goods_date,
	   t1.in_stock_date,
	   t1.pickup_goods_qty,
	   t1.in_stock_qty,
	   case when t1.in_transit_qty < 0 then 0 else t1.in_transit_qty end in_transit_qty,
	   current_timestamp() as dw_last_update_date ,
	   case when t1.in_transit_amt < 0 then 0 else t1.in_transit_amt end in_transit_amt,
	   t1.return_date,
	   t1.return_qty
from DM_SW.DM_SO_TRANSIT_INVENTORY_T t1
join dwimd.dwi_md_data_material_general_t t2
on t1.material_num = t2.material_num
where t2.product_level1_code in ('30','40')
and t2.external_matl_group_code <> '10050'
and t1.stat_month >= CASE WHEN substring('${PERIOD_ID_D}',9,2) IN ('01','02','03') THEN 
                           SUBSTRING(ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3),1,7) -- 传昨天，所在月份1号的前3个月
                       ELSE
                		   SUBSTRING('${PERIOD_ID_D}',1,7) -- 传昨天，当月
                   END;
