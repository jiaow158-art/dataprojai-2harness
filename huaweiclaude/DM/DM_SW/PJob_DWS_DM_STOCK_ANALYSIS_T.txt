delete from dm_sw.dm_stock_analysis_t WHERE stat_month = substr('${PERIOD_ID_D}',1,7);
insert into dm_sw.dm_stock_analysis_t
with mat as (
select b.material_num,b.material_name,b.bathroom_channal as product_channal,b.matl_group_code,b.product_level_code
,bath_category_name as category_name
,case when (c.matl_group2_code = 'J10') then '退市库存' else '常规库存' end as 库存状态
from dwimd.dwi_md_data_material_general_t as b 
left join (select t.material_num,max(t.matl_group2_code) as matl_group2_code from dwimd.dwi_md_data_material_sales_t_his as t 
where start_date < last_day(to_date( concat( substr('${PERIOD_ID_D}',1,4),'-',substr('${PERIOD_ID_D}',6,2),'-','01'),'YYYY-MM-DD'))
and end_date >= last_day(to_date( concat( substr('${PERIOD_ID_D}',1,4),'-',substr('${PERIOD_ID_D}',6,2),'-','01'),'YYYY-MM-DD'))
group by t.material_num ) c on b.material_num = c.material_num
left join upload.upload_wm_sw_sap_prod_grp_t as k on b.product_level3_code = k.prod_code3 
)
,
tmp1 as (
select 物料编码,
	   物料描述,
       品类,
       产品渠道,
       库位分类,
       外协自产,
       库存地点,
       库存状态,
       case when 库存月份 = substr(regexp_replace(concat(substr('${PERIOD_ID_D}',1,7),'-01')+ INTERVAL '-1 month','\D','','g'),1,6) then 月末库存数量 else 0 end as 期初库存数量, --所选上一月
       case when 库存月份 = substr(regexp_replace(concat(substr('${PERIOD_ID_D}',1,7),'-01')+ INTERVAL '-1 month','\D','','g'),1,6) then 月末库存金额/10000 else 0 end as 期初库存金额,
       case when 库存月份 = regexp_replace(substr('${PERIOD_ID_D}',1,7),'\D','','g') then 月末库存数量 else 0 end as 期末库存数量, --所选月
       case when 库存月份 = regexp_replace(substr('${PERIOD_ID_D}',1,7),'\D','','g') then 月末库存金额/10000 else 0 end as 期末库存金额,
       case when 库存月份 = substr(regexp_replace(concat(substr('${PERIOD_ID_D}',1,7),'-01')+ INTERVAL '-1 year','\D','','g'),1,6) then 月末库存数量 else 0 end as 去年当期期末库存数量, --所选月上一年
       case when 库存月份 = substr(regexp_replace(concat(substr('${PERIOD_ID_D}',1,7),'-01')+ INTERVAL '-1 year','\D','','g'),1,6) then 月末库存金额/10000 else 0 end as 去年当期期末库存金额
  from (
 SELECT 库存月份,物料编码,m.category_name as 品类,m.product_channal as 产品渠道,m.material_name as 物料描述,m.库存状态,库位分类,库存地点,外协自产,sum(月末库存金额+月末外协寄售金额) 月末库存金额,sum(月末库存数量+月末外协寄售数量) 月末库存数量 
 FROM  dm.dm_own_inventory_t a  
 left join mat m on a.物料编码 = m.material_num
 where a."工厂" not like '39%' and a.库存月份 >= '202101' and 库存地点 in ('GA','TJ','WX','GYCK')
 group by 库存月份,物料编码,m.category_name,m.product_channal,库位分类,外协自产,库存地点,m.库存状态,m.material_name
) t1
)
,tmp as (
	select
	 物料编码
	,物料描述
	,品类
	,产品渠道
	,库位分类
	,库存地点
	,库存状态
	,coalesce(sum(期初库存数量),0) as 期初库存数量
	,coalesce(sum(期末库存数量),0) as 期末库存数量
    ,coalesce(sum(期初库存金额),0) as 期初库存金额
	,coalesce(sum(期末库存金额),0) as 期末库存金额
	,coalesce(SUM(去年当期期末库存数量),0) as 去年当期期末库存数量
	,coalesce(SUM(去年当期期末库存金额),0) as 去年当期期末库存金额
	,sum(case when 外协自产 = '自产' then 期初库存数量 else 0 end ) as 期初自产数量
	,sum(case when 外协自产 = '外协' then 期初库存数量 else 0 end ) as 期初外协数量
	,sum(case when 外协自产 = '自产' then 期末库存数量 else 0 end ) as 期末自产数量
	,sum(case when 外协自产 = '外协' then 期末库存数量 else 0 end ) as 期末外协数量
	,sum(case when 外协自产 = '自产' then 期初库存金额 else 0 end ) as 期初自产金额
	,sum(case when 外协自产 = '外协' then 期初库存金额 else 0 end ) as 期初外协金额
	,sum(case when 外协自产 = '自产' then 期末库存金额 else 0 end ) as 期末自产金额
	,sum(case when 外协自产 = '外协' then 期末库存金额 else 0 end ) as 期末外协金额
	from tmp1
	group by 物料编码,物料描述,品类,产品渠道,库位分类,库存地点,库存状态
)
,tmp2 as (
 select 
 a.MATERIAL_NUM as 物料编码
,m.material_name as 物料描述
,m.category_name as 品类
,m.product_channal as 产品渠道
,i.location_category as 库位分类
,m.库存状态
,sum(a.ACTUAL_BILLED_QTY) as 实际数量
,sum(case when b.期初库存数量 <> 0 then b.期初库存金额 / b.期初库存数量 * a.ACTUAL_BILLED_QTY
    when c.mov_avg_price <> 0 and c.price_unit <> 0 then c.mov_avg_price * a.ACTUAL_BILLED_QTY / c.price_unit / 10000
    when d.mov_avg_price <> 0 and d.price_unit <> 0 then d.mov_avg_price * a.ACTUAL_BILLED_QTY / d.price_unit / 10000 end ) as 实际成本 
from DM.DM_GL_OUT_INVOICE_T as a
left join tmp as b on a.MATERIAL_NUM = b.物料编码 and a.INV_LOCATION = b.库存地点 
left join mat as m on a.MATERIAL_NUM = m.material_num
left join DM.DM_RPT_STOCK_PLACE_T as i on a.INV_LOCATION = i.stock_place_name and i.stock_place_flag = '库存地点不为空' and i.stock_place_name is not null 
left join DWRDIM.DWR_DIM_MATERIAL_ACCT_HIS_D c on a.MATERIAL_NUM = c.MATERIAL_NUM and a.factory_code = c.value_range and a.batch_num = c.assess_type and c.year_mon = substr('${PERIOD_ID_D}',1,7)
left join DWRDIM.DWR_DIM_MATERIAL_ACCT_HIS_D d on a.MATERIAL_NUM = d.MATERIAL_NUM and a.factory_code = d.value_range and d.assess_type is null and d.year_mon = substr(concat(substr('${PERIOD_ID_D}',1,7),'-01')+ INTERVAL '-1 month',1,7)
where 1=1 and a.INV_LOCATION in ('GA','TJ','WX','GYCK') 
and a.sale_org_code between '3210' and '3920'
and a.payer_id not in ('0000003200','0000003300','0000003500','0000003600','0000003700','0000003800','0000003900','0000003A00') --不包含公司间
and substr(a.BILL_DATE,1,7) = substr('${PERIOD_ID_D}',1,7)
group by 
 a.MATERIAL_NUM
,m.material_name
,m.category_name 
,m.product_channal 
,i.location_category
,m.库存状态
--,a.factory_code
)
select 
     substr('${PERIOD_ID_D}',1,7)  stat_month
	,coalesce(a.物料编码,b.物料编码) 物料编码
	,coalesce(a.物料描述,b.物料描述) 物料描述
	,coalesce(a.品类,b.品类) 品类
	,coalesce(a.产品渠道,b.产品渠道) 产品渠道
	,coalesce(a.库位分类,b.库位分类) 库位分类
	,coalesce(a.库存状态,b.库存状态) 库存状态
	,coalesce(期初库存数量,0) 期初库存数量
	,coalesce(期末库存数量,0) 期末库存数量
	,coalesce(去年当期期末库存数量,0) 去年当期期末库存数量
	,coalesce(期初自产数量,0) 期初自产数量
	,coalesce(期初外协数量,0) 期初外协数量
	,coalesce(期末自产数量,0) 期末自产数量
	,coalesce(期末外协数量,0) 期末外协数量
    ,coalesce(期初库存金额,0) 期初库存金额
	,coalesce(期末库存金额,0) 期末库存金额
	,coalesce(去年当期期末库存金额,0) 去年当期期末库存金额
	,coalesce(期初自产金额,0) 期初自产金额
	,coalesce(期初外协金额,0) 期初外协金额
	,coalesce(期末自产金额,0) 期末自产金额
	,coalesce(期末外协金额,0) 期末外协金额
	,coalesce(b.实际数量,0) as 销量数量
	,coalesce(b.实际成本,0) as 销量金额
	,CURRENT_TIMESTAMP()	as dw_last_update_date
 from tmp a
 full outer join tmp2 b on a.物料编码 = b.物料编码 and a.品类 = b.品类 and a.产品渠道 = b.产品渠道 and a.库位分类 = b.库位分类 
 where 1=1
