delete from dm_sw.dm_retail_store_warning_t
where start_month = substr(TO_DATE('${PERIOD_ID_D}','YYYY-MM-DD'), 1, 7);
INSERT INTO dm_sw.dm_retail_store_warning_t
with A as (
    select          a.marketing_department                                                sales_center,  
                    d.emp_num                                                             salesman_id,
                    d.emp_name                                                            salesman_name,
                    a.distributor_code                                                    distributor_code,
                    e.b1shortname                                                         sales_dep,
                    a.distributor_code || '-' || e.b1shortname                            distributor_join,
                    a.store_code                                                          store_code,
                    a.store_name                                                          store_name,
                    a.store_address                                                       store_address 
    from DWIMD.DWI_MD_RTM_VALID_STORE_DATA_T a
             left join dm.DM_WF_EMP_INFO_T d on a.salesman_code = d.emp_num
             left join upload.upload_shopcustcode_b1shortname e
                       on lpad(a.customer_code, 10, '0') = lpad(e.custcode, 10, '0')
    where a.del_flag = 'N'
      and store_status = 'STORE_STATUS_01'
      and nvl(store_channel, 'RETAIL') = 'RETAIL'
      and a.store_type in ('STORE_TYPE_01','STORE_TYPE_02')
    group by marketing_department, d.emp_num , d.emp_name 
    , a.distributor_code, a.distributor_code || '-' || e.b1shortname , a.store_code ,a.store_name,a.store_address ,e.b1shortname
       )
       
,B as (      
select 
    substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'),1,7) start_month,
    a.sales_center,
    a.sales_dep,
    a.store_name,
    a.store_code,
    a.store_address,
    a.salesman_id,
    a.salesman_name,
    sum(case when to_char(b.doc_date, 'yyyy-MM') =  SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) then b.line_total_amt else '0' end )  recent1_month_amt, -- 
    sum(case when to_char(b.doc_date, 'yyyy-MM') <= SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM')>=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '2 month', 1, 7) then b.line_total_amt else '0' end )  recent2_month_amt, -- 
    sum(case when to_char(b.doc_date, 'yyyy-MM') <= SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM')>=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '3 month', 1, 7) then b.line_total_amt else '0' end ) recent3_month_amt -- 
from A a
      left join DM.DM_WM_B1_SALES_ORDER_T b
                on (b.cust_code || '-' || b.cust_byname = a.distributor_join and b.cust_code != a.store_code) 
                    and customer_type = 'SW'
      left join UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T c
                on b.material_num = c.material_num
where (b.CUR_MATL_GROUP_CODE <> '1000010'or b.CUR_MATL_GROUP_CODE is null)
and  to_char(b.doc_date, 'yyyy-MM') <=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM') >=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '3 month', 1, 7)
and c.material_type in ('智能', '大五金', '卫浴件', '小五金', '浴室柜', '休闲', '陶瓷', '其它')
group by 
a.sales_center,
a.sales_dep,
a.store_name,
a.store_code,
a.store_address,
a.salesman_id,
a.salesman_name

union all 

select 
    substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'),1,7) start_month,
    a.sales_center,
    a.sales_dep,
    a.store_name,
    a.store_code,
    a.store_address,
    a.salesman_id,
    a.salesman_name,
    sum(case when to_char(b.doc_date, 'yyyy-MM') =  SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) then b.line_total_amt else '0' end )  recent1_month_amt, -- 
    sum(case when to_char(b.doc_date, 'yyyy-MM') <= SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM')>=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '2 month', 1, 7) then b.line_total_amt else '0' end )  recent2_month_amt, -- 
    sum(case when to_char(b.doc_date, 'yyyy-MM') <= SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM')>=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '3 month', 1, 7) then b.line_total_amt else '0' end ) recent3_month_amt -- 
from A a
      left join DM.DM_WM_B1_SALES_ORDER_T b
                on b.cust_code = a.store_code
                    and customer_type = 'SW'
      left join UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T c
                on b.material_num = c.material_num
where 1=1
and (b.CUR_MATL_GROUP_CODE <> '1000010'or b.CUR_MATL_GROUP_CODE is null)
 and  to_char(b.doc_date, 'yyyy-MM') <=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) and to_char(b.doc_date, 'yyyy-MM') >=SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '3 month', 1, 7)
and c.material_type in ('智能', '大五金', '卫浴件', '小五金', '浴室柜', '休闲', '陶瓷', '其它')
group by 
a.sales_center,
a.sales_dep,
a.store_name,
a.store_code,
a.store_address,
a.salesman_id,
a.salesman_name
)

,C as (
select 
	start_month,
	sales_center,
    sales_dep,
    store_name,
    store_code,
    store_address,
    salesman_id,
    salesman_name,
    case when sum(recent1_month_amt) = 0 then 1 else 0 end as recent1_month,
    case when sum(recent2_month_amt) = 0 then 1 else 0 end as recent2_month,
    case when sum(recent3_month_amt) = 0 then 1 else 0 end as recent3_month
from B
--where store_code = 'DPWY03700124'
group by 
	start_month,
	sales_center,
	sales_dep,
	store_name,
	store_code,
	store_address,
	salesman_id,
	salesman_name
)

select 
	start_month,
    sales_center,
    sales_dep,
    store_name,
    store_code,
    store_address,
    salesman_id,
    salesman_name,
    case when recent3_month = 1 or recent2_month = 1 then 0 else recent1_month end recent1_month,
    case when recent3_month = 1   then 0 else recent2_month end recent2_month,
    recent3_month,
    current_timestamp() as dw_last_update_date
from C
