--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
with material_data as(
	select 
		 b.material_num,
		 b.product_level3_code ,
		 b.product_level3_name,
		 b.matl_type_code,
		 c.matl_group4_code
	from dwimd.dwi_md_data_material_general_t as b 
	left join (select material_num,max(matl_group4_code) as matl_group4_code 
	from dwimd.dwi_md_data_material_sales_t group by material_num ) c 
	on b.material_num = c.material_num
	where b.matl_type_code ='FERT'
--	and (product_level3_code like '4012%' or product_level3_code like '4014%')
)
,tmp_data_01 as(  -- 龙头厂、沐浴房、高明数据
	select substr(t1.voucher_post_date,1,10) as start_date,
	       t1.factory_werks_code,
		   t2.product_level3_code,
	       t2.product_level3_name,
	       t2.matl_group4_code,
	       t1.material_num ,
	       t1.material_name,
	       sum(case when t1.move_type in ('101','911','162','123','131')  then t1.quantity  
	            	when t1.move_type in ('102','912','161','122','132')  then - t1.quantity
	            else 0 end ) as product_num
	from DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F t1
	join material_data t2
	on t1.material_num = t2.material_num 
	left join (select distinct purchase_voucher_num,purchase_voucher_type from DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F) t3 
	on  t1.purchase_voucher_num = t3.purchase_voucher_num
	where t1.factory_werks_code in ('3500','3700','3200')
	and substr(t1.voucher_post_date,1,7) >= '2022-06'
	and (substr(t1.supplier_or_creditor_acct,1,6) <> '000000' or t1.supplier_or_creditor_acct is null )
	and (t3.purchase_voucher_type <> 'ZSUB' or t3.purchase_voucher_type is null )
	group by substr(t1.voucher_post_date,1,10),
	         t1.factory_werks_code,
	         t2.product_level3_code,
	         t2.product_level3_name,
	         t2.matl_group4_code,
	         t1.material_num ,
	         t1.material_name
)
,tmp_data_02 as( 
select t.start_date,
       t.factory_werks_code,
       t.product_level3_code,
       t.product_level3_name,
       t.material_num ,
	   t.material_name,
       case when t.material_name LIKE '%纳米%' and t.material_name like '%银裳%' then '纳米'
		      when t.product_level3_code = '4012001' then '花洒'
		      when t.product_level3_code like '4012%' and t.product_level3_code <> '4012001' then '龙头'
		      when t.product_level3_code like '4014%' and t.matl_group4_code in ('DJ','DZ') then '淋浴房-非标'
		      when t.product_level3_code like '4014%' and t.matl_group4_code not in ('DJ','DZ') then '淋浴房-常规'
		      when t.factory_werks_code = '3700' and t.matl_group4_code in ('DJ','DZ') then '高明-非标' 
	          when t.factory_werks_code = '3700' and t.matl_group4_code not in ('DJ','DZ') then '高明-常规'
	     else t.product_level3_name  end as project_name,
       t.matl_group4_code,
       t.product_num
from tmp_data_01 t
where factory_werks_code in ('3500','3700')
)
,tmp_data_03 as( 
select t.start_date,
       t.project_name as product_level3_name,
       sum(t.product_num) as product_num
from tmp_data_02 t
where t.project_name in('纳米','花洒','龙头','淋浴房-非标','淋浴房-常规','高明-非标','高明-常规')
group by t.start_date,
         t.project_name
)     
,tmp_data_04 as(
SELECT T1.day_dt as start_date,  -- 日期
	   t3.product_level3_name,  -- 产品层次3
	   nvl(t2.product_num,0) as product_qty  -- 产品数量
FROM DM.dm_dim_date_d  T1
left join (select distinct product_level3_name from tmp_data_03) t3
on 1 = 1
left join tmp_data_03 t2
on T1.day_dt = t2.start_date
and t3.product_level3_name = t2.product_level3_name
WHERE T1.day_dt >= '2022-06-01'
AND T1.day_dt <= date_sub(CURRENT_DATE() ,1) --CURRENT_DATE()-- date_sub(CURRENT_DATE() ,1)
),
tmp_data_01_02 as(   -- 江西江门基地生产入库数据
	SELECT t.start_date,
		   t.factory_werks_code,
		   t.product_level3_code,
	       t.product_level3_name,
	       t.material_num ,
		   t.material_name,
		   case when t.product_level3_code not in ('3007001','3006003','3006001','3006004','3006002') then '其他'
		        else t.product_level3_name  end as project_name, -- 智能齐套一体机  蹲便器  坐便器 小便器  盆  
		   t.matl_group4_code,
	       t.product_num     
	FROM tmp_data_01 T   -- 生产入库 (江西/江门基地)
	where t.factory_werks_code in ('3200','3500')   -- 江西3200 江门3500
	and t.product_level3_code like '30%'
),
tmp_data_01_03 as(
select t.start_date,
       t.factory_werks_code,
       t.project_name as product_level3_name,
       sum(t.product_num) as product_num
from tmp_data_01_02 t
group by t.start_date,
         t.factory_werks_code,
         t.project_name
),
tmp_data_01_04 as(
SELECT T1.day_dt as start_date,  -- 日期
       t3.factory_werks_code, -- 工厂
	   t3.product_level3_name,  -- 产品层次3
	   nvl(t2.product_num,0) as product_qty  -- 产品数量
FROM DM.dm_dim_date_d  T1
left join (select distinct factory_werks_code,product_level3_name from tmp_data_01_03) t3
on 1 = 1
left join tmp_data_01_03 t2
on T1.day_dt = t2.start_date
and t3.factory_werks_code = t2.factory_werks_code
and t3.product_level3_name = t2.product_level3_name
WHERE T1.day_dt >= '2022-06-01'
AND T1.day_dt <= date_sub(CURRENT_DATE() ,1)--CURRENT_DATE()
)
INSERT OVERWRITE TABLE DM_SW.DM_PAM_BASE_MANAGE_RPT_T01
select  CASE WHEN T.product_level3_name in ('花洒','龙头','纳米')  THEN '龙头厂'
             WHEN T.product_level3_name in ('淋浴房-非标','淋浴房-常规') THEN '淋浴房'
             WHEN T.product_level3_name in ('高明-非标','高明-常规') THEN '高明'
             END   AS PICK_UP_BASE_NAME, -- 提取基地
		t.start_date,
        case when T.product_level3_name in ('淋浴房-非标','淋浴房-常规') then  replace(t.product_level3_name,'淋浴房-','')
             when T.product_level3_name in ('高明-非标','高明-常规')then  replace(t.product_level3_name,'高明-','')
        else t.product_level3_name end as product_level3_name,
        t.product_qty,
        CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
from tmp_data_04 t   -- 龙头厂、淋浴房、高明 数据
union all
select case when t.factory_werks_code = '3200' then '江西基地'
            when t.factory_werks_code = '3500' then '江门基地'
        else '' end as PICK_UP_BASE_NAME, -- 提取基地
        t.start_date,  -- 日期
        t.product_level3_name,  -- 产品层次3
        t.product_qty,
        CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
from tmp_data_01_04 t  -- 江西基地、江门基地 数据
