truncate table dm_sw.DM_SW_CUST_SALE_STAT_T; 
with 子母客编关联 as (select distinct '汇总'                                                            一级,
                                bezei                                                           二级,
                                c.txtmd                                                         三级,
                                kunnr                                                           子客编,
                                case kunnr when '0000321540' then '1000004847' else zzkunnr end 客编
                from dwipi.DWI_WF_ZSD_ZMKB_WY_T a
                         left join (select distinct "ACCOUNT_NUMBER" customer, "NAME" txtmd
                                    from dm.dm_rpt_zmaster_cus_t) c
                                   on case kunnr when '0000321540' then '1000004847' else zzkunnr end = c.customer
                where vkbur in ('J001', 'J002', 'J004', 'J006', 'J016', 'J018', 'J019')
),
     data as (
         select 二级,
                客编,
                三级,
                sum(当期对厂提货销售额)   当期对厂提货销售额,
                sum(年累计对厂提货销售额)  年累计对厂提货销售额,
                sum(当期B1出货销售额)   当期B1出货销售额,
                sum(当期云订货下单销售额)  当期云订货下单销售额,
                sum(年累计B1出货销售额)  年累计B1出货销售额,
                sum(年累计云订货下单销售额) 年累计云订货下单销售额,
                sum(现有库存金额)      现有库存金额,
                sum(上月库存金额)      上月库存金额
         from (
                  select 二级,
                         客编,
                         三级,
                         sum(case
                                 when calday between to_char(to_date('${PERIOD_ID_D}','yyyy-MM-dd'),'yyyyMM01') 
								             and replace('${PERIOD_ID_D}','-','') then ambperformance
                                 else '0' end) / 10000 当期对厂提货销售额,
                         sum(ambperformance) / 10000            年累计对厂提货销售额,
                         0                             当期B1出货销售额,
                         0                             当期云订货下单销售额,
                         0                             年累计B1出货销售额,
                         0                             年累计云订货下单销售额,
                         0                             现有库存金额,
                         0                             上月库存金额
                  from DM.dm_ambv2_zzt_xssr_t a
                           join 子母客编关联 b on lpad(a.customer,10, '0') = b.子客编
						   join dm.dm_rpt_zmaster_cus_t bb
                             on a.salesorg = bb."SALES_ORGANIZATION_CODE" 
							 and bb."SALES_CHANNEL_CODE" = a.distr_chan
                             and a.customer = bb."ACCOUNT_NUMBER"
							 and bb."PRODUCT_GROUP_CODE" = '00'
							 and bb."PARVW" = 'AG'
                           join dm.dm_rpt_sales_group_t c on bb."SALES_GROUP_CODE" = c.node_name9
                           left join upload.upload_keyproducts d on a.distr_chan = d.原名称 and d.类型 = '卫浴渠道'
                  where calday between left('${PERIOD_ID_D}',4) || '0101' and replace('${PERIOD_ID_D}','-','')
                    and c.node_name2 = 'H11000002'
					and c.node_name4 = 'H03230643'
                    and d.新名称 = '零售'
                  group by 二级, 客编, 三级
                  union all
                  select 二级,
                      客编,
                      三级,
                      0 当期对厂提货销售额,
                      0 年累计对厂提货销售额,
                      sum(
                      case
                      when doc_date between to_char(to_date('${PERIOD_ID_D}','yyyy-MM-dd'),'yyyy-MM-01') and '${PERIOD_ID_D}'
                      then line_total_amt
                      else '0' end)
                      / 10000 当期B1出货销售额,
                      sum(
                      case
                      when cloud_order_flag = '是' and
                      to_char(doc_date, 'yyyyMMdd') between to_char(to_date('${PERIOD_ID_D}','yyyy-MM-dd'),'yyyy-MM-01') and '${PERIOD_ID_D}'
                      then line_total_amt
                      else '0' end)
                      / 10000 当期云订货下单销售额,
                      sum(line_total_amt) / 10000 年累计B1出货销售额,
                      sum(case
                      when cloud_order_flag = '是' then line_total_amt
                      else '0' end) / 10000 年累计云订货下单销售额,
                      0 现有库存金额,
                      0 上月库存金额
                  from dm.DM_WM_B1_SALES_ORDER_T a
                      join 子母客编关联 b
                  on lpad(a.dealer_code, 10, '0') = b.子客编
                      left join UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T c
                      on A.material_num = c.material_num
                  where customer_type = 'SW'
                    and (A.CUR_MATL_GROUP_CODE <> '1000010'
                     or A.CUR_MATL_GROUP_CODE is null)
                    and c.material_type in ('智能'
                      , '大五金'
                      , '卫浴件'
                      , '小五金'
                      , '浴室柜'
                      , '休闲'
                      , '陶瓷'
                      , '其它')
                    and left(doc_date,4) = left ('${PERIOD_ID_D}',4)
                    and doc_date <= '${PERIOD_ID_D}'
                  group by 二级, 客编, 三级
                  union all
                  select 二级,
                      客编,
                      三级,
                      0 当期对厂提货销售额,
                      0 年累计对厂提货销售额,
                      0 当期B1出货销售额,
                      0 当期云订货下单销售额,
                      0 年累计B1出货销售额,
                      0 年累计云订货下单销售额,
                      sum(
                      case
                      when stat_date = '${PERIOD_ID_D}'
                      then inventory_amount
                      else '0' end)
                      / 10000 现有库存金额,
                      sum(
                      case
                      when stat_date =
                      to_char(to_date('${PERIOD_ID_D}') - interval '1 month', 'yyyy-MM-dd')
                      then inventory_amount
                      else '0' end)
                      / 10000 上月库存金额
                  from DM.DM_WM_B1_WYJX_INVENTORY_T a
                      join 子母客编关联 b
                  on lpad(a.dealer_code, 10, '0') = b.子客编
                      left join UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T c
                      on A.material_num = c.material_num
					  left join upload.upload_b1_data_category_mate_t BDCM
                      on A.cur_sap_product_level3_code = BDCM.prod_level_code3 
                  where stat_date in('${PERIOD_ID_D}',to_char(to_date('${PERIOD_ID_D}') - interval '1 month', 'yyyy-MM-dd'))
                    and (
                      A.CUR_SAP_MATL_GROUP_CODE <> '1000010'
                     or A.CUR_SAP_MATL_GROUP_CODE is null
                      )
                    and nvl(BDCM.classification_name,c.material_type) in ('智能'
                      , '大五金'
                      , '卫浴件'
                      , '小五金'
                      , '浴室柜'
                      , '休闲'
                      , '陶瓷')
                    and nvl(c.material_type
                      , '其它') != '非东鹏'
                  group by 二级, 客编, 三级
              ) a
         group by 二级,
                  客编,
                  三级
     )
insert into dm_sw.dm_sw_cust_sale_stat_t
(
stat_date,
dept_name,
cust_num,
cust_name,
month_sale_amt,
month_b1_sale_amt,
month_cloud_sale_amt,
year_sale_amt,
year_b1_sale_amt,
year_cloud_sale_amt,
month_stock_amt,
last_month_stock_amt,
DW_LAST_UPDATE_DATE --DW数据同步时间
)
select '${PERIOD_ID_D}' as stat_date,
       a.二级 as dept_name,
       a.客编 as cust_num,
       a.三级 as cust_name,
       当期对厂提货销售额 as month_sale_amt,
       当期B1出货销售额 as month_b1_sale_amt,
       当期云订货下单销售额 as month_cloud_sale_amt,
       年累计对厂提货销售额 as year_sale_amt,
       年累计B1出货销售额 as year_b1_sale_amt,
       年累计云订货下单销售额 as year_cloud_sale_amt,
       现有库存金额 as month_stock_amt,
       上月库存金额 as last_month_stock_amt,
	   current_timestamp as DW_LAST_UPDATE_DATE --DW数据同步时间
from data a
