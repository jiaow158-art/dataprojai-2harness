set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;

set hive.exec.dynamic.partition=true;

WITH DWR_WM_B1_SALES_ORDER_F_DATA AS(
  SELECT substring(t.doc_date,1,7) as stat_month ,
         t2.area_code,
         t2.area_name,
         t2.operation_center_code as operation_center_code,
	     t2.operation_center_name as operation_center_name ,
	     t.material_num ,
	     sum(t.material_qty) as  material_qty
  from DWROTD.DWR_WM_B1_SALES_ORDER_F t 
  JOIN DM_SW.DM_SW_PREDICT_CUST_ORG  t2
  on case when length(t.dealer_code)<= 6 then concat('0000',t.dealer_code) else t.dealer_code end = t2.cust_num
  where t.customer_type = 'SW' 
  --and t.out_return_goods = '出货'
  --and t.material_qty >0
  and substring(t.doc_date,1,10) <= '${PERIOD_ID_D}'
  and substring(t.doc_date,1,10) >= add_months('${PERIOD_ID_D}',-24)
  group by substring(t.doc_date,1,7) ,
           t2.area_code,
           t2.area_name,
           t2.operation_center_code ,
	       t2.operation_center_name ,
	       t.material_num 
  having sum(t.material_qty) >0
union all       
    select  substring(t.stat_date_d,1,7) as stat_month ,
            '子公司',
            '子公司',
            operation_center_name as operation_center_code,
            operation_center_name,
            material_num, 
            sum(OUTSTOCK_QTY)
    from  dm_sw.DM_WM_CENTER_OUTINSTOCK_T_D T
    where T.stat_date_d >= concat(substring(add_months('${PERIOD_ID_D}',-23),1,7),'-01')
	AND T.stat_date_d  <= '${PERIOD_ID_D}'   -- 每月1号，传昨天，如：5月1日跑4月30日的数
	group by substring(t.stat_date_d,1,7) ,
            operation_center_name,
            material_num
),
FORECAST_DISTRIBUTION_ACTUAL_DATA as(  -- 分销预测明细数据(N1~N3)
	SELECT substring('${PERIOD_ID_D}',1,7) as stat_month, 
		   t1.area_code, 
		   t1.area_name,
		   t1.operation_center_code, 
		   t1.operation_center_name, 
		   t1.material_num,
		   sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-23),1,7) and substring('${PERIOD_ID_D}',1,7) then t1.material_qty else 0 end ) as n1_lj_distribution_qty,  -- 3年前36个月到昨天
		   sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-22),1,7) and substring('${PERIOD_ID_D}',1,7) then t1.material_qty else 0 end ) as n2_lj_distribution_qty,  -- 3年前35个月到昨天
		   sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-21),1,7) and substring('${PERIOD_ID_D}',1,7) then t1.material_qty else 0 end ) as n3_lj_distribution_qty  -- 3年前34个月到昨天
	FROM DWR_WM_B1_SALES_ORDER_F_DATA t1
	group by --t1.stat_month, 
			 t1.area_code, 
			 t1.area_name, 
			 t1.operation_center_code, 
			 t1.operation_center_name, 
			 t1.material_num
),
MATERIAL_MIN_DATE AS (
SELECT T.material_num,MIN(doc_date) AS material_min_date
	from DWROTD.DWR_WM_B1_SALES_ORDER_F t
	where t.customer_type = 'SW' 
	and t.out_return_goods = '出货'
	and t.material_qty > 0
	and doc_date > add_months('${PERIOD_ID_D}',-24) and doc_date <= '${PERIOD_ID_D}'
GROUP BY T.material_num
),
WORKDAY  as (
	SELECT distinct substr(doc_date,1,10) as stat_date_d
		from DWROTD.DWR_WM_B1_SALES_ORDER_F t
	where t.customer_type = 'SW' 
	and t.out_return_goods = '出货'
	and t.material_qty > 0
	and doc_date > add_months('${PERIOD_ID_D}',-24) and doc_date <= '${PERIOD_ID_D}'
),
FORECAST_DISTRIBUTION_YXTS_DATA as(  -- 实际的有效天数
	SELECT substring('${PERIOD_ID_D}',1,7) as stat_month,
	       T2.material_num,
		   count(distinct case when stat_date_d > add_months('${PERIOD_ID_D}',-24) and stat_date_d <= '${PERIOD_ID_D}' and stat_date_d>= t2.material_min_date then stat_date_d else null end) as n1_distribution_ts_cnt,
		   count(distinct case when stat_date_d > add_months('${PERIOD_ID_D}',-23) and stat_date_d <= '${PERIOD_ID_D}' and stat_date_d>= t2.material_min_date then stat_date_d else null end) as n2_distribution_ts_cnt,
		   count(distinct case when stat_date_d > add_months('${PERIOD_ID_D}',-22) and stat_date_d <= '${PERIOD_ID_D}' and stat_date_d>= t2.material_min_date then stat_date_d else null end) as n3_distribution_ts_cnt
	FROM WORKDAY T1
	JOIN MATERIAL_MIN_DATE T2 ON 1=1 
	GROUP BY T2.material_num
),
holiday_detail_data as( -- 假期明细数据
	select substring(t.day_dt,1,7) as stat_month, -- 月份
		   max(substring(t.day_dt,9,2)) as month_days,--月份天数,
	       sum(case when t.holiday_name is null then 1 else 0 end ) as valid_days,--有效天数,
	       sum(case when t.holiday_name is not null then 1 else 0 end ) as holiday_days -- 假期天数
	from (
		select t1.day_dt ,t2.holiday_date,t2.holiday_name
		from dm.dm_dim_date_d t1
		left join SDI.SDI_SALES_FORECAST_HOLIDAY_1037 t2
		on t1.day_dt = t2.holiday_date
		and t2.del_flag = 'N'
	   )t
	   group by substring(t.day_dt,1,7)
),
holiday_detail_data01 as(  -- 假期数据，如：5月1日，跑4月30日的数据，日期存4月，预测5月，N+1是6月，以此内推
select t.stat_month,
       t1.valid_days as n1_valid_days,
       t2.valid_days as n2_valid_days,
       t3.valid_days as n3_valid_days
from holiday_detail_data t
left join holiday_detail_data t1
on substring(add_months(concat(t.stat_month,'-01'),2),1,7) = t1.stat_month
left join holiday_detail_data t2
on substring(add_months(concat(t.stat_month,'-01'),3),1,7) = t2.stat_month
left join holiday_detail_data t3
on substring(add_months(concat(t.stat_month,'-01'),4),1,7) = t3.stat_month
where t.stat_month = substring('${PERIOD_ID_D}',1,7)  -- 每月1号传昨天
),
NO_DELIVER_DATA AS (
select operation_center_code,material_num,sum(not_delivered_qty) as not_delivered_qty from (
SELECT  T2.operation_center_code,
        T1.ItemCode as material_num,
        SUM(T1.OpenQty) as not_delivered_qty
FROM DWIOTD.DWI_WM_B1_RDR1_T T1  
LEFT JOIN  DM_SW.DM_SW_PREDICT_CUST_ORG  T2 ON  case when length(T1.CUSTOMER_CODE) < 10 then concat('0000',T1.CUSTOMER_CODE) else  T1.CUSTOMER_CODE end = T2.CUST_NUM
GROUP BY  T2.operation_center_code,T1.ItemCode
having  SUM(T1.OpenQty)>0
union all  
select  
    t3.operation_center_code,
    t1.product_code,
    sum(t1.not_delivered_qty) as  not_delivered_qty
from SDI.SDI_DTWMS_SO_LINE_1031 t1 
join SDI.SDI_DTWMS_SO_HEADERS_1031 t2 ON t1.HEADER_ID = t2.HEADER_ID
LEFT JOIN  DM_SW.DM_SW_PREDICT_CUST_ORG  T3 ON T2.DISTRIBUTOR_CODE = T3.CUST_NUM
group by t3.operation_center_code,t1.product_code
having  sum(t1.not_delivered_qty)>0
) tmp 
group by operation_center_code,material_num
)
INSERT OVERWRITE TABLE DM_SW.DM_SW_PREDICT_TRANSPORT_DISTRIBUTION_T PARTITION(stat_month)
select t1.area_code, 
	   t1.area_name,
	   t1.operation_center_code, 
	   t1.operation_center_name, 
	   t1.material_num,
	   t1.n1_lj_distribution_qty,  -- 3年前36个月到昨天
	   t2.n1_distribution_ts_cnt,  -- 3年前36个月到昨天的有效天数
	   t3.n1_valid_days,		   -- N1的有效假期天数
	   t1.n2_lj_distribution_qty,  -- 3年前35个月到昨天
	   t2.n2_distribution_ts_cnt,  -- 3年前35个月到昨天的有效天数
	   t3.n2_valid_days	,		   -- N2的有效假期天数
	   t1.n3_lj_distribution_qty,  -- 3年前34个月到昨天
	   t2.n3_distribution_ts_cnt,   -- 3年前34个月到昨天的有效天数	   
	   t3.n3_valid_days	,		  	-- N3的有效假期天数
	    -- 向上取整((N+1的累计数 除以 N+1的有效天数) 乘以 N+1 的假期天数) 
	   ceil((t1.n1_lj_distribution_qty / t2.n1_distribution_ts_cnt) * t3.n1_valid_days) as n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
	   -- 向上取整(( (N+2的累计数 + N+1月系统提货预测建议值) 除以 (N+2 的有效天数 + N+1 的假期天数) ) 乘以 N+2 的假期天数) 
	   ceil(
		    (
	          (
		       t1.n2_lj_distribution_qty 
		       + 
		       ((t1.n1_lj_distribution_qty / t2.n1_distribution_ts_cnt) * t3.n1_valid_days)
		       ) /
		       (t2.n2_distribution_ts_cnt + t3.n1_valid_days)
		     ) * t3.n2_valid_days
	        ) as n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
	    /*
   	    * 向上取整(((N+3的累计数 + N+1月系统提货预测建议值  + N+2月系统提货预测建议值) 
   	    * 除以 (N+3 的有效天数 + N+1 的假期天数 + N+2 的假期天数)) 乘以 N+3 的假期天数) 
   	    * */ 
	   ceil(	
	        (  
	         (	
	            t1.n3_lj_distribution_qty 
	      		 +
	      		((t1.n1_lj_distribution_qty / t2.n1_distribution_ts_cnt) * t3.n1_valid_days)
	      		+
	      		(
				    (
			          (
				       t1.n2_lj_distribution_qty 
				       + 
				       ((t1.n1_lj_distribution_qty / t2.n1_distribution_ts_cnt) * t3.n1_valid_days)
				       ) /
				       (t2.n2_distribution_ts_cnt + t3.n1_valid_days)
				     ) * t3.n2_valid_days
			     )
			   ) /
			   (t2.n3_distribution_ts_cnt + t3.n1_valid_days + t3.n2_valid_days)
			  ) * t3.n3_valid_days
			 )as n3_distribution_forecast_suggest_qty, -- N+3月系统分销预测建议值
		 current_timestamp() as dw_last_update_date ,  -- 数据同步时间
		 coalesce(t4.not_delivered_qty,0) as not_delivered_qty,
		 t1.stat_month
from FORECAST_DISTRIBUTION_ACTUAL_DATA t1
left join FORECAST_DISTRIBUTION_YXTS_DATA t2
on t1.stat_month = t2.stat_month and t1.material_num = t2.material_num
left join holiday_detail_data01 t3
on t1.stat_month = t3.stat_month
left join NO_DELIVER_DATA t4 ON t1.operation_center_code =t4.operation_center_code and t1.material_num = t4.material_num ;
