WITH SDI_B1_CUSTOMER_DATA_1037_DATA AS(  -- B1卫浴有效的客编数据
   select case when length(cust_code)<= 6 then concat('0000',cust_code) else cust_code end as cust_code ,
   		  abbreviation customer_short_name 
   from SDI.SDI_B1_CUSTOMER_DATA_1037 
   where cust_type = 'SW'
),
ORG_DATA as( -- 主客编（运中）-大区关系
select t2.code as operation_center_code,
       t2.name as operation_center_name, 
       t3.code as area_code,
       t3.name as area_name
 from SDI.SDI_WYAPS_INSERVICE_RELATION_1007 t1
JOIN (select id,code,name from sdi.sdi_bd_catalog_1007 where catalog_type_id =37 and is_phantom=0) t2 ON t1.operation_center_id  = t2.id
JOIN (select id,code,name from sdi.sdi_bd_catalog_1007 where catalog_type_id =39 and is_phantom=0) t3 ON t1.area_id  = t3.id
),
DWR_WM_B1_SALES_ORDER_F_DATA AS(
  SELECT substring(t.doc_date,1,7) as stat_month ,
         t2.area_code,
         t2.area_name,
         case when length(t.dealer_code)<= 6 then concat('0000',t.dealer_code) else t.dealer_code end as operation_center_code,
	     t1.customer_short_name as operation_center_name ,
	     t.material_num ,
	     sum(t.material_qty) as  outstock_qty,
	     sum(t.total_sales_amt) as sales_amt
  from DWROTD.DWR_WM_B1_SALES_ORDER_F t 
  join SDI_B1_CUSTOMER_DATA_1037_DATA t1
  on case when length(t.dealer_code)<= 6 then concat('0000',t.dealer_code) else t.dealer_code end = t1.cust_code
  JOIN ORG_DATA t2
  on case when length(t.dealer_code)<= 6 then concat('0000',t.dealer_code) else t.dealer_code end = t2.operation_center_code
  where t.customer_type = 'SW' 
  and t.out_return_goods = '出货'
  and t.material_qty >0
  and substring(t.doc_date,1,10) <= '${PERIOD_ID_D}'
  group by substring(t.doc_date,1,7) ,
           t2.area_code,
           t2.area_name,
           case when length(t.dealer_code)<= 6 then concat('0000',t.dealer_code) else t.dealer_code end ,
	       t1.customer_short_name ,
	       t.material_num 
),
dim_data as(   -- 数据发散  '2022-01-01' 发散的开始时间可以调整
  select SUBSTRING(day_dt,1,7) as stat_month
  from DM.dm_dim_date_d T
  where day_dt between '2022-01-01' and '${PERIOD_ID_D}' -- date_sub(from_unixtime(unix_timestamp(),'yyyy-MM-dd'),1)
  group by SUBSTRING(day_dt,1,7)
),
DISTRIBUTION_DETAIL_DATA01 as(
	SELECT t1.stat_month,
	       t2.operation_center_code,
		   t2.operation_center_name,
		   t2.area_code,
		   t2.area_name,
	       t2.material_num,
	       t4.material_name,
	       nvl(t3.outstock_qty,0) as outstock_qty,
	       nvl(t3.sales_amt,0) as sales_amt
	FROM dim_data t1
	left join(
		select distinct 
		       operation_center_code,
		       operation_center_name,
		       material_num,
		       area_code,
               area_name
		from DWR_WM_B1_SALES_ORDER_F_DATA
	) t2
	on 1 = 1
	left join DWR_WM_B1_SALES_ORDER_F_DATA t3
	on t1.stat_month = t3.stat_month
	and nvl(t2.operation_center_code,'ISNULL') = nvl(t3.operation_center_code,'ISNULL')
	and nvl(t2.operation_center_name,'ISNULL') = nvl(t3.operation_center_name,'ISNULL')
	and nvl(t2.material_num,'ISNULL') = nvl(t3.material_num,'ISNULL')
	join DWIMD.dwi_md_data_material_general_t t4 ON t2.material_num = t4.material_num
),
B1_SALES_HIS_DATA AS(
select --t1.stat_month,
	   t1.area_code,
	   t1.area_name,
       t1.operation_center_code,
	   t1.operation_center_name,
       t1.material_num,
       t1.material_name,
       case when substring(max(t1.stat_month),6,1) = '0' then concat(substring(max(t1.stat_month),7,1),'月')
            else concat(substring(max(t1.stat_month),6,2),'月') end as start_month, -- 开始月份
 	   ceil((sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-11),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  max(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-11),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  min(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-11),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end ))/10
		   ) as mon_avg_12_pickup_qty, -- 近12个月月均提货数
	   ceil((sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-5),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  max(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-5),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  min(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-5),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end ))/4
		   ) as mon_avg6_pickup_qty, -- 近6个月月均提货数
	   ceil((sum(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-2),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  max(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-2),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end )
	            -  min(case when t1.stat_month between substring(add_months('${PERIOD_ID_D}',-2),1,7) and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0) else 0 end ))/1
		   ) as mon_avg3_pickup_qty, -- 近3个月月均提货数
	   --t1.stat_month,
	   --t1.outstock_qty,
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-11),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_12m_sales_qty, -- 过去第12个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-10),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_11m_sales_qty, -- 过去第11个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-9),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_10m_sales_qty, -- 过去第10个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-8),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_9m_sales_qty, -- 过去第9个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-7),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_8m_sales_qty, -- 过去第8个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-6),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_7m_sales_qty, -- 过去第7个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-5),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_6m_sales_qty, -- 过去第6个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-4),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_5m_sales_qty, -- 过去第5个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-3),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_4m_sales_qty, -- 过去第4个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-2),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_3m_sales_qty, -- 过去第3个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-1),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_2m_sales_qty, -- 过去第2个月销售数
	   sum(case when t1.stat_month = substring(add_months('${PERIOD_ID_D}',-0),1,7)  then  nvl(t1.outstock_qty,0) else 0 end )  as last_1m_sales_qty, -- 过去第1个月销售数
	   sum(case when t1.stat_month between concat(substring('${PERIOD_ID_D}',1,4),'-01') and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.outstock_qty,0)  else 0 end ) as ytd_sales_qty,   -- 年累计数量
	   sum(case when t1.stat_month between concat(substring('${PERIOD_ID_D}',1,4),'-01') and substring('${PERIOD_ID_D}',1,7) then  nvl(t1.sales_amt,0)  else 0 end ) as ytd_sales_amt
from DISTRIBUTION_DETAIL_DATA01 t1
group by --t1.stat_month,
	   t1.area_code,
	   t1.area_name,
       t1.operation_center_code,
	   t1.operation_center_name,
       t1.material_num,
       t1.material_name
)
INSERT OVERWRITE TABLE DM_SW.DM_SW_B1_SALES_HIS_T PARTITION(STAT_MONTH)
SELECT CONCAT(replace( substring(add_months('${PERIOD_ID_D}',1),1,7),'-',''),ROW_NUMBER() OVER()) SEQ_NO, -- 数据同步ID
	   t1.operation_center_code,  -- 运中
	   T1.operation_center_name,
	   t1.area_code,    -- 大区
	   t1.area_name,
	   '01' as channel_code,  -- 渠道
	   '经销' as channel_name,
	   t1.material_num,   -- 物料
	   T1.material_name,
	   t1.start_month, -- 开始月份
 	   t1.mon_avg_12_pickup_qty, -- 近12个月月均提货数
	   t1.mon_avg6_pickup_qty, -- 近6个月月均提货数
	   t1.mon_avg3_pickup_qty, -- 近3个月月均提货数
	   t1.last_12m_sales_qty, -- 过去第12个月销售数
	   t1.last_11m_sales_qty, -- 过去第11个月销售数
	   t1.last_10m_sales_qty, -- 过去第10个月销售数
	   t1.last_9m_sales_qty, -- 过去第9个月销售数
	   t1.last_8m_sales_qty, -- 过去第8个月销售数
	   t1.last_7m_sales_qty, -- 过去第7个月销售数
	   t1.last_6m_sales_qty, -- 过去第6个月销售数
	   t1.last_5m_sales_qty, -- 过去第5个月销售数
	   t1.last_4m_sales_qty, -- 过去第4个月销售数
	   t1.last_3m_sales_qty, -- 过去第3个月销售数
	   t1.last_2m_sales_qty, -- 过去第2个月销售数
	   t1.last_1m_sales_qty, -- 过去第1个月销售数
	   current_timestamp() as dw_last_update_date,  -- 数据同步时间 
	   ytd_sales_qty,
	   ytd_sales_amt,
       substring(add_months('${PERIOD_ID_D}',1),1,7) as stat_month -- 日期 t1.stat_month
FROM B1_SALES_HIS_DATA T1;
