set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with dz_chose as (
select t.material_num,
	sum(case when  t.matl_group4_code in('DZ') then 1 else 0 end) as dz_flag --DZ定制过滤标识
from dwimd.dwi_md_data_material_sales_t_his  t
where 1=1
and date_add('${PERIOD_ID_D}',0) >= t.start_date 
and date_add('${PERIOD_ID_D}',0) < t.end_date 
group by t.material_num
)
,mat as (
select t1.material_num,t1.product_level2_name  as cat,t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        t1.width * t1.length_1/1000/1000 as area,
        t1.product_brand_code,t1.product_brand_name,
        t1.product_level2_code,
        t1.busi_volume,
        t1.reserved_field_1,
        case 
            when t1.reserved_field_1 in ('G101','G111','G108') then '零售'
            when t1.reserved_field_1 in ('G102','G112') then '工程'
            when t1.reserved_field_1 in ('G103','G113') then '家装'
            when t1.reserved_field_1 in ('G104','G116') then '流通'
            when t1.reserved_field_1 in ('G107') then '国贸'
            when t1.reserved_field_1 in ('G109OEM') then 'OEM'
            when t1.reserved_field_1 in ('G110') then '电商'
            else t1.reserved_field1_name 
        end reserved_field1_name,
        t1.bathroom_channal,
        t1.matl_group_code,t1.matl_group_name,t1.external_matl_group_code,t1.external_matl_group_name,
        case
	        when t1.product_level3_code in ('4012009') then '小五金'
	        when t1.product_level1_code='30' and t1.product_level2_code in ('3005','3007') then '智能坐便器'
		    when t1.product_level1_code='30' and t1.product_level2_code not in ('3005','3007') then '陶瓷'
		    when t1.product_level2_code in ('4002','4007') and t2.dz_flag <= 0 then '浴室柜（标品）'
		    when t1.product_level2_code in ('4002','4007') and t2.dz_flag > 0 then '浴室柜（特（客）定）'
		    when t1.product_level2_code in ('4004','4005','4012')  then '大五金'
		    when t1.product_level2_code in ('4006','4009','4010','4011','4013','4109','4103')  then '小五金'
		    when t1.product_level2_code in ('4003','4014')  then '休闲类'
        end as category_name,
        basic_calc_unit_name  --物料单位
from dwimd.dwi_md_data_material_general_t t1
left join dz_chose t2 on t1.material_num = t2.material_num
where 1=1 
and (t1.matl_group_code in ('3000010','4000010') or (t1.matl_group_code = '9000010' and external_matl_group_code = '10050'))
and 1=
case 
    when t1.product_level1_code in ('30','40')  then 1
    when t1.product_level1_code in ('41') and t1.product_level2_code in ('4109','4103') then 1 
    else 0
end
)
,dwr_wm_all_type_stock_month_t as 
(
select 
t1.month_date as month_date
,t1.factory_werks_code
,t1.material_num
,t1.stock_qty
,t1.inv_location
from dwrotd.dwr_wm_all_type_stock_month_t t1
join mat t2 on t1.material_num = t2.material_num
where t1.month_date >= replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,4),'-01-01'),1),1,7),'-','')
and t1.month_date <= replace(substr('${PERIOD_ID_D}',1,7),'-','')
-- and t1.VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1)
and t1.inv_location in  ('GR','GX','GB','GK','GO','GR90','GR99','GC','GD','GZ','GD90','GQ','GF','GA')
and t1.factory_werks_code in   ('3210','3220','3510','3520','3310','3320','3810','3820','3610','3620','3603','3660','3720','3710')
),
sto as(
-- 基地内的转储单
select distinct ekko.ebeln  
FROM sdi.sdi_ekko_1001 as ekko 
INNER JOIN sdi.sdi_ekpo_1001 as  ekpo ON ekko.ebeln = ekpo.ebeln
where ekko.bsart = 'ZSUB'  
and ekpo.lgort IN('GR','GX','GB','GK','GO','GR90','GR99','GC','GD','GZ','GD90','GQ','GF','GA')
)
,DWR_WM_PRODSTOCK_OUT_IN_F as (
select 
t1.VOUCHER_POST_DATE
,t1.factory_werks_code --工厂代码
,t1.material_num --物料名称
,t1.inv_location
,case when (t1.move_type in ('301','161','162') or (t1.move_type in ('101','102') and t3.ebeln is null))
		then 
			t1.stock_qty
		else 0 end in_stock_qty--入库数量
,case when t1.move_type in ('601','687','643','931','933','951','Z13','Z15','Z17','Z61','702',
							 '602','688','651','652','653','654','644','932','934','952','Z14','Z16','Z18','Z62','701'
								)  and t3.ebeln is null
	then 
		-1*t1.stock_qty
	else 0 end out_stock_qty--出库数量
from dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join mat t2 on t1.material_num = t2.material_num
left join sto t3 ON t1.purchase_voucher_num = t3.ebeln 
where 1=1
and t1.VOUCHER_POST_DATE >= concat(substr('${PERIOD_ID_D}',1,4),'-01-01')
and t1.VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1)
and t1.inv_location in  ('GR','GX','GB','GK','GO','GR90','GR99','GC','GD','GZ','GD90','GQ','GF','GA')
and t1.factory_werks_code in  ('3210','3220','3510','3520','3310','3320','3810','3820','3610','3620','3603','3660','3720','3710')
)
,month_stock as
(
select 
t1.factory_werks_code
,t1.material_num
,t1.inv_location
,sum(case when  t1.month_date = replace(substr('${PERIOD_ID_D}',1,7),'-','') then t1.stock_qty else 0 end) as stock_qty_month_end --本月库存
,sum(case when  t1.month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,7),'-01'),1),1,7),'-','') then t1.stock_qty else 0 end) as stock_qty_month_start --月初库存
,sum(case when  t1.month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,4),'-01-01'),1),1,7),'-','') then t1.stock_qty else 0 end) as stock_qty_year_start --年初库存
from dwr_wm_all_type_stock_month_t t1
group by  t1.factory_werks_code,t1.material_num,t1.inv_location
)
,in_out_data AS 
(
select 
t1.factory_werks_code --工厂代码
,t1.material_num --物料名称
,t1.inv_location
,sum(case when substr(t1.VOUCHER_POST_DATE,1,10) = substr('${PERIOD_ID_D}',1,10) then in_stock_qty else 0 end) in_stock_qty_d--入库数量 当日
,sum(case when substr(t1.VOUCHER_POST_DATE,1,10) = substr('${PERIOD_ID_D}',1,10) then out_stock_qty  else 0 end) out_stock_qty_d--出库数量 当日

,sum(case when substr(t1.VOUCHER_POST_DATE,1,7) = substr('${PERIOD_ID_D}',1,7) and VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1) then in_stock_qty else 0 end) in_stock_qty_m--入库数量 当月
,sum(case when substr(t1.VOUCHER_POST_DATE,1,7) = substr('${PERIOD_ID_D}',1,7) and VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1) then out_stock_qty  else 0 end) out_stock_qty_m--出库数量 当月

,sum(case when substr(t1.VOUCHER_POST_DATE,1,4) = substr('${PERIOD_ID_D}',1,4) and VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1) then in_stock_qty else 0 end) in_stock_qty_y--入库数量 当年
,sum(case when substr(t1.VOUCHER_POST_DATE,1,4) = substr('${PERIOD_ID_D}',1,4) and VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1) then out_stock_qty  else 0 end) out_stock_qty_y--出库数量 当年
from DWR_WM_PRODSTOCK_OUT_IN_F t1
group by t1.factory_werks_code,t1.material_num,t1.inv_location
)

,total_data as 
(
select 
t1.factory_werks_code
,t1.material_num
,t1.inv_location
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.stock_qty_month_end/3.6) else t1.stock_qty_month_end end  stock_qty_month_end--本月库存
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.stock_qty_month_start/3.6) else t1.stock_qty_month_start end  stock_qty_month_start--月初库存
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.stock_qty_year_start/3.6) else t1.stock_qty_year_start end stock_qty_year_start --年初库存
,0 in_stock_qty_d--入库数量 当日
,0 out_stock_qty_d--出库数量 当日
,0 in_stock_qty_m--入库数量 当月
,0 out_stock_qty_m--出库数量 当月
,0 in_stock_qty_y--入库数量 当年
,0 out_stock_qty_y--出库数量 当年
from month_stock t1
left join mat t2 on t1.material_num = t2.material_num
where not(t1.stock_qty_month_end=0 and t1.stock_qty_month_start=0 and t1.stock_qty_year_start=0)
union all 
select 
t1.factory_werks_code
,t1.material_num
,t1.inv_location
,0 stock_qty_month_end --本月库存
,0 stock_qty_month_start --月初库存
,0 stock_qty_year_start --年初库存
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.in_stock_qty_d/3.6) else t1.in_stock_qty_d end in_stock_qty_d--入库数量 当日
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.out_stock_qty_d/3.6) else t1.out_stock_qty_d end out_stock_qty_d--出库数量 当日
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.in_stock_qty_m/3.6) else t1.in_stock_qty_m end in_stock_qty_m--入库数量 当月
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.out_stock_qty_m/3.6) else t1.out_stock_qty_m end out_stock_qty_m--出库数量 当月
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.in_stock_qty_y/3.6) else t1.in_stock_qty_y end in_stock_qty_y--入库数量 当年
,case when t2.product_level2_code = '4014' and t2.basic_calc_unit_name = 'M2' then CEIL(t1.out_stock_qty_y/3.6) else t1.out_stock_qty_y end out_stock_qty_y--出库数量 当年
from in_out_data t1
left join mat t2 on t1.material_num = t2.material_num
where not (t1.in_stock_qty_y = 0 and t1.out_stock_qty_y = 0)
)
INSERT OVERWRITE TABLE dwrotd.DWR_WM_RPT_WM_SW_CXC_DAY_FACTORY_F partition(stat_date)
select
t1.factory_werks_code `工厂`
,t1.material_num `编码`
,t1.inv_location
,case 
	when t2.material_source = 'OEM' then '外购'
	when t1.factory_werks_code in ('3210','3220') then '江西基地'
	when t1.factory_werks_code in ('3510','3520') then '江门基地'
	when t1.factory_werks_code in ('3610','3620','3603','3660') and t3.product_level2_code like '30%' then '江门基地'
	when t1.factory_werks_code in ('3710','3720') then '江门基地'
	when t1.factory_werks_code in ('3310','3320','3810','3820','3610','3620','3603','3660') then '外购'
	else '其他基地'
end `基地`
,case 
    when t3.category_name = '小五金' and t1.factory_werks_code = '3520' and coalesce(t2.material_source,'NAL') <> 'OEM' then '大五金'
    else coalesce(t3.category_name,'其他品类')
end `品类`
,sum(t1.stock_qty_month_end)  `本月库存`
,sum(t1.stock_qty_month_start) `月初库存`
,sum(t1.stock_qty_year_start)  `年初库存`
,sum(t1.in_stock_qty_d)  `入库数量 当日`
,sum(t1.out_stock_qty_d) `出库数量 当日`
,sum(t1.in_stock_qty_m) `入库数量 当月`
,sum(t1.out_stock_qty_m)  `出库数量 当月`
,sum(t1.in_stock_qty_y) `入库数量 当年`
,sum(t1.out_stock_qty_y) `出库数量 当年`
,sum(t1.stock_qty_month_end) as `实时库存（件）` 
,sum(t1.stock_qty_month_end*t3.busi_volume) as `实时库存（立方米）` 
,current_timestamp() as dw_last_update_date 
,replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','') as stat_date 
from total_data t1
left join dm_sw.dm_otd_material_source_t t2 on t1.factory_werks_code = t2.factory_werks_code and t1.material_num = t2.material_num
left join mat t3 on t1.material_num = t3.material_num
group by 
case 
	when t2.material_source = 'OEM' then '外购'
	when t1.factory_werks_code in ('3210','3220') then '江西基地'
	when t1.factory_werks_code in ('3510','3520') then '江门基地'
	when t1.factory_werks_code in ('3610','3620','3603','3660') and t3.product_level2_code like '30%' then '江门基地'
	when t1.factory_werks_code in ('3710','3720') then '江门基地'
	when t1.factory_werks_code in ('3310','3320','3810','3820','3610','3620','3603','3660') then '外购'
	else '其他基地'
end
,case 
    when t3.category_name = '小五金' and t1.factory_werks_code = '3520' and coalesce(t2.material_source,'NAL') <> 'OEM' then '大五金'
    else coalesce(t3.category_name,'其他品类')
end ,t1.factory_werks_code,t1.material_num,t1.inv_location
