--销售排名逻辑：
--产品层次30开头的只取3006001-3006004开头，且层次为10位数的
--记录：还有3007001跟3007002，层次是10位数字的，是属于智能的，这个需要保留的
--Union all
--产品层次40开头的只取4002001、4007001、4012（剔除4012003/4012004/4012007）、4013、4014（剔除4014002001/4014002003/4014003/4014003002）开头的，4002005取描述有 主柜字眼的
--Union all
--4109；9006#04001中只保留21-JTN4033AD、21-JTN4019AD、21-JTN4005ADQ、21-JTN4000ADQ这4款感应器，且这4款感应器销量需再加上对应配套的小便器的销量
----如果编码是10位数，简码则取前面5位数；如果编码是11位数，简码则取前面6位数，然后结尾是0的，需剔除0；如果编码是12位数，简码则取前面7位数，然后结尾是0的，需剔除0

with dwi_md_data_material_general_t as (
	select *,
		CASE
      		WHEN length(SPLIT_PART(a.material_num,'_',1)) = 10 THEN substr(a.material_num, 1, 5)
        	WHEN length(SPLIT_PART(a.material_num,'_',1)) = 11 THEN rtrim(substr(a.material_num, 1, 6),'0')
   			WHEN length(SPLIT_PART(a.material_num,'_',1)) = 12 THEN rtrim(substr(a.material_num, 1, 7),'0')
   		END AS material_num_short
	from dwimd.dwi_md_data_material_general_t a
    where (
	   	substr(a.product_level_code, 1, 2) = '30' and
       	substr(a.product_level_code, 1, 7) >= '3006001' and
       	substr(a.product_level_code, 1, 7) <= '3006004' and
       	length(a.product_level4_code) = 10
          	) 
          or (
        substr(a.product_level_code, 1, 7) in ('3007001','3007002') and 
        length(a.product_level4_code) = 10
        	)
    union all
    select *,
        CASE
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 10 THEN substr(a.material_num, 1, 5)
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 11 THEN rtrim(substr(a.material_num, 1, 6),'0')
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 12 THEN rtrim(substr(a.material_num, 1, 7),'0')
        END AS material_num_short
    from dwimd.dwi_md_data_material_general_t a
    where (
    	substr(a.product_level_code, 1, 7) = '4002001' or
        substr(a.product_level_code, 1, 7) = '4007001' or
        (substr(a.product_level_code, 1, 4) = '4012' and
        a.product_level_code not in
        ('4012003', '4012004', '4012007')) or
        substr(a.product_level_code, 1, 4) = '4013' or
        (substr(a.product_level_code, 1, 4) = '4014' and
        a.product_level_code not in
        ('4014002001', '4014002003','4014003', '4014003002') ) or
        (a.product_level_code = '4002005' and a.material_name like '%主柜%')
        	) 
    union all
    select *,
    	CASE
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 10 THEN substr(a.material_num, 1, 5)
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 11 THEN rtrim(substr(a.material_num, 1, 6),'0')
            WHEN length(SPLIT_PART(a.material_num,'_',1)) = 12 THEN rtrim(substr(a.material_num, 1, 7),'0')
        END AS material_num_short
    from dwimd.dwi_md_data_material_general_t a
    where 
    	(substr(a.product_level_code, 1, 4) = '4109') or(
        substr(a.product_level_code, 1, 10) = '9006#04001' and
    	a.material_num in ('21-JTN4033AD','21-JTN4019AD','21-JTN4005ADQ','21-JTN4000ADQ'))
			),

material_bom as (
	select
		t2.bom_component as material_num,--子项
		t.material_num as material_bom,--母项
		row_number() over(partition by bom_component order by material_num  ) as RN --只需要一个bom
	from (
        select matl_list,material_num
        from 
            (
            select 
            *,row_number() over(partition by material_num order by rec_crt_date) as RN
            from DWRDIM.DWR_DIM_MATERIAL_LIST_LINK_F 
             --使用的是销售的bom是5，生产还有一个bom为1
            where bom_use_code = '5' 
            ) as t1
        where RN = 1 
	        ) t
	left join DWRDIM.DWR_DIM_BOM_ITEM_D t2 on t2.matl_list = t.matl_list 
)
