set hive.exec.dynamic.partition.mode=nonstrict;
with tem as (
SELECT A.MATERIAL_NUM ,
		A.FACTORY_CODE ,
	    A.INV_LOCATION ,
	    A.BATCH_NUM ,
	    A.STOCKOUT_ORDER_NO,
	    A.stockout_order_item_no,
	    A.WBS_ELEMENT ,
		A.cust_num,
		A.cust_name,
		A.SALES_ORG_CODE , 
	    A.SUBJECT_DISTR_TYPE,
		A.PRO_SKU_PREPARE_QTY,
		A.TOTAL_PREPARE_QTY,
		A.ORDER_SOURCE,
		A.REFERENCE_BILL_NUM,
		A.REFERENCE_PROJ_NUM,
		A.prepare_stock_date,
		A.prepare_stock_days,
		A.distribution_channel_code,
		A.creator_id,
	    A.res_stock_date,
		A.busi_volume,
		A.unit_price,
		A.TOTAL_PREPARE_VOLUMe,
		A.TOTAL_PREPARE_AMT,
		A.org_lev2_code,
		A.org_lev2_name,
		A.org_lev3_code,
		A.org_lev3_name,
		A.org_lev4_code,
		A.org_lev4_name,
		A.org_lev5_code,
		A.org_lev5_name,
		A.DW_LAST_UPDATE_DATE,
		A.require_date,
		A.promise_date,
		A.creator_name,
		case 
			when date_format(date_sub(current_date(),1),'EEEE') = 'Wednesday' then date_sub(current_date(),1) --周三
			when day(last_day(date_sub(current_date(),1))) = day(date_sub(current_date(),1)) then date_sub(current_date(),1)--月末
			when date_format(date_sub(current_date(),1), 'MMdd') = '1231' then date_sub(current_date(),1)--年底
		end as partition_ymd
from DM_SW.DM_WM_PREPARE_STOCK_DTL A

union ALL 

SELECT 
    A.MATERIAL_NUM ,
	A.FACTORY_CODE ,
    A.INV_LOCATION ,
    A.BATCH_NUM ,
    A.STOCKOUT_ORDER_NO,
    A.stockout_order_item_no,
    A.WBS_ELEMENT ,
	A.cust_num,
	A.cust_name,
	A.SALES_ORG_CODE , 
    A.SUBJECT_DISTR_TYPE,
	A.PRO_SKU_PREPARE_QTY,
	A.TOTAL_PREPARE_QTY,
	A.ORDER_SOURCE,
	A.REFERENCE_BILL_NUM,
	A.REFERENCE_PROJ_NUM,
	A.prepare_stock_date,
	A.prepare_stock_days,
	A.distribution_channel_code,
	A.creator_id,
    A.res_stock_date,
	A.busi_volume,
	A.unit_price,
	A.TOTAL_PREPARE_VOLUMe,
	A.TOTAL_PREPARE_AMT,
	A.org_lev2_code,
	A.org_lev2_name,
	A.org_lev3_code,
	A.org_lev3_name,
	A.org_lev4_code,
	A.org_lev4_name,
	A.org_lev5_code,
	A.org_lev5_name,
	A.DW_LAST_UPDATE_DATE,
	A.require_date,
	A.promise_date,
	A.creator_name,
	'4712-12-31' as partition_ymd --实时
from DM_SW.DM_WM_PREPARE_STOCK_DTL A
)

insert overwrite table DM_SW.DM_WM_PREPARE_STOCK_DTL_YMD partition(partition_ymd)
SELECT 
    A.MATERIAL_NUM ,
	A.FACTORY_CODE ,
    A.INV_LOCATION ,
    A.BATCH_NUM ,
    A.STOCKOUT_ORDER_NO,
    A.stockout_order_item_no,
    A.WBS_ELEMENT ,
	A.cust_num,
	A.cust_name,
	A.SALES_ORG_CODE , 
    A.SUBJECT_DISTR_TYPE,
	A.PRO_SKU_PREPARE_QTY,
	A.TOTAL_PREPARE_QTY,
	A.ORDER_SOURCE,
	A.REFERENCE_BILL_NUM,
	A.REFERENCE_PROJ_NUM,
	A.prepare_stock_date,
	A.prepare_stock_days,
	A.distribution_channel_code,
	A.creator_id,
    A.res_stock_date,
	A.busi_volume,
	A.unit_price,
	A.TOTAL_PREPARE_VOLUMe,
	A.TOTAL_PREPARE_AMT,
	A.org_lev2_code,
	A.org_lev2_name,
	A.org_lev3_code,
	A.org_lev3_name,
	A.org_lev4_code,
	A.org_lev4_name,
	A.org_lev5_code,
	A.org_lev5_name,
	A.DW_LAST_UPDATE_DATE,
	A.require_date,
	A.promise_date,
	A.creator_name,
case 
		when a.org_lev3_name = '供应链中心' or a.org_lev4_name = '供应链中心' then '供应链中心'
		when a.org_lev3_name = '卫浴国际营销' then '卫浴国际营销'
		when a.org_lev3_name = '品类运营部' then '品类运营部'
		when (a.org_lev3_name = '商用中心' or a.org_lev4_name = '商用中心')  then '商用中心' --这里需要区分自营、经销
		when a.org_lev3_name = '国内营销中心' then '卫浴子公司-STO转储订单' --卫浴子公司-STO转储订单
		when (a.org_lev3_name = '营销中心' or  a.org_lev4_name = '营销中心') and a.distribution_channel_code <> '06' then '营销中心'
		when (a.org_lev3_name = '营销中心' or  a.org_lev4_name = '营销中心') and a.distribution_channel_code = '06' then '商用中心'
		when a.org_lev3_name = '客户服务部' then '客户服务部'
		when a.org_lev3_name = '鹏嘉家居' then '鹏嘉家居'
		when a.org_lev3_name = '国内直营板块' then '瓷砖事业部'
		when a.org_lev3_name = '国内经销板块' then '瓷砖事业部'
		when a.org_lev3_name = '国内营销系统' then '瓷砖事业部'
		when a.org_lev3_name = '国际营销系统' then '瓷砖事业部'
		when a.org_lev3_name = '瓷砖新零售中心' then '瓷砖事业部'
		when a.org_lev3_name = '瓷砖营销中心' then '瓷砖事业部'
		when a.org_lev3_name = '其他' then '其他'
		else org_lev2_name
	end as sub_grp,
	case 
		when a.org_lev3_name = '供应链中心' or a.org_lev4_name = '供应链中心' then '供应链中心'
		when a.org_lev3_name = '卫浴国际营销' then '卫浴国际营销'
		when a.org_lev3_name = '品类运营部' then a.org_lev5_name
		when (a.org_lev3_name = '商用中心' or a.org_lev4_name = '商用中心') and d.cust_group_code in ('A0','A4','A1') then '商用中心-自营' 
		when (a.org_lev3_name = '商用中心' or a.org_lev4_name = '商用中心') and d.cust_group_code not in ('A0','A4','A1') then '商用中心-经销'--这里需要区分自营、经销
		when a.org_lev3_name = '国内营销中心' then a.org_lev4_name --卫浴子公司-STO转储订单
		when (a.org_lev3_name = '营销中心' or  a.org_lev4_name = '营销中心') and a.distribution_channel_code <> '06' then a.org_lev5_name
		when (a.org_lev3_name = '营销中心' or  a.org_lev4_name = '营销中心') and a.distribution_channel_code = '06' and d.cust_group_code in ('A0','A4','A1')  then '商用中心-自营'
		when (a.org_lev3_name = '营销中心' or  a.org_lev4_name = '营销中心') and a.distribution_channel_code = '06' and d.cust_group_code not in ('A0','A4','A1')  then '商用中心-经销'
		when a.org_lev3_name = '客户服务部' then '客户服务部'
		when a.org_lev3_name = '鹏嘉家居' then '鹏嘉家居'
		when a.org_lev3_name = '国内直营板块' then a.org_lev4_name
		when a.org_lev3_name = '国内经销板块' then a.org_lev4_name
		when a.org_lev3_name = '国内营销系统' then a.org_lev5_name
		when a.org_lev3_name = '国际营销系统' then '国际营销系统'
		when a.org_lev3_name = '瓷砖新零售中心' then '瓷砖新零售中心'
		when a.org_lev3_name = '瓷砖营销中心' and a.org_lev4_name like '%子公司' then a.org_lev4_name
		when a.org_lev3_name = '瓷砖营销中心' and a.org_lev4_name not like '%子公司' then a.org_lev5_name
		when a.org_lev3_name = '其他' then '其他'
		else concat(org_lev2_name,'-其他部门')
	end as sub_grp1,
	d.cust_group_code,
	A. partition_ymd
from tem A
left join dwrdim.dwr_dim_cust_sales_area_d d on a.cust_num = d.cust_num and a.distribution_channel_code = d.distribution_chnl_code 
			and a.sales_org_code = d.sales_org_code and substring(d.end_date,1,10) = '4712-12-31'
where A.partition_ymd is not null
and 
(
COALESCE(A.res_stock_date,'NAL') <> date_format(current_date(),'yyyy-MM-dd')
or COALESCE(A.prepare_stock_date,'NAL') <> date_format(current_date(),'yyyy-MM-dd')
)
