-- 卫浴运营中心可用库存报表
--CREATE TABLE DM_SW.DM_OTD_OPERCENTER_USABLESTOCK_T as
with sdi_b1_customer_data_1037_DATA AS(
  SELECT  CASE WHEN LENGTH(cust_code) < 10 THEN CONCAT('0000',cust_code) ELSE cust_code END as cust_code,
		 company_name,
		 abbreviation,
		 is_main,
		 dw_last_update_date,
		 is_rtm,
		 rtm_cust_code,
		 rtm_online_date,
		 cust_type,
		 source_b1_flag
  FROM SDI.sdi_b1_customer_data_1037
  where is_main = '是'
),
SAP_B1_RESULT_DATA as(  -- SAP B1 运营中心可用库存
	select '${PERIOD_ID_D}' day_dt, -- 日期，传昨天
		   CASE WHEN LENGTH(t1.cust_code) < 10 THEN CONCAT('0000',t1.cust_code) ELSE t1.cust_code END as customer_code,  --  客编
	       t1.ItemCode as material_num, -- 物料编码
		   nvl(t3.material_name ,t2_1.itemname) AS material_name, -- 物料名称
		   t4.whscode as warehouse_code, -- 仓库编码
		   t4.whsname as warehouse_name, -- 仓库名称
		   t2.DistNumber as batch_code, -- 批次编号
		   nvl(t1.Quantity,0) as tot_stock_qty,  -- 总库存数量
		   nvl(t1.quantity,0) - nvl(t1.CommitQty,0) as available_stock_qty,--'可用库存数量(可再分配)',
		   nvl(t1.CommitQty,0) as return_stock_qty, -- 留货库存数量(批次已分配)
		   nvl(t1.Quantity,0) * nvl(t5.AvgPrice,0) as tot_stock_amt, --'总库存金额',
		   (nvl(t1.quantity,0) - nvl(t1.CommitQty,0)) * nvl(t5.AvgPrice,0) as available_stock_amt, -- '可用库存金额',
		   nvl(t1.CommitQty,0) * nvl(t5.AvgPrice,0) as return_stock_amt -- '留货库存金额'
	from DWIOTD.DWI_wm_B1_OBTQ_T t1    -- 批次库存表    
	join DWIOTD.DWI_wm_B1_obtn_T t2 -- 批次编号 
	on t1.customer_type = t2.customer_type
	and t1.cust_code = t2.customer_code
	AND t1.ItemCode=t2.ItemCode 
	and  t1.SysNumber=t2.SysNumber 
	join DWIOTD.DWI_WM_B1_OITM_T t2_1
	on t1.customer_type = t2_1.customer_type
	and t1.cust_code = t2_1.customer_code
	and t1.itemcode = t2_1.itemcode
	left join DWIMD.dwi_md_data_material_general_t t3
	on t1.itemcode = t3.material_num 
	left join DWIOTD.DWI_WM_B1_OWHS_T t4 -- 仓库
	on t1.customer_type = t4.customer_type
	and t1.cust_code = t4.customer_code
	and t1.WhsCode=t4.WhsCode
	join DWIOTD.DWI_WM_B1_oitw_T t5 -- 仓库表 
	on t1.customer_type = t5.customer_type
	and t1.cust_code = t5.customer_code
	and t1.itemcode=t5.itemcode 
	and  t1.WhsCode=t5.WhsCode 
	where t1.customer_type = 'SW'
	and t1.quantity > 0 
),
sdi_dtwms_inventory_batch_1031_data as(  -- RTM批次库存表
	SELECT t1.distributor_code as customer_code,
		   t1.product_code as material_num,
		   NVL(t2.material_name,t1.product_name) as material_name,
		   t3.warehouse_code,
		   t3.warehouse_name,
		   t1.batch_code ,
		   sum(nvl(t1.quantity,0)) as tot_stock_qty, -- 总库存数量
		   sum(nvl(t1.stay_qty,0)) as return_stock_qty, -- 留货库存数量
		   sum(nvl(t1.quantity,0) - nvl(t1.stay_qty,0)) as available_stock_qty -- 可用库存数量
	FROM SDI.sdi_dtwms_inventory_batch_1031 t1
	LEFT JOIN DWIMD.dwi_md_data_material_general_t T2
	ON T1.product_code = T2.material_num 
	LEFT JOIN SDI.SDI_DTWMS_WAREHOUSE_INFO_1031 T3
	ON t1.distributor_id = T3.distributor_id 
	and t1.warehouse_id = T3.warehouse_id 
	group by t1.distributor_code ,
			 t1.product_code,
			 NVL(t2.material_name,t1.product_name),
			 t3.warehouse_code,
		     t3.warehouse_name,
			 t1.batch_code 
),
sdi_dtwms_inv_cost_1031_data as( -- RTM成本价取值
	select t1.distributor_id,
		   T3.distributor_code as customer_code,
		   T3.distributor_name as customer_name,
		   t1.warehouse_id,
	       t2.warehouse_code,
	       t2.warehouse_name,
	       t1.product_id ,
	       t4.product_code as material_num,
	       t4.sap_product_name as material_name,
	       t1.batch_code ,
	       t1.cost_price
	from SDI.sdi_dt_cost_1031 t1
	LEFT JOIN SDI.SDI_DTWMS_WAREHOUSE_INFO_1031 T2
	ON t1.distributor_id = T2.distributor_id 
	and t1.warehouse_id = T2.warehouse_id
	LEFT JOIN sdi.sdi_DT_DISTRIBUTOR_INFO_1029 T3
	ON T1.distributor_id = T3.distributor_id 
	LEFT JOIN SDI.SDI_PROD_INFO_1031 T4
	ON T1.product_id = T4.product_id 
	where concat(t1.year,'-',case when length(t1.month) < 2 then concat('0',t1.month) else t1.month end) in
	(SELECT max(concat(year,'-',case when length(month) < 2 then concat('0',month) else month end)) as stat_month
	FROM SDI.sdi_dt_cost_1031)
),
BASE_COST_XS_DATA AS(  -- RTM标准成本系数
	SELECT 
	      T.distributor_code, 
	      T.customer_group_code1, 
	      T.customer_group_code3,
	      CASE WHEN T.customer_group_code1 = 'J1' AND T.customer_group_code3 = 'J1' THEN 1
	      	   WHEN T.customer_group_code1 = 'J1' AND T.customer_group_code3 = 'J2' THEN 1 + 0.02
	      	   WHEN T.customer_group_code1 = 'J1' AND T.customer_group_code3 = 'J3' THEN 1 + 0.04
	      	   WHEN T.customer_group_code1 = 'J1' AND T.customer_group_code3 = 'J4' THEN 1 + 0.06
	      	   ELSE 0
	      END AS XS -- 系数
	FROM(
		select distributor_code,
			   customer_group_code1,
			   customer_group_code3,
			   ROW_NUMBER() OVER(PARTITION BY distributor_code,customer_group_code1,customer_group_code3 ORDER BY sales_area_id ASC) RN
		from sdi.sdi_sap_distributor_sale_area_1029
		where delivery_frozen='N'
		and price_group_code='JK'
		and distribution_channel_code='01'
		and substring(sale_organization_code,1,1) = '3'
	) T
	WHERE T.RN= 1
),
BASE_COST_DATA AS(  -- RTM标准成本
	SELECT T1.distributor_code AS customer_code,
		   T1.product_code AS material_num,
		   T1.warehouse_code,
		   T1.batch_code,
		   NVL(T3.cost_price,0) AS cost_price,
	       NVL(T2.XS,0) AS XS,
	       NVL(T3.cost_price,0) * NVL(T2.XS,0) AS BASE_COST
	FROM SDI.sdi_dtwms_inventory_batch_1031 T1
	LEFT JOIN BASE_COST_XS_DATA T2
	ON T1.distributor_code = T2.distributor_code
	LEFT JOIN SDI.SDI_DT_BASE_COST_1031 T3
	ON t1.product_id = t3.product_id 
	WHERE 1 = 1
	AND t3.channel_code='10001'
	and t3.product_id is not null
),
RTM_B1_RESULT_DATA AS( --- RTM 运中可用库存数据
	SELECT '${PERIOD_ID_D}' day_dt, -- 日期 ，传昨天
		   t1.customer_code,  -- 客户编码
		   t1.material_num, -- 物料编码
		   t1.material_name, -- 物料名称
		   t1.warehouse_code, -- 仓库编码
		   t1.warehouse_name, -- 仓库名称
		   t1.batch_code,   -- 批次编码
		   t1.tot_stock_qty, -- 总库存数量
		   t1.available_stock_qty, -- 可用库存数量
		   t1.return_stock_qty, -- 留货库存数量
		   t1.tot_stock_qty * nvl(t2.cost_price,t3.BASE_COST) as tot_stock_amt , -- 总库存金额
		   t1.available_stock_qty * nvl(t2.cost_price,t3.BASE_COST) as available_stock_amt, -- 可用库存数量
		   t1.return_stock_qty * nvl(t2.cost_price,t3.BASE_COST) as return_stock_amt -- 留货库存金额
	FROM sdi_dtwms_inventory_batch_1031_data T1
	LEFT JOIN sdi_dtwms_inv_cost_1031_data T2
	ON t1.customer_code = t2.customer_code
	and t1.material_num = t2.material_num
	and t1.warehouse_code = t2.warehouse_code
	and t1.batch_code = t2.batch_code
	LEFT JOIN BASE_COST_DATA T3
	ON t1.customer_code = T3.customer_code
	and t1.material_num = T3.material_num
	and t1.warehouse_code = T3.warehouse_code
	and t1.batch_code = T3.batch_code
),
RESULT_DATA AS(   -- 合并SAP B1/RTM B1/福建数据
	SELECT t.day_dt, -- 日期 
		   t2.cust_code as customer_code,  -- 客户编码
		   t.material_num, -- 物料编码
		   t.material_name, -- 物料名称
		   t.warehouse_code, -- 仓库编码
		   t.warehouse_name, -- 仓库名称
		   t.batch_code,   -- 批次编码
		   t.tot_stock_qty, -- 总库存数量
		   t.available_stock_qty, -- 可用库存数量
		   t.return_stock_qty, -- 留货库存数量
		   t.tot_stock_amt , -- 总库存金额
		   t.available_stock_amt, -- 可用库存数量
		   t.return_stock_amt -- 留货库存金额
	FROM SAP_B1_RESULT_DATA t    -- SAP B1 数据
	JOIN sdi_b1_customer_data_1037_DATA T2
	--ON T.customer_code = T2.cust_code
	on case when t.customer_code = '1000005469' then '1000006985' -- 安徽
	 	when t.customer_code = '321540' then '1000004847'  -- 苏沪
	 else t.customer_code end =t2.cust_code
	and T.day_dt < case when t2.is_rtm = 'Y' then SUBSTRING(CAST(T2.rtm_online_date AS STRING),1,10) else '4712-12-31' end
	UNION ALL
	SELECT t.day_dt, -- 日期 
		   t.customer_code,  -- 客户编码
		   t.material_num, -- 物料编码
		   t.material_name, -- 物料名称
		   t.warehouse_code, -- 仓库编码
		   t.warehouse_name, -- 仓库名称
		   t.batch_code,   -- 批次编码
		   t.tot_stock_qty, -- 总库存数量
		   t.available_stock_qty, -- 可用库存数量
		   t.return_stock_qty, -- 留货库存数量
		   t.tot_stock_amt , -- 总库存金额
		   t.available_stock_amt, -- 可用库存数量
		   t.return_stock_amt -- 留货库存金额
	FROM RTM_B1_RESULT_DATA t    -- RTM B1 数据
	JOIN sdi_b1_customer_data_1037_DATA T2
	ON T.customer_code = T2.cust_code
	and T.day_dt >= case when t2.is_rtm = 'Y' then SUBSTRING(CAST(T2.rtm_online_date AS STRING),1,10) else '4712-12-31' end
)

		t1.available_stock_qty, -- 可用库存数量
		t1.return_stock_qty, -- 留货库存数量
