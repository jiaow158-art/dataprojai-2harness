--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with DISTRIBUTION_FORECAST_DIM_DATA as(   -- 为了用函数算近12个月的数据,从预测近13个月开发散
  select SUBSTRING(day_dt,1,7) as stat_month
  from DM.dm_dim_date_d T
  where day_dt between add_months('${PERIOD_ID_D}',-13) and '${PERIOD_ID_D}' 
  group by SUBSTRING(day_dt,1,7)
),
DM_SALES_FORECAST_DISTRIBUTION_DETAIL_DATA AS(
	SELECT t1.operation_center_code,
		   t1.material_num,
		   SUM(t1.outstock_qty) as outstock_qty,
		   SUM(t1.instock_qty) as instock_qty,
		   SUM(t1.qm_qty) as qm_qty,
		   substring(t1.stat_date_d ,1,7) as stat_month
	FROM DM_SW.DM_SALES_FORECAST_DISTRIBUTION_DETAIL_TMP t1
	where t1.stat_date_d <= '${PERIOD_ID_D}' -- 每月1号，传昨天，如：5月1日跑4月30日的数据
	  and t1.stat_date_d >= add_months('${PERIOD_ID_D}',-13)
	group by t1.operation_center_code,
		   t1.material_num,
		   substring(t1.stat_date_d ,1,7) 
)
INSERT OVERWRITE TABLE DM_SW.DM_SALES_FORECAST_DISTRIBUTION_DETAIL_TMP01 PARTITION(stat_month)
SELECT t2.operation_center_code,
       null ,
       t2.material_num,
       nvl(t3.outstock_qty,0) as outstock_qty,
       nvl(t3.instock_qty,0) as instock_qty,
       nvl(t3.qm_qty,0) as qm_qty,
       current_timestamp() as dw_last_update_date ,
       t1.stat_month
FROM DISTRIBUTION_FORECAST_DIM_DATA t1
left join(
	select distinct 
	       operation_center_code,
	       material_num
	from DM_SALES_FORECAST_DISTRIBUTION_DETAIL_DATA
	where stat_month = substring('${PERIOD_ID_D}',1,7)
) t2
on 1 = 1
left join DM_SALES_FORECAST_DISTRIBUTION_DETAIL_DATA t3
on t1.stat_month = t3.stat_month
and nvl(t2.operation_center_code,'ISNULL') = nvl(t3.operation_center_code,'ISNULL')
and nvl(t2.material_num,'ISNULL') = nvl(t3.material_num,'ISNULL');
