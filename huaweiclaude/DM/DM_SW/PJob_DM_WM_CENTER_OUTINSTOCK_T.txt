union all

out_fact as(
-- 实际发票数据
select
	substr(a.bill_date,1,7) as stat_month,
	a.sale_org_code,
	a.distribution_channel_code , 
	a.cust_sal,
	--d.operation_center_code ,   --
	case when a.sale_org_code = '3830' then '津冀'
	     when a.sale_org_code = '3831' then '贵州'
		 when a.sale_org_code = '3832' then '苏沪'
	else d.operation_center_code end as operation_center_code,
	--d.operation_center_name,
	case when a.sale_org_code = '3830' then '津冀'
	     when a.sale_org_code = '3831' then '贵州'
		 when a.sale_org_code = '3832' then '苏沪'
	else d.operation_center_name end as operation_center_name,
	d.area_code,
	d.area_name,
	a.material_num,
	sum(nvl(a.actual_billed_qty,0)) as actual_billed_qty,   -- 开票数量
	case when a.distribution_channel_code = '01' then '经销'
         when a.distribution_channel_code = '02' then '展厅'
		 when a.distribution_channel_code = '03' and a.cust_sal in('0000003A00') then '艺耐'
		 when a.distribution_channel_code = '03' then '内部交易'
		 when a.distribution_channel_code = '05' and b.cust_group_code='F1'  then 'OEM'
		 when a.distribution_channel_code = '05' and b.cust_group_code='F0' then '国贸'
		 when a.distribution_channel_code = '06' and b.cust_group_code in('A0','A4') then '自营工程'
		 when a.distribution_channel_code = '06'  /*and b.cust_group_code in('A6','A7','B0')*/ then '经销工程'
		 when a.distribution_channel_code = '07' then '电商' 
		  when a.distribution_channel_code = '08' then '家装'
		   when a.distribution_channel_code = '10' then '流通'end   def_channnel,
    sum(a.subtotal_price_process1) as actual_billed_amt, -- 开票金额
    a.inv_location
from DM_GL_OUT_INVOICE_T_DATA a  -- DM.DM_GL_OUT_INVOICE_T
left join (select cust_num,sales_org_code,distribution_chnl_code,max(cust_group_code) as cust_group_code from dwimd.dwi_md_cust_sales_area_t
            where  product_group ='00' 
            and  end_date = '4712-12-31'
            group by cust_num,sales_org_code,distribution_chnl_code) b On a.cust_sal  = b.cust_num 
                                                                          and a.sale_org_code = b.sales_org_code 
                                                                          and a.distribution_channel_code = b.distribution_chnl_code
left join DWIPI.DWI_WF_ZSD_ZMKB_WY_T c ON a.sale_org_code  = c.vkorg and a.cust_sal =c.kunnr  and a.distribution_channel_code = c.vtweg  and c.spart ='00'
left join org d ON c.zzkunnr = d.operation_center_code 
where a.sale_org_code between '3210' and '3920'  -- 卫浴供应链
and a.sale_org_code in ('3830','3831','3832')  -- 排除 3830 是津冀、3831 是 贵州、3832 是苏沪
--and a.inv_location not in ('WX','GYCK','TJ') -- 排除 WX 苏沪  GYCK 贵州  TJ  津冀
and a.payer_id not in ('0000003200','0000003300','0000003500','0000003600','0000003700','0000003800','0000003900')  -- 去除内部发票
and bill_date >=concat('${PERIOD_ID_M}','-01')   -- 跑昨日所在月份1号数据 2022-11-01
and a.actual_billed_qty>0
and a.sale_voucher_num not like '4%'
group by substr(a.bill_date,1,7) ,
a.sale_org_code,
a.distribution_channel_code , 
a.cust_sal,
--d.operation_center_code ,   
case when a.sale_org_code = '3830' then '津冀'
	     when a.sale_org_code = '3831' then '贵州'
		 when a.sale_org_code = '3832' then '苏沪'
	else d.operation_center_code end,
case when a.sale_org_code = '3830' then '津冀'
	     when a.sale_org_code = '3831' then '贵州'
		 when a.sale_org_code = '3832' then '苏沪'
	else d.operation_center_name end,
d.area_code,
d.area_name,
a.material_num,
case when a.distribution_channel_code = '01' then '经销'
     when a.distribution_channel_code = '02' then '展厅'
		 when a.distribution_channel_code = '03' and a.cust_sal in('0000003A00') then '艺耐'
		 when a.distribution_channel_code = '03' then '内部交易'
		 when a.distribution_channel_code = '05' and b.cust_group_code='F1'  then 'OEM'
		 when a.distribution_channel_code = '05' and b.cust_group_code='F0' then '国贸'
		 when a.distribution_channel_code = '06' and b.cust_group_code in('A0','A4') then '自营工程'
		 when a.distribution_channel_code = '06'  /*and b.cust_group_code in('A6','A7','B0')*/ then '经销工程'
		 when a.distribution_channel_code = '07' then '电商' 
		  when a.distribution_channel_code = '08' then '家装'
		   when a.distribution_channel_code = '10' then '流通'end ,
		   a.inv_location
),
ekbe as  
(
select t1.ebeln ,
	   t1.ebelp , 
	   t1.budat,   -- 凭证日期
	   sum(case when t1.bewtp='L' then t1.menge else 0 end) as lmeng, -- 生产提货单数量
	   sum(case when t1.bewtp='U' AND t1.shkzg = 'H' then t1.menge 
			    when t1.bewtp='U' AND t1.shkzg = 'S' then 0-t1.menge 
			    else 0 end) as out_emeng,   -- 发货数量
	   sum(case when t1.bewtp='E' AND t1.shkzg = 'S' then t1.menge 
			    when t1.bewtp='E' AND t1.shkzg = 'H' then 0-t1.menge 
			    else 0 end) as in_emeng   -- 收货数量
from sdi.sdi_ekbe_1001 t1
where 1=1
group by t1.ebeln ,
	   t1.ebelp ,
	   t1.budat
),
sub_data_01 as(
 select concat(substring(cast(t3.budat as string),1,4),'-',substring(cast(t3.budat as string),5,2)) as stat_month,
		case when t2.lgort = 'WX' then '苏沪'
       		  when t2.lgort = 'GYCK' then '贵州'
       		  when t2.lgort = 'TJ' then '津冀'
       	   else '' end as operation_center_code,
         case when t2.lgort = 'WX' then '苏沪'
       		  when t2.lgort = 'GYCK' then '贵州'
       		  when t2.lgort = 'TJ' then '津冀'
       	   else '' end as operation_center_name,
       	T2.matnr AS material_num,
       	SUM(NVL(T3.in_emeng,0)) AS actual_billed_qty
  FROM sdi.sdi_ekko_1001 as T1 
  INNER JOIN sdi.sdi_ekpo_1001 as  T2 
  ON T1.ebeln = T2.ebeln
  JOIN  ekbe AS T3 
  on T2.ebeln  = T3.ebeln 
  and T2.ebelp  = T3.ebelp
  where T2.lgort in('WX','GYCK','TJ')   -- 库存地点
  and t3.budat >= replace(concat('${PERIOD_ID_M}','-01'),'-','')
  GROUP BY concat(substring(cast(t3.budat as string),1,4),'-',substring(cast(t3.budat as string),5,2)),
  		   case when t2.lgort = 'WX' then '苏沪'
       		  when t2.lgort = 'GYCK' then '贵州'
       		  when t2.lgort = 'TJ' then '津冀'
       	   else '' end,
       	   T2.matnr
)
,sub_com_2 as(
-- 子公司入库
select t1.stat_month,
	   t1.operation_center_code,
	   t1.operation_center_name,
	   t1.material_num,
	   t1.actual_billed_qty as INSTOCK_QTY,
	   0 as OUTSTOCK_QTY
  from out_fact t1
 where t1.inv_location not in ('WX','GYCK','TJ') 
   and t1.operation_center_name is not null

	union all

select t2.stat_month,
	   t2.operation_center_code,
	   t2.operation_center_name,
	   t2.material_num,
	   t2.actual_billed_qty as INSTOCK_QTY,
	   0 as OUTSTOCK_QTY
	from sub_data_01 t2

union all

select t3.stat_month,
	   t3.operation_center_code,
	   t3.operation_center_name,
	   t3.material_num,
	   0 as INSTOCK_QTY,
	   t3.actual_billed_qty as OUTSTOCK_QTY
from out_fact t3
where t3.operation_center_name is not null
)
INSERT OVERWRITE TABLE dm_sw.DM_WM_CENTER_OUTINSTOCK_T partition(stat_month)
select operation_center_name,
	   material_num,
	   sum(nvl(INSTOCK_QTY,0)) as INSTOCK_QTY,
	   sum(nvl(OUTSTOCK_QTY,0)) as OUTSTOCK_QTY,
	   current_timestamp() as dw_last_update_date ,
	   operation_center_code,
	   stat_month
from sub_com_2 t
group by stat_month,
		 operation_center_code,
		 operation_center_name,
		 material_num
