truncate table DM_SW.DM_WM_B1_INOUT_WAREHOUSE_T; --where stat_month >= '${PERIOD_ID_M}';  -- 近2个月
with dm_otd_sales_forecast_stat_data as(   --  客户区域关系
 	select operation_center_code as customer_code,
 	       operation_center_name as customer_short_name,
 	       area_code,
 	       area_name 
 	from dwimd.DWI_MD_APS_CUSTOMER_RELATION_T
	union all
	select '苏沪子公司' as customer_code,'苏沪子公司' as customer_short_name,'子公司' as area_code,'子公司' as area_name
	union all
	select '津冀子公司' as customer_code,'津冀子公司' as customer_short_name,'子公司' as area_code,'子公司' as area_name
	union all
	select '贵州子公司' as customer_code,'贵州子公司' as customer_short_name,'子公司' as area_code,'子公司' as area_name
)
,DM_WM_B1_INVENTORY_T_D_DATA as(  -- 出入库数据
    select stat_month,
		   customer_code,
		   customer_short_name,
		   material_num,
		   SUM(INSTOCK_QTY) as INSTOCK_QTY,
		   SUM(OUTSTOCK_QTY) as OUTSTOCK_QTY
	from (
		select SUBSTRING(T."日期",1,7) as stat_month, 
			   CASE WHEN LENGTH(t.customer_code) < 10 then concat('0000',t.customer_code) else t.customer_code end customer_code,
			   T."客户别名" as customer_short_name , 
			   T."物料编码" as material_num, 
			   sum(T."期间采购收货数量") as INSTOCK_QTY, 
			   sum(T."期间销售出库数量") as OUTSTOCK_QTY  
		from DM.DM_WM_B1_INVENTORY_T_D T
		group by SUBSTRING(T."日期",1,7),
				 T."客户别名",
				 T."物料编码",
				 CASE WHEN LENGTH(t.customer_code) < 10 then concat('0000',t.customer_code) else t.customer_code end
		union ALL
		select T.stat_month ,
			   T.operation_center_code AS customer_code,
			   T.operation_center_name as customer_short_name,
			   T.material_num ,
			   T.instock_qty ,
			   T.outstock_qty 
		from DM_SW.DM_WM_CENTER_OUTINSTOCK_T T
	) T
	group by stat_month,
			 customer_code,
			 customer_short_name,
			 material_num
),
detail_data as(
select t1.stat_month,  -- 单据日期
	   t2.customer_code, -- 客户编码
	   t2.customer_short_name, -- 客户别名
	   t2.area_code,  -- 区域编码
	   t2.area_name,  -- 区域名称
	   t1.material_num, -- 物料编码
	   t3.material_NAME,-- 物料名称
	   T3.product_level2_code as PROD_CATEGORY_CODE,  -- 产品类别编码
	   T3.product_level2_name as PROD_CATEGORY_NAME,  -- 产品类别名称
	   CASE WHEN T4.matl_group4_code = 'TS' THEN 'TS'
		   ELSE 'ZS' END AS PROD_STATUS_CODE,    -- 产品状态编码
	   CASE WHEN T4.matl_group4_code = 'TS' THEN '退市'
		  ELSE '在市' END AS PROD_STATUS_NAME,    -- 产品状态名称
	   nvl(t1.INSTOCK_QTY,0) as INSTOCK_QTY,  -- 入库数量
	   nvl(t1.OUTSTOCK_QTY,0) as  OUTSTOCK_QTY -- 出期出库
from DM_WM_B1_INVENTORY_T_D_DATA t1
join dm_otd_sales_forecast_stat_data t2
--on t1.customer_short_name = t2.customer_short_name
on t1.customer_code = t2.customer_code
JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T3  -- 物料主数据
ON t1.material_num = T3.material_num
join DWIMD.DWI_MD_PRODUCT_LEVEL_APS_T T5  -- APS产品品类数据
on case when nvl(T3.product_level3_code,'NULL') = nvl(T5.prod_level_code,'NULL') then T3.product_level3_code = T5.prod_level_code
        else T3.product_level2_code = T5.prod_level_code end
--on T3.product_level_code = T5.prod_level_code
LEFT JOIN (
   select material_num,
		   max(matl_group4_code) as matl_group4_code ,
		   max(matl_group4_name) as matl_group4_name
	from dwimd.dwi_md_data_material_sales_t   -- 物料销售
	group by material_num 
)T4
ON t1.material_num = T4.material_num
where abs(t1.INSTOCK_QTY) + abs(t1.OUTSTOCK_QTY) <> 0
),
dim_data as(   -- 数据发散，从2021年1月开始
  select SUBSTRING(day_dt,1,7) as stat_month
  from DM.dm_dim_date_d T
  where day_dt between '2021-01-01' and CURRENT_DATE -1 -- concat('${PERIOD_ID_M}','-01')
  group by SUBSTRING(day_dt,1,7)
),
tmp_data_01 as(
select t1.stat_month,  -- 单据日期
       t2.customer_code, -- 客户编码
	   t2.customer_short_name, -- 客户别名
	   t2.area_code,  -- 区域编码
	   t2.area_name,  -- 区域名称
	   t2.material_num, -- 物料编码
	   t2.material_NAME,-- 物料名称
	   T2.PROD_CATEGORY_CODE,  -- 产品类别编码
	   T2.PROD_CATEGORY_NAME,  -- 产品类别名称
	   T2.PROD_STATUS_CODE,    -- 产品状态编码
	   T2.PROD_STATUS_NAME,    -- 产品状态名称
	   nvl(t3.INSTOCK_QTY,0) as INSTOCK_QTY,  -- 入库数量
	   nvl(t3.OUTSTOCK_QTY,0) as OUTSTOCK_QTY -- 出期出库
from dim_data t1  -- 发散维度
left join (
   select distinct
       customer_code, 
	   customer_short_name,  
	   area_code,  
	   area_name,  
	   material_num, 
	   material_NAME,
	   PROD_CATEGORY_CODE,  
	   PROD_CATEGORY_NAME,  
	   PROD_STATUS_CODE,    
	   PROD_STATUS_NAME  
    from detail_data 
    )t2   -- 维度
on 1 = 1
left join detail_data t3
on t1.stat_month = t3.stat_month
and nvl(t2.customer_code,'ISNULL') = nvl(t3.customer_code,'ISNULL')
and nvl(t2.customer_short_name,'ISNULL') = nvl(t3.customer_short_name,'ISNULL')
and nvl(t2.area_code,'ISNULL') = nvl(t3.area_code,'ISNULL')
and nvl(t2.area_name,'ISNULL') = nvl(t3.area_name,'ISNULL')
and nvl(t2.material_num,'ISNULL') = nvl(t3.material_num,'ISNULL')
and nvl(t2.material_NAME,'ISNULL') = nvl(t3.material_NAME,'ISNULL')
and nvl(t2.PROD_CATEGORY_CODE,'ISNULL') = nvl(t3.PROD_CATEGORY_CODE,'ISNULL')
and nvl(t2.PROD_CATEGORY_NAME,'ISNULL') = nvl(t3.PROD_CATEGORY_NAME,'ISNULL')
and nvl(t2.PROD_STATUS_CODE,'ISNULL') = nvl(t3.PROD_STATUS_CODE,'ISNULL')
and nvl(t2.PROD_STATUS_NAME,'ISNULL') = nvl(t3.PROD_STATUS_NAME,'ISNULL')
)
INSERT INTO DM_SW.DM_WM_B1_INOUT_WAREHOUSE_T
select t1.*,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
from tmp_data_01 t1;  -- tmp_data_01  detail_data
