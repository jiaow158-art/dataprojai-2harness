INSERT OVERWRITE TABLE DM.DM_BO_SHIP_ORDER_T
SELECT 
       A.DELV_SHIP_ST_NAME --出货计划状态描述
      ,A.DELV_SHIP_LINE_ID --出货计划行号
	  ,A.DELV_SHIP_CODE --出货计划编码
	  ,A.CREATE_DATE --创建时间
	  ,A.MATERIAL_NUM --物料编码
	  ,A.MATERIAL_NAME --物料名称
	  ,A.SHIP_PLAN_QTY --本次出货数
	  ,A.SAP_ORDER_NUM --SAP订单号
	  ,B.SAP_ORDER_LINE_NUM --SAP订单行号
	  ,C.CREATE_DATE AS SAP_CREATE_DATE --销售订单创建日期
	  ,B.BATCH_NUM --批次
	  ,B.PLAN_FACTORY_CODE --计划出货工厂编码
	  ,B.PLAN_WH_CODE --库存地点
	  ,B.ACTL_SHIP_QTY --实际开单
	  ,A.RSDU_SHIP_QTY --出货数剩余未开单数
	  ,A.DMAND_NUM --要货单编号
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,B.DMAND_LINE_ID AS DMAND_LINE_ID --要货单行ID
  FROM DWRLTC.DWR_BO_DELV_SHIP_ORDER_F A
  LEFT JOIN DWROTD.DWR_BO_DELV_SHIP_BILL_F B
    ON A.DELV_SHIP_LINE_ID = B.DELV_SHIP_LINE_ID
   AND A.DELV_SHIP_ID = B.DELV_SHIP_ID
  LEFT JOIN DWROTD.DWR_SO_HEAD_SAP_F C --SAP订单
    ON A.SAP_ORDER_NUM = C.SAL_VCH_NUM
 WHERE COALESCE(A.DELV_SHIP_ST_NAME,'') <> '已失效'
;
