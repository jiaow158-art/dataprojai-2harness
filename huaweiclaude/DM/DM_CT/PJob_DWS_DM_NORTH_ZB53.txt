--特定产品销售面积ZB53
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB53';
with base as (
select 
'${PERIOD_ID_M}' as calmonth,
'ZB53' as zb_code,
sum(sales_stock_out_area) as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩

from  dm.dm_wm_low_turnover_stockout_detail_t
where  stock_type  = '特惠品' 
and  concat(substring(stat_month,1,4),'-',substring(stat_month,5,2))  = '${PERIOD_ID_M}'


union all 

select 
'${PERIOD_ID_M}' as calmonth,
'ZB53' as zb_code,
0 as 当月实绩,
SUM(sales_stock_out_area) as 累计实绩,
0 as 季度累计实绩
from 
dm.dm_wm_low_turnover_stockout_detail_t
where 
stock_type  = '特惠品' 
and 
concat(substring(stat_month,1,4),'-',substring(stat_month,5,2))  between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'

union all 

select 
'${PERIOD_ID_M}' as calmonth,
'ZB53' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
SUM(sales_stock_out_area) as 季度累计实绩
from 
dm.dm_wm_low_turnover_stockout_detail_t
where    
stock_type  = '特惠品' 
and 
to_char((SUBSTRING( concat(substring(stat_month,1,4),'-',substring(stat_month,5,2)) ,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING( concat(substring(stat_month,1,4),'-',substring(stat_month,5,2)) ,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)

)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)/10000 as CM_STAT_AMT,
sum(累计实绩)/10000 as ACC_STAT_AMT,
sum(季度累计实绩)/10000 as QTR_TGT_AMT
from 
base
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
