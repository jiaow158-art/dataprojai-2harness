--逾期应收总额（工程市场中心-新增逾期与历史逾期）ZB62 ZB63
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}'  
and zb_code IN ('ZB62','ZB63');
with CREDIT_DEVALUE as (
select 
SUBSTRING(T1.query_date,1,7) as CALMONTH 
,
SUM(
case when
T1.general_ledger_account in ('0011310800','0011310100','0011310700','0011110100','0011310600','0011310300')
and T1.is_overdue_cust = 'N'
and coalesce (T1.comp_code,'#')  <> '8000'
and coalesce (T1.cust_group_code,'#')<> 'Z0'
and T2.node_name4 = 'H04230685'
then T1.overdue_receivables_2023_after 
else 0 END) as "逾期应收总额（工程市场中心-新增逾期）"

,
SUM(
case when
T1.general_ledger_account in ('0011310800','0011310100','0011310700','0011110100','0011310600','0011310300')
and T1.is_overdue_cust = 'N'
and coalesce (T1.comp_code,'#')  <> '8000'
and coalesce (T1.cust_group_code,'#')<> 'Z0'
and T2.node_name5 = 'H05230609'
then T1.overdue_receivables_2023_ago 
else 0 END) as "逾期应收总额（战略工程-历史逾期）"
from 
DWRFIN.DWR_AR_RECEIVABLE_AGING_2023_INFO_F T1
left join  
dm.dm_rpt_sales_group_t  t2 
on t1.wbs_sales_group = t2.node_name9
where  
query_year = SUBSTRING('${PERIOD_ID_M}' ,1,4)
and 
T1.query_date = case when SUBSTRING('${PERIOD_ID_M}' ,1,7) = substring(current_date,1,7)  
then substring(current_date- INTERVAL '1 day',1,10) 
else SUBSTR(concat(CONCAT (SUBSTRING ('${PERIOD_ID_M}' ,1,7)) ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)  end 
group by SUBSTRING(T1.query_date,1,7)

)
,AMT as (
select 
CALMONTH ,
'ZB62' as  zb_code,
 "逾期应收总额（工程市场中心-新增逾期）"/10000 as CM_STAT_AMT,
 "逾期应收总额（工程市场中心-新增逾期）"/10000 as ACC_STAT_AMT,
 "逾期应收总额（工程市场中心-新增逾期）"/10000 as QTR_TGT_AMT
 from 
CREDIT_DEVALUE
union all 
select 
CALMONTH ,
'ZB63' as  zb_code,
 "逾期应收总额（战略工程-历史逾期）"/10000 as CM_STAT_AMT,
 "逾期应收总额（战略工程-历史逾期）"/10000 as ACC_STAT_AMT,
 "逾期应收总额（战略工程-历史逾期）"/10000 as QTR_TGT_AMT
 from 
CREDIT_DEVALUE
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
