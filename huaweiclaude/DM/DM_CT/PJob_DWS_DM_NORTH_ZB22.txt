delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB22';

--大规格产品销售面积占比（零售渠道）
with bigmjzb as (
select 
calmonth
,SUM(
case when  
T2.node_name3 ='H03230618'	
and coalesce (t2.node_name5,'#') not in ('H03230605','H03230612','H03230604','H03230627')
and (t1.integrate_channel = 'GD01' or t1.integrate_channel is null)
and t1.data_source in ('S','T','D') 
and (mt.prod_property in ('A0001','C0002','A0002','D0001','C0001','F0001') or mt.product_series_name in ('918','超大板'))
then T1.ambperformance else 0  end 
)as "大规格产品"
,SUM(
case when  
T2.node_name3 ='H03230618'	
and coalesce (t2.node_name5,'#') not in ('H03230605','H03230612','H03230604','H03230627')
and (t1.integrate_channel = 'GD01' or t1.integrate_channel is null)
and t1.data_source in ('S','T','D') 
then T1.ambperformance else 0  end 
)as "整体销售面积"
from dm.dm_ambv2_zzt_xssr_t t1
left join  
dm.dm_rpt_sales_group_t  t2   
on t1.sales_grp  = t2.node_name9
left join 
dwimd.dwi_md_data_material_general_t mt
on t1.material  = mt.material_num 
left join 
DWRLTC.DWR_BO_PROJECT_DETAIL_F PROJECT_DETAIL
on t1.zsjbm = PROJECT_DETAIL.bo_code
group by calmonth
)
,bigmjzb2 as (
select 
calmonth,
'ZB22' as zb_code,
case when 整体销售面积 = 0 then 0 else 大规格产品/整体销售面积 end  as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
bigmjzb
where calmonth = '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB22' as zb_code,
0 as 当月实绩,
SUM(大规格产品)/SUM(整体销售面积) as 累计实绩,
0 as 季度累计实绩
from 
bigmjzb
where CALMONTH between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB22' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
SUM(大规格产品)/SUM(整体销售面积) as 季度累计实绩
from 
bigmjzb
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)*100 as CM_STAT_AMT,
sum(累计实绩)*100 as ACC_STAT_AMT,
sum(季度累计实绩)*100 as QTR_TGT_AMT
from 
bigmjzb2
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.PJFX
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
