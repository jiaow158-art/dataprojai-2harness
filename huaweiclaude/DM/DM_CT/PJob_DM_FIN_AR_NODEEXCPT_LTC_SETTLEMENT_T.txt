INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_SETTLEMENT_T
SELECT DISTINCT
       D.WBS
      ,A.BUSINESS_CODE --商机编码
      ,A.REGISTER_STATUS --结算状态
	  ,A.BACK_DATE --回单日期
	 -- ,A.IS_SOURCE_FILE --结算单是否回收 回单是否签收
	  ,A.SETTLEMENT_BILL_CODE --结算单号
	  ,B.MEANING AS REGISTER_STATUS_NAME --结算状态描述
	  ,C.ESTIMATED_COMPLETE_DATE --预计竣工日期
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_BILL_1033 A
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B
    ON A.REGISTER_STATUS = B.LOOKUP_CODE
   AND B.LOOKUP_TYPE = 'REGISTER_STATUS'
   AND B.ENABLED_FLAG = 'Y'
  LEFT JOIN(SELECT DISTINCT PR.BUSINESS_CODE
		                   ,PR.ESTIMATED_COMPLETE_DATE --预计竣工日期
	          FROM SDI.SDI_PROJECT_REVIEW_1033 PR
	         WHERE PR.CONTRACT_FLAG = 'Y'
	           AND PR.DELETE_FLAG = 0 
           ) C
	ON A.BUSINESS_CODE = C.BUSINESS_CODE
  LEFT JOIN SDI.SDI_BUSINESS_OPPORTUNITY_1033 D
    ON A.BUSINESS_CODE = D.BUSINESS_CODE
   AND D.DELETE_FLAG = 0    
 WHERE A.DELETE_FLAG = 0
