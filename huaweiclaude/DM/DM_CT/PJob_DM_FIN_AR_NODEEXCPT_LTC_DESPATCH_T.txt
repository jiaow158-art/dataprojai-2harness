WITH T1 AS(
SELECT          REFERENCE_BILL_NUM
               ,BILL_VOUCHER_NUM
  FROM DWIFIN.DWI_GL_INVOICE_ITEM_T A
 WHERE A.PERIOD_ID_M >= '202001'
   AND REFERENCE_BILL_NUM IN(SELECT DELIVERY_NO
                               FROM SDI.SDI_DELIVER_DESPATCH_HEADER_1033
                            )
),T2 AS(
SELECT DISTINCT
       T1.REFERENCE_BILL_NUM
  FROM T1
  LEFT JOIN SDI.SDI_ZKHFP_1001 B
    ON B.VBELN = T1.BILL_VOUCHER_NUM
 WHERE LENGTH(B.EERT) > 0
)
INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_DESPATCH_T
SELECT DISTINCT
           D.WBS
		  ,DDHS.LOGISTICS_TYPE --运输方式编码
		  ,B2.MEANING AS LOGISTICS_TYPE_NAME --运输方式描述		  
		  ,DDHS.DELIVERY_NO --提货单号
		  ,DSH.SIGN_STATUS --签收状态
		  ,CASE WHEN T2.REFERENCE_BILL_NUM IS NOT NULL THEN 'Y' ELSE 'N' END AS IS_SOURCE_FILE --结算单是否回收 回单是否签收
		  ,DSH.CREATION_DATE AS PICK_DATE --提货日期
		  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	FROM SDI.SDI_DELIVER_DESPATCH_HEADER_1033 DDHS
   INNER JOIN SDI.SDI_DELIVER_DESPATCH_LINE_1033 DDL
	ON DDHS.DESPATCH_ID = DDL.DESPATCH_ID
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B2
          ON DDHS.LOGISTICS_TYPE = B2.LOOKUP_CODE
         AND B2.LOOKUP_TYPE = 'DELIVERY_METHOD_C'
         AND B2.ENABLED_FLAG = 'Y'
   INNER JOIN SDI.SDI_CONTRACT_1033 D
	  ON DDHS.CONTRACT_ID = D.CONTRACT_ID	 
	LEFT JOIN SDI.SDI_DELIVER_SIGN_HEADER_1033 DSH --签收单
	  ON DDHS.BUSINESS_ID = DSH.BUSINESS_ID
	 AND DDHS.DESPATCH_ID = DSH.DESPATCH_ID 
	 AND DDHS.DELIVERY_NO =  DSH.DELIVERY_NO
	 AND D.WBS = DSH.WBS
	 AND DSH.SIGN_STATUS = 'ALREADY_SIGNED'
	 AND DSH.DELETE_FLAG = 0
	LEFT JOIN SDI.SDI_SETTLEMENT_BILL_1033 E
	  ON DSH.BUSINESS_CODE = E.BUSINESS_CODE
	 AND E.DELETE_FLAG = 0 
	LEFT JOIN(SELECT B.CONTRACT_CODE
					,MAX(A.PICK_DATE) AS PICK_DATE --提货日期
	             FROM SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_DETAIL_1033 A
				 LEFT JOIN SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_1033 B
				   ON A.REQUEST_PAYOUT_ID = B.REQUEST_PAYOUT_ID
				WHERE B.DELETE_FLAG = 0
				  AND A.DELETE_FLAG = 0
			   GROUP BY B.CONTRACT_CODE
	         ) F
      ON D.CONTRACT_CODE = F.CONTRACT_CODE
	LEFT JOIN T2
	  ON T2.REFERENCE_BILL_NUM = DDHS.DELIVERY_NO
   WHERE DDHS.DELIVERY_NO IS NOT NULL
	 AND DSH.BUSINESS_ID IS NOT NULL
	 AND DDHS.DELETE_FLAG = 0
	 AND DSH.DELETE_FLAG = 0
;
