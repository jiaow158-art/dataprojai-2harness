--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

with tmp_data00 as (
select T.*
       ,dense_rank() over(order by CALENDAR_MONTH desc) as rk
from DWRFIN.DWR_GL_GROUP_DAYRPT_F T
),
tmp_data01 as(     -- 瓷砖事业部业务需要维护的运营中心数据
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'浙江运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'湘鄂运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'山东运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'闽琼运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'晋豫运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'江苏运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'黑吉辽运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'广东运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'京津冀运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'西北运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'西南运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '区域运营中心' AS SUB_SYSTEM_NAME,'赣皖运营中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'陕西子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'云南子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'广西子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'上海子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'深圳子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'广州子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '直营经营管理板块' AS SUB_SYSTEM_NAME,'佛山子公司'AS subcompany_big_area_NAME UNION ALL
	SELECT '工程市场中心' AS SUB_SYSTEM_NAME,'战略工程中心'AS subcompany_big_area_NAME UNION ALL
	SELECT '工程市场中心' AS SUB_SYSTEM_NAME,'商机资源开发部'AS subcompany_big_area_NAME UNION ALL
	SELECT '工程市场中心' AS SUB_SYSTEM_NAME,'幕墙工程中心'AS subcompany_big_area_NAME
),
tmp_data02 as(    -- 取最新一个月的数据做为维度（排除工程市场中心的数据）
	SELECT T.UTILITIES_DEPT_CODE,
	       T.UTILITIES_DEPT,  -- 事业部
	       T.SYSTEM_CODE,
	       T.SYSTEM_NAME,    -- 系统
	       T.SUBCOMPANY_BIG_AREA_CODE,
	       T.SUBCOMPANY_BIG_AREA_NAME   -- 子公司大区
	FROM tmp_data00 T
	where t.SUBCOMPANY_BIG_AREA_NAME not in ('战略工程中心','商机资源开发部','幕墙工程中心')
	AND t.rk = 1
	GROUP BY T.UTILITIES_DEPT_CODE,
	         T.UTILITIES_DEPT,
	         T.SYSTEM_CODE,
	         T.SYSTEM_NAME,
	         T.SUBCOMPANY_BIG_AREA_CODE,
	         T.SUBCOMPANY_BIG_AREA_NAME
 ) ,
 tmp_data03 as( -- 强制给工程市场中心挂在子系统下
 SELECT  T2.UTILITIES_DEPT_CODE,
         T2.UTILITIES_DEPT,  -- 事业部
         T2.SYSTEM_CODE,
         T2.SYSTEM_NAME,    -- 系统
         CASE WHEN T2.UTILITIES_DEPT = '瓷砖事业部' AND T2.SYSTEM_NAME = '国内营销系统' AND T2.SUBCOMPANY_BIG_AREA_NAME = '工程市场中心'
              THEN ''--T2.subcompany_big_area_code
           END  AS SUB_SYSTEM_CODE,
         CASE WHEN T2.UTILITIES_DEPT = '瓷砖事业部' AND T2.SYSTEM_NAME = '国内营销系统' AND T2.SUBCOMPANY_BIG_AREA_NAME = '工程市场中心'
              THEN T2.SUBCOMPANY_BIG_AREA_NAME
            ELSE T1.SUB_SYSTEM_NAME
         END SUB_SYSTEM_NAME, -- 子系统
         T2.SUBCOMPANY_BIG_AREA_CODE,
         T2.SUBCOMPANY_BIG_AREA_NAME   -- 运营中心
 FROM tmp_data02 T2
 LEFT JOIN tmp_data01 T1
 ON T2.SUBCOMPANY_BIG_AREA_NAME = T1.subcompany_big_area_NAME
 ),
 tmp_data04 as(  -- 取出工程市场中心数据，且给子公司挂在直营经营管理板块下
 SELECT T1.UTILITIES_DEPT_CODE,
         T1.UTILITIES_DEPT,  -- 事业部
         T1.SYSTEM_CODE,
         T1.SYSTEM_NAME,    -- 系统
         T1.SUB_SYSTEM_CODE,
         T1.SUB_SYSTEM_NAME, -- 子系统
         '' AS SUBCOMPANY_BIG_AREA_CODE,
         T2.SUBCOMPANY_BIG_AREA_NAME   -- 运营中心
 FROM tmp_data03 T1
 LEFT JOIN tmp_data01 T2
 ON T1.SUB_SYSTEM_NAME = T2.SUB_SYSTEM_NAME
 WHERE T1.SUB_SYSTEM_NAME = '工程市场中心'
 UNION ALL
 SELECT T1.UTILITIES_DEPT_CODE,
         T1.UTILITIES_DEPT,  -- 事业部
         CASE WHEN T1.UTILITIES_DEPT = '瓷砖事业部' AND T1.sub_system_name = '直营经营管理板块' AND T1.subcompany_big_area_name IN ('深圳子公司','广州子公司','上海子公司','陕西子公司','广西子公司','云南子公司','佛山子公司')
              THEN '22000001'
           ELSE T1.SYSTEM_CODE
         END SYSTEM_CODE,
         CASE WHEN T1.UTILITIES_DEPT = '瓷砖事业部' AND T1.sub_system_name = '直营经营管理板块' AND T1.subcompany_big_area_name IN ('深圳子公司','广州子公司','上海子公司','陕西子公司','广西子公司','云南子公司','佛山子公司')
             THEN '国内营销系统'
           ELSE T1.SYSTEM_NAME
         END SYSTEM_NAME,    -- 系统
         T1.SUB_SYSTEM_CODE,
         T1.SUB_SYSTEM_NAME, -- 子系统
         T1.SUBCOMPANY_BIG_AREA_CODE,
         T1.SUBCOMPANY_BIG_AREA_NAME   -- 运营中心
 FROM tmp_data03 T1
 WHERE nvl(T1.SUB_SYSTEM_NAME,'NA') <> NVL('工程市场中心','NA')
 ) ,
 TMP_DATA05 AS( -- 取销售部门名称
SELECT DISTINCT T1.utilities_dept as UTILITIES_DEPT_CODE,
	   T1.system as SYSTEM_CODE, 
       T1.subcompany_big_area as SUBCOMPANY_BIG_AREA_CODE,
	   T1.sales_dept as sale_dept_code,
	   T2.sales_dept_desc,
       T1.sales_group
FROM DWRFIN.DWR_GL_SALES_GROUP_F T1
LEFT JOIN DWRFIN.DWR_GL_SALES_DEPT_F T2
ON T1.sales_dept = T2.sales_dept_code 
AND t2.language_no = '1'
),
tmp_data06 as( -- 取出维度到销售组的数据
select T1.UTILITIES_DEPT_CODE,
       T1.UTILITIES_DEPT AS UTILITIES_DEPT_NAME,  -- 事业部
       T1.SYSTEM_CODE,
       T1.SYSTEM_NAME,    -- 系统
       T1.SUB_SYSTEM_CODE,
       T1.SUB_SYSTEM_NAME, -- 子系统
       T1.SUBCOMPANY_BIG_AREA_CODE AS OPERATION_CENTER_CODE,
       T1.SUBCOMPANY_BIG_AREA_NAME AS OPERATION_CENTER_NAME,   -- 运营中心
       T2.SALE_DEPT_CODE,
	   T2.SALES_DEPT_DESC AS SALES_DEPT_NAME, -- 销售部门
       T2.SALES_GROUP AS SALES_GROUP_CODE -- 销售组
from tmp_data04 t1
left join TMP_DATA05 t2
on t1.UTILITIES_DEPT_CODE = t2.UTILITIES_DEPT_CODE
and t1.SYSTEM_CODE = t2.SYSTEM_CODE
and t1.SUBCOMPANY_BIG_AREA_CODE = t2.SUBCOMPANY_BIG_AREA_CODE
),
project_data01 as(  -- 取出工程市场中心下的销售部门及销售组
	SELECT distinct 
	       t.utilities_dept as utilities_dept_code,
	       t.system as system_code,
	       t.subcompany_big_area as sub_system_code, 
	       '工程市场中心' as sub_system_name,
	       T.sales_dept as sales_dept_code,
	       T.sales_group as sales_group_code
	FROM DWRFIN.DWR_GL_SALES_GROUP_F T
	WHERE T.subcompany_big_area = '33000046'
	AND T.sales_dept <> '' 
),
project_data02 as(  -- 取出工程市场中心下的销售部门及销售组
select t1.utilities_dept_code,   -- 事业部编码
       t1.system_code,     -- 系统编码
       t1.sub_system_code,   -- 子系统
       t1.sub_system_name,
       '' as  operation_center_code,-- 运营中心
       case when t1.sales_dept_code = 'Z031' then '幕墙工程中心' -- 幕墙工程中心
            when t1.sales_dept_code = 'Z033' then '商机资源开发部' -- 商机资源开发部
          else '战略工程中心'
       end operation_center_name,-- 运营中心
       t1.sales_dept_code,    -- 销售部门
       t2.sales_dept_desc as sales_dept_name ,
       t1.sales_group_code
from project_data01 t1
left join DWRFIN.DWR_GL_SALES_DEPT_F t2
on t1.sales_dept_code = t2.sales_dept_code
AND t2.language_no = '1'
),
RESULT_DATA01 AS(  -- 取出所有数据，需要判断工程市场特殊情况
select T1.UTILITIES_DEPT_CODE,
       T1.UTILITIES_DEPT_NAME,  -- 事业部
       T1.SYSTEM_CODE,
       T1.SYSTEM_NAME,    -- 系统
       T1.SUB_SYSTEM_CODE,
       T1.SUB_SYSTEM_NAME, -- 子系统
       T1.OPERATION_CENTER_CODE,
       T1.OPERATION_CENTER_NAME,   -- 运营中心
       CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '工程市场中心' THEN
             T2.sales_dept_code
         ELSE T1.SALE_DEPT_CODE
        END SALE_DEPT_CODE,
       CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '工程市场中心' THEN
            T2.sales_dept_name
	     ELSE T1.SALES_DEPT_NAME
	   END SALES_DEPT_NAME, -- 销售部门
	   CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '工程市场中心' THEN
	        t2.SALES_GROUP_CODE
         else T1.SALES_GROUP_CODE
       end SALES_GROUP_CODE -- 销售组
from tmp_data06 t1
left join project_data02 t2
on t1.utilities_dept_code = t2.utilities_dept_code
and t1.system_code = t2.system_code
and t1.sub_system_name = t2.sub_system_name
and t1.operation_center_name = t2.operation_center_name
),
sub_company_data01 as(  -- 取出子公司下的销售部门及销售组
	SELECT distinct 
	       t.utilities_dept as utilities_dept_code,
	       '22000001' as system_code,
	       '' AS sub_system_code,
	       '直营经营管理板块' as sub_system_name,
	       t.subcompany_big_area as operation_center_code,
	       T.sales_dept as sales_dept_code,
	       T.sales_group as sales_group_code
	FROM DWRFIN.DWR_GL_SALES_GROUP_F T
	WHERE T.subcompany_big_area in ('33000003','33000004','33000005','33000006','33000007','33000008','33000013')
),
sub_company_data02 AS( -- 取出子公司下的销售部门及销售组
SELECT T1.utilities_dept_code,
       T1.system_code,
       T1.sub_system_code,
       T1.sub_system_name,
       T1.operation_center_code,
       T1.sales_dept_code,
       T2.sales_dept_desc as sales_dept_name,
       T1.sales_group_code
FROM sub_company_data01 T1
left join DWRFIN.DWR_GL_SALES_DEPT_F t2
on t1.sales_dept_code = t2.sales_dept_code
AND t2.language_no = '1'
),
RESULT_DATA02 AS(  -- 取出所有数据，需要判断直营经营管理板块特殊情况
SELECT T1.UTILITIES_DEPT_CODE,
       T1.UTILITIES_DEPT_NAME,  -- 事业部
       T1.SYSTEM_CODE,
       T1.SYSTEM_NAME,    -- 系统
       T1.SUB_SYSTEM_CODE,
       T1.SUB_SYSTEM_NAME, -- 子系统
       T1.OPERATION_CENTER_CODE,
       T1.OPERATION_CENTER_NAME,   -- 运营中心
       CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '直营经营管理板块' THEN
            T2.sales_dept_code
        ELSE T1.SALE_DEPT_CODE
        END SALES_DEPT_CODE,
       CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '直营经营管理板块' THEN
            T2.sales_dept_name
	     ELSE T1.SALES_DEPT_NAME
	   END SALES_DEPT_NAME, -- 销售部门
	   CASE WHEN T1.UTILITIES_DEPT_NAME = '瓷砖事业部' AND T1.SUB_SYSTEM_NAME = '直营经营管理板块' THEN
	        t2.SALES_GROUP_CODE
         else T1.SALES_GROUP_CODE
       end SALES_GROUP_CODE -- 销售组
FROM RESULT_DATA01 T1
LEFT JOIN sub_company_data02 T2
ON T1.utilities_dept_code = T2.utilities_dept_code
AND T1.system_code = T2.system_code
AND T1.sub_system_name = T2.sub_system_name
AND T1.operation_center_code = T2.operation_center_code
)
INSERT OVERWRITE TABLE DM.DM_GL_GROUP_STRUCTURE_D
SELECT T1.*,
       T2.txtsh as SALES_GROUP_NAME, -- 销售组名称 
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE  -- 数据同步时间
FROM RESULT_DATA02 T1
LEFT JOIN SDI.SDI_TSALES_GRP_1009 T2  -- 销售组
ON T1.SALES_GROUP_CODE = T2.sales_grp
AND T2.langu = '1'
AND T2.DEL_FLAG = 'N'
