,busi_name	
,cust_num

 busi_num 
,busi_name	
,cust_num

,coalesce(cntc_qty,0)
,coalesce(sale_amt,0)

 busi_num 
,busi_name	
,cust_num	
,cust_name	 
,busi_belong_name

,sum(zxssl) as zxssl

	  when sum(v_aply_qty) <> 0 then sum(v_sale_amt) / sum(v_aply_qty)
	  when sum(v_zxssl) <> 0 then sum(v_ambperformance) / sum(v_zxssl)
	  else 0 end as "单价(有效)"

	  when sum(aply_qty) <> 0 then sum(sale_amt) / sum(aply_qty)
	  when sum(zxssl) <> 0 then sum(ambperformance) / sum(zxssl)
	  else 0 end as "单价(无效)"

 busi_num 
,busi_name	
,cust_num	
,cust_name	 
,busi_belong_name
,busi_4_lev_desc,busi_5_lev_desc,busi_6_lev_desc,busi_7_lev_desc,busi_stage_name,material_num,unit_area

select  
 busi_num 
,busi_name	
,cust_num	
,cust_name
,busi_4_lev_desc,busi_5_lev_desc,busi_6_lev_desc,busi_7_lev_desc,busi_stage_name,material_num,unit_area,busi_belong_name

,sum(case when busi_belong_name like '%经销%' then (case when "合同用量(有效)"-"订单数量(基于合同)"+"订单退货数量(基于合同)" < 0 then 0

                    else "合同用量(有效)"-GREATEST("订单数量(基于合同)"+"订单退货数量(基于合同)" ,review_plan_qty) end) end) as "已签合同未下单数量"
,sum(case when (GREATEST(qty,review_plan_qty) - proj_rmn_qty - rmn_qty) < 0 then 0 
	  else GREATEST(qty,review_plan_qty) - proj_rmn_qty - rmn_qty end) as "已下单未排产数量"
,sum(case when (proj_rmn_qty + rmn_qty + t_rmn_qty- zxssl) < 0 then 0 
	  else proj_rmn_qty + rmn_qty + t_rmn_qty - zxssl end) as "已生产未出库数量"
,sum(coalesce("合同用量(有效)",0)) as "合同用量(有效)"
,sum(coalesce("订单数量(基于合同)",0)) as "订单数量(基于合同)"
,sum(coalesce("订单退货数量(基于合同)",0)) as "订单退货数量(基于合同)"

,sum(qty) as qty
,sum(rmn_qty) as rmn_qty
,sum(t_rmn_qty) as t_rmn_qty
,sum(proj_rmn_qty) as proj_rmn_qty
,sum(zxssl) as zxssl
,sum(ambperformance) as ambperformance 
,sum(sale_amt) as sale_amt
,sum(aply_qty) as aply_qty
,sum(cntc_qty) as cntc_qty
,sum(case when "单价(有效)" = 0 then "单价(无效)" else "单价(有效)" end) as "单价"

 busi_num 
,busi_name	
,cust_num	
,cust_name
,busi_4_lev_desc,busi_5_lev_desc,busi_6_lev_desc,busi_7_lev_desc,busi_stage_name,material_num,unit_area ,busi_belong_name

,sum(coalesce("已下单未排产数量",0)) as "已下单未排产数量"

 busi_num
)

 coalesce(z_gcbbh,zsjbm) as busi_code
--,customer as cust_code
,min(calmonth) as calmonth 
,sum(case when substr(calmonth,1,4) = substr('${PERIOD_ID_D}',1,4) and calmonth <= substr('${PERIOD_ID_D}',1,7) then coalesce(ambperformance,0) else 0 end) as ambperformance
,sum(case when substr(calmonth,1,4) = substr('${PERIOD_ID_D}',1,4) and calmonth <= substr('${PERIOD_ID_D}',1,7) then coalesce(zxsmj,0) else 0 end) as zxsmj
,sum(case when calmonth = substr('${PERIOD_ID_D}',1,7) then coalesce(ambperformance,0) else 0 end) as m_yj_amt
,sum(case when calmonth = to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD') - interval '1 year','YYYY-MM') then coalesce(ambperformance,0) else 0 end) as m_yj_amt_prev
from dm.dm_ambv2_zzt_xssr_t T
left join dm.dm_rpt_sales_group_t T6
on T.sales_grp = T6.node_name9

and data_source not in ('D','U') and T6.node_name5 <> 'H03230605' 
and T.integrate_channel = 'GD03' and T6.node_name2 = 'H11000001' 
group by 
coalesce(z_gcbbh,zsjbm)
--customer
)
,tmp1 as (

,tmp0 as (
select 
 '0' as flag  --有商机有业绩
,substr('${PERIOD_ID_D}',1,7) as STAT_MONTH 
,T.busi_code 
,case when T.busi_belong_name like '%经销商%' then T.CUST_NUM else '' end as cust_num  --经销商
--,case when T.busi_belong_name like '%经销商%' then '' else T.cust_comp_name end as cust_name  --经销商
,'' as par_cust_num  --经销商母客编
,T.city_name --商机城市
,case when T.busi_belong_name like '%经销商%' then coalesce(HT.city_name,T6.city_name) else T5.city_name end as HT_CITY_NAME--合同出货城市
,case when T.busi_belong_name like '%经销商%' then T.emp_num else '' end as employee_code--销售雇员
,case when T.busi_belong_name like '%经销商%' then T.emp_name else '' end as employee_name--销售雇员

,case when T.busi_belong_name like '%经销商%' then HT.CUST_TYPE_CODE else '' end as CUST_TYPE_CODE

,T.busi_belong_name--商机归属方
,case when T.busi_belong_name like '%经销商%' then coalesce(HT.cust_yj_goat,0) else '' end as cust_yj_goat-- 商的业绩目标
,sum(T2.ambperformance) as act_ache_amt  --实际达成业绩（万元）
,sum(T2.m_yj_amt) as m_yj_amt--当月实际业绩
,sum(T2.m_yj_amt_prev) as m_yj_amt_prev --同期实际业绩
,sum(case when HT.city_name = T.city_name then T2.ambperformance else 0 end ) as cust_city_amt --商在本市产出业绩（万元）
,sum(case when substr(T.busi_succ_date,1,4) = substr('${PERIOD_ID_D}',1,4) and substr(T.busi_succ_date,1,7) <= substr('${PERIOD_ID_D}',1,7) then 1 else 0 end) as y_sum_busi_QTY --本年累计报备数量(条)
,sum(case when substr(T.busi_succ_date,1,4) = substr('${PERIOD_ID_D}',1,4) and substr(T.busi_succ_date,1,7) <= substr('${PERIOD_ID_D}',1,7) then coalesce(T.build_area,0) else 0 end) as y_sum_busi_area --本年累计报备面积（万㎡）
,sum(case when substr(T.s45_tran_time,1,4) = substr('${PERIOD_ID_D}',1,4) and substr(T.s45_tran_time,1,7) <= substr('${PERIOD_ID_D}',1,7) then 1 else 0 end) as tran_busi_qty--本年商机转化数量（条）
,sum(case when substr(T.s45_tran_time,1,4) = substr('${PERIOD_ID_D}',1,4) and substr(T.s45_tran_time,1,7) <= substr('${PERIOD_ID_D}',1,7) then T3.aply_area else 0 end) as tran_busi_area--本年商机转化面积（万M²）
,sum(case when substr(T.s45_tran_time,1,4) = substr('${PERIOD_ID_D}',1,4) and substr(T.s45_tran_time,1,7) <= substr('${PERIOD_ID_D}',1,7) then T3.sale_amt else 0 end) as tran_busi_amt--本年商机转化金额（万元）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') 
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then 1 else 0 end) as S14_EFF_BUSI_QTY --S1-S4阶段的有效商机数量（条）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') 
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then coalesce(T.build_area,0) else 0 end) as S14_EFF_BUSI_AREA --S1-S4阶段商机采购面积（万M²）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') and T.predict_pur_amt >= 300
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then 1 else 0 end) as S14_300_EFF_BUSI_QTY --S1-S4阶段的300万以上商机数量（条）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') and T.predict_pur_amt >= 300
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then coalesce(T.predict_pur_amt,0) else 0 end) as S14_300_EFF_BUSI_amt --S1-S4阶段的300万以上商机金额（万元）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') and T.tactic_cus_lev_name = 'NKA'
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then 1 else 0 end) as S14_NKA_EFF_BUSI_QTY --S1-S4阶段的NKA商机数量（条）
,sum(case when substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') and T.tactic_cus_lev_name = 'NKA'
and (T.apply_staus_name <> '供货完毕' or T.apply_staus_name is null) and T.busi_stage_name in ('S1','S2','S3','S4') then coalesce(T.predict_pur_amt,0) else 0 end) as S14_NKA_EFF_BUSI_AMT--S1-S4阶段的NKA商机金额（万元）
,sum(case when T.busi_stage_name <> 'SX' and substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') then T4."已签合同未下单金额" else 0 end ) as signed_no_ord_amt --已价审未下单金额
,sum(T4."已下单未排产金额" ) as orded_no_prod_amt --已下单未生产金额
,sum(T4."已生产未出库金额" ) as proded_no_out_amt --已生产未出库金额
,sum(case when T.busi_stage_name <> 'SX' and substr(T.state_value_end_date,1,10) > last_day('${PERIOD_ID_D}') then T4."已签合同未下单数量" else 0 end ) as signed_no_ord_qty --已价审未下单数量
,sum(T4."已下单未排产数量" ) as orded_no_prod_qty --已下单未生产数量
,sum(T4."已生产未出库数量" ) as proded_no_out_qty --已生产未出库数量
,sum("合同用量(有效)") as "合同用量(有效)"
,sum("订单数量(基于合同)") as "订单数量(基于合同)"
from DWRLTC.DWR_PROJ_BUSI_GATHER_T T
left join HT  
on HT.CUST_NUM = case when T.busi_belong_name like '%经销商%' then T.CUST_NUM else cust_comp_code end 
and HT.RN = 1
left join tmp T2 
on T.busi_code = T2.busi_code
left join tmp1 T3
on T.busi_code = T3.busi_num
left join tran2 T4
on T.busi_code = T4.busi_num 
left join UPLOAD.UPLOAD_T65_CITY_GRP_T T5
on T.sales_grp_code = T5.SALES_GRP
left join org_city T6 
on coalesce(T.sales_org,'#') = T6.sale_org
where 1=1 and dept_name = '瓷砖事业部'
and (busi_type_name <> '战略集采' or busi_type_name is null )
group by 
 T.busi_code 
,case when T.busi_belong_name like '%经销商%' then T.CUST_NUM else '' end    --经销商
,T.city_name --商机城市
,case when T.busi_belong_name like '%经销商%' then coalesce(HT.city_name,T6.city_name) else T5.city_name end--合同出货城市
,case when T.busi_belong_name like '%经销商%' then T.emp_num else '' end --销售雇员
,case when T.busi_belong_name like '%经销商%' then T.emp_name else '' end --销售雇员
