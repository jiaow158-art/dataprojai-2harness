set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

INSERT OVERWRITE TABLE DM.DM_WM_ECO_COST_RCVY_STAT_T PARTITION(STAT_MONTH)
SELECT MATERIAL_CODE --物料编号
      ,MATERIAL_NAME --物料描述
      ,CASE WHEN MATERIAL_NAME = '通用编码' THEN '深加工'
	        WHEN COST_CENTER = '2100MC0001' THEN '深加工'
            WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE NOT LIKE '%_A'
			 AND MATERIAL_CODE NOT LIKE '%_B' THEN '半成品'
			WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE LIKE '%_A' 
			 AND COST_CENTER = '2100MA0002' THEN '优等品'
			WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE LIKE '%_B'
			 AND COST_CENTER = '2100MA0002' THEN '合格品'
			 END AS MATERIAL_TYPE --产品分类
	  ,CASE WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
	         AND (MATERIAL_CODE LIKE '%_A' OR MATERIAL_CODE LIKE '%_B')
	        THEN REGEXP_REPLACE(MATERIAL_NAME,'[A-Za-z0-9]','')
			 END AS MATERIAL_COLOUR_TYPE --颜色分类
	  ,SUM(CASE WHEN MATERIAL_CODE LIKE '%_B' THEN 0.75 * QTY
	            ELSE QTY
				 END
		  )	AS TOTAL_QTY --数量
	  ,SUM(CASE WHEN MATERIAL_CODE LIKE '%_B' THEN 0.75 * QTY * UNIT_AREA
	            ELSE QTY * UNIT_AREA
				 END
		  )	AS TOTAL_AREA --面积
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
      ,STAT_MONTH --年月
  FROM DWROTD.DWR_VM_INPUT_OUTPUT_T
 WHERE STAT_MONTH = '${PERIOD_ID_M}'
   AND COM_CODE = '2100'
   AND FACTORY_WERKS_CODE = '2100'
 GROUP BY
       MATERIAL_CODE --物料编号
      ,MATERIAL_NAME --物料描述
      ,CASE WHEN MATERIAL_NAME = '通用编码' THEN '深加工'
	        WHEN COST_CENTER = '2100MC0001' THEN '深加工'
            WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE NOT LIKE '%_A'
			 AND MATERIAL_CODE NOT LIKE '%_B' THEN '半成品'
			WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE LIKE '%_A' 
			 AND COST_CENTER = '2100MA0002' THEN '优等品'
			WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
			 AND MATERIAL_CODE LIKE '%_B'
			 AND COST_CENTER = '2100MA0002' THEN '合格品'
			 END --产品分类
	  ,CASE WHEN SUBSTR(MATERIAL_CODE,1,1) = 'E'
	         AND (MATERIAL_CODE LIKE '%_A' OR MATERIAL_CODE LIKE '%_B')
	        THEN REGEXP_REPLACE(MATERIAL_NAME,'[A-Za-z0-9]','')
			 END
	  ,STAT_MONTH
;
