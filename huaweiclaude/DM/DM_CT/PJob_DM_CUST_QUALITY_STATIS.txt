set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置成非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
WITH  TMP AS (
select 
T1.appeal_num,
T1.RESP_CONV, --责任方
T1.COST_AMB * T1.COMP_QTY as compensation_amt,
T1.batch AS batch_act , --批次
T1.cost_amb AS cost_amb
from  
DWRCMP.DWR_CMP_CUST_COMPENSATE_INFO_F t1
union all  
select 
T1.appeal_num,
T1.RESP_CONV, --责任方
T1.compensation_amt,
t1.compensation_batch  AS  batch_act,--批次
'' AS cost_amb
from  
DWRCMP.DWR_CMP_CUST_COMPENSATE_AMT_F  t1
)
, aps 	 as (
	select distinct batch_start,product_factory_code from  (			select
			batch_start,
			product_factory_code
		from
			DWRDIM.DWR_DIM_PRODUCT_AREA_D
		where
			factory_type = '自制'
			and product_factory_code is not null
		union all
		select distinct
			batch_start,
			'外协'
		from
			DWRDIM.DWR_DIM_PRODUCT_AREA_D
		where
			factory_type = '外协') aps
	)

,	XSSR AS (
		SELECT
		T1.calmonth AS calmonth,
			SUM(zxsmj) as zxsmj,
CASE WHEN t2.product_factory_code  = '1000' THEN '总部' 
                               WHEN t2.product_factory_code  = '1200' THEN '淄博' 
                               WHEN t2.product_factory_code  = '1300' THEN '清远' 
                               WHEN t2.product_factory_code  = '1500' THEN '丰城' 
                               WHEN t2.product_factory_code  = '1600' THEN '澧县' 
                               WHEN t2.product_factory_code  = '1800' THEN '重庆' 
                               WHEN t2.product_factory_code  = '1900' THEN '山西' 
                               ELSE '外协'  END as product_factory_code
		FROM
			DM.DM_AMBV2_PROFIT_T T1 
			LEFT JOIN  aps  T2 
			ON T1.zchanqbd = T2.batch_start
			LEFT JOIN  dm.dm_rpt_sales_group_t T4 
			ON T1.sales_grp = T4.node_name9
			 left join DM.DM_RPT_PROD_LEVEL_T T5 
			 on T1.ZPRODH3 = T5.prod_code3 and T5.PROD_CODE3 IS NOT NULL
		WHERE
-- 			substr(period_id_m, 1, 4) IN ('2023', '2024')
             t4.node_desc2 = '瓷砖事业部'
             AND 
			T1.calmonth = '${period_id_m}'
			AND coalesce (T1.cust_group,'#') <> 'Z0'
			AND coalesce (T1.doc_type,'#') <> 'ZT50'
		    AND T5.lve2_code
		    IN ('715岩板','918岩板','ART+','DPI','世界印象','仿古砖','其他瓷砖','原石','大板','抛釉砖','景观砖','玻化砖','瓷片','精瓷中板','超大板')
		GROUP BY
		CASE WHEN t2.product_factory_code  = '1000' THEN '总部' 
                               WHEN t2.product_factory_code  = '1200' THEN '淄博' 
                               WHEN t2.product_factory_code  = '1300' THEN '清远' 
                               WHEN t2.product_factory_code  = '1500' THEN '丰城' 
                               WHEN t2.product_factory_code  = '1600' THEN '澧县' 
                               WHEN t2.product_factory_code  = '1800' THEN '重庆' 
                               WHEN t2.product_factory_code  = '1900' THEN '山西' 
                               ELSE '外协'  end, 
			T1.calmonth

	)
--计算各产地的销值	
,XSSR_XZ AS (
		SELECT
			zchanqbd,
			material,
			sum(
				dept_inner_cost + zcbtzz + cross_cost + standard_price_a + cost_1n
			) as standard_price_a_s,
			sum(zxssl) AS SALE_QTY_s,  --年累计销售数量
			'${period_id_m}' as calmonth,
			sum(case when A.period_id_m = '${period_id_m}' then zxssl else 0 end ) as zxssl  --当月销售数量
		from
			DM.DM_AMBV2_PROFIT_T a
			left join dm.dm_rpt_sales_group_t d on a.sales_grp = d.node_name9
		where
			A.cust_group <> 'Z0'
			AND A.doc_type <> 'ZT50'
			and d.node_name2 = 'H11000001'
			and A.period_id_m >= concat(substr('${period_id_m}', 1, 4), '-01')
			AND A.period_id_m <= '${period_id_m}'
		group by
			zchanqbd,
			material
	)
,XSSR_XZ2 as (
SELECT
	T1.material,
	T1.zchanqbd,
	T2.product_factory_code,
	T3.matnr,
	case
		when t3.matnr is not null then t3.price2 else
		case when t1.SALE_QTY_S = 0 then 0 else 
		t1.standard_price_a_s * 1.13 / t1.SALE_QTY_S / 0.85
		end 
	end as COST_AMB, --阿米巴成本
	T1.zxssl,
(T1.zxssl * case
		when t3.matnr is not null then t3.price2 else
		case when t1.SALE_QTY_S = 0 then 0 else 
		t1.standard_price_a_s * 1.13 / t1.SALE_QTY_S / 0.85
		end 
	end ) as xz
FROM
	XSSR_XZ T1
	left join 
	(select matnr, datef,dated,min(price2) as price2 from SDI.SDI_ZTAMB_COST_1001
	where replace (concat('${period_id_m}','-01'),'-','') between datef AND dated 
	group by  matnr, datef,dated ) t3 
	on t1.material = t3.matnr
	LEFT JOIN APS T2 ON T1.zchanqbd = T2.batch_start
)
,XSSR_XZ3 as 
(
	select
	'${period_id_m}' as calmonth,
	CASE WHEN product_factory_code  = '1000' THEN '总部' 
                               WHEN product_factory_code  = '1200' THEN '淄博' 
                               WHEN product_factory_code  = '1300' THEN '清远' 
                               WHEN product_factory_code  = '1500' THEN '丰城' 
                               WHEN product_factory_code  = '1600' THEN '澧县' 
                               WHEN product_factory_code  = '1800' THEN '重庆' 
                               WHEN product_factory_code  = '1900' THEN '山西' 
                               WHEN product_factory_code  = '外协' THEN '外协' 
                               ELSE product_factory_code  end as product_factory_code,
	sum(xz) as xz 
	from  
	XSSR_XZ2 
group by 
CASE WHEN product_factory_code  = '1000' THEN '总部' 
                               WHEN product_factory_code  = '1200' THEN '淄博' 
                               WHEN product_factory_code  = '1300' THEN '清远' 
                               WHEN product_factory_code  = '1500' THEN '丰城' 
                               WHEN product_factory_code  = '1600' THEN '澧县' 
                               WHEN product_factory_code  = '1800' THEN '重庆' 
                               WHEN product_factory_code  = '1900' THEN '山西' 
                               WHEN product_factory_code  = '外协' THEN '外协' 
                               ELSE product_factory_code  end 
                              
)
,DM_CUST_QUALITY_STATIS_01 AS (
select 
SUBSTRING(t1.approval_date,1,7) AS approval_month,
t1.appeal_num,
t1.cc_class,
t2.RESP_CONV,-- 责任方
t1.cust_comp_qty,
t1.resp_conv_qty,
case when   t1.resp_conv_qty = 0 or t1.resp_conv_qty is null then 0 else t1.cust_comp_qty/t1.resp_conv_qty end as resp_rate ,
t2.compensation_amt,--,
-- coalesce(T4.product_factory_code,T3.product_factory_code) as product_factory_code --工厂代码
t2.cost_amb
from 
DWRCMP.DWR_CMP_CUST_COMPLAINT_INFO_F  t1
left join  (		select
			ROW_NUMBER() OVER (
				PARTITION BY
					appeal_num
				ORDER BY
					compensation_amt desc
			) AS rn,
appeal_num,
RESP_CONV,
compensation_amt,
 batch_act ,
 cost_amb
		from
			TMP)     t2 
on t1.appeal_num = t2.appeal_num
and t2.rn = 1 
LEFT JOIN  aps  T3 
ON SUBSTR(t2.batch_act,1,2) = T3.batch_start
LEFT JOIN aps  T4 
ON SUBSTR(t2.batch_act,1,3) = T4.batch_start
WHERE SUBSTRING(approval_date,1,7) =  '${period_id_m}'  
)
INSERT
	OVERWRITE TABLE dm.dm_cust_quality_statis  PARTITION (period_id_m)
SELECT 
t1.approval_month,
t1.cc_class,
coalesce(t1.RESP_CONV,'#') as RESP_CONV,
sum(t1.resp_rate) as cust_comp_qty_H,
count(distinct t1.appeal_num) as cust_comp_qty_F,
max(t2.zxsmj) as zxsmj,
sum(t1.compensation_amt) as comp_amount,
max(xz) as zxz,
current_timestamp AS dw_last_update_date,
T1.approval_month as period_id_m
from   
DM_CUST_QUALITY_STATIS_01  t1 
left JOIN 
XSSR  t2 
on  
coalesce(t1.RESP_CONV,'#') = T2.product_factory_code
left join  
XSSR_XZ3 T3
ON  
coalesce(t1.RESP_CONV,'#') = T3.product_factory_code
group by 
t1.approval_month,
t1.cc_class,
coalesce(t1.RESP_CONV,'#')
