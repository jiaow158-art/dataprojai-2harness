--新品销售收入占比ZB79
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB79';
with xpxs as (
select
calmonth,
SUM(case when mt.product_listed_date between  
SUBSTR(concat(CONCAT (SUBSTRING (calmonth ,1,7)) ,'-01')- INTERVAL '1 YEAR' + INTERVAL '1 MONTH', 1, 7) 
and  calmonth then t1.ambperformance end) as XSZB_FZ,
SUM(ambperformance) as XSZB_FM
from dm.dm_ambv2_zzt_xssr_t t1 
left join 
dwimd.dwi_md_data_material_general_t mt
on t1.material  = mt.material_num 
left join dm.dm_rpt_sales_group_t  t2 
on t1.sales_grp = t2.node_name9
where t2.node_name2 =  'H11000001'
and t1.data_source in ('S','T','D')
group by 
calmonth
)
,xpxs2 as (

select 
'${PERIOD_ID_M}' as calmonth,
'ZB79' as zb_code,
case when XSZB_FM = 0 then 0 else XSZB_FZ/XSZB_FM end as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
xpxs
where  calmonth = '${PERIOD_ID_M}' 

union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB79' as zb_code,
0 as 当月实绩,
SUM(XSZB_FZ)/SUM(XSZB_FM)  as 累计实绩,
0 as 季度累计实绩
from 
xpxs
where 
calmonth between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'

union all 

select 
'${PERIOD_ID_M}' as calmonth,
'ZB79' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
SUM(XSZB_FZ)/SUM(XSZB_FM) as 季度累计实绩
from 
xpxs
where  
 to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)

,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩) * 100 as CM_STAT_AMT,
sum(累计实绩) * 100 as ACC_STAT_AMT,
sum(季度累计实绩)* 100 as QTR_TGT_AMT
from 
xpxs2
group by 
calmonth,
zb_code
)

insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.PJFX
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
