set tez.grouping.max-size=134217728;  -- 128MB（单位字节）

INSERT OVERWRITE TABLE DM.DM_TRA_DELV_DTL_T
SELECT A.SALES_NUM AS SAP_ORDER_NUM --订单号
      ,A.DELV_ITEM_NUM --提货单行号
      ,A.PLAN_ITEM_NUM --计划项目号
	  ,A.PLAN_NUM --提货单组号
	  ,A.DELV_NUM --提货单号
	  ,COALESCE(B.DELV_QTY,0) AS DELV_QTY --提货单数量
	  ,B.BATCH_NUM --批次
	  ,B.WH_CODE --库存地点
	  ,COALESCE(C.OUT_STOCK_QTY,0) AS OUT_QTY --出库数量
	  ,B.PICKUP_DATE AS DELV_APLY_DATE --提货申请时间
	  ,DATEDIFF(CURRENT_DATE,D.CREATE_DATE) - 1 AS PREP_DAY --备货天数
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.CREATE_DATE) - UNIX_TIMESTAMP(B.PICKUP_DATE)) / 86400,0) || '天'
      || TRUNC((MOD((UNIX_TIMESTAMP(A.CREATE_DATE) - UNIX_TIMESTAMP(B.PICKUP_DATE)),86400)) / 3600,0) || '小时','.0','') AS DELV_APPR_CNSUM_TIME --提单审批耗时 创建提货单时间-提货单申请时间
      ,REPLACE(TRUNC((UNIX_TIMESTAMP(C.VOUCHER_POST_DATE) - UNIX_TIMESTAMP(A.CREATE_DATE)) / 86400,0) || '天'
      || TRUNC((MOD((UNIX_TIMESTAMP(C.VOUCHER_POST_DATE) - UNIX_TIMESTAMP(A.CREATE_DATE)),86400)) / 3600,0) || '小时','.0','') AS PREP_CNSUM_TIME --备货耗时 出库时间-创建提货单时间
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.SALES_ITEM_NUM AS SALE_LINE_NUM --SAP订单行号
  FROM DWROTD.DWR_OTD_LOADING_PLAN_T A
  LEFT JOIN(SELECT LOADING_PLAN_NUMBER
                  ,LOADING_PLAN_NUMBER_ITEM
				  ,BATCH_NUMBER AS BATCH_NUM
				  ,STOCK_LOCATION_CODE AS WH_CODE
				  ,PICKUP_DATE
                  ,SUM(PICKUP_QUANTITY) AS DELV_QTY
              FROM DWROTD.DWR_OTD_SO_PICKUP_ORDERS_F
             WHERE APPROVAL_STATUS = '审批成功'
			 GROUP BY LOADING_PLAN_NUMBER
			         ,LOADING_PLAN_NUMBER_ITEM
					 ,BATCH_NUMBER
					 ,STOCK_LOCATION_CODE
					 ,PICKUP_DATE
           ) B
	ON A.PLAN_NUM = B.LOADING_PLAN_NUMBER
   AND A.PLAN_ITEM_NUM = B.LOADING_PLAN_NUMBER_ITEM
  LEFT JOIN(SELECT DELIVERY_IM AS DELV_NUM --交货单号
                  ,DELIVERY_PROJECT AS DELV_LINE_NUM --交货行号
				  ,MAX(VOUCHER_POST_DATE) AS VOUCHER_POST_DATE
                  ,SUM(CASE WHEN DR_CR_ID = 'H_SHKZG' THEN QUANTITY ELSE 0 END) AS OUT_STOCK_QTY --出库数量
              FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F
			 --WHERE SPECIAL_STOCK_FLAG_2 = 'Q'
               --AND MOVE_TYPE <> '313'
			 GROUP BY DELIVERY_IM
			         ,DELIVERY_PROJECT
           ) C
    ON A.DELV_NUM = C.DELV_NUM
   AND A.DELV_ITEM_NUM = C.DELV_LINE_NUM
  LEFT JOIN DWIOTD.DWI_WM_LADINGBILL_HEADER_T D
    ON A.DELV_NUM = D.DELIVERY
