with batch_area  as(
	select batch_start,
		   max(case when batch_start='0Y' then '华盛昌基地'
		            when factory_type <>'外协' or factory_type is null then belong_area_name else 'OEM' end) as belong_area_name
	from DWRDIM.DWR_DIM_PRODUCT_AREA_D 
	group by batch_start  
)
, mat as (
-- 产区
	select t1.material_num,t1.product_level2_name  as cat,t1.dimension,
        t1.def_prod_channel  as prod_channel ,t1.product_level2_code as product_level2,
        t1.width * t1.length_1/1000/1000 as area,
        t1.reserved_field_6 as PRO_LEV
  from dwimd.dwi_md_data_material_general_t t1 
  where matl_group_code ='1000010'
	and external_matl_group_code in('10010','10080')
    and t1.product_level2_code  like '1%'
    and t1.product_level2_code  not in('1011','1012','1013','1014','1016')
	and matl_type_code='FERT'
)
-- 出入库
,tmp as (
select  replace(substr(t1.VOUCHER_POST_DATE,1,7),'-','') as stat_month,
		t1.material_num,  -- 物料
		t1.batch_num ,  --批次
		coalesce(t3.belong_area_name,t4.belong_area_name,'广东产区') as belong_area_name, --产区
		t2.product_level2,
		t2.cat,--品类
		t2.dimension,--规格
		t2.prod_channel,--渠道
		t2.PRO_LEV,--产品等级
		sum(case when MOVE_TYPE IN(
						 '601','602','643','644','653','654') then (case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end) 
						 end*t2.area) as sales_stock_out_area, --出库面积
		sum(case when MOVE_TYPE IN(
						 '911','912','913','914','131','132','531','532' ) 
						 then (case when DR_CR_ID='S_SHKZG' then QUANTITY else  -QUANTITY end*t2.area) end) as stock_in_area --入库面积

from  dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join mat t2 on t1.material_num  = t2.material_num 
left join batch_area t3  on substr(t1.batch_num,1,3) = t3.batch_start
left join batch_area t4  on substr(t1.batch_num,1,2) = t4.batch_start

where VOUCHER_POST_DATE >= '2022-07-31'  and ( t1.batch_num not LIKE '%888888' and t1.batch_num not LIKE '%999999')--虚拟批次
and MOVE_TYPE IN(
					 '601','602','643','644','653','654',  -- 销售出库
					 '911','912','913','914','131','132','531','532','101','102'   -- 生产入库&采购入库
					 )
    group by replace(substr(t1.VOUCHER_POST_DATE,1,7),'-','') 
    ,t1.material_num
    ,coalesce(t3.belong_area_name,t4.belong_area_name,'广东产区')
    ,t1.batch_num
    ,t2.product_level2
    ,t2.cat
    ,t2.dimension
    ,t2.prod_channel
    ,t2.PRO_LEV
 )
INSERT OVERWRITE TABLE DM.DM_PAM_BASE_CAPACITY_T
select  stat_month,
		material_num,  -- 物料
		batch_num ,  --批次
	    belong_area_name, --产区
		product_level2,
		cat,--品类
		dimension,--规格
		prod_channel,--渠道
		PRO_LEV,--产品等级
		sum(sales_stock_out_area) as sales_stock_out_area, --出库面积
		sum(stock_in_area) as stock_in_area, --入库面积
	    sum(case when PRO_LEV in ('A','B')  then stock_in_area else 0 end) as SALES_PRO_LEV_AREA, 
		sum(case when PRO_LEV in ('A')  then stock_in_area else 0 end) as SALES_PRO_LEV_A_AREA, 
		CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 最后更新时间 	
		from tmp
group by 
stat_month,
		material_num,  -- 物料
		batch_num ,  --批次
	    belong_area_name, --产区
		product_level2,
		cat,--品类
		dimension,--规格
		prod_channel,--渠道
		PRO_LEV--产品等级
