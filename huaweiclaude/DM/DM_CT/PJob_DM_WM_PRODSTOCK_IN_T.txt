--OEM入库量
--获取采购计划单的要求交期，回复交期，下单数量
/*
业务对象：
采购订单-订单类型：ZnB1，nb；创建时间；数量（面积）
会计凭证-物料凭证：移动类型101K、102K；
生产工厂8A*、77* 、75*
物料主数据-产品渠道、产品层次2、大小/量纲、外部物料组：10010，物料组1000010
供应商-供应商编码、省份城市（对应区分产区），BP查看
物料凭证-：移动类型101K、102K；
产区：APS产区关系表
生产工厂8A*、77* 、75*
*/
WITH T AS(
SELECT OEM.PURCHASE_ORDER_NO
      ,OEM.PURCHASE_ORDER_PROJ_NO
	  ,O.REQUIRE_DELIVERY
      ,OEM.REPLY_DELIVERY
	  ,SUM(COALESCE(D.PLAN_AMOUNT,0)) AS PLAN_AMOUNT
	  ,SUM(COALESCE(O.CANCEL_QTY,0)) AS CANCEL_QTY
	  ,SUM(COALESCE(D.PLAN_AMOUNT,0) + COALESCE(O.CANCEL_QTY,0)) AS ORIGINAL_PLAN_AMOUNT
  FROM DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F D
 INNER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F P 
    ON P.ID = D.PROCESS_TECH_TASK_ID 
   AND P.IS_PHANTOM = '0'
 INNER JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F O 
    ON O.ID = P.PRODUCT_ORDER_ID 
   AND P.INV_ORG_ID = O.INV_ORG_ID   
 INNER JOIN SDI.SDI_APS_OEM_PO_SUM_1007 OEM 
    ON OEM.APS_PLAN_TASK_DTL_ID = D.ID
 GROUP BY
       OEM.PURCHASE_ORDER_NO
      ,OEM.PURCHASE_ORDER_PROJ_NO
	  ,O.REQUIRE_DELIVERY
      ,OEM.REPLY_DELIVERY	
)
INSERT OVERWRITE TABLE dm.DM_WM_PRODSTOCK_IN_T
SELECT
       A.DELIVERY_IM
	  ,A.MATERIAL_ID --物料ID
	  ,A.MATERIAL_NUM --物料编码
	  ,B.MATERIAL_NAME --物料名称
	  ,A.MATERIAL_VCH_NO_MBLNR
	  ,A.BATCH_NUM --批次
	  ,A.MOVE_TYPE --移动类型
	  ,A.VOUCHER_POST_DATE --过账日期
	  ,A.SPECIAL_STOCK_FLAG_2 --特殊库存标识
	  ,A.VOUCHER_DATE --凭证日期
	  ,A.PURCHASE_VOUCHER_NUM --采购订单号
	  ,A.PURCHASE_VOUCHER_ITEM_NUM --采购凭证的项目编号
	  ,A.FACTORY_WERKS_ID --工厂
	  ,A.FACTORY_WERKS_CODE --工厂
	  ,A.FACTORY_WERKS_NAME --工厂
	  ,B.EXTERNAL_MATL_GROUP_CODE --外部物料组
	  ,B.MATL_GROUP_CODE --物料组
	  ,B.SIZE_UNIT --规格单位
	  ,B.LENGTH_1 --长度
	  ,B.WIDTH --宽度
	  ,B.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道编码
	  ,B2.CONFIG_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称
	  ,B1.VTEXT AS PRODUCT_STAGE2 --产品层次2
	  ,A.QUANTITY --数量
	  ,A.INV_LOCATION --库存地点
	  ,A.SUPPLIER_OR_CREDITOR_ID --供应商
	  ,A.SUPPLIER_OR_CREDITOR_ACCT --供应商
	  ,A.SUPPLIER_OR_CREDITOR_NAME --供应商
	  ,CASE WHEN A.DR_CR_ID = 'S_SHKZG' THEN A.QUANTITY
            WHEN A.DR_CR_ID = 'H_SHKZG' THEN -1 * A.QUANTITY
            ELSE 0
             END AS INTOWAREHOUSE_QUANTITY --OEM入库量
      ,CASE WHEN A.DR_CR_ID = 'S_SHKZG' THEN A.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000 
            WHEN A.DR_CR_ID = 'H_SHKZG' THEN -1 * A.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
            ELSE 0
             END AS INTOWAREHOUSE_AREA --采购入库面积
      ,T.REQUIRE_DELIVERY --要求交期
      ,T.REPLY_DELIVERY --回复交期
	  ,T.PLAN_AMOUNT --下单数量	
	  ,E.PURCHASE_ORDER_QTY --采购数量
	  ,COALESCE(T.PLAN_AMOUNT,E.PURCHASE_ORDER_QTY) AS PURCHASE_ORDER_QUANTITY --采购数量
	  ,CASE WHEN COALESCE(COALESCE(T.PLAN_AMOUNT,E.PURCHASE_ORDER_QTY),0) = 0 THEN 0
	        ELSE COALESCE(T.PLAN_AMOUNT,E.PURCHASE_ORDER_QTY) * B.LENGTH_1 * B.WIDTH /1000000
             END AS PURCHASE_ORDER_AREA --采购面积（平方米）
	  ,C.PURCHASE_VOUCHER_TYPE --采购凭证类型
	  ,C.PURCHASE_ORG --采购组织
	  ,C.PURCHASE_GROUP --采购组
      ,D.PRODUCT_AREA_ID --产区
      ,D.PRODUCT_AREA_CODE --产区
      ,D.PRODUCT_AREA_NAME --产区	  
      ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE	
      ,A.LC_AMT AS INTOWAREHOUSE_AMOUNT --入库金额
      ,A.CURRENCY_CODE --货币单位
	  ,F.LATEST_VOUCHER_POST_DATE --批次最晚入库日期
	  ,F.EARLIEST_VOUCHER_POST_DATE --批次最早入库日期
	  ,B.DIMENSION --规格
	  ,A.CUST_ID --客户
	  ,A.CUST_NUM --客户
	  ,A.CUST_NAME --客户	 
      ,COALESCE(T.CANCEL_QTY,0) AS CANCEL_QTY --采购订单取消量
      ,COALESCE(T.CANCEL_QTY,0) * B.LENGTH_1 * B.WIDTH /1000000 AS CANCEL_AREA --采购订单取消面积（平方米）
      ,COALESCE(T.ORIGINAL_PLAN_AMOUNT,0) AS ORIGINAL_PLAN_AMOUNT --原采购数量
      ,COALESCE(T.ORIGINAL_PLAN_AMOUNT,0) * B.LENGTH_1 * B.WIDTH /1000000 AS ORIGINAL_PURCHASING_AREA --原采购面积（平方米）	  
      ,CASE WHEN COALESCE(BD.BRAND_NAME,'') <> '' THEN BD.BRAND_NAME
	        WHEN COALESCE(BD1.BRAND_NAME,'') <> '' THEN BD1.BRAND_NAME
	        WHEN COALESCE(BD2.BRAND_NAME,'') <> '' THEN BD2.BRAND_NAME
			ELSE 'NA'
			 END AS BRAND_NAME --品牌名称
	  ,A.wbs_element
	  ,P.PRODUCT_PROPERTY
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D B
    ON A.MATERIAL_ID = B.MATERIAL_ID
   AND B.end_date = '4712-12-31'
  LEFT JOIN SDI.SDI_T179T_1001 B1 --产品品类
    ON B1.MANDT = B.GROUP_NUM
   AND B1.SPRAS = '1'
   AND B1.DEL_FLAG = 'N'
   AND SUBSTR(B.PRODUCT_LEVEL_CODE,1,4) = B1.PRODH
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R B2
    ON B.RESERVED_FIELD_1 = B2.SRC_SYS_CONFIG_CODE
   AND B2.SOURCE_CODE = 'ZTYL01'
  LEFT JOIN T
    ON T.PURCHASE_ORDER_NO = A.PURCHASE_VOUCHER_NUM
   AND T.PURCHASE_ORDER_PROJ_NO = A.PURCHASE_VOUCHER_ITEM_NUM
  LEFT JOIN DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F C
    ON A.PURCHASE_VOUCHER_NUM = C.PURCHASE_VOUCHER_NUM
  LEFT JOIN (SELECT A.SUPPLIER_OR_CREDITOR_ID
                 ,A.PRODUCT_AREA_ID
	             ,A.PRODUCT_AREA_CODE
                 ,A.PRODUCT_AREA_NAME
                 ,row_number() over(partition by A.SUPPLIER_OR_CREDITOR_ID order by a.update_date DESC) AS RN
             FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D A
             WHERE IS_PHANTOM=0
		  ) D
    ON A.SUPPLIER_OR_CREDITOR_ID = D.SUPPLIER_OR_CREDITOR_ID AND D.RN=1
  LEFT JOIN DWRPUR.DWR_PTP_PURCHASE_VCH_ITEM_F E
    ON A.PURCHASE_VOUCHER_NUM = E.PURCHASE_VOUCHER_NUM
   AND A.PURCHASE_VOUCHER_ITEM_NUM = E.PURCHASE_VCH_ITEM_NUM1
  LEFT JOIN(SELECT 
                   MATERIAL_NUM,BATCH_NUM
				  ,MAX(VOUCHER_POST_DATE) AS LATEST_VOUCHER_POST_DATE --批次最晚入库日期
				  ,MIN(VOUCHER_POST_DATE) AS EARLIEST_VOUCHER_POST_DATE --批次最早入库日期
			  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F
			 WHERE MOVE_TYPE in('101','102') 
			 GROUP BY MATERIAL_NUM,
			       BATCH_NUM
		   ) F
    ON A.MATERIAL_NUM = F.MATERIAL_NUM AND A.BATCH_NUM = F.BATCH_NUM
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD
    ON A.FACTORY_WERKS_CODE = BD.FACTORY_WERKS_CODE
   AND B.EXTERNAL_MATL_GROUP_CODE = BD.EXTERNAL_MATL_GROUP_CODE
   AND B.MATL_GROUP_CODE = BD.MATL_GROUP_CODE
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD1
    ON A.FACTORY_WERKS_CODE = BD1.FACTORY_WERKS_CODE
   AND BD1.EXTERNAL_MATL_GROUP_CODE = ''
   AND B.MATL_GROUP_CODE = BD1.MATL_GROUP_CODE 
  LEFT JOIN (SELECT DISTINCT FACTORY_WERKS_CODE
                            ,BRAND_NAME
               FROM DWRDIM.DWR_DIM_MATERIAL_BRAND_D
			  WHERE MATL_GROUP_CODE = ''
			) BD2
    ON A.FACTORY_WERKS_CODE = BD2.FACTORY_WERKS_CODE
  LEFT JOIN(SELECT DISTINCT code, PRODUCT_PROPERTY from SDI.SDI_ITEM_1007 WHERE IS_PHANTOM = '0'  AND DEL_FLAG = 'N') P
    ON A.MATERIAL_NUM = P.CODE
 WHERE A.MOVE_TYPE in('101','102')
   AND COALESCE(A.SPECIAL_STOCK_FLAG_2,'') = 'K_T148T'
   --AND SUBSTR(A.FACTORY_WERKS_CODE,1,2) IN('8A','77','75')
;
