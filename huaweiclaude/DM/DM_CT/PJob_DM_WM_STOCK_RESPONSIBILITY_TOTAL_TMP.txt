--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with wbs as(
	select a.pspnr,
		   b.zxszz, 
		   b.zunnr , 
		   c.vkgrp  
	from sdi.sdi_prps_1001 a 
	join sdi.sdi_proj_1001 b 
	ON substr(a.posid,1,10) = b.pspid 
	LEFT JOIN sdi.sdi_knvv_1001 c 
	ON b.zunnr = c.kunnr 
	AND c.vkorg = b.zxszz 
	AND c.vtweg = '06' 
	AND c.DEL_FLAG ='N'
)
insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_TOTAL_TMP PARTITION(STAT_MONTH)
select t1.material_num, 
	   t1.batch_num, 
	   t1.factory_werks_code ,
	   t1.inv_location,
       t4.lev1_name,
	   t4.lev2_name,
	   t4.lev3_name,
	   t4.lev4_name,
	   t4.lev5_name,
	   t4.lev6_name,
       case when t4.lev2_name = '瓷砖事业部'  and t4.lev3_name='国内营销系统' and t4.lev4_name='区域运营中心' then '区域运营中心'
            when t4.lev2_name = '瓷砖事业部'  and t4.lev3_name='国内营销系统' and t4.lev4_name='直营经营管理板块' then '直营经营管理板块'
            when t4.lev2_name = '瓷砖事业部'  and t4.lev3_name='国内营销系统' and t4.lev4_name='工程市场中心' then '工程市场中心'
            when t4.lev2_name = '瓷砖事业部'  and t4.lev3_name='国际营销系统' then '国际营销系统'
            when t4.lev2_name = '瓷砖事业部'  and t4.lev3_name='梦之家事业部' then '梦之家'
            else '其他部门留备货' end as resp_system  -- 项目库存归属
		,sum(stock_qty) as stock_qty   -- 总库存
		,sum(case when COALESCE(special_stock_flag_2,'') in('','K_T148T') AND COALESCE(goods_move_stock_type_sid,'') = '01' then COALESCE(STOCK_QTY,0) else 0 end) as zkkc--非限制库存
		,sum(case when COALESCE(special_stock_flag_2,'') = 'Q_T148T'  then COALESCE(STOCK_QTY,0) else 0 end) as q_stock ---项目库存数()
	,sum(case when COALESCE(special_stock_flag_2,'') in('','K_T148T') AND COALESCE(goods_move_stock_type_sid,'') = '02' then COALESCE(STOCK_QTY,0) else 0 end) as zjkc--质检数
	,sum(case when COALESCE(special_stock_flag_2,'') in('','K_T148T') AND COALESCE(goods_move_stock_type_sid,'') = '07' then COALESCE(STOCK_QTY,0) else 0 end) as djkc--冻结数
	,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
	,substr('${PERIOD_ID_D}',1,7) AS STAT_MONTH
from dwrotd.dwr_wm_all_type_stock_month_t t1
join  dm.dm_wm_stock_responsibility_mat_tmp t2 on t1.material_num  = t2.material_num 
left join wbs t3 ON t1.wbs_element  = t3.pspnr 
left join dm.dm_rpt_sale_grp_t t4 ON t3.vkgrp  = t4.sale_grp
 where  month_date = replace(substr('${PERIOD_ID_D}',1,7),'-','')   -- 传昨天
group by  t1.material_num, 
		  t1.batch_num, 
		  t1.factory_werks_code ,
		  t1.inv_location ,
          t4.lev1_name,
		  t4.lev2_name,
		  t4.lev3_name,
		  t4.lev4_name,
		  t4.lev5_name,
		  t4.lev6_name;
