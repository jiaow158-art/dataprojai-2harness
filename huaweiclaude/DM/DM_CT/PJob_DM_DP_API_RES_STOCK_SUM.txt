--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
-- set hive.exec.dynamic.partition=true; 
-- set hive.exec.dynamic.partition.mode=nonstrict;

with tem as
(
--全部留货数据
select 
	substr(t1.user_delivery_date,1,4) as stat_year,
	from_unixtime(unix_timestamp(t1.user_delivery_date, 'yyyyMMdd'), 'yyyy-MM')  as stat_month,
	t2.sales_center_code,
	t2.sales_center,
	case when lb_type = '专属客编留货' then 'Y' else 'N' end as is_spec_cust,
	'N' as is_wbs,
	zh_channel_code4 as integrate_channel,
	t2.sales_dep_code,	
	t2.sales_dep,
	t2.sales_region_code,	
	t2.sales_region,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t4.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,	
	t1.dimension as dimension,
	t1.material_num,
	t1.material_name,	
	t1.sal_pre_res_area as order_area
from  DM.DM_RPT_SALES_DELIVER_2023_T t1
join dm.dm_dp_api_sales_group_struct t2 on t1.sale_grp = t2.sales_grp
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL   
where 1=1
and lb_type in ('销售订单留货','专属客编留货')
and stat_month = replace(substring(current_timestamp(),1,7),'-','')

union all

--剩余项目库存
SELECT 
	substr(expect_date,1,4) as stat_year,
	from_unixtime(unix_timestamp(t1.expect_date, 'yyyyMMdd'), 'yyyy-MM')  as stat_month,
	t2.sales_center_code,
	t2.sales_center,
	'N' is_spec_cust,
	'Y' as is_wbs,
	zh_channel_code4 as integrate_channel,
	t2.sales_dep_code,	
	t2.sales_dep,
	t2.sales_region_code,	
	t2.sales_region,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t4.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,	
	t4.DIMENSION as dimension,
	t1.material_num,
	t4.material_name,	
	t1.available_stock_qty*t4.length_1*t4.width/1000000 as order_area
from dwrotd.dwr_wm_wbs_stock_f t1
join dm.dm_dp_api_sales_group_struct t2 on t1.sale_grp = t2.sales_grp
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL   
where 1=1
and t1.start_month= replace(substring(current_timestamp(),1,7),'-','')
and t1.available_stock_qty > 0
)

INSERT OVERWRITE TABLE dm.dm_dp_api_res_stock_sum --partition (stat_month_ym)
select 
	stat_year,
	stat_month,
	sales_center_code,
	sales_center,
	is_spec_cust,
	is_wbs,
	integrate_channel,
	sales_dep_code,	
	sales_dep,
	sales_region_code,	
	sales_region,
	fin_cate,	
	dimension,
	material_num,
	material_name,	
	order_area,
	current_timestamp() as dw_last_update_date,
	stat_month as stat_month_ym
from tem
