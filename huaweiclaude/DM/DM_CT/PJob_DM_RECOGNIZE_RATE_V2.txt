WITH TMP AS (
SELECT 
t1.period_id_m,t1.type,
count(t2.invisible_code) as code_inventory,  --条码入库数
COUNT(PACK_code) - COUNT(CASE WHEN PACK_code = 'NULL' OR PACK_code = 'NOREAD' THEN 1 ELSE NULL END)  - COUNT(DISTINCT PACK_CODE) AS PACK_repeat_QTY --打包重复数
FROM SDI.SDI_PACK_RELATION3_1043  t1 
	left join (select 
	distinct  invisible_code,substring(create_time,1,10)  as cdate 
	from SDI.SDI_CODE_RELATION_1042 
	where period_id_m = substring ('${PERIOD_ID_D}',1,7) 
	and invisible_code is not null  and invisible_code <> '' 
	) t2
	on t1.period_id_m = t2.cdate and t1.pack_code = t2.invisible_code
WHERE period_id_m ='${PERIOD_ID_D}'
GROUP BY 
t1.period_id_m,t1.type
)
,PACK_RELATION3 as (
select PACK_code as PACK_code, PACK_DATE,type from SDI.SDI_PACK_RELATION3_1043
where PACK_DATE = '${PERIOD_ID_D}' AND PACK_CODE <> 'NULL' AND PACK_CODE <>  'NOREAD'
group by PACK_code, PACK_DATE,type
)
,INKJET_RELATION as (
select INKJET_code as INKJET_code,type  from  SDI.SDI_INKJET_RELATION_1043 
where INKJET_DATE >= DATE_ADD( '${PERIOD_ID_D}', -6)
AND INKJET_DATE < DATE_ADD( '${PERIOD_ID_D}', 1)
group by type,INKJET_code
)

,FRACTIONATION_RELATION as (
select FRACTIONATION_code  as FRACTIONATION_code ,type  from  SDI.SDI_FRACTIONATION_RELATION_1043 
where FRACTIONATION_DATE >= DATE_ADD( '${PERIOD_ID_D}', -6)
AND FRACTIONATION_DATE < DATE_ADD( '${PERIOD_ID_D}', 1)
group by FRACTIONATION_code ,type

    COUNT(DISTINCT T3.FRACTIONATION_code) AS FRACTIONATION_ASCEND_QTY, -- 分级可追溯数量
    COUNT(DISTINCT CASE WHEN T2.INKJET_code IS NOT NULL AND T3.FRACTIONATION_code IS NOT NULL THEN T1.PACK_code END) AS INKJET_FRACTIONATION_ASCEND_QTY -- 分级&喷墨追溯数
FROM
    PACK_RELATION3 T1
LEFT JOIN
    INKJET_RELATION T2 ON T1.PACK_code = T2.INKJET_code
    AND T2.type IN ('清远三线喷墨工序', '清远四线喷墨工序2', '清远四线喷墨工序3', '清远四线喷墨工序4')
LEFT JOIN
    FRACTIONATION_RELATION T3 ON T1.PACK_code = T3.FRACTIONATION_code
    AND T3.type IN ('清远三线分级工序', '清远四线分级工序')
WHERE
    T1.type = '清远三线打包工序'
GROUP BY
    T1.PACK_DATE
UNION ALL 
SELECT

    '清远四线打包工序' AS process,
    COUNT(DISTINCT T2.INKJET_code) AS INKJET_ASCEND_QTY, -- 喷墨可追溯数量
    COUNT(DISTINCT T3.FRACTIONATION_code) AS FRACTIONATION_ASCEND_QTY, -- 分级可追溯数量
    COUNT(DISTINCT CASE WHEN T2.INKJET_code IS NOT NULL AND T3.FRACTIONATION_code IS NOT NULL THEN T1.PACK_code END) AS INKJET_FRACTIONATION_ASCEND_QTY -- 分级&喷墨追溯数
FROM

INSERT OVERWRITE TABLE DM.DM_RECOGNIZE_RATE_V2 
SELECT
	t1.PACK_DATE,
	t1.process,
	t1.INKJET_ASCEND_QTY,
	t1.FRACTIONATION_ASCEND_QTY,
	t1.INKJET_FRACTIONATION_ASCEND_QTY,
	t2.PACK_repeat_QTY,
	'N' AS del_flag,
    CURRENT_TIMESTAMP AS dw_last_update_date,
    t2.code_inventory     --2025.02.19新增条码入库数
FROM
	DM_RECOGNIZE_RATE t1
	LEFT JOIN tmp t2 ON t1.PACK_DATE = t2.period_id_m
	AND t1.process = t2.type
where t1.PACK_DATE IS NOT NULL  and t1.PACK_DATE <> '' AND t1.PACK_DATE = '${PERIOD_ID_D}'
union all 
SELECT
pack_date,
process,
inkjet_ascend_qty,
fractionation_ascend_qty,
inkjet_fractionation_ascend_qty,
pack_repeat_qty,
del_flag,
dw_last_update_date,
code_inventory
FROM 
DM.DM_RECOGNIZE_RATE_V2 
WHERE pack_date <> '${PERIOD_ID_D}'
