--共享仓辐射中小经销商收入同比增长率ZB71
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}'  
and zb_code ='ZB71';
with tbzzl as (
 select 
 calmonth,
       sum(case
               when d.prod_property = 'A0001'
                   and (d.product_series_name in ('918', '超大板')
                       or 
                           d.product_brand_code in ('A1', 'A13', 'A14', 'A16', 'A18'))
                   and coalesce(integrate_channel2, '#') != 'GD04' then ambperformance
               else 0 end) as 高值业绩,
       sum(ambperformance) as 业绩
from dm.dm_ambv2_zzt_xssr_t a
         left join dm.dm_rpt_zmaster_cus_t z
                   on a.customer = z."ACCOUNT_NUMBER"
                       and a.distr_chan = z."SALES_CHANNEL_CODE"
                       and a.salesorg = z."SALES_ORGANIZATION_CODE"
                       and z."PARVW" = 'AG'
         left join DWRDIM.DWR_DIM_WBS_BASIS_INFO_F zz
                   on a.wbs_elemt = zz.WBS_LEVEL3
         left join dm.dm_md_sale_grp_reset_t c on case
                                                      when a.data_source in ('S', 'T')
                                                          then COALESCE(
                                                              COALESCE(zz.sales_grp, z."SALES_GROUP_CODE"),
                                                              a.sales_grp)
                                                      else a.sales_grp
                                                      end = c.node_name9
         left join dwimd.dwi_md_data_material_general_t d on a.material = d.material_num
         join upload.UPLOAD_CT_SHARE_CUSTOMER_REL_1037 e on a.customer = lpad(e.customer_code, 10, '0')
  
where c.node_name3 = 'H03230618'
  and integrate_channel = 'GD01'
  and e.is_small_and_medium = '是'
  and customer_end_date = '99991231'
  and substring(a.calmonth,1,4) >= '2024'
group by calmonth
)
,tbzzl2 as (
select 
calmonth ,
'ZB71' as zb_code,
(高值业绩)/业绩 as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
tbzzl
where calmonth = '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB71' as zb_code,
0 as 当月实绩,
SUM(高值业绩)/SUM(业绩) as 累计实绩,
0 as 季度累计实绩
from 
tbzzl
where CALMONTH between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB71' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
SUM(高值业绩)/SUM(业绩)  as 季度累计实绩
from 
tbzzl
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)*100 as CM_STAT_AMT,
sum(累计实绩)*100 as ACC_STAT_AMT,
sum(季度累计实绩)*100 as QTR_TGT_AMT
from 
tbzzl2
group by 
calmonth,
zb_code
)
  
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
