insert overwrite into DM.DM_SO_OMS_POLICY_T

WITH BELNR AS (
SELECT DISTINCT VOUCHER_CODE,DIVIDE FROM DWRFIN.DWR_AR_CUST_DETAIL_F  
WHERE DIVIDE IN (SELECT DISTINCT POLICY_ACTIVITY_CODE FROM DWROTD.DWR_SO_OMS_POLICY_T )
)
,TMP AS (
SELECT 
 T.POLICY_ACTIVITY_CODE 
,T.POLICY_CODE
,T.POLICY_NAME
,T.POLICY_START_DATE
,T.POLICY_END_DATE
,T.POLICY_POSITION ---报名档位
,T.DEPOSIT_RETURN
,T.CUST_CODE-- 客户编码
,T.CUST_NAME-- 客户名称
,T.SALES_ORG_CODE
,T.DIS_CHNL_CODE 
,T.SALES_GROUP_CODE
,T.NODE_NAME5
,T.NODE_DESC5
,T.PAR_CUST_NUM
,T.PAR_CUST_NAME
,T.POLICY_APPLY_DATE --报名时间
,T.CUST_LEV_NAME  --客户等级
,T.CUST_ATTR_NAME --客户属性
,T.CUST_ACHE_CLASS
,T.CUST_ACHE_CLASS_NAME  --客户分类名称 
,T.TASK_SUM_AMT -- 任务合计      
,T.DH_TARGET_AMT--订货目标金额
,T.DEPOSIT_AMT --定金金额
,SUM(T.ORD_AMT) AS ORD_AMT --订单金额
,SUM(T.FAC_YJ_AMT) AS FAC_YJ_AMT --出库金额（对厂）
,SUM(T.WARE_YJ_AMT) AS WARE_YJ_AMT--出库金额（对仓）
,SUM(T.FAC_YJ_AMT + T.WARE_YJ_AMT) AS PAR_CUST_AMT --母客编业绩
,SUM(T.POLICY_CUST_AMT) AS POLICY_CUST_AMT   --报名客编实际业绩
,SUM(T.PER_COMP_RATE) AS PER_COMP_RATE --保证金返回门槛（业绩完成率）
FROM DWROTD.DWR_SO_OMS_POLICY_T T
GROUP BY 
 T.POLICY_ACTIVITY_CODE 
,T.POLICY_CODE
,T.POLICY_NAME
,T.POLICY_START_DATE
,T.POLICY_END_DATE
,T.POLICY_POSITION ---报名档位
,T.DEPOSIT_RETURN 
,T.CUST_CODE-- 客户编码
,T.CUST_NAME-- 客户名称
,T.SALES_ORG_CODE
,T.DIS_CHNL_CODE 
,T.SALES_GROUP_CODE
,T.NODE_NAME5
,T.NODE_DESC5
,T.PAR_CUST_NUM
,T.PAR_CUST_NAME
,T.POLICY_APPLY_DATE --报名时间
,T.CUST_LEV_NAME  --客户等级        
,T.CUST_ATTR_NAME --客户属性
,T.CUST_ACHE_CLASS
,T.CUST_ACHE_CLASS_NAME  --客户分类名称 
,T.TASK_SUM_AMT -- 任务合计      
,T.DH_TARGET_AMT--订货目标金额
,T.DEPOSIT_AMT --定金金额
)
,TMPP AS (
SELECT 
 T.POLICY_ACTIVITY_CODE 
,T.POLICY_CODE
,T.POLICY_NAME
,T.POLICY_START_DATE
,T.POLICY_END_DATE
,T.POLICY_POSITION ---报名档位
,T.DEPOSIT_RETURN
,T.CUST_CODE-- 客户编码
,T.CUST_NAME-- 客户名称
,T.SALES_ORG_CODE
,T.DIS_CHNL_CODE 
,T.SALES_GROUP_CODE
,T.NODE_NAME5
,T.NODE_DESC5
,T.PAR_CUST_NUM
,T.PAR_CUST_NAME
,T.POLICY_APPLY_DATE --报名时间
,T.CUST_LEV_NAME  --客户等级
,T.CUST_ATTR_NAME --客户属性
,T.CUST_ACHE_CLASS
,T.CUST_ACHE_CLASS_NAME  --客户分类名称 
,T.TASK_SUM_AMT -- 任务合计      
,T.DH_TARGET_AMT--订货目标金额
,T.DEPOSIT_AMT --定金金额
,T.ORD_AMT --订单金额
,T.FAC_YJ_AMT --出库金额（对厂）
,T.WARE_YJ_AMT--出库金额（对仓）
,T.POLICY_CUST_AMT  --报名客编实际业绩
,T.PAR_CUST_AMT --母客编业绩
,CASE WHEN T.DH_TARGET_AMT <> 0 THEN T.ORD_AMT / T.DH_TARGET_AMT ELSE 0 END AS ORD_FIN_RATE  --订单完成率
,CASE WHEN T.DH_TARGET_AMT <> 0 THEN T.PAR_CUST_AMT / T.DH_TARGET_AMT ELSE 0 END AS PERFORM_FIN_RATE  --业绩完成率
,T.PER_COMP_RATE --保证金返回门槛（业绩完成率）
,T1.VOUCHER_CODE as BELNR  --退订凭证
FROM TMP T
LEFT JOIN BELNR T1 
ON T.POLICY_ACTIVITY_CODE = T1.DIVIDE
)
SELECT 
 T.POLICY_ACTIVITY_CODE 
,T.POLICY_CODE
,T.POLICY_NAME
,T.POLICY_START_DATE
,T.POLICY_END_DATE
,T.POLICY_POSITION ---报名档位
,T.DEPOSIT_RETURN
,T.CUST_CODE-- 客户编码
,T.CUST_NAME-- 客户名称
,T.SALES_ORG_CODE
,T.DIS_CHNL_CODE 
,T.SALES_GROUP_CODE
,T.NODE_NAME5
,T.NODE_DESC5  --运营中心 
,T.PAR_CUST_NUM
,T.PAR_CUST_NAME
,T.POLICY_APPLY_DATE --报名时间
,T.CUST_LEV_NAME  --客户等级
,T.CUST_ATTR_NAME --客户属性
,T.CUST_ACHE_CLASS
,T.CUST_ACHE_CLASS_NAME  --客户分类名称 
,T.TASK_SUM_AMT -- 任务合计      
,T.DH_TARGET_AMT--订货目标金额
,T.DEPOSIT_AMT --定金金额
,T.ORD_AMT --订单金额
,T.FAC_YJ_AMT --出库金额（对厂）
,T.WARE_YJ_AMT--出库金额（对仓）
,T.POLICY_CUST_AMT  --报名客编实际业绩
,T.PAR_CUST_AMT --母客编业绩
,T.ORD_FIN_RATE  --订单完成率
,T.PERFORM_FIN_RATE  --业绩完成率
,CASE WHEN T.DEPOSIT_RETURN = 'PROPORTION_RETURN' THEN CASE WHEN T.PERFORM_FIN_RATE < T.PER_COMP_RATE THEN 0 ELSE T.PERFORM_FIN_RATE * T.DEPOSIT_AMT END 
      WHEN T.DEPOSIT_RETURN = 'NOT_RETURNELSE' THEN 0 
      WHEN T.DEPOSIT_RETURN = 'FINISH_RETURN' THEN CASE WHEN T.PERFORM_FIN_RATE < T.PER_COMP_RATE THEN 0 ELSE T.DEPOSIT_AMT END ELSE 0 END AS RE_ORD_AMT --退订金额
,T.DEPOSIT_AMT - CASE WHEN T.DEPOSIT_RETURN = 'PROPORTION_RETURN' THEN CASE WHEN T.PERFORM_FIN_RATE < T.PER_COMP_RATE THEN 0 ELSE T.PERFORM_FIN_RATE * T.DEPOSIT_AMT END 
                      WHEN T.DEPOSIT_RETURN = 'NOT_RETURNELSE' THEN 0 
                      WHEN T.DEPOSIT_RETURN = 'FINISH_RETURN' THEN CASE WHEN T.PERFORM_FIN_RATE < T.PER_COMP_RATE THEN 0 ELSE T.DEPOSIT_AMT END ELSE 0 END AS MS_ORD_AMT -- 没收金额
,T.BELNR 
,case when BELNR is null then '否' else '是' end as is_return -- 是否退订
,current_date() AS DW_LAST_UPDATE_DATE
,T.PER_COMP_RATE --保证金返回门槛（业绩完成率）
from TMPP T
