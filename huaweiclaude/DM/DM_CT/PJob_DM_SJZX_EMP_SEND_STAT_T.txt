WITH T1 AS(
SELECT LOG_TIME
      ,TEMPL_CODE
	  ,USER_NUM
	  ,EMP_NUM
	  ,TASK_NAME
	  ,TRML_TYPE_NAME
  FROM DWRBIT.DWR_SJZX_TASK_LOG_T
LATERAL VIEW EXPLODE(SPLIT(REPLACE(REPLACE(REPLACE(USER_NUM,'"',''),'[',''),']',''),',')) TABLE_VIEW AS EMP_NUM 
 WHERE SUBSTR(LOG_TIME,1,7) = '${PERIOD_ID_M}'
)
INSERT OVERWRITE TABLE DM.DM_SJZX_EMP_SEND_STAT_T
SELECT
       A.EMP_NUM --工号
	  ,B.EMP_NAME --员工姓名
	  ,A.TEMPL_CODE --模板编码
	  ,A.TASK_NAME AS RPT_NAME --调度任务名称
	  ,SUM(CASE WHEN C.TYPE IN ('0','1','2','3','4','5','6','7','8') THEN 1 ELSE 0 END) AS SEND_CNT --推送次数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.TRML_TYPE_NAME --终端类型
	  ,A.STAT_MONTH --统计月份
  FROM(SELECT DISTINCT
              SUBSTR(LOG_TIME,1,7) AS STAT_MONTH --统计月份
             ,EMP_NUM
			 ,TEMPL_CODE
			 ,TASK_NAME
			 ,TRML_TYPE_NAME
         FROM T1
		--WHERE SUBSTR(LOG_TIME,1,7) = '${PERIOD_ID_M}'
      ) A
 INNER JOIN DM.DM_WF_EMP_INFO_T B
    ON A.EMP_NUM = B.EMP_NUM
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_RECORD_EXECUTE_T C
    ON A.EMP_NUM = C.EMP_NUM
   AND A.TASK_NAME = C.TASK_NAME
   AND A.STAT_MONTH = SUBSTR(C.BEGIN_TIME,1,7)
 WHERE COALESCE(B.DEPT_NAME,'SNULL') <> '数字赋能平台数据及财经产品部'
   AND COALESCE(B.DEPT_NAME,'SNULL') <> '审计法务部审计内控组'
 GROUP BY A.STAT_MONTH --统计月份
      ,A.EMP_NUM --工号
	  ,B.EMP_NAME --员工姓名
	  ,A.TEMPL_CODE --模板编码
	  ,A.TASK_NAME --调度任务名称
	  ,A.TRML_TYPE_NAME
	  ;
