INSERT OVERWRITE TABLE DM.DM_LTC_BO_RPT_T
select t.BO_NO, -- 商机号
	   t.AREA_NAME, --  区域
	   t.BO_VALID_DATE,  -- 商机有效期
	   t.BO_STAGE, -- 商机阶段
	   t.PROJECT_NAME, -- 项目名称
	   t.APPLY_DATE, -- 申请日期
	   t.PROJECT_TYPE, -- 工程类别
	   t.deliver_tot_area, -- 发货总面积
	   t.customer_code, -- 客户编码
	   t.customer_name,  -- 客户名称
	   t.cust_group_name,  -- 客户组
	   t1.BO_YEAR_TARGET , -- 新增商机年度目标
	   t1.BO_YEAR_TARGET_FOR_MONTH , -- 新增商机年度目标所属月份
	   t1.BO_YEAR_TARGET_FOR_QUARTER , -- 新增商机年度目标所属季度
	   t1.BO_YEAR_TRAN_TARGET , -- 商机转化年度目标
	   t1.BO_YEAR_TRAN_TARGET_FOR_MONTH , -- 商机转化年度目标所属月份
	   t1.BO_YEAR_TRAN_TARGET_FOR_QUARTER , -- 商机转化年度目标所属季度
	   t.bo_type, -- 商机类型/工程大类
	   t.forecast_use_qty, --预计用量
	   t2.dqxsz as sales_grp, -- 当前线组
	   t2.dqxszms as sales_grp_desc, -- 当前线组描述
	   t3.node_name5 as sales_dept_name, -- 当前销售部门 层级5
	   t3.node_desc5 as sales_dept_desc, -- 当前销售部门描述 层级5描述
	   t3.node_name4 as sales_area_name, -- 当前销售区域 层级4
	   t3.node_desc4 as sales_area_desc, -- 当前销售区域描述 层级4描述
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE 
from DWRLTC.DWR_BO_PROJECT_CHARLEDGER_F t
left join SDI.SDI_CT_BO_YEAR_TARGET_1037 t1
on t.BO_NO = t1.bo_no
left join SDI.SDI_ZSD_SJXX_1001 t2
on t.BO_NO = t2.sjh 
left join DM.DM_RPT_SALES_GROUP_T t3
on t2.dqxsz = t3.node_name9;
