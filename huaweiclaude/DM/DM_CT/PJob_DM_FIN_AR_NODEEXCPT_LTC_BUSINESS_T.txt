WITH T1 AS(
SELECT B.WBS_SPIT_ELEMENT AS WBS
      ,A.PARTNER AS CUST_NUM
	  ,SUM(A.CREDIT_LIMIT) AS CREDIT_AMT
  FROM SDI.SDI_ZTSDXD01_1001 A
  LEFT JOIN (SELECT DISTINCT WBS_SPIT_ELEMENT,WBS_ELEMENT FROM DWROTD.DWR_SO_PROJECT_WBS_F) B
    ON A.PS_PSP_PNR = B.WBS_ELEMENT
 WHERE A.MANDT = 800
   AND A.CREDIT_SGMNT <> 0000
 GROUP BY B.WBS_SPIT_ELEMENT
         ,A.PARTNER
)
INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_BUSINESS_T
SELECT
      SUBSTRING(CAST(BO.CREATION_DATE AS STRING),1,10) AS BUSINESS_CREATE_DATE --商机创建日期
	 ,BO.APPLY_USER_AREA_CODE --销售区域编码
	 ,BO.APPLY_USER_AREA AS APPLY_USER_AREA_NAME --销售区域名称
	 ,BO.APPLY_USER_DEPT_CODE --销售部门编码
	 ,BO.APPLY_USER_DEPT AS APPLY_USER_DEPT_NAME --销售部门名称
	 ,BO.STRATEGIC_CUSTOMER_ID --大客户ID
	 ,BO.STRATEGIC_CUSTOMER_NAME --大客户名称
	 ,BO.APPLY_USER_ID --业务员ID
	 ,BO.APPLY_USER AS APPLY_USER_NAME -- 业务员名称
	 ,BO.BUSINESS_ID --商机ID
	 ,BO.BUSINESS_CODE --商机编号
	 ,BO.BUSINESS_NAME --商机名称（项目名称）
	 ,BO.CUSTOMER_CODE --客户编码
	 ,BO.CUSTOMER_NAME --客户名称
	 ,BO.WBS --WBS号
	 ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	 ,T1.CREDIT_AMT --授信金额
	 ,T2.USER_NAME AS APPLY_EMP_NUM --业务员工号
  FROM SDI.SDI_BUSINESS_OPPORTUNITY_1033 BO
  LEFT JOIN T1
    ON BO.WBS = T1.WBS
   AND BO.CUSTOMER_CODE = T1.CUST_NUM	
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T2
    ON BO.APPLY_USER_ID = T2.USER_ID
   AND T2.DELETE_FLAG = 0 
 WHERE BO.DELETE_FLAG = 0
