      ,T.SALE_CHNL_CODE --分销渠道编码
	  ,T.SALE_CHNL_NAME --分销渠道名称
      ,T.STAT_YEAR --年份
	  ,CONCAT(T.STAT_YEAR,'-12') AS STAT_MONTH --月份	  
      ,T.VAILD_DATE --合同有效期
	  ,T.GC12Y AS SALE_TGT_AMT --整装任务目标
	  ,A.S --省份
	  ,A.SHI1 --城市
	  ,T.MAIN_CUST_NUM --母客编
	  ,T.IS_MAIN_CUST --是否母客编
	  ,T.IS_SAME_TGT --是否与母客编共同使用销售任务
  FROM T
  LEFT JOIN T5 AS A
    ON T.ID = A.MAINID
 WHERE T.RN = 1
   AND A.S IS NOT NULL
   AND T.SALE_CHNL_CODE = 'GD02'   
)
INSERT OVERWRITE TABLE DM.DM_CS_SALE_TGT_T
SELECT DISTINCT
       LPAD(T1.CUST_NUM,10,'0') AS CUST_NUM --客编
      ,T1.SALE_CHNL_CODE --分销渠道编码
	  ,T1.SALE_CHNL_NAME --分销渠道名称
      ,T1.STAT_YEAR --年份
	  ,T1.STAT_MONTH --月份	  
      ,T1.VAILD_DATE --合同有效期
	  ,T1.SALE_TGT_AMT --整装任务目标
	  ,COALESCE(A.NEW_PROVINCE,T1.SHENG) AS PROV_NAME --省份
	  ,COALESCE(B.NEW_CITY,T1.SHI) AS CITY_NAME --城市
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,LPAD(T1.MAIN_CUST_NUM,10,'0') AS MAIN_CUST_NUM --母客编
	  ,T1.IS_MAIN_CUST --是否母客编
	  ,T1.IS_SAME_TGT --是否与母客编共同使用销售任务
 FROM (SELECT *
             ,ROW_NUMBER() OVER(PARTITION BY CUST_NUM,SALE_CHNL_CODE,STAT_MONTH ORDER BY VAILD_DATE DESC) AS RN
         FROM T1) AS T1
 LEFT JOIN (SELECT DISTINCT PROVINCE,NEW_PROVINCE
              FROM SDI.SDI_SALES_MAP_PROVINCE_1037) A --针对一个经销商有多个营地的，嘉颖自己做了合并
   ON T1.SHENG = A.PROVINCE
 LEFT JOIN (SELECT DISTINCT CITY,NEW_CITY
              FROM SDI.SDI_SALES_MAP_PROVINCE_1037) B --针对一个经销商有多个营地的，嘉颖自己做了合并
   ON T1.SHI = B.CITY
WHERE T1.RN = 1
