--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH T1 AS(
SELECT A.STAT_DATE
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
	  ,COUNT(1) AS GLANCE_COUNT --浏览次数
  FROM DWRCS.DWR_WT_USER_SHARE_EVENT_F A
  LEFT JOIN DWIMD.DWI_MD_DMT_DEPT_INFO_T B
    ON A.STORE_ID = B.STORE_ID
   AND B.STORE_ID IS NOT NULL
 WHERE A.ITEM_CODE IS NOT NULL
 GROUP BY A.STAT_DATE
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME	
),T2 AS(
SELECT SUBSTR(A.CREATE_TIME,1,10) AS STAT_DATE
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
	  ,COUNT(1) AS SHARE_COUNT --分享次数
  FROM DWRCS.DWR_WT_USER_BEH_ANALY_F A
  LEFT JOIN DWIMD.DWI_MD_DMT_DEPT_INFO_T B
    ON A.STORE_ID = B.STORE_ID
   AND B.STORE_ID IS NOT NULL    
 WHERE A.ITEM_CODE IS NOT NULL	
 GROUP BY SUBSTR(A.CREATE_TIME,1,10)
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
),T3 AS(
SELECT A.STAT_DATE
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,A.LEV1_DEPT_NAME
	  ,A.LEV2_DEPT_NAME
	  ,A.LEV3_DEPT_NAME
	  ,A.LEV4_DEPT_NAME
	  ,A.LEV5_DEPT_NAME
	  ,A.LEV6_DEPT_NAME
	  ,A.GLANCE_COUNT --浏览次数
	  ,0 AS SHARE_COUNT --分享次数
  FROM T1 A
  
 UNION ALL

SELECT A.STAT_DATE
      ,A.STAT_MONTH
	  ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,A.LEV1_DEPT_NAME
	  ,A.LEV2_DEPT_NAME
	  ,A.LEV3_DEPT_NAME
	  ,A.LEV4_DEPT_NAME
	  ,A.LEV5_DEPT_NAME
	  ,A.LEV6_DEPT_NAME
	  ,0 AS GLANCE_COUNT --浏览次数
	  ,A.SHARE_COUNT --分享次数
  FROM T2 A
)
INSERT OVERWRITE TABLE DM.DM_DMT_PROD_EVENT_T PARTITION(STAT_MONTH)
SELECT A.STAT_DATE
      ,A.ITEM_NO
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,A.LEV1_DEPT_NAME
	  ,A.LEV2_DEPT_NAME
	  ,A.LEV3_DEPT_NAME
	  ,A.LEV4_DEPT_NAME
	  ,A.LEV5_DEPT_NAME
	  ,A.LEV6_DEPT_NAME
	  ,SUM(A.GLANCE_COUNT) AS GLANCE_COUNT --浏览次数
	  ,SUM(A.SHARE_COUNT) AS SHARE_COUNT --分享次数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
      ,A.STAT_MONTH	  
  FROM T3 A
 GROUP BY
       A.STAT_DATE
	  ,A.ITEM_NO 
      ,A.ITEM_CODE
	  ,A.ITEM_NAME
      ,A.LEV1_DEPT_NAME
	  ,A.LEV2_DEPT_NAME
	  ,A.LEV3_DEPT_NAME
	  ,A.LEV4_DEPT_NAME
	  ,A.LEV5_DEPT_NAME
	  ,A.LEV6_DEPT_NAME 
      ,A.STAT_MONTH
;
