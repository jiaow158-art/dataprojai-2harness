DELETE FROM dm.dm_dp_api_sales_sum ;
INSERT INTO dm.dm_dp_api_sales_sum 
with tem as(
select 
    a.calmonth ,   -- 年月
    e.sales_center_code, -- 业务部门代码
    e.sales_center, -- 业务部门
    e.sales_dep_code,   -- 营销部代码
    e.sales_dep,   -- 营销部
    e.sales_region_code,   -- 销区代码
    e.sales_region,   -- 销区
    case
       when a.integrate_channel = 'GD03'  then 'GD03'
    --   when a.integrate_channel2 = 'GD04' then 'GD04' --2023.12.14 取消特惠品渠道
       else a.integrate_channel
       end as integrate_channel,   -- 整合渠道2
    case
       when a.integrate_channel = 'GD03'  then '工程渠道'
    --   when  a.integrate_channel2 = 'GD04' then '特惠品' --2023.12.14 取消特惠品渠道
       when a.integrate_channel = 'GD01' then '零售'
       when a.integrate_channel = 'GD02' then '整装头部'
       end   integrate_channel_desc,  -- 渠道
    coalesce(case when c0.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when c0.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN c0.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN c0.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN c0.product_level2_code = '1003' AND c0.dimension = '1500X750' OR c0.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN c0.product_level2_code = '1003' AND (c0.dimension != '1500X750' AND c0.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN c0.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN c0.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN c0.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN c0.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate2,   -- 财务品类
    coalesce(a.size_dim, '其他规格')  as dimension,   -- 财务规格
    a.material as material_num,--产品编码
    a.material___t as material_name,  -- 产品名称
    a.ambperformance,  -- 总金额（元）
    zxsmj,          -- 面积（平方米）
    case when coalesce(commission_group, '#') != '02' then zsyjf end as zsyjf,  -- 使用积分
    case when a.data_source = 'T' then 'Y'
    	when substring(d.CREATE_DATE,1,7) = a.calmonth then 'Y'
	    else 'N'
	    end is_current_month,   -- 是否当月订单，待加工
	case 
	    when c0.prod_property_name = 'ART产品' then 'ART'
	    when c0.prod_property_name = 'DPI产品' then 'DPI'
	    when c0.prod_property_name in('瓷砖产品','梦之家产品')  then '瓷砖'
		else  '1+N' end n_cate_flag,  -- 1+n 品类
	case when coalesce(a.bus_area, '#') != 'G000' and a.data_source = 'S'  then 'Y' else  'N' end as is_price_cal,
	a.data_source
from dm.dm_ambv2_zzt_xssr_t a
join dm.dm_dp_api_sales_group_struct e on a.sales_grp = e.sales_grp
    left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
    left join dwimd.dwi_md_data_material_general_t c0 on a.material = c0.material_num
    --left join DM.DM_RPT_PROD_LEVEL_T c on coalesce(c0.product_level3_code,c0.product_level2_code,c0.product_level1_code) = c.prod_code3 and c.PROD_CODE3 IS NOT NULL
    left join DWROTD.DWR_SO_HEAD_SAP_F d on a.doc_number = d.sal_vch_num
where 1=1 
and (NODE_NAME2 = 'H11000001' or (coalesce(b.node_name2, '#') = '#' AND A.PROD_PROPERTY_NAME IN ('整装产品', '瓷砖产品')))
)
select 
	substring(calmonth,1,4) as stat_year ,   -- 年月
	calmonth as stat_month_ym,
	sales_center, -- 业务部门
    sales_center_code, -- 业务部门代码
    sales_dep,   -- 营销部
    sales_dep_code,   -- 营销部代码
    sales_region,   -- 销区
    sales_region_code,   -- 销区代码
    integrate_channel_desc,  -- 渠道
	integrate_channel,   -- 整合渠道代码
    fin_cate2 as fin_cate,   -- 财务品类
    dimension,   -- 财务规格
    material_num,--产品编码
    material_name,  -- 产品名称
    sum(ambperformance) as sales_amt,  -- 总金额（元）
    sum(zxsmj) as sales_area,          -- 面积（平方米）
    sum(zsyjf) as zsyjf,  -- 使用积分
    is_current_month,   -- 是否当月订单，待加工
	n_cate_flag,  -- 1+n 品类
    current_timestamp() as dw_last_update_date,
    calmonth as stat_month_ym,
    is_price_cal,
    data_source
from tem 
group by  
    calmonth ,   -- 年月
    sales_center_code, -- 业务部门代码
    sales_center, -- 业务部门
    sales_dep_code,   -- 营销部代码
    sales_dep,   -- 营销部
    sales_region_code,   -- 销区代码
    sales_region,   -- 销区
	integrate_channel_desc,  
	integrate_channel,  
    fin_cate2, 
    dimension, 
    material_num,
    material_name,
    is_current_month,
	n_cate_flag,
	is_price_cal,
	data_source
