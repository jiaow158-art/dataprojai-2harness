		 -- ,t1.gl_account_code

		,t2.sales_group
		,t2.wbs_code
		,sum(tsl) tsl
		,sum(asset_original_value)  +
		 sum(current_period_depreciation) +sum(fixed_asset_impairment) as  net_fixed_asset_value
		,sum(asset_original_value)  asset_original_value
		,sum(current_period_depreciation)  accumulated_depreciation
		,sum(current_period_depreciation)  current_period_depreciation
		,sum(asset_scrap_value) asset_scrap_value

    from dwrfin.dwr_cost_fund_cost_asset_d t1

    on t1.asset_code = t2.asset_code 
    and t1.asset_sub_code = t2.asset_sub_code

),tmp1 as 
(select
	 t2.month_id        stat_month -- `月份`
	,company_code        -- 公司
	,asset_code          -- 资产号
	,asset_desc         -- 资产描述

	,sum(case when t2.month_id = t1.months then remaining_useful_life_years else 0 end ) remaining_useful_life_years -- 尚可使用年限
	,sum(sum(case when t2.month_id = t1.months then tsl else 0 end ))  over(partition by  t1.asset_code,t1.asset_sub_code,t1.company_code order by t2.month_id) tsl -- 资产净值(辅助)
	,sum(sum(case when t2.month_id = t1.months then net_fixed_asset_value else 0 end ))  over(partition by  t1.asset_code,t1.asset_sub_code,t1.company_code order by t2.month_id) net_fixed_asset_value -- 资产净值
	,sum(sum(case when t2.month_id = t1.months then asset_original_value else 0 end ))  over(partition by  t1.asset_code,t1.asset_sub_code,t1.company_code order by t2.month_id) asset_original_value -- 购置原值

	,sum(sum(case when t2.month_id = t1.months then fixed_asset_impairment else 0 end ))  over(partition by  t1.asset_code,t1.asset_sub_code,t1.company_code order by t2.month_id) fixed_asset_impairment -- 资产减值
