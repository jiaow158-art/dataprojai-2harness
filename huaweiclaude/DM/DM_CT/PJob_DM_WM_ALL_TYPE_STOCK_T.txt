set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;

with t1 as(
select
     a.MATERIAL_ID
	,a.MATERIAL_NUM AS MATERIAL_NUM  -- matnr
    ,a.MATERIAL_NAME	
	,a.FACTORY_ID
	,a.FACTORY_CODE AS FACTORY_CODE  -- werks
	,a.FACTORY_NAME
	,a.STOCK_PLACE as INV_LOCATION  -- lgort
	,a.BATCH_NUM AS BATCH_NUM  -- CHARG
	,RETAIL_VOUCHER_NO AS SAL_VCH_NUM --销售订单号  SO_NO
	,RETAIL_VOUCHER_PROJECT_NO AS SAL_VCH_ITEM_NUM --销售订单项目  SO_ITEM
	,WBS_ELEMENT AS WBS_ELEMENT  -- WBS_NUM
	,CUST_ID
	,CUST_NUM --客户  CUSTOMER 
	,CUST_NAME
	,SUPPLIER_OR_CREDITOR_ID
	,SUPPLIER_OR_CREDITOR_NUM --供应商  VENDOR
	,SUPPLIER_OR_CREDITOR_NAME
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') in('','K_T148T') AND COALESCE(STOCK_TYPE,'') = '01' then COALESCE(STOCK_QTY,0) else 0 end) as zkkc--总库
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') = 'Q_T148T'  AND COALESCE(STOCK_TYPE,'') = '01' then COALESCE(STOCK_QTY,0) else 0 end) as xmkc--项目库存数
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') in('','K_T148T') AND COALESCE(STOCK_TYPE,'') = '02' then COALESCE(STOCK_QTY,0) else 0 end) as zjkc--质检数
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') in('','K_T148T') AND COALESCE(STOCK_TYPE,'') = '07' then COALESCE(STOCK_QTY,0) else 0 end) as djkc--冻结数
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') = 'Q_T148T' AND COALESCE(STOCK_TYPE,'') = '02' then COALESCE(STOCK_QTY,0) else 0 end) as xmzjkc--项目质检数
	,sum(case when COALESCE(SPECIAL_STOCK_FLAG,'') = 'Q_T148T' AND COALESCE(STOCK_TYPE,'') = '07' then COALESCE(STOCK_QTY,0) else 0 end) as xmdjkc--项目冻结数
	,SUM(COALESCE(STOCK_QTY,0)) AS STOCK_QTY
from DWROTD.DWR_WM_ALL_TYPE_STOCK_F a  -- ZCDSV_TOTALSTCK
group by a.MATERIAL_ID
	    ,a.MATERIAL_NUM
        ,a.MATERIAL_NAME	
	    ,a.FACTORY_ID
	    ,a.FACTORY_CODE
	    ,a.FACTORY_NAME
	    ,a.STOCK_PLACE 
	    ,a.BATCH_NUM
	    ,RETAIL_VOUCHER_NO
	    ,RETAIL_VOUCHER_PROJECT_NO
	    ,WBS_ELEMENT
	    ,CUST_ID
	    ,CUST_NUM
	    ,CUST_NAME
	    ,SUPPLIER_OR_CREDITOR_ID
	    ,SUPPLIER_OR_CREDITOR_NUM
	    ,SUPPLIER_OR_CREDITOR_NAME
)
--留货库存
,t11 as(
SELECT 
--SUM(COALESCE(A.SKU_CONFIRM_QTY,0)) SKU_CONFIRM_QTY -- VMENG
	sum(case when COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'Q_T163I' then COALESCE(A.SKU_CONFIRM_QTY,0) else 0 end) as SKU_PROCONFIRM_QTY --项目留货数
   ,SUM(case when (COALESCE(A.SUBJECT_DISTR_TYPE,'') = '' OR  (COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'M_T163I' AND COALESCE(A.REQ_TYPE,'') = '01'))
	then COALESCE(A.SKU_CONFIRM_QTY,0) else 0 end) as SKU_CONFIRM_QTY -- VMENG 留货数
    ,A.MATERIAL_ID
	,A.MATERIAL_NUM
    ,A.MATERIAL_NAME
	,A.FACTORY_ID
	,A.FACTORY_CODE
	,A.FACTORY_NAME
	,A.INV_LOCATION -- LGORT
	,A.BATCH_NUM -- CHARG
	,A.SAL_VCH_NUM --销售凭证 VBELN
	,A.SAL_VCH_ITEM_NUM --销售凭证项目 POSNR
	,A.WBS_ELEMENT --WBS 要素 PSPEL
	,A.CUST_ID
	,A.CUST_NUM--客户  CUSTOMER KUNNR
	,A.CUST_NAME	
	,NULL AS SUPPLIER_OR_CREDITOR_ID
	,NULL AS SUPPLIER_OR_CREDITOR_NUM --供应商 VENDOR
	,NULL AS SUPPLIER_OR_CREDITOR_NAME
FROM DWROTD.DWR_SO_SAL_REQ_SAP_F A-- VBBE
WHERE COALESCE(A.SALES_DIS_VCH_TYPE,'') = 'C_VBTYPL'  -- VBTYP
AND ( (COALESCE(A.SUBJECT_DISTR_TYPE,'') = '' OR  (COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'M_T163I' AND COALESCE(A.REQ_TYPE,'') = '01'))  -- KNTTP  bdart
    OR  COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'Q_T163I' )--项目留货数
AND COALESCE(A.SKU_CONFIRM_QTY,0) > 0 
	group by MATERIAL_ID
	,MATERIAL_NUM
    ,MATERIAL_NAME
	,FACTORY_ID
	,FACTORY_CODE
	,FACTORY_NAME
	,INV_LOCATION -- LGORT
	,BATCH_NUM -- CHARG
	,SAL_VCH_NUM --销售凭证 VBELN
	,SAL_VCH_ITEM_NUM --销售凭证项目 POSNR
	,WBS_ELEMENT --WBS 要素 PSPEL
	,CUST_ID
	,CUST_NUM --客户  CUSTOMER KUNNR
	,CUST_NAME		
union all
SELECT --'转储单留货'
	0  as SKU_PROCONFIRM_QTY --项目留货数
   ,SUM(t1.res_stock_qty) as SKU_CONFIRM_QTY -- VMENG 留货数
    ,t3.MATERIAL_ID
	,t3.MATERIAL_NUM
    ,t3.MATERIAL_NAME
	,null as FACTORY_ID
	,t1.stockout_FACTORY_CODE as FACTORY_CODE --发出
	,null as   FACTORY_NAME
	,t1.inv_location as INV_LOCATION -- LGORT
	,t1.batch_num as BATCH_NUM -- CHARG
	,t1.purchase_order_num as SAL_VCH_NUM --销售凭证 VBELN
	,t1.purchase_order_item_num as SAL_VCH_ITEM_NUM --销售凭证项目 POSNR
	,t1.wbs_element as WBS_ELEMENT --WBS 要素 PSPEL
	,null as CUST_ID
	,t1.CUST_NUM as CUST_NUM--客户  CUSTOMER KUNNR
	,t1.CUST_NAME as CUST_NAME	
	,NULL AS SUPPLIER_OR_CREDITOR_ID
	,NULL AS SUPPLIER_OR_CREDITOR_NUM --供应商 VENDOR
	,NULL AS SUPPLIER_OR_CREDITOR_NAME
from  dwrotd.dwr_wm_stock_res_from_purchase t1
left join (select distinct lev1_name,lev2_name,lev3_name,lev4_name,lev5_name from dm.dm_rpt_sale_grp_t) t2 
	on t1.belong_company = t2.lev5_name
left join dwimd.dwi_md_data_material_general_t t3
	on t1.material_num = t3.material_num
group by 
    t3.MATERIAL_ID
	,t3.MATERIAL_NUM
    ,t3.MATERIAL_NAME
    ,t1.stockout_FACTORY_CODE
    ,t1.inv_location 
	,t1.batch_num 
	,t1.purchase_order_num 
	,t1.purchase_order_item_num 
	,t1.wbs_element
	,t1.CUST_NUM 
	,t1.CUST_NAME 
)

,t22 as(
SELECT 
     T.MATERIAL_ID
	,a.matnr AS MATERIAL_NUM
	,T.MATERIAL_NAME
	,T1.FACTORY_ID
	,a.werks AS FACTORY_CODE
	,T1.FACTORY_NAME
	,a.lgort AS INV_LOCATION
	,a.CHARG AS BATCH_NUM
	,a.AUFNR AS SAL_VCH_NUM --销售凭证
	,a.AUFPS AS SAL_VCH_ITEM_NUM--销售凭证项目
	,a.PSPEL AS WBS_ELEMENT --WBS要素
	,NULL AS CUST_ID--客户
	,NULL AS CUST_NUM
	,NULL AS CUST_NAME
	,D.SUPPLIER_OR_CREDITOR_ID
    ,D.SUPPLIER_OR_CREDITOR_NUM -- LIFNR
    ,D.SUPPLIER_OR_CREDITOR_NAME
	,SUM(COALESCE(a.erfmg,0)) erfmg
	,SUM(COALESCE(a.enmng,0)) enmng
FROM SDI.SDI_RESB_1001 a
left join DWIMD.DWI_MD_MATERIAL_GENERAL_T T	--关联物料主数据表,取MATERIAL_NAME
ON T.MATERIAL_NUM = a.matnr		--物料ID
AND T.GROUP_NUM = a.MANDT			--集团
AND T.END_DATE ='4712-12-31'				--结束时间'4712-12-31'
AND A.DEL_FLAG = 'N'					--删除标识'N'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T T1			--关联工厂主数据表,取FACTORY_NAME
ON  T1.FACTORY_CODE = a.werks		--工厂ID
AND T1.GROUP_NUM = a.MANDT			--集团
AND T1.END_DATE ='4712-12-31'				--结束时间'4712-12-31'
AND A.DEL_FLAG = 'N'					--删除标识'N'
left join SDI.SDI_RKPF_1001 b
on a.RSNUM = b.RSNUM
and COALESCE(b.NPLNR,'') <> ''
left join(
  SELECT PURCHASE_VOUCHER_NUM, -- EBELN
         PURCHASE_VCH_ITEM_NUM1, -- EBELP
		 CUST_NUM -- KUNNR
  FROM DWRPUR.DWR_PTP_PURCHASE_VCH_ITEM_F -- EKPO    可以干掉这段逻辑，因为关联是left join,也没有取出字段
  GROUP BY PURCHASE_VOUCHER_NUM,
           PURCHASE_VCH_ITEM_NUM1,
		   CUST_NUM) c
on a.BANFN = c.PURCHASE_VOUCHER_NUM
and a.BNFPO = c.PURCHASE_VCH_ITEM_NUM1
left join(
	SELECT PURCHASE_VOUCHER_NUM, -- EBELN
		   SUPPLIER_OR_CREDITOR_ID,
		   SUPPLIER_OR_CREDITOR_NUM, -- LIFNR
		   SUPPLIER_OR_CREDITOR_NAME
	FROM DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F -- EKkO
	GROUP BY PURCHASE_VOUCHER_NUM,
	         SUPPLIER_OR_CREDITOR_ID,
		     SUPPLIER_OR_CREDITOR_NUM, -- LIFNR
		     SUPPLIER_OR_CREDITOR_NAME
) d
on a.BANFN = d.PURCHASE_VOUCHER_NUM
WHERE COALESCE(a.xloek,'') = ''
AND COALESCE(a.kzear,'') = ''
AND COALESCE(a.erfmg,0) > 0 
and a.bdart not in ('BB','SB')
and b.NPLNR is null
group by T.MATERIAL_ID
	,a.matnr
	,T.MATERIAL_NAME
	,T1.FACTORY_ID
	,a.werks
	,T1.FACTORY_NAME
	,a.lgort
	,a.CHARG
	,a.AUFNR
	,a.AUFPS
	,a.PSPEL
	,D.SUPPLIER_OR_CREDITOR_ID
    ,D.SUPPLIER_OR_CREDITOR_NUM -- LIFNR
    ,D.SUPPLIER_OR_CREDITOR_NAME
)
,t23 as(

	    ,A.CUST_NAME
	    ,A.SUPPLIER_OR_CREDITOR_ID

	    ,B.CUST_NUM
	    ,B.CUST_NAME
	    ,B.SUPPLIER_OR_CREDITOR_ID
        ,B.SUPPLIER_OR_CREDITOR_NUM -- LIFNR

group by A.MATERIAL_ID
	    ,a.MATERIAL_NUM
	    ,A.MATERIAL_NAME
	    ,A.FACTORY_ID
	    ,a.FACTORY_CODE
	    ,A.FACTORY_NAME
	    ,a.INV_LOCATION
	    ,a.BATCH_NUM
	    ,a.SAL_VCH_NUM --销售凭证
	    ,a.SAL_VCH_ITEM_NUM--销售凭证项目
	    ,a.WBS_ELEMENT --WBS要素
	    ,A.CUST_ID--客户
	    ,A.CUST_NUM
	    ,A.CUST_NAME
	    ,A.SUPPLIER_OR_CREDITOR_ID
        ,A.SUPPLIER_OR_CREDITOR_NUM -- LIFNR
        ,A.SUPPLIER_OR_CREDITOR_NAME
)
