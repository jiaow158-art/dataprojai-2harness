-- 		Ontime_QTY DECIMAL(32, 8) COMMENT '准时单量',
-- 		Daily_appointment_count DECIMAL(32, 8) COMMENT '当日预约量',
-- 		Total_vehicles DECIMAL(32, 8) COMMENT '总车辆',
-- 		Ontime_vehicles DECIMAL(32, 8) COMMENT '准到车辆',
-- 		missed_vehicles DECIMAL(32, 8) COMMENT '未准到车辆',
-- 		Twait_time DECIMAL(32, 8) COMMENT '准到等待时间',
-- 		Fwait_time DECIMAL(32, 8) COMMENT '未准到等待时间',

WITH  reservation_record as (
select   
*
FROM 
dwrotd.dwr_tm_reservation_record
where 
reservation_state in ('LOAD_END','BE_LOAD','WAIT_QUEUE','COMPLETE','LINE_UP') 

)
,reservation_wait_time AS (
		SELECT
			SUBSTRING(T1.load_end_time, 1, 10) AS load_end_time,
			T1.warehouse_id,
			T1.warehouse_name,
			T1.reservation_group_code,
			MAX(wait_time) AS Twait_time,
			'' AS Fwait_time
		FROM
		reservation_record T1
		where
			HOUR(T1.load_end_time) >= 5
		    AND T1.is_abnormal <> 1
		GROUP BY
			T1.reservation_group_code,
			T1.warehouse_id,
			T1.warehouse_name,
			SUBSTRING(T1.load_end_time, 1, 10)
		HAVING
			COUNT(*) = COUNT(
				CASE
					WHEN is_accurate = 'Y' THEN 1
				END
			)
		UNION ALL
		SELECT
			SUBSTRING(T2.load_end_time, 1, 10) AS load_end_time,
			T2.warehouse_id,
			T2.warehouse_name,
			T2.reservation_group_code,
			'' AS Twait_time,
			MAX(T2.wait_time) AS Fwait_time
		FROM
			reservation_record T2
		where
			HOUR(T2.load_end_time) >= 5
			AND T2.is_abnormal <> 1
		GROUP BY
			T2.reservation_group_code,
			T2.warehouse_id,
			T2.warehouse_name,
			SUBSTRING(T2.load_end_time, 1, 10)
		having
			min(is_accurate) = 'N'
	),
	reservation_wait_time_grp as (
		SELECT
			load_end_time,
			warehouse_id,
			warehouse_name,
			reservation_group_code,
			sum(Twait_time) as Twait_time,
			sum(Fwait_time) as Fwait_time
		from
			reservation_wait_time
		GROUP BY
			load_end_time,
			warehouse_id,
			warehouse_name,
			reservation_group_code
	),
	TMP as (
		select
			SUBSTRING(T1.sign_time, 1, 10) AS load_end_time,
			T1.warehouse_id,
			T1.warehouse_name,
			count(t1.reservation_code) AS all_order_qty, --总单量
			count(
				case
					WHEN T1.is_staff = 'N' THEN 1
					ELSE NULL
				END
			) AS appointment_qty, --预约单量
			COUNT(
				CASE
					WHEN (
						T1.is_staff = 'N'
						AND T1.reservation_date = substring(T1.sign_time, 1, 10)
					) THEN 1
					ELSE NULL
				END
			) AS Ontime_QTY, --准时单量
			COUNT(
				CASE
					WHEN (
						T1.is_staff = 'N'
						AND SUBSTRING(T1.creation_date, 1, 10) = SUBSTRING(T1.sign_time, 1, 10)
					) THEN 1
					ELSE NULL
				END
			) - COUNT(
				CASE
					WHEN (
						T1.is_staff = 'N'
						AND SUBSTRING(T1.creation_date, 1, 10) = SUBSTRING(T1.sign_time, 1, 10)
					) THEN CASE
						WHEN T1.reservation_type = 'WITHOUT_SUPPORT'
						AND HOUR(T1.creation_date) < 10 THEN 1
						ELSE NULL
					END
					ELSE NULL
				END
			) AS Daily_appointment_count, --当日预约量
			'' AS Total_vehicles, --总车辆
			'' AS Ontime_vehicles, --准到车辆
			'' AS missed_vehicles, --未准到车辆
			'' AS Twait_time,
			'' AS Fwait_time,
			'' AS is_abnormal_qty  --异常车辆
		FROM
			reservation_record  T1
		GROUP BY
			SUBSTRING(T1.sign_time, 1, 10),
			T1.warehouse_id,
			T1.warehouse_name
		UNION ALL ------------准到车辆逻辑
		SELECT
			SUBSTRING(T1.load_end_time, 1, 10) AS load_end_time,
			T1.warehouse_id,
			T1.warehouse_name,
			'' AS all_order_qty, --总单量
			'' AS appointment_qty, --预约单量
			'' AS Ontime_QTY, --准时单量
			'' AS Daily_appointment_count, --当日预约量
			'' AS Total_vehicles, --总车辆
			COUNT(
				DISTINCT CASE
					WHEN t2.all_accurate = 1 then T1.reservation_group_code
					else null
				end
			) as Ontime_vehicles, --准到车辆
            '' AS missed_vehicles, --未准到车辆
			'' as Twait_time, --准到等待时间
			'' AS Fwait_time, --未准到等待时间
			'' AS is_abnormal_qty  --异常车辆
		FROM
	    	reservation_record T1
			LEFT JOIN (
				select
					sum(wait_time) as Twait_time,
					count(DISTINCT is_accurate) as all_accurate,
					reservation_group_code
				from
					reservation_record
				where load_end_time IS NOT NULL  and load_end_time <> ''
				group by
					reservation_group_code
				having
					count(DISTINCT is_accurate) = 1
					and min(is_accurate) = 'Y'
			) T2 ON T1.reservation_group_code = T2.reservation_group_code
			LEFT JOIN (
				select
					sum(wait_time) as Fwait_time,
					count(DISTINCT is_accurate) as all_accurate,
					reservation_group_code
				from
					reservation_record
				where load_end_time IS NOT NULL  and load_end_time <> ''
				group by
					reservation_group_code
				having
					count(DISTINCT is_accurate) = 1
					and min(is_accurate) = 'N'
			) T3 ON T1.reservation_group_code = T3.reservation_group_code
    	WHERE T1.is_abnormal <> 1
		GROUP BY
			SUBSTRING(T1.load_end_time, 1, 10),
			T1.warehouse_id,
			T1.warehouse_name
		UNION ALL ------------准到等待时间和未准到等待时间
		SELECT
			T1.load_end_time,
			T1.warehouse_id,
			T1.warehouse_name,
			'' AS all_order_qty, --总单量
			'' AS appointment_qty, --预约单量
			'' AS Ontime_QTY, --准时单量
			'' AS Daily_appointment_count, --当日预约量
			'' AS Total_vehicles, --总车辆
			'' AS Ontime_vehicles, --准到车辆
			'' AS missed_vehicles, --未准到车辆
			SUM(T1.Twait_time) AS Twait_time,
			SUM(T1.Fwait_time) AS Fwait_time,
			'' AS is_abnormal_qty  --异常车辆			
		FROM
			reservation_wait_time_grp T1
		GROUP BY
			T1.load_end_time,
			T1.warehouse_id,
			T1.warehouse_name
		UNION ALL ---------总车辆与异常车辆逻辑
			SELECT 
	        SUBSTRING(T1.load_end_time, 1, 10) AS load_end_time,
			T1.warehouse_id,
			T1.warehouse_name,
			'' AS all_order_qty, --总单量
			'' AS appointment_qty, --预约单量
			'' AS Ontime_QTY, --准时单量
			'' AS Daily_appointment_count, --当日预约量
			COUNT(DISTINCT T1.reservation_group_code) AS Total_vehicles, --总车辆
			'' AS Ontime_vehicles, --准到车辆
			'' AS missed_vehicles, --未准到车辆
			'' AS Twait_time,
			'' AS Fwait_time,
			COUNT(DISTINCT CASE WHEN T1.is_abnormal = 1 then T1.reservation_group_code else null end) as is_abnormal_qty -- 异常车辆
			FROM
			reservation_record  T1
			GROUP BY 
			SUBSTRING(T1.load_end_time, 1, 10),
			T1.warehouse_id,
			T1.warehouse_name
	)
INSERT
	OVERWRITE TABLE   DM.DM_DAILY_APPOINTMENT_STATUS 
select
	load_end_time,
	CASE WHEN warehouse_id = '27' THEN '13'
	     WHEN warehouse_id = '19' THEN '15'
	     WHEN warehouse_id = '26' THEN '14'
	     WHEN warehouse_id = '23' THEN '16'
	     ELSE warehouse_id
	     END AS warehouse_id,
	CASE WHEN warehouse_name = '清远配送仓' THEN '清远仓'
	     WHEN warehouse_name = '江西配送仓' THEN '丰城仓'
	     WHEN warehouse_name = '重庆配送仓' THEN '重庆仓'
	     WHEN warehouse_name = '湖南配送仓' THEN '湖南仓'
	     ELSE warehouse_name
	     END AS warehouse_name,
	sum(all_order_qty) as all_order_qty,
	sum(appointment_qty) as appointment_qty,
	sum(ontime_qty) as ontime_qty,
	sum(daily_appointment_count) as daily_appointment_count,
	sum(total_vehicles) as total_vehicles,
	sum(ontime_vehicles) as ontime_vehicles,
	sum(missed_vehicles) as missed_vehiclesm,
	SUM(Twait_time) AS Twait_time,
	SUM(Fwait_time) AS Fwait_time,
	'N' AS del_flag,
	current_timestamp () as dw_last_update_date,
	SUM(is_abnormal_qty) AS is_abnormal_qty
from
	tmp
WHERE load_end_time <> '' AND load_end_time IS NOT NULL
GROUP BY
	load_end_time,
	CASE WHEN warehouse_id = '27' THEN '13'
	     WHEN warehouse_id = '19' THEN '15'
	     WHEN warehouse_id = '26' THEN '14'
	     WHEN warehouse_id = '23' THEN '16'
	     ELSE warehouse_id
	     END ,
	CASE WHEN warehouse_name = '清远配送仓' THEN '清远仓'
	     WHEN warehouse_name = '江西配送仓' THEN '丰城仓'
	     WHEN warehouse_name = '重庆配送仓' THEN '重庆仓'
	     WHEN warehouse_name = '湖南配送仓' THEN '湖南仓'
	     ELSE warehouse_name
	     END
