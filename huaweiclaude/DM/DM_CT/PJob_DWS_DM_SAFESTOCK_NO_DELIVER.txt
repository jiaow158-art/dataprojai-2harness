delete from   dm.dm_safestock_no_deliver ;  
insert into dm.dm_safestock_no_deliver
select material_num,product_area  as prod_area,sum(nodeliver_qty_aps) as nodeliver_qty_aps
from  dm.dm_rpt_sales_deliver_t 
where nodeliver_qty_aps >0
and stat_month ='${PERIOD_ID_M}'
group by material_num,product_area;
