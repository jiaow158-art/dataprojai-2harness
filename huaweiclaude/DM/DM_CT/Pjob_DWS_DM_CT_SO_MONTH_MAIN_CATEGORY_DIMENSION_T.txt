-- update：2024/03/29 修改逻辑：出库面积使用业绩表，补充订单表过滤条件
delete
from dm.dm_ct_so_month_main_category_dimension_t
where stat_month = '${PERIOD_ID_M}';
insert into dm.dm_ct_so_month_main_category_dimension_t
select to_char(to_date(A.audat), 'yyyy-MM')                                 stat_month,
       case
           when zh_channel_code1 = 'GD03' then '工程'
           when zh_channel_code1 = 'GD01' then '零售'
           when zh_channel_code1 = 'GD02'
               then '整装' end                                                channel_text,
       case when z.category_code is null then '其他' else c.category_name end category_name,
       c.dimension,
       sum(A.MWSBP + A.netwr)                                               amount,
       sum((vmeng + wqrsl) * a.width * a.length_1)                          area,
       current_timestamp
from DM.DM_OTD_SALES_ORDER_DET_T A
         LEFT JOIN dm.dm_rpt_sale_grp_t B
                   ON A.vkgrp = B.sale_grp
         left join dwimd.dwi_md_data_material_general_t c on a.matnr = c.material_num
         left join (select distinct category_code
                    from upload.upload_group_main_category_specification_t
                    where division_code = 'H11000001') z on c.category = z.category_code
where case
          when A.audat in ('', '00000000') then ''
          when A.audat is null then ''
          else to_char(to_date(A.audat), 'yyyy-MM')
          end = '${PERIOD_ID_M}'
  and B.lev2_name = '瓷砖事业部'
  and coalesce(abgru, '11') in ('11', '56')
  and c.matl_group_code = '1000010'
  and c.external_matl_group_code = '10010'
  and substr(werks, 1, 2) in ('12', '13', '15', '16', '17', '18', '19', '7D', '72', '75', '77', '79', '8A')
  and substr(lgort, 1, 2) != '00'
  and lgort not in ('BC', 'GFZ', '7F21')
group by stat_month, channel_text, case when z.category_code is null then '其他' else c.category_name end, c.dimension
