with tmp_data as(
	SELECT T1.VBELN,
		   T1.POSNR,
		   T1.ID_BOM,
		   T2.COMPLETE_CODE
	FROM SDI.SDI_VBAP_1001 T1
	LEFT JOIN (
		SELECT DISTINCT T.ID_BOM,T.COMPLETE_CODE
		   FROM(
			   SELECT ITEM_NO AS ID_BOM,
			          PRODUCT_CODE AS COMPLETE_CODE
			   FROM SDI.SDI_S_ORDER_ITEM_LT_1029
			   WHERE ITEM_NO IS NOT NULL
			   UNION ALL
			  	SELECT T2.ROW_ID AS ID_BOM,
			  	       T1.NAME AS COMPLETE_CODE
				FROM SDI.SDI_S_PROD_INT_1002 T1
				JOIN SDI.SDI_S_ORDER_ITEM_1002 T2
				ON T1.ROW_ID = T2.PROD_ID
			 ) T
	)T2
	ON T1.ID_BOM = T2.ID_BOM
	WHERE T1.ID_BOM IS NOT NULL 
	AND T1.ID_BOM <> ''
 )
