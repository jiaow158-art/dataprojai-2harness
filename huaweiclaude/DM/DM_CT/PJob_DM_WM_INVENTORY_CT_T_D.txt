set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;

set hive.exec.dynamic.partition=true;

--CREATE TABLE DM.DM_WM_INVENTORY_CT_T_D AS
with CT_SALES_DATA as(   -- 瓷砖销售数据
	select SUBSTRING(doc_date,1,10) as stat_date_d,   -- 日期
	   	   dealer_code AS CUSTOMER_CODE,   -- 客户编码
	       material_num,    -- 物料编码
	       sum(t.cost_total_amt) as cost_total_amt, -- 成本合计
	       sum(t.material_qty) as sales_outgoods_qty, -- 销售出库数量
	       sum(t.line_total_amt) as sales_outgoods_amt --销售出库金额
	from DWROTD.DWR_WM_B1_SALES_ORDER_F t
	where t.customer_type = 'CT'
	group by SUBSTRING(doc_date,1,10),
			 dealer_code,
			 material_num
),
tmp_data as(
SELECT T1.period_id_d AS day_dt, -- 日期
	   T1.data_source_falg , -- 数据来源
	   T2.share_flag , -- 共享仓标识(A:即是共享又是其它 Y：是 N：否)
	   T1.customer_code,  -- 客户编码
	   T1.customer_name,  -- 客户名称
	   T1.sales_org_name , -- 销售组织
	   T1.material_num,   -- 物料编码
	   T1.material_name,   -- 物料名称
	   T1.dimension, -- 规格
	   T1.menu_desc_name , -- 是否主菜单
	   nvl(T3.cost_total_amt,0) as cost_total_amt,  -- 成本
	   T1.qc_qty, -- 起初数量
	   T1.qj_instock_qty, -- 本期入库
	   T1.qj_pp_del_qty, -- 期间采购收货数量
	   T1.qj_sale_return_qty, -- 期间销售退货数量
	   T1.qj_other_instock_qty, -- 期间其他入库数量
	   0 as qj_pp_return_qty , -- 期间采购退货数量
	   T1.qj_outstock_qty, -- 本期出库
	   T1.qj_sale_outstock_qty, -- 期间销售出库数量
	   T1.qj_other_outstock_qty, -- 期间其他出库数量
	   T1.qm_qty, -- 期末数量
	   T1.unit_area_qty, -- 单位面积
	   T1.qm_area_qty, -- 期末面积
	   T1.qj_happen_qty, -- 期间发生
	   T1.qc_amt, -- 期初金额
	   T1.bq_instock_amt, -- 本期入库金额
	   T1.bq_outstock_amt, -- 本期出库金额
	   T1.qm_amt, -- 期末金额
	   T1.bq_happen_amt, -- 本期发生金额
	   T1.last_instock_date, -- 最末入库日期
	   nvl(t3.sales_outgoods_qty, 0) as sales_outgoods_qty,-- 销售出库数量
	   nvl(t3.sales_outgoods_amt , 0) as sales_outgoods_amt, --销售出库金额
	   nvl(t3.sales_outgoods_qty, 0)  * T4.length_1 * T4.width / 1000 / 1000 AS sales_outgoods_area_qty, --销售出库面积
	   T4.material_id,   -- 物料ID
       T4.matl_type_code, -- SAP物料类型编码
       T4.matl_type_name, -- SAP物料类型名称
       T4.product_group_code, --SAP产品组
       T4.product_group_name, -- SAP产品组名称
       T4.product_classify_code, -- 产品分级编码
       T4.product_classify_name, -- 产品分级名称
       T4.product_level_code,  -- 产品层级编码
       T4.product_level_name,  -- 产品层级名称
       T4.product_level1_code,-- 产品层级1编码
       T4.product_level1_name,-- 产品层级1名称
       T4.product_level2_code,-- 产品层级2编码
       T4.product_level2_name,-- 产品层级2名称
       T4.product_level3_code,-- 产品层级3编码
       T4.product_level3_name,-- 产品层级3名称
       T4.product_level4_code,-- 产品层级4编码
       T4.product_level4_name,-- 产品层级4名称
       T4.busi_volume,			-- 体积
       T4.length_1 ,--长
       T4.width   -- 宽
FROM DWROTD.DWR_WM_INVENTORY_CT_T_D T1
JOIN SDI.SDI_CT_CUSTOMER_DATA_1037 T2
ON T1.data_source_falg = T2.data_source_falg 
AND T1.customer_code = T2.customer_code 
LEFT JOIN CT_SALES_DATA T3
ON T1.PERIOD_ID_D = T3.STAT_DATE_D
AND T1.customer_code = T3.customer_code
AND T1.material_num = T3.material_num
JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T4
ON T1.material_num = T4.material_num 
where T1.period_id_d >= date_sub('${PERIOD_ID_D}',3)	-- 传昨天(3天前) --'2022-09-01'
AND T1.period_id_d <= '${PERIOD_ID_D}'	-- 传昨天 --'2021-10-31'
)
INSERT OVERWRITE TABLE DM.DM_WM_INVENTORY_CT_T_D PARTITION(period_id_d)
select t.*,
	   current_timestamp() as dw_last_update_date ,
	   T.day_dt as period_id_d -- 分区字段
from tmp_data t;
