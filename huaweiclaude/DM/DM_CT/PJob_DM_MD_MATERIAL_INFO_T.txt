WITH T AS(
SELECT MATERIAL_NUM
      ,MATL_GROUP2_CODE --主次
      ,ROW_NUMBER() OVER(PARTITION BY MATERIAL_NUM ORDER BY END_DATE DESC,START_DATE DESC) AS RN
  FROM DWIMD.DWI_MD_DATA_MATERIAL_SALES_T
 WHERE GROUP_NUM = '800'
),T1 AS(--获取物料在去年年末时的菜单说明
SELECT A.MATERIAL_NUM
      ,A.MENU_DESC_CODE --菜单说明
      ,A.MENU_DESC_NAME --菜单说明名称
	  ,ROW_NUMBER() OVER(PARTITION BY A.MATERIAL_NUM ORDER BY A.END_DATE DESC) AS RN
  FROM DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T_HIS A
  LEFT JOIN DM.DM_DIM_DATE_D B
    ON 1 = 1
   AND B.DAY_DT = '${PERIOD_ID_D}'	
 WHERE A.START_DATE >= B.DAY_YEND_YAGO_DT
   AND B.DAY_YEND_YAGO_DT <= A.END_DATE
)
INSERT OVERWRITE TABLE DM.DM_MD_MATERIAL_INFO_T
SELECT A.MATERIAL_NUM --物料编码
      ,A.MATERIAL_NAME --名称
      ,A.TEXTURE_1	--产品材质
      ,A.TEXTURE_1_NAME	--产品材质名称
      ,A.COLOR_CODE	--色系
      ,A.COLOR_NAME	--色系名称
      ,A.DIMENSION	--尺寸
      ,A.FACES_CODE	--面相
      ,A.FACES_NAME	--面相名称
      ,A.PRODUCT_PLACE_CODE	--产品工艺编码
      ,A.PRODUCT_PLACE_NAME	--产品工艺名称
      ,A.RESERVED_FIELD_7 AS PRODUCT_BODY_CODE	--产品胚体
      ,A.RESERVED_FIELD7_NAME AS PRODUCT_BODY_NAME	--产品胚体名称
      ,A.RESERVED_FIELD_8 AS PRODUCT_FUNC_CODE	--产品功能
      ,A.RESERVED_FIELD8_NAME AS PRODUCT_FUNC_NAME	--产品功能名称
      ,A.RESERVED_FIELD_1 AS PRODUCT_CHNL_CODE --产品渠道 渠道
      ,A.RESERVED_FIELD1_NAME AS PRODUCT_CHNL_NAME --产品渠道名称 渠道
      ,A.DEF_PROD_CHANNEL --四大渠道
      ,A.PRODUCT_POSITION1 AS PRODUCT_POSITION_CODE --产品定位
      ,A.PRODUCT_POSITION_NAME1 AS PRODUCT_POSITION_NAME --产品定位名称
      ,A.RESERVED_FIELD_6 AS PRODUCT_LVL_CODE --产品等级(主次)
      ,A.RESERVED_FIELD6_NAME AS PRODUCT_LVL_NAME --产品等级名称(主次)
	  ,A.CATEGORY AS PRODUCT_CATG_CODE --品类
      ,A.CATEGORY_NAME AS PRODUCT_CATG_NAME --品类名称
      ,B.CONFIG_NAME AS PRODUCT_SERIES_NAME --产品系列
	  ,T.MATL_GROUP2_CODE --主次
	  ,C.URL_TEXT --URL路径
	  ,A.MENU_DESC_CODE --菜单说明
	  ,A.MENU_DESC_NAME --菜单说明名称
	  ,T1.MENU_DESC_CODE AS LY_MENU_DESC_CODE --去年菜单说明
      ,T1.MENU_DESC_NAME AS LY_MENU_DESC_NAME --去年菜单说明名称	
	  ,D.PRICE_AMT * 0.85 AS PRICE_AMT --EP价格	  
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN D.PRICE_AMT * 0.85 < 40 THEN '40以下'
	        WHEN D.PRICE_AMT * 0.85 < 45 THEN '40-45'
			WHEN D.PRICE_AMT * 0.85 < 50 THEN '45-50'
			WHEN D.PRICE_AMT * 0.85 < 60 THEN '50-60'
			WHEN D.PRICE_AMT * 0.85 < 70 THEN '60-70'
			WHEN D.PRICE_AMT * 0.85 < 90 THEN '70-90'
			WHEN D.PRICE_AMT * 0.85 >= 90 THEN '90以上'
	         END AS PRICE_STAGE --价格段
  FROM DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T A
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R B
    ON SUBSTR(A.PRODUCT_LEVEL_CODE,1,7) = B.SRC_SYS_CONFIG_CODE
   AND B.SOURCE_CODE = 'T179T'
   AND B.LANG_CODE = '1'
  LEFT JOIN T
    ON A.MATERIAL_NUM = T.MATERIAL_NUM
   AND T.RN = 1
  LEFT JOIN(SELECT MATERIAL_NUM
                  ,SPLIT(URL_TEXT,',')[0] AS URL_TEXT --URL路径
				  ,ROW_NUMBER() OVER(PARTITION BY MATERIAL_NUM ORDER BY UPDATE_TIME DESC) AS RN
              FROM DWRDIM.DWR_DIM_MATERIAL_URL_D
		     WHERE IS_HIDE_CODE = 'true'	
		   ) C
    ON A.MATERIAL_NUM = C.MATERIAL_NUM
   AND C.RN = 1
  LEFT JOIN(SELECT MATERIAL_NUM
                  ,PRICE_AMT
				  ,ROW_NUMBER() OVER(PARTITION BY MATERIAL_NUM ORDER BY END_DATE DESC) AS RN
              FROM DWRDIM.DWR_DIM_MATERIAL_PRICE_D
		     WHERE PRICE_GRP_CODE = 'ZA'	--取EP价格
		   ) D
    ON A.MATERIAL_NUM = D.MATERIAL_NUM
   AND D.RN = 1		
  LEFT JOIN T1
    ON A.MATERIAL_NUM = T1.MATERIAL_NUM
   AND T1.RN = 1 
 WHERE A.GROUP_NUM = 800
   AND A.MATL_GROUP_CODE = '1000010'
;
