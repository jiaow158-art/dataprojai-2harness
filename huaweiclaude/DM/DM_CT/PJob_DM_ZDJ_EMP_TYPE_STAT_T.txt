WITH T1 AS(
SELECT A.EMP_ID
      ,A.POST_NAME
      ,A.USER_ID
	  ,SUBSTR(A.CREATE_TIME,1,7) AS CREATE_MONTH --创建月份
  FROM DWRDIM.DWR_DIM_DMT_EMP_INFO_D A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.STATUS = 1
   AND A.DEPT_NAME NOT LIKE '%测试%'
   AND A.DEPT_NAME NOT LIKE '%台湾%'
   AND A.POST_NAME IN('装到家-销售顾问','装到家-鹏筑师傅')
)
INSERT OVERWRITE TABLE DM.DM_ZDJ_EMP_TYPE_STAT_T
SELECT REPLACE(A.POST_NAME,'装到家-','') AS EMP_TYPE --员工类型
      ,COUNT(DISTINCT A.EMP_ID) AS EMP_CNT --员工数
	  ,COUNT(DISTINCT C.EMPLOYEE_ID) AS QY_EMP_CNT --启用员工数
	  ,COUNT(DISTINCT CASE WHEN A.CREATE_MONTH = SUBSTR(DATE_ADD(CURRENT_DATE(),-1),1,7) THEN A.EMP_ID END) AS NEW_EMP_CNT --新增员工数
      ,COUNT(DISTINCT CASE WHEN SUBSTR(C.CREATE_TIME,1,7) = SUBSTR(DATE_ADD(CURRENT_DATE(),-1),1,7) THEN C.EMPLOYEE_ID END) AS CUR_MONTH_QY_EMP_CNT --当月启用员工数
	  ,COUNT(DISTINCT B.USER_ID) AS CUR_MONTH_HY_EMP_CNT --当月活跃员工数 
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T1 AS A
  LEFT JOIN (SELECT DISTINCT USER_ID
               FROM SDI.SDI_UAA_AUDIT_LOG_1030
			  WHERE SUBSTR(CREATE_TIME,1,7) = SUBSTR(DATE_ADD(CURRENT_DATE(),-1),1,7) 
			    AND STATUS = 1
            ) B
	ON A.USER_ID = B.USER_ID
  LEFT JOIN SDI.SDI_HD_PROVIDER_EMPLOYEE_1030 C
    ON A.EMP_ID = C.EMPLOYEE_ID
   AND C.EP_ID = 3044
   AND C.APP_ID = 1831
   AND C.ABLE_STATUS = 1
 GROUP BY REPLACE(A.POST_NAME,'装到家-','')
