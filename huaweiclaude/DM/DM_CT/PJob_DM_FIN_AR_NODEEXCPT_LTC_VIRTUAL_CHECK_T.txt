INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_VIRTUAL_CHECK_T
SELECT
       A.WBS
	  ,A.VIRTUAL_CHECK_ID
	  ,A.VIRTUAL_CHECK_CODE
	  ,B.PRODUCT_CHECK_STATUS --货物验收状态
	  ,CASE WHEN B.PRODUCT_CHECK_STATUS = 'WAIT_OVERDUE' THEN '待办逾期'
		    WHEN B.PRODUCT_CHECK_STATUS = 'WAIT_NOT_OVERDUE' THEN '待办未逾期'
		    WHEN B.PRODUCT_CHECK_STATUS = 'COMPLETE_OVERDUE' THEN '已办逾期'
			WHEN B.PRODUCT_CHECK_STATUS = 'COMPLETE_NOT_OVERDUE' THEN '已办未逾期'
			 END AS PRODUCT_CHECK_STATUS_NAME --货物验收状态描述
	  ,SUM(COALESCE(B.TOTAL_VIRTUAL_QTY,0)) AS  TOTAL_VIRTUAL_QTY
	  ,SUM(COALESCE(B.TOTAL_FORMAL_QTY,0)) AS TOTAL_FORMAL_QTY
	  ,SUM(COALESCE(B.TOTAL_VIRTUAL_AMOUNT,0)) AS TOTAL_VIRTUAL_AMOUNT
	  ,SUM(COALESCE(B.TOTAL_FORMAL_AMOUNT,0)) AS TOTAL_FORMAL_AMOUNT
	  ,SUM(COALESCE(B.TOTAL_VIRTUAL_QTY,0) - COALESCE(B.TOTAL_FORMAL_QTY,0)) AS DB_ACCEPT_QTY -- 待办验收数量 --验收数量
	  ,SUM(COALESCE(B.TOTAL_VIRTUAL_AMOUNT,0) - COALESCE(B.TOTAL_FORMAL_AMOUNT,0)) AS DB_ACCEPT_AMT -- 待办验收金额 -- 验收金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,B.MATERIAL_CODE --物料编码
      ,B.MATERIAL_NAME --物料描述
	  ,B.PICK_CODE AS DELIVERY_NO --提货单号
	  ,B.DEMAND_CODE --要货单号	  
  FROM SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_1033 A
  LEFT JOIN SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_DETAIL_1033 B
    ON A.VIRTUAL_CHECK_ID = B.VIRTUAL_CHECK_ID
   AND A.DELETE_FLAG = 0    
 WHERE B.DELETE_FLAG = 0    
 GROUP BY 
       A.WBS
	  ,A.VIRTUAL_CHECK_ID
	  ,A.VIRTUAL_CHECK_CODE
	  ,B.PRODUCT_CHECK_STATUS --货物验收状态
	  ,CASE WHEN B.PRODUCT_CHECK_STATUS = 'WAIT_OVERDUE' THEN '待办逾期'
		    WHEN B.PRODUCT_CHECK_STATUS = 'WAIT_NOT_OVERDUE' THEN '待办未逾期'
		    WHEN B.PRODUCT_CHECK_STATUS = 'COMPLETE_OVERDUE' THEN '已办逾期'
			WHEN B.PRODUCT_CHECK_STATUS = 'COMPLETE_NOT_OVERDUE' THEN '已办未逾期'
			 END --货物验收状态描述
	  ,B.MATERIAL_CODE --物料编码
      ,B.MATERIAL_NAME --物料描述
	  ,B.PICK_CODE --提货单号
	  ,B.DEMAND_CODE --要货单号
