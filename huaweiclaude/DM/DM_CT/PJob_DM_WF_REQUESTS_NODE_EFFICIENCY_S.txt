--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH T1 AS(
SELECT A.STAT_MONTH
      ,A.L1LC
	  ,A.WF_TYPE_NAME
	  ,A.WF_CODE
	  ,A.WF_NAME
	  ,SUM(A.NODE_COUNT) AS WF_NODE_COUNT
	  ,SUM(CASE WHEN CURRENT_NODE_TYPE = '归档' THEN NODE_COUNT ELSE 0 END) AS GD_NODE_COUNT
	  ,ROUND(SUM(CASE WHEN CURRENT_NODE_TYPE = '归档' THEN A.GZ_HOUR ELSE 0 END)/
	   SUM(CASE WHEN CURRENT_NODE_TYPE = '归档' THEN NODE_COUNT ELSE 0 END),2) AS AVG_WF_HOUR --流程平均耗时（工作时间：小时）
  FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_T A
 WHERE STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
   AND SUBSTR(RECEIVE_DATE,1,7) = SUBSTR('${PERIOD_ID_D}',1,7)
 GROUP BY
       A.STAT_MONTH
      ,A.L1LC
	  ,A.WF_TYPE_NAME
	  ,A.WF_CODE
	  ,A.WF_NAME
), T2 AS(
SELECT A.STAT_MONTH
      ,A.L1LC
	  ,A.WF_TYPE_NAME
	  ,A.WF_CODE
	  ,A.WF_NAME
	  ,A.NODE_NAME
	  ,SUM(A.NODE_COUNT) AS NODE_COUNT
	  ,ROUND(SUM(A.GZ_HOUR)/SUM(A.NODE_COUNT),2) AS AVG_NODE_HOUR --节点平均耗时（工作时间：小时）
	  ,ROUND(SUM(A.HS_MINUTE)/SUM(A.NODE_COUNT),2) AS AVG_NODE_MINUTE --节点平均耗时（工作时间）
  FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_T A
 WHERE STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
   AND SUBSTR(RECEIVE_DATE,1,7) = SUBSTR('${PERIOD_ID_D}',1,7)
 GROUP BY
       A.STAT_MONTH
      ,A.L1LC
	  ,A.WF_TYPE_NAME
	  ,A.WF_CODE
	  ,A.WF_NAME
	  ,A.NODE_NAME
)
INSERT OVERWRITE TABLE DM.DM_WF_REQUESTS_NODE_EFFICIENCY_S PARTITION (STAT_MONTH)
SELECT
       A.L1LC
	  ,A.WF_TYPE_NAME
	  ,A.WF_CODE
	  ,A.WF_NAME
	  ,A.WF_NODE_COUNT
	  ,A.GD_NODE_COUNT
	  ,B.NODE_NAME
	  ,B.NODE_COUNT
	  ,B.AVG_NODE_HOUR --节点平均耗时（工作时间：小时）
	  ,CASE WHEN B.AVG_NODE_MINUTE >= 1440 
	        THEN CAST(B.AVG_NODE_MINUTE/1440 AS DECIMAL) || '天' || CAST(PMOD(B.AVG_NODE_MINUTE,1440)/60 AS DECIMAL) || '小时' || CAST(PMOD(PMOD(B.AVG_NODE_MINUTE,1440),60) AS DECIMAL) || '分钟'
			WHEN B.AVG_NODE_MINUTE < 1440 AND B.AVG_NODE_MINUTE >= 60
			THEN '0天'||CAST(B.AVG_NODE_MINUTE/60 AS DECIMAL) || '小时' || CAST(PMOD(B.AVG_NODE_MINUTE,60) AS DECIMAL) || '分钟'
			WHEN B.AVG_NODE_MINUTE < 60
			THEN '0天0小时'||CAST(B.AVG_NODE_MINUTE AS DECIMAL)||'分钟'
			 END AS AVG_NODE_MINUTE
	  ,A.AVG_WF_HOUR --流程平均耗时（工作时间：小时）
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,A.STAT_MONTH
  FROM T1 AS A
  LEFT JOIN T2 AS B
    ON A.WF_CODE = B.WF_CODE
   AND A.STAT_MONTH = B.STAT_MONTH
;
