WITH LOADING_CONTACT_DATA AS(  -- 提货人数据
	SELECT T2.EXTERNNO ,
		   T2.externlineno,
		   MAX(T1.LOADING_CONTACT) AS LOADING_CONTACT
		FROM SDI.SDI_TMS_DO_1005 T1
		LEFT JOIN SDI.SDI_TMS_DO_DETAIL_1005 T2
		ON T1.DO_NO = T2.DO_NO 
		AND T1.DEL_FLAG = 'N'
		AND T2.DEL_FLAG  = 'N'
		GROUP BY T2.EXTERNNO ,
		         T2.externlineno
),OSS_SALE_ORDER_DATA AS( --OSS订单数据
SELECT a.VBELN
      ,a.POSNR
	  ,b.OSS_VBELN AS ORDER_NUM
	  ,a.ID_ITEM AS ORDER_ITEM_NUM
	  ,a.KWMENG AS REQUIRE_QUANTITY
	  ,CASE WHEN COALESCE(d.BSTDK,'') NOT IN('','00000000') THEN d.BSTDK
	        ELSE c.BSTDK
			 END AS REQUIRE_DELIVERY_DATE
FROM SDI.SDI_VBAP_1001 a
INNER JOIN SDI.SDI_VBAK_1001 b
ON a.VBELN = b.VBELN
AND a.MANDT = b.MANDT 
AND a.DEL_FLAG = 'N'
AND b.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_VBKD_1001 c
ON a.VBELN = c.VBELN
AND a.POSNR = c.POSNR 
AND a.MANDT = c.MANDT 
AND c.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_VBKD_1001 d
ON a.VBELN = d.VBELN
AND CAST(d.POSNR AS int) = 0
AND a.MANDT = d.MANDT 
AND d.DEL_FLAG = 'N'
WHERE a.MANDT = 800
AND COALESCE(b.OSS_VBELN,'') <> ''
),PLAN_ORDER_DATA AS(   -- 计划订单数据
 SELECT A.POSNR AS SAP_ITEM_NUM
 ,D.POSID AS PLAN_ITEM_NUM
 ,B.POSNR AS SAP_PLAN_NUM
 ,E.POSID AS PLAN_NUM 
 ,C.POSNR AS SAP_WBS_NUM
 ,F.POSID AS WBS_NUM  
 ,G.post1 AS WBS_NAME_CN
 FROM SDI.SDI_PRHI_1001 A
 LEFT JOIN SDI.SDI_PRHI_1001 B
 ON A.UP = B.POSNR
 AND A.MANDT = B.MANDT
 AND B.DEL_FLAG = 'N'
 LEFT JOIN SDI.SDI_PRHI_1001 C
 ON B.UP = C.POSNR
 AND C.MANDT = B.MANDT
 AND C.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 D
 ON A.POSNR = D.PSPNR
 AND A.MANDT = D.MANDT
 AND D.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 E
 ON B.POSNR = E.PSPNR 
 AND E.MANDT = B.MANDT
 AND E.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 F
 ON C.POSNR = F.PSPNR  
 AND C.MANDT = F.MANDT
 AND F.DEL_FLAG = 'N' 
 LEFT JOIN SDI.sdi_prps_1001 G
 ON F.POSID = G.posid
 AND G.MANDT = '800'
 AND G.del_flag = 'N'
 WHERE A.MANDT = '800'
 AND A.DEL_FLAG = 'N'
),
DELIVERY_NO_DATA AS(
SELECT A2.DELIVERY AS DELIVERY_NO--提货单号
      ,A1.DELIVERY_PROJECT --交货项目
	  ,A1.REFERENCE_BILL_NUM AS SALE_ORDER_NUM -- SAP订单头
	  ,A1.REFERENCE_PROJ_NUM AS SALE_ORDER_ITEM_NUM -- SAP订单行
	  ,T2.ORDER_NUM AS OSS_ORDER_NUM -- OSS订单号
	  ,T5.plan_num AS PLAN_ORDER_NUM -- 计划订单头
	  ,T5.plan_item_num AS PLAN_ORDER_ITEM_NUM -- 计划订单行号
	  ,T5.WBS_NUM   -- WBS编码
      ,T5.WBS_NAME_CN  -- 项目名称
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.CREATE_DATE,'yyyyMMdd'),'yyyy-MM-dd') as CREATE_DATE --记录建立日期
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.DELIVERY_TIME,'yyyyMMdd'),'yyyy-MM-dd') as DELIVERY_TIME --交货时间
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.DELIVERY_DATE,'yyyyMMdd'),'yyyy-MM-dd') as DELIVERY_DATE --交货日期
	  ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.PLAN_GOODS_MOVE_DATE,'yyyyMMdd'),'yyyy-MM-dd') as PLAN_GOODS_MOVE_DATE --计划发货日期
      ,A2.SHIP_TERM_CODE --装运条件
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.VOUCHER_DATE,'yyyyMMdd'),'yyyy-MM-dd') as VOUCHER_DATE --凭证中的凭证日期
	  ,A2.FACTORY_ID AS DELIVERY_FACTORY_ID   -- 交货的接收工厂ID
      ,A2.FACTORY_CODE AS DELIVERY_FACTORY_CODE --交货的接收工厂编码
	  ,'' AS DELIVERY_FACTORY_NAME -- 交货的接收工厂
	  ,t2.REQUIRE_QUANTITY  -- 需求数量
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(t2.REQUIRE_DELIVERY_DATE,'yyyyMMdd'),'yyyy-MM-dd') as REQUIRE_DELIVERY_DATE -- 需求时间
      ,A2.SHIPPING_TYPE --装运类型
	  ,A2.SALES_DIS_VCH_TYPE -- 销售和分销凭证类别
	  ,'' as SALES_DIS_VCH_TYPE_NAME -- 销售和分销凭证类别名称
	  ,A2.CUST_SAL_ID -- 售达方ID
      ,A2.CUST_SAL_NUM --售达方编码
	  ,'' as CUST_SAL_NAME -- 售达方名称
      ,A2.SHIPMENT_PLAN_STATUS --运输计划状态	  
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.CON_TIME,'yyyyMMdd'),'yyyy-MM-dd') as CON_TIME --确认时间
      ,A2.GROSS_WGHT --总重量
	  ,A2.CUST_DEL_ID-- 送达方ID
      ,A2.CUST_DEL_NUM --送达方编码
	  ,'' as CUST_DEL_NAME -- 送达方名称
      ,A2.WHOLE_FORZEN_STATUS --总体冻结的状态
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.BILL_DATE,'yyyyMMdd'),'yyyy-MM-dd') as BILL_DATE --开票日期
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.COMPANY_BILL_DATE,'yyyyMMdd'),'yyyy-MM-dd') as COMPANY_BILL_DATE --公司间开票的开票日期
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.LOAD_DATE,'yyyyMMdd'),'yyyy-MM-dd') as LOAD_DATE --装载日期
      ,A2.SHIP_GROUP --运输组
	  ,'' as SHIP_GROUP_NAME -- 运输组名称
      ,A2.PROJ_VOL --业务量
      ,A2.COMPANY_BILL_SALES_ORG --公司间开票的销售组织
      ,A2.SALES_ORG_CODE --销售组织
	  ,'' as SALES_ORG_NAME -- 销售组织名称
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.TRNSP_PLAN_DATE,'yyyyMMdd'),'yyyy-MM-dd') as TRNSP_PLAN_DATE --运输计划日期
      ,A2.CUST_GROUP --客户组
	  ,'' as CUST_GROUP_NAME -- 客户组名称
      ,A2.SHIP_TYPE_FLAG --运输方式标识
      ,A2.PROJ_NET_WEIGHT --净重量
      ,A2.WEIGHT_UNIT --重量单位
      ,A2.SHIP_TYPE	 --运输方式
	  ,A2.WHOLE_VCH_DEAL_STATUS -- 凭证的总体处理状态
	  ,'' as WHOLE_VCH_DEAL_STATUS_NAME -- 凭证的总体处理状态名称
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.MOVE_PLACE_DATE,'yyyyMMdd'),'yyyy-MM-dd') as MOVE_PLACE_DATE --移交地点的移交日期
      ,A2.DELIVERY_TYPE --交货类型
      ,A2.TITLE_ROUTINE_STATUS --抬头的常规不完整状态
      ,A2.ORDER_COMB_FLAG --订单组合标识
      ,A2.TOTAL_PICKING AS OPS_STATUS --OPS状态
	  ,CASE WHEN A2.TOTAL_PICKING = 'A' THEN '没有处理'
	        WHEN A2.TOTAL_PICKING = 'B' THEN '部分处理'
			WHEN A2.TOTAL_PICKING = 'C' THEN '完全地处理'
	  END OPS_STATUS_NAME -- OPS状态名称
      ,A2.WAREHOUSE_MANG_ACTION_STATUS AS VM_STATUS --VM状态
	  ,CASE WHEN A2.WAREHOUSE_MANG_ACTION_STATUS = 'A' THEN '没有处理'
	        WHEN A2.WAREHOUSE_MANG_ACTION_STATUS = 'B' THEN '部分处理'
			WHEN A2.WAREHOUSE_MANG_ACTION_STATUS = 'C' THEN '完全地处理'
	  END VM_STATUS_NAME -- VM状态名称
      ,A2.PICKING_CONFIRM_STATUS --拣配确认状态
	  ,CASE WHEN A2.PICKING_CONFIRM_STATUS = 'A' THEN '没有处理'
	        WHEN A2.PICKING_CONFIRM_STATUS = 'B' THEN '部分处理'
			WHEN A2.PICKING_CONFIRM_STATUS = 'C' THEN '完全地处理'
	  END PICKING_CONFIRM_STATUS_NAME -- 拣配确认状态名称
      ,A2.TOTAL_GOODS_MOVE_STATUS AS DELIVERY_NO_STATUS--提货单状态(货物移动)
	  ,CASE WHEN A2.TOTAL_GOODS_MOVE_STATUS = 'A' THEN '没有处理'
	        WHEN A2.TOTAL_GOODS_MOVE_STATUS = 'B' THEN '部分处理'
			WHEN A2.TOTAL_GOODS_MOVE_STATUS = 'C' THEN '完全地处理'
	  END DELIVERY_NO_STATUS_NAME -- 提货单状态名称
	  ,FROM_UNIXTIME(UNIX_TIMESTAMP(A2.ACTUAL_GOODS_MOVE_DATE,'yyyyMMdd'),'yyyy-MM-dd') as ACTUAL_GOODS_MOVE_DATE --提货时间(实际发货日期)
	  ,A1.ACTUAL_DELIVERY_QTY --提货数量(实际已交货量)
	  ,T1.LOADING_CONTACT -- 提货人
      ,A2.ISSUE_INVOICE_STATUS --开票状态
      ,A2.COMPANY_BILL_TOTAL_STATUS --总体状态
	  ,A1.MATERIAL_ID -- 物料ID
	  ,A1.MATERIAL_NUM --物料编号
	  ,A1.MATERIAL_NAME -- 物料名称
	  ,A1.MATERIAL_GROUP -- 物料组
	  ,A1.MATERIAL_GROUP_NAME -- 物料组名称
	  ,T3.LENGTH_1 --长度
	  ,T3.WIDTH --宽度
	  ,T3.DIMENSION --规格
	  ,T3.EXTERNAL_MATL_GROUP_CODE --外部物料组
	  ,T3.PRODUCT_LEVEL_CODE --产品层次
	  ,T3.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道编码
	  ,T4.CONFIG_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称
	  ,A1.FACTORY_ID -- 工厂ID
      ,A1.FACTORY_CODE --工厂编码
	  ,A1.FACTORY_NAME -- 工厂名称
      ,A1.COMPONENT_QTY --组件数量
      ,A1.SO_ITEM_TEXT --销售订单项目的短文本
      ,A1.BASE_MEASURE_UNIT --基本计量单位
      ,A1.GOODS_MOVE_REASON --货物移动原因
      ,A1.PROJ_TYPE --项目类型
	  ,A1.PROJ_TYPE_NAME -- 项目类型名称
	  ,A1.BATCH_NUM --批号
      ,A1.SPECIAL_INV_FLAG --特殊库存标识
	  ,A1.SPECIAL_INV_FLAG_NAME -- 特殊库存标识名称
      ,A1.STORAGE_TYPE --仓储类型
      ,A1.DELIVERY_PROJ_ORIGINAL_QTY --交货项目的原始数量
      ,A1.ACCEPT_DELIVERY_WBS --接受/发货(WBS要素)
      ,A1.ASSESSMENT_TYPE --评估类型
      ,A1.PROFIT_CENTER --利润中心
	  ,A1.PROFIT_CENTER_NAME -- 利润中心名称
      ,A1.KC_VOL_UNIT --体积单位
      ,A1.VOL_UNIT --体积单位
      ,A1.INVENTORY_STORE_PLACE --库存仓位
      ,A1.TOTAL_REQ --需求总计
      ,A1.INV_LOCATION --库存地点
      ,A1.IN_WBS_ELEMENT --项目（内部）
      ,A1.DIS_CHANNEL --分销渠道
	  ,A1.DIS_CHANNEL_NAME -- 分销渠道名称
      ,A1.MATERIAL_SHIP_GROUP --物料运输组
	  ,A1.MATERIAL_SHIP_GROUP_NAME --物料运输组名称
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A1.PRODUCTION_DATE,'yyyyMMdd'),'yyyy-MM-dd') as PRODUCTION_DATE --生产日期
      ,A1.SAL_GROUP --销售组
      ,A1.MOVE_TYPE --移动类型(库存管理)
      ,A1.MOVE_FLAG --移动标识
	  ,A1.BATCH_SPLIT_TOTAL_VOL --所有批分裂项目的累计净重量
      ,A1.KC_WEIGHT_UNIT --重量单位
	  ,A1.PICKING_INTO_STATUS --拣配状态
	  ,CASE WHEN A1.PICKING_INTO_STATUS = 'A' THEN '没有处理'
	        WHEN A1.PICKING_INTO_STATUS = 'B' THEN '部分处理'
			WHEN A1.PICKING_INTO_STATUS = 'C' THEN '完全地处理'
	  END PICKING_INTO_STATUS_NAME -- 拣配状态名称
	  ,A1.GOODS_MOVE_STATUS --货物移动
	  ,A1.GOODS_MOVE_STATUS_NAME --货物移动名称
	  ,A1.DELIVERY_BILL_STATUS --开票状态
	  ,CASE WHEN A1.DELIVERY_BILL_STATUS = 'A' THEN '没有处理'
	        WHEN A1.DELIVERY_BILL_STATUS = 'B' THEN '部分处理'
			WHEN A1.DELIVERY_BILL_STATUS = 'C' THEN '完全地处理'
	   END DELIVERY_BILL_STATUS_NAME -- 开票状态名称
	  ,A1.COMPANY_BILL_STATUS --IBS状态
	  ,CASE WHEN A1.COMPANY_BILL_STATUS = 'A' THEN '没有处理'
	        WHEN A1.COMPANY_BILL_STATUS = 'B' THEN '部分处理'
			WHEN A1.COMPANY_BILL_STATUS = 'C' THEN '完全地处理'
	   END COMPANY_BILL_STATUS_NAME -- IBS状态名称
	  ,A3.SORTL AS CUST_SHORT_NAME--客户简写
	  ,A3.STRAS AS STREET_HOUSE_NUM --街道和房屋号
	  ,A3.PSTLZ AS ZIP_CODE--邮政编码
	  ,A3.ORT01 AS CITY_NAME--城市
	  ,A3.SORTL||' / '||A3.STRAS||A3.PSTLZ||' / '||A3.ORT01 AS CUST_ADDRESS  -- 客户地址
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
	  ,T2.ORDER_ITEM_NUM AS OSS_ITEM_NUM --OSS销售订单行号
FROM DWROTD.DWR_WM_LADINGBILL_ITEM_F A1   -- SAP提货单行表
LEFT JOIN DWIOTD.dwi_wm_ladingbill_header_t A2  -- SAP提货单头表
ON A2.DELIVERY = A1.DELIVERY
AND A2.DEL_FLAG = 'N'
AND A1.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_KNA1_1001 A3  -- 客户地址信息
ON A2.CUST_DEL_NUM = A3.KUNNR
AND A3.DEL_FLAG = 'N'
LEFT JOIN LOADING_CONTACT_DATA T1  -- 提货人
ON A2.DELIVERY = T1.EXTERNNO
AND A1.DELIVERY_PROJECT = T1.externlineno
left join OSS_SALE_ORDER_DATA t2   -- LTC销售订单表_OSS
on A1.REFERENCE_BILL_NUM = T2.VBELN
AND A1.REFERENCE_PROJ_NUM = t2.POSNR
LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D t3 -- 物料
ON A1.MATERIAL_NUM = t3.MATERIAL_NUM
AND t3.END_DATE = '4712-12-31'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R t4 -- 配置表
ON t3.RESERVED_FIELD_1 = t4.SRC_SYS_CONFIG_CODE
AND t4.SOURCE_CODE = 'ZTYL01'
LEFT JOIN PLAN_ORDER_DATA T5   -- 计划订单数据
ON A1.WBS_ELEMENT =  T5.SAP_ITEM_NUM 
)
