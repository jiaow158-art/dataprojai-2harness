INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_DEMAND_T
SELECT A.WBS
      ,DDH.DEMAND_CODE --要货单号
	  ,DDH.CREATION_DATE AS DELIVER_CREATION_DATE --要货单创建日期
	  ,DDH.DEMAND_STATUS --要货单状态
	  ,DDH.RECOVE_FLAG -- 要货单是否回收
	  ,C.ORDERS_PUSH_METHOD --要货单推送方式
	  ,B.MEANING AS ORDERS_PUSH_METHOD_NAME --要货单推送方式描述
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,D.MEANING AS DEMAND_STATUS_NAME --要货单状态描述
  FROM SDI.SDI_DELIVER_DEMAND_HEADER_1033 DDH
  LEFT JOIN SDI.SDI_BUSINESS_OPPORTUNITY_1033 A
    ON DDH.BUSINESS_ID = A.BUSINESS_ID
   AND A.DELETE_FLAG = 0	
  LEFT JOIN (SELECT DISTINCT WBS
                            ,ORDERS_PUSH_METHOD
               FROM SDI.SDI_CONTRACT_1033
			  WHERE COALESCE(ORDERS_PUSH_METHOD,'') <> ''
                AND DELETE_FLAG = 0			  
			) C
	ON A.WBS = C.WBS
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B
    ON C.ORDERS_PUSH_METHOD = B.LOOKUP_CODE
   AND B.LOOKUP_TYPE = 'ORDER_PUSH_METHOD'
   AND B.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 D
    ON DDH.DEMAND_STATUS = D.LOOKUP_CODE
   AND D.LOOKUP_TYPE = 'DEMAND_STATUS'
   AND D.ENABLED_FLAG = 'Y' 
 WHERE DDH.DELETE_FLAG = 0  
;
