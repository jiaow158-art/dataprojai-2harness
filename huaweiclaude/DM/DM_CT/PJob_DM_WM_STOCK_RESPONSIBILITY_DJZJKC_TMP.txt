INSERT OVERWRITE TABLE  DM.DM_WM_STOCK_RESPONSIBILITY_DJZJKC_TMP
SELECT A.MATERIAL_NUM,
       A.BATCH_NUM,
       A.FACTORY_CODE,
       A.INV_LOCATION ,
       '冻结+质检' AS ITEM_TYPE,
       LEFT_DJZJKC AS STOCK_QTY,
	   current_timestamp() as dw_last_update_date
FROM  DM.DM_WM_STOCK_RESPONSIBILITY_LEFT_2_TMP A  
WHERE LEFT_DJZJKC >0;
