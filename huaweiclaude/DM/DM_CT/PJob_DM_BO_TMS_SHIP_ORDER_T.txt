WITH T AS(
SELECT DISTINCT
       A.DELV_NUM --发运单号
      ,B.DMAND_NUM --要货单号
	  ,A.DELV_LINE_NUM --提货单行号
	  ,A.DMAND_LINE_ID --要货单行号
  FROM DWRLTC.DWR_BO_DSPT_ORDER_F A
  LEFT JOIN DWRLTC.DWR_BO_DMAND_ORDER_F B
    ON A.DMAND_ID = B.DMAND_ID
)
INSERT OVERWRITE TABLE DM.DM_BO_TMS_SHIP_ORDER_T
SELECT A.DO_NUM --运单号
      ,A.ST_NAME --状态
	  ,A.HNDL_TIME --运单处理时间
	  ,A.PBLSH_TIME --承运商接单时间
	  ,A.DSPCH_TIME --派车时间
	  ,A.CARR_NAME --承运商
	  ,A.LOAD_SITE_NAME	--提货站点
	  ,A.ACTL_DELV_QTY AS SIGN_QTY --物流签收数量
	  ,B.RCPT_ST_NAME --签收状态
	  ,CASE WHEN A.LGSTCS_TYPE_NAME = '自提' THEN '自提'
	        ELSE '配送'
			 END AS PICK_TYPE_NAME --提货方式
	  ,A.DSPT_NUM --发运单号
	  ,A.DRVR_NAME --提货司机
	  ,A.DRVR_PHONE_NUM --司机电话
	  ,A.DRVR_CARD_NUM --证件号
	  ,A.DRVR_CAR_NUM --车牌号
	  ,A.DELV_PROV_NAME	--收货省
      ,A.DELV_CITY_NAME	--收货市
      ,A.DELV_CNTY_NAME	--收货县
      ,A.DELV_DIST_NAME	--收货区
	  ,A.DELV_ADDR --详细地址
	  ,A.SIGN_TIME --签收时间
	  ,T.DMAND_NUM --要货单号
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.SIGN_TIME) - UNIX_TIMESTAMP(C.LEAVE_TIME)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.SIGN_TIME) - UNIX_TIMESTAMP(C.LEAVE_TIME)),86400)) / 3600,0) || '小时','.0','') AS LGSTCS_CNSUM_TIME --物流耗时 物流签收时间-离场时间
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.SAP_ORDER_NUM --SAP订单号
	  ,A.SAP_ORDER_LINE_NUM --SAP订单行号
	  ,T.DMAND_LINE_ID --要货单行号
	  ,CASE WHEN T.DMAND_NUM IS NOT NULL THEN '自营工程' ELSE '零售' END ORDER_CAT_NAME
  FROM DWROTD.DWR_TRA_TMS_SHIP_DTL_F A
  LEFT JOIN(SELECT DISTINCT
                   DO_NUM
                  ,RCPT_ST_NAME
              FROM DWROTD.DWR_TRA_RCPT_ORDER_F
		   ) B
	ON A.DO_NUM = B.DO_NUM
  LEFT JOIN T
    ON A.DELV_NUM = T.DELV_NUM
   AND A.DELV_LINE_NUM = T.DELV_LINE_NUM
  LEFT JOIN(SELECT DISTINCT 
                   LEAVE_TIME --离场时间
                  ,DELIVERY_ORDER_NUMBER
              FROM DWROTD.DWR_TM_RESERVATION_RECORD_DETAIL_F
             WHERE COALESCE(RESERVATION_STATE,'') NOT IN('CANCELED','EXIT')
		   ) C
    ON A.DELV_NUM = C.DELIVERY_ORDER_NUMBER
