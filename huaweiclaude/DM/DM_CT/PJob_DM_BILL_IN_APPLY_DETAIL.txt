CREATE TABLE IF NOT EXISTS  DM.DM_BILL_IN_APPLY_DETAIL (
  invisible_code string COMMENT '隐形码',
  apply_code string COMMENT '入库单号',
  storage_in_time timestamp COMMENT '入库时间',
  product_code string COMMENT '产品编码',
  batch_number string COMMENT '批次',
  base string COMMENT '填报表基地',
  production_line string COMMENT '填报表产线',
  box_code string COMMENT '箱码',
  warehouse_id string COMMENT '库存地点',
  actual_num int COMMENT '入库数量',
  CREATE_TIME timestamp COMMENT '记录创建日期',
  dw_last_update_date timestamp COMMENT '最后更新时间',
  del_flag string COMMENT '删除标识'
)
PARTITIONED BY (period_id_m string);

INSERT OVERWRITE TABLE DM.DM_BILL_IN_APPLY_DETAIL
SELECT
t1.invisible_code as invisible_code, --隐形码
t1.storage_in_id,     --入库单号
t1.storage_in_time,   --入库时间
t1.product_code,--产品编码
t1.batch,   --批次
t2.Base, --填报表'基地'
t2.Production_line,           --填报表'产线'
t1.box_code,       --箱码
t1.warehouse_id,  --库存地点
1 as actual_num ,  --入库数量
t1.create_time as create_time,--创建时间
CURRENT_TIMESTAMP() as dw_last_update_date,--最后更新时间
'N' AS del_flag,
substring(t1.create_time,1,7) as period_id_m
FROM
--SDI.sdi_wms_bill_in_apply_detail_1040 t1
--JOIN SDI.SDI_WMS_BILL_IN_APPLY_1040 t2  SDI_CODE_RELATION_1042
--ON t1.apply_id = t2.ID
--JOIN
SDI.SDI_CODE_RELATION_1042 t1
LEFT JOIN dm.dm_stock_fill t2
ON substr(t1.invisible_code, 1, 2) = t2.invisible_code
where   t1.invisible_code <> ''
AND storage_in_id IS NOT NULL  and DATE(t1.CREATE_TIME ) >= '2023-07-19'
