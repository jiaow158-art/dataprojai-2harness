WITH DWR_DIM_PRODUCT_AREA_D_DATA AS(  -- 产区数据
  select  product_factory_code,
        final_area_name 
	from DWRDIM.DWR_DIM_PRODUCT_AREA_D 
	where product_factory_code is not null 
	group by product_factory_code,
			 final_area_name 
),
DWI_MD_DATA_MATERIAL_GENERAL_T_DATA AS(  -- 物料数据
   select 
		B.material_num,
		B.product_level2_name, -- as 品类,
		b.length_1 ,b.width 
	from DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T b
	where B.matl_group_code = '1000010'
)
INSERT OVERWRITE TABLE DM.DM_FIN_COST_RETURN_T
select 
	concat(A.lfgja,A.lfmon) as months,-- 年月
	werks as factory_code,-- 工厂
	A.ktext as factory_name,-- 工厂描述
	A.plnbez as material_num,-- 物料编码
	A.maktx as material_name,-- 物料描述
	A.bstmg as order_OUTPUT,-- 订单产量
	A.meins as UNIT,-- 单位
	A.zcb_sum as COST_TOTAL, --成本合计
	A.zcb_peinh as COST_PIRCIE,--成本单价
	A.groes as SPECIFICATIONS,--规格
	B.final_area_name , -- 产区
	C.product_level2_name, -- 品类
	case when  substr(werks,1,2) ='13' then '清远基地'
		 when  substr(werks,1,2) ='15' then '江西基地'
		 when  substr(werks,1,2) ='17' then '佛山基地'
		 when  substr(werks,1,2) ='16' then '湖南基地'
		 when  substr(werks,1,2) ='12' then '山东基地'
		 when  substr(werks,1,2) ='18' then '重庆基地'
		 when  substr(werks,1,2) ='19' then '山西基地'
		else '' end as BASE_NAME, --基地
	sum(A.zcl_sn) as CLNS_AMT,-- 金额_材料沙泥
	sum(A.zsn_sl) as CLNS_QTY, -- 数量_材料沙泥
	sum(A.zcl_hg) as CLHG_AMT,-- 金额_材料化工
	sum(A.zhg_sl) as CLHG_QTY,-- 数量_材料化工
	sum(A.zcl_bz) as CLBZ_AMT, -- 金额_材料包装
	sum(A.zbz_sl) as CLBZ_QTY, -- 数量_材料包装
	sum(A.zcl_qt) as CLQT_AMT, -- 金额_材料其它
	sum(A.zqt_sl) as CLQT_QTY, -- 数量_材料其它
	sum(A.zbcp) as BCP_AMT, -- 金额_半成品
	sum(A.zbcp_sl) as BCP_QTY, -- 数据_半成品
	sum(A.zccp) as CCP_AMT,-- 金额_产成品
	sum(A.zccp_sl) as CCP_QTY, -- 数量_产成品
	sum(A.zfcp) as FCP_AMT, -- 金额_副产品,
	sum(A.zfcp_sl) as FCP_QTY , -- 数量_副产品
	sum(A.zfghy) as FGHY_AMT , -- 金额_返工耗用,
	sum(A.zfghy_sl) as FGHY_QTY, -- 数量_返工耗用
	sum(A.zrgf) as RGF_AMT, -- 金额_人工费
	sum(A.zrgf_sl) as RGF_QTY, -- 数量_人工费
	sum(A.zrl_m) as RLM_AMT , -- 金额_燃料煤
	sum(A.zm_sl) as RLM_QTY, -- 数量_燃料煤
	sum(A.zrl_q) as RLQ_AMT,-- 金额_燃料气
	sum(A.zq_sl) as RLQ_QTY, -- 数量_燃料气
	sum(A.zdf) as DF_AMT, -- 金额_电费
	sum(A.zdf_sl) as DF_QTY, -- 数量_电费
	sum(A.zzjfx) as ZJJTX_AMT, -- 金额_折旧及摊销
	sum(A.zzjfx_sl) as ZJJTX_QTY, -- 数量_折旧及摊销
	sum(A.zww_pg) as WWJGPGF_AMT, -- 金额_委外加工抛光费
	sum(A.zww_sl) as WWJGPGF_QTY, -- 数量_委外加工抛光费
	sum(A.zsf) as SF_AMT, -- 金额_水费
	sum(A.zsf_sl) as SF_QTY, -- 数量_水费
	sum(A.zwlxh) as WLXH_AMT, -- 金额_物料消耗
	sum(A.zwlxh_sl) as WLXH_QTY, -- 数量_物料消耗
	sum(A.zwxf) as WXF_AMT, -- 金额_维修费
	sum(A.zwxf_sl) as WXF_QTY, -- 数量_维修费
	sum(A.zcy) as CY_AMT, -- 金额_柴油
	sum(A.zcy_sl) as CY_QTY, -- 数量_柴油
	sum(A.zwscl) as WSCL_AMT, -- 金额_污水处理
	sum(A.zwscl_sl) as WSCL_QTY, -- 数量_污水处理
	sum(A.zqtfy) as QTFY_AMT, -- 金额_其他费用
	sum(A.zcbcy) as QTFY_QTY , -- 数量_其他费用
	CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	A.bstmg * c.width * c.length_1/1000/1000 AS order_area
from DWIFIN.DWI_COST_RESTORE_CURRENT_BALANCE_T A
LEFT JOIN DWR_DIM_PRODUCT_AREA_D_DATA B
ON SUBSTR(A.WERKS,1,2) = SUBSTR(B.PRODUCT_FACTORY_CODE,1,2)
LEFT JOIN DWI_MD_DATA_MATERIAL_GENERAL_T_DATA C
on A.plnbez = C.material_num 
GROUP BY A.lfgja,
		 A.lfmon,
		 A.werks,
		 A.ktext,
		 A.plnbez,
		 A.maktx,
		 A.bstmg,
		 A.meins,
		 A.zcb_sum,
		 A.zcb_peinh,
		 A.groes,
		 B.final_area_name,
		 C.product_level2_name,
		 case when  substr(werks,1,2) ='13' then '清远基地'
		 when  substr(werks,1,2) ='15' then '江西基地'
		 when  substr(werks,1,2) ='17' then '佛山基地'
		 when  substr(werks,1,2) ='16' then '湖南基地'
		 when  substr(werks,1,2) ='12' then '山东基地'
		 when  substr(werks,1,2) ='18' then '重庆基地'
		 when  substr(werks,1,2) ='19' then '山西基地'
		else '' end,
		 A.bstmg * C.width * C.length_1/1000/1000;
