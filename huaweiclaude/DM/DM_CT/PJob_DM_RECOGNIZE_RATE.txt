with TMP AS (
SELECT 
T1.PACK_date AS RECOGNIZE_DATE,
T2.Base AS BASE,
T2.Production_line AS Production_line,
COUNT(T1.PACK_code) - COUNT(CASE WHEN T1.PACK_code = 'NULL' THEN 1 ELSE NULL END) AS ALLPACK_QTY, --打包总数
COUNT(T1.PACK_code) - COUNT(CASE WHEN T1.PACK_code = 'NULL' OR T1.PACK_code = 'NOREAD' THEN 1 ELSE NULL END) AS PACK_RECOGNIZE_QTY,--打包识别数
COUNT(CASE WHEN T1.PACK_code = 'NOREAD' THEN 1 ELSE NULL END) AS PACK_UNRECOGNIZE_QTY,--打包未识别数
COUNT(CASE WHEN T1.PACK_code = 'NULL' THEN 1 ELSE NULL END) AS PACK_EMPTY_QTY, --打包空砖数
COUNT(T1.PACK_code) - COUNT(CASE WHEN T1.PACK_code = 'NULL' OR T1.PACK_code = 'NOREAD' THEN 1 ELSE NULL END)  - COUNT(DISTINCT T1.PACK_CODE) AS PACK_repeat_QTY, --打包重复数
0 as FRACTIONATION_QTY,  --分级总数
0 AS FRACTIONATION_RECOGNIZE_QTY,--分级识别数
0 AS FRACTIONATION_UNRECOGNIZE_QTY,--分级未识别数
0 AS FRACTIONATION_EMPTY_QTY,--分级空砖数
0 AS INKJET_QTY,    --喷墨总数
0 AS INKJET_RECOGNIZE_QTY,--喷墨识别数
0 AS INKJET_UNRECOGNIZE_QTY,--喷墨未识别数 
0 AS INKJET_EMPTY_QTY--喷墨空砖数
FROM SDI.SDI_PACK_RELATION3_1043 T1 
LEFT JOIN dm.dm_stock_fill T2
ON substr(t1.PACK_code, 1, 2) = t2.invisible_code
WHERE T1.PACK_date = '${PERIOD_ID_D}'
GROUP BY PACK_date,T2.Base,
T2.Production_line  

UNION ALL 
SELECT 
T1.FRACTIONATION_date AS RECOGNIZE_DATE,
T2.Base AS BASE,
T2.Production_line AS Production_line,
0 AS ALLPACK_QTY, --打包总数
0 AS PACK_RECOGNIZE_QTY, --打包识别数
0 AS PACK_UNRECOGNIZE_QTY, --打包未识别数
0 AS PACK_EMPTY_QTY, --打包空砖数
0 AS PACK_repeat_QTY, --打包重复数
COUNT(T1.FRACTIONATION_code) - COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' THEN 1 ELSE NULL END) AS FRACTIONATION_QTY, --分级总数
COUNT(T1.FRACTIONATION_code) - COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' OR T1.FRACTIONATION_code = 'NOREAD' THEN 1 ELSE NULL END) AS FRACTIONATION_RECOGNIZE_QTY, --分级识别数
COUNT(CASE WHEN T1.FRACTIONATION_code = 'NOREAD' THEN 1 ELSE NULL END) AS FRACTIONATION_UNRECOGNIZE_QTY, --分级未识别数
COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' THEN 1 ELSE NULL END) AS FRACTIONATION_EMPTY_QTY, --分级空砖数
0 AS INKJET_QTY, --喷墨总数
0 AS INKJET_RECOGNIZE_QTY, --喷墨识别数
0 AS INKJET_UNRECOGNIZE_QTY, --喷墨未识别数
0 AS INKJET_EMPTY_QTY --喷墨空砖数
FROM SDI.SDI_FRACTIONATION_RELATION_1043 T1
LEFT JOIN dm.dm_stock_fill T2
ON substr(t1.FRACTIONATION_code, 1, 2) = t2.invisible_code
WHERE  T1.FRACTIONATION_date = '${PERIOD_ID_D}'
GROUP BY 
T1.FRACTIONATION_date,T2.Base,
T2.Production_line
UNION ALL 
SELECT 
T1.INKJET_date AS RECOGNIZE_DATE, --识别日期
T2.Base AS BASE,
T2.Production_line AS Production_line,
0 AS ALLPACK_QTY, --打包总数
0 AS PACK_RECOGNIZE_QTY, --打包识别数
0 AS PACK_UNRECOGNIZE_QTY, --打包未识别数
0 AS PACK_EMPTY_QTY, --打包空砖数
0 AS PACK_repeat_QTY, --打包重复数
0 AS FRACTIONATION_QTY, --分级总数
0 AS FRACTIONATION_RECOGNIZE_QTY, --分级识别数
0 AS FRACTIONATION_UNRECOGNIZE_QTY, --分级未识别数
0 AS FRACTIONATION_EMPTY_QTY, --分级空砖数
COUNT(T1.INKJET_code) - COUNT(CASE WHEN T1.INKJET_code = 'NULL' THEN 1 ELSE NULL END) AS INKJET_QTY, --喷墨总数
COUNT(T1.INKJET_code) - COUNT(CASE WHEN T1.INKJET_code = 'NULL' OR T1.INKJET_code = 'NOREAD' THEN 1 ELSE NULL END) AS INKJET_RECOGNIZE_QTY, --喷墨识别数
COUNT(CASE WHEN T1.INKJET_code = 'NOREAD' THEN 1 ELSE NULL END) AS INKJET_UNRECOGNIZE_QTY, --喷墨未识别数
COUNT(CASE WHEN T1.INKJET_code = 'NULL' THEN 1 ELSE NULL END) AS INKJET_EMPTY_QTY --喷墨空砖数
FROM SDI.SDI_INKJET_RELATION_1043 T1
LEFT JOIN dm.dm_stock_fill T2
ON substr(t1.INKJET_code, 1, 2) = t2.invisible_code
WHERE T1.INKJET_date = '${PERIOD_ID_D}'
GROUP BY 
T1.INKJET_date,T2.Base,T2.Production_line
)

,TMP1  AS(
SELECT 
RECOGNIZE_DATE,
BASE,
Production_line,
SUM(ALLPACK_QTY) AS ALLPACK_QTY,--打包总数
SUM(PACK_RECOGNIZE_QTY) AS PACK_RECOGNIZE_QTY,--打包识别数
SUM(PACK_UNRECOGNIZE_QTY) AS PACK_UNRECOGNIZE_QTY ,--打包未识别数
SUM(PACK_EMPTY_QTY) AS PACK_EMPTY_QTY, --打包空砖数
SUM(PACK_repeat_QTY) AS PACK_repeat_QTY,--打包重复数
SUM(FRACTIONATION_QTY) AS FRACTIONATION_QTY,  --分级总数
SUM(FRACTIONATION_RECOGNIZE_QTY)  AS FRACTIONATION_RECOGNIZE_QTY,--分级识别数
SUM(FRACTIONATION_UNRECOGNIZE_QTY) AS FRACTIONATION_UNRECOGNIZE_QTY,--分级未识别数
SUM(FRACTIONATION_EMPTY_QTY) AS FRACTIONATION_EMPTY_QTY,--分级空砖数
SUM(INKJET_QTY) AS INKJET_QTY,    --喷墨总数
SUM(INKJET_RECOGNIZE_QTY ) AS INKJET_RECOGNIZE_QTY,--喷墨识别数
SUM(INKJET_UNRECOGNIZE_QTY ) AS INKJET_UNRECOGNIZE_QTY,--喷墨未识别数
SUM(INKJET_EMPTY_QTY) AS INKJET_EMPTY_QTY,--喷墨空砖数,
SUM(PACK_RECOGNIZE_QTY)/SUM(ALLPACK_QTY) AS PACK_RATE,--打包识别率
SUM(FRACTIONATION_RECOGNIZE_QTY)/SUM(FRACTIONATION_QTY) AS FRACTIONATION_RATE,--分级识别率
SUM(INKJET_RECOGNIZE_QTY)/SUM(INKJET_QTY) AS INKJET_RATE--喷墨识别率
FROM TMP 
GROUP BY RECOGNIZE_DATE ,BASE,
Production_line
)
,TMP2 AS (
     SELECT
        T1.PACK_DATE AS PACK_DATE,--识别日期
        COUNT( DISTINCT T2.INKJET_code)AS INKJET_ASCEND_QTY,--喷墨可追溯数量
        count( DISTINCT t3.FRACTIONATION_code)AS FRACTIONATION_ASCEND_QTY,--分级可追溯数量
        count( DISTINCT case when   T2.INKJET_code IS NOT NULL and t3.FRACTIONATION_code is not null then T1.PACK_code end )AS INKJET_FRACTIONATION_ASCEND_QTY--分级&喷墨追溯数
FROM
        (SELECT DISTINCT PACK_code as PACK_code, PACK_DATE
    FROM SDI.SDI_PACK_RELATION3_1043 )  T1
        LEFT JOIN (select distinct INKJET_code  from  SDI.SDI_INKJET_RELATION_1043 t2   
        where   T2.INKJET_DATE >= DATE_ADD( '${PERIOD_ID_D}', -2)
        AND T2.INKJET_DATE < DATE_ADD( '${PERIOD_ID_D}', 1)) T2 ON T1.PACK_code = T2.INKJET_code
        left join  (select distinct FRACTIONATION_code  from  SDI.SDI_FRACTIONATION_RELATION_1043 t2   
        where   T2.FRACTIONATION_DATE >= DATE_ADD( '${PERIOD_ID_D}', -2)
        AND T2.FRACTIONATION_DATE < DATE_ADD( '${PERIOD_ID_D}', 1)) t3 ON t1.PACK_code = t3.FRACTIONATION_code
WHERE 1=1
        AND T1.PACK_DATE = '${PERIOD_ID_D}' AND T1.PACK_CODE <> 'NULL' AND T1.PACK_CODE <>  'NOREAD'
        GROUP BY T1.PACK_DATE
 

)
,TMP3 AS (
SELECT 
T1.RECOGNIZE_DATE,--识别日期
T1.BASE, --基地
T1.Production_line,--产线
T1.ALLPACK_QTY,--打包总数
T1.PACK_RECOGNIZE_QTY,--打包识别数
T1.PACK_UNRECOGNIZE_QTY ,--打包未识别数
T1.PACK_EMPTY_QTY, --打包空砖数
T1.PACK_repeat_QTY,--打包重复数
T1.FRACTIONATION_QTY,  --分级总数
T1.FRACTIONATION_RECOGNIZE_QTY,--分级识别数
T1.FRACTIONATION_UNRECOGNIZE_QTY,--分级未识别数
T1.FRACTIONATION_EMPTY_QTY,--分级空砖数
T1.INKJET_QTY,    --喷墨总数
T1.INKJET_RECOGNIZE_QTY,--喷墨识别数
T1.INKJET_UNRECOGNIZE_QTY,--喷墨未识别数
T1.INKJET_EMPTY_QTY,--喷墨空砖数,
T1.PACK_RATE,--打包识别率
T1.FRACTIONATION_RATE,--分级识别率
T1.INKJET_RATE,--喷墨识别率
T2.INKJET_ASCEND_QTY,--喷墨可追溯数量
T2.FRACTIONATION_ASCEND_QTY,--分级可追溯数量
T2.INKJET_FRACTIONATION_ASCEND_QTY,--分级&喷墨追溯数
T2.INKJET_ASCEND_QTY/T1.ALLPACK_QTY AS INKJET_ASCEND_RATE,--打包喷墨追溯率
T2.FRACTIONATION_ASCEND_QTY/T1.ALLPACK_QTY AS FRACTIONATION_ASCEND_RATE,--打包分级追溯率
T2.INKJET_FRACTIONATION_ASCEND_QTY/T1.ALLPACK_QTY AS INKJET_FRACTIONATION_ASCEND_RATE--打包分级喷墨追溯率
FROM TMP1 T1
JOIN TMP2 T2
ON T1.RECOGNIZE_DATE = T2.PACK_DATE
)

Production_line,--产线
ALLPACK_QTY,--打包总数
PACK_RECOGNIZE_QTY,--打包识别数
PACK_UNRECOGNIZE_QTY ,--打包未识别数

INKJET_FRACTIONATION_ASCEND_RATE--打包分级喷墨追溯率
FROM TMP3
UNION ALL 
SELECT 
RECOGNIZE_DATE,--识别日期
BASE, --基地
Production_line,--产线
ALLPACK_QTY,--打包总数
PACK_RECOGNIZE_QTY,--打包识别数
PACK_UNRECOGNIZE_QTY ,--打包未识别数

FRACTIONATION_RECOGNIZE_QTY,--分级识别数
FRACTIONATION_UNRECOGNIZE_QTY,--分级未识别数
FRACTIONATION_EMPTY_QTY,--分级空砖数

FROM 
DM.DM_RECOGNIZE_RATE
WHERE substring(RECOGNIZE_DATE,1,10) <>'${PERIOD_ID_D}'
