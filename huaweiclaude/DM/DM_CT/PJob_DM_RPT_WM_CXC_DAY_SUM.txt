--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with mat as (

	select t1.material_num,t1.product_level2_name  as cat,t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        t1.width * t1.length_1/1000/1000 as area
  from dwimd.dwi_md_data_material_general_t t1 
  where matl_group_code ='1000010'
	and external_matl_group_code in('10010','10080','10060')
    and t1.product_level2_code  like '1%'
    and t1.product_level2_code  not in('1011','1012','1013','1016')
	and matl_type_code='FERT')
	,
mid1 as(
select 
       a.material_num ,a.batch_num, a.inv_location ,a.factory_werks_code ,a.factory_werks_name ,a.move_type ,
       row_number() over(partition by  a.material_num ,a.batch_num order by a.voucher_post_date,a.material_vch_no_mblnr  asc) rn
 from  dwrotd.dwr_wm_prodstock_out_in_f a  
 join mat t1 on a.material_num  = t1.material_num 
 ),
 tmp as(
select batch_start,factory_type,product_factory_code ,belong_area_name,
case when  b.factory_type='外协' then 'OEM'
                      when  b.product_factory_code like '12%' then '山东基地'
                      when b.product_factory_code like '13%' then '清远基地'
                      when b.product_factory_code like '15%' then '江西基地'
                      when b.product_factory_code like '16%' then '湖南基地'
                      when b.product_factory_code like '17%' then '华盛昌基地'
                      when b.product_factory_code like '18%' then '重庆基地'
                      when b.product_factory_code like '19%' then '山西基地' 
                   end factory,
row_number () over(partition by batch_start order by case when coalesce(factory_type,' ')=' ' then 1 else 0 end ) rn
from DWRDIM.DWR_DIM_PRODUCT_AREA_D b
),
batch_area as(
select a.material_num ,a.batch_num, a.inv_location ,a.factory_werks_code ,a.factory_werks_name ,a.move_type ,
     case when inv_location like 'BE%' OR  inv_location like 'YX%' then 'OEM' 
            when factory_werks_code like '12%' then '山东基地'
            when factory_werks_code like '13%' then '清远基地'
            when factory_werks_code like '15%' then '江西基地'
            when factory_werks_code like '16%' then '湖南基地'
            when factory_werks_code like '17%' then '华盛昌基地'
            when factory_werks_code like '18%' then '重庆基地'
            when factory_werks_code like '19%' then '山西基地'
            when factory_werks_code in('7520','7560','7580','7920','7980','7960','8A20','8A80','8A60','7720','7760','7780','7220','7260','7280') 
              then 'OEM' 
            when b.batch_start is not null  then b.factory
            when c.batch_start is not null  then case when c.batch_start in('0B','AB') then '清远基地' when  c.batch_start in('0Y') then '华盛昌基地' else c.factory end else 
            'OEM' end as factory
            from  mid1 as a
            left join tmp b on substr(a.batch_num,1,3) = b.batch_start and b.rn=1
            left join tmp c on substr(a.batch_num,1,2) = c.batch_start and c.rn=1
            where 1=1
            and a.rn=1
            ),
stock_chg_det as(
-- 出入库
select  t1.VOUCHER_POST_DATE,
		t1.material_num,  -- 关联品类、渠道
		t1.batch_num ,    -- 关联基地
		t2.cat,
		t2.dimension,
		t2.prod_channel,
		sum(case when MOVE_TYPE IN(
						 '601','602','653','654') then (case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end) end*t2.area) as sales_stock_out_area,
		sum(case when MOVE_TYPE IN(
						 '911','912','913','914','131','132','531','532' ) or (move_type IN('101','102','161') and t1.SPECIAL_STOCK_FLAG_2='K_T148T') then (case when DR_CR_ID='S_SHKZG' then QUANTITY else  -QUANTITY end*t2.area) end) as stock_in_area
from  dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join mat t2 
on t1.material_num  = t2.material_num 
where VOUCHER_POST_DATE>= concat(substr('${PERIOD_ID_D}',1,4),'-01-01') --date_trunc('year',date '${PERIOD_ID_D}') 
and  VOUCHER_POST_DATE< date_add('${PERIOD_ID_D}',1)
	and MOVE_TYPE IN(
					 '601','602','653','654',  -- 销售出库
					 '911','912','913','914','131','132','531','532','101','102','161'   -- 生产入库&采购入库
					 )
    group by t1.VOUCHER_POST_DATE ,t1.material_num,t1.batch_num,t2.cat,t2.dimension,t2.prod_channel
 ),
 stock_chg_sum as
 (
 select t2.factory as belong_area_name,
        t1.cat,t1.dimension,t1.prod_channel,
        sum(sales_stock_out_area) sales_stock_out_area_year,
        sum(stock_in_area) stock_in_area_year,
        sum(case when VOUCHER_POST_DATE>= concat(substr('${PERIOD_ID_D}',1,7),'-01') then sales_stock_out_area else 0 end) sales_stock_out_area_month,
        sum(case when VOUCHER_POST_DATE>=concat(substr('${PERIOD_ID_D}',1,7),'-01') then stock_in_area else 0 end ) stock_in_area_month,
        sum(case when VOUCHER_POST_DATE>='${PERIOD_ID_D}' then sales_stock_out_area else 0 end) sales_stock_out_area_day,
        sum(case when VOUCHER_POST_DATE>='${PERIOD_ID_D}' then stock_in_area else 0 end ) stock_in_area_day
 from stock_chg_det t1
 left join batch_area t2  on t1.material_num = t2.material_num and t1.batch_num  = t2.batch_num
 group by t2.factory,cat,dimension,prod_channel
 ),
 stock_det as (
 select t3.factory as belong_area_name,cat,dimension,prod_channel,
        sum(case when stock_flg='本月库存' then stock_qty end*t2.area ) stock_area_month_end,
        sum(case when stock_flg='月初库存' then stock_qty end*t2.area ) stock_area_month_start,
        sum(case when stock_flg='年初库存' then stock_qty end*t2.area ) stock_area_year_start
 from 
 (
 -- 月初库存、年初库存、本月库存
select batch_num,material_num,sum(stock_qty) as stock_qty, '本月库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr('${PERIOD_ID_D}',1,7),'-','')
group by  batch_num,material_num
 union all 
select batch_num,material_num,sum(stock_qty) as stock_qty,'月初库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,7),'-01'),1),1,7),'-','')
group by  batch_num,material_num
union all
 select batch_num,material_num,sum(stock_qty) as stock_qty,'年初库存' stock_flg  from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,4),'-01-01'),1),1,7),'-','')
group by  batch_num,material_num
) t1
join mat t2 on t1.material_num  = t2.material_num 
left join batch_area t3  on t1.material_num = t3.material_num and t1.batch_num  = t3.batch_num
group by t3.factory ,cat,dimension,prod_channel
),
mid as (
select  belong_area_name,cat,dimension,prod_channel,
         stock_area_month_end,
        stock_area_month_start,
        stock_area_year_start,
        0.0 as sales_stock_out_area_year,0.0 as stock_in_area_year,
        0.0 as sales_stock_out_area_month,0.0 as stock_in_area_month,
        0.0 as sales_stock_out_area_day,0.0 as stock_in_area_day
        from stock_det
union all 
select  belong_area_name,cat,dimension,prod_channel,
        0,0,0,
        sales_stock_out_area_year,stock_in_area_year,
        sales_stock_out_area_month,stock_in_area_month,
        sales_stock_out_area_day,stock_in_area_day
        from stock_chg_sum
        )
INSERT OVERWRITE TABLE DM.DM_RPT_WM_CXC_DAY_SUM PARTITION(stat_date)
 select 
        belong_area_name,  -- 产区
		cat,    -- 产品层级2
		dimension, -- 规格
		prod_channel, -- 产品渠道
        sum(stock_area_month_end) as stock_area_month_end,  -- 月末库存
        sum(stock_area_month_start) as stock_area_month_start, -- 月初库存
        sum(stock_area_year_start) as stock_area_year_start, -- 年初库存
        sum(sales_stock_out_area_year) as sales_stock_out_area_year,  -- 年累计销售出库
		sum(stock_in_area_year) as stock_in_area_year, -- 年累计入库（自产或外协）
        sum(sales_stock_out_area_month) as sales_stock_out_area_month,  -- 月累计销售出库
		sum(stock_in_area_month) as stock_in_area_month, -- 月累计入库 （自产或外协）
        sum(sales_stock_out_area_day) as sales_stock_out_area_day,  -- 日销售出库
		sum(stock_in_area_day) as stock_in_area_day, -- 日入库 （自产或外协）
		current_timestamp() as dw_last_update_date, -- 数据同步时间
		replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','') as stat_date --分区字段(按天)
 from mid 
 group by belong_area_name,cat,dimension,prod_channel;
