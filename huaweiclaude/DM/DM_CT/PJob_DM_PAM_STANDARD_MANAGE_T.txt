WITH tmp_data as(     --  数据字典数据
   select h.selectvalue,
          t.fieldname,
          h.SELECTNAME
 from SDI.SDI_workflow_billfield_1004 t , 
      SDI.SDI_workflow_selectitem_1004 h  
 WHERE  t.billid = '-1116' 
 and h.fieldid = t.id
),
tmp_data_01 as(
 SELECT T1.SCJD2     AS APPLY_BASE_CODE,           -- 申请基地编码
        T3.SELECTNAME    AS APPLY_BASE_NAME,           -- 申请基地名称
        T2.MX1_BZBBH  AS STANDARD_CODE,     -- 标准版编号
        T2.MX1_BZBPC  AS STANDARD_BATCH_CODE,    -- 标准版批次
        T2.MX1_BZBLX     AS STANDARD_TYPE_CODE,  -- 标准版类型编码
        T4.SELECTNAME    AS STANDARD_TYPE_NAME,  -- 标准版类型名称
        T1.ZTLX    AS STATUS_TYPE_CODE,   -- 状态类型编码
        T5.SELECTNAME   AS STATUS_TYPE_NAME,-- 状态类型名称
        T2.MX1_WJDW2  AS OUT_BASE_CODE, --外借基地编码
        T6.SELECTNAME AS OUT_BASE_NAME, -- 外借基地名称
        NVL(T2.MX1_KYSL,0) AS USABLE_QTY,  -- 可用数量
        CASE WHEN T5.SELECTNAME = '存贮' THEN
           NVL(t2.MX1_KYSL,0) 
         ELSE 0
        END AS begin_QTY ,  -- 期初数量
        CASE WHEN T5.SELECTNAME = '借入' THEN
           NVL(t2.MX1_KYSL,0) 
         ELSE 0
        END AS borrow_QTY   -- 借入数量
 FROM SDI.SDI_FORMTABLE_MAIN_1116_1004 T1  -- 头表
 LEFT JOIN SDI.SDI_FORMTABLE_MAIN_1116_DT1_1004 T2  -- 行表
 ON T1.ID  = T2.MAINID 
 LEFT JOIN tmp_data T3 -- 生产基地
 ON T1.SCJD2 = T3.selectvalue 
 AND T3.FIELDNAME = 'SCJD2'
 LEFT JOIN tmp_data T4 -- 标准版类型
 ON T2.MX1_BZBLX = T4.selectvalue 
 AND T4.FIELDNAME = 'MX1_BZBLX'
 LEFT JOIN tmp_data T5 -- 状态类型
 ON T1.ZTLX = T5.selectvalue 
 AND T5.FIELDNAME = 'ZTLX'
 LEFT JOIN tmp_data T6 -- 外借单位
 ON T2.MX1_WJDW2 = T6.selectvalue 
 AND T6.FIELDNAME = 'MX1_WJDW2'
 WHERE T1.ZTLX IS NOT NULL
),
tmp_data_02 AS(
SELECT T1.APPLY_BASE_NAME AS BASE_NAME,
    T1.STANDARD_CODE,
    T1.STANDARD_BATCH_CODE,
    T1.STANDARD_TYPE_NAME,
    SUM(T1.BEGIN_QTY) AS BEGIN_QTY,
    SUM(T1.BORROW_QTY) AS BORROW_QTY,
    SUM(0) AS LEND_QTY
FROM TMP_DATA_01 T1
GROUP BY T1.APPLY_BASE_NAME,
   T1.STANDARD_CODE,
   T1.STANDARD_BATCH_CODE,
   T1.STANDARD_TYPE_NAME
UNION ALL
SELECT T.OUT_BASE_NAME AS BASE_NAME,
    T.STANDARD_CODE,
    T.STANDARD_BATCH_CODE,
    T.STANDARD_TYPE_NAME,
    SUM(0) AS BEGIN_QTY,
    SUM(0) AS BORROW_QTY,
    SUM(T.BORROW_QTY) AS LEND_QTY
FROM TMP_DATA_01 T
WHERE OUT_BASE_NAME IS NOT null
GROUP BY T.OUT_BASE_NAME,
   T.STANDARD_CODE,
   T.STANDARD_BATCH_CODE,
   T.STANDARD_TYPE_NAME
)
INSERT OVERWRITE TABLE DM.DM_PAM_STANDARD_MANAGE_T  -- 生产标准版管理报表
SELECT T.BASE_NAME,
	   T.STANDARD_CODE,
	   T.STANDARD_BATCH_CODE,
	   T.STANDARD_TYPE_NAME,
	   SUM(T.BEGIN_QTY) AS BEGIN_QTY,
       SUM(T.BORROW_QTY) AS BORROW_QTY,
       SUM(T.LEND_QTY) AS LEND_QTY,
       SUM(T.BEGIN_QTY) + SUM(T.BORROW_QTY) - SUM(T.LEND_QTY) AS END_QTY,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM tmp_data_02 T
GROUP BY T.BASE_NAME,
	     T.STANDARD_CODE,
	     T.STANDARD_BATCH_CODE,
	     T.STANDARD_TYPE_NAME;
