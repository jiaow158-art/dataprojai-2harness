-- 设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with A as (
-- 工程计划单
select 
	substring(a.BSTDK,1,4) as stat_year--期望交期
    ,concat(substring(a.BSTDK,1,4),'-',substring(a.BSTDK,5,2)) as stat_month   -- 年月
	,e.sales_center_code -- 业务部门代码
    ,e.sales_center -- 业务部门
    ,e.sales_dep_code   -- 营销部代码
    ,e.sales_dep   -- 营销部
    ,e.sales_region_code   -- 销区代码
    ,e.sales_region   -- 销区
    ,case 
        when a.vtweg = '06' then 'GD03'
        -- when  a.werks like '%80' then 'GD04' -- 取消特惠品渠道
        when a.vtweg = '08' then 'GD02'
        else 'GD01' end as integrate_channel   -- 整合渠道2
	,a.kunnr
	,a.werks
	,a.vtweg
	,a.menge --使用的是这个数量
	 ,if(substr(thdat,1,6)=substr(BSTDK,1,6),zljcksl*t3.length_1*t3.width/1000000,0) as stockout_area_ondue--准时出库总面积（平方米）
    ,a.menge*t3.length_1*t3.width/1000000 as stockout_area        -- 面积（平方米）--计算总面积
    
    ,a.pspnr --wbs号
from dm.dm_otd_wbs_plan_order_det_t a
join dm.dm_dp_api_sales_group_struct e on a.vkgrp = e.sales_grp
left join dwimd.dwi_md_data_material_general_t t3 on a.matnr = t3.material_num
--where a.pspnr = '00112893'
)

, B as (
-- 销售订单 ： 需要剔除工程计划单中的部分
select 
	substring(a.BSTDK,1,4) as stat_year--期望交期
    ,concat(substring(a.BSTDK,1,4),'-',substring(a.BSTDK,5,2)) as stat_month   -- 年月
    ,e.sales_center_code -- 业务部门代码
    ,e.sales_center -- 业务部门
    ,e.sales_dep_code   -- 营销部代码
    ,e.sales_dep   -- 营销部
    ,e.sales_region_code   -- 销区代码
    ,e.sales_region   -- 销区
    ,case 
        when a.vtweg = '06' then 'GD03'
        -- when  a.werks like '%80' and a.AUART = 'ZA02' then 'GD04' -- 取消特惠品渠道
        when a.vtweg = '08' then 'GD02'
        else 'GD01' end as integrate_channel   -- 整合渠道2
    ,if(substr(thdat,1,6)=substr(BSTDK,1,6),MENGE*material_area,0) as stockout_area_ondue -- 准时出库总面积（平方米）
    ,a.kwmeng*material_area  as stockout_area        -- 面积（平方米）--计算总面积
    
    ,ps_psp_pnr --wbs号
from DM.DM_OTD_SALES_ORDER_DET_T a
join dm.dm_dp_api_sales_group_struct e on a.vkgrp = e.sales_grp
)

,C as (
select 
	A.stat_year
	,A.stat_month
	,A.sales_center_code
	,A.sales_center
	,A.sales_dep_code
	,A.sales_dep
	,A.sales_region_code
	,A.sales_region
	,A.integrate_channel
	,A.stockout_area_ondue
	,A.stockout_area
from A 
union all
select 
	B.stat_year
	,B.stat_month
	,B.sales_center_code
	,B.sales_center
	,B.sales_dep_code
	,B.sales_dep
	,B.sales_region_code
	,B.sales_region
	,B.integrate_channel
	,B.stockout_area_ondue
	,B.stockout_area
from B 
left join A on B.ps_psp_pnr = A.pspnr
where A.pspnr is null
)

INSERT OVERWRITE TABLE dm.dm_dp_api_stockout_oudue partition (stat_month_ym)

select 
	stat_year
	,stat_month
	,sales_center_code
	,sales_center
	,sales_dep_code
	,sales_dep
	,sales_region_code
	,sales_region
	,integrate_channel
	,sum(stockout_area_ondue) as stockout_area_ondue
	,sum(stockout_area) as stockout_area
	,current_timestamp() as dw_last_update_date
	,stat_month as stat_month_ym
from C
group by 
	stat_year
	,stat_month
	,sales_center_code
	,sales_center
	,sales_dep_code
	,sales_dep
	,sales_region_code
	,sales_region
	,integrate_channel
