--应收周转天数（工程市场中心）ZB33
--应收周转天数（工程市场中心）ZB33
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB33';
with base as (
    select node_name9,
           sum(this_month_receive)      this_month_receive,
           sum(last_month_receive)      last_month_receive,
           sum(this_year_begin_receive) this_year_begin_receive,
           sum(this_month_notax)        this_month_notax,
           sum(last_month_notax)        last_month_notax,
           sum(this_year_begin_notax)   this_year_begin_notax
    from (
             select node_name9,
                    sum(case
                            when left(query_date, 7) in
                                 (to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 years', 'yyyy-MM'),
                                  to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-MM'))
                                then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                            else 0 end) / 2
                        + sum(case
                                  when left(query_date, 7) between
                                      to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '11 months', 'yyyy-MM') and
                                      to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months', 'yyyy-MM')
                                      then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                                  else 0 end)
                        this_month_receive,
                    sum(case
                            when left(query_date, 7) in
                                 (to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months' - interval ' 1 years',
                                          'yyyy-MM'),
                                  to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months', 'yyyy-MM'))
                                then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                            else 0 end) / 2
                        + sum(case
                                  when left(query_date, 7) between
                                      to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months' -
                                              interval '11 months',
                                              'yyyy-MM') and
                                      to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months' - interval '1 months',
                                              'yyyy-MM')
                                      then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                                  else 0 end)
                        last_month_receive,
                    sum(case
                            when left(query_date, 7) in
                                 (to_char(to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                          interval '1 years' -
                                          interval '1 years', 'yyyy-MM'),
                                  to_char(to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                          interval '1 years',
                                          'yyyy-MM'))
                                then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                            else 0 end) / 2
                        + sum(case
                                  when left(query_date, 7) between
                                      to_char(to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                              interval '1 years' -
                                              interval '11 months', 'yyyy-MM') and
                                      to_char(to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                              interval '1 years' - interval '1 months',
                                              'yyyy-MM')
                                      then OVERDUE_RECEIVABLES + N_OVERDUE_RECEIVABLES
                                  else 0 end)
                        this_year_begin_receive,
                    0   this_month_notax,
                    0   last_month_notax,
                    0   this_year_begin_notax
             from DWRFIN.DWR_AR_CREDIT_DEVALUE_F a
                      left join dm.dm_rpt_sales_group_t b on a.sales_group_master_file = b.node_name9
             where ((node_name4 = 'H04230655' and strpos(node_desc6, '工程') > 0)
                 or (node_name4 in ('H04230685', 'H03230613'))
                 or (node_name2 in ('H11000003', 'H11000010')))
               and left(query_date, 7) between to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '2 years', 'yyyy-MM')
                 and to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-MM')
                and comp_code != '8000'
               and is_overdue_cust = 'N'
               and coalesce(cust_group_code, '#') != 'Z0'
               and general_ledger_account in ('0011310100', '0011310600', '0011310700', '0011310800', '0011110100','0011310300')
             group by node_name9
             union all
             select node_name9,
                    0                   this_month_receive,
                    0                   last_month_receive,
                    0                   this_year_begin_receive,
                    sum(case
                            when calmonth between to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '11 months',
                                                          'yyyy-MM')
                                and to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-MM') then notax_sales_net_amt
                            else 0 end) this_month_notax,
                    sum(case
                            when calmonth between to_char(
                                        to_date('${PERIOD_ID_M}' || '-01') - interval '1 months' - interval '11 months',
                                        'yyyy-MM')
                                and to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '1 months', 'yyyy-MM')
                                then notax_sales_net_amt
                            else 0 end) last_month_notax,
                    sum(case
                            when calmonth between to_char(
                                        to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                        interval '1 years' -
                                        interval '11 months', 'yyyy-MM')
                                and to_char(to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-12-01')::date -
                                            interval '1 years',
                                            'yyyy-MM') then notax_sales_net_amt
                            else 0 end) this_year_begin_notax
             from dm.dm_ambv2_profit_t a
                      left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
             where ((node_name4 = 'H04230655' and strpos(node_desc6, '工程') > 0)
                 or (node_name4 in ('H04230685', 'H03230613'))
                 or (node_name2 in ('H11000003', 'H11000010')))
               and coalesce(cust_group, '#') != 'Z0'
               and calmonth between to_char(to_date('${PERIOD_ID_M}' || '-01') - interval '2 years', 'yyyy-MM')
                 and to_char(to_date('${PERIOD_ID_M}' || '-01'), 'yyyy-MM')
             group by node_name9
         ) a
    group by node_name9
)
,base2 as (
select 
	  '${PERIOD_ID_M}' as calmonth,
		b.node_name4,
       b.node_desc4,
--       b.node_name3,
--       b.node_desc3,

--       case

--           else b.node_desc4 end            node_desc4,
--       case
--           when (b.node_name2 not in ('H11000002') and b.node_name3 not in ('H03230618'))
--               or node_name4 in ('H04230672', 'H04230680') then ''

--               or node_name4 in ('H04230672', 'H04230680') then ''

       sum(this_month_receive) / 10000      this_month_receive,
       sum(last_month_receive) / 10000      last_month_receive,
       sum(this_year_begin_receive) / 10000 this_year_begin_receive,
       sum(this_month_notax) / 10000        this_month_notax,
       sum(last_month_notax) / 10000        last_month_notax,
       sum(this_year_begin_notax) / 10000   this_year_begin_notax
from base a
         left join dm.dm_rpt_sales_group_t b on a.node_name9 = b.node_name9
where 1 = 1
    and node_name4='H04230685'
group by b.node_name4, b.node_desc4--,

--    b.node_desc5
) 
,AMT as (
select 
CALMONTH,
'ZB33' as zb_code,
case when this_month_notax/360 = 0 then 0 
	else (this_month_receive/12)/(this_month_notax/360)end  as CM_STAT_AMT,
case when this_month_notax/360 = 0 then 0 
	else (this_month_receive/12)/(this_month_notax/360)end  as ACC_STAT_AMT,
case when this_month_notax/360= 0 then 0 
	else (this_month_receive/12)/(this_month_notax/360)end  as QTR_TGT_AMT
from 
base2
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx 
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
