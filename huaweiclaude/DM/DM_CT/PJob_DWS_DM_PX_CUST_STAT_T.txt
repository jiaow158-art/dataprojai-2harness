delete from dm.dm_px_cust_stat_t
where stat_date = to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd')
;

with t1 as( --业绩
select case when a.data_source in ('S', 'T')
        then COALESCE(COALESCE(zz.sales_grp,z."SALES_GROUP_CODE"),a.sales_grp)
        else a.sales_grp
         end as sale_grp_code --销售组
	  ,calmonth --月份
	  ,calday --日期
	  ,lpad(a.customer,10,'0') as cust_num --客户编码
	  ,lpad(coalesce(g."PARENT_ACCNT_NUMBER", a.customer),10,'0') as mth_cust_num --母客编
	  ,coalesce(a.ambperformance,0) as sale_amt --业绩
	  ,a.integrate_channel as channel_code --整合渠道编码
	  ,case when a.integrate_channel = 'GD01'
             and d.product_brand_code in('A5','A26')
			 and d.reserved_field_6 = 'A'
			then coalesce(a.ambperformance,0)
			else 0
			 end as qjcp_sale_amt --旗舰产品业绩
	  --系列=净奢石、微韵石且产品等级=A，整合渠道=零售
from dm.dm_ambv2_zzt_xssr_t a
left join dm.dm_rpt_zmaster_cus_t z
on a.customer = z."ACCOUNT_NUMBER"
and a.distr_chan = z."SALES_CHANNEL_CODE"
and a.salesorg = z."SALES_ORGANIZATION_CODE"
and z."PARVW" = 'AG'
left join (select distinct WBS_LEVEL3 as WBS_NUM,sales_grp from DWRDIM.DWR_DIM_WBS_BASIS_INFO_F)zz
on a.wbs_elemt = zz.WBS_NUM
left join dm.dm_rpt_sales_group_t b 
on case when a.data_source in ('S', 'T')
        then COALESCE(COALESCE(zz.sales_grp,z."SALES_GROUP_CODE"),a.sales_grp)
        else a.sales_grp
         end = b.node_name9
left join dm.dm_rpt_zmaster_cus_t g
                            on a.salesorg = g."SALES_ORGANIZATION_CODE"
						   and coalesce(a.zsales_channel, a.distr_chan) = g."SALES_CHANNEL_CODE"
                           and a.customer = g."ACCOUNT_NUMBER"
						   and g."PARVW" = 'AG'
left join dwimd.dwi_md_data_material_general_t d on a.material = d.material_num
         where calmonth between to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyy-01') 
		   and to_char('${P_PERIOD_D}'::date, 'yyyy-MM')
		   and a.calday <= to_char('${P_PERIOD_D}'::date, 'yyyyMMdd')

),t2 as (
         SELECT
                b.node_name9 as org_code,
                lpad(客户编码,10,'0') as cust_num,
				coalesce(c.parent_customer,lpad(客户编码,10,'0')) as mth_cust_num,
				case when 渠道 = '零售渠道' then 'GD01'
				     when 渠道 = '整装头部' then 'GD02'
					 when 渠道 = '工程渠道' then 'GD03'
					  end as channel_code,
                unnest(string_to_array((
                                                                                                                               COALESCE(一月, '0') || ',' ||
                                                                                                                               COALESCE(二月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(三月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(四月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(五月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(六月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(七月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(八月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(九月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(十月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(十一月, '0') ||
                                                                                                                               ',' ||
                                                                                                                               COALESCE(十二月, '0')),
                                       ','))                                                      目标,
                年份 || unnest(string_to_array('01,02,03,04,05,06,07,08,09,10,11,12', ',')) || '01' 月份
         FROM upload.upload_achievementcustomer a
		 left join (select row_number() over(PARTITION by node_desc9 order by case when node_name4 in('H04230600','H04230655') then 1 else 2 end asc) as rn,* from dm.dm_rpt_sales_group_t) b
           on a.销售地区 = b.node_desc9
		  and b.rn = 1
		 left join(select distinct level4,customer,parent_customer
		             from dm.ct_sales_performance_daily_t
                    where calday = to_char('${P_PERIOD_D}'::date, 'yyyyMMdd')
				  ) c
		   on a.销售地区 = c.level4
		  and lpad(客户编码,10,'0') = c.customer
         where 年份 = to_char(to_date('${P_PERIOD_D}'), 'yyyy')
),t3 as(
select sale_grp_code --销售组
	  ,calmonth --月份
	  ,calday --日期
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sale_amt --业绩
	  ,qjcp_sale_amt --旗舰产品业绩
	  ,cast(0 as numeric(38,10)) as sale_target --目标
  from t1
 union all
select org_code
      ,to_char(月份::date,'yyyy-MM') as calmonth
	  ,null as calday
	  ,cust_num
	  ,mth_cust_num
	  ,channel_code --渠道编码
	  ,cast(0 as numeric(38,10)) as sale_amt
	  ,cast(0 as numeric(38,10)) as qjcp_sale_amt --旗舰产品业绩
	  ,cast(目标 as numeric(38,10))*10000 as sale_target --目标
  from t2
 where to_char(月份::date,'yyyy-MM') <= to_char(to_date('${P_PERIOD_D}'), 'yyyy-MM')
),t4 as(
select to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd') as stat_date --统计日期
      ,cast('年度' as varchar) as stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sum(case when left(calmonth,4) = to_char(to_date('${P_PERIOD_D}'), 'yyyy') then sale_amt else 0 end) as sale_amt --业绩
	  ,sum(case when left(calmonth,4) = to_char(to_date('${P_PERIOD_D}') - interval '1 years', 'yyyy')
	             and calday <= to_char(to_date('${P_PERIOD_D}') - interval '1 years', 'yyyyMMdd')
	            then sale_amt else 0 end) as ly_sale_amt --去年同期业绩
	  ,sum(case when left(calmonth,4) = to_char(to_date('${P_PERIOD_D}'), 'yyyy') then qjcp_sale_amt else 0 end) as qjcp_sale_amt --旗舰产品业绩
	  ,sum(case when left(calmonth,4) = to_char(to_date('${P_PERIOD_D}'), 'yyyy') then sale_target else 0 end) as sale_target --目标
  from t3 as a
 group by sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
 union all
select to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd') as stat_date --统计日期
      ,cast('半年度' as varchar) as stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sum(case when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('1','2')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('1','2') 
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date, 'yyyy')
				then sale_amt
                when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('3','4')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('3','4')
				then sale_amt
				 else 0 end) as sale_amt --业绩
	  ,sum(case when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd') - interval '1 years','yyyyQ'),1) in('1','2')
				 and right(to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyQ'),1) in('1','2')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyy')
				 and calday <= to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyMMdd')
				then sale_amt
                when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd') - interval '1 years','yyyyQ'),1) in('3','4')
				 and right(to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyQ'),1) in('3','4')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyy')
				 and calday <= to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyMMdd')
				then sale_amt
				 else 0 end) as ly_sale_amt --去年同期业绩
	  ,sum(case when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('1','2')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('1','2')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date, 'yyyy')
				then qjcp_sale_amt
                when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('3','4')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('3','4')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date, 'yyyy')
				then qjcp_sale_amt
				 else 0 end) as qjcp_sale_amt --旗舰产品业绩
	  ,sum(case when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('1','2')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('1','2')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date, 'yyyy')
				then sale_target
                when right(to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ'),1) in('3','4')
				 and right(to_char('${P_PERIOD_D}'::date, 'yyyyQ'),1) in('3','4')
				 and left(calmonth,4) = to_char('${P_PERIOD_D}'::date, 'yyyy')
				then sale_target
				 else 0 end) as sale_target --目标
  from t3 as a
 group by sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
 union all
select to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd') as stat_date --统计日期
      ,cast('季度' as varchar) as stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ') = to_char('${P_PERIOD_D}'::date, 'yyyyQ')
	            then sale_amt else 0 end) as sale_amt --业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ') = to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyQ')
	             and calday <= to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyMMdd') 
	            then sale_amt else 0 end) as ly_sale_amt --去年同期业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ') = to_char('${P_PERIOD_D}'::date, 'yyyyQ')
	            then qjcp_sale_amt else 0 end) as qjcp_sale_amt --旗舰产品业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyyQ') = to_char('${P_PERIOD_D}'::date, 'yyyyQ')
	            then sale_target else 0 end) as sale_target --目标
  from t3 as a
 group by sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
 union all
select to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd') as stat_date --统计日期
      ,cast('月度' as varchar) as stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyy-MM') = to_char('${P_PERIOD_D}'::date, 'yyyy-MM')
	            then sale_amt else 0 end) as sale_amt --业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyy-MM') = to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyy-MM')
	             and calday <= to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyMMdd') 
	            then sale_amt else 0 end) as ly_sale_amt --去年同期业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyy-MM') = to_char('${P_PERIOD_D}'::date, 'yyyy-MM')
	            then qjcp_sale_amt else 0 end) as qjcp_sale_amt --旗舰产品业绩
	  ,sum(case when to_char(to_date(a.calmonth||'-01','yyyy-MM-dd'),'yyyy-MM') = to_char('${P_PERIOD_D}'::date, 'yyyy-MM')
	            then sale_target else 0 end) as sale_target --目标
  from t3 as a
 group by sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
 union all
select to_char('${P_PERIOD_D}'::date, 'yyyy-MM-dd') as stat_date --统计日期
      ,cast('日' as varchar) as stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
	  ,sum(case when calday = to_char('${P_PERIOD_D}'::date, 'yyyyMMdd')
	            then sale_amt else 0 end) as sale_amt --业绩
	  ,sum(case when calday = to_char('${P_PERIOD_D}'::date - interval '1 years', 'yyyyMMdd')
	            then sale_amt else 0 end) as ly_sale_amt --去年同期业绩
	  ,sum(case when calday = to_char('${P_PERIOD_D}'::date, 'yyyyMMdd')
	            then qjcp_sale_amt else 0 end) as qjcp_sale_amt --旗舰产品业绩
	  ,null as sale_target --目标
  from t3 as a
 group by sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,mth_cust_num --母客编
	  ,channel_code --渠道编码
)insert into dm.dm_px_cust_stat_t(
       stat_date --统计日期
      ,stat_type --统计类型
	  ,sale_grp_code --销售组
	  ,cust_num --客户编码
	  ,cust_name --客户名称
	  ,mth_cust_num --母客编
	  ,mth_cust_name --母客编描述
	  ,channel_code --整合渠道编码
	  ,channel_name --整合渠道名称
	  ,sale_amt --业绩
	  ,ly_sale_amt --去年同期业绩
	  ,qjcp_sale_amt --旗舰产品业绩
	  ,sale_target --目标
	  ,dw_last_update_date --DW数据同步时间
)
select a.stat_date --统计日期
      ,a.stat_type --统计类型
	  ,a.sale_grp_code --销售组
	  ,a.cust_num --客户编码
	  ,c.cust_name --客户名称
	  ,a.mth_cust_num --母客编
	  ,d.cust_name as mth_cust_name --母客编描述
	  ,a.channel_code --整合渠道编码
	  ,b.channel_name --整合渠道名称
	  ,a.sale_amt --业绩
	  ,a.ly_sale_amt --去年同期业绩
	  ,a.qjcp_sale_amt --旗舰产品业绩
	  ,a.sale_target --目标
	  ,current_timestamp as dw_last_update_date --DW数据同步时间
  from t4 as a
  left join dm.dm_dim_integrate_channel_d b
    on a.channel_code = b.channel_code
   and b.channel_type = '整合渠道1'
  left join (select distinct z."ACCOUNT_NUMBER" as cust_num
                            ,z."DESCRIPTION" as cust_name
               from dm.dm_rpt_zmaster_cus_t z
              where z."PARVW" = 'AG'
            ) c
	on a.cust_num = c.cust_num
  left join (select distinct z."ACCOUNT_NUMBER" as cust_num
                            ,z."DESCRIPTION" as cust_name
               from dm.dm_rpt_zmaster_cus_t z
              where z."PARVW" = 'AG'
            ) d
	on a.mth_cust_num = d.cust_num
  left join dm.dm_rpt_sales_group_t e
    on a.sale_grp_code = e.node_name9
 where e.node_name2 = 'H11000001'
