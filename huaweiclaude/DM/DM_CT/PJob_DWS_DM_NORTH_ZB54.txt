--区域化交付率ZB54
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB54';
with tmp as(
select
stat_month,
case
when
b.product_area_name
is
null
then'区域化'else'非区域化'
end as area_flag,
sum(sales_stock_out_area)/10000 as sales_stock_out_area
from dm.dm_otd_area_delivery_detail_m a
left join upload.upload_otd_no_area_supply_line_cfg b 
on a.belong_area_name=b.product_area_name
and a.receiving_province=b.receiving_province
where 
not(coalesce(a.product_level2_name,'#')IN('景观砖','玻化砖')
or
coalesce(a.product_level3_name,'#')IN('超大板'))
group by
stat_month,dis_channel_name,op_partment,belong_area_name,product_level2_name,dimension,area_flag
)
,base as (

select 
'${PERIOD_ID_M}' as calmonth,
'ZB54' as zb_code,
sum(case when area_flag='区域化' then sales_stock_out_area else 0 end)/sum(sales_stock_out_area) as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
tmp
where stat_month = '${PERIOD_ID_M}'


union all 

select 
'${PERIOD_ID_M}' as calmonth,
'ZB54' as zb_code,
0 as 当月实绩,
sum(case when area_flag='区域化' then sales_stock_out_area else 0 end)/sum(sales_stock_out_area) as 累计实绩,
0 as 季度累计实绩
from 
tmp
where 
stat_month between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'


union all 

select 
'${PERIOD_ID_M}' as calmonth,
'ZB54' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
sum(case when area_flag='区域化' then sales_stock_out_area else 0 end)/sum(sales_stock_out_area)  as 季度累计实绩
from 
tmp
where    
to_char((SUBSTRING(stat_month,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(stat_month,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)*100 as CM_STAT_AMT,
sum(累计实绩)*100 as ACC_STAT_AMT,
sum(季度累计实绩)*100 as QTR_TGT_AMT
from 
base
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
