WITH DALIY_USE AS ( --能源消耗
select 
 A.start_date
,A.site_code
,A.product_line_code
,A.work_shop_code
,sum(case when A.business_type_code = 1 then A.use_qty else 0 end)  as ELEC_USE_QTY --用电量
,sum(case when A.business_type_code = 2 then A.use_qty else 0 end ) as GAS_USE_QTY --用气量
,sum(case when A.business_type_code = 3 then A.use_qty else 0 end ) as COAL_USE_QTY --用煤量
,max(B1.conversion_coefficient) as gas_coefficient --天然气折标系数
,max(B2.conversion_coefficient) as elec_coefficient --电力折标系数
,max(B3.conversion_coefficient) as coal_coefficient --煤折标系数
,max(B4.conversion_coefficient) as jhq_coefficient --焦化气折标系数
from DWROTD.DWR_PAM_BASE_ENERGYCONSUME_D_F A
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B1
on A.site_code = B1.site_code
and A.work_shop_code = B1.work_shop_code
and A.start_date between B1.start_date and B1.end_date
and B1.delete_flag = 1 
and B1.fuel_category = '天然气'
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B2
on A.site_code = B2.site_code
and A.work_shop_code = B2.work_shop_code
and A.start_date between B2.start_date and B2.end_date
and B2.delete_flag = 1 
and B2.fuel_category = '电力'
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B3
on A.site_code = B3.site_code
and A.work_shop_code = B3.work_shop_code
and A.start_date between B3.start_date and B3.end_date
and B3.delete_flag = 1 
and B3.fuel_category = '煤'
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B4
on A.site_code = B4.site_code
and A.work_shop_code = B4.work_shop_code
and A.start_date between B4.start_date and B4.end_date
and B4.delete_flag = 1 
and B4.fuel_category = '焦化气'
where  A.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}'  AND A.SITE_CODE = '1500'
group by 
 A.start_date
,A.site_code
,A.product_line_code
,A.work_shop_code
)

,WAREHOUSE_PROD as (  --过仓产量

 shift_date as start_date
,site_code
,work_shop_code
,replace(product_line_code,'FJ','YX') AS product_line_code  --分级产线转换成成型产线
,sum(case when grade in ('一级','优等','次品') then emit_num * unit_area else 0 end) as emit_area
from  DWROTD.DWR_PAM_BASE_WAREHOUSE_PROD_F
where SITE_CODE = '1500' AND shift_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}'
group by 
 shift_date
,site_code
,work_shop_code
,replace(product_line_code,'FJ','YX')
)
,gas_use as (
select

,site_code
,work_shop_code -- 车间
,product_line_code
,sum(gas_qty) as gas_qty
,sum(gas_qty_jlq) as gas_qty_jlq -- 焦炉气用量
,sum(gas_qty_trq) as gas_qty_trq  -- 天然气用量
from DWROTD.DWR_PAM_BASE_NATURALGAS_F
where site_code = '1500' AND start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}'
group by 
 start_date
,site_code
,work_shop_code -- 车间
,product_line_code
)
,dry_prod as (  --原料车间过仓产量
SELECT
 shift_date as start_date
,site as site_code
,work_shop as work_shop_code
,product_line as product_line_code
,SUM(dry_weight_of_powder) as dry_area
,SUM(case when fuel_name = '天然气' then dry_weight_of_powder else 0 end) trq_dry_area
,SUM(case when fuel_name = '焦化气' then dry_weight_of_powder else 0 end) jhq_dry_area
,SUM(case when fuel_name = '直烧煤' then dry_weight_of_powder else 0 end) zsm_dry_area
FROM sdi.sdi_dj_statistics_of_spray_tower_powder_output_1006
WHERE  delete_flag = 1 and site = '1500' AND shift_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}'
GROUP BY 
 site
,work_shop
,shift_date
,product_line
)

	and T2.product_line_code = T1.product_line_code

	GROUP BY

		 t1.product_line_code
)

select
 T.start_date
,T.site_code
,T2.site_name
,T.work_shop_code
,T3.work_shop_name
,T.product_line_code
,T1.product_line_name

,T.emit_area --过仓产量
,T.ELEC_USE_QTY --用电量
,T.COAL_USE_QTY -- 用煤量
,T.elec_coefficient ----电力折标系数
,T.gas_coefficient ----天然气折标系数
,T.coal_coefficient ---煤折折标标系数
,T.jhq_coefficient --焦化气折标系数
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,T.trq_dry_area --天然气过仓产量
,T.jhq_dry_area --焦化气过仓产量
,T.zsm_dry_area --直烧煤过仓产量
,T.grade_qty --分级产量
,T.start_date AS period_id_d
from tmp1 T
left join DWRDIM.DWR_DIM_PAM_PRODUCT_LINE_D T1 --产线维表
on T.site_code = T1.site_code
and T.work_shop_code = T1.work_shop_code
and T.product_line_code = T1.product_line_code
left join DWRDIM.DWR_DIM_PAM_SITE_D T2 -- 基地维表
on T.site_code = T2.site_code
left join DWRDIM.DWR_DIM_PAM_WORK_SHOP_D T3 --车间维表
on T.site_code = T3.site_code 
and T.work_shop_code = T3.work_shop_code
