-- 瓷砖各基地能源消耗表(月)
WITH DM_PAM_BASE_INSTOCKYIELD_S_DATA AS(  -- 入库产量数据（电、物耗）
	SELECT period_id_m, -- 分区字段
		   start_month, -- 月份
		   site_code, -- 基地
		   work_shop_code, -- 车间
		   product_line_code, -- 产线
		   SUM(stock_qty) AS instock_qty  -- 入库数量
	FROM DM.DM_PAM_BASE_INSTOCKYIELD_S -- 瓷砖各基地入库产量表
	WHERE del_flag = 'N'
	AND period_id_m >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7) -- 前3月，调度传昨天,格式：YYYY-MM-DD 
	GROUP BY period_id_m,
			 start_month,
			 site_code,
			 work_shop_code,
			 product_line_code
),
DM_PAM_BASE_INSTOCKYIELD_S_NO_CQ_GAS_DATA AS(  -- 入库产量数据（非重庆的气）
	SELECT period_id_m, -- 分区字段
		   start_month, -- 月份
		   site_code, -- 基地
		   work_shop_code, -- 车间
		   product_line_code, -- 产线
		   '2' AS business_type_code, -- 气
		   '气' AS business_type_name, -- 气
		   SUM(stock_qty) AS instock_qty  -- 入库数量
	FROM DM.DM_PAM_BASE_INSTOCKYIELD_S -- 瓷砖各基地入库产量表
	WHERE site_code <> '1800'
	AND del_flag = 'N'
	AND period_id_m >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7) -- 前3月，调度传昨天,格式：YYYY-MM-DD 
	GROUP BY period_id_m,
			 start_month,
			 site_code,
			 work_shop_code,
			 product_line_code
),
DM_PAM_BASE_INSTOCKYIELD_S_GAS_DATA AS(  -- 入库产量数据（重庆）
	SELECT period_id_m, -- 分区字段
		   start_month, -- 日期
		   site_code, -- 基地
		   product_line_code, -- 产线
		   SUM(instock_qty) AS instock_qty  -- 入库数量
	FROM DM_PAM_BASE_INSTOCKYIELD_S_DATA -- 瓷砖各基地入库产量表
	WHERE site_code = 1800
	GROUP BY period_id_m,
			 start_month,
			 site_code,
			 product_line_code
),
COST_CENTER_DATA_MATERIAL as( -- 成本中心数据
	SELECT T1.start_month, -- 月份
		   T1.business_type_code,
		   T1.business_type_name, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
		   1 AS ANALYSIS_TYPE_CODE,
		   '成本中心' AS ANALYSIS_TYPE_NAME, -- 分析类型名称(1:成本中心 2:工序 3:产线 4:车间 5:基地)
		   T1.site_code,  -- 基地编码
		   T3.work_shop AS work_shop_code, -- 车间编码
		   T3.product_line AS product_line_code, -- 产线编码
		   T3.work_no_code AS work_no_code, -- 工序编码
		   T1.cost_center_code, -- 成本中心编码
		   T1.use_qty, -- 实际用量
		   NVL(T2.instock_qty,0) AS instock_qty,-- 入库产量
		   T1.period_id_m 
	FROM DWROTD.DWR_PAM_BASE_ENERGYCONSUME_M_F T1 -- 瓷砖各基地能源用量表(月)
	LEFT JOIN DM_PAM_BASE_INSTOCKYIELD_S_GAS_DATA T2 -- 瓷砖各基地入库产量表(重庆)
	ON T1.period_id_m = T2.period_id_m  -- 分区字段
	AND T1.start_month = T2.start_month  -- 日期
	AND T1.site_code = T2.site_code -- 基地
	AND T1.product_line_code = T2.product_line_code -- 产线
	LEFT JOIN SDI.sdi_dj_powerconsumption_costcenter_maintenance_1006 T3 -- 清远-用电与成本中心维护表
	ON T1.site_code = T3.SITE
	AND T1.cost_center_code = T3.cost_center 
	AND T3.delete_flag = 1
	AND T3.site = 1800
	WHERE T1.del_flag = 'N'
	AND T1.business_type_code = 2  -- 气
	AND T1.site_code = '1800'  -- 重庆
	AND t1.period_id_m >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7) -- 前3月，调度传昨天,格式：YYYY-MM-DD
	AND T1.period_id_m >= '2023-01'  -- 固定条件，原因见注释
	UNION ALL
	SELECT COALESCE(T1.start_month,T2.start_month) AS start_month, -- 月份
		   COALESCE(T1.business_type_code,T2.business_type_code) AS business_type_code,
		   COALESCE(T1.business_type_name,T2.business_type_name) AS business_type_name, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
		   1 AS ANALYSIS_TYPE_CODE,
		   '成本中心' AS ANALYSIS_TYPE_NAME, -- 分析类型名称(1:成本中心 2:工序 3:产线 4:车间 5:基地)
		   COALESCE(T1.site_code,T2.site_code) AS site_code,  -- 基地编码
		   COALESCE(T1.work_shop_code,T2.work_shop_code) AS work_shop_code, -- 车间编码
		   COALESCE(T1.product_line_code,T2.product_line_code) AS product_line_code, -- 产线编码
		   CASE WHEN COALESCE(T1.site_code,T2.site_code) = '1300' 
		        AND COALESCE(T1.work_shop_code,T2.work_shop_code) = 'CJ_QY7'
		        AND COALESCE(T1.product_line_code,T2.product_line_code) = '0B8_CX08'
		        AND T1.work_no_code IS NULL THEN '0B8_CX08'
		   ELSE T1.work_no_code
		   END AS work_no_code, -- 工序编码
		   CASE WHEN COALESCE(T1.site_code,T2.site_code) = '1300' 
		        AND COALESCE(T1.work_shop_code,T2.work_shop_code) = 'CJ_QY7'
		        AND COALESCE(T1.product_line_code,T2.product_line_code) = '0B8_CX08'
		        AND T1.cost_center_code IS NULL THEN '1300MA0060'
		   ELSE T1.cost_center_code
		   END AS cost_center_code, -- 成本中心编码
		   COALESCE(T1.use_qty,0) AS use_qty,  -- 实际用量
		   COALESCE(T2.instock_qty,0) AS instock_qty,-- 入库产量
		   COALESCE(T1.period_id_m,T2.period_id_m) AS period_id_m 
	FROM DWROTD.DWR_PAM_BASE_ENERGYCONSUME_M_F T1 -- 瓷砖各基地能源用量表(月)
	FULL OUTER JOIN DM_PAM_BASE_INSTOCKYIELD_S_NO_CQ_GAS_DATA T2 -- 瓷砖各基地入库产量表
    ON T1.period_id_m = T2.period_id_m  -- 分区字段
	AND T1.start_month = T2.start_month  -- 月份
	AND T1.site_code = T2.site_code -- 基地
	AND T1.work_shop_code = T2.work_shop_code -- 车间
	AND T1.product_line_code = T2.product_line_code -- 产线
	AND T1.business_type_code = T2.business_type_code
	AND T1.business_type_name = t2.business_type_name
	WHERE 1 = 1
	AND COALESCE(t1.period_id_m,T2.period_id_m) >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7) -- 前3月，调度传昨天,格式：YYYY-MM-DD
	AND COALESCE(T1.site_code,T2.site_code) <> '1800'   -- 非重庆的气
	AND COALESCE(T1.business_type_code,T2.business_type_code) = 2 
	AND COALESCE(T1.period_id_m,T2.period_id_m) >= '2023-01' -- 固定条件，原因见注释
	UNION ALL
	SELECT T1.start_month, -- 月份
		   T1.business_type_code,
		   T1.business_type_name, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
		   1 AS ANALYSIS_TYPE_CODE,
		   '成本中心' AS ANALYSIS_TYPE_NAME, -- 分析类型名称(1:成本中心 2:工序 3:产线 4:车间 5:基地)
		   T1.site_code,  -- 基地编码
		   T1.work_shop_code, -- 车间编码
		   T1.product_line_code, -- 产线编码
		   T1.work_no_code, -- 工序编码
		   T1.cost_center_code, -- 成本中心编码
		   T1.use_qty, -- 实际用量
		   NVL(T2.instock_qty,0) AS instock_qty,-- 入库产量
		   T1.period_id_m 
	FROM DWROTD.DWR_PAM_BASE_ENERGYCONSUME_M_F T1 -- 瓷砖各基地能源用量表(月)
	LEFT JOIN DM_PAM_BASE_INSTOCKYIELD_S_DATA T2 -- 瓷砖各基地入库产量表
	ON T1.period_id_m = T2.period_id_m  -- 分区字段
	AND T1.start_month = T2.start_month  -- 月份
	AND T1.site_code = T2.site_code -- 基地
	AND T1.work_shop_code = T2.work_shop_code -- 车间
	AND T1.product_line_code = T2.product_line_code -- 产线
	WHERE T1.del_flag = 'N'
	AND t1.period_id_m >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7) -- 前3月，调度传昨天,格式：YYYY-MM-DD
	AND T1.business_type_code IN (1,4)   -- 电 、 物耗
	AND T1.period_id_m >= '2024-01'
),
