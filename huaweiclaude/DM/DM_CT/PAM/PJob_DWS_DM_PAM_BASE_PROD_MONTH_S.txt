DELETE FROM DM.DM_PAM_BASE_PROD_MONTH_S WHERE STAT_MONTH BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}';
INSERT INTO DM.DM_PAM_BASE_PROD_MONTH_S
with base_line as (
select *
from (select distinct zz.site_code
                    , zz.site_short_name
                    , case when zz.site_code = '1300' and work_shop_type != '原料车间' then zz.product_line_code
                           else zz.work_shop_code end work_code
                    , case when zz.site_code = '1300' and work_shop_type != '原料车间' then zz.product_line_name
                           else zz.work_shop_name end work_name
          from DWRDIM.DWR_DIM_PAM_PRODUCT_LINE_D zz
          where zz.site_code in ('1300', '1600', '1500', '1800', '1900')
          and zz.work_shop_code in
                ('CJ_HN1', 'CJ_HN2', 'CJ_HN3', 'CJ_HN4', 'CJ_HN5', 'CJ_HNYLCJ', 'CJ_YC5', 'CJ_YC6', 'CJ_YC7', 'CJ_YC8',
                 'CJ_YCYL1',
                 'CJ_FC8', 'CJ_FC7', 'CJ_FC6', 'CJ_FC5', 'CJ_FC4', 'CJ_FC3', 'CJ_FC2', 'CJ_FC11', 'CJ_FC1',
                 'CJ_YL1', 'WS_YL',
                 'CJ_QY1', 'CJ_QY3', 'CJ_QY5', 'CJ_QY7', 'CJ_QY9', 'CJ_SX1', 'CJ_SX2', 'CJ_YL')
         ) aa
)
,tmp as (
select
 substr(to_char((a.start_date)::date + interval '1 days', 'yyyy-MM-dd'),1,7) as stat_month
,zz.site_code
,zz.site_short_name
,zz.work_code
,zz.work_name
,sum(a.grade_area) as m_grade_area
,max(daily_avg_prod_goat) as daily_avg_prod_goat
,max(a.prod_day) as prod_day
,sum(a.superior_area) as superior_area
-- 压机数-产量+接班数-交班数-中试
,sum(a.production_area - a.grade_area + a.K_TAKE_AREA + a.Y_TAKE_AREA - a.K_HAND_AREA - a.Y_HAND_AREA - a.loss_slice_area) as loss_ntr --损耗分子
,sum(a.production_area + a.K_TAKE_AREA + a.Y_TAKE_AREA - a.K_HAND_AREA - a.Y_HAND_AREA - a.loss_slice_area) as loss_dtr --损耗分母
,sum(gas_use_qty) as gas_use_qty --用气量
,sum(a.coal_use_qty) as coal_use_qty --用煤量
,sum(a.elec_use_qty) as elec_use_qty --用电量
,sum(a.mc_use_qty) as mc_use_qty --物耗量
,max(a.elec_coefficient) as elec_coefficient --用电折标系数
,max(a.gas_coefficient) as gas_coefficient --用气折标系数
,max(a.coal_coefficient) as coal_coefficient --用煤折标系数
from base_line zz
left join dm.dm_pam_base_prod_daily_t a
on case when a.site_code = '1300' and a.product_line_code not in ('OB_YLFB', 'OB_YLFZ', 'OB_YLFB01', 'OB_YLFB02') 
then a.product_line_code else a.work_shop_code end = zz.work_code
--where a.site_code = '1300'
group by 
substr(to_char((a.start_date)::date + interval '1 days', 'yyyy-MM-dd'),1,7)
,zz.site_code
,zz.site_short_name
,zz.work_code
,zz.work_name
)
select 
 a.stat_month  --年月
,a.site_code --基地编码
,a.site_short_name --基地名称
,a.work_code --车间/产线代码名称
,a.work_name --车间/产线名称
,coalesce(a.m_grade_area,0) as m_grade_area -- 月产量
,coalesce(a.daily_avg_prod_goat,0) as daily_avg_prod_goat --日均产量目标
,sum(a.m_grade_area) over(partition by a.stat_month,a.site_code ) as sum_grade_area --总产量
,coalesce(a.superior_area,0) as superior_area -- 优等品产量
,coalesce(a.prod_day,0) as prod_day --生产天数
,coalesce(b.prod_qty,0) as prod_qty  --总产能
,coalesce(b.emp_cnt,0) as emp_cnt  --员工人数
,coalesce(a.loss_ntr,0) as loss_ntr--损耗分子
,coalesce(a.loss_dtr,0) as loss_dtr --损耗分母
,coalesce(a.gas_use_qty,0) as gas_use_qty--用气量
,coalesce(a.coal_use_qty,0) as coal_use_qty--用煤量
,coalesce(a.elec_use_qty,0) as elec_use_qty--用电量
,coalesce(a.mc_use_qty,0) as mc_use_qty--物耗量
,coalesce(a.elec_coefficient,0) as elec_coefficient--用电折标系数
,coalesce(a.gas_coefficient,0) as gas_coefficient --用气折标系数
,coalesce(a.coal_coefficient,0) as coal_coefficient --用煤折标系数
,case when a.stat_month = substr(current_date() - 1,1,7) then EXTRACT(DAY FROM (DATE_TRUNC('DAY', CURRENT_DATE))) 
 else EXTRACT(DAY FROM (DATE_TRUNC('DAY', last_day(to_date(a.stat_month,'YYYY-MM-DD'))))) end as day_cnt --当期天数
,case when a.stat_month = substr(current_date() - 1,1,7) then sum(a.m_grade_area) over(partition by a.stat_month,a.site_code ) else coalesce(c.act_grade_area,0) end as act_grade_area --实际产量(成本还原)
,current_timestamp() as dw_last_update_date 
from tmp a
left join (select * from DM.DM_SAP_HR_BASE_PROD_T where data_type = '产能') b
on a.site_code = b.dept_code
and a.stat_month = b.stat_month
left join (select stat_month,plant,sum(sum_area/3) as act_grade_area from DWRFIN.DWR_COST_RECOVERY_RESEARCH_T group by stat_month,plant) c
on a.stat_month = c.stat_month and a.site_code = c.plant
WHERE a.STAT_MONTH BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
