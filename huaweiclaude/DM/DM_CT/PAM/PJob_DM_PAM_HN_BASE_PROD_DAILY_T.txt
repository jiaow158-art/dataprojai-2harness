--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true;

WITH DAILY_PROD AS ( --分级产量
select 
 T.start_date
,T.site_code
,T.work_shop_code
,T.product_line_code
,T.group_code
,T.material_code AS material_num --当日产品编号
,T.product_model AS prod_size --当日产品规格
,sum(T.DIVIDEYIELD_AREA * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000) as grade_area --产量(㎡)
,sum(T.DIVIDEYIELD_AREA) as grade_qty --产量数
,sum(T.first_rate_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as superior_area --优等品产量(㎡)
,sum(T.first_rate_area) as superior_qty --优等品产量数（㎡）
,sum(T.one_grade_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as one_grade_area --一级品产量(㎡)
,sum(T.one_grade_area) as one_grade_qty -- 一级品产量数
,sum(T.second_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as second_area --次品产量(㎡)
,sum(T.second_area) as second_qty --次品产量数
from DWROTD.DWR_PAM_BASE_DIVIDEYIELD_F T
where T.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}' 
and T.site_code = '1600'
group by 
 T.start_date
,T.site_code
,T.work_shop_code
,T.product_line_code
,T.group_code
,T.material_code
,T.product_model
)
,prod_defect as (
 select 
 a.shift_date as start_date
,a.site_code
,a.cx_work_shop_code as work_shop_code
,a.cx_product_line_code as product_line_code
,a.group_code
,a.product_code as material_num
,a.product_dim as prod_size
,b.work_no_code
,b.work_no_desc as work_no_name
,a.defect_code
,a.defect_name
,sum(nvl(a.first_rate_qty,0)) as first_rate_qty --一级数量
,sum(nvl(a.first_rate_qty,0) * a.unit_area) as first_rate_area --一级面积
,sum(nvl(a.substandard_qty,0)) as substandard_qty --次品数量
,sum(nvl(a.substandard_qty,0) * a.unit_area) as substandard_area --次品面积
,sum(nvl(a.waste_qty,0)) as waste_qty  --废品数量
,sum(nvl(a.waste_qty,0) * a.unit_area) as waste_area --废品面积
from DWROTD.DWR_PAM_BASE_GRADE_DEFECT_F a 
left join SDI.SDI_DP_DATA_FENJI_GRADE_RADING_PROCESS_CONTRAST_1006 b 
on a.site_code = b.site and a.defect_code = b.defect_code and b.delete_flag = 1
where a.shift_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}' 
and a.site_code = '1600' 
group by 
 a.shift_date
,a.site_code
,a.cx_work_shop_code
,a.cx_product_line_code
,a.group_code
,a.product_code
,b.work_no_code
,b.work_no_desc
,a.defect_code
,a.defect_name
,a.product_dim
)
,tmp as (
select
 a.start_date  --日期
,a.site_code  --基地编码
,a.work_shop_code  -- 车间编码
,a.product_line_code --产线编码
,a.group_code  --班组
,a.material_num --当日产品编号
,a.prod_size --当日产品规格
,b.work_no_code --工序编码
,b.work_no_name --工序名称
,b.defect_code  --缺陷编码
,b.defect_name --缺陷名称
,a.grade_area--产量(㎡)
,a.grade_qty--产量数
,a.superior_area--优等品产量(㎡)
,a.superior_qty--优等品产量数（㎡）
,a.one_grade_area--一级品产量(㎡)
,a.one_grade_qty-- 一级品产量数
,a.second_area--次品产量(㎡)
,a.second_qty--次品产量数
,0 as first_rate_qty--一级数量
,0 as first_rate_area--一级面积
,0 as substandard_qty--次品数量
,0 as substandard_area--次品面积
,0 as waste_qty --废品数量
,0 as waste_area --废品面积
from DAILY_PROD a
left join prod_defect b
on a.start_date = b.start_date
and a.site_code = b.site_code
and a.work_shop_code = b.work_shop_code
and a.product_line_code = b.product_line_code
and a.group_code = b.group_code
and a.material_num = b.material_num
union all 
select
 start_date  --日期
,site_code  --基地编码
,work_shop_code  -- 车间编码
,product_line_code --产线编码
,group_code  --班组
,material_num --当日产品编号
,prod_size --当日产品规格
,work_no_code --工序编码
,work_no_name --工序名称
,defect_code  --缺陷编码
,defect_name --缺陷名称
,0 as grade_area--产量(㎡)
,0 as grade_qty--产量数
,0 as superior_area--优等品产量(㎡)
,0 as superior_qty--优等品产量数（㎡）
,0 as one_grade_area--一级品产量(㎡)
,0 as one_grade_qty-- 一级品产量数
,0 as second_area--次品产量(㎡)
,0 as second_qty--次品产量数
,first_rate_qty--一级数量
,first_rate_area--一级面积
,substandard_qty--次品数量
,substandard_area--次品面积
,waste_qty --废品数量
,waste_area --废品面积
from prod_defect
)
,tmp1 as (
select
 start_date  --日期
,site_code  --基地编码
,work_shop_code  -- 车间编码
,product_line_code --产线编码
,group_code  --班组
,material_num --当日产品编号
,prod_size --当日产品规格
,work_no_code --工序编码
,work_no_name --工序名称
,defect_code  --缺陷编码
,defect_name --缺陷名称
,sum(grade_area) as grade_area--产量(㎡)
,sum(grade_qty) as grade_qty--产量数
,sum(superior_area) as superior_area--优等品产量(㎡)
,sum(superior_qty) as superior_qty--优等品产量数（㎡）
,sum(one_grade_area) as one_grade_area--一级品产量(㎡)
,sum(one_grade_qty) as one_grade_qty-- 一级品产量数
,sum(second_area) as second_area--次品产量(㎡)
,sum(second_qty) as second_qty--次品产量数
,sum(first_rate_qty) as first_rate_qty--一级数量
,sum(first_rate_area) as first_rate_area--一级面积
,sum(substandard_qty) as substandard_qty--次品数量
,sum(substandard_area) as substandard_area--次品面积
,sum(waste_qty) as waste_qty --废品数量
,sum(waste_area) as waste_area --废品面积
from tmp a
group by 
 start_date  --日期
,site_code  --基地编码
,work_shop_code  -- 车间编码
,product_line_code --产线编码
,group_code  --班组
,material_num --当日产品编号
,prod_size --当日产品规格
,work_no_code --工序编码
,work_no_name --工序名称
,defect_code  --缺陷编码
,defect_name --缺陷名称
)
insert overwrite table dm.dm_pam_hn_base_prod_daily_t partition(period_id_d)
select
 T.start_date  --日期
,T.site_code --基地编码
,T2.site_name --基地名称
,T.work_shop_code  --车间编码
,T3.work_shop_name --车间名称
,T.product_line_code  --产线编码
,T1.product_line_name --产线名称
,T.group_code  --班组
,T.material_num --当日产品编号
,T.prod_size --当日产品规格
,T.work_no_code --工序编码
,T.work_no_name --工序名称
,T.defect_code  --缺陷编码
,T.defect_name --缺陷名称
,T.grade_qty --产量数
,T.grade_area --产量(㎡)
,T.superior_area --优等品产量(㎡)
,T.superior_qty --优等品产量数
,T.one_grade_area --一级品产量(㎡)
,T.one_grade_qty -- 一级品产量数
,T.second_area --次品产量(㎡)
,T.second_qty --次品产量数
,T.first_rate_qty --一级数量
,T.first_rate_area --一级面积
,T.substandard_qty --次品数量
,T.substandard_area --次品面积
,T.waste_qty  --废品数量
,T.waste_area --废品面积
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,T.start_date AS period_id_d
from tmp1 T
left join DWRDIM.DWR_DIM_PAM_PRODUCT_LINE_D T1 --产线维表
on T.site_code = T1.site_code
and T.work_shop_code = T1.work_shop_code
and T.product_line_code = T1.product_line_code
left join DWRDIM.DWR_DIM_PAM_SITE_D T2 -- 基地维表
on T.site_code = T2.site_code
left join DWRDIM.DWR_DIM_PAM_WORK_SHOP_D T3 --车间维表
on T.site_code = T3.site_code 
and T.work_shop_code = T3.work_shop_code
