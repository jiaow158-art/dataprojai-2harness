WITH base as (--基地与日期笛卡尔积
select a.month_id AS STAT_MONTH,b.site_code from 
(select substr(day_dt,1,7) as month_id  
from dm.dm_dim_date_d  
where year_id = '${PERIOD_ID_Y}'
group by substr(day_dt,1,7) ) a 
left join (select site_code from DWRDIM.DWR_DIM_PAM_SITE_D where site_short_name <> '') b on 1=1 
)
,SAP AS ( --获取成本还原有生产的基地日期
SELECT T1.PLANT AS SITE_CODE,STAT_MONTH 
FROM DWRFIN.DWR_COST_RECOVERY_RESEARCH_T T1
LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T2 
ON T1.MATERIAL_NUM = T2.MATERIAL_NUM
WHERE SUBSTR(T1.PROD_VERSION,1,1) = 'R'
AND T2.MATL_TYPE_CODE = 'FERT'
AND T1.PROD_LEV = '优等品'
AND T1.SUM_AREA <> 0 
AND SUBSTR(T1.STAT_MONTH,1,4) = '${PERIOD_ID_Y}' 
GROUP BY PLANT,STAT_MONTH
)
,DATE_REP AS ( --获取无生产的基地日期，并且获取其最近有生产的日期
SELECT T3.site_code,t3.STAT_MONTH,min(T2.STAT_MONTH) AS N_STAT_MONTH
FROM BASE T3  
LEFT JOIN SAP T1
ON T1.SITE_CODE = T3.site_code
and t1.STAT_MONTH = t3.STAT_MONTH
JOIN SAP T2 
ON T3.SITE_CODE = T2.SITE_CODE
WHERE 1=1 AND T1.SITE_CODE IS NULL AND T2.STAT_MONTH > T3.STAT_MONTH
GROUP BY T3.site_code,t3.STAT_MONTH
)
,LOCKOUT_FEE_D AS (

 '00' AS FLAG ---生产线标识
,NVL(T.N_STAT_MONTH,T1.STAT_MONTH) AS STAT_MONTH
,T1.SITE_CODE 
,T1.SITE_NAME

,T1.PCS_SEG_NAME

,T1.COST_CENTER_NAME

,CASE WHEN T1.COST_ELEMENT_CODE = 'C' THEN T1.LOCKOUT_AMT ELSE 0 END AS ELEC_AMT--成本要素-电
,CASE WHEN T1.COST_ELEMENT_CODE = 'D' THEN T1.LOCKOUT_AMT ELSE 0 END AS WATER_AMT--成本要素-水
,CASE WHEN T1.COST_ELEMENT_CODE = 'H' THEN T1.LOCKOUT_AMT ELSE 0 END AS REPAIR_AMT--成本要素-修理
,CASE WHEN T1.COST_ELEMENT_CODE = 'J' THEN T1.LOCKOUT_AMT ELSE 0 END AS MAT_AMT--成本要素-物耗
,CASE WHEN T1.COST_ELEMENT_CODE = 'L' THEN T1.LOCKOUT_AMT ELSE 0 END AS LABOR_AMT--成本要素-人工
,CASE WHEN T1.COST_ELEMENT_CODE = 'M' THEN T1.LOCKOUT_AMT ELSE 0 END AS DAA_AMT--成本要素-折旧
-- ,T1.LOCKOUT_DAYS 
-- ,CASE WHEN T1.STAT_MONTH = SUBSTR(CURRENT_DATE,1,7) THEN DAY(DATE_ADD(CURRENT_DATE ,-1)) ELSE DAY(LAST_DAY(T1.STAT_MONTH||'-01')) END AS DAYS_IN_MONTH  
FROM DM.DM_PAM_BASE_PROD_LOCKOUT_FEE_T T1
LEFT JOIN DATE_REP T 
ON T.STAT_MONTH = T1.STAT_MONTH
AND T.SITE_CODE = T1.SITE_CODE
-- WHERE T1.PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
WHERE CASE WHEN T.N_STAT_MONTH IS NULL THEN T1.PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
ELSE T1.PERIOD_ID_M BETWEEN T.STAT_MONTH AND T.N_STAT_MONTH END

SELECT  --非生产线停产费用明细表
 T2.FLAG
,NVL(T.N_STAT_MONTH,T2.STAT_MONTH) AS STAT_MONTH
,T2.SITE_CODE 
,T2.SITE_NAME
,T2.S_PRODUCT_LINE_CODE
,T2.S_PRODUCT_LINE_NAME
,T2.PCS_SEG_CODE
,T2.PCS_SEG_NAME
,T2.COST_CENTER_CODE 
,T2.COST_CENTER_NAME
,CASE WHEN T2.COST_ELEMENT_CODE = 'B' THEN T2.LOCKOUT_AMT ELSE 0 END AS GAS_AMT--成本要素-气
,CASE WHEN T2.COST_ELEMENT_CODE = 'C' THEN T2.LOCKOUT_AMT ELSE 0 END AS ELEC_AMT--成本要素-电
,CASE WHEN T2.COST_ELEMENT_CODE = 'D' THEN T2.LOCKOUT_AMT ELSE 0 END AS WATER_AMT--成本要素-水
,CASE WHEN T2.COST_ELEMENT_CODE = 'H' THEN T2.LOCKOUT_AMT ELSE 0 END AS REPAIR_AMT--成本要素-修理
,CASE WHEN T2.COST_ELEMENT_CODE = 'J' THEN T2.LOCKOUT_AMT ELSE 0 END AS MAT_AMT--成本要素-物耗
,CASE WHEN T2.COST_ELEMENT_CODE = 'L' THEN T2.LOCKOUT_AMT ELSE 0 END AS LABOR_AMT--成本要素-人工
,CASE WHEN T2.COST_ELEMENT_CODE = 'M' THEN T2.LOCKOUT_AMT ELSE 0 END AS DAA_AMT--成本要素-折旧
-- ,T2.LOCKOUT_DAYS 
-- ,CASE WHEN T2.STAT_MONTH = SUBSTR(CURRENT_DATE,1,7) THEN DAY(DATE_ADD(CURRENT_DATE ,-1)) ELSE DAY(LAST_DAY(T2.STAT_MONTH||'-01')) END AS DAYS_IN_MONTH 
FROM DM.DM_PAM_BASE_NON_PROD_LOCKOUT_FEE_T T2
LEFT JOIN DATE_REP T 
ON T.STAT_MONTH = T2.STAT_MONTH
AND T.SITE_CODE = T2.SITE_CODE
-- WHERE T2.PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
WHERE CASE WHEN T.N_STAT_MONTH IS NULL THEN T2.PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
ELSE T2.PERIOD_ID_M BETWEEN T.STAT_MONTH AND T.N_STAT_MONTH END
