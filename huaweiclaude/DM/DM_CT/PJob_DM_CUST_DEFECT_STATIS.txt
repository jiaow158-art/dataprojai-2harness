WITH  TMP AS (
select 
T1.appeal_num,
T1.RESP_CONV, --责任方
T1.COST_AMB * T1.COMP_QTY as compensation_amt
from  
DWRCMP.DWR_CMP_CUST_COMPENSATE_INFO_F t1
union all  
select 
T1.appeal_num,
T1.RESP_CONV, --责任方
T1.compensation_amt
from  
DWRCMP.DWR_CMP_CUST_COMPENSATE_AMT_F  t1
)
, DM_CUST_DEFECT_STATIS_01 AS (
select
SUBSTRING(t1.approval_date,1,7) AS approval_month,
t1.appeal_num,
T2.RESP_CONV,
T3.qxfl3 AS defect_cat3, --缺陷分类3
t1.cust_comp_qty,
t1.resp_conv_qty,
case when   t1.resp_conv_qty = 0 or t1.resp_conv_qty is null then 0 else t1.cust_comp_qty/t1.resp_conv_qty end as resp_rate ,--客诉宗数按户数
coalesce(t2.compensation_amt,0) as compensation_amt
from
DWRCMP.DWR_CMP_CUST_COMPLAINT_INFO_F T1 
LEFT JOIN (		select
			ROW_NUMBER() OVER (
				PARTITION BY
					appeal_num
				ORDER BY
					compensation_amt desc
			) AS rn,
appeal_num,
RESP_CONV,
compensation_amt
		from
			TMP)   T2 
ON T1.appeal_num = T2.appeal_num
AND T2.RN = 1 
LEFT JOIN  
(select  distinct qxfl3,MAINID from  SDI.SDI_FORMTABLE_MAIN_565_dt1_1004) T3
ON T1.ID = T3.MAINID
where T1.cc_class = '质量问题'
AND SUBSTRING(t1.approval_date,1,4) >= '2023'
)
insert
	OVERWRITE table dm.dm_cust_defect_statis
SELECT 
approval_month,
RESP_CONV,-- 责任方
defect_cat3, --缺陷分类3
sum(resp_rate) as  cust_comp_qty_H ,--客诉宗数按户数
count(distinct t1.appeal_num) as cust_comp_qty_F, --客诉宗数按流程
current_timestamp as dw_last_update_date
from DM_CUST_DEFECT_STATIS_01 t1 
group by 
approval_month,
RESP_CONV,-- 责任方
defect_cat3 --缺陷分类3
