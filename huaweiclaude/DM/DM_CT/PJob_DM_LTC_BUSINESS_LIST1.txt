--商机预测
with business_forcast as(
    select 
    bsf.created_by
    ,bsf.business_id
    ,sum(use_quantity) as use_quantity 
    ,sum(estimated_amount) as estimated_amount 
    ,sum(estimated_year_deliver_amount) as estimated_year_deliver_amount 
    ,max(estimated_signed_date) as estimated_signed_date
    ,max(estimated_delivery_date) as estimated_delivery_date 
from STG.STG_BUSINESS_SALE_FORECAST_1033 bsf
join SDI.SDI_BUSINESS_OPPORTUNITY_1033 bo on bo.business_id  = bsf.business_id and bo.created_by = bsf.created_by 
where bsf.delete_flag = 0 
group by bsf.created_by,bsf.business_id
)
--商机升级信息记录
,business_operate_log as (
SELECT 
bo.business_id
,bol.current_business_stage
,bol.after_business_stage
,bol.upgrade_time
FROM SDI.SDI_business_operate_log_1031 BOl
LEFT JOIN SDI.SDI_BUSINESS_OPPORTUNITY_1033 BO ON BOL.BUSINESS_ID = BO.BUSINESS_ID
WHERE bol.delete_flag = 0
)
--项目评审用量
,project_review as (
    select sum(prmp.review_quotation * prmp.review_dosage) as review_dosage
            ,pr.business_code
            ,pr.project_code
    FROM SDI.SDI_PROJECT_REVIEW_1033 pr
    left join SDI.SDI_PROJECT_REVIEW_MATERIAL_PRICE_1033  prmp on prmp.project_id = pr.project_id and prmp.delete_flag = 0
    where pr.delete_flag = 0
    group by pr.business_code  ,pr.project_code
)
