WITH T AS(
SELECT STORE_ID
      ,COUNT(1) AS EMP_CNT
  FROM SDI.SDI_BUZ_EMPLOYEE_1030
 WHERE STATUS = 1
   AND EP_ID = 3044
   AND APP_ID = 1831
 GROUP BY STORE_ID
)
INSERT OVERWRITE TABLE DM.DM_DMT_STORE_STAT_T
SELECT A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LVL1_DEPT_NAME
	  ,B.LVL2_DEPT_NAME
	  ,B.LVL3_DEPT_NAME
	  ,B.LVL4_DEPT_NAME
	  ,B.LVL5_DEPT_NAME
	  ,B.LVL6_DEPT_NAME
	  ,A.START_DATE
	  ,A.END_DATE
	  ,COUNT(DISTINCT A.STORE_ID) AS STORE_COUNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,A.IS_OPEN
	  ,T.EMP_CNT --导购数
	  ,A.STORE_NAME --门店名称
  FROM DWIMD.DWI_MD_DMT_STORE_INFO_T A
  LEFT JOIN DWRDIM.DWR_DIM_DMT_DEPT_INFO_D B
    ON A.STORE_ID = B.STORE_ID
  LEFT JOIN T
    ON T.STORE_ID = A.STORE_ID  
 WHERE --A.STATUS = 1
    A.STORE_TYPE = 0 
 GROUP BY
       A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LVL1_DEPT_NAME
	  ,B.LVL2_DEPT_NAME
	  ,B.LVL3_DEPT_NAME
	  ,B.LVL4_DEPT_NAME
	  ,B.LVL5_DEPT_NAME
	  ,B.LVL6_DEPT_NAME
	  ,A.START_DATE
	  ,A.END_DATE
	  ,A.IS_OPEN
	  ,T.EMP_CNT
	  ,A.STORE_NAME --门店名称
;
