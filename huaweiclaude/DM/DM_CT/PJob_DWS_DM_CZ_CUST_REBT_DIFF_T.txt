DELETE FROM DM.DM_CZ_CUST_REBT_DIFF_T
 WHERE STAT_TYPE = '月度'
   AND STAT_DATE = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy-MM')
;

DELETE FROM DM.DM_CZ_CUST_REBT_DIFF_T
 WHERE STAT_TYPE = '季度'
   AND STAT_DATE = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ')
;

DELETE FROM DM.DM_CZ_CUST_REBT_DIFF_T
 WHERE STAT_TYPE = '半年度'
   AND STAT_DATE = CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('1','2')
                        THEN TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy')||'H1'
			            WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('3','4')
                        THEN TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy')||'H2'
			             END
;

DELETE FROM DM.DM_CZ_CUST_REBT_DIFF_T
 WHERE STAT_TYPE = '年度'
   AND STAT_DATE = TO_CHAR('${P_PERIOD_D}'::DATE,'yyyy')
;

WITH PARENT_CUSTOMER_INFO AS( --母客编信息
SELECT DISTINCT LPAD(CUSTOMER,10,'0') AS CUSTOMER,LPAD(PARENT_CUSTOMER,10,'0') AS PARENT_CUSTOMER,LEVEL4_CODE,LEVEL4,LEVEL3_CODE,LEVEL3
  FROM DM.CT_SALES_PERFORMANCE_DAILY_T A
 WHERE LPAD(CUSTOMER,10,'0') <> LPAD(PARENT_CUSTOMER,10,'0')
   AND CALDAY = TO_CHAR(TO_DATE(CURRENT_DATE - INTERVAL '1 DAY'),'yyyyMMdd')
   AND DOUBLE_COUNT = 0
   AND A.CUSTOMER IS NOT NULL
   AND LEVEL3_CODE = 'H04230600'
   AND INTEGRATE_CHANNEL_CODE = 'GD01'
),CUSTOMER_INFO1 AS( --客编信息
SELECT DISTINCT LPAD(CUSTOMER,10,'0') AS CUSTOMER,LPAD(PARENT_CUSTOMER,10,'0') AS PARENT_CUSTOMER,LEVEL4
               ,ROW_NUMBER() OVER(PARTITION BY LPAD(CUSTOMER,10,'0') ORDER BY CASE WHEN LPAD(CUSTOMER,10,'0') <> LPAD(PARENT_CUSTOMER,10,'0') THEN 1 ELSE 2 END ASC) AS RN
  FROM DM.CT_SALES_PERFORMANCE_DAILY_T A
 WHERE CALDAY = TO_CHAR(TO_DATE(CURRENT_DATE - INTERVAL '1 DAY'),'yyyyMMdd')
   AND LEVEL3_CODE = 'H04230600'
   AND INTEGRATE_CHANNEL_CODE = 'GD01'
),CUSTOMER_INFO AS( --客编信息
SELECT CUSTOMER,PARENT_CUSTOMER,LEVEL4
  FROM CUSTOMER_INFO1
 WHERE RN = 1
),SPEC_CUSTOMER_INFO AS(
SELECT DISTINCT "ACCOUNT_NUMBER" AS SPEC_CUST_NUM --一级B客编
  FROM DM.DM_RPT_ZMASTER_CUS_T
 WHERE "ACCOUNT_GROUP_CODE" IN('B1')
   AND "PARVW" = 'AG'
),T0 AS( --合同台账信息
SELECT DISTINCT 
       IS_SUB_CUST_NAME --是否为子客编描述
	  ,CNTC_STATUS_NAME --合同状态名称
	  ,IS_WF_FIL_NAME --流程是否归档描述（瓷砖）
	  ,CUST_TYPE_NAME --客户类型描述
	  ,CUST_NUM --客户编码
	  ,APLY_DATE --申请日期
	  ,CUST_ATTR_NAME --客户属性描述
	  ,LCGD_DATE AS GD_DATE --归档日期
  FROM DWRDIM.DWR_DIM_DEALER_CNTC_D
 WHERE COALESCE(IS_SUB_CUST_NAME,'否') = '否'
   AND DATA_SRC_NAME = '台账'
   AND CNTC_STATUS_NAME = '履约中'
   AND LEFT(APLY_DATE,4) = '2025'
   AND LEFT(CNTC_END_DATE,4) = '2025'
   AND IS_WF_FIL_NAME = '已归档'
   AND CUST_TYPE_NAME IN('零售商A','零售商B')
   AND SALE_RGN_NAME LIKE '%运营中心%'
),T1 AS( --经销商目标 第一步
SELECT 'GD01' AS CHANNEL_CODE
	  ,销售地区 AS NODE_DESC5
	  ,渠道 AS CAHNNEL_NAME
	  ,LPAD(客户编码,10,'0') CUST_NUM
      ,UNNEST(STRING_TO_ARRAY((
                                COALESCE(一月, '0') || ',' ||
                                COALESCE(二月, '0') ||
                                ',' ||
                                COALESCE(三月, '0') ||
                                ',' ||
                                COALESCE(四月, '0') ||
                                ',' ||
                                COALESCE(五月, '0') ||
                                ',' ||
                                COALESCE(六月, '0') ||
                                ',' ||
                                COALESCE(七月, '0') ||
                                ',' ||
                                COALESCE(八月, '0') ||
                                ',' ||
                                COALESCE(九月, '0') ||
                                ',' ||
                                COALESCE(十月, '0') ||
                                ',' ||
                                COALESCE(十一月, '0') ||
                                ',' ||
                                COALESCE(十二月, '0')),
                                          ',')) 目标,
    年份 || UNNEST(STRING_TO_ARRAY('01,02,03,04,05,06,07,08,09,10,11,12',','))  STAT_MONTH
  FROM UPLOAD.UPLOAD_ACHIEVEMENTCUSTOMER
 WHERE 年份 = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy')
   AND 渠道 = '零售渠道'
),T2 AS( --经销商目标
SELECT T1.CHANNEL_CODE
	  ,T1.NODE_DESC5
	  ,T1.CAHNNEL_NAME
	  --一级B子客编独立统计
	  ,CASE WHEN B.SPEC_CUST_NUM IS NOT NULL THEN T1.CUST_NUM
	        ELSE COALESCE(COALESCE(T.PARENT_CUSTOMER,A.PARENT_CUSTOMER),T1.CUST_NUM)
  			 END AS CUST_NUM --客户编码
	  ,SUM(CASE WHEN RIGHT(T1.STAT_MONTH,2) = TO_CHAR('${P_PERIOD_D}'::DATE,'MM')
                THEN CAST(T1.目标 AS NUMERIC(38,10))
                ELSE 0 END) AS MONTH_TARGET --月目标
	  ,SUM(CASE WHEN TO_CHAR((T1.STAT_MONTH || '01')::DATE,'yyyyQ') = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ')
     	        THEN CAST(T1.目标 AS NUMERIC(38,10))
                ELSE 0 END) AS QUARTER_TARGET --季度目标
	  ,SUM(CASE WHEN RIGHT(TO_CHAR((T1.STAT_MONTH || '01')::DATE,'yyyyQ'),1) IN('1','2')
	             AND RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('1','2')
     	        THEN CAST(T1.目标 AS NUMERIC(38,10))
				WHEN RIGHT(TO_CHAR((T1.STAT_MONTH || '01')::DATE,'yyyyQ'),1) IN('3','4')
	             AND RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('3','4')
     	        THEN CAST(T1.目标 AS NUMERIC(38,10))
                ELSE 0 END) AS HY_TARGET --半年目标
	  ,SUM(CAST(T1.目标 AS NUMERIC(38,10))) AS YEAR_TARGET --全年目标
  FROM T1
  LEFT JOIN CUSTOMER_INFO A
    ON T1.CUST_NUM = LTRIM(A.CUSTOMER,'0')
   AND T1.NODE_DESC5 = A.LEVEL4
  LEFT JOIN PARENT_CUSTOMER_INFO T
	ON T1.NODE_DESC5 = T.LEVEL4
   AND A.PARENT_CUSTOMER = T.CUSTOMER
  LEFT JOIN SPEC_CUSTOMER_INFO B
    ON T1.CUST_NUM = B.SPEC_CUST_NUM
 GROUP BY T1.CHANNEL_CODE
	  ,T1.NODE_DESC5
	  ,T1.CAHNNEL_NAME
	  ,CASE WHEN B.SPEC_CUST_NUM IS NOT NULL THEN T1.CUST_NUM
	        ELSE COALESCE(COALESCE(T.PARENT_CUSTOMER,A.PARENT_CUSTOMER),T1.CUST_NUM)
  			 END
),T3 AS(
SELECT CASE WHEN A.DATA_SOURCE IN ('S','T')
            THEN COALESCE(COALESCE(ZZ.SALES_GRP, Z."SALES_GROUP_CODE"),A.SALES_GRP)
            ELSE A.SALES_GRP
             END AS ORG_CODE --销售组编码
	  ,C.NODE_NAME5 --运营中心编码
      ,C.NODE_DESC5 --运营中心
      ,'零售渠道' AS CHANNEL_NAME --整合渠道
	  ,C.NODE_DESC6 --部门
	  ,C.NODE_DESC7 --销区
	  ,C.NODE_DESC8 --线组
	  --一级B子客编独立统计
	  ,CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN LPAD(A.CUSTOMER,10,'0')
	        ELSE COALESCE(COALESCE(LPAD(H.PARENT_CUSTOMER,10,'0'),LPAD(G.PARENT_CUSTOMER,10,'0')),LPAD(A.CUSTOMER,10,'0'))
 			 END AS CUST_NUM --客户编码
	  ,E.CUST_NAME --客户名称
	  ,CASE WHEN E1.CUSTOMER IS NOT NULL THEN '一类'
	        WHEN E2.CUSTOMER IS NOT NULL THEN '一类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 1000 THEN '一类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 500 THEN '二类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 120 THEN '三类'
			WHEN COALESCE(F.YEAR_TARGET,0) < 120 THEN '四类'
			 END AS CUST_CAT_NAME --客户类型
	  ,SUM(CASE WHEN CALMONTH = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy-MM')
	             AND COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001','$$$') --产品所有权=瓷砖或者产品所有权为空
	            THEN AMBPERFORMANCE
				ELSE 0
				 END
	      ) / 10000 AS MONTH_ACHIEVEMENT --月业绩
      ,SUM(CASE WHEN TO_CHAR((CALMONTH || '-01')::DATE,'yyyyQ') = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ')
	             AND COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001','$$$') --产品所有权=瓷砖或者产品所有权为空
	            THEN AMBPERFORMANCE
				ELSE 0
				 END
	      ) / 10000 AS QUARTER_ACHIEVEMENT --季度业绩
	  ,SUM(CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('1','2')
	             AND RIGHT(CALMONTH,2) IN('01','02','03','04','05','06')
				 AND COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001','$$$') --产品所有权=瓷砖或者产品所有权为空
	            THEN AMBPERFORMANCE
				WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('3','4')
	             AND RIGHT(CALMONTH,2) IN('07','08','09','10','11','12')
				 AND COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001','$$$') --产品所有权=瓷砖或者产品所有权为空
	            THEN AMBPERFORMANCE
				ELSE 0
				 END
	      ) / 10000 AS HY_ACHIEVEMENT --半年业绩
	  ,SUM(CASE WHEN COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001','$$$') --产品所有权=瓷砖或者产品所有权为空
	            THEN AMBPERFORMANCE
				ELSE 0
				 END) / 10000 AS YEAR_ACHIEVEMENT --年度业绩
	  ,SUM(CASE WHEN TO_CHAR((CALMONTH || '-01')::DATE,'yyyyQ') = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ')
	             AND COALESCE(RIGHT(A.material,2),'$$$') <> '_B' --物料编码剔除尾标未_B
	             AND COALESCE(A.INTEGRATE_CHANNEL2,'$$$') <> 'GD04' --整合品类≠特惠品
                 AND COALESCE(D.PROD_PROPERTY,'$$$') IN('A0001') --产品所有权=瓷砖
                 AND K.ZC_NAME IS NULL --剔除政策
	            THEN AMBPERFORMANCE
				ELSE 0
				 END
	      ) / 10000 AS QUARTER_REBT_ACHIEVEMENT --季度可返利业绩
	  ,SUM(CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('1','2')
	             AND RIGHT(CALMONTH,2) IN('01','02','03','04','05','06')
	             AND COALESCE(A.INTEGRATE_CHANNEL2,'$$$') <> 'GD04' --整合品类≠特惠品
				 AND COALESCE(RIGHT(A.material,2),'$$$') <> '_B' --物料编码剔除尾标未_B
                 AND D.PROD_PROPERTY = 'A0001' --产品所有权=瓷砖
                 AND K.ZC_NAME IS NULL --剔除政策
	            THEN AMBPERFORMANCE
				WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyyQ'),1) IN('3','4')
	             AND RIGHT(CALMONTH,2) IN('07','08','09','10','11','12')
	             AND COALESCE(A.INTEGRATE_CHANNEL2,'$$$') <> 'GD04' --整合品类≠特惠品
				 AND COALESCE(RIGHT(A.material,2),'$$$') <> '_B' --物料编码剔除尾标未_B
                 AND D.PROD_PROPERTY = 'A0001' --产品所有权=瓷砖
                 AND K.ZC_NAME IS NULL --剔除政策
	            THEN AMBPERFORMANCE
				ELSE 0
				 END
	      ) / 10000 AS HY_REBT_ACHIEVEMENT --半年可返利业绩
	  ,SUM(CASE WHEN COALESCE(A.INTEGRATE_CHANNEL2,'$$$') <> 'GD04' --整合品类≠特惠品
                 AND D.PROD_PROPERTY = 'A0001' --产品所有权=瓷砖
				 AND COALESCE(RIGHT(A.material,2),'$$$') <> '_B' --物料编码剔除尾标未_B
                 AND K.ZC_NAME IS NULL --剔除政策
				THEN AMBPERFORMANCE
				ELSE 0
				 END
		   ) / 10000 AS YEAR_REBT_ACHIEVEMENT --年度可返利业绩
	  ,COALESCE(F.MONTH_TARGET,0) AS MONTH_TARGET --月目标
	  ,COALESCE(F.QUARTER_TARGET,0) AS QUARTER_TARGET --季度目标
	  ,COALESCE(F.HY_TARGET,0) AS HY_TARGET --半年目标
	  ,COALESCE(F.YEAR_TARGET,0) AS YEAR_TARGET --全年目标
  FROM DM.DM_AMBV2_ZZT_XSSR_T A
  LEFT JOIN DM.DM_RPT_ZMASTER_CUS_T Z
    ON A.CUSTOMER = Z."ACCOUNT_NUMBER"
   AND A.DISTR_CHAN = Z."SALES_CHANNEL_CODE"
   AND A.SALESORG = Z."SALES_ORGANIZATION_CODE"
   AND Z."PARVW" = 'AG'
  LEFT JOIN (SELECT DISTINCT WBS_LEVEL3 AS WBS_NUM,SALES_GRP
               FROM DWRDIM.DWR_DIM_WBS_BASIS_INFO_F) ZZ
    ON A.WBS_ELEMT = ZZ.WBS_NUM
  LEFT JOIN DM.DM_RPT_SALES_GROUP_T C
    ON CASE WHEN A.DATA_SOURCE IN ('S', 'T')
            THEN COALESCE(COALESCE(ZZ.SALES_GRP, Z."SALES_GROUP_CODE"),A.SALES_GRP)
            ELSE A.SALES_GRP
             END = C.NODE_NAME9
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T D
    ON A.MATERIAL = D.MATERIAL_NUM
  LEFT JOIN CUSTOMER_INFO G
    ON LPAD(A.CUSTOMER,10,'0') = G.CUSTOMER
   AND C.NODE_DESC5 = G.LEVEL4
  LEFT JOIN PARENT_CUSTOMER_INFO H
	on C.NODE_DESC5 = H.LEVEL4
   AND LPAD(G.PARENT_CUSTOMER,10,'0') = H.CUSTOMER
  LEFT JOIN SPEC_CUSTOMER_INFO I
    ON LPAD(A.CUSTOMER,10,'0') = I.SPEC_CUST_NUM
  LEFT JOIN (SELECT DISTINCT "ACCOUNT_NUMBER" AS CUSTOMER
                            ,"DESCRIPTION" AS CUST_NAME
               FROM DM.DM_RPT_ZMASTER_CUS_T
			  WHERE "PARVW" = 'AG'   
			) E
    ON CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN LPAD(A.CUSTOMER,10,'0')
	        ELSE COALESCE(COALESCE(LPAD(H.PARENT_CUSTOMER,10,'0'),LPAD(G.PARENT_CUSTOMER,10,'0')),LPAD(A.CUSTOMER,10,'0'))
 			 END = E.CUSTOMER
  LEFT JOIN (SELECT DISTINCT "ACCOUNT_NUMBER" AS CUSTOMER
               FROM DM.DM_RPT_ZMASTER_CUS_T
			  WHERE "PARVW" = 'AG'
                AND "accnt_group8" = '强商' --客户组8			  
			) E1
    ON CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN LPAD(A.CUSTOMER,10,'0')
	        ELSE COALESCE(COALESCE(LPAD(H.PARENT_CUSTOMER,10,'0'),LPAD(G.PARENT_CUSTOMER,10,'0')),LPAD(A.CUSTOMER,10,'0'))
 			 END = E1.CUSTOMER
  LEFT JOIN (SELECT DISTINCT "ACCOUNT_NUMBER" AS CUSTOMER
               FROM DM.DM_RPT_ZMASTER_CUS_T
			  WHERE "PARVW" = 'AG'
                AND SHARED_WAREHOUSE = '06A' --共享仓			  
			) E2
    ON CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN A.CUSTOMER
	        ELSE COALESCE(COALESCE(H.PARENT_CUSTOMER,G.PARENT_CUSTOMER),LPAD(A.CUSTOMER,10,'0'))
 			 END = E2.CUSTOMER	
  LEFT JOIN T2 F
	ON CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN LPAD(A.CUSTOMER,10,'0')
	        ELSE COALESCE(COALESCE(LPAD(H.PARENT_CUSTOMER,10,'0'),LPAD(G.PARENT_CUSTOMER,10,'0')),LPAD(A.CUSTOMER,10,'0'))
 			 END = F.CUST_NUM
   AND C.NODE_DESC5 = F.NODE_DESC5
  LEFT JOIN UPLOAD.UOLOAD_CZ_ZC_REBT_DIFF_T K
    ON A.ZCMC = K.ZC_NAME
   AND K.TYPE_NAME = '零售返利'
 WHERE LEFT(A.CALMONTH,4) = TO_CHAR('${P_PERIOD_D}'::DATE, 'yyyy')
   AND CALDAY < TO_CHAR(CURRENT_DATE, 'yyyyMMdd')
   AND A.INTEGRATE_CHANNEL = 'GD01' --零售渠道
   AND C.NODE_NAME4 IN('H04230600','H04230655') --区域运营中心、国内直营中心
 GROUP BY ORG_CODE --销售组编码
	  ,C.NODE_NAME5 --运营中心编码
      ,C.NODE_DESC5 --运营中心
	  ,C.NODE_DESC6 --部门
	  ,C.NODE_DESC7 --销区
	  ,C.NODE_DESC8 --线组
	  ,CASE WHEN I.SPEC_CUST_NUM IS NOT NULL THEN LPAD(A.CUSTOMER,10,'0')
	        ELSE COALESCE(COALESCE(LPAD(H.PARENT_CUSTOMER,10,'0'),LPAD(G.PARENT_CUSTOMER,10,'0')),LPAD(A.CUSTOMER,10,'0'))
 			 END --客户编码
	  ,E.CUST_NAME --客户名称
	  ,CASE WHEN E1.CUSTOMER IS NOT NULL THEN '一类'
	        WHEN E2.CUSTOMER IS NOT NULL THEN '一类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 1000 THEN '一类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 500 THEN '二类'
			WHEN COALESCE(F.YEAR_TARGET,0) >= 120 THEN '三类'
			WHEN COALESCE(F.YEAR_TARGET,0) < 120 THEN '四类'
			 END --客户类型
	  ,COALESCE(F.MONTH_TARGET,0) --月目标
	  ,COALESCE(F.QUARTER_TARGET,0) --季度目标
	  ,COALESCE(F.HY_TARGET,0) --半年目标
	  ,COALESCE(F.YEAR_TARGET,0) --全年目标
),T4 AS( --业绩完成率
SELECT ORG_CODE --销售组编码
	  ,NODE_NAME5 --运营中心编码
      ,NODE_DESC5 --运营中心
      ,CHANNEL_NAME --整合渠道
	  ,NODE_DESC6 --部门
	  ,NODE_DESC7 --销区
	  ,NODE_DESC8 --线组
	  ,CUST_NUM --客户编码
	  ,CUST_NAME --客户名称
	  ,CUST_CAT_NAME --客户类型
	  ,MONTH_ACHIEVEMENT --月业绩
      ,QUARTER_ACHIEVEMENT --季度业绩
	  ,HY_ACHIEVEMENT --半年业绩
	  ,YEAR_ACHIEVEMENT --年度业绩
      ,QUARTER_REBT_ACHIEVEMENT --季度可返利业绩
	  ,HY_REBT_ACHIEVEMENT --半年可返利业绩
	  ,YEAR_REBT_ACHIEVEMENT --年度可返利业绩	  
	  ,MONTH_TARGET --月目标
	  ,QUARTER_TARGET --季度目标
	  ,HY_TARGET --半年目标
	  ,YEAR_TARGET --全年目标
	  ,CASE WHEN QUARTER_TARGET = 0 THEN 0 ELSE QUARTER_ACHIEVEMENT / QUARTER_TARGET END AS QUARTER_RATE --季度完成率
	  ,CASE WHEN HY_TARGET = 0 THEN 0 ELSE HY_ACHIEVEMENT / HY_TARGET END AS HY_RATE --半年完成率
	  ,CASE WHEN YEAR_TARGET = 0 THEN 0 ELSE YEAR_ACHIEVEMENT / YEAR_TARGET END AS YEAR_RATE --全年完成率
  FROM T3
),T5 AS( --返利金额
SELECT T4.ORG_CODE --销售组编码
	  ,T4.NODE_NAME5 --运营中心编码
      ,T4.NODE_DESC5 --运营中心
      ,T4.CHANNEL_NAME --整合渠道
	  ,T4.NODE_DESC6 --部门
	  ,T4.NODE_DESC7 --销区
	  ,T4.NODE_DESC8 --线组
	  ,T4.CUST_NUM --客户编码
	  ,T4.CUST_NAME --客户名称
	  ,T4.CUST_CAT_NAME --客户类型
	  ,T4.MONTH_ACHIEVEMENT --月业绩
      ,T4.QUARTER_ACHIEVEMENT --季度业绩
	  ,T4.HY_ACHIEVEMENT --半年业绩
	  ,T4.YEAR_ACHIEVEMENT --年度业绩
      ,T4.QUARTER_REBT_ACHIEVEMENT --季度可返利业绩
	  ,T4.HY_REBT_ACHIEVEMENT --半年可返利业绩
	  ,T4.YEAR_REBT_ACHIEVEMENT --年度可返利业绩
	  ,T4.MONTH_TARGET --月目标
	  ,T4.QUARTER_TARGET --季度目标
	  ,T4.HY_TARGET --半年目标
	  ,T4.YEAR_TARGET --全年目标
	  ,T4.QUARTER_RATE --季度完成率
	  ,T4.HY_RATE --半年完成率
	  ,T4.YEAR_RATE --全年完成率
	  ,CASE WHEN T4.QUARTER_RATE < 0.85 THEN 0
	        WHEN T4.QUARTER_RATE < 1 THEN T4.QUARTER_REBT_ACHIEVEMENT * T.AMT_RATE * 0.8
			WHEN T4.QUARTER_RATE >= 1 THEN T4.QUARTER_REBT_ACHIEVEMENT * T.AMT_RATE
			ELSE 0
			 END AS QUARTER_REBT_AMT --季度返利金额
	  ,CASE WHEN T4.HY_RATE < 0.85 THEN 0
	        WHEN T4.HY_RATE < 1 THEN T4.HY_REBT_ACHIEVEMENT * T.AMT_RATE * 0.8
			WHEN T4.HY_RATE >= 1 THEN T4.HY_REBT_ACHIEVEMENT * T.AMT_RATE
			ELSE 0
			 END AS HY_REBT_AMT --半年返利金额
	  ,CASE WHEN T4.YEAR_RATE < 0.85 THEN 0
	        WHEN T4.YEAR_RATE < 1 THEN T4.YEAR_REBT_ACHIEVEMENT * 0.01 * 0.8
			WHEN T4.YEAR_RATE >= 1 THEN T4.YEAR_REBT_ACHIEVEMENT * 0.01
			ELSE 0
			 END AS YEAR_REBT_AMT --全年返利金额
