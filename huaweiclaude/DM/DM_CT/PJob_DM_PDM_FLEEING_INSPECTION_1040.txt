INSERT OVERWRITE  TABLE DM.DM_PDM_FLEEING_INSPECTION_1040  
SELECT 
t1.barcode,--隐形码
t1.check_time,--查询时间
t1.house_name,  --库存地点
t1.out_customer_code, --提货客户编码
t1.out_customer_name, --客户名称
t1.check_user_name,--查询人
t1.check_address,--查询所在地点
count(t1.examine_id) as cheek_quantity , --查询数
SUM(CASE WHEN T1.status = '0' OR status = '1' THEN 1 ELSE 0 END) AS   abnormal_quantity,--异常数
SUM(CASE WHEN T1.status = '1' THEN 1 ELSE 0 END) AS Fleeing_quantity --窜货数
from SDI.SDI_PDM_FLEEING_INSPECTION_1040 T1 
GROUP BY t1.barcode,--隐形码
t1.check_time,--查询时间
t1.house_name,  --库存地点
t1.out_customer_code, --提货客户编码
t1.out_customer_name,
t1.check_user_name,
t1.check_address
