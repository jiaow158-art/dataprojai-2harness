-- 可用库存
with tmp as(
	-- 总库 - Q库存-留备货
	select t1.material_num ,
		   t1.batch_num ,
		   t1.factory_werks_code as FACTORY_CODE,
		   t1.inv_location,
		   sum(t1.stock_qty) - sum(t1.q_stock) as tmp_stock,   -- 总库 - Q库存
		   sum(zkkc) as tmp_stock1,
		   sum(djkc+zjkc) as tmp_stock2
	from dm.dm_wm_stock_responsibility_total_tmp t1
	where t1.stat_month = substr('${PERIOD_ID_D}',1,7)
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.factory_werks_code,
			 t1.inv_location
	union all 
	select t1.material_num ,
		   t1.batch_num ,
		   t1.FACTORY_CODE,
		   t1.inv_location,
		   0-(sum(res_stock_qty)+sum(sku_prepare_qty)) as tmp_stock,   -- 减去留备货
		   0-(sum(res_stock_qty)+sum(sku_prepare_qty)) as tmp_stock1,
		   0 
		   from dm.dm_wm_stock_responsibility_res_prepare_tmp t1
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.FACTORY_CODE,
			 t1.inv_location
)
insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_LEFT_0_TMP 
select t1.material_num ,
	   t1.batch_num ,
	   t1.FACTORY_CODE,
	   t1.inv_location,
       greatest(sum(tmp_stock),0) as left_stock_qty ,
       greatest(sum(tmp_stock1),0) as left_zkkc , 
       greatest(sum(tmp_stock2),0) as left_djzjkc,
	   current_timestamp() as dw_last_update_date
from tmp t1
join dm.dm_wm_stock_responsibility_mat_tmp t4 
On t1.material_num  = t4.material_num 
group by t1.material_num ,
		 t1.batch_num ,
		 t1.FACTORY_CODE,
		 t1.inv_location;
