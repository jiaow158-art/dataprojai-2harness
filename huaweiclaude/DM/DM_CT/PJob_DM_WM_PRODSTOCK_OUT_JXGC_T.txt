with get_ys_price as 
(
select  
    operation_desc,coalesce(prod_level3_name,'其他品类') as prod_level3_name,coalesce(dimension,'其他规格') as dimension,
    sum(coalesce(prod_it_amt1,0)) as prod_it_amt1,
    sum(coalesce(sales_area,0)) as sales_area,
    sum(coalesce(prod_it_amt1,0))/sum(coalesce(sales_area,0)) as ys_price --预算全年品类规格单价
FROM SDI.SDI_ACHIEVEMENT_BUDGET_T_1037
where substring(stat_month,1,4) = substring(current_date(),1,4)
group by operation_desc,coalesce(prod_level3_name,'其他品类'),coalesce(dimension,'其他规格')
)
