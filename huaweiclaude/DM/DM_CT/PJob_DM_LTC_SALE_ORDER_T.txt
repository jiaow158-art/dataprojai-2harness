INNER JOIN SDI.SDI_S_ORDER_ITEM_1002 B
ON A.ROW_ID = B.ORDER_ID
AND COALESCE(B.X_ITEM_STATUS_CD,'') NOT IN('无效','失效')
AND B.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_ORDER_ITEM_X_1002 C
ON B.ROW_ID = C.ROW_ID
AND C.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_PROD_INT_1002 D
ON B.PROD_ID = D.ROW_ID
AND D.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_PROJ_REP_1002 E
ON A.X_PROJ_REP_ID = E.ROW_ID
AND E.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_ORG_EXT_1002 F
ON A.ACCNT_ID = F.PAR_ROW_ID
AND F.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_SALES_CHAN_1002 G
ON A.X_SALES_CHAN_ID = G.ROW_ID
AND G.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_SALES_ORG_1002 H
ON A.X_ORG_ID = H.ROW_ID
AND H.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 J
ON A.X_CITY = J.NAME
AND J.`TYPE` = 'DP_CITY_TYPE'
AND J.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 I
ON A.X_PROV = I.NAME
AND I.`TYPE` = 'STATE_ABBREV'
AND I.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 K
ON A.X_OD_TYPE = K.NAME
AND K.`TYPE` = 'DP_ORDER_TYPE'
AND K.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 M
ON A.X_OD_SUBTYPE = M.NAME
AND M.`TYPE` = 'DP_ORDER_SUB_TYPE'
AND M.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_ORG_EXT_1002 N
ON A.X_PAYER_ID = N.PAR_ROW_ID
AND N.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 O
ON A.X_PAY_TYPE = O.NAME
AND O.`TYPE` = 'DP_PAYMENT_TYPE'
AND O.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_SALES_CHAN_1002 P
ON A.X_CHAN_ID = P.ROW_ID
AND P.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_ZMASTER_CUS_1001 Q 
ON F.OU_NUM = Q.ACCOUNT_NUMBER
AND Q.MANDT = 800
AND H.SALES_ORG_CD = Q.SALES_ORGANIZATION_CODE
AND G.SALES_CHAN_CD = Q.SALES_CHANNEL_CODE
and Q.PARVW = 'AG'
AND Q.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_ORDER_X_1002 R
ON A.ROW_ID = R.ROW_ID
WHERE A.DEL_FLAG = 'N'
AND SUBSTR(A.CREATED,1,4) >= '2021'
