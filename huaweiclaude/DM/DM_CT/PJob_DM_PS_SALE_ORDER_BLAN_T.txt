--插入数据到产销管理报表DM层
insert overwrite table dm.dm_ps_sale_order_blan_t
SELECT  
        case when x.Detail_State = '0' then '正常'
             when x.Detail_State = '15' then '无效'
              end as Detail_State
       ,x.CUSTOMER_NAME
       ,x.CUSTOMER_CODE
       ,x.ORDER_TYPE
       ,x.SALE_ORDER_CODE
       ,x.SALE_ORDER_LINE_NO
       ,COALESCE(x.Item_Group_One,'') as ITEM_GROUP_ONE
       ,x.ITEM_GROUP_TWO
       ,x.ITEM_CATEGORY
       ,x.CUSTOM_FLAG
       ,x.FULL_ORDER_FLAG
       ,x.ITEM_NAME as material_name
       ,x.ITEM_CODE as material_num
       ,x.SPECIFICATION_MODEL
       ,x.MAKE_TIME
       ,x.OUTPUT_TIME
       ,x.DEMAND_DELIVERY_DATE
       ,x.UNIT
       ,x.COMPLETE_TIME
       ,x.PRODUCT_ORDER_ID
       ,x.PRODUCT_ORDER_CODE
       ,x.PRODUCT_ORDER_QTY
       ,x.PRODUCT_ORDER_AREA
       ,x.SALE_ORDER_QTY
       ,x.SALE_ORDER_AREA
       ,x.SALE_ORDER_COMPLETE_QTY
       ,x.SALE_ORDER_COMPLETE_AREA
       ,x.RELATE_CODE
       ,x.RELATE_QTY
       ,x.RELATE_AREA --计划面积
       ,x.RELATE_COMPLETE_QTY
       ,x.RELATE_COMPLETE_AREA --计划完成面积
       ,x.CANCEL_QTY --合并后生产工单取消数量
       ,x.CANCEL_AREA
       ,x.RESOURCE_ID
       ,x.PLANT_CLASS
       ,x.CHANGE_TYPE
       ,x.REPORT_START_TIME
       ,x.REPORT_END_TIME
       ,x.PLAN_START_TIME
       ,x.PLAN_END_TIME
       ,COALESCE(x.PLAN_DELAY,0) as PLAN_DELAY
       ,x.ORDER_DELAY
	   ,case when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '0' then '未生成'
             when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '10' then '已生成'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '15' then '无效'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '20' then '挂起'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '24' then '部分下达'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '26' then '已下达'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '28' then '部分完成'
	         when Nvl(x.ORDER_STATE,x.PRODUCT_ORDER_STATE) = '30' then '已完成'
	          end as Order_State--订单状态
       ,X1.Code AS Resource_Code
       ,X2.Code AS Work_Shop_Code
       ,X3.Code AS factory_code
       ,X4.Code AS Product_Area_Code
       ,X1.Name AS Resource_Name
       ,X2.Name AS Work_Shop_Name --车间
       ,X3.Name AS factory_name
       ,X4.Name AS Product_Area_Name
       ,CASE WHEN (x.Product_Order_Area - x.Sale_Order_Complete_Area) < 0 THEN 0
             ELSE x.Product_Order_Area - x.Sale_Order_Complete_Area
              END AS Sale_Order_Not_Complete_Area
       ,CASE WHEN (x.Relate_Area - x.Relate_Complete_Area - x.Cancel_Area) < 0 THEN 0
             ELSE x.Relate_Area - x.Relate_Complete_Area - x.Cancel_Area
              END AS Relate_Not_Complete_Area
	   ,(cast(COALESCE(x.Relate_Area,0) as DECIMAL(38,10)) - cast(COALESCE(x.Cancel_Area,0) as DECIMAL(38,10))) as area	
       ,case when cast(COALESCE(Plan_Delay,0) as int) <= 3
              and COALESCE(Item_Group_One,'') in('A','B')
              and cast(RELATE_COMPLETE_AREA as DECIMAL(38,10)) >= (cast(Relate_Area as DECIMAL(38,10)) - cast(COALESCE(Cancel_Area,0) as DECIMAL(38,10))) * 0.9 then '是'
             when cast(COALESCE(Plan_Delay,0) as int) <= 3
              and COALESCE(Item_Group_One,'') not in('A','B')
              and cast(RELATE_COMPLETE_AREA as DECIMAL(38,10)) >= (cast(Relate_Area as DECIMAL(38,10)) - cast(COALESCE(Cancel_Area,0) as DECIMAL(38,10))) then '是'
             else '否'
              end as is_zj
	   ,X6.VTEXT AS Category
	   ,current_timestamp() as DW_LAST_UPDATE_DATE
  FROM (SELECT t.Id
              ,t.Inv_Org_Id
              ,t.Detail_State --订单状态 111
              ,t.Sale_Order_Line_No --销售订单明细行号
              ,t.Sap_Code --sap单号
              ,t.Sap_Line_No --sap行号
              ,t.Item_Group_Two --物料组2
              ,t.Custom_Flag --是否定制
              ,t.Sequence_No --订单顺序号
              ,t.Product_Area_Id ---产区
              ,t.Plant_Class --工厂分类
              ,t.Output_Time --工厂交期
              ,t.Demand_Delivery_Date
             -- ,t.Unit --主单位
              ,t.Sale_Order_Qty --订单数量
              ,t.Sale_Order_Qty * T3.Area AS Sale_Order_Area --订单面积
              ,t.Complete_Time --完成时间
              ,T2.Name AS Customer_Name --客户名称
              ,T2.CUST_NUM AS Customer_Code --客户编码
              ,t.Order_Type --订单类型
              ,t.Sale_Order_Code --销售订单号
              ,t.Full_Order_Flag --整单标识
              ,t.Make_Time --订单下单时间
              ,T3.Name AS Item_Name --物料名称
              ,T3.CODE AS Item_Code --物料编码
              ,T3.Specification_Model --物料规格
              ,T3.Product_Property AS Item_Group_One --物料组1
              ,T3.Category AS Item_Category --物料品类
              ,T4.Name AS Unit --主单位
              ,t.Product_Order_Code --合并前生产工单号
              ,t.Qty AS Product_Order_Qty --合并前生产工单数量
              ,t.Qty * T3.Area AS Product_Order_Area --合并前生产工单数量
              ,t.Complete_Qty AS Sale_Order_Complete_Qty --销售订单完成数量
              ,t.Complete_Qty * T3.Area AS Sale_Order_Complete_Area --销售订单完成面积
              ,t.Change_Type --拆合类型
              ,t.Order_State AS Product_Order_State --合并前生产订单状态
              ,t.Data_Source
              ,T5.Id AS Product_Order_Id --合并后生产ID
              ,T5.ORDER_NUM AS Relate_Code --合并后生产工单号
              ,T5.Qty AS Relate_Qty --合并后生产工单数量
              ,T5.Qty * T3.Area AS Relate_Area --合并后生产工单面积
              ,T5.Complete_Qty AS Relate_Complete_Qty --合并后生产工单完成数量
              ,T5.Complete_Qty * T3.Area AS Relate_Complete_Area --合并后生产工单完成面积
              ,T5.Cancel_Qty --合并后生产工单取消数量
              ,T5.Cancel_Qty * T3.Area AS Cancel_Area --合并后生产工单取消面积  111
              ,L.Plan_Start_Time
              ,L.Plan_End_Time
              ,L.Report_Start_Time
              ,L.Report_End_Time
              ,L.Delay_Days AS Plan_Delay
              ,t.Order_Delay
			  ,t.MATERIAL_ID
              ,T7.Resource_Id AS Resource_Id --生产资源id
              ,T5.Order_State --工单状态 111
          FROM (SELECT                 t.Id
                      ,t.Inv_Org_Id
                      ,t.Detail_State --订单状态
                      ,t.Line_No              AS Sale_Order_Line_No --销售订单明细行号
                      ,t.Sap_Code --sap单号
                      ,t.Sap_Line_No --sap行号
                      ,t.MATERIAL_GROUP_CODE2             AS Item_Group_Two --物料组2
                      ,t.CUSTOM_FLAG --是否定制
                      ,t.Sequence_No --订单顺序号
                      ,t.Product_Area_Id ---产区
                      ,t.Plant_Class --工厂分类
                      ,t.Output_Time --工厂交期
                      ,t.Demand_Delivery_Date
                      ,t.Unit --主单位
                      ,t.Net_Demand_Qty       AS Sale_Order_Qty --订单数量
                     -- ,t.Net_Demand_Qty * T3.Area AS Sale_Order_Area --订单面积
                      ,t.Complete_Time --完成时间
                      ,t.SRC_MATERIAL_ID AS Item_Id--物料 ID
					  ,t.MATERIAL_ID
                      ,t.MATERIAL_NUM
                      ,T1.SRC_CUST_ID -- 客户ID
                      ,T1.CUST_ID
                      ,T1.CUST_NUM
                      ,T1.Order_Type --订单类型
                      ,T1.Oss_Plan_Num    AS Sale_Order_Code --销售订单号
                      ,T1.Full_Order_Flag --整单标识
                      ,T1.Make_Time --订单下单时间
                      ,t.Relate_No        AS Product_Order_Code --合并前生产订单号
                      ,T4.Relevant_Code --相关单号 111
                      ,T4.Qty --数量
                      ,T4.Complete_Qty --完成数量
                      ,T4.Change_Type --拆合类型
                      ,T4.Order_State --合并前生产订单状态
                      ,11                 AS Data_Source --来源类型
                      ,L.Delay_Days       AS Order_Delay
                  FROM DWROTD.DWR_PS_SALE_ORDER_DTL_F T -- 销售订单明细  t
                  LEFT JOIN DWROTD.DWR_PS_SALE_ORDER_F T1 --销售订单主表  t1
                    ON t.Sale_Order_Id = T1.Id
                   AND T1.Is_Phantom = '0'
				   AND T1.del_flag = 'N'
                  LEFT JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F T4 --合并前生产订单  t4
                    ON t.Product_Order_Id = T4.Id
                   AND T4.Is_Phantom = '0'
				   AND T4.del_flag = 'N'
                  LEFT JOIN sdi.sdi_sale_order_delay_1007 L
                    ON L.SALE_DETAIL_ID=T.ID
                   AND L.Inv_Org_Id = T.Inv_Org_Id
                   AND L.Is_Phantom = '0'
				   AND L.del_flag = 'N'
                 WHERE T.Is_Phantom = '0'
				   AND T.del_flag = 'N'
                UNION ALL
                SELECT Manupro.Id
                      ,Manupro.Inv_Org_Id
                      ,Manupro.Order_State as Detail_State--订单状态
                      ,NULL                     AS Sale_Order_Line_No --销售订单明细行号
                      ,NULL                     AS Sap_Code --sap单号
                      ,NULL                     AS Sap_Line_No --sap行号
                      ,NULL                     AS Item_Group_Two --物料组2
                      ,Manupro.Is_Custom        AS Custom_Flag --是否定制
                      ,NULL                     AS Sequence_No --订单顺序号
                      ,Manupro.Factory_Id       AS Product_Area_Id ---产区
                      ,NULL                     AS Plant_Class --工厂分类
                      ,Manupro.Promise_Delivery AS Output_Time --工厂交期
                      ,Manupro.Require_Delivery AS Demand_Delivery_Date
                      ,NULL                     AS Unit --主单位
                      ,Manupro.ORDER_QTY              AS Sale_Order_Qty --订单数量
                       --,Manupro.Qty * T3.Area AS Sale_Order_Area --订单面积
                      ,NULL                AS Complete_Time --完成时间
                      ,Manupro.SRC_MATERIAL_ID AS Item_Id --物料 ID
                      ,Manupro.MATERIAL_ID
                      ,Manupro.MATERIAL_NUM
                      ,Manupro.SRC_CUST_ID -- 客户ID
                      ,Manupro.CUST_ID
                      ,Manupro.CUST_NUM
                      ,Ots.Name as order_type--订单类型
                      ,NULL                AS Sale_Order_Code --销售订单号
                      ,Manupro.Is_Whole    AS Full_Order_Flag --整单标识
                      ,Manupro.Order_Date as Make_Time--订单下单时间
                      ,Manupro.ORDER_NUM        AS Product_Order_Code --合并前生产订单号
                      ,T4.Relevant_Code --相关单号
                      ,T4.Qty --数量
                      ,T4.Complete_Qty --完成数量
                      ,T4.Change_Type --拆合类型
                      ,T4.Order_State --合并前生产订单状态
                      ,10                  AS Data_Source --来源类型
                      ,0                   AS Order_Delay
                  FROM DWROTD.DWR_PS_MANUAL_PROD_ORDER_F Manupro -- 手工生产订单
                  LEFT JOIN DWROTD.DWR_PS_ORDER_TYPE_SETTING_F Ots
                    ON Manupro.Order_Type_Set_Id = Ots.Id
                   AND Ots.Is_Phantom = '0'
				   AND OTS.del_flag = 'N'
                  LEFT JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F T4 --合并前生产订单  t4
                    ON Manupro.ORDER_NUM = T4.ORDER_NUM
                   AND T4.Inv_Org_Id = Manupro.Inv_Org_Id
                   AND T4.Is_Phantom = '0'
				   AND T4.del_flag = 'N'
                 WHERE Manupro.Is_Phantom = '0'
				   AND Manupro.del_flag = 'N') t
          LEFT JOIN DWIOTD.DWI_PS_CSM_CUSTOMER_T T2 --客户  t2
            ON t.SRC_CUST_ID = T2.Id
           AND T2.Is_Phantom = '0'
		   AND T2.del_flag = 'N'
          LEFT JOIN sdi.sdi_item_1007 T3 --物料  t3
            ON T.Item_Id = T3.Id
           AND T3.Is_Phantom = '0'
		   AND T3.DEL_FLAG = 'N'
          LEFT JOIN sdi.sdi_bd_unit_1007 T4 --单位  t4
            ON T3.Unit_Id = T4.Id
           AND T4.Is_Phantom = '0'
		   AND T4.DEL_FLAG = 'N'
          LEFT JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F T5 --合并后生产订单  t5
            ON T.Relevant_Code = T5.ORDER_NUM
           AND T.Inv_Org_Id = T5.Inv_Org_Id
           AND T5.Is_Phantom = '0'
		   AND T5.del_flag = 'N'
          LEFT JOIN sdi.sdi_aps_pro_order_delay_1007 L
            ON L.PRODUCT_ORDER_ID = T5.ID
           AND L.Inv_Org_Id = T5.Inv_Org_Id
           AND L.Is_Phantom = '0'
		   AND L.del_flag = 'N'
          LEFT JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F T6 --制程工艺单 t6
            ON T6.Product_Order_Id = T5.Id
           AND T6.Is_Phantom = '0'
		   AND T6.del_flag = 'N'
		  LEFT JOIN(SELECT A1.Resource_Id
		                  ,a.Process_Tech_Task_Id
						  ,ROW_NUMBER() OVER(partition by a.Process_Tech_Task_Id) AS RN
                      FROM dwrotd.DWR_PS_APS_PLAN_TASK_DTL_F a
                     INNER JOIN dwrotd.DWR_PS_APS_PLAN_TASK_F A1
                        ON a.Plan_Task_Id = A1.Id
					   AND A.del_flag = 'N'
					   AND A1.del_flag = 'N'
					) T7
		    ON T7.Process_Tech_Task_Id = T6.Id	
           AND T7.RN = 1			
		   ) x
  LEFT JOIN DWROTD.DWR_PS_RES_WIP_SCHE_F X1 --产线 x1
    ON x.Resource_Id = X1.Id
   AND X1.Is_Phantom = '0'
   AND X1.del_flag = 'N'
  LEFT JOIN DWROTD.DWR_PS_RES_ENTERPRISE_F X2 --车间 x2
    ON X1.Work_Shop_Id = X2.Id
   AND X2.Is_Phantom = '0'
   AND X2.del_flag = 'N'
  LEFT JOIN DWROTD.DWR_PS_RES_ENTERPRISE_F X3 --基地 x3
    ON X2.Tree_Pid = X3.Id
   AND X3.Is_Phantom = '0'
   AND X3.del_flag = 'N'
  LEFT JOIN DWROTD.DWR_PS_RES_ENTERPRISE_F X4 --产区 x4
    ON x.Product_Area_Id = X4.Id
   AND X4.Is_Phantom = '0'
   AND X4.del_flag = 'N'
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D X5 --物料
    ON x.MATERIAL_ID = X5.MATERIAL_ID
   AND X5.END_DATE = '4712-12-31'
   AND X5.DEL_FLAG = 'N'
  LEFT JOIN SDI.SDI_T179T_1001 X6 --产品品类
    ON X6.MANDT = X5.GROUP_NUM
   AND X6.SPRAS = '1'
   AND SUBSTR(X5.PRODUCT_LEVEL_CODE,1,4) = X6.PRODH
;
