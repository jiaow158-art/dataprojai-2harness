INSERT
	OVERWRITE TABLE dm.dm_dp_api_pred_org
SELECT
	sales_center_code,
	sales_center,
	sales_dep_code,
	sales_dep,
	sales_region_code,
	sales_region,
	CONCAT_WS(
		',',
		COLLECT_LIST(
			DISTINCT CASE
				WHEN b.DISTRIBUTION_CHNL_CODE = '06' THEN 'GD03'
				WHEN b.DISTRIBUTION_CHNL_CODE = '08' THEN 'GD02'
				ELSE 'GD01'
			END
		)
	) AS integrate_channel
FROM
	dm.dm_dp_api_sales_group_struct a
	JOIN DWRDIM.DWR_DIM_CUST_SALES_AREA_D b ON a.sales_grp = b.sales_group_code
	AND b.del_flag = 'N'
	AND b.end_date = '4712-12-31'
GROUP BY
	sales_center_code,
	sales_center,
	sales_dep_code,
	sales_dep,
	sales_region_code,
	sales_region
