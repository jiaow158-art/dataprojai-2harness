--逾期订单面积ZB56
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB56';

with tt1 as (

select
concat(substring(stat_month,1,4),'-',substring(stat_month,5,2)) as calmonth ,
sum(t.nodeliver_area_aps)/10000  as 逾期订单面积
from 
DM.DM_RPT_SALES_DELIVER_T t
where
expect_date_aps
<=
least(to_timestamp(SUBSTR(concat(CONCAT (SUBSTRING (concat(substring(stat_month,1,4),'-',substring(stat_month,5,2)),1,7)) ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10),'yyyy-MM-dd')--逾期
,current_timestamp -1 )
and 
t.nodeliver_area_aps > 0  
group by concat(substring(stat_month,1,4),'-',substring(stat_month,5,2)
)

)

,tt2 as (
select 
calmonth,
'ZB56' as zb_code,
逾期订单面积 as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
tt1
where 
calmonth = '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB56' as zb_code,
0 as 当月实绩,
avg(逾期订单面积) as 累计实绩,
0 as 季度累计实绩
from 
tt1
where CALMONTH between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB56' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
avg(逾期订单面积)  as 季度累计实绩
from 
tt1
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)

)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩) as CM_STAT_AMT,
sum(累计实绩) as ACC_STAT_AMT,
sum(季度累计实绩) as QTR_TGT_AMT
from 
TT2
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
