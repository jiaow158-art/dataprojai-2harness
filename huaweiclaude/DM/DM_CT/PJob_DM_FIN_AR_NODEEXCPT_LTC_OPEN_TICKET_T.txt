INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_OPEN_TICKET_T
SELECT 
		   SOT.WBS
		  ,SOT.OPEN_TICKET_CODE --开票单号
		  ,SOT.CREATION_DATE --开票创建日期
		  ,SOT.VERIFICATION_STATUS --核销状态
		  ,CASE WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_NO_COMMIT' THEN '未核销'
                WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_SOME_COMMIT' THEN '部分认领'
				WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_COMMIT' THEN '已认领'
				WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_F' THEN '未核销'
				WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_B' THEN '部分核销'
				WHEN SOT.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_T' THEN '已核销'
				WHEN SOT.VERIFICATION_STATUS = 'BACK_VERIFICATION' THEN '反核销'
				WHEN SOT.VERIFICATION_STATUS = 'REPORT_COST' THEN '报备费'
				WHEN SOT.VERIFICATION_STATUS = 'DELAY_COST' THEN '延期费'
				WHEN SOT.VERIFICATION_STATUS = 'ARBITRATION_EARNEST_MONEY' THEN '仲裁诚意金'
				WHEN SOT.VERIFICATION_STATUS = 'ARBITRATION_DEPOSIT' THEN '仲裁保证金'
				WHEN SOT.VERIFICATION_STATUS = 'TRANSFER_FEE' THEN '转让费'
				WHEN SOT.VERIFICATION_STATUS = 'MARGIN_BID' THEN '投标保证金'
				WHEN SOT.VERIFICATION_STATUS = 'MARGIN_QUALITY' THEN '质量保证金'
				WHEN SOT.VERIFICATION_STATUS = 'PERFORMANCE_BOND' THEN '履约保证金'
				WHEN SOT.VERIFICATION_STATUS = 'TOMARGIN_QUALITY' THEN '转质量保证金'
				WHEN SOT.VERIFICATION_STATUS = 'TOPERFORMANCE_BOND' THEN '转履约保证金'
		         END AS VERIFICATION_STATUS_NAME --核销状态描述
		  ,D.REQUEST_INVOICE_TIME2 --请款到开票时间（天）
		  ,D.INVOICE_RETURNED_TIME2 --开票到回款时间
		  ,COALESCE(SOT.CURRENT_TICKET_AMOUNT,0) --开票金额合计
		  ,COALESCE(SOT.CURRENT_TICKET_AMOUNT,0) - COALESCE(C.CURRENT_VERIFICATION_AMT,0) AS URPLUS_TICKET_AMOUNT --未核销金额
		  ,SOT.UN_CHECK_AMOUNT AS UN_CHECK_AMOUNT --未勾稽金额
		  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
          ,SOT.PROCESS_STATUS --开票状态
          ,CASE WHEN SOT.PROCESS_STATUS = 'DRAFT' THEN '草稿'
                WHEN SOT.PROCESS_STATUS = 'APPROVAL' THEN '审批中'
				WHEN SOT.PROCESS_STATUS = 'ALLOW' THEN '审批通过'
				WHEN SOT.PROCESS_STATUS = 'FEEDBACK' THEN '结果反馈'
				WHEN SOT.PROCESS_STATUS = 'COMPLETE' THEN '完成'
				WHEN SOT.PROCESS_STATUS = 'CLOSE' THEN '关闭'
				WHEN SOT.PROCESS_STATUS = 'PASS' THEN '通过'
				WHEN SOT.PROCESS_STATUS = 'REJECT' THEN '驳回'
				WHEN SOT.PROCESS_STATUS = 'FAIL' THEN '失败'
				WHEN SOT.PROCESS_STATUS = 'REFUSE' THEN '拒绝'
				WHEN SOT.PROCESS_STATUS = 'DISCARD_WAIT' THEN '待作废'
				WHEN SOT.PROCESS_STATUS = 'DISCARD' THEN '作废'
				 END AS PROCESS_STATUS_NAME --开票状态名称		  
  FROM SDI.SDI_SETTLEMENT_OPEN_TICKET_1033 SOT
  LEFT JOIN SDI.SDI_CONTRACT_1033 A
    ON A.CONTRACT_CODE = SOT.CONTRACT_CODE
   AND A.DELETE_FLAG = 0    
  LEFT JOIN (SELECT MASTER_CODE
                   ,SUM(COALESCE(GOODS_TOTAL_PRICE,0) + COALESCE(GOODS_TOTAL_TAX,0)) AS CURRENT_TICKET_AMOUNT
               FROM SDI.SDI_SETTLEMENT_FEE_1033
              WHERE DELETE_FLAG = 0
                 AND MASTER_FLAG IN('TICKET_A','PRE_TICKET_B')
			  GROUP BY MASTER_CODE 
			) B
	ON SOT.OPEN_TICKET_CODE = B.MASTER_CODE
  LEFT JOIN (SELECT INVOICE_CODE
                   ,SUM(CURRENT_VERIFICATION_AMT) AS CURRENT_VERIFICATION_AMT --已核销金额
               FROM SDI.SDI_RETURN_RECIPIENT_INFO_DETAIL_1033
              WHERE DELETE_FLAG = 0 
			  GROUP BY INVOICE_CODE 
            ) C
	ON SOT.OPEN_TICKET_CODE = C.INVOICE_CODE		
  LEFT JOIN (SELECT DISTINCT 
                             CONTRACT_ID
							,REQUEST_INVOICE_TIME2
							,INVOICE_RETURNED_TIME2
               FROM SDI.SDI_CONTRACT_SETTLEMENT_HEAD_1033
              WHERE DELETE_FLAG = 0 
            ) D
	ON A.CONTRACT_ID = D.CONTRACT_ID
 WHERE SOT.DELETE_FLAG = 0
