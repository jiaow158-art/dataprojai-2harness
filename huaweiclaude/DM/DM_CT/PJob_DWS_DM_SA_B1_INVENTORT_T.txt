-- 需求人：zhouhaochang
-- update：2023/06/20 zhouhaochang 销售出库由区间改为当天
-- update：2023/06/21 zhouhaochang 补充期末面积、单位面积，方便前端计算，库存物料主档改为join关联
-- update：2023/06/25 zhouhaochang 补充用于计算库存周转率等字段
-- update：2023/07/17 zhouhaochang 使用胖哥新底表DM_WM_INVENTORY_CT_T_D
-- update：2023/07/18 zhouhaochang 增加销售组织字段
-- update：2023/08/30 zhouhaochang 使用胖子新底表DM.DM_WM_INVENTORY_CT_T
-- ******************************************************************** --
do
$$
    declare
        P_DATE varchar(500) ;
    begin
        for P_DATE in (select day_id
                       from dm.dm_dim_date_d
                       where day_id >= to_char(to_date('${PERIOD_ID_D1}'), 'yyyyMMdd')
                         and day_id <= to_char(to_date('${PERIOD_ID_D2}'), 'yyyyMMdd')
                       order by day_id)
            loop
                delete
                from dm.dm_sa_b1_inventort_t
                where stat_day = to_char(P_DATE::date, 'yyyy-MM-dd');
                insert into dm.dm_sa_b1_inventort_t
                select to_char(P_DATE::date, 'yyyy-MM-dd')                                                stat_day,
                       a.data_source_falg,
                       a.share_flag,
                       a.customer_code,
                       a.customer_name,
                       a.material_num,
                       a.material_name,
                       d.lve2_code                                                                        prod_type,
                       a.dimension,
                       a.menu_desc_name,
                       case b.is_high_value when 1 then '是' else '否' end                                  is_high_value,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then cost_total_amt
                               else 0 end)                                                                sales_cost,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qc_qty
                               else 0 end)                                                                qc_qty,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_instock_qty
                               else 0 end)                                                                本期入库,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_pp_del_qty
                               else 0 end)                                                                期间采购收货数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_sale_return_qty
                               else 0 end)                                                                期间销售退货数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_other_instock_qty
                               else 0 end)                                                                期间其他入库数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_pp_return_qty
                               else 0 end)                                                                期间采购退货数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_outstock_qty
                               else 0 end)                                                                本期出库,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_sale_outstock_qty
                               else 0 end)                                                                期间销售出库数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_other_outstock_qty
                               else 0 end)                                                                期间其他出库数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qm_qty
                               else 0 end)                                                                qm_qty,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qj_happen_qty
                               else 0 end)                                                                期间发生,
                       sum(case when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qc_amt else 0 end) 期初金额,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then bq_instock_amt
                               else 0 end)                                                                本期入库金额,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then bq_outstock_amt
                               else 0 end)                                                                本期出库金额,
                       sum(case when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then qm_amt else 0 end) 期末金额,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then bq_happen_amt
                               else 0 end)                                                                本期发生金额,
                       to_char(max(last_instock_date)::date, 'yyyyMMdd')::int                             最末入库日期,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then sales_outgoods_qty
                               else 0 end)                                                                销售出库数量,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then sales_outgoods_amt
                               else 0 end)                                                                销售出库金额,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd') then sales_outgoods_area_qty
                               else 0 end)                                                                销售出库面积,
                       a.material_id,
                       a.matl_type_code,
                       a.matl_type_name,
                       a.product_group_code,
                       a.product_group_name,
                       a.product_classify_code,
                       a.product_classify_name,
                       a.product_level_code,
                       a.product_level_name,
                       a.product_level1_code,
                       a.product_level1_name,
                       a.product_level2_code,
                       a.product_level2_name,
                       a.product_level3_code,
                       a.product_level3_name,
                       a.product_level4_code,
                       a.product_level4_name,
                       a.busi_volume,
                       current_timestamp,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd')
                                   then qm_qty * b.width * b.length_1 / 1000 / 1000
                               else 0 end)                                                                期末面积,
                       unit_area_qty                                                                      单位面积,
                       sum(case
                               when substr(day_dt, 1, 7) = to_char(P_DATE::date, 'yyyy-MM')
                                   then sales_outgoods_area_qty
                               else 0 end)                                                                本月销售面积,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-01')
                                   then qc_qty * b.width * b.length_1 / 1000 / 1000
                               else 0 end)                                                                月初库存面积,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd')
                                   then qm_qty * b.width * b.length_1 / 1000 / 1000
                               else 0 end)                                                                月末库存面积,
                       sum(sales_outgoods_area_qty)                                                       年初至月末销售面积,
                       sum(case
                               when day_dt = to_char(P_DATE::date, 'yyyy-01-01')
                                   then qc_qty * b.width * b.length_1 / 1000 / 1000
                               else 0 end)                                                                年初库存面积,
                       sum(case when right(day_dt, 2) = '01' then qc_qty else 0 end + case
                                                                                          when day_dt = to_char(P_DATE::date, 'yyyy-MM-dd')
                                                                                              then qm_qty * b.width * b.length_1 / 1000 / 1000
                                                                                          else 0 end)     每月末库存面积求和,
                       sales_org_name
                from DM.DM_WM_INVENTORY_CT_T a
                         left join dwimd.dwi_md_data_material_general_t_01 b on a.material_num = b.material_num
                         left join dm.dm_rpt_prod_level_t d
                                   on coalesce(b.product_level3_code, b.product_level2_code, '#') =
                                      coalesce(d.prod_code3, '#')
                         left join upload.upload_ct_customer_data_t c on a.customer_code = c.customer_code
                where (day_dt <= to_char(P_DATE::date, 'yyyy-MM-dd')
                    and left(day_dt, 4) = to_char(P_DATE::date, 'yyyy'))
                  and c.stop_flag is null
                group by a.data_source_falg,
                         a.share_flag, a.customer_code,
                         a.customer_name,
                         a.material_num,
                         a.material_name,
                         prod_type,
                         a.dimension,
                         a.menu_desc_name,
                         is_high_value,
                         a.material_id,
                         a.matl_type_code,
                         a.matl_type_name,
                         a.product_group_code,
                         a.product_group_name,
                         a.product_classify_code,
                         a.product_classify_name,
                         a.product_level_code,
                         a.product_level_name,
                         a.product_level1_code,
                         a.product_level1_name,
                         a.product_level2_code,
                         a.product_level2_name,
                         a.product_level3_code,
                         a.product_level3_name,
                         a.product_level4_code,
                         a.product_level4_name,
                         a.busi_volume,
                         单位面积,
                         sales_org_name;
            end loop;
    end;
$$
