INSERT OVERWRITE TABLE DM.DM_DMT_TC_CUST_STAT_T
SELECT SUBSTR(A.FST_PRTCT_TIME,1,7) AS STAT_MONTH --统计月份
      ,SUBSTR(A.FST_PRTCT_TIME,1,10) AS STAT_DATE --统计日期
      ,A.PRTCT_SRC_ID AS CUST_SRC_CODE --客户来源
	  ,A.PRTCT_SRC_NAME --客户来源名称
	  ,A.PARNT_PRTCT_SRC_NAME AS CUST_SRC_CAT_NAME --客户来源类别名称
	  ,B.MKT_NAME AS DEPT_NAME --部门名称
	  ,CASE WHEN SUBSTR(B.PROV_NAME,1,2) IN('上海','天津','重庆','北京') THEN B.PROV_NAME
	        ELSE B.CITY_NAME
	         END AS CITY_NAME --城市
	  ,B.STORE_NAME --门店名称
	  ,NULL AS SALE_NAME --导购员名称
	  ,'门店' AS DATA_TYPE --统计类型
      ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已报备','已到店','已交定','已测量','已出图','已确图','已大定','已小定','交易成功','已完成','已流失') THEN C.CUST_PHONE_NUM END) AS YBB_CUST_CNT --S0 已报备数
	  ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已到店','已交定','已测量','已出图','已确图','已大定','已小定','交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YDS_CUST_CNT --S1 已到店数
	  ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已交定','已测量','已出图','已确图','已大定','已小定','交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YXD_CUST_CNT --S2 已小定数 已交定
      ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已测量','已出图','已确图','已大定','交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YCL_CUST_CNT --S3 已测量数 已出图
	  ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已出图','已确图','已大定','交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YQT_CUST_CNT --S4 已确图数
	  ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('已大定','交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YDD_CUST_CNT --S5 已大定数
      ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN('交易成功','已完成') THEN C.CUST_PHONE_NUM END) AS YCJ_CUST_CNT --S6 已成交数
	  ,0 AS YLS_CUST_CNT --S7 已流失数
	  ,0 AS CJ_AMT --成交金额
	  ,0 AS DQCJ_AMT --当期成交金额
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,A.PRTCT_STR_ID AS STORE_ID --门店ID
      ,NULL AS SALE_EMP_ID --导购ID
      ,B.OA_STORE_CODE AS OUT_STORE_ID --OA门店编码
      ,B.BUSI_ID AS BUSI_TYPE --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）	  
      ,B.DEALER_NAME AS DEALER_NAME --经销商名称
	  ,B.CUST_NUM AS DEALER_NUM --经销商编号
	  ,0 AS EXAM_CNT --撞单数
	  ,0 AS CONFLICT_CNT --冲突数
	  ,B.IS_TC --是否参与同城业务
  FROM DWRCS.DWR_CS_MOS_PRTCT_INFO_F A
 INNER JOIN DWROTD.DWR_SO_MOS_ORDER_INFO_F C
    ON A.ORDER_ID = C.ORDER_ID
  LEFT JOIN DWRDIM.DWR_DIM_MOS_STORE_INFO_T B
    ON A.PRTCT_STR_ID = B.ID
 WHERE SUBSTR(A.FST_PRTCT_TIME,1,4) >= '2023'
   AND(B.MKT_NAME LIKE '%运营中心' OR B.MKT_NAME LIKE '%子公司')
 GROUP BY
       SUBSTR(A.FST_PRTCT_TIME,1,7) --统计月份
      ,SUBSTR(A.FST_PRTCT_TIME,1,10) --统计日期
      ,A.PRTCT_SRC_ID --客户来源
	  ,A.PRTCT_SRC_NAME --客户来源名称
	  ,A.PARNT_PRTCT_SRC_NAME --客户来源类别名称
	  ,B.MKT_NAME --部门名称
	  ,CASE WHEN SUBSTR(B.PROV_NAME,1,2) IN('上海','天津','重庆','北京') THEN B.PROV_NAME
	        ELSE B.CITY_NAME
	         END --城市
	  ,B.STORE_NAME --门店名称
	  ,A.PRTCT_STR_ID  --门店ID
      ,B.OA_STORE_CODE --OA门店编码
      ,B.BUSI_ID --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）	  
      ,B.DEALER_NAME --经销商名称
	  ,B.CUST_NUM --经销商编号
	  ,B.IS_TC --是否参与同城业务
	  
 UNION ALL

SELECT SUBSTR(C.LOSE_TIME,1,7) AS STAT_MONTH --统计月份
      ,SUBSTR(C.LOSE_TIME,1,10) AS STAT_DATE --统计日期
      ,A.PRTCT_SRC_ID AS CUST_SRC_CODE --客户来源
	  ,A.PRTCT_SRC_NAME --客户来源名称
	  ,A.PARNT_PRTCT_SRC_NAME AS CUST_SRC_CAT_NAME --客户来源类别名称
	  ,B.MKT_NAME AS DEPT_NAME --部门名称
	  ,CASE WHEN SUBSTR(B.PROV_NAME,1,2) IN('上海','天津','重庆','北京') THEN B.PROV_NAME
	        ELSE B.CITY_NAME
	         END AS CITY_NAME --城市
	  ,B.STORE_NAME --门店名称
	  ,A.PRTCT_EMP_NAME AS SALE_NAME --导购员名称
	  ,'导购' AS DATA_TYPE --统计类型
      ,0 AS YBB_CUST_CNT --S0 已报备数
	  ,0 AS YDS_CUST_CNT --S1 已到店数
	  ,0 AS YXD_CUST_CNT --S2 已小定数 已交定

	  ,0 AS YDD_CUST_CNT --S5 已大定数

	  ,COUNT(DISTINCT CASE WHEN C.ORDER_ST_NAME IN ('已流失') THEN C.CUST_PHONE_NUM END) AS YLS_CUST_CNT --S7 已流失数
	  ,0 AS CJ_AMT --成交金额

      ,A.PRTCT_STR_ID AS STORE_ID --门店ID
      ,A.PRTCT_EMP_ID AS SALE_EMP_ID --导购ID
      ,B.OA_STORE_CODE AS OUT_STORE_ID --OA门店编码
      ,B.BUSI_ID AS BUSI_TYPE --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）	  
      ,B.DEALER_NAME AS DEALER_NAME --经销商名称

	  ,COUNT(1) AS EXAM_CNT --撞单数
	  ,0 AS CONFLICT_CNT --冲突数
	  ,B.IS_TC --是否参与同城业务

    ON A.PRTCT_STR_ID = B.ID
 INNER JOIN DWROTD.DWR_SO_MOS_ORDER_INFO_F C
    ON A.ORDER_ID = C.ORDER_ID
 WHERE SUBSTR(C.LOSE_TIME,1,4) >= '2023'
   AND(B.MKT_NAME LIKE '%运营中心' OR B.MKT_NAME LIKE '%子公司')
 GROUP BY
       SUBSTR(C.LOSE_TIME,1,7) --统计月份
      ,SUBSTR(C.LOSE_TIME,1,10) --统计日期
      ,A.PRTCT_SRC_ID --客户来源

	  ,A.PARNT_PRTCT_SRC_NAME --客户来源类别名称
	  ,B.MKT_NAME --部门名称
	  ,CASE WHEN SUBSTR(B.PROV_NAME,1,2) IN('上海','天津','重庆','北京') THEN B.PROV_NAME
	        ELSE B.CITY_NAME
	         END --城市
	  ,B.STORE_NAME --门店名称
	  ,A.PRTCT_STR_ID  --门店ID
      ,B.OA_STORE_CODE --OA门店编码
      ,B.BUSI_ID --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）	  
      ,B.DEALER_NAME --经销商名称

	  ,A.PRTCT_EMP_ID --导购ID
	  ,A.PRTCT_EMP_NAME --导购员名称
	  ,B.IS_TC --是否参与同城业务
