DROP TABLE IF EXISTS DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP00;
CREATE TABLE DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP00
AS
SELECT REQUESTID
      ,MAX(CONCAT(CONCAT(OPERATEDATE,' '),OPERATETIME)) AS OPERATE_DAY
  FROM SDI.SDI_WORKFLOW_REQUESTLOG_1004
 WHERE DEL_FLAG = 'N' 
   AND OPERATEDATE >= '2023-01-01'
   AND LOGTYPE = 'r'
GROUP BY REQUESTID
;

DROP TABLE IF EXISTS DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP01;
CREATE TABLE DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP01
AS
SELECT REQUEST_CODE
      ,OPERATION_NODE_CODE
	  ,OA_GROUPID
	  ,COUNT(1) AS CNT
  FROM DWRPI.DWR_WF_REQUESTS_NODE_CREATOR_F C
 WHERE COALESCE(C.OPERATION_TYPE,'') NOT IN('1','8')
   AND COALESCE(C.BEFORE_CHANGE_OPERATION_TYPE,'') NOT IN('1','8','11')
   AND C.RECEIVE_DATE >= '2021-01-01'
GROUP BY REQUEST_CODE
        ,OPERATION_NODE_CODE
		,OA_GROUPID
;

DROP TABLE IF EXISTS DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP02;
CREATE TABLE DM.DM_WF_REQUESTS_CONSUMING_TIME_T_TMP02
AS
SELECT A.REQUESTID
      ,A.OPERATOR AS REQUEST_EMP_ID --发起人ID
      ,CONCAT(CONCAT(A.OPERATEDATE,' '),A.OPERATETIME) AS REQUEST_DAY --流程发起时间
      ,ROW_NUMBER() OVER(PARTITION BY A.REQUESTID ORDER BY CONCAT(CONCAT(A.OPERATEDATE,' '),A.OPERATETIME) ASC) AS RN
  FROM SDI.SDI_WORKFLOW_REQUESTLOG_1004 A
  LEFT JOIN (SELECT A.REQUESTID
                   ,MAX(CONCAT(CONCAT(A.OPERATEDATE,' '),A.OPERATETIME)) AS REQUEST_DAY --流程发起时间                   
			   FROM SDI.SDI_WORKFLOW_REQUESTLOG_1004 A
			  WHERE A.DEL_FLAG = 'N'
                AND A.LOGTYPE IN('r')
			  GROUP BY A.REQUESTID	
			) B --获取撤回流程的时间
	ON A.REQUESTID = B.REQUESTID
 WHERE A.DEL_FLAG = 'N'
   AND A.LOGTYPE IN('2')
   --提交流程的时间要大于撤回流程时间
   AND CONCAT(CONCAT(A.OPERATEDATE,' '),A.OPERATETIME) >= COALESCE(B.REQUEST_DAY,'1900-01-01 00:00:00')
;

      ,case when coalesce(B.LEV2_DEPT_NAME,'') = '' then B.LEV1_DEPT_NAME

			when coalesce(B.LEV6_DEPT_NAME,'') = '' then concat(B.LEV1_DEPT_NAME,'>',B.LEV2_DEPT_NAME,'>',B.LEV3_DEPT_NAME,'>',B.LEV4_DEPT_NAME,'>',B.LEV5_DEPT_NAME)

	  ,count(distinct A.node_id) as node_count --节点数
	  ,sum(case when A.HS_GZ3 < 0 then 0
	            else A.HS_GZ3
				 end
		  )/3600 gz_hour --耗时(工作时间:小时)
	  ,B.LEV1_DEPT_NAME --部门层级1
	  ,B.LEV2_DEPT_NAME --部门层级2
	  ,B.LEV3_DEPT_NAME --部门层级3
	  ,B.LEV4_DEPT_NAME --部门层级4

	  ,B.LEV6_DEPT_NAME --部门层级6
	  ,B.LEV7_DEPT_NAME --部门层级7
	  ,B.LEV8_DEPT_NAME --部门层级8

	  ,A.EMP_POST_NAME
	  ,A.WF_TYPE_NAME --流程分类
	  ,A.wf_code
	  ,A.is_zx

      ,A.REQUEST_DAY --流程发起时间
      ,A.GD_DAY --流程归档时间
      ,A.REQUEST_EMP_ID --发起人ID
	  ,C.SAPID AS REQUEST_EMP_NUM --发起人工号
	  ,C.LASTNAME AS REQUEST_EMP_NAME --发起人名称
	  ,C.DEPARTMENTID AS REQUEST_DEPT_ID --发起人所在部门ID
	  ,E.DEPT_NAME AS REQUEST_DEPT_NAME --发起人所属部门
	  ,SUM(A.HS_GZ3) AS WORK_HS --工作耗时（秒）
      ,D.LEV1_DEPT_NAME AS REQUEST_LEV1_DEPT_NAME --发起人所在一级部门

	  ,D.LEV4_DEPT_NAME AS REQUEST_LEV4_DEPT_NAME --发起人所在四级部门
	  ,D.LEV5_DEPT_NAME AS REQUEST_LEV5_DEPT_NAME --发起人所在五级部门
	  ,D.LEV6_DEPT_NAME AS REQUEST_LEV6_DEPT_NAME --发起人所在六级部门
	  ,D.LEV7_DEPT_NAME AS REQUEST_LEV7_DEPT_NAME --发起人所在七级部门
	  ,D.LEV8_DEPT_NAME AS REQUEST_LEV8_DEPT_NAME --发起人所在八级部门
	  ,A.ACTL_EMP_ID --实际操作人ID
	  ,A.ACTL_EMP_NUM --实际操作人工号
	  ,A.ACTL_EMP_NAME --实际操作人姓名	
  FROM TE A
  LEFT JOIN DM.DM_WF_DEPT_INFO_T B
    ON A.DEPT_ID = B.DEPT_ID
  --修改发起人的取值表，不过滤离职员工
  LEFT JOIN SDI.SDI_HRMRESOURCE_1004 C
    ON A.REQUEST_EMP_ID = C.ID
   AND C.DEL_FLAG = 'N'
  LEFT JOIN DM.DM_WF_DEPT_INFO_T D
    ON C.DEPARTMENTID = D.DEPT_ID
  LEFT JOIN DWRPI.DWR_WF_OA_DEPT_INTO_F E
    ON C.DEPARTMENTID = E.DEPT_ID
 GROUP BY
       A.request_code --请求ID
	  ,A.request_explain --流水号

	  ,A.node_name --节点名称
	  ,A.l1lc --L1流程
	  ,A.wf_name --流程名称
	  ,A.receive_date --接收日期
	  ,A.receive_time --接收时间

	  ,A.operate_time --处理时间
	  ,A.operatedateSJ --实际处理日期
	  ,A.operatetimeSJ --实际处理时间
      ,A.operate_type --操作类型
	  ,A.last_operate_type --上一操作类型
      ,case when coalesce(B.LEV2_DEPT_NAME,'') = '' then B.LEV1_DEPT_NAME
      	    when coalesce(B.LEV3_DEPT_NAME,'') = '' then concat(B.LEV1_DEPT_NAME,'>',B.LEV2_DEPT_NAME)

			when coalesce(B.LEV5_DEPT_NAME,'') = '' then concat(B.LEV1_DEPT_NAME,'>',B.LEV2_DEPT_NAME,'>',B.LEV3_DEPT_NAME,'>',B.LEV4_DEPT_NAME)
			when coalesce(B.LEV6_DEPT_NAME,'') = '' then concat(B.LEV1_DEPT_NAME,'>',B.LEV2_DEPT_NAME,'>',B.LEV3_DEPT_NAME,'>',B.LEV4_DEPT_NAME,'>',B.LEV5_DEPT_NAME)

			when coalesce(B.LEV8_DEPT_NAME,'') = '' then concat(B.LEV1_DEPT_NAME,'>',B.LEV2_DEPT_NAME,'>',B.LEV3_DEPT_NAME,'>',B.LEV4_DEPT_NAME,'>',B.LEV5_DEPT_NAME,'>',B.LEV6_DEPT_NAME,'>',B.LEV7_DEPT_NAME)

			 end --部门架构
	  ,B.LEV1_DEPT_NAME --部门层级1
	  ,B.LEV2_DEPT_NAME --部门层级2
	  ,B.LEV3_DEPT_NAME --部门层级3

	  ,B.LEV5_DEPT_NAME --部门层级5
	  ,B.LEV6_DEPT_NAME --部门层级6
	  ,B.LEV7_DEPT_NAME --部门层级7
	  ,B.LEV8_DEPT_NAME --部门层级8 
	  ,A.current_node_type
	  ,A.EMP_POST_NAME
	  ,A.WF_TYPE_NAME --流程分类
	  ,A.wf_code
	  ,A.is_zx
	  ,A.REQUEST_STATUS --流程状态
      ,A.REQUEST_DAY --流程发起时间
      ,A.GD_DAY --流程归档时间
      ,A.REQUEST_EMP_ID --发起人ID
	  ,C.SAPID --发起人工号
	  ,C.LASTNAME --发起人名称
	  ,C.DEPARTMENTID --发起人所在部门ID
	  ,E.DEPT_NAME --发起人所属部门
