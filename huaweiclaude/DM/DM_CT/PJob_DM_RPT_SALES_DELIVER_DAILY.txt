-- 净交付 = 今日交付昨日未交付+今日下单今日交付
INSERT OVERWRITE TABLE dm.dm_rpt_sales_deliver_daily 
select 
t1.vbeln , 
t1.posnr, 
t1.matnr , 
t1.res_stock_date , 
t1.res_stock_qty , 
t1.bstdk , 
t1.product_area , 
t1.dw_last_update_date,
t1.erdat , 
least(t1.res_stock_qty*t1.product_area, t2.nodeliver_area_aps) as net_deliver_area,
t1.plant_code,
t1.self_foreign_flag,
t1.znka,
t1.znka_type,
t1.znka_kunnr
from dm.dm_otd_so_res_stock_dtl t1   
join dm.dm_rpt_sales_deliver_daily_tmp t2 ON t1.vbeln = t2.order_no and t1.posnr =t2.order_item_no-- 昨日未交付明细
where t1.res_stock_date = SUBSTRING(replace(date_sub(current_date,1),'-',''),1,8)
union all 
select 
t1.vbeln , 
t1.posnr, 
t1.matnr , 
t1.res_stock_date , 
t1.res_stock_qty , 
t1.bstdk , 
t1.product_area , 
t1.dw_last_update_date,
t1.erdat , 
t1.res_stock_qty*t1.product_area as net_deliver_area,
t1.plant_code,
t1.self_foreign_flag,
t1.znka,
t1.znka_type,
t1.znka_kunnr
from dm.dm_otd_so_res_stock_dtl t1   
left join dm.dm_rpt_sales_deliver_daily_tmp t2 ON t1.vbeln = t2.order_no and t1.posnr =t2.order_item_no
where t1.res_stock_date =SUBSTRING(replace(date_sub(current_date,1),'-',''),1,8)
and t2.order_no is null
and t1.erdat = SUBSTRING(replace(date_sub(current_date,1),'-',''),1,8);

      delivernosend_area, 
      dw_last_update_date1, 
      prod_name3, 
      expect_date, 
      first_confirm_date, 
      order_type, 
      dept_lv3_code, 
      dept_lv4_code, 
      dept_lv5_code, 
      dept_lv3_name, 
      dept_lv4_name, 
      dept_lv5_name, 
      cust_num, 
      cust_name, 
      jhddlx, 
      submi, 
      salesman_num_list, 
      salesman_name_list, 
      nodeliver_area_aps, 
      expect_date_aps, 
      stat_month,
      plant_code,
      self_foreign_flag
from dm.dm_rpt_sales_deliver_t     
where nodeliver_area_aps>0
and stat_month = SUBSTRING(replace(date_sub(current_date,1),'-',''),1,6);
