	  ,A.LARGE_MONTH_REBT_ACHIEVEMENT AS LARGE_SALE_ACHIEVEMENT --大规格业绩

	  ,NULL AS LARGET_ACTL_REBT_AMT --大规格实际返利金额

SELECT A.ORG_CODE --销售组编码
	  ,A.NODE_NAME5 --运营中心编码
      ,A.NODE_DESC5 --运营中心
      ,A.CHANNEL_NAME --整合渠道
	  ,A.NODE_DESC6 --部门
	  ,A.NODE_DESC7 --销区
	  ,A.NODE_DESC8 --线组
	  ,A.CUST_NUM --客户编码
	  ,A.CUST_NAME --客户名称
	  ,A.QUARTER_RAD_CUST_CNT AS RAD_CUST_CNT --辐射客户数
	  ,A.QUARTER_TARGET AS SALE_TARGET --目标
	  ,A.LARGE_QUARTER_ACHIEVEMENT AS LARGE_ACHIEVEMENT --大规格业绩
	  ,A.LARGET_QUARTER_RATE AS LARGET_RATE --大规格完成率
	  ,A.QUARTER_SALE_QTY AS B1_SALE_QTY --B1销售数量
	  ,A.QUARTER_SALE_AMT AS B1_SALE_AMT --B1销售金额
	  ,A.CNVN_QUARTER_QTY AS CNVN_SALE_QTY --常规产品销售数量
	  ,A.CNVN_QUARTER_ACHIEVEMENT AS CNVN_ACHIEVEMENT --常规产品业绩
	  ,A.QUARTER_REBT_JDGMT AS CNVN_REBT_JDGMT --常规产品返利判断
	  ,A.CNVN_QUARTER_REBT_ACHIEVEMENT AS CNVN_REBT_ACHIEVEMENT --常规产品可结算业绩
	  ,A.EXPTD_QUARTER_REBT_AMT AS CNVN_EXPTD_REBT_AMT --常规产品拟达标可返利金额
	  ,CASE WHEN A.LARGET_ACCM_RATE >= 1 AND A.CNVN_ACCM_RATE >= 1
            THEN A.CNVN_ACTL_QUARTER_REBT_AMT + COALESCE(B.CNVN_DIFF_REBT_AMT,0)
	        ELSE A.CNVN_ACTL_QUARTER_REBT_AMT
	         END AS CNVN_ACTL_REBT_AMT --常规产品实际返利金额
	  ,A.LARGE_QUARTER_QTY AS LARGE_SALE_QTY --大规格销售数量
	  ,A.LARGE_QUARTER_REBT_ACHIEVEMENT AS LARGE_SALE_ACHIEVEMENT --大规格业绩
	  ,A.LARGET_EXPTD_QUARTER_REBT_AMT AS LARGET_EXPTD_REBT_AMT --大规格拟达标可返利金额
	  ,CASE WHEN A.LARGET_ACCM_RATE < 1 THEN A.LARGET_ACTL_QUARTER_REBT_AMT
            ELSE A.LARGET_ACTL_QUARTER_REBT_AMT + COALESCE(B.LARGET_DIFF_REBT_AMT,0)
	         END AS LARGET_ACTL_REBT_AMT --大规格实际返利金额
	  ,A.LARGET_ACCM_RATE --大规格累计完成率
	  ,A.CNVN_ACCM_RATE --常规产品累计完成率
	  ,'季度' AS STAT_TYPE --统计类型
	  ,TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ') AS STAT_DATE --统计周期
	  ,TO_CHAR('${P_PERIOD_D}'::DATE,'yyyy') AS STAT_YEAR --统计年份
	  ,CASE WHEN A.LARGET_QUARTER_RATE < 1 THEN '季度未达标100%，不进行季度返利'
	        ELSE '季度达标100%，进行季度返利'
			 END ||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '2'
	        AND A.LARGET_ACCM_RATE > 1
			AND C.Q1_LARGET_BC = 1
			AND C.Q1_LARGET_ACCM_RATE < 1
		   THEN '补差Q1差额'
			END ||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '2'
            AND A.LARGET_ACCM_RATE > 1
            AND A.LARGET_QUARTER_RATE < 1
           THEN '补差Q2差额'
		    END ||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '3'
	        AND A.LARGET_ACCM_RATE > 1
			AND C.Q2_LARGET_BC = 1
			AND C.Q2_LARGET_ACCM_RATE < 1
		   THEN '补差Q2差额'
		    END ||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '3'
            AND A.LARGET_ACCM_RATE > 1
            AND A.LARGET_QUARTER_RATE < 1
           THEN '补差Q3差额'
		    END ||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '4'
	        AND A.LARGET_ACCM_RATE > 1
            AND A.LARGET_QUARTER_RATE < 1
		   THEN '补差Q4差额'
			END	||
	  CASE WHEN RIGHT(TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ'),1) = '4'
	        AND A.LARGET_ACCM_RATE > 1
			AND C.Q3_LARGET_BC = 1
			AND C.Q3_LARGET_ACCM_RATE < 1
		   THEN '补差Q3差额'
			END AS REMARK --备注说明
  FROM T7 A
  LEFT JOIN (SELECT A.NODE_NAME5 --运营中心编码
	               ,A.CUST_NUM --客户编码
	               ,SUM(A.CNVN_ACTL_REBT_AMT) AS CNVN_ACTL_REBT_AMT --常规产品实际返利金额
	               ,SUM(A.LARGET_ACTL_REBT_AMT) AS LARGET_ACTL_REBT_AMT --大规格实际返利金额
				   ,SUM(COALESCE(A.CNVN_EXPTD_REBT_AMT,0) - COALESCE(A.CNVN_ACTL_REBT_AMT,0)) AS CNVN_DIFF_REBT_AMT --常规产品返利金额差额
				   ,SUM(COALESCE(A.LARGET_EXPTD_REBT_AMT,0) - COALESCE(A.LARGET_ACTL_REBT_AMT,0)) AS LARGET_DIFF_REBT_AMT --大规格返利金额差额
               FROM DM.DM_CZ_GXC_CUST_REBT_DIFF_T A
              WHERE STAT_TYPE = '季度'
                AND STAT_DATE < TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ')
                AND STAT_YEAR = TO_CHAR('${P_PERIOD_D}'::DATE,'yyyy')
              GROUP BY A.NODE_NAME5 --运营中心编码
	               ,A.CUST_NUM --客户编码  
            ) B
	ON A.NODE_NAME5 = B.NODE_NAME5
   AND A.CUST_NUM = B.CUST_NUM
  LEFT JOIN (SELECT A.NODE_NAME5 --运营中心编码
	               ,A.CUST_NUM --客户编码
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '1' AND LARGET_RATE < 1 THEN 1 END) AS Q1_LARGET_BC --Q1季度大规格是否达标
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '2' AND LARGET_RATE < 1 THEN 1 END) AS Q2_LARGET_BC --Q2季度大规格是否达标
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '3' AND LARGET_RATE < 1 THEN 1 END) AS Q3_LARGET_BC --Q3季度大规格是否达标
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '1' THEN LARGET_ACCM_RATE ELSE 0 END) AS Q1_LARGET_ACCM_RATE --Q1季度大规格累计完成率
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '2' THEN LARGET_ACCM_RATE ELSE 0 END) AS Q2_LARGET_ACCM_RATE --Q2季度大规格累计完成率
				   ,MAX(CASE WHEN RIGHT(STAT_DATE,1) = '3' THEN LARGET_ACCM_RATE ELSE 0 END) AS Q3_LARGET_ACCM_RATE --Q3季度大规格累计完成率
               FROM DM.DM_CZ_GXC_CUST_REBT_DIFF_T A
              WHERE STAT_TYPE = '季度'
                AND STAT_DATE < TO_CHAR('${P_PERIOD_D}'::DATE,'yyyyQ')
                AND STAT_YEAR = TO_CHAR('${P_PERIOD_D}'::DATE,'yyyy')
              GROUP BY A.NODE_NAME5 --运营中心编码
	               ,A.CUST_NUM --客户编码
            ) C
	ON A.NODE_NAME5 = C.NODE_NAME5
   AND A.CUST_NUM = C.CUST_NUM
)
