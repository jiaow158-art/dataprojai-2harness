INSERT OVERWRITE TABLE DM.DM_WF_EMP_INFO_T
select hr.WORKCODE as emp_num --工号
      ,hr.lastname as emp_name --姓名
      ,COALESCE(F.USRID,hr.email) AS email --邮箱
      ,concat('Dp',hr.WORKCODE) as Password --密码
      ,HR.jobactivitydesc AS EMP_POST_NAME --岗位层级
      ,HR.jobactivitydesc as Role_name --角色
      ,HR.status --用户状态 0:试用1:正式2:临时3:试用延期4:解聘5:离职6:退休7:无效
      ,b.dept_id --部门ID
      ,case when coalesce(LEV2_DEPT_NAME,'') = '' then LEV1_DEPT_NAME
      	    when coalesce(LEV3_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME)
			when coalesce(LEV4_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME)
			when coalesce(LEV5_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME,'>',LEV4_DEPT_NAME)
			when coalesce(LEV6_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME,'>',LEV4_DEPT_NAME,'>',LEV5_DEPT_NAME)
			when coalesce(LEV7_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME,'>',LEV4_DEPT_NAME,'>',LEV5_DEPT_NAME,'>',LEV6_DEPT_NAME)
			when coalesce(LEV8_DEPT_NAME,'') = '' then concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME,'>',LEV4_DEPT_NAME,'>',LEV5_DEPT_NAME,'>',LEV6_DEPT_NAME,'>',LEV7_DEPT_NAME)
		    else concat(LEV1_DEPT_NAME,'>',LEV2_DEPT_NAME,'>',LEV3_DEPT_NAME,'>',LEV4_DEPT_NAME,'>',LEV5_DEPT_NAME,'>',LEV6_DEPT_NAME,'>',LEV7_DEPT_NAME,'>',LEV8_DEPT_NAME)
			 end as DEPT_JG_NAME --部门架构
      ,b.LEV1_DEPT_NAME --一级部门
	  ,REPLACE(LEV2_DEPT_NAME,'东鹏||','') AS LEV2_DEPT_NAME --二级部门
	  ,LEV3_DEPT_NAME --三级部门
	  ,LEV4_DEPT_NAME --四级部门
	  ,LEV5_DEPT_NAME --五级部门
	  ,LEV6_DEPT_NAME --六级部门
	  ,LEV7_DEPT_NAME --七级部门
	  ,LEV8_DEPT_NAME --八级部门
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
	  ,CASE WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('505') THEN C.SUPERIOR_DEPT_CODE
	        WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('504') THEN -1
			WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('2143') THEN -3
			ELSE C.SUPERIOR_DEPT_CODE
			 END AS FATHER_DEPT_ID --上级部门ID
	  ,C.DEPT_NAME AS DEPT_NAME --部门名称
	  ,CASE WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('505') THEN '东鹏'
            WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('504') THEN '董事会'
			WHEN C.SUPERIOR_DEPT_CODE = 0 AND C.BELONGING_BRANCH in('2143') THEN '丽适岩板'
    	    ELSE D.DEPT_NAME
 			 END AS FATHER_DEPT_NAME --上级部门名称
	  ,CASE WHEN C.SUPERIOR_DEPT_CODE = 0 THEN -2
	        WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('505') THEN 0
            WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('504') THEN -1
			WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('2143') THEN -3
    	    ELSE E.DEPT_ID
			 END AS last_father_dept_id --上上级部门ID
	  ,CASE WHEN C.SUPERIOR_DEPT_CODE = 0 THEN '东鹏控股'
	        WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('505') THEN '东鹏'
            WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('504') THEN '董事会'
			WHEN D.SUPERIOR_DEPT_CODE = 0 AND D.BELONGING_BRANCH in('2143') THEN '丽适岩板'
    	    ELSE E.DEPT_NAME
			 END AS last_father_dept_name --上上级部门名称
	  ,HR.ID AS EMP_ID --员工ID：对应企业微信的微信成员名
      ,CASE WHEN HR.STATUS = '0' THEN '试用'
	        WHEN HR.STATUS = '1' THEN '正式'	
			WHEN HR.STATUS = '2' THEN '临时'	
			WHEN HR.STATUS = '3' THEN '试用延期'	
			WHEN HR.STATUS = '4' THEN '解聘'	
			WHEN HR.STATUS = '5' THEN '离职'	
			WHEN HR.STATUS = '6' THEN '退休'	
			WHEN HR.STATUS = '7' THEN '无效'	
			ELSE ''
			 END AS STATUS_NAME --用户状态名称
	  ,HR.MOBILE AS PHONE_NUM --手机号码
	  ,HR.CREATED AS CREATE_TIME --创建时间
	  ,HR.MODIFIED AS UPDATE_TIME --更新时间
	  ,B.DEPT_CODE --部门编码
FROM (SELECT *
            ,ROW_NUMBER() OVER(PARTITION BY WORKCODE
			 ORDER BY CASE WHEN SAPID = WORKCODE THEN 0 ELSE 1 END ASC,LASTMODDATE DESC
					 ) AS RN
        FROM SDI.SDI_HRMRESOURCE_1004
	 ) HR
INNER JOIN (SELECT LEV1_DEPT_LEVEL
                  ,LEV1_DEPT_ID
                  ,LEV1_DEPT_NAME
                  ,LEV2_DEPT_LEVEL
                  ,LEV2_DEPT_ID
                  ,REPLACE(REPLACE(LEV2_DEPT_NAME,'东鹏||',''),'董事会||','') AS LEV2_DEPT_NAME
                  ,LEV3_DEPT_LEVEL
                  ,LEV3_DEPT_ID
                  ,LEV3_DEPT_NAME
                  ,LEV4_DEPT_LEVEL
                  ,LEV4_DEPT_ID
                  ,LEV4_DEPT_NAME
                  ,LEV5_DEPT_LEVEL
                  ,LEV5_DEPT_ID
                  ,LEV5_DEPT_NAME
                  ,LEV6_DEPT_LEVEL
                  ,LEV6_DEPT_ID
                  ,LEV6_DEPT_NAME
                  ,LEV7_DEPT_LEVEL
                  ,LEV7_DEPT_ID
                  ,LEV7_DEPT_NAME
                  ,LEV8_DEPT_LEVEL
                  ,LEV8_DEPT_ID
                  ,LEV8_DEPT_NAME
                  ,DEPT_ID
                  ,DW_LAST_UPDATE_DATE
                  ,DEPT_CODE
              FROM DM.DM_WF_DEPT_INFO_T) B
ON HR.DEPARTMENTID = B.DEPT_ID
LEFT JOIN DWRPI.DWR_WF_OA_DEPT_INTO_F C
ON B.DEPT_ID = C.DEPT_ID
AND COALESCE(C.SEAL_MARK,'') <> '1'
LEFT JOIN DWRPI.DWR_WF_OA_DEPT_INTO_F D
ON D.DEPT_ID = C.SUPERIOR_DEPT_CODE
AND COALESCE(D.SEAL_MARK,'') <> '1'
LEFT JOIN DWRPI.DWR_WF_OA_DEPT_INTO_F E
ON E.DEPT_ID = D.SUPERIOR_DEPT_CODE
AND COALESCE(E.SEAL_MARK,'') <> '1'
LEFT JOIN (SELECT PERNR
                 ,USRID
				 ,ROW_NUMBER() OVER(PARTITION BY PERNR ORDER BY ENDDA DESC,BEGDA DESC) AS RN
             FROM SDI.SDI_PA0105_1001
			WHERE MANDT = '800'
              AND SUBTY = 'EMAI'
		  ) F
ON HR.WORKCODE = F.PERNR
AND F.RN = 1
WHERE --HR.STATUS IN ('0', '1', '2', '3')
 COALESCE(hr.lastname,'') not in('OSS账号','PLM账号','直营价审非瓷砖事业部产品副总裁授权子公司总经理'
,'流程归档','系统传输账号','举报邮箱')
AND coalesce(hr.WORKCODE,'') <> ''
AND HR.RN = 1
;
