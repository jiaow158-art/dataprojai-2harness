--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DM.DM_LTC_ORDER_PLAN_PRODUCT_T
SELECT nvl(t.sales_order,t2.SALE_ORDER_HEAD_NO) as SALE_ORDER_HEAD_NO, -- 销售订单号  
       nvl(t.sales_order_line_number,t2.SALE_ORDER_LINE_NO) as SALE_ORDER_LINE_NO,-- 销售订单行号
	   t2.WBS_NUM, -- WBS编码
	   t2.WBS_NAME_CN, -- WBS名称
       t.shop_order as PLAN_NO, -- 报工号
	   t.REPRINT_SITUATION , -- 复版状态(0:未复版  1：已复版)
       t.TYPE as CHANGE_TYPE, -- 变更类型
       t.site_id  as FACTORY_ID, -- 工厂ID
       t.site as FACTORY_NUM, -- 工厂编码
       t.site_describe as FACTORY_NAME_CN, -- 工厂名称
       t.WORK_SHOP_ID as WORKSHOP_ID, -- 车间ID
       t.WORK_SHOP as WORKSHOP_CODE, -- 车间代码
       t.work_shop_desc as WORKSHOP_NAME_CN, -- 车间名称
       t.change_work_shop as CHANGE_WORKSHOP_CODE , -- 变更后车间代码
       t.CHANGE_WORK_SHOP_DESC as CHANGE_WORKSHOP_NAME_CN,  -- 变更后车间名称
       t.PRODUCT_LINE_ID,  -- 产品线id
       t.product_line as PRODUCT_LINE_CODE , -- 产线代码
       t.PRODUCT_LINE_DESC as PRODUCT_LINE_NAME_CN, -- 产线名称
       t.CHANGE_PRODUCT_LINE as CHANGE_PRODUCT_LINE_CODE, -- 变更后生产线代码
       t.change_product_line_desc as CHANGE_PRODUCT_LINE_NAME_CN , -- 变更后生产线名称
	   T2.CUSTOMER_NUM, 			-- 客户编码
       T2.CUSTOMER_NAME_CN, 		-- 客户名称
       t.material_code as MATERIAL_NUM , -- 物料编码
       t.MATERIAL_DESCRIPTION as MATERIAL_NAME_CN , -- 物料描述
       t.specifications_model , -- 规格型号
       t.WHETHER_CUSTOMIZED,  -- 是否定制
       t.planned_quantity,    -- 计划数量
       (t.planned_quantity * t1.area) as PLANNED_QUANTITY_AREA,  -- 计划数量面积
       t.change_quantity,  -- 变更后数量
       (t.change_quantity * t1.area) as CHANGE_QUANTITY_AREA,   -- 变更后数量面积
       t.inventory_quantity ,  -- 入库数量
       (t.inventory_quantity * t1.area) as INVENTORY_QUANTITY_AREA,  -- 入库数量面积
        case when t.INVENTORY_SITUATION = 0 then '未开始'
             when t.INVENTORY_SITUATION = 1 then '进行中'
             when t.INVENTORY_SITUATION = 2 then '已完成'
        end INVENTORY_SITUATION, --入库情况(0:未开始，1:进行中，2:已完成)
		t.PLANNED_START_TIME, --计划开始时间
		t.PLANNED_END_TIME, --计划结束时间
		t.CHANGE_PLAN_START_DATE, --变更后计划开始时间
		t.CHANGE_PLAN_END_DATE, --变更后计划结束时间
		t.RELEASE_TIME, --下达日期
		t.CHANGE_RELEASE_TIME, --变更后下达日期
		t.PRODUCT_CODE, --产品编号
		t.PACKAGE_CODE, --包装编码
		t.SPECIAL_PACKAGING, --特殊包装物包装
		t.SPECIAL_BRICK, --特殊砖底商标
		t.SPECIAL_SIZE, --特殊尺寸
		t.SPECIAL_THICKNESS, --特殊厚度
		t.TRIM_TEXTURE_COLOR, --微调纹理或颜色
		t.OTHER_SPECIAL, --其他特殊定制要求
		current_timestamp() as DW_LAST_UPDATE_DATE -- 同步数据时间
FROM SDI.sdi_dp_production_order_info_1006 t -- 生产工单数据表
left join SDI.SDI_ITEM_1007 t1   --物料
on t.material_code = t1.code 
and t1.is_phantom = 0
left join DM.DM_LTC_ORDER_PLAN_PS_T T2 -- 排产表
on t.shop_order = t2.plan_no;
