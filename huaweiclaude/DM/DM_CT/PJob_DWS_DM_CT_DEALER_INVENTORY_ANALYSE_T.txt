delete from DM.DM_CT_DEALER_INVENTORY_ANALYSE_T where PERIOD_ID_D = '${PERIOD_ID_D}';
insert into DM.DM_CT_DEALER_INVENTORY_ANALYSE_T
with MATERIAL_GENERAL_data as(  -- 物料数据
  select material_num,
  		 material_name,
  		 dimension,
  		 width,
  		 length_1,
  		 product_level2_name,
  		 reserved_field_6
  from DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T
  where matl_group_code = '1000010'
)
,dealer as  --经销商所属大区
(
select t.kunnr,
	   t.BZIRK_NAME,
	   t.name1,
	   t.freeze,
	   t.rn
 from(
 select distinct t.KUNNR as kunnr
	        ,case when t.BZIRK = 'B00018' then t1.subcompany_big_area_name
	          else t.BZIRK_NAME end as BZIRK_NAME
			,name1
	        ,t.freeze
	        ,row_number() over(partition by t.KUNNR order by nvl(t.freeze,1) asc) rn
	from DWIPI.DWI_WF_ZSD_ZMKB_T t
	left join dwrfin.dwr_gl_sales_group_v T1
	on t.vkgrp = t1.sales_group
	WHERE (t.BZIRK LIKE 'S%' OR t.BZIRK = 'B00018') 
	) t
	where rn = 1
)
,b1_customer_data as ( -- 瓷砖的经销商数据
select DISTINCT CUSTOMER_TYPE,
		case when length(customer_code) < 10 then concat('0000',customer_code) else customer_code end as customer_code,
		customer_name_cn 
 from dm.dm_b1_customer_data_t 
 where customer_type in ('CT')  
 and customer_flag = 'N'
 and share_stock_flag = 'N'
),
dim_data as(
  select 'dq' as day_type,'${PERIOD_ID_D}' day_date
  union all
  select 'hb' as day_type,substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) day_date
  union all
  select 'tb' as day_type,substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) day_date
),
ww_detail_data as (
select a.customer,
       --a.customername ,
	   a.material,
       sum(case when caltype='+' then quantity else 0-quantity end) ww_kc_qty_dq,
       sum(case when concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) <= substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) then  case when caltype='+' then quantity else 0-quantity end else 0 end) ww_kc_qty_hb,
       sum(case when concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) <= substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) then  case when caltype='+' then quantity else 0-quantity end else 0 end) ww_kc_qty_tb,
       max(case when caltype='+' then concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) else '' end) as ww_kc_date_dq,
       max(case when concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) <= substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) then  case when caltype='+' then concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) else '' end else '' end) as ww_kc_date_hb,
       max(case when concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) <= substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) then  case when caltype='+' then concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) else '' end else '' end) as ww_kc_date_tb
from dwiotd.DWI_WM_ZWW_STOCK_DATA_T a
where concat(substring(a.transmissiondate,1,4),'-',substring(a.transmissiondate,5,2),'-',substring(a.transmissiondate,7,2)) <= '${PERIOD_ID_D}'
group by a.customer,
		 --a.customername,
		 a.material
),
ww_detail_data_01 as(
select '${PERIOD_ID_D}' as stat_date, a.customer ,/*a.customername,*/a.material , ww_kc_qty_dq as kc_qty,ww_kc_date_dq as kc_date from ww_detail_data a
union all 
select substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) as stat_date, a.customer ,/*a.customername,*/a.material , ww_kc_qty_hb as kc_qty,ww_kc_date_hb as kc_date from ww_detail_data a
union all 
select substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) as stat_date, a.customer ,/*a.customername,*/a.material , ww_kc_qty_tb as kc_qty,ww_kc_date_tb as kc_date from ww_detail_data a
),
tmp_data as(
-- B1的明细数据
select  a.日期 as date_inv,
        coalesce(d.BZIRK_NAME,d2.BZIRK_NAME) BZIRK_NAME,
        case when length(a.客户别名) < 10 then concat('0000',a.客户别名) else a.客户别名 end as CUST_NUM,
	    coalesce(d.name1,d2.name1) customer_name_cn,
		--c.customer_name_cn,
		b.dimension,
		b.product_level2_name AS material_TYPE,
	    A.物料编码 as material_num,
	    A.物料描述 as material_name,
-- 		b.reserved_field_6 as wlz_desc,
		case when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 180 then '半年内'
		     when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 360 then '半年至一年'
			 when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 720 then '一年至两年'
			 else '两年以上'
			  end as kl_date,
	    sum(A.期末金额) as final_amount,
	    sum(A.期末数量) as final_quality,
		sum(a.期末数量 * b.width * b.length_1)/1000000 as area,
		'B1' as data_source_flag -- 数据来源标识
	from DM.DM_WM_B1_INVENTORY_CT_T_D  A 
	join MATERIAL_GENERAL_data   B 	on A.物料编码 = B.material_num 
	join b1_customer_data c	        on case when length(a.客户别名) < 10 then concat('0000',a.客户别名) else a.客户别名 end= c.customer_code
	left join dealer d on d.kunnr = case when length(a.客户别名) < 10 then concat('0000',a.客户别名) else a.客户别名 end
	and d.freeze isnull
	left join dealer d2 on d2.kunnr = case when length(a.客户别名) < 10 then concat('0000',a.客户别名) else a.客户别名 end 
	and d2.freeze = 'X'
	where 日期 IN (--'${PERIOD_ID_D}', -- 当前(传昨天)
				   --substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10), -- 环比
				   --substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) -- 同比
	           select day_date from dim_data
				   )
	group by a.日期,
	         coalesce(d.BZIRK_NAME,d2.BZIRK_NAME),
	         A.客户别名, 
			 coalesce(d.name1,d2.name1),
			 --c.customer_name_cn,
			 A.物料编码,  
			 A.物料描述,
			 b.dimension,
		     b.product_level2_name,
			 --b.reserved_field_6,
			 case when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 180 then '半年内'
		     when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 360 then '半年至一年'
			 when date_part('day',cast(日期 as date)-cast(最末入库日期 as date)) <= 720 then '一年至两年'
			 else '两年以上'
			  end
--万维的明细数据
	union all
	select a.stat_date as date_inv,
       coalesce(d.BZIRK_NAME,d2.BZIRK_NAME) BZIRK_NAME,
        a.customer as CUST_NUM,
		coalesce(d.name1,d2.name1) as customer_name_cn,
	    --a.customername as customer_name_cn,
		b.dimension,
		b.product_level2_name AS material_TYPE,
	    A.material as material_num,
	    b.material_name as material_name,
	    case when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 180 then '半年内'
		     when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 360 then '半年至一年'
			 when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 720 then '一年至两年'
			 else '两年以上'
			  end as kl_date,
	    0 as final_amount,
	    sum(A.kc_qty) as final_quality,
		sum(a.kc_qty * b.width * b.length_1)/1000000 as area,
		'WW' as data_source_flag -- 数据来源标识
	from ww_detail_data_01 a
	join MATERIAL_GENERAL_data   B 	on A.material = B.material_num 
	left join dealer d on d.kunnr = a.customer and d.freeze is null
	left join dealer d2 on d2.kunnr = case when length(a.customer) < 10 then concat('0000',a.customer) else a.customer end 
	and d2.freeze = 'X'
	where 
	/*a.customer in (
		'0000310184', -- 湖南富信供应链
		'0000310332', -- 兰州万恒公司
		'0000313056', -- 宜兴市聚品汇装饰材料有限公司
		'0000314175', -- 江苏镇江典尚
		'0000314852', -- 湖南张家界鹏越建材
		'0000315732', -- 江苏泰州经鼎名家
		'0000316040', -- 江苏盐城皇巢东鹏
		'0000317932', -- 湖北武汉连天红装饰
		'0000319572', -- 天河马会家居连锁专卖店
		'0000320098'  -- 辽宁大连鹏东建筑
	)
	and */ a.stat_date in(select day_date from dim_data)
	group by a.stat_date,
	         coalesce(d.BZIRK_NAME,d2.BZIRK_NAME),
	         A.customer, 
			 coalesce(d.name1,d2.name1),
			 --a.customername,
			 A.material,  
			 b.material_name,
			 case when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 180 then '半年内'
			     when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 360 then '半年至一年'
				 when date_part('day',cast(stat_date as date)-cast(kc_date as date)) <= 720 then '一年至两年'
				 else '两年以上'
				  end,
			 b.dimension,
		     b.product_level2_name
)
select  di.BZIRK_NAME,--大区
        di.CUST_NUM,--经销商编码
	    di.customer_name_cn,--经销商名称
	    di.dimension,--
	    di.material_TYPE,--物料类别
	    di.material_num,--物料编码
	    di.material_name,--物料名称
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}'  and kl_date in('半年内') then final_quality end ),0) as half_year,--半年内方数 -- 
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('半年内') then area end ),0) as half_year_area,--半年内面积数
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('半年至一年') then final_quality end ),0) as half_to_one_year,--半年-一年方数 -- 
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('半年至一年') then area end ),0) as half_to_one_year_area,--半年-一年面积数
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('一年至两年') then final_quality end ),0) as one_to_two_year,--一年-两年方数 -- 
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('一年至两年') then area end ),0) as one_to_two_year_area,--一年-两年面积数
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('两年以上') then final_quality end ),0) as over_two_year,--两年以上方数 --
	    nvl(sum(case when date_inv = '${PERIOD_ID_D}' and kl_date in('两年以上') then area end ),0) as over_two_year_area,--两年以上面积数
		current_timestamp as dw_last_update_date,    -- 数据同步时间
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) and kl_date in('半年内') then final_quality end ),0) as half_year_t,--半年内方数_同比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) and kl_date in('半年至一年') then final_quality end ),0) as half_to_one_year_t,---半年-一年方数_同比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) and kl_date in('一年至两年') then final_quality end ),0) as one_to_two_year_t,--一年-两年方数_同比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-12),1,10) and kl_date in('两年以上') then final_quality end ),0) as over_two_year_t,--两年以上方数_同比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) and kl_date in('半年内') then final_quality end ),0) as half_year_h,--半年内方数_环比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) and kl_date in('半年至一年') then final_quality end ),0) as half_to_one_year_h,---半年-一年方数_环比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) and kl_date in('一年至两年') then final_quality end ),0) as one_to_two_year_h,--一年-两年方数_环比
	    nvl(sum(case when date_inv = substring(add_months(to_char(to_date('${PERIOD_ID_D}','YYYY-MM-DD'),'YYYY-MM-DD'),-1),1,10) and kl_date in('两年以上') then final_quality end ),0) as over_two_year_h,--两年以上方数_环比
	    '${PERIOD_ID_D}' as PERIOD_ID_D, -- 当期日期
	    di.data_source_flag -- 数据来源标识
from tmp_data di
group by di.BZIRK_NAME,
        di.CUST_NUM,
	    di.customer_name_cn,
	    di.dimension,
	    di.material_TYPE,
	    di.material_num,
	    di.material_name,
	    di.data_source_flag;
