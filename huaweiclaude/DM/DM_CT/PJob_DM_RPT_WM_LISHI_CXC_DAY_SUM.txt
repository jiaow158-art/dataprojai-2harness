--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
----设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with mat as (
	select t1.material_num,
	    t1.product_level2_name  as cat,
	    t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        case 
            when t1.dimension in ('2780X1200','3200X1600','2600X800','2400X1200','2700X1200') then t1.dimension
            else '其他'
        end other_dimension,
        t2.matl_group1_code,
        t1.width * t1.length_1/1000/1000 as area
  from dwimd.dwi_md_data_material_general_t t1 
  left join (select material_num,max(matl_group1_code) matl_group1_code from dwimd.dwi_md_data_material_sales_t group by material_num) t2
        on t1.material_num = t2.material_num
  where 1=1
	and t1.prod_property = 'F0001'
	)
,stock_chg_det as(
-- 出入库
select  t1.VOUCHER_POST_DATE,
		t1.material_num,
		sum(case when MOVE_TYPE IN(
						 '601','602','653','654') then (case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end) end*t2.area) as sales_stock_out_area,
		sum(case when MOVE_TYPE IN(
						 '911','912','913','914','131','132','531','532' ) or (move_type IN('101','102','161') and t1.SPECIAL_STOCK_FLAG_2='K_T148T') then (case when DR_CR_ID='S_SHKZG' then QUANTITY else  -QUANTITY end*t2.area) end) as stock_in_area
from  dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join mat t2 
on t1.material_num  = t2.material_num 
where VOUCHER_POST_DATE>= concat(year('${PERIOD_ID_D}') - 1, '-01-01')
and  VOUCHER_POST_DATE< date_add('${PERIOD_ID_D}',1)
	and MOVE_TYPE IN(
					 '601','602','653','654',  -- 销售出库
					 '911','912','913','914','131','132','531','532','101','102','161'   -- 生产入库&采购入库
					 )
    group by t1.VOUCHER_POST_DATE ,t1.material_num
 ),
 stock_chg_sum as
 (
 select 
        t1.material_num,
        sum(case when VOUCHER_POST_DATE>= concat(substr('${PERIOD_ID_D}',1,4),'-01-01') then sales_stock_out_area else 0 end) sales_stock_out_area_year,
        sum(case when VOUCHER_POST_DATE>= concat(substr('${PERIOD_ID_D}',1,4),'-01-01') then stock_in_area else 0 end) stock_in_area_year,
        sum(case when VOUCHER_POST_DATE>= concat(substr('${PERIOD_ID_D}',1,7),'-01') then sales_stock_out_area else 0 end) sales_stock_out_area_month,
        sum(case when VOUCHER_POST_DATE>=concat(substr('${PERIOD_ID_D}',1,7),'-01') then stock_in_area else 0 end ) stock_in_area_month,
        sum(case when VOUCHER_POST_DATE>='${PERIOD_ID_D}' then sales_stock_out_area else 0 end) sales_stock_out_area_day,
        sum(case when VOUCHER_POST_DATE>='${PERIOD_ID_D}' then stock_in_area else 0 end ) stock_in_area_day,
        sum(case when substring(VOUCHER_POST_DATE,1,4) = year('${PERIOD_ID_D}') - 1 then sales_stock_out_area else 0 end) sales_stock_out_area_lastyear
 from stock_chg_det t1
 group by 
    t1.material_num
 ),
 stock_det as (
 select 
        t1.material_num,
        sum(case when stock_flg='本月库存' then stock_qty end*t2.area ) stock_area_month_end,
        sum(case when stock_flg='月初库存' then stock_qty end*t2.area ) stock_area_month_start,
        sum(case when stock_flg='年初库存' then stock_qty end*t2.area ) stock_area_year_start
 from 
 (
 -- 月初库存、年初库存、本月库存
select material_num,sum(stock_qty) as stock_qty, '本月库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr('${PERIOD_ID_D}',1,7),'-','')
group by  material_num
 union all 
select material_num,sum(stock_qty) as stock_qty,'月初库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,7),'-01'),1),1,7),'-','')
group by  material_num
union all
 select material_num,sum(stock_qty) as stock_qty,'年初库存' stock_flg  from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = replace(substr(date_sub(concat(substr('${PERIOD_ID_D}',1,4),'-01-01'),1),1,7),'-','')
group by  material_num
) t1
left join mat t2 on t1.material_num  = t2.material_num 
group by t1.material_num
),
mid as (
select  material_num,
        stock_area_month_end,
        stock_area_month_start,
        stock_area_year_start,
        0.0 as sales_stock_out_area_year,0.0 as stock_in_area_year,
        0.0 as sales_stock_out_area_month,0.0 as stock_in_area_month,
        0.0 as sales_stock_out_area_day,0.0 as stock_in_area_day,
        0.0 as sales_stock_out_area_lastyear
        from stock_det
union all 
select  material_num,
        0,0,0,
        sales_stock_out_area_year,stock_in_area_year,
        sales_stock_out_area_month,stock_in_area_month,
        sales_stock_out_area_day,stock_in_area_day,
        sales_stock_out_area_lastyear
        from stock_chg_sum
        )
INSERT OVERWRITE TABLE DM.DM_RPT_WM_LISHI_CXC_DAY_SUM PARTITION(start_date)
 select 
 		t2.matl_group1_code, --物料组1
		t2.other_dimension, --规格
        sum(t1.stock_area_month_end) as stock_area_month_end,  -- 月末库存
        sum(t1.stock_area_month_start) as stock_area_month_start, -- 月初库存
        sum(t1.stock_area_year_start) as stock_area_year_start, -- 年初库存
        sum(t1.sales_stock_out_area_year) as sales_stock_out_area_year,  -- 年累计销售出库
		sum(t1.stock_in_area_year) as stock_in_area_year, -- 年累计入库（自产或外协）
        sum(t1.sales_stock_out_area_month) as sales_stock_out_area_month,  -- 月累计销售出库
		sum(t1.stock_in_area_month) as stock_in_area_month, -- 月累计入库 （自产或外协）
        sum(t1.sales_stock_out_area_day) as sales_stock_out_area_day,  -- 日销售出库
		sum(t1.stock_in_area_day) as stock_in_area_day, -- 日入库 （自产或外协）
		sum(t1.sales_stock_out_area_lastyear) as sales_stock_out_area_lastyear,  -- 去年累计销售出库
		current_timestamp() as dw_last_update_date, -- 数据同步时间
		replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','') as start_date --分区字段(按天)
 from mid t1
 join mat t2 on t1.material_num = t2.material_num
 group by matl_group1_code,other_dimension
