WITH T1 AS(
SELECT SUBSTR(A.ENTRYDT,1,10) AS STAT_DATE --统计日期
	  ,COUNT(DISTINCT A.EXTERNAL_ORDER_CODE) AS TMS_ORDER_CNT --TMS订单数
  FROM SDI.SDI_TMS_ORDERDETAIL_1005 A
 INNER JOIN SDI.SDI_TMS_ORDER_1005 B
    ON A.ORDER_CODE = B.CODE
 WHERE A.EXTERNAL_ORDER_CODE IS NOT NULL
   AND SUBSTR(A.ENTRYDT,1,10) >= '2022-01-01'
 GROUP BY SUBSTR(A.ENTRYDT,1,10)
 )
,T2 AS(
SELECT SUBSTR(ERDAT,1,4) || '-' || SUBSTR(ERDAT,5,2) || '-' || SUBSTR(ERDAT,7,2) AS STAT_DATE
      ,COUNT(1) AS SAP_ORDER_CNT --SAP订单数
  FROM SDI.SDI_LIKP_1001
 WHERE MANDT = '800'
   AND ERDAT >= '20220101'
 GROUP BY SUBSTR(ERDAT,1,4) || '-' || SUBSTR(ERDAT,5,2) || '-' || SUBSTR(ERDAT,7,2)
) 
,T3 AS(
SELECT STAT_DATE --统计日期
      ,TMS_ORDER_CNT --TMS订单数
	  ,0 AS SAP_ORDER_CNT --SAP订单数
  FROM T1

 UNION ALL

SELECT STAT_DATE --统计日期
      ,0 AS TMS_ORDER_CNT --TMS订单数
	  ,SAP_ORDER_CNT --SAP订单数
  FROM T2 
)
INSERT OVERWRITE TABLE DM.DM_VM_TMS_ORDER_STAT_T
SELECT STAT_DATE --统计日期
      ,SUM(TMS_ORDER_CNT) AS TMS_ORDER_CNT --TMS订单数
	  ,SUM(SAP_ORDER_CNT) AS SAP_ORDER_CNT --SAP订单数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T3
 GROUP BY STAT_DATE
