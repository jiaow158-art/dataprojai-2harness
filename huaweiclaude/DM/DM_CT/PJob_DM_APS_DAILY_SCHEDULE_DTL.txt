with APS_PRO_ORDER
as(
SELECT  
        PRODUCTORDER.ID PRODUCT_ORDER_ID, 
        PRODUCTORDER.CODE PRODUCT_ORDER_CODE, 
        PRODUCTORDER.REQUIRE_DELIVERY PROMISE_DELIVERY, 
        PRODUCTORDER.REQUIRE_DELIVERY REQUIRE_DELIVERY, 
        PRODUCTORDER.ITEM_ID,
        coalesce(PRODUCTORDER1.ORDER_TYPE_SET_ID, 
        PRODUCTORDER.ORDER_TYPE_SET_ID) as ORDER_TYPE_SET_ID,  -- 合单前订单类型
        ordertype.name as ORDER_TYPE_SET_NAME
        
    FROM  sdi.sdi_APS_PRO_ORDER_1007 PRODUCTORDER 
    LEFT OUTER JOIN sdi.sdi_APS_PRO_ORDER_1007 PRODUCTORDER1 ON  PRODUCTORDER1.RELEVANT_CODE = PRODUCTORDER.CODE  -- 合并前工单
    LEFT OUTER JOIN sdi.sdi_order_type_setting_1007 ordertype ON  coalesce(PRODUCTORDER1.ORDER_TYPE_SET_ID, PRODUCTORDER.ORDER_TYPE_SET_ID) = ordertype.id
),
APS_PRO_ORDER1 as(
SELECT PRODUCT_ORDER_ID,
       PRODUCT_ORDER_CODE,
       PROMISE_DELIVERY,
       REQUIRE_DELIVERY,
       ITEM_ID,
      COUNT(DISTINCT ORDER_TYPE_SET_ID) ALL_CNT,
       COUNT(DISTINCT CASE WHEN ORDER_TYPE_SET_ID IN(1,13) THEN ORDER_TYPE_SET_ID END) MANUAL_CNT,
       COUNT(DISTINCT CASE WHEN ORDER_TYPE_SET_ID NOT IN(1,13) THEN ORDER_TYPE_SET_ID END) SALES_CNT,
       concat_ws(',',collect_set(ORDER_TYPE_SET_NAME)) AS ORDER_TYPE_SET_NAME_SET
       
FROM APS_PRO_ORDER
GROUP BY 
PRODUCT_ORDER_ID,
       PRODUCT_ORDER_CODE,
       PROMISE_DELIVERY,
       REQUIRE_DELIVERY,
       ITEM_ID
),
APS_PRO_ORDER2 as(
/*若报工单号对应的工单，合单前的生产订单类型
1、仅包含一种订单类型，赋值对应订单类型；
2、仅包含“新品首批订单”、“安全库存”两种，赋值“多类型手工单合单”；
3、仅包含“历史订单”、“工程样板间销售订单”、“工程补货销售订单”、“单体工程批量销售订单”、“战略工程计划订单”、“门店装修计划订单”、“零售计划订单”、“补货订单”两种及两种以上订单类型，赋值“多类型销售单合单”；
4、只要包含“新品首批订单”or“安全库存”一种及一种以上，且包含“历史订单”or“工程样板间销售订单”or“工程补货销售订单”or“单体工程批量销售订单”or“战略工程计划订单”or“门店装修计划订单”or“零售计划订单”or“补货订单”一种及一种以上，就赋值“销售与手工单合单”
*/
SELECT PRODUCT_ORDER_ID,
       PRODUCT_ORDER_CODE,
       PROMISE_DELIVERY,
       REQUIRE_DELIVERY,
       ITEM_ID,
       CASE WHEN ALL_CNT=1 THEN ORDER_TYPE_SET_NAME_SET
            WHEN MANUAL_CNT>1 AND SALES_CNT=0 THEN '多类型手工单合单'
            WHEN SALES_CNT>1 AND MANUAL_CNT=0 THEN '多类型销售单合单'
            WHEN MANUAL_CNT>=1 AND SALES_CNT>=1 THEN '销售与手工单合单' END AS  DEF_ORDER_TYPE_SET_NAME,
       ORDER_TYPE_SET_NAME_SET
FROM APS_PRO_ORDER1
),OEM_TASK AS(
SELECT DISTINCT A.APS_PLAN_TASK_DTL_ID
               ,TT.CODE AS SUPPLIER_CODE --供应商编码
			   ,TT.NAME AS SUPPLIER_NAME --供应商名称
			   ,A.PURCHASE_ORDER_NO AS PRCH_ORDER_NUM --采购订单号
			   ,A.PURCHASE_ORDER_PROJ_NO AS PRCH_ORDER_TIME_NUM --采购订单项目号
  FROM SDI.SDI_APS_OEM_PO_SUM_1007 A
  LEFT JOIN SDI.SDI_SUPPLIER_1007 TT
    ON A.SUPPLIER_ID = TT.ID
   AND TT.DEL_FLAG = 'N'	
 WHERE A.IS_PHANTOM = 0 --有效
   AND A.SUPPLIER_ID IS NOT NULL
   AND A.DEL_FLAG = 'N'
)
INSERT OVERWRITE TABLE dm.dm_aps_daily_schedule_dtl 
SELECT 
PLANTASKDETAIL.ID ID, 
PLANTASKDETAIL.ID PLAN_TASK_DETAIL_ID, 
TECHORDER.ID PROCESS_TECH_TASK_ID, 
TASK.ID PLAN_TASK_ID, 
PRODUCTORDER.PRODUCT_ORDER_ID, 
PRODUCTORDER.PRODUCT_ORDER_CODE, 
PRODUCTORDER.PROMISE_DELIVERY, 
PRODUCTORDER.REQUIRE_DELIVERY, 
PLANTASKDETAIL.WORK_ORDER OEM_PURCHASE_ORDER_NO, 
PROCESSTECH.ID PROCESS_TECH_ID, 
ITEM.ID ITEM_ID, 
ITEM.CODE ITEM_CODE, 
ITEM.NAME ITEM_NAME, 
ITEM.AREA,
ITEM.CATEGORY,  -- 品类
ITEM.PRODUCT_PROPERTY, -- 物料组1
WIPRES.ID RESOURCE_ID, 
WIPRES.CODE RESOURCE_CODE, 
WIPRES.NAME RESOURCE_NAME, 
T0.ID PRODUCT_AREA_ID, 
T0.CODE PRODUCT_AREA_CODE, 
T0.NAME PRODUCT_AREA_NAME, 
PLANTASKDETAIL.UPDATE_DATE LAST_UPDATE_DATE, 
PLANTASKDETAIL.PLAN_AMOUNT,
(PLANTASKDETAIL.PLAN_AMOUNT*ITEM.AREA) PLAN_AREA, 
TASK.PLAN_NO PLAN_NO, 
TASK.START_TIME START_TIME, 
TASK.END_TIME END_TIME, 
TASK.STATUS STATUS, 
TASK.IS_CANCEL IS_CANCEL, 
TASK.RE_STATE RE_STATE, 
TASK.RE_DATE RE_DATE,
PLANTASKDETAIL.RELEASE_START_TIME RELEASE_START_TIME,  -- 下达日期
TASK.CLOSE_FLAG CLOSE_FLAG,
TASK.CLOSE_QTY CLOSE_QTY,
TASK.CLOSE_TIME AS CLOSE_TIME,
-- 1、自制类入库批次从APS_DAILY_SCHEDULE获取 2、外协入库批次从sap获取
case when FINISHEDVM.BATCH_NO is null then sapstockin.STOCKIN_QTY else  FINISHEDVM.STOCKIN_QTY end as STOCKIN_QTY,
coalesce(FINISHEDVM.BATCH_NO,sapstockin.BATCH_NO) as batch_no,
coalesce(FINISHEDVM.stockin_time,sapstockin.stockin_time) as stockin_time,
PRODUCTORDER.DEF_ORDER_TYPE_SET_NAME,
PRODUCTORDER.ORDER_TYPE_SET_NAME_SET,
T2.ID WORKSHOP_ID, 
T2.CODE WORKSHOP_CODE, 
T2.NAME WORKSHOP_NAME, 
T3.ID FACTORY_ID, 
T3.CODE FACTORY_CODE, 
T3.NAME FACTORY_NAME,
uarfc.factory,
case when  coalesce(FINISHEDVM.stockin_time,sapstockin.stockin_time) <'2022-05-13' and uarfc.factory='湖南' and substr(coalesce(FINISHEDVM.BATCH_NO,sapstockin.BATCH_NO),-1,1)<>'7' 
	 then (if(FINISHEDVM.BATCH_NO is null, sapstockin.STOCKIN_QTY,FINISHEDVM.STOCKIN_QTY))*ITEM.AREA
     when  coalesce(FINISHEDVM.stockin_time,sapstockin.stockin_time) <'2022-05-13' and uarfc.factory<>'湖南' and substr(coalesce(FINISHEDVM.BATCH_NO,sapstockin.BATCH_NO),-1,1) not in('7','9') 
     then (if(FINISHEDVM.BATCH_NO is null, sapstockin.STOCKIN_QTY,FINISHEDVM.STOCKIN_QTY))*ITEM.AREA
     when  coalesce(FINISHEDVM.stockin_time,sapstockin.stockin_time) >='2022-05-13' and substr(coalesce(FINISHEDVM.BATCH_NO,sapstockin.BATCH_NO),-1,1) not in('7','9') then (if(FINISHEDVM.BATCH_NO is null, sapstockin.STOCKIN_QTY,FINISHEDVM.STOCKIN_QTY))*ITEM.AREA 
     else 0 end qc_stockin_area,
T4.SUPPLIER_CODE, --供应商编码
T4.SUPPLIER_NAME, --供应商名称
T4.PRCH_ORDER_NUM, --采购订单号
T4.PRCH_ORDER_TIME_NUM --采购订单项目号
FROM sdi.sdi_APS_PLAN_TASK_DTL_1007 PLANTASKDETAIL
    LEFT OUTER JOIN sdi.sdi_APS_PLAN_TASK_1007 TASK ON PLANTASKDETAIL.PLAN_TASK_ID = TASK.ID  
    LEFT OUTER JOIN sdi.sdi_RES_WIP_SCHE_1007 WIPRES ON TASK.RESOURCE_ID = WIPRES.ID  
    LEFT OUTER JOIN sdi.sdi_APS_PROTECH_ORDER_1007 TECHORDER ON PLANTASKDETAIL.PROCESS_TECH_TASK_ID = TECHORDER.ID  
    LEFT OUTER JOIN APS_PRO_ORDER2 AS PRODUCTORDER ON PRODUCTORDER.PRODUCT_ORDER_ID = TECHORDER.PRODUCT_ORDER_ID
    LEFT OUTER JOIN sdi.sdi_ITEM_1007 ITEM ON PRODUCTORDER.ITEM_ID = ITEM.ID  
    LEFT OUTER JOIN sdi.sdi_APS_PROTECH_1007 PROCESSTECH ON TECHORDER.PROCESS_TECH_ID = PROCESSTECH.ID  
    left join sdi.sdi_aps_resource_factory_cfg_1037 uarfc ON WIPRES.NAME = uarfc.resource_name
    LEFT OUTER JOIN (SELECT DAILYSCHEDULE.PLAN_TASK_ID PLANTASKID, 
                            DAILYSCHEDULE.PLAN_TASK_ID PLAN_TASK_ID, 
                            DAILYSCHEDULE.PROCESS_TECH_ORDER_ID PROCESS_TECH_TASK_ID,
                            DAILYSCHEDULE.QTY AS  STOCKIN_QTY,
                            BATCH AS BATCH_NO ,  
                            DAILYSCHEDULE.create_date as  stockin_time            
FROM sdi.sdi_APS_DAILY_SCHEDULE_1007 DAILYSCHEDULE
) FINISHEDVM
ON PLANTASKDETAIL.PLAN_TASK_ID = FINISHEDVM.PLAN_TASK_ID 
AND PLANTASKDETAIL.PROCESS_TECH_TASK_ID = FINISHEDVM.PROCESS_TECH_TASK_ID
LEFT OUTER JOIN 
( -- 外协入库批次信息
SELECT AN.BANFN AS work_order
        ,KO.LIFNR ,PO.WERKS ,PO.LGORT ,BE.CHARG as batch_no, from_unixtime(unix_timestamp( BE.BUDAT,'yyyymmdd'),'yyyy-mm-dd') as stockin_time,
CASE WHEN BE.SHKZG = 'S' 
THEN BE.MENGE ELSE 0 - BE.MENGE END AS STOCKIN_QTY 
FROM sdi.sdi_EBAN_1001 AN  
left JOIN sdi.sdi_EKPO_1001 AS PO ON PO.BANFN = AN.BANFN AND PO.BNFPO = AN.BNFPO 
INNER JOIN sdi.sdi_EKKO_1001 AS KO ON KO.EBELN = PO.EBELN  
LEFT JOIN sdi.sdi_EKBE_1001 AS BE ON PO.EBELN = BE.EBELN 
AND PO.EBELP = BE.EBELP  WHERE AN.BNFPO = '00010'
AND AN.BSART = 'ZNB1' AND BE.BEWTP = 'E'  
) sapstockin ON FINISHEDVM.batch_no is null and PLANTASKDETAIL.WORK_ORDER = sapstockin.work_order
 LEFT OUTER JOIN sdi.sdi_RES_ENTERPRISE_1007 T0 ON WIPRES.FACTORY_ID = T0.ID
 LEFT OUTER JOIN sdi.sdi_RES_ENTERPRISE_1007 T2 ON WIPRES.WORK_SHOP_ID = T2.ID 
 LEFT OUTER JOIN sdi.sdi_RES_ENTERPRISE_1007 T3 ON T2.TREE_PID = T3.ID
 LEFT JOIN OEM_TASK T4 ON T4.APS_PLAN_TASK_DTL_ID = PLANTASKDETAIL.ID
 WHERE 1=1 
 AND IS_CANCEL =0 
 AND RELEASE_start_time  IS NOT NULL;
