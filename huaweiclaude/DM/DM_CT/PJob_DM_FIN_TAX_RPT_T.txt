WITH T0 AS(
SELECT DISTINCT
       C.ERP_VOUCHER_NO --财务凭证号
      ,A.ERP_ITEM_NUMBER --财务凭证行
      ,SUBSTR(B.POSTING_DATE,1,4) AS POST_YEAR --年份
	  ,B.CORPORATION_CODE AS COMP_CODE --公司代码
  FROM SDI.SDI_VOUCHER_ITEM_1039 A
  LEFT JOIN SDI.SDI_VOUCHER_HEAD_1039 B ON A.HEAD_ID = B.ID 
  LEFT JOIN SDI.SDI_VOUCHER_STATE_1039 C ON C.HEAD_ID = B.ID 
  LEFT JOIN SDI.SDI_SHARE_ORDER_HEAD_1039 D ON B.SHARE_ORDER_ID = D.ID
 INNER JOIN SDI.SDI_BUSINESS_TYPE_1039 E ON D.BUSINESS_TYPE_ID = E.ID
   AND E.CODE = '001621' --结转全年增值税二级科目余额
)
,T AS(
SELECT  A.FIN_VOUCHER_LINE_CODE --财务凭证行
       ,A.FIN_VOUCHER_CODE --财务凭证号
       ,A.POSTING_DATE --日期
       ,A.MONTH AS STAT_MONTH --月份
       ,A.YEAR --年份
       ,A.COST_CENTER_CODE --成本中心编码
       ,A.COST_CENTER_DESC --成本中文名称
       ,A.FUNC_SCOPE_CODE --功能范围编码
       ,A.FUNC_SCOPE_NAME --功能范围名称
       ,A.GENERAL_LEDGER_ACCOUNT --总账科目
       ,A.GENERAL_LEDGER_ACCOUNT_DESC --总账科目名称
       ,A.MATERIAL_VOUCHER_CODE --物料凭证
       ,A.COMP_CODE --公司代码
       ,B.COMPANY_NAME AS COMP_DESC --公司代码描述
       ,A.PROFIT_CENTER_CODE --利润中心编码
       ,A.PROFIT_CENTER_DESC --利润中心名称
       ,A.SALES_GROUP_CODE --销售组编码
       ,A.SALES_GROUP_DESC --销售组名称
       ,A.LOCAL_CURRENCY_UNIT --金额单位
       ,A.LOCAL_CURRENCY_AMT --金额
	   ,A.CUST_GROUP_CODE --客户组
	   ,A.CHANNEL_CODE --分销渠道
	   ,A.MATERIAL_CODE --物料编码
	   ,CASE WHEN T0.POST_YEAR IS NOT NULL THEN 1 ELSE 0 END AS IS_GX --是否001621凭证
  FROM DWRFIN.DWR_FIN_COST_D_COMPRE_SUBJ_T A
  LEFT JOIN T0
    ON A.FIN_VOUCHER_CODE = T0.ERP_VOUCHER_NO
   AND CAST(A.FIN_VOUCHER_LINE_CODE AS DECIMAL(38)) = T0.ERP_ITEM_NUMBER
   AND A.YEAR = T0.POST_YEAR
   AND A.COMP_CODE = T0.COMP_CODE
  LEFT JOIN (SELECT DISTINCT B.COMPANY_CODE,B.COMPANY_NAME
               FROM DWRDIM.DWR_DIM_COMPANY_D B
			  WHERE B.GROUP_NUM = '800'
                AND SUBSTR(B.END_DATE,1,10) = '4712-12-31'
			) B
    ON A.COMP_CODE = B.COMPANY_CODE
 WHERE A.GENERAL_LEDGER_ACCOUNT IN('0021710101','0021710102','0021710104','0021710105','0021710106','0021710107','0021710108','0021710109','0021710110'
                                  ,'0021710200','0021713000','0021710500','0057010100','0054020200','0054020700','0054020800','0054020600','0054021000'
                                  ,'0054020500','0054020300','0054020400','0021710400','0051020200','0051020300','0051020400','0051020100','0051020110'
                                  ,'0051020210','0051029900','0051010300','0051010101','0051010102','0051010400','0051010103','0051010900','0051010104'
                                  ,'0051010200','0021710105','0021710101','0021710107')
),T1 AS( --当月
SELECT  T.COMP_CODE --公司代码
       ,T.COMP_DESC --公司代码描述
	   ,T.STAT_MONTH --月份
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710200' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_OUT_DRZ_AMT --外部待认证科目余额（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021713000' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_IN_DRZ_AMT --公司间待认证余额（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710500' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_VAT_AMT --增值税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT IN('0021710101','0021710102','0021710104','0021710105','0021710106','0021710107','0021710108','0021710109','0021710110') AND T.IS_GX = 0 THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_IA_VAT_AMT --账载增值税（系统增值税二级科目）（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0057010100' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_CIT_AMT --企业所得税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020200' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_CMCT_AMT --城市维护建设税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020700' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_ES_AMT --教育附加费（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020800' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_LES_AMT --地方教育费附加（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020600' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_EBM_AMT --堤围费（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054021000' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_ST_AMT --印花税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020500' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_EPT_AMT --环境保护税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020300' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_PT_AMT --房产税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0054020400' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_LST_AMT --土地使用税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710400' THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_IIT_AMT --个人所得税（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT IN('0051020200','0051020300','0051020400','0051020100','0051020110','0051020210','0051029900','0051010300','0051010101','0051010102','0051010400','0051010103','0051010900','0051010104','0051010200') THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_OR_AMT --营业收入（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710105' AND T.IS_GX = 0 THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_OT_AMT --销项税额累计数（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710101' AND T.IS_GX = 0 THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_IT_AMT --进项税累计数（当月）
	   ,SUM(CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0021710107' AND T.IS_GX = 0 THEN LOCAL_CURRENCY_AMT ELSE 0 END) AS CM_ITT_AMT --进项税转出（当月）
  FROM T
 GROUP BY T.COMP_CODE --公司代码
       ,T.COMP_DESC --公司代码描述
	   ,T.STAT_MONTH --月份
),T2 AS(
SELECT  B.COMP_CODE --公司代码
       ,B.COMP_DESC --公司代码描述
	   ,A.STAT_MONTH --月份
	   ,COALESCE(C.CM_OUT_DRZ_AMT,0) AS CM_OUT_DRZ_AMT --外部待认证科目余额（当月）
	   ,COALESCE(C.CM_IN_DRZ_AMT,0) AS CM_IN_DRZ_AMT --公司间待认证余额（当月）
	   ,COALESCE(C.CM_VAT_AMT,0) AS CM_VAT_AMT --增值税（当月）
	   ,COALESCE(C.CM_IA_VAT_AMT,0) AS CM_IA_VAT_AMT --账载增值税（系统增值税二级科目）（当月）
	   ,COALESCE(C.CM_CIT_AMT,0) AS CM_CIT_AMT --企业所得税（当月）
	   ,COALESCE(C.CM_CMCT_AMT,0) AS CM_CMCT_AMT --城市维护建设税（当月）
	   ,COALESCE(C.CM_ES_AMT,0) AS CM_ES_AMT --教育附加费（当月）
	   ,COALESCE(C.CM_LES_AMT,0) AS CM_LES_AMT --地方教育费附加（当月）
	   ,COALESCE(C.CM_EBM_AMT,0) AS CM_EBM_AMT --堤围费（当月）
	   ,COALESCE(C.CM_ST_AMT,0) AS CM_ST_AMT --印花税（当月）
	   ,COALESCE(C.CM_EPT_AMT,0) AS CM_EPT_AMT --环境保护税（当月）
	   ,COALESCE(C.CM_PT_AMT,0) AS CM_PT_AMT --房产税（当月）
	   ,COALESCE(C.CM_LST_AMT,0) AS CM_LST_AMT --土地使用税（当月）
	   ,COALESCE(C.CM_IIT_AMT,0) AS CM_IIT_AMT --个人所得税（当月）
	   ,COALESCE(C.CM_OR_AMT,0) AS CM_OR_AMT --营业收入（当月）
	   ,COALESCE(C.CM_OT_AMT,0) AS CM_OT_AMT --销项税额累计数（当月）
	   ,COALESCE(C.CM_IT_AMT,0) AS CM_IT_AMT --进项税累计数（当月）
	   ,COALESCE(C.CM_ITT_AMT,0) AS CM_ITT_AMT --进项税转出（当月）
  FROM (SELECT DISTINCT T1.STAT_MONTH
          FROM T1
	   ) AS A
  LEFT JOIN (SELECT DISTINCT T1.COMP_CODE,T1.COMP_DESC
               FROM T1
            ) AS B
	ON 1 = 1
  LEFT JOIN T1 AS C
    ON B.COMP_CODE = C.COMP_CODE
   AND A.STAT_MONTH = C.STAT_MONTH
)
INSERT OVERWRITE TABLE DM.DM_FIN_TAX_RPT_T
SELECT  COMP_CODE --公司代码
       ,COMP_DESC --公司代码描述
	   ,STAT_MONTH --月份
	   ,CM_OUT_DRZ_AMT --外部待认证科目余额（当月）
	   ,SUM(CM_OUT_DRZ_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_OUT_DRZ_AMT --外部待认证科目余额（累计）
	   ,CM_IN_DRZ_AMT --公司间待认证余额（当月）
	   ,SUM(CM_IN_DRZ_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_IN_DRZ_AMT --公司间待认证余额（累计）
	   ,CM_VAT_AMT --增值税（当月）
	   ,SUM(CM_VAT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_VAT_AMT --增值税（累计）
	   ,CM_IA_VAT_AMT --账载增值税（系统增值税二级科目）（当月）
	   ,SUM(CM_IA_VAT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_IA_VAT_AMT --账载增值税（系统增值税二级科目）（累计）
	   ,CM_CIT_AMT --企业所得税（当月）
	   ,SUM(CM_CIT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_CIT_AMT --企业所得税（累计）
	   ,CM_CMCT_AMT --城市维护建设税（当月）
	   ,SUM(CM_CMCT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_CMCT_AMT --城市维护建设税（累计）
	   ,CM_ES_AMT --教育附加费（当月）
	   ,SUM(CM_ES_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_ES_AMT --教育附加费（累计）
	   ,CM_LES_AMT --地方教育费附加（当月）
	   ,SUM(CM_LES_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_LES_AMT --地方教育费附加（累计）
	   ,CM_EBM_AMT --堤围费（当月）
	   ,SUM(CM_EBM_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_EBM_AMT --堤围费（累计）
	   ,CM_ST_AMT --印花税（当月）
	   ,SUM(CM_ST_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_ST_AMT --印花税（累计）
	   ,CM_EPT_AMT --环境保护税（当月）
	   ,SUM(CM_EPT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_EPT_AMT --环境保护税（累计）
	   ,CM_PT_AMT --房产税（当月）
	   ,SUM(CM_PT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_PT_AMT --房产税（累计）
	   ,CM_LST_AMT --土地使用税（当月）
	   ,SUM(CM_LST_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_LST_AMT --土地使用税（累计）
	   ,CM_IIT_AMT --个人所得税（当月）
	   ,SUM(CM_IIT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_IIT_AMT --个人所得税（累计）
	   ,CM_OR_AMT --营业收入（当月）
	   ,SUM(CM_OR_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_OR_AMT --营业收入（累计）
	   ,CM_OT_AMT --销项税额累计数（当月）
	   ,SUM(CM_OT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_OT_AMT --销项税额累计数（累计）
	   ,CM_IT_AMT --进项税累计数（当月）
	   ,SUM(CM_IT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_IT_AMT --进项税累计数（累计）
	   ,CM_ITT_AMT --进项税转出（当月）
	   ,SUM(CM_ITT_AMT) OVER(PARTITION BY COMP_CODE ORDER BY STAT_MONTH ASC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS TOT_ITT_AMT --进项税转出（累计）
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T2
;
