set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH RK_DATE1 AS( --入库日期
SELECT
        MATERIAL_NUM --物料编码
       ,BATCH_NUM --批次
	   ,CASE WHEN COALESCE(SUBSTR(CAST(PRODUCTION_DATE AS STRING),1,10),'') IN('','1900-01-01','4712-12-31') THEN VOUCHER_POST_DATE
	         ELSE PRODUCTION_DATE
			  END AS VOUCHER_POST_DATE 
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F --物料移动表   

 UNION ALL
 
SELECT
        MATNR as MATERIAL_NUM --物料编码
       ,charg as BATCH_NUM --批次
	   ,CASE WHEN COALESCE(SUBSTR(CAST(PRODUCTION_DATE AS STRING),1,10),'') IN('','1900-01-01','4712-12-31') THEN VOUCHER_POST_DATE
	         ELSE PRODUCTION_DATE
			  END AS VOUCHER_POST_DATE 
  FROM sdi.sdi_matdoc_1026 --物料移动表
 WHERE mjahr < '2019'
),RK_DATE AS(
SELECT
        MATERIAL_NUM --物料编码
       ,BATCH_NUM --批次
	   ,MIN(VOUCHER_POST_DATE) AS MIN_VOUCHER_POST_DATE
	   ,MAX(VOUCHER_POST_DATE) AS MAX_VOUCHER_POST_DATE
  FROM RK_DATE1
 GROUP BY 
        MATERIAL_NUM --物料编码
       ,BATCH_NUM --批次
)

,PRICE AS(
SELECT MATNR
      ,BWKEY
	  ,COALESCE(E.BWTAR,'') BWTAR
	  ,LFGJA
	  ,LFMON
	  ,VERPR
	  ,PEINH
	  ,CAST(CONCAT(LFGJA,LFMON) AS INT) AS STAT_MONTH
  FROM SDI.SDI_MBEWH_1001 E
 WHERE E.DEL_FLAG = 'N'
   AND E.VPRSV = 'V'

 UNION ALL

SELECT MATNR
      ,BWKEY
	  ,COALESCE(E.BWTAR,'') BWTAR
	  ,LFGJA
	  ,LFMON
	  ,VERPR
	  ,PEINH
	  ,CAST(CONCAT(LFGJA,LFMON) AS INT) AS STAT_MONTH
  FROM SDI.SDI_MBEW_1001 E
 WHERE E.DEL_FLAG = 'N'
   AND E.VPRSV = 'V' 
),PRICE1 AS(
SELECT MATNR
      ,BWKEY
	  ,BWTAR
	  ,LFGJA
	  ,LFMON
	  ,VERPR
	  ,PEINH
	  ,STAT_MONTH
	  ,ROW_NUMBER() OVER(PARTITION BY MATNR,BWKEY,BWTAR ORDER BY STAT_MONTH DESC) AS RN
  FROM PRICE
),PRICE2 AS(
SELECT MATNR
      ,BWKEY
	  ,BWTAR
	  ,LFGJA
	  ,LFMON
	  ,VERPR
	  ,PEINH
	  ,STAT_MONTH
	  ,ROW_NUMBER() OVER(PARTITION BY MATNR,BWKEY ORDER BY STAT_MONTH DESC) AS RN
  FROM PRICE
 WHERE COALESCE(BWTAR,'') = ''
),SUPPLIER_OR_CREDITOR_DATA AS(
  SELECT MATERIAL_NUM
        ,BATCH_NUM
		,SUPPLIER_OR_CREDITOR_ID
		,SUPPLIER_OR_CREDITOR_ACCT
		,SUPPLIER_OR_CREDITOR_NAME
		,ROW_NUMBER() OVER(PARTITION BY MATERIAL_NUM,BATCH_NUM ORDER BY VOUCHER_POST_DATE ASC) AS RN
    FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
   WHERE COALESCE(A.SUPPLIER_OR_CREDITOR_ACCT,'') NOT IN('','00000000') 
)

INSERT OVERWRITE TABLE DM.DM_WM_PRODSTOCK_OUT_IN_T PARTITION(STAT_MONTH)
SELECT
       A.MATERIAL_ID --物料ID
      ,A.MATERIAL_NUM --物料编码
	  ,A.MATERIAL_NAME --物料名称
	  ,C.EXTERNAL_MATL_GROUP_CODE --外部物料组
	  ,C.MATL_GROUP_CODE --物料组
	  ,C.SIZE_UNIT --规格单位
	  ,C.LENGTH_1 --长度
	  ,C.WIDTH --宽度
	  ,C.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道编码
	  ,C2.CONFIG_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称
	  ,C1.VTEXT AS PRODUCT_STAGE2 --产品层次2
	  ,CASE WHEN COALESCE(BD.BRAND_NAME,'') <> '' THEN BD.BRAND_NAME
	        WHEN COALESCE(BD1.BRAND_NAME,'') <> '' THEN BD1.BRAND_NAME
	        WHEN COALESCE(BD2.BRAND_NAME,'') <> '' THEN BD2.BRAND_NAME
			ELSE 'NA'
			 END AS BRAND_NAME --品牌名称
      ,A.FACTORY_WERKS_ID --工厂ID
      ,A.FACTORY_WERKS_CODE --工厂编码
	  ,A.FACTORY_WERKS_NAME --工厂名称	  
      ,A.BATCH_NUM --批次
	  ,A.MOVE_TYPE --移动类型
	  ,A.SPECIAL_STOCK_FLAG_2 --特殊库存标识
	  ,A.ORDER_NUM --订单号
	  ,A.ORDER_PROJECT_NO --订单项目号
	  ,A.INV_LOCATION --库存地点
	  ,A.STOCK_QTY --库存数量
      ,A.DR_CR_ID --借方/贷方标识
	  ,A.PURCHASE_VOUCHER_NUM --采购订单号
	  ,A.PURCHASE_VOUCHER_ITEM_NUM --采购凭证的项目编号
	  ,CASE WHEN COALESCE(A.SUPPLIER_OR_CREDITOR_ACCT,'') IN('','00000000') 
			THEN SOC.SUPPLIER_OR_CREDITOR_ID
			ELSE A.SUPPLIER_OR_CREDITOR_ID
			 END SUPPLIER_OR_CREDITOR_ID --供应商ID
	  ,CASE WHEN COALESCE(A.SUPPLIER_OR_CREDITOR_ACCT,'') IN('','00000000') 
			THEN SOC.SUPPLIER_OR_CREDITOR_ACCT
			ELSE A.SUPPLIER_OR_CREDITOR_ACCT
			 END SUPPLIER_OR_CREDITOR_ACCT --供应商编码
	   ,CASE WHEN COALESCE(A.SUPPLIER_OR_CREDITOR_ACCT,'') IN('','00000000') 
			 THEN SOC.SUPPLIER_OR_CREDITOR_NAME
			 ELSE A.SUPPLIER_OR_CREDITOR_NAME
			  END SUPPLIER_OR_CREDITOR_NAME --供应商名称
      ,D.PRODUCT_AREA_ID --产区ID
      ,D.PRODUCT_AREA_CODE --产区编码
      ,D.PRODUCT_AREA_NAME --产区名称	  
	  ,A.PRODUCTION_DATE --生产日期
	  ,A.VOUCHER_DATE --凭证中的凭证日期
      ,A.VOUCHER_POST_DATE --凭证中的过帐日期
      ,A.REFER_VCH_NO --参考凭证的凭证号
      ,A.CONSUME_QTY --消耗数量
      ,A.VOUCHER_HEADER_TEXT --凭证抬头文本
      ,A.MATERIAL_VCH_NO_MBLNR --物料凭证编号
      ,A.COMMITMENT_PROJ_ID --承诺项目
	  ,A.CUST_ID --客户
	  ,A.CUST_NUM --客户
	  ,A.CUST_NAME --客户
      ,A.QUANTITY --数量
	  ,COALESCE(E.VERPR,F.VERPR) AS AVERAGE_PRICE --移动平均价
	  ,E.PEINH AS PRICE_UNIT --价格单位
	  ,CASE WHEN A.DR_CR_ID = 'H_SHKZG' THEN 0 - A.QUANTITY
		    WHEN A.DR_CR_ID = 'S_SHKZG' THEN A.QUANTITY
			END AS WAREHOUSE_QUANTITY --库存量	   
      ,CASE WHEN A.DR_CR_ID = 'S_SHKZG' AND A.BASE_MEASURE_UNIT = 'M2' THEN A.QUANTITY
	        WHEN A.DR_CR_ID = 'H_SHKZG' AND A.BASE_MEASURE_UNIT = 'M2' THEN 0 - A.QUANTITY
            WHEN A.DR_CR_ID = 'S_SHKZG' AND A.BASE_MEASURE_UNIT = 'PC' THEN A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000 
	        WHEN A.DR_CR_ID = 'H_SHKZG' AND A.BASE_MEASURE_UNIT = 'PC' THEN 0 - A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000
            WHEN A.DR_CR_ID = 'S_SHKZG' AND A.BASE_MEASURE_UNIT = 'SET' THEN A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000 
	        WHEN A.DR_CR_ID = 'H_SHKZG' AND A.BASE_MEASURE_UNIT = 'SET' THEN 0 - A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000
            WHEN A.DR_CR_ID = 'S_SHKZG' AND A.BASE_MEASURE_UNIT = 'Z33' THEN A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000 
	        WHEN A.DR_CR_ID = 'H_SHKZG' AND A.BASE_MEASURE_UNIT = 'Z33' THEN 0 - A.QUANTITY * C.LENGTH_1 * C.WIDTH /1000000
	         END AS WAREHOUSE_AREA --库存面积（平方米）
	   
	  ,CASE WHEN A.DR_CR_ID = 'H_SHKZG' THEN 0 - A.QUANTITY * COALESCE(E.VERPR,F.VERPR)
		    WHEN A.DR_CR_ID = 'S_SHKZG' THEN A.QUANTITY * COALESCE(E.VERPR,F.VERPR)
			END AS WAREHOUSE_PRICE --库存金额（元）		 
	  ,A.CURRENCY_CODE --货币码
      ,A.WMS_PALLET_QTY --WMS 货盘数
      ,A.STOCK_TYPE_INSMK --库存类型
      ,COALESCE(B.MIN_VOUCHER_POST_DATE,from_unixtime(unix_timestamp('20191231','yyyyMMdd'),'yyyy-MM-dd')) AS MIN_VOUCHER_POST_DATE --批次最早过账日期
	  ,COALESCE(B.MAX_VOUCHER_POST_DATE,from_unixtime(unix_timestamp('20191231','yyyyMMdd'),'yyyy-MM-dd')) AS MAX_VOUCHER_POST_DATE --批次最晚过账日期
	  ,A.WBS_ELEMENT --WBS号
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,REPLACE(SUBSTR(date_format('${PERIOD_ID_D}','yyyy-MM-dd'),1,7),'-','') AS STAT_MONTH
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
  LEFT JOIN RK_DATE B
    ON A.MATERIAL_NUM = B.MATERIAL_NUM
   AND COALESCE(A.BATCH_NUM,'') = COALESCE(B.BATCH_NUM,'')
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D C
    ON A.MATERIAL_ID = C.MATERIAL_ID
   AND C.end_date = '4712-12-31'
  LEFT JOIN SDI.SDI_T179T_1001 C1 --产品品类
    ON C1.MANDT = C.GROUP_NUM
   AND C1.SPRAS = '1'
   AND C1.DEL_FLAG = 'N'
   AND SUBSTR(C.PRODUCT_LEVEL_CODE,1,4) = C1.PRODH
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R C2
    ON C.RESERVED_FIELD_1 = C2.SRC_SYS_CONFIG_CODE
   AND C2.SOURCE_CODE = 'ZTYL01'
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD
    ON A.FACTORY_WERKS_CODE = BD.FACTORY_WERKS_CODE
   AND C.EXTERNAL_MATL_GROUP_CODE = BD.EXTERNAL_MATL_GROUP_CODE
   AND C.MATL_GROUP_CODE = BD.MATL_GROUP_CODE
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD1
    ON A.FACTORY_WERKS_CODE = BD1.FACTORY_WERKS_CODE
   AND BD1.EXTERNAL_MATL_GROUP_CODE = ''
   AND C.MATL_GROUP_CODE = BD1.MATL_GROUP_CODE 
  LEFT JOIN (SELECT DISTINCT FACTORY_WERKS_CODE
                            ,BRAND_NAME
               FROM DWRDIM.DWR_DIM_MATERIAL_BRAND_D
			  WHERE MATL_GROUP_CODE = ''
			) BD2
    ON A.FACTORY_WERKS_CODE = BD2.FACTORY_WERKS_CODE
  LEFT JOIN (
       select 
        	 SUPPLIER_OR_CREDITOR_ID
             ,PRODUCT_AREA_ID
             ,PRODUCT_AREA_CODE
             ,PRODUCT_AREA_NAME
        from 
            (
            SELECT SUPPLIER_OR_CREDITOR_ID
                 ,PRODUCT_AREA_ID
                 ,PRODUCT_AREA_CODE
                 ,PRODUCT_AREA_NAME
                 ,ROW_NUMBER() over (partition by SUPPLIER_OR_CREDITOR_ID order by PRODUCT_AREA_ID desc )  rk
            FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D
            GROUP BY 
                  SUPPLIER_OR_CREDITOR_ID
                 ,PRODUCT_AREA_ID
                 ,PRODUCT_AREA_CODE
                 ,PRODUCT_AREA_NAME
            )q1
         where rk = 1
		  ) D
    ON A.SUPPLIER_OR_CREDITOR_ID = D.SUPPLIER_OR_CREDITOR_ID
  LEFT JOIN PRICE1 E --移动平均价
    ON A.MATERIAL_NUM = E.MATNR
   AND A.FACTORY_WERKS_CODE = E.BWKEY
   AND COALESCE(A.BATCH_NUM,'') = COALESCE(E.BWTAR,'')
   AND E.RN = 1
  LEFT JOIN PRICE2 F --移动平均价
    ON A.MATERIAL_NUM = F.MATNR
   AND A.FACTORY_WERKS_CODE = F.BWKEY
   --AND COALESCE(A.BATCH_NUM,'') = COALESCE(E.BWTAR,'')
   AND F.RN = 1   
  LEFT JOIN SUPPLIER_OR_CREDITOR_DATA SOC
  ON A.MATERIAL_NUM = SOC.MATERIAL_NUM
  AND A.BATCH_NUM = SOC.BATCH_NUM
  AND SOC.RN = 1
  WHERE (substr(INV_LOCATION,1,2) = 'BE' OR INV_LOCATION = 'BJ01')
  --年月日的判定
  AND REPLACE(SUBSTR(A.VOUCHER_POST_DATE,1,10),'-','')  <=  REPLACE(SUBSTR(date_format('${PERIOD_ID_D}','yyyy-MM-dd'),1,10),'-','') --统计日
