			 WHEN SRC_OEM_STATUS = 80 THEN '已完成交付'
			 WHEN SRC_OEM_STATUS = 90 THEN '特批退回'
			 WHEN SRC_OEM_STATUS = 95 THEN '超期审批中'
             WHEN SRC_OEM_STATUS = 110 THEN '已关闭'
			  END SRC_OEM_STATUS_NAME, --源采购计划状态名称
		CLOSE_BILL_REASON, --关单原因
		CLOSE_BILL_REASON_NAME, --关单原因名称
		CASE WHEN OEM_STATUS = 10 THEN '未处理'
		     WHEN OEM_STATUS = 20 THEN '审批中'

			 WHEN OEM_STATUS = 80 THEN '已完成交付'

                     WHEN COALESCE(BT.CHARG,'') <> '' THEN BT.CHARG

                CASE WHEN COALESCE(S.INVENTORY_LOCATION,'') <> '' THEN S.INVENTORY_LOCATION
				     ELSE BT.LGORT
					  END AS INVENTORY_LOCATION,
                COALESCE(BT.BUDAT,S.PRODUCTION_DATE) PRODUCTION_DATE,
                CASE WHEN BT.BANFN IS NOT NULL THEN BT.MENGE
    				 ELSE S.QTY
					  END AS QTY,
                T.*
            from (
                    select
                        T0.ID,
                        T0.CLOSE_FLAG,
                        --T0.CLOSE_QTY,
                        T0.END_TIME,
                        T0.FINISHED_AMOUNT,
                        T0.FINISHED_AREA,
                        T0.IS_CANCEL,
                        T0.ITEM_CODE,
                        --T0.ITEM_NAME,
                        T0.ITEM_SPECIFICATION_MODEL,
                        T0.ITEM_AREA,
                        (
                            CASE
                                WHEN T0.ITEM_PRODUCT_PROPERTY IS NULL
                                OR T0.ITEM_PRODUCT_PROPERTY IN ('FGB', 'TGB', 'YFB')
                                OR T0.ITEM_CODE LIKE 'Z%'
                                OR T0.ITEM_CODE LIKE 'W%'
                                OR T0.ITEM_CODE LIKE 'A%' THEN 'A'
                                ELSE T0.ITEM_PRODUCT_PROPERTY
                            END
                        ) ITEM_PRODUCT_PROPERTY --物料组1
,
                        --T0.LAST_UPDATE_DATE,
                        T0.OEM_PURCHASE_ORDER_NO,
                        T0.BASE_DELIVERY_DATE,
                        T0.Reply_Delivery,
                        --T0.IS_BY_HAND,
                        T0.SELF_FOREIGN_FLAG,
                        COALESCE(T0.NAME, '0') NAME --工单类型原名称
,
                        T1.PLAN_AMOUNT PLAN_AMOUNT,
                        T0.PLAN_AREA,
                        T0.PLAN_NO,
                        T0.PLAN_TASK_DETAIL_ID,
                        T0.PLAN_TASK_ID,
                        T0.PROCESS_TECH_ID,
                        --T2.CODE PROCESS_TECH_ID_DISPLAY,
                        --T2.NAME PROCESS_TECH_NAME,
                        T0.PROCESS_TECH_TASK_ID,
                        --T3.PROTECH_NUM PROCESS_TECH_TASK_ID_DISPLAY,
                        --T0.PRODUCT_AREA_CODE,
                        --T0.PRODUCT_AREA_ID,
                        --T0.PRODUCT_AREA_NAME,
                        T0.PRODUCT_ORDER_CODE,
                        --T0.PRODUCT_ORDER_ID,
                        T0.PROMISE_DELIVERY,
                        --T0.RE_DATE,
                        T0.RELEASE_START_TIME,
                        T0.REQUIRE_DELIVERY,
                        --T0.RESOURCE_CODE,
                        --T0.RESOURCE_ID,
                        T0.RESOURCE_NAME,
                        --T0.RE_STATE,
                        --T1.SAM SAM,
                        T0.START_TIME,
						T0.RELEVANT_CODE,
                        --T0.IS_CANCELED --,T0.OEM_STATUS
                        --,T0.STATUS
--,                     
                        (
                            CASE
                                WHEN T0.RESOURCE_NAME LIKE '%OEM%' THEN substr(T0.RESOURCE_NAME, 1, 5)
                                ELSE (
                                    CASE
                                        WHEN T0.RESOURCE_NAME LIKE '%山东%' THEN '山东'
                                        WHEN T0.RESOURCE_NAME LIKE '东华盛昌釉线%' THEN '华盛昌'
                                        WHEN T0.RESOURCE_NAME LIKE '东华盛昌介砖加工线' THEN '华盛昌加工'
                                        WHEN T0.RESOURCE_NAME LIKE '丰城介砖加工线' THEN '丰城加工'
                                        WHEN T0.RESOURCE_NAME LIKE '丰城釉线%'
                                        OR T0.RESOURCE_NAME LIKE '丰城十一车间釉线%' THEN '丰城'
                                        WHEN T0.RESOURCE_NAME LIKE '永川介砖加工线' THEN '永川加工'
                                        WHEN T0.RESOURCE_NAME LIKE '永川釉线%' THEN '永川'
                                        WHEN T0.RESOURCE_NAME LIKE '%山西%' THEN '山西'
                                        WHEN T0.RESOURCE_NAME LIKE '湖南介砖加工线' THEN '湖南加工'
                                        WHEN T0.RESOURCE_NAME LIKE '湖南釉线%' THEN '湖南'
                                        WHEN T0.RESOURCE_NAME LIKE '%釉线工序' THEN '清远'
                                        WHEN T0.RESOURCE_NAME LIKE '清远介砖加工线' THEN '清远加工'
                                    END
                                )
                            END
                        ) base --基地
,
                        (
                            CASE
                                T0.STATUS
                                WHEN 10 THEN '未完成'
                                WHEN 20 THEN '已下达'
                                WHEN 30 THEN '已完成'
                            END
                        ) STATUS --计划状态
,
                            CASE
							--2023年2月1号开始，基地原来计划结束时间+7天作为准交，现在是变成计划结束时间+5天,20250303调整为+3天

                                WHEN T0.SELF_FOREIGN_FLAG = 0 THEN date_add(T0.END_TIME,3)
                                ELSE date_add(T0.END_TIME,4)
                            END AS ZJHS_TIME --准交核算时间
,
                        (
                            CASE
                                WHEN T0.RESOURCE_NAME LIKE '%OEM%' THEN (
                                   --外协单仅保留：部分交付、已确认并转单、已完成交付、已关闭(其中"关单原因“为供应商原因或在【生产日进度报表】中已完成数量大于0的)
                                    CASE
                                        WHEN COALESCE(T0.OEM_STATUS,10) IN (10,20,30,40,50) THEN 0
										--剔除[资源名称]含OEM的工单行中“关单原因”为“客户主动取消订单”、 “计划部主动取消安全库存备货”、“客户业务要求变更产区工厂”、“供应商不接单”、“供应商不可抗原因停产，更改供应商”的工单行
										WHEN T0.RESOURCE_NAME LIKE '%OEM%' AND T0.IS_OEM_DEL = 1 THEN 0 --OEM工单是否剔除
										WHEN COALESCE(T0.CLOSE_BILL_REASON,1) = 0 AND COALESCE(T0.OEM_STATUS,10) = 110 THEN 110
										WHEN COALESCE(T0.FINISHED_AMOUNT,0) > 0 AND COALESCE(T0.OEM_STATUS,10) = 110 THEN 110
										WHEN COALESCE(T0.OEM_STATUS,10) = 110 THEN 0
                                        ELSE T0.OEM_STATUS
                                    END
                                )
                                ELSE 1010
                            END
                        ) OEM_STATUS,
				T0.SUPPLIER_OR_CREDITOR_ACCT,
				T0.SUPPLIER_OR_CREDITOR_NAME,
				T0.CLOSE_TIME,
				T0.CATEGORY,
				T0.OEM_STATUS AS SRC_OEM_STATUS, --源采购计划状态编码
				T0.CLOSE_BILL_REASON, --关单原因
				T0.CLOSE_BILL_REASON_NAME --关单原因名称
                    from
                        (
                            SELECT
                                PLANTASKDETAIL.ID ID,
                                PLANTASKDETAIL.ID PLAN_TASK_DETAIL_ID,
                                TECHORDER.ID PROCESS_TECH_TASK_ID,
                                TASK.ID PLAN_TASK_ID,
								PRODUCTORDER.RELEVANT_CODE,
                                PRODUCTORDER.ID PRODUCT_ORDER_ID,
                                PRODUCTORDER.ORDER_NUM PRODUCT_ORDER_CODE,
                                PRODUCTORDER.REQUIRE_DELIVERY PROMISE_DELIVERY,
                                PRODUCTORDER.REQUIRE_DELIVERY REQUIRE_DELIVERY,
                                PLANTASKDETAIL.WORK_ORDER OEM_PURCHASE_ORDER_NO,
                                PROCESSTECH.ID PROCESS_TECH_ID,
                                ITEM.ID ITEM_ID,
                                ITEM.CODE ITEM_CODE,
                                --ITEM.NAME ITEM_NAME,
                                ITEM.AREA ITEM_AREA,
                                ITEM.SPECIFICATION_MODEL ITEM_SPECIFICATION_MODEL,
                                ITEM.PRODUCT_PROPERTY ITEM_PRODUCT_PROPERTY,
                                WIPRES.ID RESOURCE_ID,
                                --WIPRES.CODE RESOURCE_CODE,
                                WIPRES.NAME RESOURCE_NAME,
                                --T0.ID PRODUCT_AREA_ID,
                                --T0.IS_BY_HAND,
                                --T0.CODE PRODUCT_AREA_CODE,
                                --T0.NAME PRODUCT_AREA_NAME,
                                FINISHEDVM.FINISHED_AMOUNT FINISHED_AMOUNT,
                                --PLANTASKDETAIL.UPDATE_DATE LAST_UPDATE_DATE,
                                (PLANTASKDETAIL.PLAN_AMOUNT * ITEM.AREA) PLAN_AREA,
                                (FINISHEDVM.FINISHED_AMOUNT * ITEM.AREA) FINISHED_AREA,
                                TASK.PLAN_NO PLAN_NO,
                                TASK.START_TIME START_TIME,
                                TASK.END_TIME END_TIME,
                                TASK.STATUS STATUS,
                                TASK.IS_CANCEL IS_CANCEL,
                                --TASK.RE_STATE RE_STATE,
                                --TASK.RE_DATE RE_DATE,
                                PLANTASKDETAIL.RELEASE_START_TIME RELEASE_START_TIME,
                                TASK.CLOSE_FLAG CLOSE_FLAG,
                                --TASK.CLOSE_QTY CLOSE_QTY,
                                DELIVERYDATE.BASE_DELIVERY_DATE,
                                oem.Reply_Delivery,
                                OEM.STATUS OEM_STATUS,
                                CASE WHEN WIPRES.NAME LIKE '%OEM%' THEN 1
								     ELSE 0
									  END AS SELF_FOREIGN_FLAG,
                                GD_TYPE.NAME,
                                --PRODUCTORDER.IS_CANCELED,
								TT.CODE AS SUPPLIER_OR_CREDITOR_ACCT, --供应商编码
								TT.NAME AS SUPPLIER_OR_CREDITOR_NAME, --供应商名称
								TASK.CLOSE_TIME,
								ITEM.CATEGORY,
								OEM.CLOSE_BILL_REASON,--关单原因
								OEM.CLOSE_BILL_REASON_TXT AS CLOSE_BILL_REASON_NAME, --关单原因名称
								CASE WHEN CBR.CLOSE_BILL_REASON_DESC IS NOT NULL THEN 1 ELSE 0 END IS_OEM_DEL --OEM工单是否剔除
                            FROM
                                DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F PLANTASKDETAIL --Y
                                LEFT OUTER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_F TASK ON PLANTASKDETAIL.id = TASK.PLAN_TASK_DTL_ID
                                LEFT OUTER JOIN DWROTD.DWR_PS_RES_WIP_SCHE_F WIPRES --Y
                                ON TASK.RESOURCE_ID = WIPRES.ID
                                LEFT OUTER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F TECHORDER --Y
                                ON PLANTASKDETAIL.PROCESS_TECH_TASK_ID = TECHORDER.ID
                                LEFT OUTER JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F PRODUCTORDER --Y
                                ON TECHORDER.PRODUCT_ORDER_ID = PRODUCTORDER.ID
                                LEFT OUTER JOIN SDI.sdi_ITEM_1007 ITEM ON PRODUCTORDER.SRC_MATERIAL_ID = ITEM.ID
                                and ITEM.del_flag = 'N'
                                LEFT OUTER JOIN SDI.sdi_aps_protech_1007 PROCESSTECH ON TECHORDER.PROCESS_TECH_ID = PROCESSTECH.ID
                                and PROCESSTECH.del_flag = 'N'
                                LEFT OUTER join SDI.SDI_DELIVERYDATE_CHANGE_1007 DELIVERYDATE ON DELIVERYDATE.PLAN_NO = TASK.PLAN_NO
                                AND DELIVERYDATE.del_flag = 'N'
                                LEFT OUTER join SDI.sdi_APS_OEM_PO_SUM_1007 OEM ON OEM.APS_PLAN_TASK_DTL_ID = PLANTASKDETAIL.ID
                                and OEM.del_flag = 'N'
								LEFT JOIN SDI.SDI_OEM_CLOSE_BILL_REASON_1037 CBR 
								  ON OEM.CLOSE_BILL_REASON_TXT = CBR.CLOSE_BILL_REASON_DESC
                                left join sdi.SDI_SUPPLIER_1007 tt
								on oem.SUPPLIER_ID = tt.ID
								and tt.del_flag = 'N'
								LEFT OUTER JOIN DWROTD.DWR_PS_ORDER_TYPE_SETTING_F GD_TYPE --Y
                                ON GD_TYPE.ID = PRODUCTORDER.ORDER_TYPE_SET_ID
                                LEFT OUTER JOIN (
                                    SELECT
                                        DAILYSCHEDULE.PLAN_TASK_ID PLANTASKID,
                                        DAILYSCHEDULE.PLAN_TASK_ID PLAN_TASK_ID,
                                        DAILYSCHEDULE.PROCESS_TECH_ORDER_ID PROCESS_TECH_TASK_ID,
                                        SUM(DAILYSCHEDULE.QTY) FINISHED_AMOUNT
                                    FROM
                                        DWROTD.DWR_PS_DAILY_SCHEDULE_F DAILYSCHEDULE --Y
                                    GROUP BY
                                        DAILYSCHEDULE.PLAN_TASK_ID,
                                        DAILYSCHEDULE.PROCESS_TECH_ORDER_ID
                                ) FINISHEDVM ON PLANTASKDETAIL.PLAN_TASK_ID = FINISHEDVM.PLAN_TASK_ID
                                AND PLANTASKDETAIL.PROCESS_TECH_TASK_ID = FINISHEDVM.PROCESS_TECH_TASK_ID
                                LEFT OUTER JOIN DWROTD.DWR_PS_RES_ENTERPRISE_F T0 --Y
                                ON WIPRES.SRC_BELONG_FACT_ID = T0.ID
                        ) T0
                        LEFT OUTER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F T1 --Y
                        ON T0.PLAN_TASK_DETAIL_ID = T1.ID
                        LEFT OUTER JOIN SDI.sdi_aps_protech_1007 T2 ON T0.PROCESS_TECH_ID = T2.ID
                        LEFT OUTER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F T3 ON T0.PROCESS_TECH_TASK_ID = T3.ID
                ) T
				LEFT JOIN DWROTD.DWR_PS_DAILY_SCHEDULE_F S
				ON T.Plan_Task_Id = S.PLAN_TASK_ID
                AND S.PROCESS_TECH_ORDER_ID = T.Process_Tech_Task_ID
				LEFT JOIN (
			SELECT AN.BANFN,KO.LIFNR,PO.WERKS ,PO.LGORT 
,BE.CHARG,sum(CASE WHEN BE.SHKZG = 'S' 
THEN BE.MENGE ELSE 0 - BE.MENGE END) AS MENGE ,concat(substr(BE.BUDAT,1,4),'-',substr(BE.BUDAT,5,2),'-',substr(BE.BUDAT,7,2)) as BUDAT
FROM SDI.SDI_EBAN_1001 AN  
left JOIN SDI.SDI_EKPO_1001 AS PO ON PO.BANFN = AN.BANFN 
AND PO.BNFPO = AN.BNFPO 
INNER JOIN SDI.SDI_EKKO_1001 AS KO ON KO.EBELN = PO.EBELN  
LEFT JOIN SDI.SDI_EKBE_1001 AS BE ON PO.EBELN = BE.EBELN 
AND PO.EBELP = BE.EBELP  WHERE AN.BNFPO = '00010'
AND AN.BSART = 'ZNB1' AND BE.BEWTP = 'E' 
GROUP BY AN.BANFN,KO.LIFNR,PO.WERKS ,PO.LGORT 
,BE.CHARG,concat(substr(BE.BUDAT,1,4),'-',substr(BE.BUDAT,5,2),'-',substr(BE.BUDAT,7,2))
				) BT 
		ON T.OEM_PURCHASE_ORDER_NO = BT.BANFN
		--AND T.SUPPLIER_OR_CREDITOR_ACCT = BT.LIFNR
		--AND SUBSTR(REPLACE(S.PRODUCTION_DATE,'-',''),1,8) = BT.BUDAT
        ) BASIC_DATA
    WHERE
        STATUS IN ('已完成','已下达')
        AND IS_CANCEL = 0
),
B AS (
    SELECT
        A.*,
        (
            CASE
                WHEN SUBSTR(ZJHS_TIME,1,19) >= SUBSTR(COALESCE(CLOSE_TIME,''),1,19)
                 AND CLOSE_FLAG = 1 THEN '是'   --关单标识：CLOSE_FLAG = 1 
                ELSE '否'
            END
        ) clearance_within_time1 --是否核算时间内关单1
,
        (
            CASE
                WHEN SUBSTR(REQUIRE_DELIVERY,1,19) >= SUBSTR(COALESCE(CLOSE_TIME,''),1,19) 
                 AND CLOSE_FLAG = 1 THEN '是'   --关单标识：CLOSE_FLAG = 1
                ELSE '否'
            END
        ) clearance_within_time2 --是否核算时间内关单2
,
        (
            CASE
                WHEN SUBSTR(ZJHS_TIME,1,10) >= SUBSTR(PRODUCTION_DATE,1,10) THEN (
                CASE WHEN BATCH NOT LIKE '%7'
                      AND BATCH NOT LIKE '%9' THEN QTY
                     ELSE 0
                      END
                )
                ELSE 0
            END
        ) ontime_quality1 --按时按质报工数量1
,
        (
            CASE
                WHEN SUBSTR(REQUIRE_DELIVERY,1,10) >= SUBSTR(PRODUCTION_DATE,1,10) THEN (
                    CASE WHEN BATCH NOT LIKE '%7'
                          AND BATCH NOT LIKE '%9' THEN QTY
                         ELSE 0
                          END
                )
                ELSE 0
            END
        ) ontime_quality2 --按时按质报工数量2
,
        (
            CASE
                WHEN SUBSTR(ZJHS_TIME,1,10) >= SUBSTR(PRODUCTION_DATE,1,10) THEN (
                    CASE WHEN BATCH NOT LIKE '%7'
                          AND BATCH NOT LIKE '%9' THEN JT_AREA
                         ELSE 0
                          END
                )
                ELSE 0
            END
        ) ontime_AREA1 --按时按质报工面积1
,
        (
            CASE
                WHEN SUBSTR(REQUIRE_DELIVERY,1,10) >= SUBSTR(PRODUCTION_DATE,1,10) THEN
                    CASE WHEN BATCH NOT LIKE '%7'
                          AND BATCH NOT LIKE '%9' THEN JT_AREA
                         ELSE 0
                          END
                ELSE 0
            END
        ) ontime_AREA2 --按时按质报工面积2
    FROM
        A AS A
    WHERE
        OEM_STATUS <> 0
        --AND RELEASE_START_TIME IS NOT NULL
),
qcstk AS 
(
-- 质检封仓的批次在多个报工单的时候只取数量最大的工单
select B.id, B.ITEM_CODE, B.BATCH,stk.qc_stock_qty ,
       row_number() over(partition by B.ITEM_CODE, B.BATCH order by B.PLAN_AMOUNT desc  ) rn 
from B AS B JOIN 
(
select material_num,batch_num,sum(coalesce(qc_inventory_quantity,0)+coalesce(projectqc_inventory_quantity,0)) as qc_stock_qty from dm.dm_wm_all_type_stock_t 
     group by material_num,batch_num having sum(coalesce(qc_inventory_quantity,0)+coalesce(projectqc_inventory_quantity,0)) >0
     )  stk ON B.item_code =stk.material_num and B.BATCH=stk.batch_num
),
C AS (
    SELECT
        B1.*,
        B2.ontime_quality1_1,
        B2.ontime_AREA1_1,
        ontime_quality2_2,
        ontime_AREA2_2,
        ontime_quality1_1_mid,
        ontime_quality2_2_mid,
        ontime_area1_1_mid,
        ontime_area2_2_mid,
        qc_stock_quality,
        qc_stock_area
    FROM
        B B1
        LEFT OUTER JOIN (
            select tmp.ID,
                  SUM(ontime_quality1_1) AS ontime_quality1_1,
                  SUM(ontime_AREA1_1) ontime_AREA1_1,
                  SUM(ontime_quality2_2) ontime_quality2_2,
                  SUM(ontime_AREA2_2) ontime_AREA2_2,
                  sum(greatest(ontime_quality1_1-qc_stock_qty,0)) as ontime_quality1_1_mid,  -- 按批次考虑质检冻结后的按时按质数量
                  sum(greatest(ontime_quality2_2-qc_stock_qty,0)) as ontime_quality2_2_mid,   -- 按批次考虑质检冻结后的按时按质数量
                  sum(greatest(ontime_quality1_1-qc_stock_qty,0)* ITEM_AREA) as ontime_area1_1_mid,  -- 按批次考虑质检冻结后的按时按质面积
                  sum(greatest(ontime_quality2_2-qc_stock_qty,0)* ITEM_AREA) as ontime_area2_2_mid,   -- 按批次考虑质检冻结后的按时按质面积
                  sum(qc_stock_qty) as qc_stock_quality,
                  sum(qc_stock_qty* ITEM_AREA) as qc_stock_area
            from (
            SELECT
                B.ID,
                B.item_code,
                B.BATCH,
                max(PLAN_AMOUNT) PLAN_AMOUNT,max(PLAN_AREA) PLAN_AREA,max(ITEM_AREA) ITEM_AREA,
                sum(ontime_quality1) ontime_quality1_1,
                sum(ontime_AREA1) ontime_AREA1_1,
                sum(ontime_quality2) ontime_quality2_2,
                sum(ontime_AREA2) ontime_AREA2_2,
                coalesce(max(qc_stock_qty),0) as qc_stock_qty
            FROM
                B AS B   
                LEFT JOIN 
                 qcstk AS stk ON B.id =stk.id and B.item_code =stk.item_code and B.BATCH=stk.BATCH AND stk.rn=1
            GROUP BY
                B.ID,
                B.item_code,
                B.BATCH
            ) tmp 
            GROUP BY tmp.ID
        ) B2 ON B1.ID = B2.ID
),--按时按质报工数量1_1 and 按时按质报工面积1_1 and 按时按质报工数量2_2 and 按时按质报工面积2_2
D AS (
    SELECT
        C.*,
        (
            CASE
                WHEN PLAN_AMOUNT <= ONTIME_QUALITY1_1_mid THEN PLAN_AMOUNT
                ELSE ONTIME_QUALITY1_1_mid
            END
        ) Submitted_quantity1 --准交数量1
,
        (
            CASE
                WHEN PLAN_AMOUNT <= ONTIME_QUALITY2_2_mid THEN PLAN_AMOUNT
                ELSE ONTIME_QUALITY2_2_mid
            END
        ) Submitted_quantity2 --准交数量2
,
        (
            CASE
                WHEN PLAN_AREA <= ONTIME_AREA1_1_mid THEN PLAN_AREA
                ELSE ONTIME_AREA1_1_mid
            END
        ) Submitted_AREA1 --准交面积1
,
        (
            CASE
                WHEN PLAN_AREA <= ONTIME_AREA2_2_mid THEN PLAN_AREA
                ELSE ONTIME_AREA2_2_mid
            END
        ) Submitted_AREA2 --准交面积2
    FROM
        C
),
E AS (
    SELECT
        D.*,
        ONTIME_AREA1_1_mid / PLAN_AREA Submitted_AREA_RATE1 --面积准交率1
,
        ONTIME_AREA2_2_mid / PLAN_AREA Submitted_AREA_RATE2 --面积准交率2
,
        (
            CASE
                WHEN ITEM_PRODUCT_PROPERTY IN ('A', 'B') THEN (
                    CASE
                        WHEN ONTIME_QUALITY1_1_mid / PLAN_AMOUNT >= 0.9 THEN '是'
                        ELSE '否'
                    END
                )
                WHEN ITEM_PRODUCT_PROPERTY IN ('C') THEN (
                    CASE
                        WHEN ONTIME_QUALITY1_1_mid / PLAN_AMOUNT >= 0.995 THEN '是'
                        ELSE '否'
                    END
                )
				WHEN ITEM_PRODUCT_PROPERTY IN ('D') THEN (
                    CASE
                        WHEN ONTIME_QUALITY1_1_mid / PLAN_AMOUNT >= 1 THEN '是'
                        ELSE '否'
                    END
                )
            END
        ) SKU_delivered1 --SKU是否准交1
,
        (
            CASE
                WHEN ITEM_PRODUCT_PROPERTY IN ('A', 'B') THEN (
                    CASE
                        WHEN ONTIME_QUALITY2_2_mid / PLAN_AMOUNT >= 0.9 THEN '是'
                        ELSE '否'
                    END
                )
                WHEN ITEM_PRODUCT_PROPERTY IN ('C') THEN (
                    CASE
                        WHEN ONTIME_QUALITY2_2_mid / PLAN_AMOUNT >= 0.995 THEN '是'
                        ELSE '否'
                    END
                )
				WHEN ITEM_PRODUCT_PROPERTY IN ('D') THEN (
                    CASE
                        WHEN ONTIME_QUALITY2_2_mid / PLAN_AMOUNT >= 1 THEN '是'
                        ELSE '否'
                    END
                )
            END
        ) SKU_delivered2 --SKU是否准交2
    FROM
        D
),
F as(
    SELECT
        E1.*,	
        CASE WHEN E2.RELEVANT_CODE IS NOT NULL OR E3.RELEVANT_CODE IS NOT NULL THEN '纯安全库存'
             ELSE '非纯安全库存'
			  END GD_TYPE
    FROM E E1
	LEFT JOIN(
            SELECT PRODUCTORDER.RELEVANT_CODE
			      ,COUNT(DISTINCT GD_TYPE.NAME) AS COUNT1
				  ,COUNT(DISTINCT CASE WHEN COALESCE(GD_TYPE.NAME,'') = '安全库存' THEN GD_TYPE.NAME 
				                       ELSE NULL
									    END) AS COUNT2
              FROM DWROTD.DWR_PS_APS_PRO_ORDER_F PRODUCTORDER
			  LEFT JOIN DWROTD.DWR_PS_ORDER_TYPE_SETTING_F GD_TYPE --Y
                ON GD_TYPE.ID = PRODUCTORDER.ORDER_TYPE_SET_ID
             WHERE PRODUCTORDER.RELEVANT_CODE <> PRODUCTORDER.ORDER_NUM
			 GROUP BY PRODUCTORDER.RELEVANT_CODE
        ) E2
	  ON E1.RELEVANT_CODE = E2.RELEVANT_CODE
	 AND E2.COUNT1 = E2.COUNT2
	LEFT JOIN(
            SELECT PRODUCTORDER.RELEVANT_CODE
			      ,COUNT(DISTINCT GD_TYPE.NAME) AS COUNT1
				  ,COUNT(DISTINCT CASE WHEN COALESCE(GD_TYPE.NAME,'') = '安全库存' THEN GD_TYPE.NAME 
				                       ELSE NULL
									    END) AS COUNT2
              FROM DWROTD.DWR_PS_APS_PRO_ORDER_F PRODUCTORDER
			  LEFT JOIN DWROTD.DWR_PS_ORDER_TYPE_SETTING_F GD_TYPE --Y
                ON GD_TYPE.ID = PRODUCTORDER.ORDER_TYPE_SET_ID         
			 GROUP BY PRODUCTORDER.RELEVANT_CODE
        ) E3
	  ON E1.RELEVANT_CODE = E3.RELEVANT_CODE
	 AND E3.COUNT1 = E3.COUNT2	 

),
--工单类型
 G AS (SELECT A1.* FROM F A1
