WITH T AS(
SELECT ORDER_ID
      ,COUNT(ORDER_ITEM_ID) AS ORDER_ITEM_CNT
  FROM DWROTD.DWR_SO_MOS_ORDER_LINE_INFO_F
 GROUP BY ORDER_ID
)
INSERT OVERWRITE TABLE DM.DM_MOS_CUST_ORDER_ITEM_INFO_T
SELECT A.CUST_NAME --客户名称
      ,A.CUST_PHONE_NUM --客户电话
      ,A.CUST_GNDR --性别
      ,A.AGE --年龄层
      ,A.PROV_NAME AS CUST_PROV_NAME --客户地址-省
      ,A.CITY_NAME AS CUST_CITY_NAME --客户地址-市
      ,A.DIST_NAME AS CUST_DIST_NAME --客户地址-区
      ,A.ADDR AS CUST_ADDR --客户详细地址
      ,A.DMT_SRC_NAME --客户来源
      ,E.CREATE_TIME AS FST_CREATE_TIME --客户首次创建日期
      ,A.ORDER_ST_NAME AS CUST_ST_NAME --客户状态
      ,A.ACC_EMP_NUM --导购编码
      ,A.ACC_EMP_NAME --导购名称
      ,COALESCE(B.OA_STORE_CODE,B.STORE_CODE) AS STORE_CODE --门店编码
      ,B.STORE_NAME --门店名称
      ,B.ADDR AS STORE_ADDR --门店详细地址
      ,B.PROV_NAME AS STORE_PROV_NAME --门店地址-省
      ,B.CITY_NAME AS STORE_CITY_NAME --门店地址-市
      ,B.DIST_NAME AS STORE_DIST_NAME --门店地址-区
      ,B.STORE_AREA --门店面积
      ,COALESCE(B.CUST_NUM,A.DEALER_NUM) AS DEALER_NUM  --经销商编码
      ,B.DEALER_NAME --经销商名称
      ,B.IS_TC --商是否参与同城
      ,B.TC_TIME --参与同城日期
      ,A.ORDER_NUM --订单编码
      ,A.ORDER_AMT --订单金额
      ,A.ORDER_ST_NAME --订单状态
      ,C.MATERIAL_NUM --物料编码
      ,C.MATERIAL_NAME --物料名称
      ,D.RESERVED_FIELD_1 AS MATERIAL_CNNL_CODE --产品渠道
      ,D.RESERVED_FIELD1_NAME AS MATERIAL_CNNL_NAME --产品渠道名称
      ,D.MENU_DESC_CODE --菜单说明
      ,D.MENU_DESC_NAME --菜单说明名称
      ,D.CATEGORY AS MATERIAL_CAT_CODE --品类
      ,D.CATEGORY_NAME AS MATERIAL_CAT_NAME --品类名称
      ,D.PRODUCT_BRAND_CODE AS MATERIAL_SER_CODE --产品系列
      ,D.PRODUCT_BRAND_NAME AS MATERIAL_SER_NAME --产品系列名称
      ,D.DIMENSION --大小量纲
      ,C.BUY_QTY --订单数量
      ,C.BUY_QTY * D.UNIT_AREA AS ORDER_AREA --订单面积
      ,A.NATL_SBSDZ_NAME --是否参与国补
      ,A.LITL_DEP_AMT --交定/小定金额
      ,A.BIG_DEP_AMT --大定金额
      ,A.PAY_AMT --合同金额
	  ,B.MKT_NAME --运营中心
	  ,B.SALE_GRP_NAME --销售组
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.CUST_ID AS CUST_ID --客户ID
	  ,A.CUST_NUM --客户编码
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(A.FIN_TIME,A.LOSE_TIME),A.BIG_DEP_TIME),A.LITL_DEP_TIME),A.DEP_TIME),A.RFND_DOWNPAY_TIME),A.CREATE_TIME) AS CHNG_TIME --状态变更日期
	  ,A.ORDER_AMT / COALESCE(T.ORDER_ITEM_CNT) AS LINE_ORDER_AMT --行订单金额
  FROM DWROTD.DWR_SO_MOS_ORDER_INFO_F A
  LEFT JOIN DWRDIM.DWR_DIM_MOS_STORE_INFO_T B
    ON A.STORE_ID = B.ID
  LEFT JOIN DWROTD.DWR_SO_MOS_ORDER_LINE_INFO_F C
    ON A.ORDER_ID = C.ORDER_ID
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T D
    ON C.MATERIAL_NUM = D.MATERIAL_NUM
  LEFT JOIN DWRDIM.DWR_DIM_MOS_CUST_INFO_D E
    ON A.CUST_ID = E.CUST_ID
  LEFT JOIN T
    ON A.ORDER_ID = T.ORDER_ID
;
