insert overwrite table dm.dm_md_wbs_sale_grp_d
select distinct a.poski as wbs_num,b.zvkgrp as sale_grp,current_timestamp as dw_last_update_date
from sdi.sdi_PRPS_1001 a
inner join sdi.sdi_proj_1001 b
on a.mandt = b.mandt
and a.psphi = b.pspnr
where a.mandt = 800
