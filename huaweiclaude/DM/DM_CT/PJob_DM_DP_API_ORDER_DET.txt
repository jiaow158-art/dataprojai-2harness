with tem as(
select --未交付订单明细
	from_unixtime(unix_timestamp(t1.order_date, 'yyyyMMdd'), 'yyyy-MM-dd') as order_date,
	t1.cust_num,
	t1.cust_name,
	t2.sales_center_code,
	t2.sales_center,
	t1.zh_channel_code4 as integrate_channel,
	t2.sales_dep_code,	
	t2.sales_dep,
	t2.sales_region_code,	
	t2.sales_region,	
	t1.order_no as sap_order_num,
	t1.order_item_no as sap_order_item_num,
	t1.wbs_element as wbs_element,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t4.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,	
	t1.dimension as dimension,
	t1.material_num,
	t1.material_name,	
	t1.nodeliver_qty_aps as qty,
	t1.nodeliver_area_aps as area,
	concat(substring(t1.expect_date,1,4),'-',substring(t1.expect_date,5,2),'-',substring(t1.expect_date,7,2)) as expect_date,--经销商要求交期
	t1.creator_id as creator_id,--申请人ID
	t1.creator_name as creator_name,--申请人名字
	t1.salesman_num_list as salesman_id,--业务员id
	t1.salesman_name_list as salesman_name,--业务员名字
	'未交付订单' order_stat
from  DM.DM_RPT_SALES_DELIVER_T t1
join dm.dm_dp_api_sales_group_struct t2 on t1.vkgrp = t2.sales_grp
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL   
where 1=1
and t1.stat_month = replace(substring(current_timestamp(),1,7),'-','')
and t1.nodeliver_area_aps > 0

union ALL 

select --留备货：全部
	from_unixtime(unix_timestamp(t1.create_time, 'yyyyMMdd'), 'yyyy-MM-dd') as order_date,
	t1.cust_num,
	t1.cust_name,
	t2.sales_center_code,
	t2.sales_center,
	t1.zh_channel_code4 as integrate_channel,
	t2.sales_dep_code,	
	t2.sales_dep,
	t2.sales_region_code,	
	t2.sales_region,	
	t1.sal_vch_num as sap_order_num,
	t1.sal_vch_item_num as sap_order_item_num,
	t1.wbs_element as wbs_element,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t4.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,	
	t1.dimension as dimension,
	t1.material_num,
	t1.material_name,	
	t1.sal_pre_res_qty as qty,
	t1.sal_pre_res_area as area,
	from_unixtime(unix_timestamp(t1.user_delivery_date, 'yyyyMMdd'), 'yyyy-MM-dd') as expect_date,--经销商要求交期
	t1.creator_id as creator_id,--申请人ID
	t1.creator_name as creator_name,--申请人名字
	t1.salesman_num_list as salesman_id,--业务员id
	t1.salesman_name_list as salesman_name,--业务员名字
	case when lb_type = '销售订单备货' then '备货订单' else '留货订单' end as order_stat
from  DM.DM_RPT_SALES_DELIVER_2023_T t1
join dm.dm_dp_api_sales_group_struct t2 on t1.sale_grp = t2.sales_grp
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL   
where 1=1
and lb_type in ('销售订单留货','专属客编留货','销售订单备货')
and stat_month = replace(substring(current_timestamp(),1,7),'-','')

union ALL 

SELECT --留货：剩余项目库存明细
	from_unixtime(unix_timestamp(t1.order_date, 'yyyyMMdd'), 'yyyy-MM-dd') as order_date,
	t1.cust_num,
	t1.cust_name,
	t2.sales_center_code,
	t2.sales_center,
	t1.zh_channel_code4 as integrate_channel,
	t2.sales_dep_code,	
	t2.sales_dep,
	t2.sales_region_code,	
	t2.sales_region,	
	t1.top_wbs_num as sap_order_num,
	t1.wbs_num as sap_order_item_num,
	t1.wbs_element as wbs_element,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖

               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,	
	t4.dimension as dimension,
	t1.material_num,
	t4.material_name,	
	t1.available_stock_qty as qty,
	t1.available_stock_qty*t4.length_1*t4.width/1000000 as area,
	concat(substring(t1.expect_date,1,4),'-',substring(t1.expect_date,5,2),'-',substring(t1.expect_date,7,2)) as expect_date,--经销商要求交期
	t1.creator_id as creator_id,--申请人ID
	t1.creator_name as creator_name,--申请人名字
	t1.salesman_id,--业务员id
	t1.salesman_name,--业务员名字
	'留货订单' order_stat
from dwrotd.dwr_wm_wbs_stock_f t1
join dm.dm_dp_api_sales_group_struct t2 on t1.sale_grp = t2.sales_grp
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL   
where 1=1
and t1.start_month= replace(substring(current_timestamp(),1,7),'-','')
and t1.available_stock_qty > 0
)
