INSERT overwrite INTO dm.dm_sfa_employee_info_t  --8.14初始化
with tmp as (
select 
 A.PERNR as employee_num  --员工编码
,A.NACHN as employee_name   --员工名称
,A.SUPNODE as suprrior  --上级工号 
,A.USRID_CELL as phone_num  --电话号码
,case when A.GESCH = '1' then 'M' when A.GESCH = '2' then 'F' else 'P' end as gender   -- 性别
,case when A.ZZ_OPERA = 'D' then '1' else '0' end as employee_value  --员工状态
,D.employee_value AS sfa_employee_value
,A.PLSTX as position_name --岗位
,B.dept_id as dept_id -- 部门id 
,A.ORGEH as dept_code --部门编码
,C.employee_id as employee_id --sfa员工id
,substr(A.udate,1,4)||'-'||substr(A.udate,5,2)||'-'||substr(A.udate,7,2) as update_date --修改日期
,E.user_id as suprrior_id  --上级id
,CURRENT_TIMESTAMP() as dw_last_update_date
,F.sales_grp
,F.sales_desc 
-- ,G.node_desc2
,case when G.node_name2 = 'H11000001' then '1'
      when G.node_name2 = 'H11000002' then '2' else '' end as node_desc2
,H.sap_email --SAP个人邮箱
,H.ID AS OA_id --OA人员ID
,case when node_desc4 like '%工程市场中心%' then node_name4 else node_name5 end sales_area_code  --销售区域代码
,case when node_desc4 like '%工程市场中心%' then node_desc4 else node_desc5 end sales_area_name  --销售区域名称
,case when node_desc4 like '%工程市场中心%' then node_name5 else node_name6 end sales_dept_code  --销售部门代码
,case when node_desc4 like '%工程市场中心%' then node_desc5 else node_desc6 end sales_dept_name  --销售部门名称
from DM.DM_RPT_ZHRT_OA_PERNR_T as A 
left join api_log.dws_sfa_org_dept_info_t B  
on A.ORGEH = case when length(B.dept_code) < 8 then LPAD(B.dept_code,8,'0') else B.dept_code end and B.del_flag = 0 
left join api_log.dws_sfa_org_employee_info_t C 
on A.PERNR = C.employee_number 
left join api_log.dws_sfa_org_employee_info_t D 
on A.SUPNODE = D.employee_number
left join api_log.api_to_dws_people_information E
on D.employee_id = E."_id"
left join (select emp_num,sales_grp,sales_desc,row_number() over(partition by emp_num) as rn 
from dwrdim.dwr_dim_org_cost_emp_d ) F  
on A.PERNR = F.emp_num and F.rn = 1
left join dm.dm_rpt_sales_group_t G  
on F.sales_grp = G.node_name9 
left join DWRPI.DWR_WF_HR_PERSON_INTO_F H 
on A.PERNR = H.sapid and H.sapid is not null and H.loginid is not null and H.password IS NOT NULL
left join DWIMD.DWI_ORG_PERNR_STRUCTURE_INFO_T I  --由于主表人员离职架构层级会删除掉，所以重新关联架构树
ON A.PERNR = I.employee_CODE
where I.objid_t4 = '00008706' --瓷砖营销中心
or I.objid_t5 IN ('00010371','00010375','00010379','00010384','00010389','00010394') --卫浴营销中心-华北、华东、华南、华中、西北、西南

)
,update_table as (
select 
 T1.employee_num
,CASE when T2.sfa_employee_value <> T1.employee_value THEN 'Y'
 WHEN concat(T1.employee_id,T1.employee_num,T1.employee_name,T1.suprrior,T1.suprrior_id,T1.phone_num,T1.gender,T1.employee_value,T1.position_name,T1.dept_code,T1.dept_id,T1.sales_grp,T1.sap_email,
                  T1.sales_area_code,T1.sales_dept_code,T1.sales_desc,T1.sales_area_name,T1.sales_dept_name)
         = concat(T2.employee_id,T2.employee_num,T2.employee_name,T2.suprrior,T2.suprrior_id,T2.phone_num,T2.gender,T2.employee_value,T2.position_name,T2.dept_code,T2.dept_id,T2.sales_grp,T2.sap_email,
                  T2.sales_area_code,T2.sales_dept_code,T2.sales_desc,T2.sales_area_name,T2.sales_dept_name)
      THEN 'N' ELSE 'Y' END  AS UPDATE_FLAG
FROM dm.dm_sfa_employee_info_t T1
LEFT JOIN tmp T2 
ON T1.employee_num = T2.employee_num
)
,result_table as (
-- 新增/修改数据
select 
 t1.employee_num  --员工编码
,t1.employee_name   --员工名称
,t1.suprrior  --上级工号 
,t1.phone_num  --电话号码
,t1.gender   -- 性别
,t1.employee_value  --员工状态
,t1.position_name --岗位名称 
,t1.dept_id -- 部门id 
,t1.dept_code --部门编码
,t1.employee_id --sfa员工id
,substr(current_date(),1,10) as update_date --修改日期
,t1.suprrior_id  --上级id
,t2.update_flag
,t1.sales_grp
,t1.sales_desc 
,t1.node_desc2
,t1.sap_email --SAP个人邮箱
,cast(t1.OA_id as text) as OA_id --OA人员ID
,t1.sales_area_code
,t1.sales_area_name
,t1.sales_dept_code
,t1.sales_dept_name
from tmp t1
left join update_table t2
on t1.employee_num = t2.employee_num
where t2.update_flag is null or t2.update_flag = 'Y'
union all --不变的数据
select 
 t1.employee_num  --员工编码
,t1.employee_name   --员工名称
,t1.suprrior  --上级工号 
,t1.phone_num  --电话号码
,t1.gender   -- 性别
,t1.employee_value  --员工状态
,t1.position_name --岗位名称
,t1.dept_id -- 部门id 
,t1.dept_code --部门编码
,t1.employee_id --sfa员工id
,'' as update_date --修改日期
,t1.suprrior_id  --上级id
,t2.update_flag
,t1.sales_grp
,t1.sales_desc 
,t1.node_desc2
,t1.sap_email --SAP个人邮箱
,cast(t1.OA_id as text) as OA_id --OA人员ID
,t1.sales_area_code
,t1.sales_area_name
,t1.sales_dept_code
,t1.sales_dept_name
from dm.dm_sfa_employee_info_t t1
left join update_table t2
on t1.employee_num = t2.employee_num
where t2.update_flag = 'N'
)
select distinct 
 employee_num  --员工编码
,employee_name   --员工名称
,suprrior  --上级工号 
,phone_num  --电话号码
,gender   -- 性别
,employee_value  --员工状态
,position_name --岗位名称
,dept_id -- 部门id 
,dept_code --部门编码
,employee_id --sfa员工id
,update_date --修改日期
,suprrior_id  --上级id
,CURRENT_TIMESTAMP() as dw_last_update_date
,sales_grp
,sales_desc 
,node_desc2
,sap_email --SAP个人邮箱
,OA_id --OA人员ID
,sales_area_code
,sales_area_name
,sales_dept_code
,sales_dept_name
from result_table
