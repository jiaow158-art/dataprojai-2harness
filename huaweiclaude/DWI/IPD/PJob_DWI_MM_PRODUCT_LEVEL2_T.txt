INSERT OVERWRITE TABLE DWIIPD.DWI_MM_PRODUCT_LEVEL2_T    --产品层级2
SELECT 
MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE,
FELDN AS PRODUCT_LEVEL_CODE,
TEXT20 AS PRODUCT_LEVEL_NAME,
MENGE AS QUANTITY,
MEINS AS BASE_MEASURE_UNIT,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_T157T_1001
UNION ALL
SELECT
DWI.GROUP_NUM,
DWI.LANG_CODE,
DWI.PRODUCT_LEVEL_CODE,
DWI.PRODUCT_LEVEL_NAME,
DWI.QUANTITY,
DWI.BASE_MEASURE_UNIT,
'Y' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIIPD.DWI_MM_PRODUCT_LEVEL2_T DWI
LEFT JOIN SDI.SDI_T157T_1001 SDI
ON SDI.MANDT=DWI.GROUP_NUM
AND SDI.SPRAS=DWI.LANG_CODE
AND SDI.FELDN=DWI.PRODUCT_LEVEL_CODE
WHERE SDI.MANDT IS NULL
AND SDI.SPRAS IS NULL
AND SDI.FELDN IS NULL
