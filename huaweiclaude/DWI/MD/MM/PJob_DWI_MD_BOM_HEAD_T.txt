INSERT OVERWRITE TABLE DWIMD.DWI_MD_BOM_HEAD_TMP
 SELECT 
   STKO.STLTY AS MATL_TYPE,
   STKO.STLNR AS MATL_LIST,
   STKO.STKOZ AS INNER_COUNTER,
   STKO.STLAL AS OPTIONAL_BOM,
   STKO.MANDT AS GROUP_NUM,
   FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN STKO.AEDAT  = '00000000'  OR STKO.AEDAT IS NULL THEN STKO.ANDAT
			  ELSE STKO.AEDAT END ,'yyyymmdd'),'yyyy-mm-dd') AS LAST_CHG_DATE, --修改时间如果为空，则赋值创建时间
   FROM_UNIXTIME(UNIX_TIMESTAMP(STKO.DATUV,'yyyymmdd'),'yyyy-mm-dd') AS VALID_DATE_BEGIN,
   STKO.WRKAN AS PROD_BOM_FACTORY_CODE,
   TW.NAME1 AS PROD_BOM_FACTORY_NAME,
   FROM_UNIXTIME(UNIX_TIMESTAMP(STKO.ANDAT,'yyyymmdd'),'yyyy-mm-dd') AS CRT_DATE,
   STKO.AEHLP AS DATA_CHG_FLAG,
   STKO.STKTX AS OPTIONAL_BOM_TEXT,
   STKO.BMENG AS BASIC_AMT,
   STKO.AENAM AS CHG_OBJECT_NAME,
   STKO.ANNAM AS CRT_REC_USER,
   T.CONFIG_CODE AS PROD_CLASSIFY_CODE,
   T1.CONFIG_CODE  AS BOM_STATUS_CODE,
   STKO.GUIDX AS BOM_H_CHG_G_STATUS_FLAG,
   STKO.BMEIN AS BOM_BASIC_UNIT_CODE,
   NVL(MM.MATERIAL_ID,-999998) AS MATERIAL_ID,
   NVL(MM.MATERIAL_NUM,'NOT EXISTS') AS MATERIAL_NUM,
    NVL(MM.MATERIAL_NAME,'主数据不存在') AS MATERIAL_NAME,
   'N'  AS DEL_FLAG
FROM SDI.SDI_STKO_1001 STKO
LEFT JOIN (SELECT MARA.MATERIAL_NUM,MARA.MATERIAL_ID,MARA.MATERIAL_NAME,MAST.STLNR,MAST.STLAL 
    FROM SDI.SDI_MAST_1001 MAST 
       INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MARA 
      ON MAST.MATNR = MARA.MATERIAL_NUM 
      AND MARA.END_DATE = '4712-12-31'
      AND MARA.DEL_FLAG = 'N' ) MM
      ON MM.STLNR = STKO.STLNR AND MM.STLAL = STKO.STLAL
LEFT JOIN  SDI.SDI_T001W_1001 TW
       ON  STKO.MANDT = TW.MANDT
	  AND STKO.WRKAN = TW.WERKS
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T 
       ON STKO.LABOR = T.SRC_SYS_CONFIG_CODE 
	  AND STKO.MANDT = T.GROUP_NUM 
	  AND T.SOURCE_CODE = 'T024X' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1
      ON STKO.STLST = T1.SRC_SYS_CONFIG_CODE 
     AND STKO.MANDT = T1.GROUP_NUM 
	 AND T1.SOURCE_CODE = 'T415T' ;
