set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_PURCHASE_APPLY_ACC_T		--DWI.SAP采购申请账户设置表
 SELECT
		SE1.MANDT					AS		GROUP_NUM,
		SE1.BANFN                   AS		PURCHASE_APPLY_NUM,
		SE1.BNFPO                   AS		PURCHASE_APPLY_ITEM_NUM,
		SE1.ZEBKN                   AS		PURCHASE_APPLY_ACC_SERIALNO,
		SE1.VBELP                   AS		SALE_VOUCHER_ITEM,
		SE1.MENGE                   AS		PURCHASE_APPLY_QTY,
		T.CONFIG_CODE           AS		COST_CENTER_CODE,
		SE1.AUFNR                   AS		ORDER_NUM,
		SE1.VETEN                   AS		PLAN_ITEM,
		T1.CONFIG_CODE          AS		BUSINESS_SCOPE,
		SE1.SAKTO                   AS		GL_ACCOUNT_NUM,
		T2.CONFIG_CODE          AS		CONTROL_SCOPE,
		SE1.VBELN                   AS		SALES_VCH_NUM,
		SE1.ANLN1                   AS		MAIN_ASSET_NUM,
		T3.CONFIG_CODE          AS		PROFIT_CENTER,
		SE1.FKBER                   AS		FUNCTION_SCOPE,
		SE1.ABLAD                   AS		DISCHARGE,
		SE1.FISTL                   AS		FUND_CENTER_CODE,
		SE1.ANLN2                   AS		ASSETS_SON_CODE,
		SE1.PAOBJNR                 AS		PROFITABILITY_SEGMENT_CODE,
		T4.CONFIG_CODE          AS		COMMITMENT_ITEM_ID,
		CASE WHEN SE1.ERDAT = '00000000' 	--判断SDI源表字段记录建立日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SE1.ERDAT, 1, 4) = '9999' --SDI源表字段记录建立日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.ERDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS RECORD_CREATE_DATE,
		SE1.KBLPOS                  AS		TR_RESERVE_VCH_ITEM,
		T5.CONFIG_CODE          AS		WBS_ELEMENT,
		SE1.AUFPL                   AS		ORDER_PROCESS_ROUTE_NO1,
		SE1.AUFPL_ORD               AS		ORDER_PROCESS_ROUTE_NO2,
		CASE WHEN SE1.DATEOFSERVICE = '00000000' --判断SDI源表字段Date of Service为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SE1.DATEOFSERVICE, 1, 4) = '9999' --SDI源表字段Date of Service为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.DATEOFSERVICE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS SERIVICE_DATE,
		'N' 					AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE
		FROM SDI.SDI_EBKN_1001 SE1		--SDI.SAP采购申请账户设置表
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T	--关联通用配置表
			 ON T.SRC_SYS_CONFIG_CODE = SE1.KOSTL		--取成本中心描述中文名称
			 AND T.GROUP_NUM = SE1.MANDT				--集团
			 AND T.SOURCE_CODE = 'CSKT'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1	--关联通用配置表
			 ON T1.SRC_SYS_CONFIG_CODE = SE1.GSBER		--取业务部门名称中文名称
			 AND T1.GROUP_NUM = SE1.MANDT				--集团
			 AND T1.SOURCE_CODE = 'TGSBT'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T2	--关联通用配置表
			 ON T2.SRC_SYS_CONFIG_CODE = SE1.KOKRS		--取成本控制范围中文名称
			 AND T2.GROUP_NUM = SE1.MANDT				--集团
			 AND T2.SOURCE_CODE = 'TKA01'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T3	--关联通用配置表
			 ON T3.SRC_SYS_CONFIG_CODE = SE1.PRCTR		--取利润中心主数据文本中文名称
			 AND T3.GROUP_NUM = SE1.MANDT				--集团
			 AND T3.SOURCE_CODE = 'CEPCT'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T4	--关联通用配置表
			 ON T4.SRC_SYS_CONFIG_CODE = SE1.FIPOS		--取承诺项目文本中文名称
			 AND T4.GROUP_NUM = SE1.MANDT				--集团
			 AND T4.SOURCE_CODE = 'FMCIT'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T5	--关联通用配置表
			 ON T5.SRC_SYS_CONFIG_CODE = SE1.PS_PSP_PNR	--取WBS(工作中断结构) 元素主数据中文名称
			 AND T5.GROUP_NUM = SE1.MANDT				--集团
			 AND T5.SOURCE_CODE = 'PRPS'
		UNION ALL			--SDI_EBKN_1001.SAP采购申请账户设置表 UNION ALL 拼接 DWI_PTP_PURCHASE_APPLY_ACC_T.SAP采购申请账户设置表
		SELECT
				 PAA.GROUP_NUM
				,PAA.PURCHASE_APPLY_NUM
				,PAA.PURCHASE_APPLY_ITEM_NUM
				,PAA.PURCHASE_APPLY_ACC_SERIALNO
				,PAA.SALE_VOUCHER_ITEM
				,PAA.PURCHASE_APPLY_QTY
				,PAA.COST_CENTER_CODE
				,PAA.ORDER_NUM
				,PAA.PLAN_ITEM
				,PAA.BUSINESS_SCOPE
				,PAA.GL_ACCOUNT_NUM
				,PAA.CONTROL_SCOPE
				,PAA.SALES_VCH_NUM
				,PAA.MAIN_ASSET_NUM
				,PAA.PROFIT_CENTER
				,PAA.FUNCTION_SCOPE
				,PAA.DISCHARGE
				,PAA.FUND_CENTER_CODE
				,PAA.ASSETS_SON_CODE
				,PAA.PROFITABILITY_SEGMENT_CODE
				,PAA.COMMITMENT_ITEM_ID
				,PAA.RECORD_CREATE_DATE
				,PAA.TR_RESERVE_VCH_ITEM
				,PAA.WBS_ELEMENT
				,PAA.ORDER_PROCESS_ROUTE_NO1
				,PAA.ORDER_PROCESS_ROUTE_NO2
				,PAA.SERIVICE_DATE
				,'Y' AS DEL_FLAG
				,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
				FROM DWIPUR.DWI_PTP_PURCHASE_APPLY_ACC_T PAA		--DWI.SAP采购申请账户设置表
				LEFT JOIN SDI.SDI_EBKN_1001 T1		--SDI.SAP采购申请账户设置表
				ON PAA.GROUP_NUM = T1.MANDT 		--集团						
				AND PAA.PURCHASE_APPLY_NUM = T1.BANFN 	--采购申请编号
				AND PAA.PURCHASE_APPLY_ITEM_NUM = T1.BNFPO	--采购申请的项目编号
				AND PAA.PURCHASE_APPLY_ACC_SERIALNO = T1.ZEBKN	--采购申请帐户分配段的序列号
				WHERE T1.MANDT IS NULL
				AND T1.BANFN IS NULL
				AND T1.BNFPO IS NULL
				AND T1.ZEBKN IS NULL
				AND PAA.DEL_FLAG = 'N'
