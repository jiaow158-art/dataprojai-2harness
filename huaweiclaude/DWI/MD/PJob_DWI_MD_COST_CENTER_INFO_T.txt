WITH BBB AS --成本中心层级结构
 (SELECT H.NODENAME,
         CASE
           WHEN H.IOBJNM = '0COSTCENTER' THEN
            ST.TXTSH
           ELSE
            R.TXTSH
         END NODEDESC,
         H.NODEID,
         H.PARENTID,
         H.TLEVEL,
         H.IOBJNM
    FROM SDI.SDI_BI0_HCOSTCENTER_1009 H
    LEFT JOIN SDI.SDI_BI0_TCOSTCENTER_1009 ST
      ON H.NODENAME = CONCAT(ST.CO_AREA, ST.COSTCENTER)
     AND REPLACE(CAST(DATE_SUB(CURRENT_DATE, 1) AS STRING), '-', '') BETWEEN
         ST.DATEFROM AND ST.DATETO
     AND ST.LANGU = '1'
     AND H.IOBJNM = '0COSTCENTER'
    LEFT JOIN SDI.SDI_RSTHIERNODE_1009 R
      ON R.NODENAME = H.NODENAME
     AND R.HIEID = H.HIEID
     AND R.LANGU = '1'
     AND R.OBJVERS = 'A'
   WHERE H.HIEID = 'U7IQQJ5IOXTSXJ041AZFRIE05'),
CCC AS
 (SELECT CASE
           WHEN H1.IOBJNM = '0COSTCENTER' THEN
            H1.NODENAME
           WHEN H2.IOBJNM = '0COSTCENTER' THEN
            H2.NODENAME
           WHEN H3.IOBJNM = '0COSTCENTER' THEN
            H3.NODENAME
           WHEN H4.IOBJNM = '0COSTCENTER' THEN
            H4.NODENAME

            H6.NODENAME

            H8.NODENAME

            H9.NODENAME
           WHEN H10.IOBJNM = '0COSTCENTER' THEN
            H10.NODENAME
           WHEN H11.IOBJNM = '0COSTCENTER' THEN
            H11.NODENAME
           WHEN H12.IOBJNM = '0COSTCENTER' THEN
            H12.NODENAME
           WHEN H13.IOBJNM = '0COSTCENTER' THEN
            H13.NODENAME
           WHEN H14.IOBJNM = '0COSTCENTER' THEN
            H14.NODENAME

         H4.NODEDESC AS COST_CENTER_NAME_4, --成本中心层级4,
         H5.NODENAME AS COST_CENTER_CODE_5, --节点5ID, 
         H5.NODEDESC AS COST_CENTER_NAME_5, --节点5描述,

         H11.NODENAME AS COST_CENTER_CODE_11, --节点11ID,

         H13.NODENAME AS COST_CENTER_CODE_13, --节点13ID, 
         H13.NODEDESC AS COST_CENTER_NAME_13, --节点13描述,
         H14.NODENAME AS COST_CENTER_CODE_14, --节点14ID, 
         H14.NODEDESC AS COST_CENTER_NAME_14, --节点14描述,

    LEFT JOIN BBB H2
      ON H1.NODEID = H2.PARENTID
    LEFT JOIN BBB H3
      ON H2.NODEID = H3.PARENTID
    LEFT JOIN BBB H4
      ON H3.NODEID = H4.PARENTID
    LEFT JOIN BBB H5
      ON H4.NODEID = H5.PARENTID
    LEFT JOIN BBB H6
      ON H5.NODEID = H6.PARENTID
    LEFT JOIN BBB H7

    LEFT JOIN BBB H13
      ON H12.NODEID = H13.PARENTID
    LEFT JOIN BBB H14
      ON H13.NODEID = H14.PARENTID
    LEFT JOIN BBB H15
      ON H14.NODEID = H15.PARENTID
   WHERE H1.TLEVEL = '01') 
INSERT OVERWRITE TABLE DWIMD.DWI_MD_COST_CENTER_INFO_T
SELECT CONCAT(Q.CO_AREA, Q.COSTCENTER) as zcost_center, C.KTEXT AS zcost_center_name,
      cost_center_code_1, cost_center_name_1, cost_center_code_2, cost_center_name_2, cost_center_code_3, cost_center_name_3, cost_center_code_4, cost_center_name_4, cost_center_code_5, cost_center_name_5, cost_center_code_6, cost_center_name_6, cost_center_code_7, cost_center_name_7, cost_center_code_8, cost_center_name_8, cost_center_code_9, cost_center_name_9, cost_center_code_10, cost_center_name_10, cost_center_code_11, cost_center_name_11, cost_center_code_12, cost_center_name_12, cost_center_code_13, cost_center_name_13, cost_center_code_14, cost_center_name_14, cost_center_code_15, cost_center_name_15,
       Q.SALES_GRP AS SALES_GRP_CODE, -- 销售组
       T.BEZEI AS SALES_GRP_NAME,
       Q.COMP_CODE ,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
       Q.func_area   --新增功能范围字段
  FROM SDI.SDI_BI0_QCOSTCENTER_1009 Q
  LEFT JOIN  CCC H
    ON H.ZCOST_CENTER = CONCAT(Q.CO_AREA, Q.COSTCENTER)
  LEFT JOIN SDI.SDI_TVGRT_1001 T
    ON Q.SALES_GRP = T.VKGRP AND T.SPRAS = '1' AND T.MANDT = '800'
      LEFT JOIN SDI.SDI_CSKT_1001 C
    ON Q.COSTCENTER = C.KOSTL AND Q.CO_AREA = C.KOKRS AND C.SPRAS = '1' AND C.MANDT = '800' AND C.DATBI = '99991231'
  WHERE  REPLACE(CAST(DATE_SUB(CURRENT_DATE, 1) AS STRING), '-', '') BETWEEN
       Q.DATEFROM AND Q.DATETO AND Q.CO_AREA = 'DP00'
