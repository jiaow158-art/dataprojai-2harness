SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN MTVFP ='' THEN NULL 
         ELSE MTVFP 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN MTVFP ='' THEN NULL 
         ELSE MTVFP 
         END,'SNULL'),'TMVFT') AS CONFIG_CODE,
BEZEI AS CONFIG_NAME,
'TMVFT' AS SOURCE_CODE,
'可用量检查控制：文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TMVFT_1001
