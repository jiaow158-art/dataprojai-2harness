INSERT OVERWRITE TABLE DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R 
SELECT 
DOMNAME  AS DOMAIN_NAME,
DDLANGUAGE AS LANG_CODE,
AS4LOCAL AS ACTIVE_STATUS,
VALPOS AS DOMAIN_VALUE_KEY,
DOMVALUE_L AS SRC_SYS_DOMAIN_VALUE_CODE,
DOMVALUE_L||'_'||DOMNAME AS DOMAIN_VALUE_CODE,
DDTEXT AS DOMAIN_VALUE_NAME,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_DD07T_1001
