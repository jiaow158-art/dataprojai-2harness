         ELSE STGMA	 
         END,'SNULL'),'TVSMT') AS CONFIG_CODE,

'TVSMT' AS SOURCE_CODE,
'材料: 统计组: 文本'AS SOURCE_DESC,

NULL AS EXTENDED_ATTRIBUTES2,
