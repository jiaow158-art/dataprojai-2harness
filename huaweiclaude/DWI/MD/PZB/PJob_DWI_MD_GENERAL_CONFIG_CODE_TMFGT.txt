SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN MFRGR ='' THEN NULL 
         ELSE MFRGR 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN MFRGR ='' THEN NULL 
         ELSE MFRGR 
         END,'SNULL'),'TMFGT') AS CONFIG_CODE,
BEZEI AS CONFIG_NAME,
'TMFGT' AS SOURCE_CODE, 
'物料货运组:描述'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TMFGT_1001
