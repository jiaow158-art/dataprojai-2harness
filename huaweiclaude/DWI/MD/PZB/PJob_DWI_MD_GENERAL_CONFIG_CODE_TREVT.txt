SPRAS AS LANG_CODE,
COALESCE(CASE WHEN FSTVARE ='' THEN NULL 
         ELSE FSTVARE

         ELSE FSTVARE	 
         END,'SNULL'),'TREVT') AS CONFIG_CODE,
FSTVARETXT AS CONFIG_NAME,
'TREVT' AS SOURCE_CODE,
'资金保留字段状态变式文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TREVT_1001
