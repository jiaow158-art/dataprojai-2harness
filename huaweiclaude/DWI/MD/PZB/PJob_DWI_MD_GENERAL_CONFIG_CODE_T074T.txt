INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='T074T') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN SHBKZ ='' THEN NULL 
         ELSE SHBKZ 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN SHBKZ = '' THEN NULL 
         ELSE SHBKZ	 
         END,'SNULL'),'T074T') AS CONFIG_CODE,
KTEXT AS CONFIG_NAME,
'T074T' AS SOURCE_CODE,
'特别总账'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_T074T_1001
WHERE MANDT = 800
AND SPRAS = '1'
and KOART = 'K'
