INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TWEWT') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN EXTWG ='' THEN NULL 
         ELSE EXTWG 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN EXTWG ='' THEN NULL 
         ELSE EXTWG 
         END,'SNULL'),'TWEWT') AS CONFIG_CODE,
EWBEZ AS CONFIG_NAME,
'TWEWT' AS SOURCE_CODE,
'外部物料组的分销'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TWEWT_1001
