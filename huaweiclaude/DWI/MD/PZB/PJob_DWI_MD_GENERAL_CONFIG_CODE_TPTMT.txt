INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TPTMT') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN MTPOS ='' THEN NULL 
         ELSE MTPOS 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN MTPOS ='' THEN NULL 
         ELSE MTPOS 
         END,'SNULL'),'TPTMT') AS CONFIG_CODE,
BEZEI AS CONFIG_NAME,
'TPTMT' AS SOURCE_CODE,
'材料: 物资主记录中的项目目录组: 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TPTMT_1001
