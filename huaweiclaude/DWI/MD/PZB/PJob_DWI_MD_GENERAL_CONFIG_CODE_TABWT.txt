INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TABWT') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN BWASL ='' THEN NULL 
         ELSE BWASL 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN BWASL ='' THEN NULL 
         ELSE BWASL	 
         END,'SNULL'),'TABWT') AS CONFIG_CODE,
BWATXT AS CONFIG_NAME,
'TABWT' AS SOURCE_CODE,
'资产处理类型文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TABWT_1001
