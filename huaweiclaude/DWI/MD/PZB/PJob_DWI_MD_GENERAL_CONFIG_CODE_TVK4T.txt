INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVK4T') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN KATR4 ='' THEN NULL 
         ELSE KATR4 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN KATR4 ='' THEN NULL 
         ELSE KATR4 
         END,'SNULL'),'TVK4T') AS CONFIG_CODE,  
VTEXT AS CONFIG_NAME,
'TVK4T' AS SOURCE_CODE,
'属性4文本（客户主文件）'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVK4T_1001
