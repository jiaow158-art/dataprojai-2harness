SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN LPRIO ='' THEN NULL 
         ELSE LPRIO 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN LPRIO ='' THEN NULL 
         ELSE LPRIO 
         END,'SNULL'),'TPRIT') AS CONFIG_CODE,
BEZEI AS CONFIG_NAME,
'TPRIT' AS SOURCE_CODE,
'客户: 交货优先: 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
