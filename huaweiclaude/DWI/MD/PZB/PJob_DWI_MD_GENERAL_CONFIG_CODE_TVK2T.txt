INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVK2T') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN KATR2 ='' THEN NULL 
         ELSE KATR2 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN KATR2 ='' THEN NULL 
         ELSE KATR2 
         END,'SNULL'),'TVK2T') AS CONFIG_CODE,   
VTEXT AS CONFIG_NAME,
'TVK2T' AS SOURCE_CODE,
'属性2文本（客户主文件）'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVK2T_1001
