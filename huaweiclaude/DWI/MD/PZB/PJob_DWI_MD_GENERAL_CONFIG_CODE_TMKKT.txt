SPRAS AS LANG_CODE,
COALESCE(CASE WHEN KALSK ='' THEN NULL 
         ELSE KALSK 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,

         ELSE KALSK 
         END,'SNULL'),'TMKKT') AS CONFIG_CODE,
KALSB AS CONFIG_NAME,
'TMKKT' AS SOURCE_CODE, 
'有关计算方案的组(供应商): 描述'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TMKKT_1001
