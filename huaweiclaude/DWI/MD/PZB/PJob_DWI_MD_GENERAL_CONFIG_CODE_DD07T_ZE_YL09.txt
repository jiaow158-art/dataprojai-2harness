INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='DD07T_ZE_YL09') 
SELECT '800' AS GROUP_NUM,                 
'1' AS LANG_CODE,
COALESCE(CASE WHEN VALUE_CODE ='' THEN NULL 
         ELSE VALUE_CODE 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN VALUE_CODE ='' THEN NULL 
         ELSE VALUE_CODE	 
         END,'SNULL'),'DD07T_ZE_YL09') AS CONFIG_CODE,
VALUE_DESC AS CONFIG_NAME,
'DD07T_ZE_YL09' AS SOURCE_CODE,
'产品所有权' AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZMMT014_1001
WHERE FIELDNAME = 'YL09'
