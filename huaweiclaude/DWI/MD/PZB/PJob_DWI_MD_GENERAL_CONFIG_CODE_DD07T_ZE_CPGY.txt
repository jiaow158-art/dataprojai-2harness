INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D = 'DD07T_ZE_CPGY') 
SELECT '800' AS GROUP_NUM,                 
DDLANGUAGE AS LANG_CODE,
COALESCE(CASE WHEN DOMVALUE_L ='' THEN NULL 
         ELSE DOMVALUE_L 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN DOMVALUE_L ='' THEN NULL 
         ELSE DOMVALUE_L	 
         END,'SNULL'),'DD07T_ZE_CPGY') AS CONFIG_CODE,
DDTEXT AS CONFIG_NAME,
'DD07T_ZE_CPGY' AS SOURCE_CODE,
'产品工艺' AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_DD07T_1001
WHERE DOMNAME = 'ZE_CPGY'
