SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN PARVW ='' THEN NULL 
         ELSE PARVW 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN PARVW ='' THEN NULL 
         ELSE PARVW 
         END,'SNULL'),'TPART') AS CONFIG_CODE,
VTEXT AS CONFIG_NAME,
'TPART' AS SOURCE_CODE,
'业务合作伙伴功能: 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TPART_1001
