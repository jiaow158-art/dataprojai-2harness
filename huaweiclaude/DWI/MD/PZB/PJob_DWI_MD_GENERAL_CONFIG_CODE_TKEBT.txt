INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TKEBT') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN ERKRS ='' THEN NULL 
         ELSE ERKRS 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN ERKRS ='' THEN NULL 
         ELSE ERKRS	 
         END,'SNULL'),'TKEBT') AS CONFIG_CODE,
ERKRS_BZ AS CONFIG_NAME,
'TKEBT' AS SOURCE_CODE,
'工序修改的描述'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TKEBT_1001
