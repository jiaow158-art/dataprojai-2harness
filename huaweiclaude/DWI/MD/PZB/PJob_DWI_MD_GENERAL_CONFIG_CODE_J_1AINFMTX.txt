         ELSE J_1AINFMET	 
         END,'SNULL'),'J_1AINFMTX') AS CONFIG_CODE,
INFMETTXT AS CONFIG_NAME,
'J_1AINFMTX' AS SOURCE_CODE,
'通货膨胀方法： 描述'AS SOURCE_DESC,

NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
