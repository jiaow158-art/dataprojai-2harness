INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVKTT') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN KTGRD ='' THEN NULL 
         ELSE KTGRD 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN KTGRD ='' THEN NULL 
         ELSE KTGRD 
         END,'SNULL'),'TVKTT') AS CONFIG_CODE,  
VTEXT AS CONFIG_NAME,
'TVKTT' AS SOURCE_CODE,
'客户：帐户确定组：文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVKTT_1001
