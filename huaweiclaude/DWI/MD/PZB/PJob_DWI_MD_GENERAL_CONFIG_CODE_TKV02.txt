SPRAS AS LANG_CODE,
COALESCE(CASE WHEN AWSLS ='' THEN NULL 
         ELSE AWSLS 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,

         ELSE AWSLS	 
         END,'SNULL'),'TKV02') AS CONFIG_CODE,
AWSTX AS CONFIG_NAME,
'TKV02' AS SOURCE_CODE,
'差异关键字名'AS SOURCE_DESC,
AWOBA AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TKV02_1001
