INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='SAPAPO_RRPTYPET') 
SELECT MANDT AS GROUP_NUM,                 
LANGU AS LANG_CODE,
COALESCE(CASE WHEN RRP_TYPE ='' THEN NULL 
         ELSE RRP_TYPE 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN RRP_TYPE ='' THEN NULL 
         ELSE RRP_TYPE	 
         END,'SNULL'),'SAPAPO_RRPTYPET') AS CONFIG_CODE,
TEXT AS CONFIG_NAME,
'SAPAPO_RRPTYPET' AS SOURCE_CODE,
'规划程序文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_SAPAPO_RRPTYPET_1001
