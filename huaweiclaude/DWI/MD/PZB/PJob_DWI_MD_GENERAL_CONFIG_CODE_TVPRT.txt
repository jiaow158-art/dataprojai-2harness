INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVPRT') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE, 
COALESCE(CASE WHEN PROVG ='' THEN NULL 
         ELSE PROVG 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN PROVG ='' THEN NULL 
         ELSE PROVG 
         END,'SNULL'),'TVPRT') AS CONFIG_CODE,
VTEXT AS CONFIG_NAME,
'TVPRT' AS SOURCE_CODE,
'佣金组文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVPRT_1001
