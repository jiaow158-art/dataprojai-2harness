INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TGSBT') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN GSBER ='' THEN NULL 
         ELSE GSBER 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN GSBER ='' THEN NULL 
         ELSE GSBER 
         END,'SNULL'),'TGSBT') AS CONFIG_CODE,
GTEXT AS CONFIG_NAME,
'TGSBT' AS SOURCE_CODE, 
'业务部门名称'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TGSBT_1001
