insert OVERWRITE table dwimd.dwi_md_data_material_factory_t_his_test
SELECT 
      T7.MATNR AS MATERIAL_NUM,--物料编号
      MARC.MANDT AS GROUP_NUM,--集团
      MARC.WERKS AS FACTORY_CODE,--工厂编码
	  R.DDTEXT AS INTEGRATE_MRP_NAME,--综合名称
	  T.VTEXT AS LOAD_GROUP_NAME,--装载组名称
      T1.SPBEZ  AS REPE_PARAMETER_FILE_NAME,--重复生产参数文件名称
      MARC.WEBAZ AS RECEIVE_GOODS_DAYS,--以天计的收货处理时间
      MARC.VINT2 AS FORWARD_CONSUM_PERIOD,--消耗时期-向前
      MARC.VINT1 AS REVERSE_CONSUM_PERIOD,--消耗时期-逆向
	  R1.DDTEXT AS CONSUM_PATTERNS_NAME,--消耗模式名称	  
      T2.BEZEI  AS MATL_SHIP_GROUP_NAME,--物料运输组名称
      T3.DIBEZ  AS MATL_REQUIRE_PLAN_TYPE_NAME,--物料需求计划类型名称
      MARC.DISPO AS MATL_REQ_PLAN_CTRER_CODE,--物料需求计划控制员
      MARC.DISLS  AS MATL_P_BATCH_PROCEDURE_CODE,--物料计划内批量程序
      R2.DDTEXT AS NO_COST_ACCT_NAME,--无成本核算名称
      MARC.SFCPF AS PROD_PLAN_PARAMETER_FILE,--生产计划参数文件
      MARC.PRCTR AS PROFIT_CENTER_CODE,--利润中心
      MARC.MTVFP AS USABLE_CHECK_GROUP_CODE,--可用性检查的检查组
      MARC.PLIFZ AS PLAN_DELIVERY_DAYS,--计划交货时间（天）
      T6.TEXT40  AS PLAN_STRATEGY_GROUP_NAME,--计划策略组名称
      R3.DDTEXT AS POST_TO_INSPECT_STOCK_NAME,--过帐到检验库存名称
      MARC.FHORI AS FLOAT_P_P_CODE,--浮动排产时间容余码
      MARC.LGPRO AS DELIVERY_LOC_CODE,--发货存储地点
      MARC.SBDKZ AS RELATED_REQUIRE_FLAG,--对于独立和集中需求的相关需求标识   
      MARC.DZEIT AS FACTORY_PROD_DAYS,--厂内生产时间天数     
      T5.EKNAM AS PUR_GROUP_NAME,--采购组名称
      R4.DDTEXT AS PUR_TYPE_NAME,--采购类型名称
      MARC.SAUFT AS ALLOW_REMANUFACTURE_FLAG,--标识：允许的重复制造
      MARC.RGEKZ AS RECOIL_FLAG,--标识：反冲
      MARC.KORDB AS SOURCE_LIST_REQUIRE_FLAG,--标识: 源清单要求
      'N' AS DEL_FLAG,
      MD.update_date start_date,
      '4712-12-31' end_date,
      MD.update_date AS  update_date,
t7.LGORT as WAREHOUSE_CODE,	--仓库地点 
MARC.MFRGR AS MATL_SHIP_GROUP_CODE,--物料运输组名称
MARC.LADGR AS LOAD_GROUP_CODE,--装载组
T10.BEZEI AS USABLE_CHECK_GROUP_NAME,-- 可用性检查名称 TMVFT-BEZEI
MARC.EKGRP AS PUR_GROUP_CODE,--采购组 EKGRP
T12.MCTXT AS PROFIT_CENTER_NAME,--利润中心名称 M_PRCTN-MCTXT
MARC.DISMM AS MATL_REQUIRE_PLAN_TYPE_CODE,--物料需求计划类型 DISMM
T4.LOSLT AS MATL_P_BATCH_PROCEDURE_NAME, --批量大小 T439T-LOSLT 
MARC.BESKZ AS PUR_TYPE_CODE,-- 采购类型 BESKZ
MARC.SOBSL AS SPE_PUR_TYPE_CODE,--特殊采购类型 MARC-SOBSL
T11.LTEXT AS SPE_PUR_TYPE_NAME,--特殊采购类型  T460T-LTEXT
MARC.STRGR AS PLAN_STRATEGY_GROUP_CODE,--策略组 STRGR
MARC.VRMOD AS CONSUM_PATTERNS_CODE,--消耗模式 VRMOD
MARC.MISKZ AS INTEGRATE_MRP_FLAG,--综合 MISKZ
MARC.SFEPR AS REPE_PARAMETER_FILE_CODE,-- 重复生产参数文件 SFEPR
MARC.FEVOR AS PRODUTION_MANAGER,--生产管理员 FEVOR
MD.create_date AS create_date,
from_unixtime(unix_timestamp() ,'yyyy-MM-dd HH:mm:ss') dw_last_update_date
  FROM 
   SDI.sdi_MARD_test_1001 T7 --LGORT       
  LEFT JOIN SDI.SDI_MARC_test_1001 MARC
  ON MARC.MANDT = T7.MANDT 
	  AND MARC.MATNR = T7.matnr 
	  AND MARC.WERKS = T7.WERKS 
LEFT JOIN dwimd.dwi_md_data_material_general_t_his_test MD
      ON T7.MANDT = MD.GROUP_NUM
	  AND T7.MATNR = MD.MATERIAL_NUM
LEFT JOIN  SDI.SDI_DD07T_1001 R
       ON R.DOMNAME = 'MISKZ'
      AND MARC.MISKZ = R.DOMVALUE_L
	  AND R.DDLANGUAGE = '1'
LEFT JOIN  SDI.SDI_DD07T_1001 R1
      ON  R1.DOMNAME  = 'VRMOD'
      AND MARC.VRMOD = R1.DOMVALUE_L 
	  AND R1.DDLANGUAGE = '1'
LEFT JOIN  SDI.SDI_DD07T_1001 R2
      ON  R2.DOMNAME  = 'CK_NO_COSTING'
      AND MARC.NCOST = R2.DOMVALUE_L 
	  AND R2.DDLANGUAGE = '1'
LEFT JOIN  SDI.SDI_DD07T_1001 R3
      ON  R3.DOMNAME  = 'QKZ'
      AND MARC.INSMK = R3.DOMVALUE_L 
	  AND R3.DDLANGUAGE = '1'
LEFT JOIN  SDI.SDI_DD07T_1001 R4
      ON  R4.DOMNAME  = 'BESKZ'
      AND MARC.BESKZ = R4.DOMVALUE_L  
      AND R4.DDLANGUAGE = '1'		  
LEFT JOIN SDI.SDI_TLGRT_1001 T --VTEXT
       ON MARC.LADGR = T.LADGR 
	  AND MARC.MANDT = T.MANDT 
	  AND T.SPRAS = '1'
LEFT JOIN SDI.SDI_T437R_1001 T1 --SPBEZ
       ON MARC.SFEPR = T1.SFPRO 
	  AND MARC.MANDT = T1.MANDT 
	  AND T1.SPRAS = '1'
LEFT JOIN SDI.SDI_TMFGT_1001 T2 --BEZEI
       ON MARC.MFRGR = T2.MFRGR 
	  AND MARC.MANDT = T2.MANDT 
	  AND T2.SPRAS = '1'
LEFT JOIN SDI.SDI_T438T_1001 T3 --DIBEZ
       ON MARC.DISMM = T3.DISMM 
	  AND MARC.MANDT = T3.MANDT 
	  AND T3.SPRAS = '1'
LEFT JOIN SDI.SDI_T439T_1001 T4 --LOSLT
       ON MARC.DISLS = T4.DISLS 
	  AND MARC.MANDT = T4.MANDT 
	  AND T4.SPRAS = '1'
LEFT JOIN SDI.SDI_T024_1001 T5 --EKNAM
       ON MARC.EKGRP = T5.EKGRP
	  AND MARC.MANDT = T5.MANDT
LEFT JOIN SDI.SDI_T461X_1001 T6 --TEXT40
       ON MARC.STRGR = T6.STRGR 
	  AND MARC.MANDT = T6.MANDT 
	  AND T6.SPRAS = '1'	
LEFT JOIN SDI.sdi_TMVFT_1001 T10 --BEZEI
       ON MARC.MTVFP = T10.MTVFP 
	  AND MARC.MANDT = T10.MANDT 
	  AND T10.SPRAS = '1' 
LEFT JOIN SDI.sdi_T460T_1001 T11 --LTEXT
       ON MARC.SOBSL = T11.SOBSL 
	  AND MARC.MANDT = T11.MANDT 
	  AND MARC.WERKS = T11.WERKS
	  AND T11.SPRAS = '1'	
LEFT JOIN SDI.sdi_M_PRCTN_1001 T12 --MCTXT
       ON MARC.PRCTR = T12.PRCTR 
	  AND MARC.MANDT = T12.MANDT 
	  AND T12.SPRAS = '1'
