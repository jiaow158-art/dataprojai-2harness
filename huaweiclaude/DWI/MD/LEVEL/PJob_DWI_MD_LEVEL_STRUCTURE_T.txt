INSERT OVERWRITE TABLE DWIMD.DWI_MD_LEVEL_STRUCTURE_T          --层次结构表
SELECT 
'HCOSTCENTER'            AS           SOURCE_TYPE,      
S.HIEID                  AS           INNER_LEVEL_ID,      
S.OBJVERS                AS           OBJECT_VERSION,       
S.NODEID                 AS           NODE_ID,              
S.IOBJNM                 AS           OBJECT_NAME,          
S.CHILDID                AS           CHILD_ID,            
S.NODENAME               AS           NODE_NAME,            
S.NEXTID                 AS           NEXT_ID,              
S.PARENTID               AS           PARENT_ID,            
S.TLEVEL                 AS           NODE_LEVEL,           
S.LINK                   AS           LINK_FLAG,            
'N'                      AS           DEL_FLAG,             
CURRENT_TIMESTAMP()      AS           DW_LAST_UPDATE_DATE 
FROM SDI.SDI_BI0_HCOSTCENTER_1001 S
UNION ALL
SELECT 
'HSALES_GRP'              AS           SOURCE_TYPE,      
S1.HIEID                  AS           INNER_LEVEL_ID,      
S1.OBJVERS                AS           OBJECT_VERSION,       
S1.NODEID                 AS           NODE_ID,              
S1.IOBJNM                 AS           OBJECT_NAME,          
S1.CHILDID                AS           CHILD_ID,            
S1.NODENAME               AS           NODE_NAME,            
S1.NEXTID                 AS           NEXT_ID,              
S1.PARENTID               AS           PARENT_ID,            
S1.TLEVEL                 AS           NODE_LEVEL,           
S1.LINK                   AS           LINK_FLAG,            
'N'                       AS           DEL_FLAG,             
CURRENT_TIMESTAMP()       AS           DW_LAST_UPDATE_DATE 
FROM SDI.SDI_BI0_HSALES_GRP_1001 S1
