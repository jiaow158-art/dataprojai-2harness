INSERT OVERWRITE TABLE DWIMD.DWI_MD_PRODUCT_AREA_T
--产区关系主数据
SELECT 
       A.ID
      ,A.BATCH_START
      ,A.DISTRIBUTION_FLAG
      ,A.INV_ORG_ID
      ,A.INVENTORY_LOCATION
      ,A.IS_PHANTOM
      ,A.PLANT_CLASS
      ,A.PRODUCT_AREA_ID
	  ,B.CODE AS PRODUCT_AREA_CODE
      ,B.NAME AS PRODUCT_AREA_NAME
      ,A.PRODUCT_BASE AS SALE_FACTORY_CODE
      ,A.PRODUCT_FACTORY AS PRODUCT_FACTORY_CODE
      ,A.SUPPLIER_CODE AS SUPPLIER_OR_CREDITOR_ACCT
      ,A.TYPE_ AS FACTORY_TYPE 
      ,A.CREATE_BY
      ,A.CREATE_DATE	  
      ,A.UPDATE_BY
      ,A.UPDATE_DATE
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_PRODUCTIONAREARELATION_1007 A
LEFT JOIN DWIOTD.DWI_PS_RES_ENTERPRISE_T B
ON B.ID = A.PRODUCT_AREA_ID
AND B.DEL_FLAG = 'N'
WHERE A.DEL_FLAG = 'N'
AND A.is_phantom = 0
and A.inv_org_id = 1
;
