       A.ID	AS USER_ID --自增ID
      ,A.FACE_ID	 --	人脸ID
      ,A.WX_EXT_USER_ID	 --	企业微信外部联系人ID
      ,A.WX_UNION_ID	 --	微信UNIONID
      ,A.WX_OPEN_ID	 --	微信OPENID

      ,A.SOURCE	 --	客户来源,参考CLIENTSOURCEENUM类
      ,A.REAL_NAME	 --	真实姓名
      ,A.NICK_NAME	 --	昵称
      ,A.PHONE AS TEL_NUM --手机号
      ,A.BIRTHDAY	 --	生日YYYY-MM-DD
      ,A.GENDER	 --	性别 1-男性;2-女性;0-未知

      ,C.PROVINCE_ID --	省区域ID
      ,C.PROVINCE AS PROVINCE_NAME --省区域名
      ,C.CITY_ID	 --	市区域ID
      ,C.CITY AS CITY_NAME --市区域名
      ,C.DISTRICT_ID	 --	区区域ID
      ,C.DISTRICT AS DISTRICT_NAME -- 区区域名
      ,C.COMMUNITY_ADDRESS	 --	小区地址
      ,C.DETAILED_ADDRESS	 --	详细地址
      ,C.DETAILED_ADDRESS_LONGITUDE	 --	详细地址经度

      ,A.BIND_EMPLOYEE_ID	 --	归属员工ID
      ,A.BIND_RULE_ID	 --	客户归属关系绑定规则的ID
      ,A.BIND_TIME AS LAST_BIND_TIME --最后绑定时间
      ,C.CREATE_TIME	 --	创建时间
      ,C.UPDATE_TIME	 --	更新时间
      ,A.PROMOTER	 --	推广人
      ,A.GRADE	 --	客户等级
      ,A.PROMOTION_TYPE	 --	推广方式， 1-导购推广 2-员工推广 3-客户推广
      ,A.CORP_NAME	 --	企业名称(只有添加的企业外部联系人才有)
      ,A.INVALID_CAUSE	 --	无效原因:1-客户不想要 2-客户选择其他品牌 3-客户收房延期 4-内部人员 5-其他 参考CLIENTINVALIDENUM枚举类
      ,A.TIKTOK_ID	 --	抖音数据ID
      ,A.FOLLOW_INFO	 --	跟进情况
      ,A.INVALID_TIME	 --	无效时间
      ,A.TELESALES_TIME	 --	电销时间
      ,A.DESCRIPTION	 --	备注/描述
      ,A.APP_FOLLOW_INFO	 --	APP跟进情况
      ,A.PROMOTER_ID	 --	推广人客户ID
      ,A.AMBASSADOR_STORE_ID	 --	推广大使门店
      ,A.AMBASSADOR_CREATE_TIME	 --	成为大使时间
      ,A.AGE_GROUP	 --	年龄层，1-18~25岁，2-26~35岁，3-36岁及以上
      ,A.DECORATE_PREDILECTION	 --	装修喜好，1-中国风，2-欧美风，3-日式
      ,A.DEALER_ID	 --	经销商ID
      ,A.DEALER_NAME	 --	归属经销商
      ,A.LONGITUDE	 --	经度
      ,A.LATITUDE	 --	纬度
      ,A.IS_ASSET_CLIENT	 --	是否多资产客户:0-否/1-是
      ,A.ASSET_TYPE	 --	多资产类型:1-主客户/2-多资产客户
      ,A.IS_OLD_CLIENT_ADD	 --	是否老客户新增
      ,B.ACTIVATION_TIME	 --	激活时间
      ,B.ACTIVATION_EMPLOYEE_ID	 --	激活导购ID
      ,B.FOLLOW_TIME	 --	跟进时间
      ,B.FOLLOW_EMPLOYEE_ID	 --	跟进导购ID
      ,B.REACH_STORE_TIME	 --	到店时间
      ,B.REACH_STORE_EMPLOYEE_ID	 --	到店导购ID
      ,B.ALREADY_PAY_TIME	 --	交定时间
      ,B.ALREADY_PAY_TYPE	 --	交定方式
      ,B.ALREADY_PAY_MONEY	 --	交定金额
      ,B.ALREADY_PAY_EMPLOYEE_ID	 --	交定导购ID
      ,B.ORDER_TIME	 --	开单时间
      ,B.ORDER_MONEY	 --	开单金额
      ,B.ORDER_EMPLOYEE_ID	 --	开单导购ID
      ,B.ACCOMPLISH_TIME	 --	完成时间
      ,B.ACCOMPLISH_EMPLOYEE_ID	 --	完成导购ID

      ,B.CLOSE_TYPE	 --	关闭原因
      ,B.CLOSE_EMPLOYEE_ID	 --	关闭导购ID

      ,C.UPDATE_STAGE_REMARK --	客户阶段变更备注
      ,C.BIND_TIME --客户绑定时间
      ,C.CUSTOMER_SOURCE --企业自定义来源
      ,C.IS_WXWORK_EXT_USER	 --客户是否是企微外部联系人
      ,C.IS_WHITELIST	 --是否白名单，1：是；0：否；
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_CLIENT_INFO_1030 A
 INNER JOIN SDI.SDI_CLIENT_STAGE_CHANGE_DETAIL_1030 B
    ON B.CLIENT_ID = A.ID
   AND B.EP_ID = 3044
   AND B.IS_DELETED = 0
 INNER JOIN SDI.SDI_CLIENT_EMPLOYEE_BIND_1030 C
    ON B.STORE_ID = C.STORE_ID
   AND B.CLIENT_ID = C.CLIENT_ID
   AND C.EP_ID = 3044
   AND C.IS_DELETED = 0
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.DELETED = 0
