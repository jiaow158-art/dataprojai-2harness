INSERT OVERWRITE TABLE DWIMD.DWI_MD_APS_CUSTOMER_RELATION_T
select t2.code as operation_center_code,
       t4.cust_short_name as operation_center_name,
       --t2.name as operation_center_name, 
       t3.code as area_code,
       t3.name as area_name,
       'N' as del_flag,
       current_timestamp() as dw_last_update_date
from SDI.SDI_WYAPS_INSERVICE_RELATION_1007 t1
JOIN (select id,code,name from sdi.sdi_bd_catalog_1007 where catalog_type_id =37 and is_phantom=0) t2 
ON t1.operation_center_id  = t2.id
JOIN (select id,code,name from sdi.sdi_bd_catalog_1007 where catalog_type_id =39 and is_phantom=0) t3 
ON t1.area_id  = t3.id
JOIN (select case when length(t.cust_code)<10 then concat('0000',t.cust_code) else t.cust_code end as cust_code,
	   t.abbreviation as cust_short_name 
	from sdi.sdi_b1_customer_data_1037 t 
	where t.is_main = '是'
) t4
on t2.code = t4.cust_code;
