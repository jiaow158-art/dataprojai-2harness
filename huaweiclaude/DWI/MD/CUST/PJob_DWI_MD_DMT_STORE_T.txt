set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

INSERT OVERWRITE TABLE DWIMD.DWI_MD_DMT_STORE_T
SELECT
	   ID AS STORE_ID 							-- 门店ID
      ,NAME AS STORE_NAME 						-- 门店名称
      ,ABBREVIATION AS STORE_BYNAME 			-- 门店简称
      ,STATUS 									-- 状态
      ,PROVINCE_ID 								-- 省区域ID      
      ,PROVINCE	AS PROVINCE_NAME 				-- 省
      ,CITY_ID 									-- 市区域ID      
      ,CITY AS CITY_NAME 						-- 市
      ,DISTRICT_ID 								-- 区区域ID      
      ,DISTRICT AS DISTRICT_NAME 				-- 区
      ,ADDRESS AS ADDR_NAME 					-- 详细地址
      ,OWNER_ID									-- 店长用户ID
      ,TEL AS TEL_NUM 							-- 店铺联系方式
      ,LENGTH 									-- 店铺长度
      ,WIDTH 									-- 店铺宽度
      ,BUSINESS_START_HOUR						-- 经营开始时间	
      ,BUSINESS_END_HOUR 						-- 经营结束时间		
      ,LONGITUDE 								-- 门店经度
      ,LATITUDE 								-- 门店纬度
      ,BD_LONGITUDE 							-- 百度地图经度
      ,BD_LATITUDE 								-- 百度地图纬度
      ,FIELD 									-- 门店所属板块（领域）
      ,APP_ID 									-- 店铺ID
      ,`DESCRIBE` 								-- 描述
      ,`TYPE` AS STORE_TYPE						-- 门店类型（0：门店 1：商圈）
      ,BUSINESS_DAYOFWEEK 						-- 门店营业星期
      ,UPDATE_TIME								-- 最后更新时间
      ,CREATE_TIME 								-- 创建时间	
      ,IS_OPEN 									-- 是否营业 1-营业，0-打烊
      ,OUT_STORE_ID 							-- 外部门店编号
      ,PARENT_ID 								-- 父ID
      ,MODE										-- 模式, 0:直营; 1:经销商
      ,DEALER_ID 								-- 经销商ID
      ,MERCHANT_TYPE 							-- 0-普通商户 1-平台自营 2-品牌自营
      ,CONTACT_NAME								-- 联系人
      ,CONTACT_PHONE 							-- 联系人手机号
      ,SYN_BRAND_TEMPLATE 						-- 是否同步品牌模板
      ,LOCK_BRAND_TEMPLATE						-- 锁定门店装修模板
      ,CATEGORY									-- 经营类目
      ,OUT_DEALER_ID 							-- 外部经销商ID
      ,HOTEL_NAME 								-- 酒店名称
      ,STORE_TYPE AS HOTEL_TYPE					-- 酒店类型:0:非酒店1:酒店
      ,STORE_TAG 								-- 门店标签
      ,OPEN_STATUS_CHANGE_TIME 					-- 营业状态变更时间
      ,OPEN_STATUS_CHANGE_PER_USER_ID 			-- 营业状态变更人ID
      ,OPEN_STATUS_CHANGE_PER_USER_NAME			-- 营业状态变更人名称
      ,SUBSTR(CREATE_TIME,1,10) AS START_DATE 	-- 开始日期
      ,SUBSTR(UPDATE_TIME,1,10) AS UPDATE_DATE	-- 更新日期
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(), 'YYYY-MM-DD HH:MM:SS') AS DW_LAST_UPDATE_DATE -- 同步数据时间
FROM SDI.SDI_BUZ_STORE_1030
WHERE EP_ID = 3044
AND APP_ID = 1831
;
