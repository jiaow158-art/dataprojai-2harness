WITH LOOKUP_DATA AS(
	SELECT T.LOOKUP_TYPE,T.LOOKUP_CODE ,T.MEANING 
	FROM SDI.SDI_BASE_LOOKUP_VALUES_1027 T 
	WHERE T.ENABLED_FLAG = 'Y' 
	AND  T.LOOKUP_TYPE IN ('ICPOS_STORE_TYPE','ICPOS_STORE_STATUS','ICPOS_STORE_NATURE','ICPOS_SHOP_TYPE')
)
INSERT OVERWRITE TABLE DWIMD.DWI_MD_STORE_DATA_TEMP
SELECT /* +MAPJOIN(ICPOS_STORE) */
	T1.STORE_ID							AS STORE_ID							-- 门店ID

	,'' 								AS OPERATION_CENTER_NAME			-- 运营中心名称

	,T1.CUSTOMER_NAME	AS CUSTOMER_NAME 					-- 经销商名称
	--删除逻辑：当分销商编码在主档为空时，获取经销商编码，改为直接获取分销商编码，分销商名称也是同样的逻辑
	/*
	,(CASE WHEN T1.DISTRIBUTOR_CODE = '' OR T1.DISTRIBUTOR_CODE IS NULL OR T1.DISTRIBUTOR_CODE ='NULL'
		THEN T1.CUSTOMER_CODE ELSE T1.DISTRIBUTOR_CODE 
		END ) 							AS DISTRIBUTOR_CODE					-- 分销商编码
	,(CASE WHEN T1.DISTRIBUTOR_NAME = '' OR T1.DISTRIBUTOR_NAME IS NULL OR T1.DISTRIBUTOR_NAME ='NULL'
		THEN T1.CUSTOMER_NAME ELSE T1.DISTRIBUTOR_NAME 
		END ) 							AS DISTRIBUTOR_NAME	                -- 分销商名称 */
	,T1.DISTRIBUTOR_CODE                AS DISTRIBUTOR_CODE					-- 分销商编码
    ,T1.DISTRIBUTOR_NAME	            AS DISTRIBUTOR_NAME	                -- 分销商名称
	,INFO.OLD_STORE_CODE				AS OLD_STORE_CODE					-- 旧门店编号
	,T1.STORE_CODE						AS STORE_CODE						-- 门店编码
	,T1.STORE_NAME						AS STORE_NAME						-- 门店名称
	,T1.DESIGN_AREA						AS DESIGN_AREA						-- 店面面积
	,T1.STORE_AREA						AS STORE_AREA						-- 计划东鹏产品展示面积（M²）
	,T6.MEANING 						AS SHOP_TYPE						-- 门店类型
	,T3.MEANING 						AS STORE_TYPE						-- 店面类型
	,T5.MEANING 						AS STORE_NATURE						-- 店面性质
	,T4.MEANING 						AS STORE_STATUS						-- 店面状态
	,T1.PROVINCE						AS PROVINCE							-- 省份
	,T1.CITY							AS CITY								-- 城市
	,T1.COUNTY							AS COUNTY							-- 区县
	,T1.ADDRESS							AS ADDRESS							-- 详细地址
	,SEAT.IS_GET_LOCATION				AS IS_GET_LOCATION					-- 是否获取经纬度
	,SEAT.LONGITUDE						AS LONGITUDE						-- 经度
	,SEAT.LATITUDE						AS LATITUDE							-- 纬度
	,SUBSTR(T1.STORE_FILING_DATE,1,10) 	AS STORE_FILING_DATE				-- 店面建档时间
	,T1.COMPLETION_DATE					AS COMPLETION_DATE 					-- 完工时间
	,T1.OPERATION_TYPE					AS OPERATION_TYPE 					-- 是否经营中
	,INFO.IS_PRODUCT_TO_DP				AS IS_PRODUCT_TO_DP 				-- 是否竞品转东鹏
	,INFO.PRODUCT_BRAND					AS PRODUCT_BRAND 					-- 竞品品牌
	,INFO.IS_NEED_RESET					AS IS_NEED_RESET 					-- 是否需要重装
	,INFO.REVOKE_DATE					AS REVOKE_DATE 						-- 撤销日期
	,INFO.REVOKE_REASON					AS REVOKE_REASON 					-- 撤销原因
	,INFO.ADDRESS_CONFLICT_REASON		AS ADDRESS_CONFLICT_REASON 			-- 地址冲突原因
	,'N' 								AS DEL_FLAG							-- 删除标识
	,SUBSTR(T1.LAST_UPDATE_DATE,1,10) 	AS START_DATE                       -- 开始时间
	,'4712-12-31' 						AS END_DATE                         -- 结束时间
	,SUBSTR(T1.LAST_UPDATE_DATE,1,10) 	AS UPDATE_DATE                      -- 更新时间
FROM 
	SDI.SDI_ICPOS_STORE_1027 T1
	LEFT JOIN SDI.SDI_ICPOS_STORE_CONTACTS_1027 T2
	ON T1.STORE_ID = T2.STORE_ID 
	LEFT JOIN SDI.SDI_ICPOS_STORE_SEAT_1027 SEAT
	ON T1.STORE_ID = SEAT.STORE_ID 
	LEFT JOIN SDI.SDI_ICPOS_STORE_INFO_1027 INFO
	ON T1.STORE_ID = INFO.STORE_ID 
	LEFT JOIN LOOKUP_DATA T3
	ON T1.STORE_TYPE = T3.LOOKUP_CODE 
	AND T3.LOOKUP_TYPE = 'ICPOS_STORE_TYPE'
	LEFT JOIN LOOKUP_DATA T4
	ON T1.STORE_STATUS = T4.LOOKUP_CODE 
	AND T4.LOOKUP_TYPE = 'ICPOS_STORE_STATUS'
	LEFT JOIN LOOKUP_DATA T5
	ON T1.STORE_NATURE = T5.LOOKUP_CODE 
	AND T5.LOOKUP_TYPE = 'ICPOS_STORE_NATURE'
	LEFT JOIN LOOKUP_DATA T6
	ON T1.SHOP_TYPE = T6.LOOKUP_CODE 
	AND T6.LOOKUP_TYPE = 'ICPOS_SHOP_TYPE'
;
