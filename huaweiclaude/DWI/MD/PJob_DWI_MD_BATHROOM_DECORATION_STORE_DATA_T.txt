set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE  DWIMD.DWI_MD_BATHROOM_DECORATION_STORE_DATA_T
SELECT 
t.dph 						AS dph,--门店号
t.khbm						AS khbm,--经销商编码
lsh							AS lsh,--流水号
req.createdate				AS createdate,--创建日期
isc.marketing_department	AS marketing_department, ---门店所属大区
'N'                 		AS DEL_FLAG,--删除标识
current_timestamp() 		AS DW_LAST_UPDATE_DATE  --数据同步时间
from SDI.SDI_FORMTABLE_MAIN_1015_1004 t 
left join sdi.sdi_workflow_requestbase_1004 req on req.requestid = t.requestid--当前节点
left join SDI.SDI_ICPOS_STORE_1027 ist  on ist.store_code = t.dph
left join sdi.sdi_ICPOS_STORE_CONTACTS_1027 isc on ist.store_id = isc.store_id
where  req.currentnodeid not in ('55739','55738','55744');
