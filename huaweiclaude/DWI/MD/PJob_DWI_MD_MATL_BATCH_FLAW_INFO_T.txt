set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE  DWIMD.DWI_MD_MATL_BATCH_FLAW_INFO_T
SELECT 
	   A.CHARG 													AS BATCH_NUM 				--批次编号
      ,A.MATNR 													AS MATERIAL_NUM 			--物料编号			
	  ,D.ATWRT 													AS FLAW_CODE 				--缺陷编码
	  ,E.ATWTB 													AS FLAW_DESC 				--缺陷描述    
	  ,A.ERSDA 													AS START_DATE 				--开始日期 
	  ,A.LAEDA 													AS END_DATE					--结束日期
	  ,from_unixtime(unix_timestamp() ,'yyyy-MM-dd HH:mm:ss') 	AS DW_LAST_UPDATE_DATE		--同步数据时间
FROM SDI.SDI_MCH1_1001  A 
LEFT JOIN SDI.SDI_AUSP_1001 B
ON A.CUOBJ_BM = B.OBJEK
AND A.MANDT = B.MANDT 
AND B.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CABN_1001 C
ON B.ATINN = C.ATINN 
AND A.MANDT = C.MANDT 
AND C.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CAWN_1001 D
ON B.ATWRT = D.ATWRT
AND A.MANDT = D.MANDT 
AND D.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CAWNT_1001 E
ON D.MANDT = E.MANDT
AND D.ATINN = E.ATINN
AND D.ATZHL = E.ATZHL
AND E.DEL_FLAG = 'N'
AND E.SPRAS = '1'
LEFT JOIN(SELECT MATNR,CHARG,FIRSTDATE
             FROM SDI.SDI_ZSD_CXYY_1001
            WHERE FIRSTDATE <> '00000000'
              AND MANDT = '800'
          GROUP BY MATNR,CHARG,FIRSTDATE ) F
ON 	A.MATNR = F.MATNR
AND A.CHARG = F.CHARG		  
WHERE A.MANDT = 800
AND A.DEL_FLAG = 'N' AND A.CUOBJ_BM <> '000000000000000000' 
AND C.ATNAM ='Z_DEFECT';
