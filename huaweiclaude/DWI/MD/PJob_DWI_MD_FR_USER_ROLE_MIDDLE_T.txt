INSERT OVERWRITE TABLE DWIMD.DWI_MD_FR_USER_ROLE_MIDDLE_T
SELECT A.ID,
       A.ROLEID,
       A.ROLETYPE,
       A.USERID,
       A.TENANTID,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
       'N' AS DEL_FLAG,
       B.NAME AS ROLE_NAME --角色名称
  FROM SDI.SDI_FR_FINE_USER_ROLE_MIDDLE_1035 A
  LEFT JOIN SDI.SDI_FR_FINE_CUSTOM_ROLE_1035 B
    ON A.ROLEID = B.ID
