INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATL_BATCH_INFO_T_HIS_001
SELECT ROW_NUMBER() OVER() 							      AS ID									--代理主键
	  ,A.CHARG 												  AS BATCH_NUM 							--批次编号
      ,A.MATNR 												  AS MATERIAL_NUM 						--物料编号			
      ,MAX(CASE WHEN C.ATNAM = 'Z_CXREASON' THEN D.ATWRT END) AS PROMOTE_REASON_CODE 				--促销原因编码
      ,MAX(CASE WHEN C.ATNAM = 'Z_QTYPE' 	THEN D.ATWRT END) AS CLEARANCE_REASON_CODE 				--清仓原因编码
      ,MAX(CASE WHEN C.ATNAM = 'Z_CXREASON' THEN E.ATWTB END) AS PROMOTE_REASON 					--促销原因
      ,MAX(CASE WHEN C.ATNAM = 'Z_QTYPE' 	THEN E.ATWTB END) AS CLEARANCE_REASON 					--清仓原因
	  ,MAX(CASE WHEN C.ATNAM = 'Z_LENGTH' 	THEN B.ATFLV END) AS LENGTH_1 							--长度
	  ,MAX(CASE WHEN C.ATNAM = 'Z_WIDTH'  	THEN B.ATFLV END) AS WIDTH 								--宽度
	  ,MAX(CASE WHEN C.ATNAM = 'Z_THICK'  	THEN B.ATFLV END) AS THICK 								--厚度
	  ,MAX(CASE WHEN C.ATNAM = 'Z_PACKING' 	THEN B.ATFLV END) AS PACKING_PIECES 					--包装片数
	  ,MAX(CASE WHEN C.ATNAM = 'Z_STOCK' 	THEN B.ATFLV END) AS STOCK_PIECES 						--托盘片数
	  ,MAX(CASE WHEN C.ATNAM = 'Z_BATCH' 	THEN D.ATWRT END) AS SMALL_COLOR_CODE 					--小色号编码
	  ,MAX(CASE WHEN C.ATNAM = 'Z_BATCH' 	THEN E.ATWTB END) AS SMALL_COLOR_DESC					--小色号描述
	  ,from_unixtime(unix_timestamp(cast(cast(MAX(CASE WHEN C.ATNAM = 'Z_ERDAT' THEN B.ATFLV  END) as decimal) as string),'yyyyMMdd'),'yyyy-MM-dd') AS PRODUCE_DATE 						--生产日期
	  ,MAX(CASE WHEN C.ATNAM = 'Z_BTYPE' 	THEN D.ATWRT END) AS CLEARANCE_TYPE_CODE				--清仓类型编码
	  ,MAX(CASE WHEN C.ATNAM = 'Z_BTYPE' 	THEN E.ATWTB END) AS CLEARANCE_TYPE_DESC				--清仓类型描述
	  ,MAX(CASE WHEN C.ATNAM = 'Z_WEIGHT' 	THEN B.ATFLV END) AS WEIGHT 							--重量
	  ,from_unixtime(unix_timestamp(cast(cast(MAX(CASE WHEN C.ATNAM = 'Z_VFDAT' THEN B.ATFLV ELSE A.VFDAT END) as decimal) as string),'yyyyMMdd'),'yyyy-MM-dd') AS SHELF_END_DATE 					--货架寿命到期日
	  ,from_unixtime(unix_timestamp(cast(cast(MAX(CASE WHEN C.ATNAM = 'Z_LWEDT' THEN B.ATFLV END) as decimal) as string),'yyyyMMdd'),'yyyy-MM-dd') AS RECENT_RECEIPT_DATE				--最近的收货日期   
	  ,MAX(CASE WHEN C.ATNAM = 'Z_CUSTOMER' THEN B.ATFLV END) AS CUST_NUM 							--客户编号
      ,from_unixtime(unix_timestamp(coalesce(F.FIRSTDATE,MIN(G.udate)),'yyyyMMdd'),'yyyy-MM-dd') AS FIRST_PROMOTE_DATE 	--首次维护促销原因日期
	  ,from_unixtime(unix_timestamp(case when A.ERSDA = '00000000' then '19700101' else A.ERSDA end,'yyyymmdd'),'yyyy-mm-dd') AS START_DATE 	--开始日期														--开始日期
	  ,from_unixtime(unix_timestamp(case when A.LAEDA = '00000000' then '19700101' else A.LAEDA end,'yyyymmdd'),'yyyy-mm-dd') AS UPDATE_DATE	--更新日期
	  ,'4712-12-31' 										  AS END_DATE							--结束日期
	  ,'初始化数据' 										  AS DATA_TYPE							--数据类型
	  ,'N'          										  AS DEL_FLAG							--删除标识
	  ,from_unixtime(unix_timestamp() ,'yyyy-MM-dd HH:mm:ss') AS DW_LAST_UPDATE_DATE		        --同步数据时间
	  ,MAX(CASE WHEN C.ATNAM = 'Z_VOLUME' 	THEN B.ATFLV END) AS VOLUME  --体积

    FROM SDI.SDI_MCH1_1001 A
    LEFT JOIN SDI.SDI_AUSP_1001 B
      ON A.CUOBJ_BM = B.OBJEK
     AND A.MANDT = B.MANDT
     AND B.DEL_FLAG = 'N'
    LEFT JOIN SDI.SDI_CABN_1001 C
      ON B.ATINN = C.ATINN
     AND A.MANDT = C.MANDT
     AND C.DEL_FLAG = 'N'
     AND C.ATNAM IN ('Z_QTYPE',
                     'Z_CXREASON',
                     'Z_LENGTH',
                     'Z_WIDTH',
                     'Z_PACKING',
                     'Z_STOCK',
                     'Z_BATCH',
                     'Z_ERDAT',
                     'Z_BTYPE',
                     'Z_WEIGHT',
                     'Z_VFDAT',
                     'Z_LWEDT',
                     'Z_THICK',
                     'Z_CUSTOMER',
                     'Z_VOLUME'
                     )
    LEFT JOIN SDI.SDI_CAWN_1001 D
      ON B.ATWRT = D.ATWRT
     AND A.MANDT = D.MANDT
     AND D.DEL_FLAG = 'N'
    LEFT JOIN SDI.SDI_CAWNT_1001 E
      ON D.MANDT = E.MANDT
     AND D.ATINN = E.ATINN
     AND D.ATZHL = E.ATZHL
     AND E.DEL_FLAG = 'N'
     AND E.SPRAS = '1'
    LEFT JOIN (SELECT MATNR, CHARG, FIRSTDATE
                FROM SDI.SDI_ZSD_CXYY_1001
               WHERE FIRSTDATE <> '00000000'
                 AND MANDT = '800'
               GROUP BY MATNR, CHARG, FIRSTDATE) F
      ON A.MATNR = F.MATNR
     AND A.CHARG = F.CHARG
     LEFT JOIN (
        select cdpos.tabkey,min(CDHDR.udate) udate
        FROM 
        sdi.sdi_cdpos_1001 CDPOS
        INNER JOIN sdi.sdi_cdhdr_1001 CDHDR ON CDPOS.MANDANT = CDHDR.MANDANT and cdpos.objectclas = cdhdr.objectclas and cdhdr.objectid = cdpos.objectid and cdhdr.changenr = cdpos.changenr
        where cdpos.fname='ATWRT'
        and cdpos.tabname = 'ABAUSP'
        AND CDPOS.MANDANT='800'
        AND coalesce(CDPOS.value_old,'') ='' and coalesce(CDPOS.value_new,'')<>''
        group by cdpos.tabkey
     ) G ON CONCAT(B.objek,B.atinn,B.atzhl,B.mafid) = G.tabkey AND C.ATNAM='Z_CXREASON'
   WHERE A.MANDT = 800
     AND A.DEL_FLAG = 'N'
     /*
     AND A.CUOBJ_BM <> '000000000000000000'
     AND C.ATNAM IN ('Z_QTYPE',
                     'Z_CXREASON',
                     'Z_LENGTH',
                     'Z_WIDTH',
                     'Z_PACKING',
                     'Z_STOCK',
                     'Z_BATCH',
                     'Z_ERDAT',
                     'Z_BTYPE',
                     'Z_WEIGHT',
                     'Z_VFDAT',
                     'Z_LWEDT',
                     'Z_THICK',
                     'Z_CUSTOMER')
        */  -- 2023/06/09 1、取消条件 A.CUOBJ_BM <> '000000000000000000'；2、在表关联时过滤特征值C.ATNAM
   GROUP BY A.CHARG, A.MATNR, F.FIRSTDATE, A.ERSDA, A.LAEDA;


WITH BATCH_AREA AS
(SELECT BATCH_START,FACTORY_TYPE,PRODUCT_FACTORY_CODE ,BELONG_AREA_CODE ,BELONG_AREA_NAME,FINAL_AREA_NAME,SUPPLIER_OR_CREDITOR_NUM ,SUPPLIER_OR_CREDITOR_NAME ,
ROW_NUMBER () OVER(PARTITION BY BATCH_START ORDER BY CASE WHEN COALESCE(FACTORY_TYPE,' ')=' ' THEN 1 ELSE 0 END ) RN,
PRODUCT_AREA_NAME 
FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D B
),
batch_in_time as (
SELECT 
      MATNR AS MATERIAL_NUM,
      CHARG AS BATCH_NUM,
      MIN(CASE WHEN BUDAT<>'00000000' THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') END) AS EARLIEST_VOUCHER_POST_DATE, --批次最早入库日期
      MIN(CASE WHEN ((BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG <> 'K') 
                or (BSTTYP_SG = 'K' and BWART in ('411','101','991'))) and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30' 
               THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') 
               WHEN ((BWART in ('101','911','913','Z01','131','531','561','653') and BSTTYP_SG <> 'K') or (BSTTYP_SG = 'K' and BWART in ('411','101','991'))) 
               THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') END) AS EARLIEST_VOUCHER_POST_DATE_BW, ---批次最早入库日期(BW通用)
      MIN(CASE WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and WERKS = '7C20' 
               THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') END) AS earliest_voucher_post_date_fc1, --批次最早入库日期1(辅材专用)
      MIN(CASE WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and BSTTYP_SG <> 'K' and EBELN not like '47%'
               THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') END) AS earliest_voucher_post_date_fc3 --批次最早入库日期3(辅材专用)      
  FROM SDI.SDI_MATDOC_1001
 GROUP BY
      MATNR,
      CHARG
)
,mat0 AS (
SELECT a.matnr as material_num,a.charg as batch_num,a.EBELN,a.EBELP,c.bpkind
,ROW_NUMBER() OVER(partition by a.matnr,a.charg ORDER BY a.matnr,a.charg DESC) as RN 
FROM SDI.SDI_MATDOC_1001 a
INNER JOIN sdi.sdi_ekko_1001 b
ON a.EBELN = b.EBELN 
left join sdi.sdi_but000_1001 c
on a.LIFNR = c.partner and c.CLIENT = '800'
WHERE (( BSTTYP_SG <> 'K' AND BWART = '101' ) OR ( BSTTYP_SG = 'K' AND BWART IN ('101','411')))
AND a.EBELN <> '' AND ( a.EBELN LIKE '5%' OR a.EBELN LIKE '49%') AND b.BSART NOT IN ('ZSTO','ZSTR','ZSUB')
)
,mat2 as (
select MATNR AS MATERIAL_NUM
,min(CASE WHEN BUDAT<>'00000000' THEN from_unixtime(unix_timestamp(BUDAT,'yyyyMMdd'),'yyyy-MM-dd') END) earliest_voucher_post_date_fc2 --批次最早入库日期2(辅材专用)
from SDI.SDI_MATDOC_1001
where BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and WERKS = '7C20'
group by MATNR
)
INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATL_BATCH_INFO_T_HIS
SELECT T1.ID,
       T1.BATCH_NUM,
       T1.MATERIAL_NUM,
       T1.PROMOTE_REASON_CODE,
       T1.CLEARANCE_REASON_CODE,
       T1.PROMOTE_REASON,
       T1.CLEARANCE_REASON,
       T1.LENGTH_1,
       T1.WIDTH,
       T1.THICK,
       T1.PACKING_PIECES,
       T1.STOCK_PIECES,
       T1.SMALL_COLOR_CODE,
       T1.SMALL_COLOR_DESC,
       T1.PRODUCE_DATE,
       T1.CLEARANCE_TYPE_CODE,
       T1.CLEARANCE_TYPE_DESC,
       T1.WEIGHT,
       T1.SHELF_END_DATE,
       T1.RECENT_RECEIPT_DATE,
       T1.CUST_NUM,
       coalesce(T1.FIRST_PROMOTE_DATE,T5.FIRST_PROMOTE_DATE),    -- 补充手动导入的首次促销时间
       T1.START_DATE,
       T1.UPDATE_DATE,
       T1.END_DATE,
       T1.DATA_TYPE,
       T1.DEL_FLAG,
       T1.DW_LAST_UPDATE_DATE,
       COALESCE(T2.BELONG_AREA_CODE, T3.BELONG_AREA_CODE) AS PRODUCT_AREA_CODE,		-- 产区编码
       COALESCE(T2.BELONG_AREA_NAME, T3.BELONG_AREA_NAME) AS PRODUCT_AREA_NAME,		-- 产区名称		
       COALESCE(T2.FINAL_AREA_NAME, T3.FINAL_AREA_NAME)   AS FIVE_PRODUCT_AREA_NAME,	-- 产区名称（五大产区）
       COALESCE(T1.PRODUCE_DATE,T6.EARLIEST_VOUCHER_POST_DATE)  AS BATCH_AMB_IN_TIME,   --  bw字段逻辑作废，此字段作废
       T6.EARLIEST_VOUCHER_POST_DATE,
       COALESCE(T1.PRODUCE_DATE,T6.EARLIEST_VOUCHER_POST_DATE) AS BATCH_IN_TIME,
       T1.VOLUME, -- 体积
       COALESCE(T6.EARLIEST_VOUCHER_POST_DATE_BW,T1.PRODUCE_DATE) AS earliest_voucher_post_date_bw ,  ----最早入库时间(BW) 
       COALESCE(T6.earliest_voucher_post_date_fc1,mat2.earliest_voucher_post_date_fc2,T6.earliest_voucher_post_date_fc3,T1.PRODUCE_DATE) AS earliest_voucher_post_date_fc,   --最早入库时间(辅材专用)
       CASE WHEN T6.earliest_voucher_post_date_bw >= '2021-09-01' AND mat0.EBELN <> '' AND mat0.bpkind <> '98' AND ( T.external_matl_group_code = '10080' OR ( T.external_matl_group_code = '10010' AND T.matl_group_code = '1000010')) THEN SUBSTR(T1.BATCH_NUM,1,3)
	   WHEN T6.earliest_voucher_post_date_bw >= '2022-01-01' AND mat0.EBELN <> '' AND mat0.bpkind <> '98' AND T.external_matl_group_code = '10060' THEN SUBSTR(T1.BATCH_NUM,1,3) 
	   ELSE SUBSTR(T1.BATCH_NUM,1,2) END AS product_area_code_bw, --"产区" 
	   COALESCE(T2.BATCH_START, T3.BATCH_START) AS BATCH_START_CZ,		-- 瓷砖批次首两位
	   COALESCE(T2.FACTORY_TYPE, T3.FACTORY_TYPE) AS FACTORY_TYPE,	
	   COALESCE(T2.SUPPLIER_OR_CREDITOR_NUM, T3.SUPPLIER_OR_CREDITOR_NUM) AS SUPPLIER_OR_CREDITOR_NUM,
	   COALESCE(T2.SUPPLIER_OR_CREDITOR_NAME, T3.SUPPLIER_OR_CREDITOR_NAME) AS SUPPLIER_OR_CREDITOR_NAME
  FROM DWIMD.DWI_MD_MATL_BATCH_INFO_T_HIS_001 T1
  LEFT JOIN dwimd.dwi_md_data_material_general_t T 
  ON T1.MATERIAL_NUM = T.MATERIAL_NUM
  LEFT JOIN BATCH_AREA T2
    ON SUBSTR(T1.BATCH_NUM, 1, 3) = T2.BATCH_START AND T2.RN = 1
  LEFT JOIN BATCH_AREA T3
    ON SUBSTR(T1.BATCH_NUM, 1, 2) = T3.BATCH_START AND T3.RN = 1
  LEFT JOIN sdi.sdi_upload_md_material_batch_first_prom T5 ON T1.MATERIAL_NUM = T5.material_num and coalesce(T1.BATCH_NUM,'#') = coalesce(T5.BATCH_NUM,'#')
  LEFT JOIN batch_in_time T6 ON T1.material_num = T6.material_num AND  coalesce(T1.BATCH_NUM,'#*#') = coalesce(T6.BATCH_NUM,'#*#')
  left join mat2
  on mat2.material_num = T1.material_num
  left join mat0
  on mat0.material_num = T1.material_num and mat0.BATCH_NUM = T1.BATCH_NUM AND mat0.RN = 1;
