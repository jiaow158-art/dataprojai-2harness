with prod_line_data as(
	select t1.code,t1.inv_org_id ,t1.name
	from SDI.SDI_BD_CATALOG_1007 t1
	left join SDI.SDI_BD_CATALOG_TYPE_1007 t2
	on t1.CATALOG_TYPE_ID = t2.id
	where t2.CODE = 'PRODUCT_LINE'
),
prod_level_data as(
SELECT T3.name as prod_line_name,    
       T1.second_level_type AS second_level_name,
       t2.prodh as prod_level_code,
       t2.vtext as prod_level_name
FROM SDI.SDI_WYAPS_PRODUCT_SORT_MAINTAIN_1007 T1
left join SDI.SDI_WYAPS_PRODUCT_LEVEL_1007 T2
ON T1.product_level_id = T2.ID
AND t1.inv_org_id = t2.inv_org_id
left join prod_line_data t3
on t1.inv_org_id = t3.inv_org_id
and t1.product_line = t3.code
where t1.is_phantom <> '1'
)
INSERT OVERWRITE TABLE DWIMD.DWI_MD_PRODUCT_LEVEL_APS_T
select t.*,
	   current_timestamp() as dw_last_update_date 
from prod_level_data t;
