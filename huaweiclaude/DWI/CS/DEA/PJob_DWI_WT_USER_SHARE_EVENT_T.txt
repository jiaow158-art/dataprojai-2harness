INSERT OVERWRITE TABLE DWICS.DWI_WT_USER_SHARE_EVENT_T
SELECT
       CAST(ID AS DECIMAL) AS ID
      ,CAST(STORE_ID AS DECIMAL) AS STORE_ID
      ,STORE_NAME AS SRC_STORE_NAME
      ,CAST(CLIENT_ID AS DECIMAL) AS USER_ID
      ,EVENT AS EVENT_TYPE
      ,ITEM_NO
      ,ITEM_NAME AS SRC_ITEM_NAME
      ,EVENT_TIME
      ,STAT_DATE
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_ADS_WXAPP_DP_PRODUCT_SHARE_EVENT_ANALYSIS_D_1030
 WHERE EP_ID = 3044
   AND APP_ID = 1831
