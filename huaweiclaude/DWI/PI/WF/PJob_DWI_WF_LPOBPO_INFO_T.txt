with t1 as(
SELECT CASE WHEN A.LC = 0 THEN '01.IPD产品开发'
            WHEN A.LC = 1 THEN '02.MTL市场开发'
            WHEN A.LC = 2 THEN '03.RTM渠道'
            WHEN A.LC = 3 THEN '04.LTC工程'
            WHEN A.LC = 4 THEN '05.DTC零售'
            WHEN A.LC = 5 THEN '06.ITR客服'
            WHEN A.LC = 6 THEN '07.OTD订单履约'
            WHEN A.LC = 7 THEN '08.MFG制造交付'
            WHEN A.LC = 8 THEN '09.PTP采购'
            WHEN A.LC = 9 THEN '10.STE战略'
            WHEN A.LC = 10 THEN '11.MCI投融资'
            WHEN A.LC = 11 THEN '12.BTF财经'
            WHEN A.LC = 12 THEN '13.HCM人资'
            WHEN A.LC = 13 THEN '14.BIT数字化'
            WHEN A.LC = 14 THEN '15.MBS行政'
            WHEN A.LC = 15 THEN '16.EPC工程项目管'
            WHEN A.LC = 16 THEN '17.SDE服务交付'
            WHEN A.LC = 16 THEN '总体负责'
             end as lcmc
         ,case when SYB = 0 THEN '瓷砖'  
               when SYB = 1 THEN '卫浴'
               when SYB = 2 THEN '新材'
               when SYB = 3 THEN '家居事业部'
               when SYB = 4 THEN '平台'
                END AS syb_name
         ,NEW_lpo
         ,NEW_bpo
         ,a.* 
FROM sdi.sdi_uf_lcpohz_1004 a 
lateral view explode(split(lpo,',')) myTable1 as NEW_lpo
lateral view explode(split(bpo,',')) myTable2 as NEW_bpo
)
insert overwrite table dwipi.dwi_wf_lpobpo_info_t 
select b.sapid as lpo_emp_num
,b.lastname as lpo_emp_name
,c.sapid as bpo_emp_num
,c.lastname as bpo_emp_name
,t1.new_lpo
,t1.new_bpo
,t1.id
,t1.requestid
,t1.lc
,t1.lcmc
,t1.ywzg
,t1.lpo
,t1.bpo
,t1.cwdj
,t1.formmodeid
,t1.modedatacreater
,t1.modedatacreatertype
,t1.modedatacreatedate
,t1.modedatacreatetime
,t1.modedatamodifier
,t1.modedatamodifydatetime
,t1.modeuuid
,t1.syb
,t1.syb_name
,t1.lpozdrnew
,current_timestamp as DW_LAST_UPDATE_DATE
from t1 
left join sdi.sdi_hrmresource_1004 b
on t1.new_lpo = b.id
left join sdi.sdi_hrmresource_1004 c
on t1.new_bpo = c.id
;
