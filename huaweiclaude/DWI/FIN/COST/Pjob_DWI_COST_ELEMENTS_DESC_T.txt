INSERT OVERWRITE TABLE DWIFIN.DWI_COST_ELEMENTS_DESC_T
--保留DWI存在而SDI不存在的数据，删除标识置为Y
SELECT
   T1.LANG_CODE
  ,T1.GROUP_NUM
  ,T1.COST_ELEMENTS
  ,T1.ACCOUNTS_TABLE
  ,T1.COST_CENTER_DESC
  ,T1.GENERAL_NAME
  ,T1.SEARCH_TERM
  ,'Y' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_ELEMENTS_DESC_T T1
LEFT JOIN SDI.SDI_CSKU_1001 T2
ON T2.SPRAS = T1.LANG_CODE
AND T2.MANDT = T1.GROUP_NUM
AND T2.KSTAR = T1.COST_ELEMENTS
AND T2.KTOPL = T1.ACCOUNTS_TABLE
WHERE T2.SPRAS IS NULL

UNION ALL

--直接插入SDI存在而DWI不存在的数据，删除标识置为N
SELECT 
   T1.SPRAS AS LANG_CODE
  ,T1.MANDT AS GROUP_NUM
  ,T1.KSTAR AS COST_ELEMENTS
  ,T1.KTOPL AS ACCOUNTS_TABLE
  ,T1.LTEXT AS COST_CENTER_DESC
  ,T1.KTEXT AS GENERAL_NAME
  ,T1.MCTXT AS SEARCH_TERM
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_CSKU_1001 T1
;
