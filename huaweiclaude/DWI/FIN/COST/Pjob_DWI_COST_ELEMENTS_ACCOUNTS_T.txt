--成本要素 (数据与科目表相关)
--建临时表
WITH DWI_COST_ELEMENTS_ACCOUNTS_T_TMP AS(
--直接插入SDI的数据，删除标识置为N
SELECT
   0 AS SEQ_ID
  ,T1.KTOPL AS ACCOUNTS_TABLE
  ,T1.KSTAR AS COST_ELEMENTS2
  ,T1.MANDT AS GROUP_NUM
  ,T1.KSTSN AS COST_ELEMENTS1
  ,from_unixtime(unix_timestamp(T1.ERSDA,'yyyymmdd'),'yyyy-mm-dd') AS CREATE_DATE
  ,T1.USNAM AS INPUT_BY
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_CSKA_1001 T1
)

INSERT OVERWRITE TABLE DWIFIN.DWI_COST_ELEMENTS_ACCOUNTS_T
--保留DWI存在而SDI不存在的数据，删除标识置为Y
SELECT
   T1.SEQ_ID
  ,T1.ACCOUNTS_TABLE
  ,T1.COST_ELEMENTS2
  ,T1.GROUP_NUM
  ,T1.COST_ELEMENTS1
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,'Y' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_ELEMENTS_ACCOUNTS_T T1
LEFT JOIN DWI_COST_ELEMENTS_ACCOUNTS_T_TMP T2
ON T2.ACCOUNTS_TABLE = T1.ACCOUNTS_TABLE
AND T2.COST_ELEMENTS2 = T1.COST_ELEMENTS2
AND T2.GROUP_NUM = T1.GROUP_NUM
WHERE T2.ACCOUNTS_TABLE IS NULL

UNION ALL

--保留DWI存在且SDI存在的数据，删除标识置为N
SELECT 
   T1.SEQ_ID
  ,T1.ACCOUNTS_TABLE
  ,T1.COST_ELEMENTS2
  ,T1.GROUP_NUM
  ,T1.COST_ELEMENTS1
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,'N' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_ELEMENTS_ACCOUNTS_T T1
LEFT JOIN DWI_COST_ELEMENTS_ACCOUNTS_T_TMP T2
ON T2.ACCOUNTS_TABLE = T1.ACCOUNTS_TABLE
AND T2.COST_ELEMENTS2 = T1.COST_ELEMENTS2
AND T2.GROUP_NUM = T1.GROUP_NUM
WHERE T2.ACCOUNTS_TABLE IS NOT NULL

UNION ALL
--直接插入SDI存在而DWI不存在的数据，删除标识置为N，临时表的SEQ_ID加上DWI的最大SEQ_ID
SELECT
   T1.SEQ_ID + NVL(T2.SEQ_ID,0) AS SEQ_ID
  ,T1.ACCOUNTS_TABLE
  ,T1.COST_ELEMENTS2
  ,T1.GROUP_NUM
  ,T1.COST_ELEMENTS1
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM(SELECT
       ROW_NUMBER() OVER() AS SEQ_ID
	  ,T1.ACCOUNTS_TABLE
      ,T1.COST_ELEMENTS2
      ,T1.GROUP_NUM
      ,T1.COST_ELEMENTS1
      ,T1.CREATE_DATE
      ,T1.INPUT_BY
     FROM DWI_COST_ELEMENTS_ACCOUNTS_T_TMP T1
     LEFT JOIN DWIFIN.DWI_COST_ELEMENTS_ACCOUNTS_T T2
     ON T2.ACCOUNTS_TABLE = T1.ACCOUNTS_TABLE
     AND T2.COST_ELEMENTS2 = T1.COST_ELEMENTS2
     AND T2.GROUP_NUM = T1.GROUP_NUM
	 WHERE T2.ACCOUNTS_TABLE IS NULL) T1
LEFT JOIN (SELECT MAX(SEQ_ID) AS SEQ_ID FROM DWIFIN.DWI_COST_ELEMENTS_ACCOUNTS_T T1) AS T2
ON 1 = 1
;
