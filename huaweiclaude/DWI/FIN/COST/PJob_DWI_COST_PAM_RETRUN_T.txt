set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with bz_cost_qty_data as(
	SELECT t1.kalnr,  -- 没有数量结构的成本估算的成本估算编号
	       t1.jahrper, -- 过账年份/期间 YYYYPPP
	       t2.matnr,  -- 物料编码
	       substring(t1.jahrper,5,3) as month1,  -- 期间月份
	       t2.bwkey,  -- 估价范围 工厂
	       sum(t1.quant) as quant  -- 以评估数量计量单位计的交易数量
	FROM STG.stg_mldoc_1001 t1
	join STG.stg_mbew_1001 t2
	on t1.kalnr = t2.kaln1 
	AND t1.CATEG = 'ZU'
	AND t1.PTYP  = 'BF'
	where concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) >= '${PERIOD_ID_M}' -- 传昨天所在的月份
	group by t1.kalnr,
	         t1.jahrper,
	         t2.matnr,
	         t2.bwkey,
	         substring(t1.jahrper,5,3)
),
bz_cost_data_01 as(
	select  t2.*,
	        case when t1.peinh = 0 then 1 else t1.peinh end peinh,  -- 价格单位
	        t3.kst001 as cl_ns_bz , -- 材料 - 沙泥  
	        t3.kst003 as cl_hg_bz , -- 材料 - 化工  
	        t3.KST005 as cl_pj_bz , -- 材料－配件
			t3.KST007 as cl_bz_bz , -- 材料－包装
			t3.KST009 as cl_qtcl_bz , -- 材料－其他材料
			t3.KST011 as bcp_bz , -- 半成品
			t3.KST013 as ccp_bz, -- 产成品
			t3.KST015 as rgf_bz, -- 人工费
			t3.KST027 as lq_m_bz, -- 燃料-煤
			t3.KST029 as lq_q_bz, -- 燃料-气
			t3.KST019 as df_bz, -- 电费
			t3.KST021 as zj_tx_bz, -- 折旧及摊销
			t3.KST023 as wwjg_pgf_bz, -- 委外加工/抛光费
			t3.KST017 as sf_bz, -- 水费
			t3.KST025 as wlxh_bz, -- 物料消耗
			t3.KST031 as wxf_bz, -- 维修费
			t3.KST033 as cy_bz, -- 柴油
			t3.KST035 as wscl_bz, -- 污水处理
			t3.KST037 as qtfy_bz -- 其他费用
	from STG.stg_mbew_1001 t1
	left join bz_cost_qty_data t2
	on t1.matnr = t2.matnr
	and t1.bwkey = t2.bwkey
	and t1.kaln1 = t2.kalnr
	left join STG.STG_CKMLPRKEPH_1001 t3
	on t1.kaln1 = t3.kalnr
	and substring(t2.jahrper,1,4) = t3.bdatj
	and t2.month1 = t3.poper
	and t3.prtyp = 'S'
	and t3.keart = 'N'
	and t3.kkzst = ''
),
bz_cost_data_02 as(
select t1.kalnr as cost_code, -- 成本编码
       t1.bwkey as factory_code, -- 工厂编码
       t1.matnr as material_num , -- 物料编码
       t1.jahrper as period_data, -- 期间
       '标准' as cost_type, -- 类型
       t1.quant as cost_qty , -- 成本数量
       t1.cl_ns_bz * t1.quant / t1.peinh as bz_cl_ns_amt , -- 材料 - 沙泥  
       t1.cl_hg_bz * t1.quant / t1.peinh as bz_cl_hg_amt , -- 材料 - 化工  
       t1.cl_pj_bz * t1.quant / t1.peinh as bz_cl_pj_amt, -- 材料－配件
       t1.cl_bz_bz * t1.quant / t1.peinh as bz_cl_bz_amt,  -- 材料 -- 包装
       t1.cl_qtcl_bz * t1.quant / t1.peinh as bz_cl_qtcl_amt,  -- -- 材料－其他材料
       t1.bcp_bz * t1.quant / t1.peinh as bz_bcp_amt,  -- 半成品
       t1.ccp_bz * t1.quant / t1.peinh as bz_ccp_amt,  -- 产成品
       t1.rgf_bz * t1.quant / t1.peinh as bz_rgf_amt,  -- 人工费
       t1.lq_m_bz * t1.quant / t1.peinh as bz_lq_m_amt,   -- 燃料-煤
       t1.lq_q_bz * t1.quant / t1.peinh as bz_lq_q_amt,   -- 燃料-气
       t1.df_bz * t1.quant / t1.peinh as bz_df_amt,  -- 电费
       t1.zj_tx_bz * t1.quant / t1.peinh as bz_zj_tx_amt,  -- 折旧及摊销
       t1.wwjg_pgf_bz * t1.quant / t1.peinh as bz_wwjg_pgf_amt, -- 委外加工/抛光费
       t1.sf_bz * t1.quant / t1.peinh as bz_sf_amt,  -- 水费
       t1.wlxh_bz * t1.quant / t1.peinh as bz_wlxh_amt,  -- 物料消耗
       t1.wxf_bz * t1.quant / t1.peinh as bz_wxf_amt, -- 维修费
       t1.cy_bz * t1.quant / t1.peinh as bz_cy_amt,   -- 柴油
       t1.wscl_bz * t1.quant / t1.peinh as bz_wscl_amt, -- 污水处理
       t1.qtfy_bz * t1.quant / t1.peinh as bz_qtfy_amt   -- 其他费用
from bz_cost_data_01 t1
),
-- 实际成本
dim_data_01 as(
	select distinct t1.kalnr,
	                t1.jahrper,
	                t2.matnr,
	                t2.bwkey
	from STG.stg_mldoc_1001 t1
	join STG.stg_mbew_1001 t2
	on t1.kalnr = t2.kaln1 
	--where substring(t1.jahrper,1,4) =  substring('${PERIOD_ID_M}',1,4)
	where concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) >= '${PERIOD_ID_M}' -- 传昨天所在的月份
),
dim_dim_data_01 as(
	SELECT t.txele,t.elemt
	FROM STG.stg_tckh1_1001 T 
	WHERE T.spras = '1' 
	AND T.elehk = 'M1'
	AND T.elemt in ('100','110','120','130','190','200','300','410',
	                 '421','422','430','440','450','461','462','463','464','465','466')
),
dim_dim_data_02 as(
	select t1.kalnr,
	       t1.jahrper,
	       t1.matnr,
	       t1.bwkey,
	       t2.elemt,
	       t2.txele
	from dim_data_01 t1
	left join dim_dim_data_01 t2
	on 1 = 1
),
sj_tmp_data as(
	SELECT t1.quant ,
	       t2.elesmhk,
	       t2.element,
	       t1.kalnr,
	       t1.jahrper,
	       t2.tot
	FROM STG.stg_mldoc_1001 t1
	left join stg.stg_mldocccs_1001 t2
	on t1.docref = t2.docref 
	WHERE 1 = 1
	and t2.curtp = 10
	AND t1.CATEG = 'ZU'
	AND t1.PTYP  = 'BF'
	and t2.elesmhk = 'M1'
	and concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) >= '${PERIOD_ID_M}' -- 传昨天所在的月份
),
sj_tmp_data_02 as(
	select --t1.*,t2.txele
	      t2.kalnr,
	      t2.jahrper,
	      t2.matnr,
	      t2.bwkey,
	      t2.txele,
	      nvl(sum(t1.tot),0) as tot
	from dim_dim_data_02 t2
	left JOIN sj_tmp_data t1
	ON t1.kalnr =  t2.kalnr
	and t1.jahrper = t2.jahrper
	and t1.element = t2.elemt
	group by t2.kalnr,
	         t2.jahrper,
	         t2.matnr,
	         t2.bwkey,
	         t2.txele
),
sj_tmp_data_03 as(
select kalnr,
       jahrper,
       matnr,
       bwkey,
       sum(case when txele = '材料－沙泥' then tot else 0 end) sj_cl_ns_amt,
       sum(case when txele = '材料－化工' then tot else 0 end) sj_cl_hg_amt,
       sum(case when txele = '材料－配件' then tot else 0 end) sj_cl_pj_amt,
       sum(case when txele = '材料－包装' then tot else 0 end) sj_cl_bz_amt,
       sum(case when txele = '材料－其他材料' then tot else 0  end) sj_cl_qtcl_amt,
       sum(case when txele = '半成品' then tot else 0 end) sj_bcp_amt,
       sum(case when txele = '产成品' then tot else 0 end) sj_ccp_amt,
       sum(case when txele = '人工费' then tot else 0 end) sj_rgf_amt,
       sum(case when txele = '燃料-煤' then tot else 0 end) sj_rl_m_amt,
       sum(case when txele = '燃料-气' then tot else 0 end) sj_rl_q_amt,
       sum(case when txele = '电费' then tot else 0 end) sj_df_amt,
       sum(case when txele = '折旧及摊销' then tot else 0 end) sj_zjtx_amt,
       sum(case when txele = '委外加工/抛光费' then tot else 0 end) sj_wyjg_amt,
       sum(case when txele = '水费' then tot else 0 end) sj_sf_amt,
       sum(case when txele = '物料消耗' then tot else 0 end) sj_wlxh_amt,
       sum(case when txele = '维修费' then tot else 0 end) sj_wyf_amt,
       sum(case when txele = '柴油' then tot else 0 end) sj_cy_amt,
       sum(case when txele = '污水处理' then tot else 0 end) sj_wscl_amt,
       sum(case when txele = '其他费用' then tot else 0 end) sj_qtfy_amt,
       sum(case when txele in('材料－沙泥','材料－化工','材料－配件','材料－包装','材料－其他材料'
                     ,'半成品','产成品','人工费','燃料-煤','燃料-气','电费','折旧及摊销'
                     ,'委外加工/抛光费','水费','物料消耗','维修费','柴油','污水处理','其他费用') 
           then tot else 0 end) as sj_total_amt
from sj_tmp_data_02 t
group by kalnr,
         jahrper,
         matnr,
         bwkey
),
sj_cost_qty_data as(
	SELECT t1.kalnr,  -- 没有数量结构的成本估算的成本估算编号
	       t1.jahrper, -- 过账年份/期间 YYYYPPP
	       t2.matnr,  -- 物料编码
	       t2.bwkey,  -- 估价范围 工厂
	       sum(t1.quant) as quant  -- 以评估数量计量单位计的交易数量
	FROM STG.stg_mldoc_1001 t1
	join STG.stg_mbew_1001 t2
	on t1.kalnr = t2.kaln1 
	AND t1.CATEG = 'ZU'
	AND t1.PTYP  = 'BF'
	group by t1.kalnr,
	         t1.jahrper,
	         t2.matnr,
	         t2.bwkey
)
