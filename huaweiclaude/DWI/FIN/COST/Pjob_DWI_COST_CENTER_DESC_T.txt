INSERT OVERWRITE TABLE DWIFIN.DWI_COST_CENTER_DESC_T
--保留DWI存在而SDI不存在的数据，删除标识置为Y
SELECT
   T1.COST_CENTER_DESC
  ,T1.GENERAL_NAME
  ,T1.COST_CENTER
  ,T1.LANG_CODE
  ,T1.END_EFFECT_DATE
  ,T1.SEARCH_TERM
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,'Y' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_CENTER_DESC_T T1
LEFT JOIN SDI.SDI_CSKT_1001 T2
ON T2.KOSTL = T1.COST_CENTER
AND T2.SPRAS = T1.LANG_CODE
AND CAST(from_unixtime(unix_timestamp(CASE WHEN T2.DATBI = '00000000' THEN '19000101'
							          WHEN T2.DATBI = '99991231' THEN '47121231'
                                      ELSE T2.DATBI
                                       END,'yyyymmdd'),'yyyy-mm-dd') AS TIMESTAMP) = T1.END_EFFECT_DATE
AND T2.MANDT = T1.GROUP_NUM
AND T2.KOKRS = T1.CONTROL_SCOPE
WHERE T2.KOSTL IS NULL

UNION ALL

--直接插入SDI的数据，删除标识置为N
SELECT 
   T1.LTEXT AS COST_CENTER_DESC
  ,T1.KTEXT AS GENERAL_NAME
  ,T1.KOSTL AS COST_CENTER
  ,T1.SPRAS AS LANG_CODE
  ,from_unixtime(unix_timestamp(CASE WHEN T1.DATBI = '00000000' THEN '19000101'
							         WHEN T1.DATBI = '99991231' THEN '47121231'
                                     ELSE T1.DATBI
                                      END,'yyyymmdd'),'yyyy-mm-dd') AS END_EFFECT_DATE
  ,T1.MCTXT AS SEARCH_TERM
  ,T1.MANDT AS GROUP_NUM
  ,T1.KOKRS AS CONTROL_SCOPE
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_CSKT_1001 T1
;
