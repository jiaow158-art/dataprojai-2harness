INSERT
  OVERWRITE TABLE DWIFIN.DWI_PRC_CONDITIONAL_PRICE_T PARTITION(PERIOD_ID_M)
SELECT
  SPE.CLIENT AS GROUP_NUM,
  SPE.KNUMV AS VCH_TERM_NUM,
  SPE.KPOSN AS CONDITION_ITEM_NUM,
  SPE.STUNR AS STEP_NUM,
  SPE.ZAEHK AS CONDITION_COUNTER,
  SPE.ZAEKO AS CONDITION_COUNTER_RISE,
  SPE.KBETR AS CONDITION_AMT,
  SPE.KPEIN AS CONDITION_PRICE_UNIT,
  SPE.KSCHL AS CONDITION_TYPE,
  SPE.KNTYP AS CONDITION_CATEGORY,
  SPE.KOAID AS CONDITION_CLASS,
  SPE.KINAK AS CONDITION_ACTIVE_FLAG,
  SPE.KWERT AS CONDITION_VALUE,
  SPE.KAWRT AS CONDITION_BENCHMARK_VALUE,
  SPE.KNUMH AS BATCH_CONFIRM_TERM_NUM,
  SPE.KWERT_K AS CONDITION_VALUE_K,
  SPE.KSTBS AS LEVEL_BENCHMARK_VALUE,
  SPE.KDIFF AS CONDITION_ROUND_DIFFER,
  CASE
    WHEN SPE.KDATU = '00000000000000'
    OR SPE.KDATU = '' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss')
    )
    WHEN SUBSTR(SPE.KDATU, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231000000', 'yyyyMMddhhmmss')
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SPE.KDATU, 'yyyyMMddhhmmss'))
  END AS PRICING_DATE,
  SPE.LIFNR AS SUPPLIE_ACCOUNT_NUM,
  SPE.KKURS AS CURRENCY_CONDITION_RATE,
  SPE.KMEIN AS CONDITION_UNIT,
  SPE.SAKN1 AS GL_ACCOUNT_NUM,
  SPE.MWSK1 AS TAX_CODE,
  SPE.KVSL1 AS ACCOUNT_CODE,
  SPE.KVSL2 AS ACCOUNT_ACCRUED_CODE,
  SPE.KOLNR AS ACCESS_SEQUENCE,
  SPE.KRECH AS CONDITION_CALC_TYPE,
  SPE.WAERS AS CURRENCY_CODE,
  SPE.WAERK AS VOUCHER_CURRENCY_CODE,
  SPE.KSTEU AS PRICING_CONTROL,
  SPE.KUMZA AS CONVERSION_MOLECULE,
  SPE.KHERK AS CONDITION_SOURCE,
  SPE.KAPPL AS APPLICATION_PROGRAM_CODE,
  SPE.KOPOS AS CONDITION_SERIAL_NUM,
  SPE.KUMNE AS CONVERSION_DENOMINATOR,
  SPE.KONMS AS CONDITION_LEVEL_MEASURE_UNIT,
  SPE.KZBZG AS BASIC_LEVEL_FALG,
  SPE.KSTAT AS CONDITION_STATISTICS_FLAG,
  SPE.KWAEH AS CONDITION_CURRENCY_CODE,
  SPE.KVARC AS CONFIG_CONDITION_FLAG,
  SPE.KRUEK AS PRICING_ACCRUED_FLAG,
  SPE.KMPRS AS MANUAL_FLAG,
  SPE.MDFLG AS MATRIX_MAINTENANCE_FALG,
  SPE.KGRPE AS GROUP_CONDITION_FLAG,
  SPE.PRSQU AS PRICING_SOURCE,
  'N' AS DEL_FLAG,
  SPE.DW_LAST_UPDATE_DATE,
  SPE.PERIOD_ID_M
FROM
  SDI.SDI_PRCD_ELEMENTS_1001 SPE
;
