INSERT overwrite table DWIFIN.DWI_GL_DRAFT_VOUCHER_T
SELECT 
T1.MANDT       AS GROUP_NUM,
T1.BUZEI       AS ACC_VOUCHER_LINE_ITEM_QTY,
T1.BELNR       AS ACCOUNT_VOUCHER_NUM,
COM.COMPANY_ID AS COMPANY_ID,
T1.BUKRS       AS COMPANY_CODE,
T1.GJAHR       AS FISCAL_YEAR,
T1.REGIO       AS REGION_CODE,
T1.XAKTZ       AS DRAFT_ACCEPT_FLAG,
T1.XSIWE       AS DRAFT_ON_DEMAND_FLAG,
T1.WNAME       AS DRAFT_DRAWER_NAME,
T1.WORT1       AS DRAFT_DRAWER_CITY_NAME,
T1.WBZOG       AS DRAFT_DRAWEE_NAME,
T1.WORT2       AS DRAFT_DRAWEE_CITY_NAME,
T1.WEVWV       AS DRAFT_PLAN,
from_unixtime(unix_timestamp(T1.WDATE,'yyyyMMdd'),'yyyy-MM-dd')  AS DRAFT_SIGN_DATE,
T1.WSTAT       AS DRAFT_TYPE,
T1.WBANK       AS DRAFT_LONG_NUM,
T1.WLZBP       AS BANK_NAME,
T1.DEL_FLAG,
T1.DW_LAST_UPDATE_DATE
FROM SDI.SDI_BSED_1001 T1 
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM 
WHERE T1.BUKRS  = COM.COMPANY_CODE 
AND  COM.DEL_FLAG  = 'N' 
AND  COM.END_DATE  = '4712-12-31';
