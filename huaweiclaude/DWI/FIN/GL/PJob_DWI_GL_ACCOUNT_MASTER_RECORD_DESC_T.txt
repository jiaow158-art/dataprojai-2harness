--GROUP_NUM					STRING	COMMENT '集团',
--LANGUAGE_CODE				STRING	COMMENT '语言代码',
--ACCOUNT_FORM				STRING	COMMENT '帐目表',
--GENERAL_LEDGER_ACCOUNT_CODE	STRING	COMMENT '总账科目编号',
--GENERAL_LEDGER_ACCOUNT_NAME	STRING	COMMENT '总帐科目名称',
--GENERAL_LEDGER_ACCOUNT_TEXT	STRING	COMMENT '总账科目长文本',
--TIMESPAN					TIMESTAMP	COMMENT '简短格式的 UTC 时戳 (YYYYMMDDhhmmss)',
--SEARCH_TERM	STRING	COMMENT '匹配码搜索的搜索条件',
--DEL_FLAG							STRING  COMMENT '删除表示', 	
--DW_LAST_UPDATE_DATE TIMESTAMP   			COMMENT 'DW最后更新日期',
--START_DATE DATE COMMENT '开始日期',
--END_DATE   DATE COMMENT '结束日期'
--)


with sdi_tmp as (

--insert overwrite table  dwifin.DWI_GL_ACCOUNT_MASTER_RECORD_DESC_T 
SELECT 
 MANDT GROUP_NUM					
,SPRAS LANGUAGE_CODE				
,KTOPL ACCOUNT_FORM				    
,SAKNR GENERAL_LEDGER_ACCOUNT_CODE	
,TXT20 GENERAL_LEDGER_ACCOUNT_NAME	
,TXT50 GENERAL_LEDGER_ACCOUNT_TEXT	
,FROM_UNIXTIME(CASE WHEN LAST_CHANGED_TS ='0' THEN '19000101' ELSE LAST_CHANGED_TS  END ,'yyyy-MM-dd hh:mm:ss')  TIMESPAN 
,MCOD1 SEARCH_TERM					
,'N' DEL_FLAG							
,CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE    	
,SUBSTR(CURRENT_TIMESTAMP(),1,10) START_DATE
,'4712-12-31' END_DATE 
,concat(MANDT, SPRAS,KTOPL, SAKNR, TXT20, TXT50 ,MCOD1) join_key
FROM sdi.SDI_SKAT_1001
)

,dwi_tmp as (
SELECT 
 GROUP_NUM					
,LANGUAGE_CODE				
,ACCOUNT_FORM				    
,GENERAL_LEDGER_ACCOUNT_CODE	
,GENERAL_LEDGER_ACCOUNT_NAME	
,GENERAL_LEDGER_ACCOUNT_TEXT	
,TIMESPAN 
,SEARCH_TERM					
,DEL_FLAG							
,CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE    	
, START_DATE
, END_DATE 
,concat(GROUP_NUM, LANGUAGE_CODE,ACCOUNT_FORM, GENERAL_LEDGER_ACCOUNT_CODE,GENERAL_LEDGER_ACCOUNT_NAME, GENERAL_LEDGER_ACCOUNT_TEXT ,SEARCH_TERM) join_key
FROM dwifin.DWI_GL_ACCOUNT_MASTER_RECORD_DESC_T 
)

insert overwrite table   dwifin.DWI_GL_ACCOUNT_MASTER_RECORD_DESC_T 
select 
 t1.GROUP_NUM					
,t1.LANGUAGE_CODE				
,t1.ACCOUNT_FORM				    
,t1.GENERAL_LEDGER_ACCOUNT_CODE	
,t1.GENERAL_LEDGER_ACCOUNT_NAME	
,t1.GENERAL_LEDGER_ACCOUNT_TEXT	
,t1.TIMESPAN 
,t1.SEARCH_TERM					
,case when t2.join_key is null then 'Y' else t1.DEL_FLAG end DEL_FLAG
,t1.DW_LAST_UPDATE_DATE	
,t1.START_DATE
,case when t2.join_key is null then t2.START_DATE else t1.END_DATE end END_DATE
from dwi_tmp t1 left join sdi_tmp t2 on t1.join_key =t2.join_key
UNION all 
select 
 t1.GROUP_NUM					
,t1.LANGUAGE_CODE				
,t1.ACCOUNT_FORM				    
,t1.GENERAL_LEDGER_ACCOUNT_CODE	
,t1.GENERAL_LEDGER_ACCOUNT_NAME	
,t1.GENERAL_LEDGER_ACCOUNT_TEXT	
,t1.TIMESPAN 
,t1.SEARCH_TERM 
,t1.DEL_FLAG
,t1.DW_LAST_UPDATE_DATE	
,t1.START_DATE
,t1.END_DATE
from sdi_tmp t1 left join dwi_tmp t2 on t1.join_key = t2.join_key where t2.join_key is null
