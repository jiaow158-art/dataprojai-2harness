WITH SDI_TMP AS (
--INSERT OVERWRITE TABLE DWIFIN.DWI_GL_ACCOUNT_MASTER_RECORD_T
SELECT
MANDT		GROUP_NUM
,KTOPL		ACCOUNT_FORM
,SAKNR		GENERAL_LEDGER_ACCOUNT_CODE
,XBILK		BALANCE_SHEET_ACCOUNT_FLAG
,SAKAN		GENERAL_LEDGER_ACCOUNT		
,FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN ERDAT ='00000000' THEN '19000101' ELSE ERDAT  END ,'YYYYMMDD'),'YYYY-MM-DD') CREATE_DATE
,ERNAM		CREATOR
,GVTYP		PROFIT_AND_LOSS_ACCOUNT
,KTOKS		GENERAL_LEDGER_ACCOUNT_GROUP
,VBUND		TRADING_PARTNER_COMPANY_FLAG
,XLOEV		DELETE_ACCOUNT_FLAG
,XSPEA		FREEZE_ACCOUNT_CREATE_FLAG
,XSPEB		FREEZE_ACCOUNT_POSTING_FLAG
,XSPEP		FREEZE_PLAN_ACCOUNT_FLAG
,FUNC_AREA	FUNCTION_SCOPE
,GLACCOUNT_TYPE	GENERAL_LEDGER_ACCOUNT_TYPE
,LAST_CHANGED_TS	TIMESPAN
,'N' DEL_FLAG
,CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE 
FROM SDI.SDI_SKA1_1001 
)
,DWI_TMP AS (
SELECT 
GROUP_NUM					
,ACCOUNT_FORM				
,GENERAL_LEDGER_ACCOUNT_CODE	
,BALANCE_SHEET_ACCOUNT_FLAG

,DW_LAST_UPDATE_DATE 	
FROM DWIFIN.DWI_GL_ACCOUNT_MASTER_RECORD_T 
)
INSERT OVERWRITE TABLE DWIFIN.DWI_GL_ACCOUNT_MASTER_RECORD_T
SELECT 
 T1.GROUP_NUM					
,T1.ACCOUNT_FORM				
,T1.GENERAL_LEDGER_ACCOUNT_CODE	
,T1.BALANCE_SHEET_ACCOUNT_FLAG	
,T1.GENERAL_LEDGER_ACCOUNT		
,T1.CREATE_DATE					
,T1.CREATOR						
,T1.PROFIT_AND_LOSS_ACCOUNT		
,T1.GENERAL_LEDGER_ACCOUNT_GROUP
,T1.TRADING_PARTNER_COMPANY_FLAG
,T1.DELETE_ACCOUNT_FLAG			
,T1.FREEZE_ACCOUNT_CREATE_FLAG	
,T1.FREEZE_ACCOUNT_POSTING_FLAG	
,T1.FREEZE_PLAN_ACCOUNT_FLAG	
,T1.FUNCTION_SCOPE				
,T1.GENERAL_LEDGER_ACCOUNT_TYPE	
,T1.TIMESPAN					
,CASE WHEN T2.ACCOUNT_FORM IS NULL THEN 'Y' ELSE T1.DEL_FLAG END DEL_FLAG 					
,CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE FROM DWI_TMP T1
LEFT JOIN SDI_TMP T2 ON T1.GROUP_NUM = T2.GROUP_NUM AND T1.ACCOUNT_FORM = T2.ACCOUNT_FORM AND T1.GENERAL_LEDGER_ACCOUNT = T2.GENERAL_LEDGER_ACCOUNT
UNION ALL 
SELECT 
 T1.GROUP_NUM					
,T1.ACCOUNT_FORM				
,T1.GENERAL_LEDGER_ACCOUNT_CODE	
,T1.BALANCE_SHEET_ACCOUNT_FLAG	
,T1.GENERAL_LEDGER_ACCOUNT		
,T1.CREATE_DATE					
,T1.CREATOR						
,T1.PROFIT_AND_LOSS_ACCOUNT		
,T1.GENERAL_LEDGER_ACCOUNT_GROUP
,T1.TRADING_PARTNER_COMPANY_FLAG
,T1.DELETE_ACCOUNT_FLAG			
,T1.FREEZE_ACCOUNT_CREATE_FLAG	
,T1.FREEZE_ACCOUNT_POSTING_FLAG	
,T1.FREEZE_PLAN_ACCOUNT_FLAG	
,T1.FUNCTION_SCOPE				
,T1.GENERAL_LEDGER_ACCOUNT_TYPE	
,T1.TIMESPAN
,T1.DEL_FLAG
,T1.DW_LAST_UPDATE_DATE 
FROM SDI_TMP T1 
LEFT JOIN DWI_TMP T2 ON T1.GROUP_NUM = T2.GROUP_NUM AND T1.ACCOUNT_FORM = T2.ACCOUNT_FORM AND T1.GENERAL_LEDGER_ACCOUNT = T2.GENERAL_LEDGER_ACCOUNT WHERE T2.GENERAL_LEDGER_ACCOUNT IS NULL
