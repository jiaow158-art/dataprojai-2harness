insert overwrite table  dwifin.DWI_GL_ACCOUNT_VOUCHER_HEAD_T PARTITION(PERIOD_ID_M='${last_month}') 
select 
T1.GJAHR            AS FISCAL_YEAR,
T1.MONAT            AS ACCOUNTING_PERIOD,
T1.MANDT            AS GROUP_NUM,
CASE
    WHEN T1.BUKRS IS NULL OR T1.BUKRS = ''THEN -999999
    ELSE NVL(COM.COMPANY_ID, -999998)
END AS COMPANY_ID,
CASE                                        
   WHEN T1.BUKRS IS NULL OR T1.BUKRS = '' THEN 'SNULL'

END AS COMPANY_CODE,

T1.BKTXT            AS VOUCHER_HEADER_TEXT,

T1.XREF1_HD         AS VOUCHER_REFERENCE_CODE1,
T1.XREF2_HD         AS VOUCHER_REFERENCE_CODE2,
T1.PPNAM            AS PREPARATION_VOUCHER_USER_NAME,
(case 
when T1.PPDAT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.PPDAT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.PPDAT,'yyyyMMdd'),'yyyy-MM-dd') end) AS ACC_VOUCHER_PREPARATION_DATE,

when T1.CPUDT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.CPUDT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.CPUDT,'yyyyMMdd'),'yyyy-MM-dd') end) AS ACCOUNT_VOUCHER_INPUT_DATE,
(case 
when T1.BLDAT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.BLDAT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.BLDAT,'yyyyMMdd'),'yyyy-MM-dd') end) AS VOUCHER_DATE,
(case 
when T1.BUDAT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.BUDAT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.BUDAT,'yyyyMMdd'),'yyyy-MM-dd') end) AS VOUCHER_POST_DATE,
(case 
when T1.AEDAT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.AEDAT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.AEDAT,'yyyyMMdd'),'yyyy-MM-dd') end) AS ACC_LAST_UPDATE_DATE,
T1.AWKEY            AS OBJECT_KEY,
T1.XREVERSAL        AS VOUCHER_OFFSET_FLAG,
T1.XREVERSED        AS VOUCHER_OFFSETED_FLAG,
T1.XREVERSING       AS OFFSET_VOUCHER_FLAG,
T1.STBLG            AS OFFSET_VOUCHER_NUM,
R2.CONFIG_CODE      AS OFFSET_REASON,
T1.AWORG_REV        AS OFFSET_REFERENCE_ORG,
T1.AWREF_REV        AS OFFSET_REF_VOUCHER_NUM,
T1.XBLNR            AS REFERENCE_VOUCHER_NUM,
R3.CONFIG_CODE      AS REFERENCE_PROCESS,
T1.CO_REFBT         AS REFERENCE_VOUCHER_TYPE,
T1.LDGRPSPEC_PN     AS INPUT_VOUCHER_FLAG,
T1.CPUTM            AS INPUT_TIME,
T1.GRPID            AS BATCH_INPUT_SESSION_NAME,
T1.STJAH            AS REVOC_VOUCHER_FISCAL_YEAR,
T1.BVORG            AS BUSINESS_VOUCHER_NUM,
T1.USNAM            AS USER_NAME,
T1.GLVOR            AS BUSINESS_TRANSACTION,
T1.CO_VRGNG         AS CO_BUSINESS_TRANSACTION,
R4.CONFIG_CODE      AS TRANSACTION_CODE,
R5.domain_value_code            AS SOURCE_CURRENCY1,
R6.domain_value_code            AS FI_VOUCHER_TYPE,
R7.CONFIG_CODE                  AS FINANCIAL_MANAGEMENT_SCOPE,
R8.domain_value_code            AS GL_BUSINESS,
T1.HWAER            AS LOCAL_CURRENCY_CODE,
T1.WAERS            AS CURRENCY_CODE,
R9.domain_value_code            AS SECOND_LOCAL_CURRENCY_TYPE,
T1.HWAE2            AS SECOND_LOCAL_CURRENCY_CODE,
T1.KURS2            AS SECOND_LC_EX_RATE,
T1.KURSF            AS EX_RATE,

R11.CONFIG_CODE           AS EX_RATE_TYPE2,

when T1.PSOBT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.PSOBT,'yyyyMMdd'),'yyyy-MM-dd') end) AS POST_DATE,
T1.PSOZL            AS ACTUAL_ACCOUNT_FLAG,

R14.domain_value_code            AS SECOND_LC_TRANS_DATE_TYPE,
(case 
when T1.WWERT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.WWERT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.WWERT,'yyyyMMdd'),'yyyy-MM-dd') end) AS CONVERSION_DATE,
T1.XMWST            AS TAX_AUTO_CALC_FLAG,
T1.XRUEB            AS BILL_PREV_PERIOD_CREDITED_FLAG,
T1.XSECONDARY       AS SECOND_JOURNAL_ENTRY_FLAG,
T1.XSNET            AS IS_NET_AMT_FLAG,
T1.XSPLIT           AS FI_VOUCHER_SPLIT_POST_FLAG,
T1.XWVOF            AS UNDUE_CUSTOMER_INOIVCE_FLAG,
T1.del_flag,

from SDI.SDI_BKPF_1001   t1 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R           --凭证类型文本
ON R.SRC_SYS_CONFIG_CODE = T1.BLART 
AND R.GROUP_NUM = T1.MANDT
AND R.SOURCE_CODE = 'T003T' 
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R1      --凭证状态
ON  R1.domain_name  = 'BSTAT'
AND R1.src_sys_domain_value_code =T1.BSTAT
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2     --冲销过帐的原因 (名称)
ON R2.SRC_SYS_CONFIG_CODE = T1.STGRD
AND R2.GROUP_NUM = T1.MANDT 
AND R2.SOURCE_CODE = 'T041CT' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3      --会计核算的对象类型名称
ON R3.SRC_SYS_CONFIG_CODE = T1.AWTYP
AND R3.GROUP_NUM = T1.MANDT 
AND R3.SOURCE_CODE = 'TTYPT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R4        --事务代码文本
ON R4.SRC_SYS_CONFIG_CODE = T1.TCODE 
AND R4.GROUP_NUM = T1.MANDT 
AND R4.SOURCE_CODE = 'TSTCT'
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R5       --用于货币换算的源币
ON  R5.domain_name  = 'CURSR'
AND R5.src_sys_domain_value_code =T1.BASW2
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R6      --FI 凭证分类
ON  R6.domain_name  = 'DOCUMENT_CATEGORY'
AND R6.src_sys_domain_value_code =T1.DOCCAT
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R7     --资金管理域文字
ON R7.SRC_SYS_CONFIG_CODE = T1.FIKRS 
AND R7.GROUP_NUM = T1.MANDT 
AND R7.SOURCE_CODE = 'FM01T' 
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R8    --总账业务交易组
ON  R8.domain_name  = 'FINS_GLBTGRP'
AND R8.src_sys_domain_value_code =T1.GLBTGRP        
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R9      --第二本币的货币类型
ON  R9.domain_name  = 'CURSR'
AND R9.src_sys_domain_value_code =T1.CURT2
left join DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R10     --汇率类型的使用
ON R10.SRC_SYS_CONFIG_CODE = T1.KURST
AND R10.GROUP_NUM = T1.MANDT 
AND R10.SOURCE_CODE = 'TCURW' 
left join DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R11          --汇率类型的使用
ON R11.SRC_SYS_CONFIG_CODE = T1.KUTY2
AND R11.GROUP_NUM = T1.MANDT 
AND R11.SOURCE_CODE = 'TCURW' 
left join DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R12       --分类账组
ON R12.SRC_SYS_CONFIG_CODE = T1.LDGRP
AND R12.GROUP_NUM = T1.MANDT 
AND R12.SOURCE_CODE = 'FAGL_TLDGRPT' 
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R13       --FI/CO 接口中的组件子集
ON  R13.domain_name  = 'ACC_SUBSET'
AND R13.src_sys_domain_value_code =T1.SUBSET
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R14      --对于第二种本币的交易日期类型
ON  R14.domain_name  = 'CURDT'
AND R14.src_sys_domain_value_code =T1.UMRD2
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM    ------与公司主数据关联，取COMPANY_ID 
ON COM.COMPANY_CODE  = T1.BUKRS
AND COM.group_num=T1.MANDT
AND COM.DEL_FLAG  = 'N'
AND COM.END_DATE  = '4712-12-31'
WHERE T1.PERIOD_ID_M='${last_month}';
