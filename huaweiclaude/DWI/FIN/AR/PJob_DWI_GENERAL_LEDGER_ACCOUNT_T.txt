INSERT
  OVERWRITE TABLE DWIFIN.DWI_GENERAL_LEDGER_ACCOUNT_T    --科目表
SELECT
  S.MANDT AS GROUP_NUM,
  S.SPRAS AS LANG_CODE,
  T.CONFIG_CODE AS ACCOUNT_STATEMENT_CODE,
  T1.CONFIG_NAME AS ACCOUNT_STATEMENT_NAME,
  S.SAKNR AS GL_ACCOUNT_NUM,
  S.TXT20 AS GL_ACCOUNT_NAME,
  S.LAST_CHANGED_TS AS LAST_UPDATE_DATE,
  S.MCOD1 AS SEARCH_CRITERIA,
  S.TXT50 AS GL_ACCOUNT_LONG_TEXT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_SKAT_1001 S
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T ---科目表名
  ON T.SRC_SYS_CONFIG_CODE = S.KTOPL
  AND T.GROUP_NUM = S.MANDT
  AND T.SOURCE_CODE = 'T004T'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1  --科目表名
  ON T1.SRC_SYS_CONFIG_CODE = S.KTOPL
  AND T1.GROUP_NUM = S.MANDT
  AND T1.SOURCE_CODE = 'T004T'
  AND T1.LANG_CODE = '1'
UNION ALL
SELECT
  DWI.GROUP_NUM,
  DWI.LANG_CODE,
  DWI.ACCOUNT_STATEMENT_CODE,
  DWI.ACCOUNT_STATEMENT_NAME,
  DWI.GL_ACCOUNT_NUM,
  DWI.GL_ACCOUNT_NAME,
  DWI.LAST_UPDATE_DATE,
  DWI.SEARCH_CRITERIA,
  DWI.GL_ACCOUNT_LONG_TEXT,
  'Y' AS DEL_FLAG,
  ---删除标识判断
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIFIN.DWI_GENERAL_LEDGER_ACCOUNT_T DWI
  LEFT JOIN (
    SELECT
      S.MANDT,
      S.SPRAS,
      T.CONFIG_CODE AS KTOPL,
      S.SAKNR
    FROM
      SDI.SDI_SKAT_1001 S
      LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T ---科目表名
      ON T.SRC_SYS_CONFIG_CODE = S.KTOPL
      AND T.GROUP_NUM = S.MANDT
      AND T.SOURCE_CODE = 'T004T'
  ) SDI ON SDI.MANDT = DWI.GROUP_NUM
  AND SDI.SPRAS = DWI.LANG_CODE
  AND NVL(SDI.KTOPL,'SNULL') = NVL(DWI.ACCOUNT_STATEMENT_CODE,'SNULL')
  AND SDI.SAKNR = DWI.GL_ACCOUNT_NUM
WHERE 1=1
  AND SDI.MANDT IS NULL
  AND SDI.SPRAS IS NULL
  AND SDI.KTOPL IS NULL
  AND SDI.SAKNR IS NULL
