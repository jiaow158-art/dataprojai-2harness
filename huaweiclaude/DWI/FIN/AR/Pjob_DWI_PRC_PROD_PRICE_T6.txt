set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PROD_PRICE_T PARTITION(SOURCE_CODE='A943')
SELECT 
S8.MANDT AS GROUP_NUM,
S8.KNUMH AS CONDITION_RECORD_NUM,
S8.DATAB AS SRC_SYS_START_DATE,
FROM_UNIXTIME(UNIX_TIMESTAMP(S8.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE,   ---8位字符转换为日期格式
CASE WHEN SUBSTR(S8.DATBI,1,4) ='9999' 
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S8.DATBI,SUBSTR(S8.DATBI,1,4),'4712'),'yyyyMMdd'),'yyyy-MM-dd')
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(S8.DATBI,'yyyyMMdd'),'yyyy-MM-dd')
END AS END_DATE,              ---8位字符，转换为日期格式，99991231转换为4712-12-31
NULL AS APPLICATION_PROGRAM_CODE,
NULL AS CONDITION_TYPE,
S8.KFRST AS APPROVAL_STATUS_CODE,
S8.KBSTAT AS PRICE_TREATMENT_STATUS,
NULL AS SALE_OFFICE_CODE,
R.CONFIG_CODE AS SALE_ORG_CODE,
R1.CONFIG_CODE AS DIS_CHANNEL_CODE,
CASE WHEN S8.WERKS IS NULL OR S8.WERKS = '' THEN -999999             
ELSE NVL(FAC.FACTORY_ID,-999998)
END AS FACTORY_ID,
CASE                                        
   WHEN S8.WERKS IS NULL OR S8.WERKS = '' THEN 'SNULL'            
   ELSE NVL(FAC.FACTORY_CODE,'NOT EXISTS') 
END AS FACTORY_CODE,
NULL AS PAYER_ID,
'-999999' AS CUST_ID,
NULL AS CUST_NUM,
'-999999' AS MATERIAL_ID,
NULL AS MATERIAL_NUM,
NULL AS CUST_PRICE_GROUP,
'SDI_A943_1001' AS SOURCE_TYPE,
'工厂/销售机构/分销渠道' AS SOURCE_DESC,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_A943_1001 S8
LEFT JOIN  DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R   ---机构单位：销售机构：文本
ON S8.VKORG=R.Src_sys_config_code
AND S8.MANDT=R.GROUP_NUM
AND R.SOURCE_CODE='TVKOT' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1   ---机构单位：分销渠道：文本
ON S8.VTWEG=R1.Src_sys_config_code
AND S8.MANDT=R1.GROUP_NUM
AND R1.SOURCE_CODE='TVTWT'  
LEFT JOIN  DWIMD.DWI_MD_FACTORY_T FAC     ------与工厂主数据关联，取MATERIAL_ID,源字段非空处理
ON FAC.FACTORY_CODE  = S8.WERKS 
AND FAC.GROUP_NUM = S8.MANDT 
AND FAC.END_DATE ='4712-12-31'
AND FAC.DEL_FLAG='N'
UNION ALL
SELECT
DWI.GROUP_NUM,               
DWI.CONDITION_RECORD_NUM,    
DWI.SRC_SYS_START_DATE,      
DWI.START_DATE,              
DWI.END_DATE,                
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_TYPE,          
DWI.APPROVAL_STATUS_CODE,    
DWI.PRICE_TREATMENT_STATUS,  
DWI.SALE_OFFICE_CODE,       
DWI.SALE_ORG_CODE,          
DWI.DIS_CHANNEL_CODE,        
DWI.FACTORY_ID,              
DWI.FACTORY_CODE,            
DWI.PAYER_ID,                
DWI.CUST_ID,                 
DWI.CUST_NUM,                
DWI.MATERIAL_ID,             
DWI.MATERIAL_NUM,            
DWI.CUST_PRICE_GROUP,        
DWI.SOURCE_TYPE,             
DWI.SOURCE_DESC,                         
'Y' AS DEL_FLAG,            ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T DWI
LEFT JOIN SDI.SDI_A943_1001 S8
ON NVL(S8.MANDT,'SNULL')=NVL(DWI.GROUP_NUM,'SNULL')
AND NVL(S8.KNUMH,'SNULL')=NVL(DWI.CONDITION_RECORD_NUM,'SNULL')
AND NVL(S8.DATAB,'SNULL')=NVL(DWI.SRC_SYS_START_DATE,'SNULL')
WHERE S8.MANDT  IS NULL
AND S8.KNUMH  IS NULL
AND S8.DATAB  IS NULL
AND source_type='SDI_A943_1001'
