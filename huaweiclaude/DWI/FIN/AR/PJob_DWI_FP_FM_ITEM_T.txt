set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
WITH FMIFIIT_TMP AS
(SELECT T1.FMBELNR,
        T1.RLDNR,
        T1.FMBUZEI,
        T1.STUNR,
        R1.CONFIG_CODE,
        T1.GJAHR,
        T1.BTART,
        T1.MANDT,
        T1.DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FMIFIIT_1001 T1
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1   --资金管理域文字
  ON R1.SRC_SYS_CONFIG_CODE = T1.FIKRS
  AND R1.GROUP_NUM = T1.MANDT
  AND R1.SOURCE_CODE = 'FM01T'
)
INSERT
  OVERWRITE TABLE DWIFIN.DWI_FP_FM_ITEM_T                        --在资金管理中的FI行项目
SELECT
  T1.FMBELNR AS FM_ITEM_VOUCHER_NUM,
  T1.RLDNR AS LEDGER_CODE,
  T1.FMBUZEI AS FM_ITEM_RULE_NUM,
  T1.STUNR AS DB_LINE_KEY,
  R1.CONFIG_CODE AS FINANCIAL_MANAGEMENT_SCOPE,
  T1.GJAHR AS FISCAL_YEAR,
  T1.BTART AS AMOUNT_TYPE,
  T1.MANDT AS GROUP_NUM,
  R2.CONFIG_CODE AS CO_BUSINESS_TRANSACTION,
  T1.KNBELNR AS FI_VOUCHER_NUM,
  T1.KNGJAHR AS FI_VOUCHER_ACCOUNT_YEAR,
  T1.KNBUZEI AS FI_VOUCHER_ITEM,
  T1.PAYFLG AS FM_VOUCHER_PAYMENT_STATUS,
  R3.CONFIG_CODE AS BUSINESS_SCOPE,
  T1.TRBTR AS BUSINESS_CURRENCY_AMT,
  T1.TWAER AS TRANSACTIONAL_CURRENCY_CODE,
  T1.VOBELNR AS PAYMENT_VOUCHER_NUM,
  T1.FKBTR AS FM_CURRENCY_AMT,
  T1.WRTTP AS VALUE_TYPE,
  T1.VREFBT AS PRE_VOUCHER_LIST,
  T1.VREFBN AS PRE_VOUCHER_NUM,
  T1.VRFPOS AS PRE_VOUCHER_ITEM,
  T1.VRFKNT AS PRE_ACCOUNT_NUM,
  T1.BUKRS AS COMPANY_CODE,
  T1.PRCTR AS PROFIT_CENTER,
  T1.VRFTYP AS REFERENCE_PROCESS,
  T1.MEINS AS BASE_MEASURE_UNIT,
  T1.FISTL AS FUND_CENTER_CODE,
  T1.HKONT AS GL_ACCOUNT,
  T1.FIPEX AS COMMITMENT_ITEM_ID,
  T1.VOGJAHR AS VOUCHER_ACCOUNT_YEAR,
  T1.VOBUKRS AS VOUCHER_COMPANY_CODE,
  T1.MENGE AS QUANTITY,
  T1.PERIO AS ACCOUNT_PERIOD,
  R4.DOMAIN_VALUE_CODE AS CASH_FLOW,
  T1.CFLEV AS CARRY_FORWARD_LEVEL,
  T1.XREF3 AS LINE_ITEM_REFERENCE_CODE,
  T1.VBUND AS TRADE_PARTNER_COMPANY_ID,
  T1.TRANR AS FM_TRANSACTION_NUM,
  T1.FMXDOCNR AS FUND_MNGMT_REF_VOUCHER_NUM,
  T1.FMXYEAR AS FUND_MNGMT_REF_YEAR,
  T1.FMXDOCLN AS FUND_MNGMT_REF_LINE_ITEM,
(case 
when T1.ZHLDT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.ZHLDT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.ZHLDT,'yyyyMMdd'),'yyyy-MM-dd') end) AS FM_UPDATE_DATE,
(case 
when T1.PSOBT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.PSOBT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.PSOBT,'yyyyMMdd'),'yyyy-MM-dd') end) AS POST_DATE,
  T1.SGTXT AS ITEM_TEXT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FMIFIIT_1001 T1
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1   --资金管理域文字
  ON R1.SRC_SYS_CONFIG_CODE = T1.FIKRS
  AND R1.GROUP_NUM = T1.MANDT
  AND R1.SOURCE_CODE = 'FM01T'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2   --商业事务文本
  ON R2.SRC_SYS_CONFIG_CODE = T1.VRGNG
  AND R2.GROUP_NUM = T1.MANDT
  AND R2.SOURCE_CODE = 'TJ01T'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3  --业务部门名称
  ON R3.SRC_SYS_CONFIG_CODE = T1.BUS_AREA
  AND R3.GROUP_NUM = T1.MANDT
  AND R3.SOURCE_CODE = 'TGSBT'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R R4      --现金流量
  ON R4.DOMAIN_NAME = 'ZZXJLL'
  AND R4.SRC_SYS_DOMAIN_VALUE_CODE = T1.ZZXJLL
  WHERE DATE(T1.DW_LAST_UPDATE_DATE) >= '${start_date}'
UNION ALL
SELECT
  T1.FM_ITEM_VOUCHER_NUM,
  T1.LEDGER_CODE,
  T1.FM_ITEM_RULE_NUM,
  T1.DB_LINE_KEY,
  T1.FINANCIAL_MANAGEMENT_SCOPE,
  T1.FISCAL_YEAR,
  T1.AMOUNT_TYPE,
  T1.GROUP_NUM,
  T1.CO_BUSINESS_TRANSACTION,
  T1.FI_VOUCHER_NUM,
  T1.FI_VOUCHER_ACCOUNT_YEAR,
  T1.FI_VOUCHER_ITEM,
  T1.FM_VOUCHER_PAYMENT_STATUS,
  T1.BUSINESS_SCOPE,
  T1.BUSINESS_CURRENCY_AMT,
  T1.TRANSACTIONAL_CURRENCY_CODE,
  T1.PAYMENT_VOUCHER_NUM,
  T1.FM_CURRENCY_AMT,
  T1.VALUE_TYPE,
  T1.PRE_VOUCHER_LIST,
  T1.PRE_VOUCHER_NUM,
  T1.PRE_VOUCHER_ITEM,
  T1.PRE_ACCOUNT_NUM,
  T1.COMPANY_CODE,
  T1.PROFIT_CENTER,
  T1.REFERENCE_PROCESS,
  T1.BASE_MEASURE_UNIT,
  T1.FUND_CENTER_CODE,
  T1.GL_ACCOUNT,
  T1.COMMITMENT_ITEM_ID,
  T1.VOUCHER_ACCOUNT_YEAR,
  T1.VOUCHER_COMPANY_CODE,
  T1.QUANTITY,
  T1.ACCOUNT_PERIOD,
  T1.CASH_FLOW,
  T1.CARRY_FORWARD_LEVEL,
  T1.LINE_ITEM_REFERENCE_CODE,
  T1.TRADE_PARTNER_COMPANY_ID,
  T1.FM_TRANSACTION_NUM,
  T1.FUND_MNGMT_REF_VOUCHER_NUM,
  T1.FUND_MNGMT_REF_YEAR,
  T1.FUND_MNGMT_REF_LINE_ITEM,
  T1.FM_UPDATE_DATE,
  T1.POST_DATE,
  T1.ITEM_TEXT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIFIN.DWI_FP_FM_ITEM_T T1
  left join FMIFIIT_TMP  T2 
  ON NVL(T1.FM_ITEM_VOUCHER_NUM, 'SNULL') = NVL(T2.FMBELNR, 'SNULL')
  AND NVL(T1.LEDGER_CODE, 'SNULL') = NVL(T2.RLDNR, 'SNULL')
  AND NVL(T1.FM_ITEM_RULE_NUM, 'SNULL') = NVL(T2.FMBUZEI, 'SNULL')
  AND NVL(T1.DB_LINE_KEY, 'SNULL') = NVL(T2.STUNR, 'SNULL')
  AND NVL(T1.FINANCIAL_MANAGEMENT_SCOPE, 'SNULL') = NVL(T2.CONFIG_CODE, 'SNULL')
  AND NVL(T1.FISCAL_YEAR, 'SNULL') = NVL(T2.GJAHR, 'SNULL')
  AND NVL(T1.AMOUNT_TYPE, 'SNULL') = NVL(T2.BTART, 'SNULL')
  AND NVL(T1.GROUP_NUM, 'SNULL') = NVL(T2.MANDT, 'SNULL')
  AND DATE(T2.DW_LAST_UPDATE_DATE) >= '${start_date}'
WHERE
  T2.STUNR IS NULL;
