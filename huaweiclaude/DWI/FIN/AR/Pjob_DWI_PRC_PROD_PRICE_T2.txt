set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PROD_PRICE_T PARTITION(SOURCE_CODE='A900')
SELECT 
S7.MANDT AS GROUP_NUM,
S7.KNUMH AS CONDITION_RECORD_NUM,
S7.DATAB AS SRC_SYS_START_DATE,
FROM_UNIXTIME(UNIX_TIMESTAMP(S7.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE,         ---8位字符转换为日期格式
CASE WHEN SUBSTR(S7.DATBI,1,4) ='9999' 
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S7.DATBI,SUBSTR(S7.DATBI,1,4),'4712'),'yyyyMMdd'),'yyyy-MM-dd')
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(S7.DATBI,'yyyyMMdd'),'yyyy-MM-dd')
END AS END_DATE,                                 ---8位字符，转换为日期格式，99991231转换为4712-12-31
NULL AS APPLICATION_PROGRAM_CODE,
NULL AS CONDITION_TYPE,
NULL AS APPROVAL_STATUS_CODE,
NULL AS PRICE_TREATMENT_STATUS,
NULL AS SALE_OFFICE_CODE,
R.CONFIG_CODE AS SALE_ORG_CODE,
NULL AS DIS_CHANNEL_CODE,
'-999999' AS FACTORY_ID,
NULL AS FACTORY_CODE,
NULL AS PAYER_ID,
'-999999' AS CUST_ID,
NULL AS CUST_NUM,
CASE WHEN S7.MATNR IS NULL OR S7.MATNR = '' THEN -999999             
ELSE NVL(MAT.MATERIAL_ID,-999998) 
END AS MATERIAL_ID,
CASE                                        
   WHEN S7.MATNR IS NULL OR S7.MATNR = '' THEN 'SNULL'            
   ELSE NVL(MAT.MATERIAL_NUM,'NOT EXISTS') 
END AS MATERIAL_NUM,
NULL AS CUST_PRICE_GROUP,
'SDI_A900_1001' AS SOURCE_TYPE,
'销售组织/物料' AS SOURCE_DESC,
'N'  AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_A900_1001 S7
LEFT JOIN  DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R     ---机构单位：销售机构：文本
ON S7.VKORG=R.Src_sys_config_code
AND S7.MANDT=R.GROUP_NUM
AND R.SOURCE_CODE='TVKOT'
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT    ---与物料主数据关联，取MATERIAL_ID,源字段非空处理
ON MAT.MATERIAL_NUM  = S7.MATNR 
AND MAT.GROUP_NUM = S7.MANDT 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG='N'
UNION ALL
SELECT
DWI.GROUP_NUM,               
DWI.CONDITION_RECORD_NUM,    
DWI.SRC_SYS_START_DATE,      
DWI.START_DATE,              
DWI.END_DATE,                
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_TYPE,          
DWI.APPROVAL_STATUS_CODE,    
DWI.PRICE_TREATMENT_STATUS,  
DWI.SALE_OFFICE_CODE,       
DWI.SALE_ORG_CODE,          
DWI.DIS_CHANNEL_CODE,        
DWI.FACTORY_ID,              
DWI.FACTORY_CODE,            
DWI.PAYER_ID,                
DWI.CUST_ID,                 
DWI.CUST_NUM,                
DWI.MATERIAL_ID,             
DWI.MATERIAL_NUM,            
DWI.CUST_PRICE_GROUP,        
DWI.SOURCE_TYPE,             
DWI.SOURCE_DESC,                         
'Y' AS DEL_FLAG,          ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T DWI
LEFT JOIN SDI.SDI_A900_1001 S7
ON NVL(S7.MANDT,'SNULL')=NVL(DWI.GROUP_NUM,'SNULL')
AND NVL(S7.KNUMH,'SNULL')=NVL(DWI.CONDITION_RECORD_NUM,'SNULL')
AND NVL(S7.DATAB,'SNULL')=NVL(DWI.SRC_SYS_START_DATE,'SNULL')
WHERE S7.MANDT  IS NULL
AND S7.KNUMH  IS NULL
AND S7.DATAB  IS NULL
AND source_type='SDI_A900_1001'
