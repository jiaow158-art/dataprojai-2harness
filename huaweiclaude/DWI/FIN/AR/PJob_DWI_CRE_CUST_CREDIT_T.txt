insert overwrite table  dwifin.DWI_CRE_CUST_CREDIT_T        --客户授信表
select 
T1.CLIENT           AS GROUP_NUM,
CASE
    WHEN T1.PARTNER IS NULL OR T1.PARTNER = ''THEN -999999
    ELSE NVL(CUS.CUST_ID, -999998)
END AS CUST_ID,                                       
T1.PARTNER        AS CUST_NUM,
R.CONFIG_CODE    AS CREDIT_SGMNT,
T1.CREDIT_LIMIT     AS CREDIT_LIMIT_AMT,
CASE WHEN SUBSTR(T1.LIMIT_CHG_DATE,1,8) ='00000000'     --时间判断
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(T1.LIMIT_CHG_DATE,SUBSTR(T1.LIMIT_CHG_DATE,1,8),'19000101'),'yyyyMMdd'),'yyyy-MM-dd')   
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(T1.LIMIT_CHG_DATE,'yyyyMMdd'),'yyyy-MM-dd') END AS LIMIT_CHANGE_DATE,
T1.LOG_HNDL         AS LOG_HNDL,
T1.LIMIT_VALID_DATE AS LIMIT_VALID_DATE,
T1.XCRITICAL        AS CRITICAL_FLAG,
T1.X_LIMIT_ZERO     AS LIMIT_ZERO_FLAG,
T1.DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE 
FROM SDI.SDI_UKMBP_CMS_SGM_1001 T1
LEFT JOIN  DWIMD.DWI_MD_CUST_GENERAL_T CUS    --关联客户主数据取客户id
ON CUS.cust_num  = T1.PARTNER 
AND CUS.group_num = T1.CLIENT 
AND CUS.END_DATE ='4712-12-31'
AND CUS.DEL_FLAG='N'
left join DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R   --SAP 信用管理: 段数据的文本
ON R.SRC_SYS_CONFIG_CODE = T1.CREDIT_SGMNT 
AND R.GROUP_NUM = T1.CLIENT 
AND R.SOURCE_CODE = 'UKMCRED_SGM0T'
