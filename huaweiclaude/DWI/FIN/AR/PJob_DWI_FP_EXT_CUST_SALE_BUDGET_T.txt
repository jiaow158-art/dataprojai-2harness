insert overwrite table  dwifin.DWI_FP_EXT_CUST_SALE_BUDGET_T    --外部客户月度销售预算
select 
T1.ZCUST_MAT_TYPE  AS EXT_CUST_SALE_PROD_TYPE,
CASE
    WHEN T1.ZCUST_NAME IS NULL OR T1.ZCUST_NAME = '' THEN -999999
    ELSE NVL(CUS.CUST_ID, -999998)
END AS CUST_ID,
T1.ZCUST_NAME AS CUST_NUM,
T1.ZYEAR           AS FISCAL_YEAR,
T1.MANDT           AS GROUP_NUM,
T1.SAVE_TO_SAP     AS SINGLE_CHARACTER_FLAG,
T1.OALSH           AS OA_SERIAL_NUM,
T1.BZJ             AS CASH_DEPOSIT_AMT,
T1.GSQC            AS COMPANY_NAME,
T1.ZCUST_MT_DES    AS EXT_CUST_SALE_PROD_TYPE_DESC,
(case 
when T1.SDHTRQ < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.SDHTRQ > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.SDHTRQ,'yyyyMMdd'),'yyyy-MM-dd') end) AS CONTRACT_ACCEPT_DATE,
T1.JANUARY         AS MON_01_SALE_BUDGET_AMT,
T1.FEBRUARY        AS MON_02_SALE_BUDGET_AMT,
T1.MARCH           AS MON_03_SALE_BUDGET_AMT,
T1.APRIL           AS MON_04_SALE_BUDGET_AMT,
T1.MAY             AS MON_05_SALE_BUDGET_AMT,
T1.JUNE            AS MON_06_SALE_BUDGET_AMT,
T1.JULY            AS MON_07_SALE_BUDGET_AMT,
T1.AUGUST          AS MON_08_SALE_BUDGET_AMT,
T1.SEPTEMBER       AS MON_09_SALE_BUDGET_AMT,
T1.OCTOBER         AS MON_10_SALE_BUDGET_AMT,
T1.NOVEMBER        AS MON_11_SALE_BUDGET_AMT,
T1.DECEMBER        AS MON_12_SALE_BUDGET_AMT,
T1.QUART_1ST       AS QUTR_1_SALE_BUDGET_AMT,
T1.QUART_2ND       AS QUTR_2_SALE_BUDGET_AMT,
T1.QUART_3RD       AS QUTR_3_SALE_BUDGET_AMT,
T1.QUART_4TH       AS QUTR_4_SALE_BUDGET_AMT,
T1.YEAR_TOTAL      AS YEAR_SALE_BUDGET_AMT,
T1.DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZCRM_DATA_MONTH_1001 T1
LEFT JOIN  DWIMD.DWI_MD_CUST_GENERAL_T CUS    --关联客户主数据取客户id
ON CUS.cust_num  = T1.ZCUST_NAME
AND CUS.GROUP_NUM=T1.MANDT
AND CUS.DEL_FLAG = 'N'
AND CUS.END_DATE = '4712-12-31'
