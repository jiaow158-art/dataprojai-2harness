set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_TAX_INVOICE_T     --税票主表
SELECT MANDT AS GROUP_NUM,
CASE
    WHEN S.BUKRS IS NULL OR S.BUKRS = '' THEN -999999
    ELSE NVL(COM.COMPANY_ID, -999998)
END AS COMPANY_ID,
CASE                                        
   WHEN S.BUKRS IS NULL OR S.BUKRS = '' THEN 'SNULL'            
   ELSE NVL(COM.COMPANY_CODE,'NOT EXISTS') 
END AS COMPANY_CODE,
VBELN AS BILL_VOUCHER_NUM,
TAX_NO AS TAX_NUM,
TAX_CX AS ORDER_NUM,
ZHYTDH AS OCEAN_BILL_LADING_NUM,
CASE WHEN S.KUNNR IS NULL OR S.KUNNR = '' THEN -999999             
ELSE NVL(CUS.CUST_ID,-999998) 
END AS CUST_ID,
CASE                                        
   WHEN S.KUNNR IS NULL OR S.KUNNR = '' THEN 'SNULL'            
   ELSE NVL(CUS.CUST_NUM,'NOT EXISTS') 
END AS CUST_NUM,
MWSBK AS DOC_CURRENCY_TAX_AMT,
POSNR AS BILL_ITEM_NUM,
ZBGDH AS CUSTOMS_DECLARATION_NUM,
from_unixtime(unix_timestamp(DATE_D,'yyyyMMdd'),'yyyy-MM-dd') AS PRINTING_DATE,   --8位字符转换为日期格式
DEMO AS DEMO_DESC,
from_unixtime(unix_timestamp(ZDATE2,'yyyyMMdd'),'yyyy-MM-dd') AS SAP_INVOICE_ISSUE_DATE2,
from_unixtime(unix_timestamp(ZDATE,'yyyyMMdd'),'yyyy-MM-dd') AS SAP_INVOICE_ISSUE_DATE,
NAME AS INPUT_NAME,
TAX_MONEY AS NET_VALUE_AMT,
STATUS_VF AS INVOICE_STATUS,
TAX_TYPE AS TAX_TYPE,
ZTDSJ AS BILL_LADING_DATE,
TEXT_NO AS TAX_INVOICE_TEXT_NUM,
TIME_T AS PRINTING_TIME,
ZTMONEY AS TOTAL_NET_VALUE_AMT,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZINVOICE4_1001 S
LEFT JOIN  DWIMD.DWI_MD_CUST_GENERAL_T CUS   ----与客户主数据关联，取CUST_ID,源字段非空处理
ON CUS.CUST_NUM  = S.KUNNR 
AND CUS.GROUP_NUM = S.MANDT 
AND CUS.END_DATE ='4712-12-31'
AND CUS.DEL_FLAG='N'
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM    ------与公司主数据关联，取COMPANY_ID 
ON COM.COMPANY_CODE  = S.BUKRS
AND COM.GROUP_NUM=S.MANDT
AND COM.DEL_FLAG  = 'N'
AND COM.END_DATE  = '4712-12-31' 
UNION ALL
SELECT
DWI.GROUP_NUM,
DWI.COMPANY_ID,
DWI.COMPANY_CODE,
DWI.BILL_VOUCHER_NUM,
DWI.TAX_NUM,
DWI.ORDER_NUM,
DWI.OCEAN_BILL_LADING_NUM,
DWI.CUST_ID,
DWI.CUST_NUM,
DWI.DOC_CURRENCY_TAX_AMT,
DWI.BILL_ITEM_NUM,
DWI.CUSTOMS_DECLARATION_NUM,
DWI.PRINTING_DATE,
DWI.DEMO_DESC,
DWI.SAP_INVOICE_ISSUE_DATE2,
DWI.SAP_INVOICE_ISSUE_DATE,
DWI.INPUT_NAME,
DWI.NET_VALUE_AMT,
DWI.INVOICE_STATUS,
DWI.TAX_TYPE,
DWI.BILL_LADING_DATE,
DWI.TAX_INVOICE_TEXT_NUM,
DWI.PRINTING_TIME,
DWI.TOTAL_NET_VALUE_AMT,
'Y' AS DEL_FLAG,      ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_TAX_INVOICE_T DWI
LEFT JOIN SDI.SDI_ZINVOICE4_1001 SDI
ON SDI.MANDT=DWI.GROUP_NUM
AND SDI.TAX_NO=DWI.TAX_NUM
WHERE  SDI.MANDT IS NULL
AND SDI.TAX_NO IS NULL
