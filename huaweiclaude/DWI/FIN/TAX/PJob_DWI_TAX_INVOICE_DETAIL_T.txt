INSERT OVERWRITE TABLE DWIFIN.DWI_TAX_INVOICE_DETAIL_T   --税票明细表
SELECT 
MANDT AS 	GROUP_NUM,
CASE
    WHEN S1.BUKRS IS NULL OR S1.BUKRS = '' THEN -999999
    ELSE NVL(COM.COMPANY_ID, -999998)
END AS COMPANY_ID,
CASE                                        
   WHEN S1.BUKRS IS NULL OR S1.BUKRS = '' THEN 'SNULL'            
   ELSE NVL(COM.COMPANY_CODE,'NOT EXISTS') 
END AS COMPANY_CODE,
POSNR AS BILL_ITEM_NUM,
VBELN AS BILL_VOUCHER_NUM,
TAX_NO AS TAX_NUM,
TAX_CX AS ORDER_NUM,
LINESTATUS AS BILL_STATUS,
ARKTX AS SO_ITEM_SHORT_TEXT,
FKIMG AS ACT_BILLED_QTY,
KUNNR AS CUST_NUM,
MWSBP AS VC_TAX_AMT,
NETWR AS VC_NET_VAULE_AMT,
from_unixtime(unix_timestamp(DATE_D,'yyyyMMdd'),'yyyy-MM-dd') AS PRINTING_DATE, --8位字符转换为日期格式
from_unixtime(unix_timestamp(ZDATE,'yyyyMMdd'),'yyyy-MM-dd') AS SAP_INVOICE_ISSUE_DATE,
from_unixtime(unix_timestamp(ZDATE2,'yyyyMMdd'),'yyyy-MM-dd') AS SAP_INVOICE_ISSUE_DATE2,
NAME AS INPUT_NAME,
TAX_MONEY AS NET_VALUE_AMT,
STATUS_VF AS INVOICE_STATUS,
TAX_TYPE AS TAX_TYPE,
TEXT_NO AS TAX_INVOICE_TEXT_NUM,
TIME_T AS PRINTING_TIME,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZINVOICE4DETAIL_1001 S1
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM    ------与公司主数据关联，取COMPANY_ID 
ON COM.COMPANY_CODE  = S1.BUKRS
AND COM.group_num=S1.MANDT
AND COM.DEL_FLAG  = 'N'
AND COM.END_DATE  = '4712-12-31' 
UNION ALL
SELECT
DWI.GROUP_NUM,
DWI.COMPANY_ID,
DWI.COMPANY_CODE,
DWI.BILL_ITEM_NUM,
DWI.BILL_VOUCHER_NUM,
DWI.TAX_NUM,
DWI.ORDER_NUM,
DWI.BILL_STATUS,
DWI.SO_ITEM_SHORT_TEXT,
DWI.ACT_BILLED_QTY,
DWI.CUST_NUM,
DWI.VC_TAX_AMT,
DWI.VC_NET_VAULE_AMT,
DWI.PRINTING_DATE,
DWI.SAP_INVOICE_ISSUE_DATE,
DWI.SAP_INVOICE_ISSUE_DATE2,
DWI.INPUT_NAME,
DWI.NET_VALUE_AMT,
DWI.INVOICE_STATUS,
DWI.TAX_TYPE,
DWI.TAX_INVOICE_TEXT_NUM,
DWI.PRINTING_TIME,
'Y' AS DEL_FLAG,
SUBSTR(CURRENT_TIMESTAMP(),1,19) AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_TAX_INVOICE_DETAIL_T DWI
LEFT JOIN SDI.SDI_ZINVOICE4DETAIL_1001 SDI
ON SDI.BUKRS=DWI.COMPANY_CODE
AND  SDI.MANDT=DWI.GROUP_NUM 
AND SDI.POSNR=DWI.BILL_ITEM_NUM
AND SDI.VBELN=DWI.BILL_VOUCHER_NUM
AND SDI.TAX_NO=DWI.TAX_NUM
WHERE SDI.BUKRS IS NULL
AND SDI.MANDT IS NULL
AND SDI.POSNR IS NULL
AND SDI.VBELN IS NULL
AND SDI.TAX_NO IS NULL
