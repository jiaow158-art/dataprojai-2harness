insert overwrite table DWIBS .DWI_APR_NODE_BASIC_INFO_T
select 
    ID
   ,NODENAME AS NODE_NAME
   ,ISSTART AS CURR_NODE_CRT
   ,ISREJECT AS CURR_NODE_REBACK
   ,ISREOPEN AS OPEN_AGAIN
   ,ISEND AS ARCHIVE_CODE
   ,DRAWXPOS AS NODE_ABSCISSA
   ,DRAWYPOS AS NODE_ORDINATE
   ,TOTALGROUPS AS CURR_NODE_OP_GROUP_AMT
   ,NODEATTRIBUTE AS NODE_PROPERTY
   ,PASSNUM AS BRANCH_AMT
   ,ISFREENODE AS USER_DEFINED_NODE
   ,FLOWORDER  AS TRANSFER_ORDER
   ,SIGNTYPE AS SIGN_TYPE
   ,REQUESTID AS REQUEST_ID
   ,STARTNODEID AS START_NODE_ID
   ,ECOLOGY_PINYIN_SEARCH AS ECOLOGY_PINYIN_SEARCH
   ,DRAWSTYLE AS FLOWCHART_TYPE
   ,UUID     
   ,'N' AS DEL_FLAG
   ,current_timestamp() as DW_LAST_UPDATE_DATE
from SDI.SDI_WORKFLOW_NODEBASE_1004
--更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
UNION 
 SELECT 
   T1.ID						
  ,T1.NODE_NAME				   
  ,T1.CURR_NODE_CRT			
  ,T1.CURR_NODE_REBACK		
  ,T1.OPEN_AGAIN				
  ,T1.ARCHIVE_CODE			
  ,T1.NODE_ABSCISSA			
  ,T1.NODE_ORDINATE			
  ,T1.CURR_NODE_OP_GROUP_AMT	
  ,T1.NODE_PROPERTY			
  ,T1.BRANCH_AMT				
  ,T1.USER_DEFINED_NODE		
  ,T1.TRANSFER_ORDER			
  ,T1.SIGN_TYPE				
  ,T1.REQUEST_ID				
  ,T1.START_NODE_ID			
  ,T1.ECOLOGY_PINYIN_SEARCH	
  ,T1.FLOWCHART_TYPE			
  ,T1.UUID					
  ,'Y' AS DEL_FLAG			 
  ,current_timestamp() as DW_LAST_UPDATE_DATE	
FROM DWIBS .DWI_APR_NODE_BASIC_INFO_T T1
LEFT JOIN SDI.SDI_WORKFLOW_NODEBASE_1004 T2
ON T1.ID = T2.ID
WHERE T2.ID IS NULL;
