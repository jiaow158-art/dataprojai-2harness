with tmp_data01 as(
	SELECT VBELN,POSNR,concat_ws('|',collect_list(CAST(BILL_NO AS VARCHAR(80)))) as BILL_NO
		  ,concat_ws('|',collect_list(cast(DEST_LOCATION AS VARCHAR(80)))) as DEST_LOCATION
	from (SELECT distinct B.VBELN,B.POSNR,case when A.BILL_NO = 0 then '自提'
								when A.BILL_NO = 1 then '配送'
								 end as BILL_NO
	,a.DEST_LOCATION                             
	FROM SDI.SDI_ZDEL_PLAN_H_1001 A
	INNER JOIN SDI.SDI_ZDEL_PLAN_D_1001 B
	ON A.PLAN_NO = B.PLAN_NO
	AND A.MANDT = B.MANDT
	WHERE A.MANDT = 800) b
	group by VBELN,POSNR
),
tmp_data02_1 as(
    select distinct
       --A2.DELIVERY AS DELIVERY_NO--提货单号
      --,A1.DELIVERY_PROJECT --交货项目
	  A1.REFERENCE_BILL_NUM AS SALE_ORDER_NUM -- SAP订单头
	  ,A1.REFERENCE_PROJ_NUM AS SALE_ORDER_ITEM_NUM -- SAP订单行
	  ,d.RECEIVING_PROVINCE
	FROM DWROTD.DWR_WM_LADINGBILL_ITEM_F A1   -- SAP提货单行表
	INNER JOIN DWIOTD.DWI_WM_LADINGBILL_HEADER_T A2  -- SAP提货单头表
	ON A2.DELIVERY = A1.DELIVERY
	AND A2.DEL_FLAG = 'N'
	INNER JOIN dm.DM_LTC_ORDER_PLAN_SIGN_T D
	ON A2.DELIVERY = D.externno
	AND A1.DELIVERY_PROJECT = D.EXTERNLINENO
	where A1.DEL_FLAG = 'N'
),
tmp_data02 as(
SELECT  SALE_ORDER_NUM
       ,SALE_ORDER_ITEM_NUM
       ,concat_ws('|',collect_list(cast(RECEIVING_PROVINCE as string))) as RECEIVING_PROVINCE	   
  FROM tmp_data02_1
 GROUP BY SALE_ORDER_NUM
       ,SALE_ORDER_ITEM_NUM
)
INSERT OVERWRITE TABLE DWIOTD.DWI_SO_SALES_FORECAST_ORDER_T PARTITION(PERIOD_ID_M)
SELECT material as MATERIAL_NUM -- `产品编码`
      ,b.material_name as MATERIAL_NAME --`产品名称`
      ,doc_number as SALE_ORDER_HEAD_NO --`销售订单`
      ,s_ord_item as SALE_ORDER_LINE_NO --`销售订单行号`
      ,distr_chan as DIS_CHANNEL --`分销渠道`
      ,case when `/bic/zqdzh` = 'G103' then '家装大包'
            when `/bic/zqdzh` = 'G102' then '工程'
            when `/bic/zqdzh` = 'G105' then '设计师'
            when `/bic/zqdzh` = 'G101' then '零售'
             end as four_channel --`四大渠道`
      ,z_bhmj as STOCKUP_AREA_QTY -- `备货面积`
      ,z_kpmj as BILL_AREA_QTY --`开票面积`
      ,case when h.lve2_code is null then
			CASE 
			WHEN `/BIC/ZPRODH3` in('1008006','1008005') THEN '原石'
			WHEN `/BIC/ZPRODH3` in('1017012') THEN '中板'
			WHEN `/BIC/ZPRODH3` IN ('4013007','5502002','2290002','4501001','5101002','5106004') THEN '其他品类'
			END    
	  else h.lve2_code
		end as PRODUCT_LEVEL -- `产品层次`
      ,`/bic/zww010` as PRODUCT_AREA_CODE --`产区编码`
      ,g.product_area_name as PRODUCT_AREA_NAME -- `产区名称`
      ,`/bic/z_fhrq` as ACTUAL_GOODS_MOVE_DATE --  `实际发货时间`
      ,f.txtsh as ACTUAL_DELIVER_FACTORY --`实际交付工厂`
      ,size_dim as DIMENSION -- `规格`
      ,sold_to as CUST_NUM --`客户编码`
      ,c.cust_name as CUST_NAME -- `客户名称`
      ,c.region_province_name as CUST_ADDRESS_PROVINCE -- `客户所在地（省份）`
      ,a.SALES_GRP as SALES_GROUP_NUM -- `销售组编码`
      ,case when a.SALES_GRP = 'Z67' then '总裁虚拟组'
            when a.SALES_GRP = 'ZSG' then '家居渠道'
            when a.SALES_GRP = 'ZS4' then '战略工程中心'
            when a.SALES_GRP in('913','933','935','917','912','921','936','922','934','915') then '国际营销板块'
            else i.lev5_name
             end as SALES_GROUP_NAME -- `销售组`
      ,t.BILL_NO as DELIVERY_METHOD -- `配送方式`
      ,t.DEST_LOCATION as DELIVERY_DESTINATION_PROVINCE -- `送货目的地（省份）`
      ,t2.RECEIVING_PROVINCE as delivery_province
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
      ,substr(`/bic/z_fhrq`,1,6) as PERIOD_ID_M
FROM SDI.SDI_BIC_AZMSD_O2271_1009 a
left join dwrdim.DWR_DIM_MATERIAL_GENERAL_D b
on a.material = b.material_num 
and b.del_flag = 'N'
and b.end_date = '4712-12-31'
left join dwrdim.dwr_dim_cust_general_d c
on a.sold_to = c.cust_num 
and c.del_flag = 'N'
and c.end_date = '4712-12-31'
left join tmp_data01 t
on a.doc_number = t.VBELN
and a.s_ord_item = t.POSNR
left join SDI.SDI_BI0_TPLANT_1009 f
on a.plant = f.plant
left join (select distinct sale_factory_code,product_area_name
             from DWRDIM.DWR_DIM_PRODUCT_AREA_D) g
on a.plant = g.sale_factory_code
and g.sale_factory_code not in('7720','8A20')
left join DM.DM_RPT_PROD_LEVEL_T h
on a.`/bic/zprodh3` = h.prod_code3
left join dm.DM_RPT_SALE_GRP_T i
on a.SALES_GRP = i.sale_grp 
left join tmp_data02 t2
ON t2.SALE_ORDER_NUM = a.doc_number 
AND t2.SALE_ORDER_ITEM_NUM = a.s_ord_item 
where B.EXTERNAL_MATL_GROUP_CODE = '10010'
AND B.MATL_GROUP_CODE = '1000010'
and `/bic/z_fhrq` > '20220101'
and substr(`/bic/z_fhrq`,1,6) = REPLACE(SUBSTR(CURRENT_DATE - interval 1 days,1,7),'-','') ;
