--SDI_FORMTABLE_MAIN_829_1004 到 DWI_SO_SANITARY_QUO_OA_APR_T
--DEA-A1-005-洁具询价报价OA审批流程主表(工程及家装)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SANITARY_QUO_OA_APR_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.REQUESTID AS REQUEST_ID,
  SRC.LSH AS OA_CODE,
  SRC.ZDAMLE AS WHOLE_A_GAIN_LOSS_AMT,
  SRC.ZDBMLE AS WHOLE_B_GAIN_LOSS_AMT,
  SRC.ZJJE AS TOTAL_AMT,
  SRC.HTFJ AS CONTRACT_ANNEX,
  SRC.FHXX AS RETURN_INFO,
  SRC.BJDH AS SAP_QUOTE_NUM,
  SRC.BJDYXQ AS QUOTE_VALID_DATE,
  SRC.SQRQ AS APPLICATION_DATE,
  SRC.BBYXQ AS INSURANCE_DATE,
  SRC.XMMC AS PROJ_NAME,
  SRC.GCBBH AS PROJ_REP_NUM1,
  SRC.GCBBH2 AS PROJ_REP_NUM2,
  SRC.WBSH AS WBS,
  SRC.KHJC AS CUST_NAME,
  CASE
    WHEN SRC.KHBM2 IS NULL
    OR SRC.KHBM2 = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID2,
  --客户ID2
  CASE
    WHEN SRC.KHBM2 IS NULL
    OR SRC.KHBM2 = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM2,
  --客户编码2
  SRC.JGCBBH AS OLD_PROJ_REP_NUM,
  SRC.LXDH AS CONTACT_PHONE,
  SRC.SQR AS APPLY_USER,
  SRC.XSBM AS SALES_DEPT,
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN -999999
    ELSE NVL(CUST1.CUST_ID, -999998)
  END AS CUST_ID1,
  --客户ID1
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN 'SNULL'
    ELSE NVL(CUST1.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM1,
  --客户编码1
  SRC.SQBM AS APPLY_DEPT,
  SRC.FHXX2 AS OSS_RETURN_INFO,
  SRC.GCLB2 AS PROJ_CLASS2,
  SRC.SX AS PROPERTY,
  CASE
    WHEN SRC.GSDM IS NULL
    OR SRC.GSDM = '' THEN -999999
    ELSE NVL(COMP.COMPANY_ID, -999998)
  END AS COMPANY_ID,
  --公司ID
  CASE
    WHEN SRC.GSDM IS NULL
    OR SRC.GSDM = '' THEN 'SNULL'
    ELSE NVL(COMP.COMPANY_CODE, 'NOT EXISTS')
  END AS COMPANY_CODE,
  --公司代码
  WGBSPRY03 AS APP_NODE_ASSESS_EMP_ID,
  QD AS CHANNEL,
  XSZZ AS SALE_ORG_CODE,
  BMLEDXSFXYN AS GROSS_PROFIT_LESS0_FLAG,
  BSSFDD08JD AS CONFIRM_NODE_FLAG,
  CHCKLS AS APP_DISCOUNT,
  FLAG AS FLAG,
  SFQDHT AS SIGN_CONTRACT_FLAG,
  SFXQKH AS HAVED_CUST_NUM_FLAG,
  XXJBJYWFW AS QUOTE_BUSINESS_SCOPE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_829_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.KHBM2 = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST1 --客户主数据表
  ON SRC.KHBM = CUST1.CUST_NUM
  AND CUST1.END_DATE = '4712-12-31'
  AND CUST1.DEL_FLAG = 'N'
  AND CUST1.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_COMPANY_T COMP --公司主数据表
  ON SRC.GSDM = COMP.COMPANY_CODE
  AND COMP.END_DATE = '4712-12-31'
  AND COMP.DEL_FLAG = 'N'
  AND COMP.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.REQUEST_ID,
  DWI.OA_CODE,
  DWI.WHOLE_A_GAIN_LOSS_AMT,
  DWI.WHOLE_B_GAIN_LOSS_AMT,
  DWI.TOTAL_AMT,
  DWI.CONTRACT_ANNEX,
  DWI.RETURN_INFO,
  DWI.SAP_QUOTE_NUM,
  DWI.QUOTE_VALID_DATE,
  DWI.APPLICATION_DATE,
  DWI.INSURANCE_DATE,
  DWI.PROJ_NAME,
  DWI.PROJ_REP_NUM1,
  DWI.PROJ_REP_NUM2,
  DWI.WBS,
  DWI.CUST_NAME,
  DWI.CUST_ID2,
  DWI.CUST_NUM2,
  DWI.OLD_PROJ_REP_NUM,
  DWI.CONTACT_PHONE,
  DWI.APPLY_USER,
  DWI.SALES_DEPT,
  DWI.CUST_ID1,
  DWI.CUST_NUM1,
  DWI.APPLY_DEPT,
  DWI.OSS_RETURN_INFO,
  DWI.PROJ_CLASS2,
  DWI.PROPERTY,
  DWI.COMPANY_ID,
  DWI.COMPANY_CODE,
  DWI.APP_NODE_ASSESS_EMP_ID,
  DWI.CHANNEL,
  DWI.SALE_ORG_CODE,
  DWI.GROSS_PROFIT_LESS0_FLAG,
  DWI.CONFIRM_NODE_FLAG,
  DWI.APP_DISCOUNT,
  DWI.FLAG,
  DWI.SIGN_CONTRACT_FLAG,
  DWI.HAVED_CUST_NUM_FLAG,
  DWI.QUOTE_BUSINESS_SCOPE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SANITARY_QUO_OA_APR_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_829_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
