--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP交货到期表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_DELIVERY_DUE_SAP_T
SELECT
  SV.MANDT AS GROUP_NUM,
  SV.VBELN AS SAL_VCH_NUM,
  CASE
    WHEN SV.WADAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.WADAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.WADAT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS SHIPPING_DATE,
  R1.CONFIG_CODE AS RECEIVE_POINTS,
  R2.CONFIG_CODE AS ROUTE,
  R3.CONFIG_CODE AS SAL_VCH_TYPE,
  SV.KUNNR AS CUST_NUM,
  SV.KUNWE AS CUST_SRV_NUM,
  SV.ADRNR_AG AS CUST_SAL_ADDR_ID,
  SV.ADRNR_WE AS CUST_SRV_ADDR_ID,
  R4.CONFIG_CODE AS SAL_ORG_CODE,
  R5.CONFIG_CODE AS DIS_CHANNEL,
  R6.CONFIG_CODE AS PROD_GROUP,
  R7.CONFIG_CODE AS SAL_OFFICE_CODE,
  R8.CONFIG_CODE AS SAL_ORG_GROUP,
  SV.BRGEW AS PROJ_GROSS_WEIGHT,
  SV.GEWEI AS WEIGHT_UNIT,
  SV.VOLUM AS PROJ_VOL,
  CASE
    WHEN SV.LEDAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.LEDAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.LEDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS DELIVERY_CREATE_DATE,
  SV.VOLEH AS VOL_UNIT,
  SV.BEARZ AS DEAL_DAYS,
  SV.KZAZU AS BACK_SUM_FLAG,
  R9.CONFIG_CODE AS DELIVERY_PRIORITY,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_VEPVG_1001 SV --关联通用配置表，将SDI表相关字段进行转码处理
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1 ON SV.MANDT = R1.GROUP_NUM
  AND SV.VSTEL = R1.SRC_SYS_CONFIG_CODE
  AND UPPER(R1.SOURCE_CODE) = 'TVSTT' --机构单位：装运点
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2 ON SV.MANDT = R2.GROUP_NUM
  AND SV.ROUTE = R2.SRC_SYS_CONFIG_CODE
  AND UPPER(R2.SOURCE_CODE) = 'TVROT' --工艺路线
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3 ON SV.MANDT = R3.GROUP_NUM
  AND SV.AUART = R3.SRC_SYS_CONFIG_CODE
  AND UPPER(R3.SOURCE_CODE) = 'TVAKT' --销售凭证类型
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R4 ON SV.MANDT = R4.GROUP_NUM
  AND SV.VKORG = R4.SRC_SYS_CONFIG_CODE
  AND UPPER(R4.SOURCE_CODE) = 'TVKOT' --机构单位：销售机构
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R5 ON SV.MANDT = R5.GROUP_NUM
  AND SV.VTWEG = R5.SRC_SYS_CONFIG_CODE
  AND UPPER(R5.SOURCE_CODE) = 'TVTWT' --机构单位：分销渠道
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R6 ON SV.MANDT = R6.GROUP_NUM
  AND SV.SPART = R6.SRC_SYS_CONFIG_CODE
  AND UPPER(R6.SOURCE_CODE) = 'TSPAT' --组织单位: 销售分部
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R7 ON SV.MANDT = R7.GROUP_NUM
  AND SV.VKBUR = R7.SRC_SYS_CONFIG_CODE
  AND UPPER(R7.SOURCE_CODE) = 'TVKBT' --机构单位：销售办公室
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R8 ON SV.MANDT = R8.GROUP_NUM
  AND SV.VKGRP = R8.SRC_SYS_CONFIG_CODE
  AND UPPER(R8.SOURCE_CODE) = 'TVGRT' --组织单位: 销售组
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R9 ON SV.MANDT = R9.GROUP_NUM
  AND SV.LPRIO = R9.SRC_SYS_CONFIG_CODE
  AND UPPER(R9.SOURCE_CODE) = 'TPRIT' --客户: 交货优先
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SHIPPING_DATE AS SHIPPING_DATE,
  DWI.RECEIVE_POINTS AS RECEIVE_POINTS,
  DWI.ROUTE AS ROUTE,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.CUST_SRV_NUM AS CUST_SRV_NUM,
  DWI.CUST_SAL_ADDR_ID AS CUST_SAL_ADDR_ID,
  DWI.CUST_SRV_ADDR_ID AS CUST_SRV_ADDR_ID,
  DWI.SAL_ORG_CODE AS SAL_ORG_CODE,
  DWI.DIS_CHANNEL AS DIS_CHANNEL,
  DWI.PROD_GROUP AS PROD_GROUP,
  DWI.SAL_OFFICE_CODE AS SAL_OFFICE_CODE,
  DWI.SAL_ORG_GROUP AS SAL_ORG_GROUP,
  DWI.PROJ_GROSS_WEIGHT AS PROJ_GROSS_WEIGHT,
  DWI.WEIGHT_UNIT AS WEIGHT_UNIT,
  DWI.PROJ_VOL AS PROJ_VOL,
  DWI.DELIVERY_CREATE_DATE AS DELIVERY_CREATE_DATE,
  DWI.VOL_UNIT AS VOL_UNIT,
  DWI.DEAL_DAYS AS DEAL_DAYS,
  DWI.BACK_SUM_FLAG AS BACK_SUM_FLAG,
  DWI.DELIVERY_PRIORITY AS DELIVERY_PRIORITY,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DELIVERY_DUE_SAP_T DWI
  LEFT JOIN (
    SELECT
      SV.MANDT AS MANDT,
      SV.VBELN AS VBELN,
      CASE
        WHEN SV.WADAT = '00000000' THEN FROM_UNIXTIME(
          UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
          'yyyy-MM-dd'
        )
        WHEN SUBSTR(SV.WADAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
          UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
          'yyyy-MM-dd'
        )
        ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.WADAT, 'yyyyMMdd'), 'yyyy-MM-dd')
      END AS WADAT,
      R1.CONFIG_CODE AS VSTEL,
      R2.CONFIG_CODE AS ROUTE
    FROM
      SDI.SDI_VEPVG_1001 SV
      LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1 ON SV.MANDT = R1.GROUP_NUM
      AND SV.VSTEL = R1.SRC_SYS_CONFIG_CODE
      AND UPPER(R1.SOURCE_CODE) = 'TVSTT'
      LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2 ON SV.MANDT = R2.GROUP_NUM
      AND SV.ROUTE = R2.SRC_SYS_CONFIG_CODE
      AND UPPER(R2.SOURCE_CODE) = 'TVROT'
  ) TMP ON DWI.GROUP_NUM = TMP.MANDT
  AND DWI.SAL_VCH_NUM = TMP.VBELN
  AND NVL(SUBSTR(DWI.SHIPPING_DATE, 1, 10), 'SNULL') = NVL(SUBSTR(TMP.WADAT, 1, 10), 'SNULL')
  AND NVL(DWI.RECEIVE_POINTS, 'SNULL') = NVL(TMP.VSTEL, 'SNULL')
  AND NVL(DWI.ROUTE, 'SNULL') = NVL(TMP.ROUTE, 'SNULL')
WHERE
  1 = 1
  AND TMP.VBELN IS NULL;
