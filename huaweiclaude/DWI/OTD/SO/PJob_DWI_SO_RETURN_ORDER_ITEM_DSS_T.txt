--DSS退货单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RETURN_ORDER_ITEM_DSS_T
SELECT
  SBROI.ID AS ID,
  SBROI.UPDATE_TIME AS UPDATE_TIME,
  SBROI.CREATE_TIME AS CREATE_TIME,
  SBROI.RETURN_ORDER_ID AS RETURN_ORDER_ID,
  SBROI.ORDER_ITEM_ID AS ORDER_ITEM_ID,
  SBROI.PRICE AS PRICE,
  SBROI.FINAL_PRICE AS FINAL_PRICE,
  CASE
    WHEN SBROI.FACTORY_CODE IS NULL OR SBROI.FACTORY_CODE = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SBROI.FACTORY_CODE IS NULL OR SBROI.FACTORY_CODE = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  SBROI.QUANTITY AS QUANTITY,
  CASE
    WHEN SBROI.PRODUCT_CODE IS NULL OR SBROI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBROI.PRODUCT_CODE IS NULL OR SBROI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBROI.PRODUCT_NAME AS PRODUCT_NAME,
  SBROI.DISCOUNT_RATE AS DISCOUNT_RATE,
  SBROI.BATCH_CODE AS BATCH_CODE,
  SBROI.RETURN_NUM AS RETURN_NUM,
  SBROI.UPDATE_USER AS UPDATE_USER,
  SBROI.CREATE_USER AS CREATE_USER,
  SBROI.DSDD AS DSDD,
  SBROI.INVENTORY_CODE AS INVENTORY_CODE,
  SBROI.CREATE_DEPT AS CREATE_DEPT,
  SBROI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  SBROI.ITEM_NUMBER AS ITEM_NUMBER,
  SBROI.HANDLING_FEE AS HANDLING_FEE,
  SBROI.INTEGRAL AS INTEGRAL,
  SBROI.PRODUCT_UNIT AS PRODUCT_UNIT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_RETURN_ORDER_ITEM_1008 SBROI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBROI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800' 
  --关联工厂主数据表取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SBROI.FACTORY_CODE = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.RETURN_ORDER_ID AS RETURN_ORDER_ID,
  DWI.ORDER_ITEM_ID AS ORDER_ITEM_ID,
  DWI.PRICE AS PRICE,
  DWI.FINAL_PRICE AS FINAL_PRICE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.QUANTITY AS QUANTITY,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.PRODUCT_NAME AS PRODUCT_NAME,
  DWI.DISCOUNT_RATE AS DISCOUNT_RATE,
  DWI.BATCH_CODE AS BATCH_CODE,
  DWI.RETURN_NUM AS RETURN_NUM,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.DSDD AS DSDD,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.HANDLING_FEE AS HANDLING_FEE,
  DWI.INTEGRAL AS INTEGRAL,
  DWI.PRODUCT_UNIT AS PRODUCT_UNIT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RETURN_ORDER_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_RETURN_ORDER_ITEM_1008 SBROI ON DWI.ID = SBROI.ID
WHERE
  SBROI.ID IS NULL;
