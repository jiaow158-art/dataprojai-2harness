set hive.mapjoin.smalltable.filesize=100000000;

--SAP销售凭证合作伙伴表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SAL_PARTNER_SAP_T
SELECT
  SV.MANDT AS GROUP_NUM,
  SV.VBELN AS SAL_VCH_NUM,
  SV.POSNR AS SAL_VCH_ITEM_NUM,
  SV.PARVW AS PARTNER_FUNC,
  CASE
    WHEN SV.KUNNR IS NULL OR SV.KUNNR = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,--客户ID
  CASE
    WHEN SV.KUNNR IS NULL OR SV.KUNNR = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,--客户编码
  SV.ADRNR AS ADDRESS_ID,
  SV.ADRDA AS ADDRESS_FLAG,
  SV.KNREF AS BUSI_PARTNER_CUST_DESC,
  SV.PERNR AS PERSON_NUM,
  SV.LAND1 AS CNTRY_REGION_CODE,
  SV.LZONE AS GOODS_SEND_DESTINATION_CODE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_VBPA_1001 SV 
  --关联客户主数据表取客户ID
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST ON SV.MANDT = CUST.GROUP_NUM
  AND SV.KUNNR = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
;
