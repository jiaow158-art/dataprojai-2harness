--DSS报价单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_QUOTE_ITEM_DSS_T
SELECT
  SBQI.ID AS ID,
  SBQI.CREATE_TIME AS CREATE_TIME,
  SBQI.UPDATE_TIME AS UPDATE_TIME,
  SBQI.OPERATE_LOG AS OPERATE_LOG,
  SBQI.QUOTATION_ID AS QUOTATION_ID,
  SBQI.DISCOUNT AS DISCOUNT,
  SBQI.APPROVAL_DISCOUNT AS APPROVAL_DISCOUNT,
  SBQI.APPLY_TOTAL_AMOUNT AS APPLY_TOTAL_AMT,
  SBQI.APPROVAL_TOTAL_AMOUNT AS APPROVAL_TOTAL_AMT,
  SBQI.APPROVAL_PRICE AS APPROVAL_PRICE,
  SBQI.APPLY_PRICE AS APPLY_PRICE,
  SBQI.PRICE AS PRICE,
  CASE
    WHEN SBQI.FACTORY_CODE IS NULL OR SBQI.FACTORY_CODE = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SBQI.FACTORY_CODE IS NULL OR SBQI.FACTORY_CODE = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  CASE
    WHEN SBQI.PRODUCT_CODE IS NULL OR SBQI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBQI.PRODUCT_CODE IS NULL OR SBQI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBQI.SELLING_PRICE AS SELLING_PRICE,
  SBQI.PRODUCT_NAME AS PRODUCT_NAME,
  SBQI.DIRECT_SALES_COST AS DIRECT_SALES_COST,
  SBQI.QUANTITY AS QUANTITY,
  SBQI.TRANSFORM_QTY AS TRANSFORM_QTY,
  SBQI.APPROVAL_QUANTITY AS APPROVAL_QUANTITY,
  SBQI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  SBQI.BATCH_CODE AS BATCH_CODE,
  SBQI.SALES_AREA AS SALES_AREA,
  SBQI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  SBQI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  SBQI.LEVEL AS LEVEL,
  SBQI.UPDATE_USER AS UPDATE_USER,
  SBQI.WAREHOUSE_AGE AS WAREHOUSE_AGE,
  SBQI.CREATE_USER AS CREATE_USER,
  SBQI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  SBQI.CREATE_DEPT AS CREATE_DEPT,
  SBQI.ITEM_NUMBER AS ITEM_NUMBER,
  SBQI.DESIGNER_FD_POINTS AS DESIGNER_FD_POINTS,
  SBQI.ACREAGE AS ACREAGE,
  SBQI.INVENTORY_CODE AS INVENTORY_CODE,
  SBQI.LAST_PRICE AS LAST_PRICE,
  SBQI.UNIT AS UNIT,
  SBQI.QCYY AS QCYY,
  SBQI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  SBQI.PROCESSING_UNIT AS PROCESSING_UNIT,
  SBQI.STATUS AS STATUS,
  SBQI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  SBQI.COMMISSION_POINTS AS COMMISSION_POINTS,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_QUOTATION_ITEM_1008 SBQI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBQI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800' 
  --关联工厂主数据表取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SBQI.FACTORY_CODE = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.OPERATE_LOG AS OPERATE_LOG,
  DWI.QUOTATION_ID AS QUOTATION_ID,
  DWI.DISCOUNT AS DISCOUNT,
  DWI.APPROVAL_DISCOUNT AS APPROVAL_DISCOUNT,
  DWI.APPLY_TOTAL_AMT AS APPLY_TOTAL_AMT,
  DWI.APPROVAL_TOTAL_AMT AS APPROVAL_TOTAL_AMT,
  DWI.APPROVAL_PRICE AS APPROVAL_PRICE,
  DWI.APPLY_PRICE AS APPLY_PRICE,
  DWI.PRICE AS PRICE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.SELLING_PRICE AS SELLING_PRICE,
  DWI.PRODUCT_NAME AS PRODUCT_NAME,
  DWI.DIRECT_SALES_COST AS DIRECT_SALES_COST,
  DWI.QUANTITY AS QUANTITY,
  DWI.TRANSFORM_QTY AS TRANSFORM_QTY,
  DWI.APPROVAL_QUANTITY AS APPROVAL_QUANTITY,
  DWI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  DWI.BATCH_CODE AS BATCH_CODE,
  DWI.SALES_AREA AS SALES_AREA,
  DWI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  DWI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  DWI.LEVEL AS LEVEL,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.WAREHOUSE_AGE AS WAREHOUSE_AGE,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.DESIGNER_FD_POINTS AS DESIGNER_FD_POINTS,
  DWI.ACREAGE AS ACREAGE,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.LAST_PRICE AS LAST_PRICE,
  DWI.UNIT AS UNIT,
  DWI.QCYY AS QCYY,
  DWI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  DWI.PROCESSING_UNIT AS PROCESSING_UNIT,
  DWI.STATUS AS STATUS,
  DWI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  DWI.COMMISSION_POINTS AS COMMISSION_POINTS,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_QUOTE_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_QUOTATION_ITEM_1008 SBQI ON DWI.ID = SBQI.ID
WHERE
  SBQI.ID IS NULL;
