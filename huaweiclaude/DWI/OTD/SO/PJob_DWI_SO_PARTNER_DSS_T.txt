--DSS合作伙伴
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PARTNER_DSS_T
SELECT
  SBP.ID AS ID,
  SBP.CUSTOMER_ID AS CUST_ID,
  SBP.NAME1 AS NAME1,
  SBP.DESCRIPTION AS DESCRIPTION,
  SBP.SALES_ORGANIZATION_CODE AS SALES_ORG_CODE,
  SBP.UPDATE_TIME AS UPDATE_TIME,
  SBP.DISTRIBUTION_CHANNEL AS DISTRIBUTION_CHANNEL,
  SBP.DISTRIBUTION_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  SBP.PARTNER_FUNCTION AS PARTNER_FUNCTION,
  SBP.NAME AS NAME,
  SBP.PRODUCT_GROUP_CODE AS PRODUCT_GROUP_CODE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_PARTNER_1008 SBP 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.CUST_ID AS CUST_ID,
  DWI.NAME1 AS NAME1,
  DWI.DESCRIPTION AS DESCRIPTION,
  DWI.SALES_ORG_CODE AS SALES_ORG_CODE,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.DISTRIBUTION_CHANNEL AS DISTRIBUTION_CHANNEL,
  DWI.DISTR_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  DWI.PARTNER_FUNCTION AS PARTNER_FUNCTION,
  DWI.NAME AS NAME,
  DWI.PRODUCT_GROUP_CODE AS PRODUCT_GROUP_CODE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PARTNER_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_PARTNER_1008 SBP ON DWI.ID = SBP.ID
WHERE
  SBP.ID IS NULL;
