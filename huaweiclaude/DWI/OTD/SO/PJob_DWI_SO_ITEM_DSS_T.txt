--DSS销售订单行表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_ITEM_DSS_T
SELECT
  SBOI.ID AS DSS_SO_ITEM_ID,
  SBOI.ORDER_ID AS DSS_SO_ID,
  SBOI.QUOTATION_ITEM_ID AS DSS_QUOTE_ITEM_ID,
  SBOI.BATCH_CODE AS DSS_BATCH_CODE,
  SBOI.REMARK AS DSS_REMARK,
  SBOI.PRICE AS DSS_BASE_PRICE,
  SBOI.DESIGNER_FD_POINTS AS DSS_DESIGNER_POINTS,
  SBOI.CREATE_DEPT AS CREATE_DEPT,
  SBOI.UPDATE_USER AS UPDATE_USER,
  CASE
    WHEN SBOI.PRODUCT_CODE IS NULL OR SBOI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBOI.PRODUCT_CODE IS NULL OR SBOI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBOI.PRODUCT_NAME AS DSS_MATERIAL_NAME,
  SBOI.PRODUCT_UNIT AS DSS_MATERIAL_UNIT,
  SBOI.INVENTORY_CODE AS INVENTORY_CODE,
  SBOI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  SBOI.QUANTITY AS QUANTITY,
  SBOI.DISCOUNT_RATE AS DISCOUNT_RATE,
  SBOI.OPERATE_LOG AS OPERATE_LOG,
  SBOI.DIRECT_COST AS DIRECT_COST,
  SBOI.POINT_DISCOUNT AS POINT_DISCOUNT,
  SBOI.FINAL_PRICE AS FINAL_PRICE,
  SBOI.EXPECT_DATE AS EXPECT_DATE,
  SBOI.WEIGHT AS WEIGHT,
  SBOI.TOTAL_AMOUNT AS TOTAL_AMT,
  SBOI.ITEM_NUMBER AS ITEM_NUMBER,
  SBOI.STATUS AS STATUS,
  SBOI.DEFAULT_EXPECT_DATE AS DEFAULT_EXPECT_DATE,
  SBOI.GDCP_POINT_FLAG AS ADV_PROD_POINT_ACCRUAL,
  SBOI.GDCP_POINT_SCORE AS ADV_PROD_POINT_SCORE,
  SBOI.YDJF_POINT_FLAG AS MONTHS_POINT_ACCRUAL,
  SBOI.YDJF_POINT_SCORE AS MONTHS_POINT_SCORE,
  SBOI.JDJF_POINT_FLAG AS QUARTER_POINT_ACCRUAL,
  SBOI.JDJF_POINT_SCORE AS QUARTER_POINT_SCORE,
  SBOI.NDJF_POINT_FLAG AS YEARS_POINT_ACCRUAL,
  SBOI.NDJF_POINT_SCORE AS YEARS_POINT_SCORE,
  SBOI.ZXJF_POINT_FLAG AS RENOVAT_POINT_ACCRUAL,
  SBOI.ZXJF_POINT_SCORE AS DIS_ADVERT_POINT_SCORE,
  SBOI.JXGGJF_POINT_FLAG AS DIS_ADVERT_POINT_ACCRUAL,
  SBOI.JXGG_POINT_SCORE AS POINT_SCORE,
  SBOI.XCZL_POINT_FLAG AS TRAV_BROCH_POINT_ACCRUAL,
  SBOI.XCZL_POINT_SCORE AS TRAV_BROCH_POINT_SCORE,
  SBOI.QYGG_POINT_FLAG AS AREA_ADVERT_POINT_ACCRUAL,
  SBOI.QYGG_POINT_SCORE AS AREA_ADVERT_POINT_SCORE,
  SBOI.DQGG_POINT_FLAG AS LARGE_AREA_POINT_ACCRUAL,
  SBOI.DQGG_POINT_SCORE AS LARGE_AREA_POINT_SCORE,
  SBOI.COMMISSION_POINTS AS COMMISSION_POINTS,
  SBOI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  SBOI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  SBOI.PROCESSING_UNIT AS PROCESSING_UNIT,
  SBOI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  SBOI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  SBOI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  SBOI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  SBOI.COLLECT_FREIGHT AS COLLECT_FREIGHT,
  SBOI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  SBOI.STAY_GOODS_ITEM_CODE AS STAY_GOODS_ITEM_CODE,
  SBOI.STAY_GOODS_DESCRIBE AS STAY_GOODS_DESCRIBE,
  SBOI.APPROVAL_PRICE AS APPROVAL_PRICE,
  SBOI.BRICK_AREA AS BRICK_AREA,
  SBOI.STAY_GOODS_NUM AS STAY_GOODS_NUM,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_ORDER_ITEM_1008 SBOI 
  --关联物料主数据表取物料ID
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SBOI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.DSS_SO_ITEM_ID AS DSS_SO_ITEM_ID,
  DWI.DSS_SO_ID AS DSS_SO_ID,
  DWI.DSS_QUOTE_ITEM_ID AS DSS_QUOTE_ITEM_ID,
  DWI.DSS_BATCH_CODE AS DSS_BATCH_CODE,
  DWI.DSS_REMARK AS DSS_REMARK,
  DWI.DSS_BASE_PRICE AS DSS_BASE_PRICE,
  DWI.DSS_DESIGNER_POINTS AS DSS_DESIGNER_POINTS,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.DSS_MATERIAL_NAME AS DSS_MATERIAL_NAME,
  DWI.DSS_MATERIAL_UNIT AS DSS_MATERIAL_UNIT,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  DWI.QUANTITY AS QUANTITY,
  DWI.DISCOUNT_RATE AS DISCOUNT_RATE,
  DWI.OPERATE_LOG AS OPERATE_LOG,
  DWI.DIRECT_COST AS DIRECT_COST,
  DWI.POINT_DISCOUNT AS POINT_DISCOUNT,
  DWI.FINAL_PRICE AS FINAL_PRICE,
  DWI.EXPECT_DATE AS EXPECT_DATE,
  DWI.WEIGHT AS WEIGHT,
  DWI.TOTAL_AMT AS TOTAL_AMT,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.STATUS AS STATUS,
  DWI.DEFAULT_EXPECT_DATE AS DEFAULT_EXPECT_DATE,
  DWI.ADV_PROD_POINT_ACCRUAL AS ADV_PROD_POINT_ACCRUAL,
  DWI.ADV_PROD_POINT_SCORE AS ADV_PROD_POINT_SCORE,
  DWI.MONTHS_POINT_ACCRUAL AS MONTHS_POINT_ACCRUAL,
  DWI.MONTHS_POINT_SCORE AS MONTHS_POINT_SCORE,
  DWI.QUARTER_POINT_ACCRUAL AS QUARTER_POINT_ACCRUAL,
  DWI.QUARTER_POINT_SCORE AS QUARTER_POINT_SCORE,
  DWI.YEARS_POINT_ACCRUAL AS YEARS_POINT_ACCRUAL,
  DWI.YEARS_POINT_SCORE AS YEARS_POINT_SCORE,
  DWI.RENOVAT_POINT_ACCRUAL AS RENOVAT_POINT_ACCRUAL,
  DWI.DIS_ADVERT_POINT_SCORE AS DIS_ADVERT_POINT_SCORE,
  DWI.DIS_ADVERT_POINT_ACCRUAL AS DIS_ADVERT_POINT_ACCRUAL,
  DWI.POINT_SCORE AS POINT_SCORE,
  DWI.TRAV_BROCH_POINT_ACCRUAL AS TRAV_BROCH_POINT_ACCRUAL,
  DWI.TRAV_BROCH_POINT_SCORE AS TRAV_BROCH_POINT_SCORE,
  DWI.AREA_ADVERT_POINT_ACCRUAL AS AREA_ADVERT_POINT_ACCRUAL,
  DWI.AREA_ADVERT_POINT_SCORE AS AREA_ADVERT_POINT_SCORE,
  DWI.LARGE_AREA_POINT_ACCRUAL AS LARGE_AREA_POINT_ACCRUAL,
  DWI.LARGE_AREA_POINT_SCORE AS LARGE_AREA_POINT_SCORE,
  DWI.COMMISSION_POINTS AS COMMISSION_POINTS,
  DWI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  DWI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  DWI.PROCESSING_UNIT AS PROCESSING_UNIT,
  DWI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  DWI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  DWI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  DWI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  DWI.COLLECT_FREIGHT AS COLLECT_FREIGHT,
  DWI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  DWI.STAY_GOODS_ITEM_CODE AS STAY_GOODS_ITEM_CODE,
  DWI.STAY_GOODS_DESCRIBE AS STAY_GOODS_DESCRIBE,
  DWI.APPROVAL_PRICE AS APPROVAL_PRICE,
  DWI.BRICK_AREA AS BRICK_AREA,
  DWI.STAY_GOODS_NUM AS STAY_GOODS_NUM,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_ORDER_ITEM_1008 SBOI ON DWI.DSS_SO_ITEM_ID = SBOI.ID
WHERE
  SBOI.ID IS NULL;
