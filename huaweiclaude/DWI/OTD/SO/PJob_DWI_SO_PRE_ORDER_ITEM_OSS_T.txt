--SDI_CX_PRE_ORD_ITEM_1002 到 DWI_SO_PRE_ORDER_ITEM_OSS_T
--OSS准订单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PRE_ORDER_ITEM_OSS_T
SELECT
  SRC.ROW_ID AS ROW_ID,
  SRC.PLAN_ORDER_ITEM_ID AS PLAN_ORDER_ITEM_ID,
  SRC.QUOTA_ITEM_ID AS QUOTA_ITEM_ID,
  SRC.DB_LAST_UPD AS DB_LAST_UPDATE_DATE,
  SRC.CREATED AS CREATE_DATE,
  SRC.LAST_UPD AS LAST_UPDATE_DATE,
  SRC.ORDER_ID AS ORDER_ID,
  SRC.QTY AS QTY,
  SRC.SYH_PRICE AS SYH_PRICE,
  SRC.CURRENT_PRICE AS CURRENT_PRICE,
  SRC.FACTORY_ID AS SRC_FACTORY_ID,
  --源系统工厂ID
  CASE
    WHEN SRC.FACTORY_ID IS NULL
    OR SRC.FACTORY_ID = '' THEN -999999
    ELSE NVL(TMP.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.FACTORY_ID IS NULL
    OR SRC.FACTORY_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  SRC.PRODUCT_ID AS PRODUCT_ID,
  SRC.PRICE AS PRICE,
  SRC.SHIP_EXPEN AS SHIP_EXPEN,
  SRC.SHIP_CITY AS SHIP_CITY,
  SRC.LAST_UPD_BY AS LAST_UPDATE_BY,
  SRC.SERVER_EXPEN AS SERVER_EXPEN,
  SRC.SHIP_PROVINCE AS SHIP_PROVINCE,
  SRC.SHIP_TYPE AS SHIP_TYPE,
  SRC.STATUS AS STATUS,
  SRC.PROJECT_TYPE AS PROJECT_TYPE,
  SRC.CREATED_BY AS CREATED_BY,
  SRC.MODIFICATION_NUM AS MODIFICATION_NUM,
  SRC.CONFLICT_ID AS CONFLICT_ID,
  SRC.DB_LAST_UPD_SRC AS DB_LAST_UPDATE_DATE_SRC,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_PRE_ORD_ITEM_1002 SRC
  LEFT JOIN (
    SELECT
      M1.ROW_ID,
      FAC.FACTORY_ID,
      --工厂ID
      FAC.FACTORY_CODE --工厂编码
    FROM
      SDI.SDI_CX_FACTORY_1002 M1 --工厂编码表
      INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
      ON M1.FACTORY_CD = FAC.FACTORY_CODE
      AND FAC.END_DATE = '4712-12-31'
      AND FAC.DEL_FLAG = 'N'
      AND FAC.GROUP_NUM = '800'
  ) TMP ON SRC.FACTORY_ID = TMP.ROW_ID
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.PLAN_ORDER_ITEM_ID,
  DWI.QUOTA_ITEM_ID,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.ORDER_ID,
  DWI.QTY,
  DWI.SYH_PRICE,
  DWI.CURRENT_PRICE,
  DWI.SRC_FACTORY_ID,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.PRODUCT_ID,
  DWI.PRICE,
  DWI.SHIP_EXPEN,
  DWI.SHIP_CITY,
  DWI.LAST_UPDATE_BY,
  DWI.SERVER_EXPEN,
  DWI.SHIP_PROVINCE,
  DWI.SHIP_TYPE,
  DWI.STATUS,
  DWI.PROJECT_TYPE,
  DWI.CREATED_BY,
  DWI.MODIFY_QTY,
  DWI.CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PRE_ORDER_ITEM_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_PRE_ORD_ITEM_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
