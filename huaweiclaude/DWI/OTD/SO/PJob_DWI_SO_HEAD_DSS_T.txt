--DSS销售订单头表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_HEAD_DSS_T
SELECT
  SBO.ID AS DSS_SO_ID,
  SBO.ORDER_CODE AS DSS_SO_NUM,
  SBO.QUOTATION_ID AS DSS_QUOTE_ID,
  SBO.PURCHASE_CODE AS DSS_PO_NUM,
  SBO.CONTRACT_CODE AS DSS_CONTRACT_NUM,
  SBO.SHIP_TYPE AS DSS_TRANS_TYPE,
  SBO.SHIP_COMMENTS AS DSS_DELIVERY_NOTE,
  SBO.DELIVERY_ADDRESS AS DSS_DELIVERY_ADDRESS,
  SBO.CONSIGNEE_NAME AS DSS_CONSIGNEE_NAME,
  SBO.CONSIGNEE_PHONE AS DSS_CONSIGNEE_PHONE,
  SBO.CREATE_DEPT AS CREATE_DEPT,
  SBO.UPDATE_USER AS UPDATE_USER,
  SBO.ORDER_TYPE AS SO_TYPE,
  SBO.SAP_ORDER_TYPE AS SAP_ORDER_TYPE,
  SBO.SALES_ORG_CODE AS SAL_ORG_CODE,
  SBO.CHANNEL_CODE AS CHANNEL_CODE,
  CASE
    WHEN SBO.ACCOUNT_CODE IS NULL OR SBO.ACCOUNT_CODE = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,--客户ID
  CASE
    WHEN SBO.ACCOUNT_CODE IS NULL OR SBO.ACCOUNT_CODE = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,--客户编码
  SBO.ACCOUNT_NAME AS CUST_NAME,
  SBO.PAYER_CODE AS PAYER_CODE,
  SBO.PAYER_NAME AS PAYER_NAME,
  SBO.SUBMIT_DATE AS SUBMIT_DATE,
  SBO.STATUS AS STATUS,
  SBO.SALES_MAN_CODE AS SALES_MAN_CODE,
  SBO.SALES_MAN_NAME AS SALES_MAN_NAME,
  SBO.PRICE_GROUP AS PRICE_GROUP,
  SBO.SAP_CODE AS SAP_CODE,
  SBO.SAP_COMMENTS AS SAP_COMMENTS,
  SBO.CREDIT_AMOUNT AS CREDIT_AMT,
  SBO.CREDIT_TIME AS CREDIT_TIME,
  SBO.QUOTATION_CODE AS DSS_QUOTE_NUM,
  SBO.DETAIL_AMOUNT AS DETAIL_AMT,
  SBO.COMMENTS AS DSS_COMMENTS,
  SBO.STREET_ADDRESS AS STREET_ADDRESS,
  SBO.CONTACT_TYPE AS CONTACT_TYPE,
  SBO.SOURCE_CHANNEL AS SOURCE_CHANNEL,
  SBO.RESIDENT_DESIGNER AS RESIDENT_DESIGNER,
  SBO.PERCENTAGE_OF_RESIDENT_DESIGNERS AS RESIDENT_DESIG_PERC_RT,
  SBO.DELIVERY_DATE AS DSS_DELIVERY_DATE,
  SBO.OWNER_PHONE AS OWNER_PHONE,
  SBO.EXTRACT_METHOD AS EXTRACT_METHOD,
  SBO.UN_CAR_INFO AS UN_CAR_INFO,
  SBO.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  SBO.APPLICANT AS APPLICANT,
  SBO.ORDER_GROUP_CODE AS ORDER_GROUP_CODE,
  SBO.APPLICANT_NAME AS APPLICANT_NAME,
  SBO.RESIDENT_DESIGNER_NAME AS RESIDENT_DESIGNER_NAME,
  SBO.OWNER_NAME AS OWNER_NAME,
  SBO.OTHER_REMARK AS OTHER_REMARK,
  SBO.SALES_MAN_ONE_CODE AS SALES_MAN_ONE_CODE,
  SBO.SALES_MAN_ONE_NAME AS SALES_MAN_ONE_NAME,
  SBO.APPROVAL_COMMENTS AS APPROVAL_COMMENTS,
  SBO.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  SBO.STORE_SALESMAN AS STORE_SALESMAN,
  SBO.STORE_SALESMAN_PHONE AS STORE_SALESMAN_PHONE,
  SBO.STOCKPILE_TYPE AS STOCKPILE_TYPE,
  SBO.MEMBER_INFORMATION AS MEMBERS_INFO,
  SBO.LOOK_UP AS FORM_TITLE,
  SBO.UN_CAR_INFO_S_H AS SH_UNLOAD_INFO,
  SBO.CITY_S_H AS SH_CITY_AREA,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_ORDER_1008 SBO 
  --关联客户主数据表取客户ID、编码
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST ON SBO.ACCOUNT_CODE = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.DSS_SO_ID AS DSS_SO_ID,
  DWI.DSS_SO_NUM AS DSS_SO_NUM,
  DWI.DSS_QUOTE_ID AS DSS_QUOTE_ID,
  DWI.DSS_PO_NUM AS DSS_PO_NUM,
  DWI.DSS_CONTRACT_NUM AS DSS_CONTRACT_NUM,
  DWI.DSS_TRANS_TYPE AS DSS_TRANS_TYPE,
  DWI.DSS_DELIVERY_NOTE AS DSS_DELIVERY_NOTE,
  DWI.DSS_DELIVERY_ADDRESS AS DSS_DELIVERY_ADDRESS,
  DWI.DSS_CONSIGNEE_NAME AS DSS_CONSIGNEE_NAME,
  DWI.DSS_CONSIGNEE_PHONE AS DSS_CONSIGNEE_PHONE,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.SO_TYPE AS SO_TYPE,
  DWI.SAP_ORDER_TYPE AS SAP_ORDER_TYPE,
  DWI.SAL_ORG_CODE AS SAL_ORG_CODE,
  DWI.CHANNEL_CODE AS CHANNEL_CODE,
  DWI.CUST_ID AS CUST_ID,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.CUST_NAME AS CUST_NAME,
  DWI.PAYER_CODE AS PAYER_CODE,
  DWI.PAYER_NAME AS PAYER_NAME,
  DWI.SUBMIT_DATE AS SUBMIT_DATE,
  DWI.STATUS AS STATUS,
  DWI.SALES_MAN_CODE AS SALES_MAN_CODE,
  DWI.SALES_MAN_NAME AS SALES_MAN_NAME,
  DWI.PRICE_GROUP AS PRICE_GROUP,
  DWI.SAP_CODE AS SAP_CODE,
  DWI.SAP_COMMENTS AS SAP_COMMENTS,
  DWI.CREDIT_AMT AS CREDIT_AMT,
  DWI.CREDIT_TIME AS CREDIT_TIME,
  DWI.DSS_QUOTE_NUM AS DSS_QUOTE_NUM,
  DWI.DETAIL_AMT AS DETAIL_AMT,
  DWI.DSS_COMMENTS AS DSS_COMMENTS,
  DWI.STREET_ADDRESS AS STREET_ADDRESS,
  DWI.CONTACT_TYPE AS CONTACT_TYPE,
  DWI.SOURCE_CHANNEL AS SOURCE_CHANNEL,
  DWI.RESIDENT_DESIGNER AS RESIDENT_DESIGNER,
  DWI.RESIDENT_DESIG_PERC_RT AS RESIDENT_DESIG_PERC_RT,
  DWI.DSS_DELIVERY_DATE AS DSS_DELIVERY_DATE,
  DWI.OWNER_PHONE AS OWNER_PHONE,
  DWI.EXTRACT_METHOD AS EXTRACT_METHOD,
  DWI.UN_CAR_INFO AS UN_CAR_INFO,
  DWI.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  DWI.APPLICANT AS APPLICANT,
  DWI.ORDER_GROUP_CODE AS ORDER_GROUP_CODE,
  DWI.APPLICANT_NAME AS APPLICANT_NAME,
  DWI.RESIDENT_DESIGNER_NAME AS RESIDENT_DESIGNER_NAME,
  DWI.OWNER_NAME AS OWNER_NAME,
  DWI.OTHER_REMARK AS OTHER_REMARK,
  DWI.SALES_MAN_ONE_CODE AS SALES_MAN_ONE_CODE,
  DWI.SALES_MAN_ONE_NAME AS SALES_MAN_ONE_NAME,
  DWI.APPROVAL_COMMENTS AS APPROVAL_COMMENTS,
  DWI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  DWI.STORE_SALESMAN AS STORE_SALESMAN,
  DWI.STORE_SALESMAN_PHONE AS STORE_SALESMAN_PHONE,
  DWI.STOCKPILE_TYPE AS STOCKPILE_TYPE,
  DWI.MEMBERS_INFO AS MEMBERS_INFO,
  DWI.FORM_TITLE AS FORM_TITLE,
  DWI.SH_UNLOAD_INFO AS SH_UNLOAD_INFO,
  DWI.SH_CITY_AREA AS SH_CITY_AREA,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_HEAD_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_ORDER_1008 SBO ON DWI.DSS_SO_ID = SBO.ID
WHERE
  SBO.ID IS NULL;
