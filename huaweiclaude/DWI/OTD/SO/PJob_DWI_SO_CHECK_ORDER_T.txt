--SDI_CX_CHECK_ORDER_1002 到 DWI_SO_CHECK_ORDER_T
--验收单
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_CHECK_ORDER_T
SELECT
  SDI.ROW_ID AS ROW_ID,
  SDI.SIGN_STATUS AS SIGN_STATUS,
  SDI.SIGN_PERIOD AS SIGN_PERIOD,
  SDI.SIGN_ORDER_NUMBER AS SIGN_ORDER_NUMBER,
  SDI.SETTLEMENT_TYPE AS SETTLEMENT_TYPE,
  SDI.PROJECT_ID AS PROJECT_ID,
  SDI.ORDER_NUMBER AS ORDER_NUMBER,
  SDI.DB_LAST_UPD_SRC AS DB_LAST_UPD_SRC,
  SDI.CHECK_TYPE AS CHECK_TYPE,
  SDI.AMOUNT_SIGN_STATUS AS AMOUNT_SIGN_STATUS,
  SDI.ACT_UPDATED_BY AS ACT_UPDATED_BY,
  SDI.ACT_CREATED_BY AS ACT_CREATED_BY,
  SDI.QUANTITY AS QUANTITY,
  SDI.ORI_TOTAL_AMOUNT AS ORI_TOTAL_AMOUNT,
  CASE
    WHEN SDI.DB_LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.DB_LAST_UPD
  END AS DB_LAST_UPD_DATE,
  SDI.AMOUNT AS AMOUNT,
  SDI.ADJ_AMOUNT AS ADJ_AMOUNT,
  SDI.AUTO_FLAG AS AUTO_FLAG,
  SDI.CONFLICT_ID AS CONFLICT_ID,
  SDI.MODIFICATION_NUM AS MODIFICATION_NUM,
  SDI.LAST_UPD_BY AS LAST_UPD_BY,
  CASE
    WHEN SDI.LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.LAST_UPD
  END AS LAST_UPD_DATE,
  SDI.CREATED_BY AS CREATED_BY,
  CASE
    WHEN SDI.CREATED IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.CREATED
  END AS CREATED_DATE,
  SDI.X_ORIGINAL_FLAG AS ORIGINAL_FLAG,
  CASE
    WHEN SDI.X_CONFIRM_DATE IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.X_CONFIRM_DATE
  END AS CONFIRM_DATE,
  SDI.CHECK_REMARK AS CHECK_REMARK,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_CHECK_ORDER_1002 SDI
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.SIGN_STATUS,
  DWI.SIGN_PERIOD,
  DWI.SIGN_ORDER_NUMBER,
  DWI.SETTLEMENT_TYPE,
  DWI.PROJECT_ID,
  DWI.ORDER_NUMBER,
  DWI.DB_LAST_UPD_SRC,
  DWI.CHECK_TYPE,
  DWI.AMOUNT_SIGN_STATUS,
  DWI.ACT_UPDATED_BY,
  DWI.ACT_CREATED_BY,
  DWI.QUANTITY,
  DWI.ORI_TOTAL_AMOUNT,
  DWI.DB_LAST_UPD_DATE,
  DWI.AMOUNT,
  DWI.ADJ_AMOUNT,
  DWI.AUTO_FLAG,
  DWI.CONFLICT_ID,
  DWI.MODIFICATION_NUM,
  DWI.LAST_UPD_BY,
  DWI.LAST_UPD_DATE,
  DWI.CREATED_BY,
  DWI.CREATED_DATE,
  DWI.ORIGINAL_FLAG,
  DWI.CONFIRM_DATE,
  DWI.CHECK_REMARK,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_CHECK_ORDER_T DWI
  LEFT JOIN SDI.SDI_CX_CHECK_ORDER_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
