set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIOTD.DWI_SO_PROJECT_WBS_T		--DWI.SAP项目WBS结构元素表
	SELECT 	
		SP1.MANDT					AS	GROUP_NUM,
		SP1.PSPNR                   AS	WBS_ELEMENT,
		SP1.KZBWS                   AS	GIVEN_INV_REVALUED,
		SP1.POST1                   AS	DESCRIPTION_POST1,
		SP1.POSTU                   AS	DESCRIPTION_POSTU,
		CASE WHEN SP1.ERDAT = '00000000' 	--判断SDI源表字段记录建立日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.ERDAT, 1, 4) = '9999' --SDI源表字段记录建立日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.ERDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 						AS CREATE_DATE,
		SP1.ERNAM                   AS	CREATE_BY,
		SP1.AENAM                   AS	CHG_OBJECT_USER_NAME,
		CASE
        WHEN SP1.WERKS IS NULL OR SP1.WERKS	= ''	--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)			--关联不上主数据时赋值-999998
        END 					AS  FACTORY_ID, 	--工厂ID
		CASE WHEN SP1.WERKS IS NULL OR SP1.WERKS = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END      				AS	FACTORY_CODE,	--工厂编码
		SP1.PBUKR                   AS	WBS_ELE_COMPANY_CODE,
		SP1.LOEVM                   AS	MASTER_REC_DEL_FLAG,
		SP1.STUFE                   AS	PROJECT_LEVEL_GRADING,
		SP1.PRART                   AS	PROJ_TYPE,
		SP1.CLASF                   AS	PROJ_TOTAL_WBS_ELE_FLAG,
		SP1.XSTAT                   AS	CENSUS_WBS_ELEMENT,
		SP1.STORT                   AS	PLACE,
		CASE WHEN SP1.AEDAT = '00000000' 	--判断SDI源表字段对象最后更改日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.AEDAT, 1, 4) = '9999' --SDI源表字段对象最后更改日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.AEDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 						AS LAST_UPDATE_DATE,
		SP1.POSID                   AS	WBS_SPIT_ELEMENT,
		SP1.FAKKZ                   AS	INVOICE_ELEMENT,
		SP1.PRCTR                   AS	PROFIT_CENTER,
		SP1.PLAKZ                   AS	PLAN_ELEMENT,
		SP1.FUNC_AREA               AS	FUNCTION_SCOPE,
		SP1.PGSBR                   AS	WBS_ELEMENT_BUSINESS_DEPT,
		SP1.OBJNR                   AS	PROJ_OBJECT_NUM,
		SP1.ASTNA                   AS	APPLY_USER,
		SP1.SCOPE                   AS	OBJECT_TYPE,
		SP1.POSID_EDIT              AS	EDIT_WBS_SPIT_ELEMENT,
		SP1.ASTNR                   AS	APPLY_USER_NO,
		SP1.ZUORD                   AS	NETWORK_ASSIGN,
		SP1.POSKI                   AS	WBS_ELEMENT_SIMPLE_FLAG,
		SP1.SPSNR                   AS	STANDARD_PROJ_CURR_DIGIT,
		SP1.PSPHI                   AS	SUIT_PROJ_CURRENT_CODE,
		SP1.KAPPL                   AS	APPLICATION_PROGRAM,
		SP1.KVEWE                   AS	CONDITION_TABLE_PURPOSE,
		SP1.FABKL                   AS	FACTORY_CALENDAR_CODE,
		SP1.PWPOS                   AS	WBS_ELEMENT_CURRENCY,
		SP1.BELKZ                   AS	SUBJECT_ASSIGN_ELEMENT,
		SP1.SLWID                   AS	KEYWORD_ID_PS_DEFINED,
		SP1.PKOKR                   AS	WBS_ELEMENT_CONTROL_SCOPE,
		SP1.USR04                   AS	WBS_ELEMENT_QTY_ONE,
		SP1.USR05                   AS	WBS_ELEMENT_QTY_TWO,
		SP1.USR06                   AS	WBS_ELEMENT_VALUE_ONE,
		SP1.USR07                   AS	WBS_ELEMENT_VALUE_TWO,
		SP1.USR08                   AS	WBS_ELEMENT_DATE_ONE,
		SP1.USR09                   AS	WBS_ELEMENT_DATE_TWO,
		SP1.TADAT                   AS	TECHNICAL_FINISHED_DATE,
		SP1.EVGEW                   AS	POC_PS_PROGRESS,
		'N' 					AS  DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS  DW_LAST_UPDATE_DATE
		FROM SDI.SDI_PRPS_1001 SP1	--SDI.SAP项目WBS结构元素表
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF	--关联工厂主数据取工厂ID
			 ON  SP1.MANDT = DMF.GROUP_NUM		--集团
			 AND SP1.WERKS = DMF.FACTORY_CODE	--工厂编码
			 AND DMF.END_DATE = '4712-12-31'	--结束时间为'4712-12-31'
		UNION ALL				--SDI_PRPS_1001.SAP项目WBS结构元素表 UNION ALL 拼接 DWI_SO_PROJECT_WBS_T.SAP项目WBS结构元素表
		SELECT	
			   T.GROUP_NUM
			  ,T.WBS_ELEMENT
			  ,T.GIVEN_INV_REVALUED
			  ,T.DESCRIPTION_POST1
			  ,T.DESCRIPTION_POSTU
			  ,T.CREATE_DATE
			  ,T.CREATE_BY
			  ,T.CHG_OBJECT_USER_NAME
			  ,T.FACTORY_ID
			  ,T.FACTORY_CODE
			  ,T.WBS_ELE_COMPANY_CODE
			  ,T.MASTER_REC_DEL_FLAG
			  ,T.PROJECT_LEVEL_GRADING
			  ,T.PROJ_TYPE
			  ,T.PROJ_TOTAL_WBS_ELE_FLAG
			  ,T.CENSUS_WBS_ELEMENT
			  ,T.PLACE
			  ,T.LAST_UPDATE_DATE
			  ,T.WBS_SPIT_ELEMENT
			  ,T.INVOICE_ELEMENT
			  ,T.PROFIT_CENTER
			  ,T.PLAN_ELEMENT
			  ,T.FUNCTION_SCOPE
			  ,T.WBS_ELEMENT_BUSINESS_DEPT
			  ,T.PROJ_OBJECT_NUM
			  ,T.APPLY_USER
			  ,T.OBJECT_TYPE
			  ,T.EDIT_WBS_SPIT_ELEMENT
			  ,T.APPLY_USER_NO
			  ,T.NETWORK_ASSIGN
			  ,T.WBS_ELEMENT_SIMPLE_FLAG
			  ,T.STANDARD_PROJ_CURR_DIGIT
			  ,T.SUIT_PROJ_CURRENT_CODE
			  ,T.APPLICATION_PROGRAM
			  ,T.CONDITION_TABLE_PURPOSE
			  ,T.FACTORY_CALENDAR_CODE
			  ,T.WBS_ELEMENT_CURRENCY
			  ,T.SUBJECT_ASSIGN_ELEMENT
			  ,T.KEYWORD_ID_PS_DEFINED
			  ,T.WBS_ELEMENT_CONTROL_SCOPE
			  ,T.WBS_ELEMENT_QTY_ONE
			  ,T.WBS_ELEMENT_QTY_TWO
			  ,T.WBS_ELEMENT_VALUE_ONE
			  ,T.WBS_ELEMENT_VALUE_TWO
			  ,T.WBS_ELEMENT_DATE_ONE
			  ,T.WBS_ELEMENT_DATE_TWO
			  ,T.TECHNICAL_FINISHED_DATE
			  ,T.POC_PS_PROGRESS
			  ,'Y' AS DEL_FLAG
			  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
			  FROM DWIOTD.DWI_SO_PROJECT_WBS_T T	--DWI.SAP项目WBS结构元素表
			  LEFT JOIN SDI.SDI_PRPS_1001 T1		--SDI.SAP项目WBS结构元素表
			  ON T.GROUP_NUM = T1.MANDT
			  AND T.WBS_ELEMENT = T1.PSPNR
			  WHERE T1.MANDT  IS NULL		
			  AND T1.PSPNR  IS NULL
			  AND T.DEL_FLAG = 'N'
