--DSS转储单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_TRANS_STORE_ITEM_DSS_T
SELECT
  SBTSI.ID AS ID,
  SBTSI.BATCH_CODE AS BATCH_CODE,
  CASE
    WHEN SBTSI.PRODUCT_CODE IS NULL OR SBTSI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBTSI.PRODUCT_CODE IS NULL OR SBTSI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  CASE
    WHEN SBTSI.FACTORY_CODE IS NULL OR SBTSI.FACTORY_CODE = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SBTSI.FACTORY_CODE IS NULL OR SBTSI.FACTORY_CODE = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  SBTSI.QUANTITY AS QUANTITY,
  SBTSI.CREATE_TIME AS CREATE_TIME,
  SBTSI.PRODUCT_DESC AS PRODUCT_DESC,
  SBTSI.UPDATE_TIME AS UPDATE_TIME,
  SBTSI.TRANSFER_STORE_ORDER_ID AS TRANS_STORE_ORDER_ID,
  SBTSI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  SBTSI.DELIVERY_DATE AS DELIVERY_DATE,
  SBTSI.STAY_GOODS_ITEM_CODE AS STAY_GOODS_ITEM_CODE,
  SBTSI.WEIGHT AS WEIGHT,
  SBTSI.SAP_CODE AS SAP_CODE,
  SBTSI.SHIP_STORE_LOCATION AS SHIP_STORE_LOCATION,
  SBTSI.UPDATE_USER AS UPDATE_USER,
  SBTSI.CREATE_USER AS CREATE_USER,
  SBTSI.CREATE_DEPT AS CREATE_DEPT,
  SBTSI.DELETED AS DELETED,
  SBTSI.INVENTORY_CODE AS INVENTORY_CODE,
  SBTSI.ITEM_NUMBER AS ITEM_NUMBER,
  SBTSI.NET_WORTH AS NET_WORTH,
  SBTSI.UNIT AS UNIT,
  SBTSI.PRODUCT_GROUP AS PRODUCT_GROUP,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_TRANSFER_STORE_ITEM_1008 SBTSI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBTSI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800' 
  --关联工厂主数据表取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SBTSI.FACTORY_CODE = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.BATCH_CODE AS BATCH_CODE,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.QUANTITY AS QUANTITY,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.PRODUCT_DESC AS PRODUCT_DESC,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.TRANS_STORE_ORDER_ID AS TRANS_STORE_ORDER_ID,
  DWI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  DWI.DELIVERY_DATE AS DELIVERY_DATE,
  DWI.STAY_GOODS_ITEM_CODE AS STAY_GOODS_ITEM_CODE,
  DWI.WEIGHT AS WEIGHT,
  DWI.SAP_CODE AS SAP_CODE,
  DWI.SHIP_STORE_LOCATION AS SHIP_STORE_LOCATION,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.DELETED AS DELETED,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.NET_WORTH AS NET_WORTH,
  DWI.UNIT AS UNIT,
  DWI.PRODUCT_GROUP AS PRODUCT_GROUP,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_TRANS_STORE_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_TRANSFER_STORE_ITEM_1008 SBTSI ON DWI.ID = SBTSI.ID
WHERE
  SBTSI.ID IS NULL;
