--SAP项目定义表
set hive.cbo.enable=true;			--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIOTD.DWI_SO_PROJECT_DEFINED_T		--DWI.SAP项目定义表
SELECT				--查询
		SP1.MANDT				AS		GROUP_NUM,		--集团
		SP1.PSPNR               AS		IN_WBS_ELEMENT,		--项目（内部）
		T.DOMAIN_VALUE_CODE AS		GIVEN_INV_REVALUED,	--特定库存的评估
		SP1.POST1               AS		DESCRIPTION_POST1,	--PS: 短描述 (第一行文本)
		SP1.POSTU               AS		DESCRIPTION_POSTU,	--PS: 短描述(第一文本行)都是大写
		CASE WHEN SP1.ERDAT = '00000000' 	--判断SDI源表字段为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.ERDAT, 1, 4) = '9999' 	--日期截取为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.ERDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS CREATE_DATE,	--记录建立日期
		SP1.ERNAM               AS		CREATE_BY,	--创建对象的人员名称
		FROM_UNIXTIME(UNIX_TIMESTAMP(SUBSTR(CPD_UPDAT,1,16),'yyyyMMddhhmmss'))  AS  TIMESPAN,	--时戳
		SP1.AENAM               AS		CHG_OBJECT_USER_NAME,	--更改对象用户的名称
		CASE
        WHEN SP1.WERKS IS NULL OR SP1.WERKS = ''		--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)			--关联不上主数据赋值-999998
        END 				AS      FACTORY_ID,		--工厂ID
		CASE WHEN SP1.WERKS IS NULL OR SP1.WERKS = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
		END 				AS		FACTORY_CODE,	--工厂编码
		CASE
        WHEN SP1.ZUNNR IS NULL OR SP1.ZUNNR = ''	--关联客户主数据取客户ID,判断SDI源表字段客户编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(CUST.CUST_ID, -999998)		--关联不上主数据赋值-999998
        END     			AS   	CUST_ID,	--客户ID
		CASE WHEN SP1.ZUNNR IS NULL OR SP1.ZUNNR = ''	--关联客户主数据,判断SDI源表字段客户编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')		--关联不上主数据时赋值'NOT EXISTS'
		END  				 AS		CUST_NUM,	--客户编号
		CASE
        WHEN SP1.VBUKR IS NULL OR SP1.VBUKR = ''	--关联公司主数据取公司ID,判断SDI源表字段项目的公司代码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMC.COMPANY_ID, -999998)		--关联不上主数据赋值-999998
        END 				AS  	COMPANY_ID, 	--公司ID
		CASE WHEN SP1.VBUKR IS NULL OR SP1.VBUKR = ''	--关联公司主数据,判断SDI源表字段项目的公司代码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMC.COMPANY_CODE, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
		END 					AS   COMPANY_CODE,	--公司编码
		SP1.BPROF               AS		BUDGET_PARAMS_FILE,	--预算参数文件
		SP1.VPROF               AS		NETWORK_PARAMS_FILE,	--网络参数文件
		SP1.TXTSP               AS		LANG_CODE,	--语言代码
		CASE WHEN SP1.PLSEZ = '00000000' --判断SDI源表字段项目计划完成日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.PLSEZ, 1, 4) = '9999' --SDI源表字段项目计划完成日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.PLSEZ, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS PROJ_PLAN_COMPLETE_DATE,	--项目计划完成日期
		SP1.BESTA               AS		ITEM_CONFIRM_STATUS,	--项目库存
		SP1.XSTAT               AS		CENSUS_WBS_ELEMENT,	--统计 WBS 元素
		CASE WHEN SP1.AEDAT = '00000000' 	--判断SDI源表字段对象最后更改日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.AEDAT, 1, 4) = '9999' --SDI源表字段对象最后更改日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.AEDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 				AS LAST_UPDATE_DATE,	--对象最后更改日期
		T1.CONFIG_CODE      AS		PROFIT_CENTER,	--利润中心
		SP1.FUNC_AREA           AS		FUNCTION_SCOPE,	--功能范围
		SP1.OBJNR               AS		PROJ_OBJECT_NUM,	--对象号
		SP1.PSPID               AS		PROJ_CUSTMON,	--项目定义
		SP1.PSPID_EDIT          AS		PROJECT_NO,	--项目号码 (外部) 编辑
		SP1.VERNR               AS		LEADER_CODE,	--负责人编号
		SP1.ASTNA               AS		APPLY_USER,	--申请人
		T2.DOMAIN_VALUE_CODE               AS		OBJECT_TYPE,--对象类
		SP1.ZVKGRP              AS		SALES_GROUP,	--销售组
		SP1.ZXSZZ               AS		SALES_ORG,	--销售组织
		SP1.VERNA               AS		PROJ_SUPERINTENDENT,	--负责人姓名（项目管理者）
		SP1.PROFL               AS		PROJECT_PARAMS_FILE,	--项目参数文件
		SP1.ASTNR               AS		APPLY_USER_NO,	--申请人号
		SP1.ZZONE               AS		GOODS_SEND_DESTINATION_CODE,	--发送货物的目的地运输区域
		SP1.ZUORD               AS		NETWORK_ASSIGN,	--网络分配
		SP1.KZERB               AS		USE_MAIN_PROJ_TOTAL,	--标识符: 使用主数据特性的项目汇总
		SP1.STSPR               AS		WBS_STATUS_PARAMS_FILE,	--工作分解结构要素的状态参数文件
		SP1.SPSNR               AS		STANDARD_PROJ_CURR_DIGIT,	--标准项目的当前数字
		SP1.ZXSRY               AS		SALES_USER,	--销售人员
		SP1.ZGCL1               AS		ENGINEERING_CATEGORY,	--工程类别
		SP1.ZSJBH               AS		BUSINESS_CHANCE_NUM,	--商机编号
		SP1.ZUNNR_OSS           AS		OSS_CUSTOMER_ID,	--OSS 客户ID
		SP1.ZUNNR_ZL            AS		STRATEGIC_CUSTOMER_ID,	--战略客户ID
		CASE WHEN SP1.PLFAZ = '00000000' 	--判断SDI源表字段项目计划开始日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.PLFAZ, 1, 4) = '9999' 	--SDI源表字段项目计划开始日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.PLFAZ, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS PROJ_PLAN_BEGIN_DATE,	--项目计划开始日期
		SP1.PLNAW               AS		TASK_LIST_APPLICATION,	--任务清单的应用
		SP1.ZTEHT               AS		TIME_PLAN_UNIT,		--在时间计划中的时间单位
		CASE WHEN SP1.AEDTE = '00000000' 	--判断SDI源表字段总体网络的最终计划日期（基本的日期）为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd') --转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.AEDTE, 1, 4) = '9999' 	--SDI源表字段总体网络的最终计划日期（基本的日期）为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.AEDTE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS NETWORK_END_BASE_DATE,	--总体网络的最终计划日期（基本的日期）
		CASE WHEN SP1.AEDTP = '00000000' 	--判断SDI源表字段总体网络的最终计划日期（预测）为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SP1.AEDTP, 1, 4) = '9999' 	--SDI源表字段总体网络的最终计划日期（预测）为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SP1.AEDTP, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS NETWORK_END_FORECAST_DATE,	--总体网络的最终计划日期（预测）
		SP1.DPPPROF             AS		DYNAMIC_PROJ_PARAMS_FILE,	--动态项目处理者参数文件
		SP1.SCPRF               AS		WBS_PLAN_OUTLINE,	--WBS 计划的概要
		SP1.KALID               AS		FACTORY_CALENDAR_CODE,	--工厂日历码
		SP1.VGSBR               AS		PROJECT_BUSINESS_DEPT,	--项目的业务部门
		SP1.VKOKR               AS		PROJECT_CONTROL_SCOPE,	--此项目的控制范围
		SP1.EWPLF               AS		PROJECT_FORECAST_PLAN_FUN,	--项目预测日期的计划方法
		SP1.VGPLF               AS		PROJECT_BASE_PLAN_FUN,	--项目基本日期的计划方法
		SP1.PWHIE               AS		WBS_CURRENCY,	--WBS货币(项目定义)
		SP1.STSPD               AS		PROJ_CUSTMON_PARAMS_FILE,	--项目定义的状态参数文件
		'N' AS DEL_FLAG,		--删除标识
	   CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE		--DW最后更新时间
	   FROM SDI.SDI_PROJ_1001 SP1		--SDI.SAP项目定义表
	   LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF		--关联工厂主数据取工厂ID
			ON  SP1.MANDT = DMF.GROUP_NUM		--集团
			AND SP1.WERKS = DMF.FACTORY_CODE	--工厂编码
			AND DMF.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
	   LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST	--关联客户主数据取客户ID
			ON  SP1.MANDT = CUST.GROUP_NUM		--集团
			AND SP1.ZUNNR = CUST.CUST_NUM		--客户号码
			AND CUST.END_DATE = '4712-12-31'	--结束时间为'4712-12-31'
	   LEFT JOIN DWIMD.DWI_MD_COMPANY_T DMC		--关联公司主数据取公司ID
			ON  SP1.MANDT = DMC.GROUP_NUM		--集团
			AND SP1.VBUKR = DMC.COMPANY_CODE	--公司编码
			AND DMC.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
	   LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R T	--关联DD07T配置表
			ON  T.DOMAIN_NAME  = 'KZBWS'		--特定库存的评估
			AND T.SRC_SYS_DOMAIN_VALUE_CODE = SP1.KZBWS	--特定库存的评估
	   LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1	--关联通用配置表
			ON T1.SRC_SYS_CONFIG_CODE = SP1.PRCTR	--利润中心
			AND T1.GROUP_NUM = SP1.MANDT		--集团
			AND T1.SOURCE_CODE = 'CEPCT'		--伙伴利润中心
	   LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R T2	--关联DD07T配置表
			ON T2.DOMAIN_NAME  = 'SCOPE_CV'		--对象类
			AND T2.SRC_SYS_DOMAIN_VALUE_CODE = SP1.SCOPE	--对象类
		UNION ALL		--SDI_PROJ_1001.SAP项目定义表 UNION ALL 拼接DWI_SO_PROJECT_DEFINED_T.SAP项目定义表
		SELECT 	--查询
	       C.GROUP_NUM
		  ,C.IN_WBS_ELEMENT
		  ,C.GIVEN_INV_REVALUED
		  ,C.DESCRIPTION_POST1
		  ,C.DESCRIPTION_POSTU
		  ,C.CREATE_DATE
		  ,C.CREATE_BY
		  ,C.TIMESPAN
		  ,C.CHG_OBJECT_USER_NAME
		  ,C.FACTORY_ID
		  ,C.FACTORY_CODE
		  ,C.CUST_ID
		  ,C.CUST_NUM
		  ,C.COMPANY_ID
		  ,C.COMPANY_CODE
		  ,C.BUDGET_PARAMS_FILE
		  ,C.NETWORK_PARAMS_FILE
		  ,C.LANG_CODE
		  ,C.PROJ_PLAN_COMPLETE_DATE
		  ,C.ITEM_CONFIRM_STATUS
		  ,C.CENSUS_WBS_ELEMENT
		  ,C.LAST_UPDATE_DATE
		  ,C.PROFIT_CENTER
		  ,C.FUNCTION_SCOPE
		  ,C.PROJ_OBJECT_NUM
		  ,C.PROJ_CUSTMON
		  ,C.PROJECT_NO
		  ,C.LEADER_CODE
		  ,C.APPLY_USER
		  ,C.OBJECT_TYPE
		  ,C.SALES_GROUP
		  ,C.SALES_ORG
		  ,C.PROJ_SUPERINTENDENT
		  ,C.PROJECT_PARAMS_FILE
		  ,C.APPLY_USER_NO
		  ,C.GOODS_SEND_DESTINATION_CODE
		  ,C.NETWORK_ASSIGN
		  ,C.USE_MAIN_PROJ_TOTAL
		  ,C.WBS_STATUS_PARAMS_FILE
		  ,C.STANDARD_PROJ_CURR_DIGIT
		  ,C.SALES_USER
		  ,C.ENGINEERING_CATEGORY
		  ,C.BUSINESS_CHANCE_NUM
		  ,C.OSS_CUSTOMER_ID
		  ,C.STRATEGIC_CUSTOMER_ID
		  ,C.PROJ_PLAN_BEGIN_DATE
		  ,C.TASK_LIST_APPLICATION
		  ,C.TIME_PLAN_UNIT
		  ,C.NETWORK_END_BASE_DATE
		  ,C.NETWORK_END_FORECAST_DATE
		  ,C.DYNAMIC_PROJ_PARAMS_FILE
		  ,C.WBS_PLAN_OUTLINE
		  ,C.FACTORY_CALENDAR_CODE
		  ,C.PROJECT_BUSINESS_DEPT
		  ,C.PROJECT_CONTROL_SCOPE
		  ,C.PROJECT_FORECAST_PLAN_FUN
		  ,C.PROJECT_BASE_PLAN_FUN
		  ,C.WBS_CURRENCY
		  ,C.PROJ_CUSTMON_PARAMS_FILE
		   ,'Y' AS DEL_FLAG	 					
           ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
      FROM DWIOTD.DWI_SO_PROJECT_DEFINED_T C	--DWI.SAP项目定义表
	 LEFT JOIN SDI.SDI_PROJ_1001 D	--SDI.SAP项目定义表
	  ON C.GROUP_NUM = D.MANDT 	--DWI.集团 = SDI.集团						
     AND C.IN_WBS_ELEMENT = D.PSPNR	--DWI.项目（内部） = SDI.项目（内部）
   WHERE D.MANDT  IS NULL	--SDI.集团 为空
     AND D.PSPNR  IS NULL	--SDI.项目（内部） 为空
	 AND C.DEL_FLAG = 'N'
