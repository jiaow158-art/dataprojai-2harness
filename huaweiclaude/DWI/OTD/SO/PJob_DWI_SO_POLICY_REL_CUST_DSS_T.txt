--DSS折扣政策关联客户
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_POLICY_REL_CUST_DSS_T
SELECT
  SBPA.ID AS ID,
  SBPA.ACCOUNT_CODE AS ACCOUNT_CODE,
  SBPA.ACCOUNT_NAME AS ACCOUNT_NAME,
  SBPA.POLICY_UUID AS POLICY_UUID,
  SBPA.UPDATE_TIME AS UPDATE_TIME,
  SBPA.CREATE_TIME AS CREATE_TIME,
  SBPA.CREATE_USER AS CREATE_USER,
  SBPA.UPDATE_USER AS UPDATE_USER,
  SBPA.CREATE_DEPT AS CREATE_DEPT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_POLICY_ACC_1008 SBPA 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.ACCOUNT_CODE AS ACCOUNT_CODE,
  DWI.ACCOUNT_NAME AS ACCOUNT_NAME,
  DWI.POLICY_UUID AS POLICY_UUID,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_POLICY_REL_CUST_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_POLICY_ACC_1008 SBPA ON DWI.ID = SBPA.ID
WHERE
  SBPA.ID IS NULL;
