--SDI_CX_FACTORY_1002 到 DWI_SO_FACTORY_OSS_T
--OSS工厂基础表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_FACTORY_OSS_T
SELECT
  SRC.ROW_ID AS ROW_ID,
  CASE
    WHEN SRC.FACTORY_CD IS NULL
    OR SRC.FACTORY_CD = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.FACTORY_CD IS NULL
    OR SRC.FACTORY_CD = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  SRC.FACTORY_NAME AS SRC_FACTORY_NAME,
  SRC.CREATED AS CREATE_DATE,
  SRC.DB_LAST_UPD AS DB_LAST_UPDATE_DATE,
  SRC.LAST_UPD AS LAST_UPDATE_DATE,
  SRC.MODIFICATION_NUM AS MODIFY_QTY,
  SRC.CREATED_BY AS CREATED_BY,
  SRC.LAST_UPD_BY AS LAST_UPDATE_BY,
  SRC.FACTORY_CITY AS FACTORY_CITY,
  SRC.CONFLICT_ID AS CONFLICT_ID,
  SRC.DB_LAST_UPD_SRC AS DB_LAST_UPDATE_DATE_SRC,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_FACTORY_1002 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.FACTORY_CD = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.SRC_FACTORY_NAME,
  DWI.CREATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.MODIFY_QTY,
  DWI.CREATED_BY,
  DWI.LAST_UPDATE_BY,
  DWI.FACTORY_CITY,
  DWI.CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_FACTORY_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_FACTORY_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
select *  
FROM
  SDI.SDI_CX_FACTORY_1002 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.FACTORY_CD = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
