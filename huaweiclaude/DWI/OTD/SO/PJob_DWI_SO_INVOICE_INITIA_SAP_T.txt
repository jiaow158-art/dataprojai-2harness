--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP出具发票初始表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_INVOICE_INITIA_SAP_T
SELECT
  SV.MANDT AS GROUP_NUM,
  SV.VBELN AS SAL_VCH_NUM,
  R1.CONFIG_CODE AS PROD_GROUP,
  SV.LLAND AS AIM_COUNTRY,
  SV.WAERK AS SAL_CURRENCY_CODE,
  SV.FKTYP AS ISSUE_INVOICE_TYPE,
  M.DOMAIN_VALUE_CODE AS SAL_DIS_VCH_TYPE,
  SV.FKART AS BILL_TYPE,
  SV.VBART AS SAL_VCH_TYPE,
  R2.CONFIG_CODE AS DIS_CHANNEL,
  R3.CONFIG_CODE AS RECEIVE_POINTS,
  R4.CONFIG_CODE AS SAL_ORG_CODE,
  CASE
    WHEN SV.FKDAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.FKDAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.FKDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS BILL_DATE,
  SV.ADRNR AS CUST_ADDR_NUM,
  SV.KUNNR AS CUST_NUM,
  SV.NETWR AS VC_NET_WORTH_AMT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_VKDFS_1001 SV --关联通用配置表，将SDI表相关字段进行转码处理
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1 ON SV.MANDT = R1.GROUP_NUM
  AND SV.SPART = R1.SRC_SYS_CONFIG_CODE
  AND UPPER(R1.SOURCE_CODE) = 'TSPAT' --组织单位: 销售分部
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2 ON SV.MANDT = R2.GROUP_NUM
  AND SV.VTWEG = R2.SRC_SYS_CONFIG_CODE
  AND UPPER(R2.SOURCE_CODE) = 'TVTWT' --机构单位：分销渠道
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3 ON SV.MANDT = R3.GROUP_NUM
  AND SV.VSTEL = R3.SRC_SYS_CONFIG_CODE
  AND UPPER(R3.SOURCE_CODE) = 'TVSTT' --机构单位：装运点
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R4 ON SV.MANDT = R4.GROUP_NUM
  AND SV.VKORG = R4.SRC_SYS_CONFIG_CODE
  AND UPPER(R4.SOURCE_CODE) = 'TVKOT' --机构单位：销售机构
  --关联DD07T配置表，将SDI表相关字段进行转码处理
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R M ON SV.VBTYP = M.SRC_SYS_DOMAIN_VALUE_CODE
  AND UPPER(M.DOMAIN_NAME) = 'VBTYPL' --销售和分销凭证类别
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.PROD_GROUP AS PROD_GROUP,
  DWI.AIM_COUNTRY AS AIM_COUNTRY,
  DWI.SAL_CURRENCY_CODE AS SAL_CURRENCY_CODE,
  DWI.ISSUE_INVOICE_TYPE AS ISSUE_INVOICE_TYPE,
  DWI.SAL_DIS_VCH_TYPE AS SAL_DIS_VCH_TYPE,
  DWI.BILL_TYPE AS BILL_TYPE,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  DWI.DIS_CHANNEL AS DIS_CHANNEL,
  DWI.RECEIVE_POINTS AS RECEIVE_POINTS,
  DWI.SAL_ORG_CODE AS SAL_ORG_CODE,
  DWI.BILL_DATE AS BILL_DATE,
  DWI.CUST_ADDR_NUM AS CUST_ADDR_NUM,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.VC_NET_WORTH_AMT AS VC_NET_WORTH_AMT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_INVOICE_INITIA_SAP_T DWI
  LEFT JOIN SDI.SDI_VKDFS_1001 SV ON DWI.GROUP_NUM = SV.MANDT
  AND DWI.SAL_VCH_NUM = SV.VBELN
WHERE
  SV.VBELN IS NULL;
