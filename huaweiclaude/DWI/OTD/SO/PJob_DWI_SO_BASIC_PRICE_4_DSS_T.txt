--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--DSS基准价4
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_BASIC_PRICE_4_DSS_T
SELECT
  SBBP.ID AS ID,
  SBBP.NOT_LEVEL_CONDITION_PERCENTAGE AS NOT_LEVEL_CONDI_PER,
  CASE
    WHEN SBBP.MATERIAL_CODE IS NULL OR SBBP.MATERIAL_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBBP.MATERIAL_CODE IS NULL OR SBBP.MATERIAL_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBBP.CONDITION_START_TIME AS CONDITION_START_TIME,
  SBBP.CONDITION_END_TIME AS CONDITION_END_TIME,
  SBBP.CUSTOMER_PRICE_GROUP AS CUST_PRICE_GROUP,
  SBBP.SALES_ORGANIZATION AS SALES_ORGANIZATION,
  SBBP.CONDITION_UNIT1 AS CONDITION_UNIT1,
  SBBP.UPDATE_TIME AS UPDATE_TIME,
  SBBP.DISTRIBUTION_CHANNELS AS DISTRIBUTION_CHANNELS,
  SBBP.CONDITIONAL_PRICING_UNIT AS CONDITIONAL_PRICING_UNIT,
  SBBP.CONDITION_TYPE AS CONDITION_TYPE,
  SBBP.CONDITION_UNIT AS CONDITION_UNIT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_BASIC_PRICE_4_1008 SBBP 
  --关联物料主数据表取物料ID
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBBP.MATERIAL_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.NOT_LEVEL_CONDI_PER AS NOT_LEVEL_CONDI_PER,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.CONDITION_START_TIME AS CONDITION_START_TIME,
  DWI.CONDITION_END_TIME AS CONDITION_END_TIME,
  DWI.CUST_PRICE_GROUP AS CUST_PRICE_GROUP,
  DWI.SALES_ORGANIZATION AS SALES_ORGANIZATION,
  DWI.CONDITION_UNIT1 AS CONDITION_UNIT1,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.DISTRIBUTION_CHANNELS AS DISTRIBUTION_CHANNELS,
  DWI.CONDITIONAL_PRICING_UNIT AS CONDITIONAL_PRICING_UNIT,
  DWI.CONDITION_TYPE AS CONDITION_TYPE,
  DWI.CONDITION_UNIT AS CONDITION_UNIT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_BASIC_PRICE_4_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_BASIC_PRICE_4_1008 SBBP ON DWI.ID = SBBP.ID
WHERE
  SBBP.ID IS NULL;
