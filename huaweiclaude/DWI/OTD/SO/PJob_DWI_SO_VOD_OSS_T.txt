--SDI_S_VOD_1002 到 DWI_SO_VOD_OSS_T
--OSS系统标准表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_VOD_OSS_T
SELECT
  SRC.OBJECT_NUM,
  SRC.ROW_ID,
  SRC.VOD_NAME,
  SRC.CREATED,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.MODIFICATION_NUM,
  SRC.DESC_TEXT,
  SRC.LOCKED_DT,
  SRC.LAST_UPD_BY,
  SRC.VOD_TYPE_CD,
  SRC.CREATED_BY,
  SRC.LOCKED_BY,
  SRC.DB_LAST_UPD_SRC,
  SRC.LOCKED_FLG,
  SRC.CONFLICT_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_S_VOD_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.OBJECT_NUM,
  DWI.ROW_ID,
  DWI.VOD_NAME,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.MODIFY_QTY,
  DWI.DESC_TEXT,
  DWI.LOCKED_DATE,
  DWI.LAST_UPDATE_BY,
  DWI.VOD_TYPE_CD,
  DWI.CREATED_BY,
  DWI.LOCKED_BY,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.LOCKED_FLAG,
  DWI.CONFLICT_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_VOD_OSS_T DWI
  LEFT JOIN SDI.SDI_S_VOD_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
