--DSS退货单头
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RETURN_ORDER_HEAD_DSS_T
SELECT
  SBRO.ID AS ID,
  SBRO.CREATE_TIME AS CREATE_TIME,
  SBRO.UPDATE_TIME AS UPDATE_TIME,
  SBRO.RETURN_ORDER_CODE AS RETURN_ORDER_CODE,
  SBRO.SAP_RETURN_CODE AS SAP_RETURN_CODE,
  SBRO.SAP_INVOICE_NUM AS SAP_INVOICE_NUM,
  SBRO.APPLICANT_DATE AS APPLICANT_DATE,
  SBRO.ORDER_ID AS ORDER_ID,
  SBRO.SAP_CODE AS SAP_CODE,
  SBRO.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  SBRO.DELIVERY_ADDRESS AS DELIVERY_ADDRESS,
  SBRO.CONSIGNEE_PHONE AS CONSIGNEE_PHONE,
  SBRO.UN_CAR_INFO AS UN_CAR_INFO,
  SBRO.DELIVERY_DATE AS DELIVERY_DATE,
  SBRO.SALES_MAN_NAME AS SALES_MAN_NAME,
  SBRO.SALES_MAN_CODE AS SALES_MAN_CODE,
  SBRO.APPLICANT_NAME AS APPLICANT_NAME,
  SBRO.APPLICANT AS APPLICANT,
  SBRO.CREATE_USER AS CREATE_USER,
  SBRO.ACCOUNT_NAME AS ACCOUNT_NAME,
  SBRO.UPDATE_USER AS UPDATE_USER,
  SBRO.ACCOUNT_CODE AS ACCOUNT_CODE,
  SBRO.PAYER_CODE AS PAYER_CODE,
  SBRO.DESIGNER AS DESIGNER,
  SBRO.OTHER_REMARK AS OTHER_REMARK,
  SBRO.CREATE_DEPT AS CREATE_DEPT,
  SBRO.CONTRACT_CODE AS CONTRACT_CODE,
  SBRO.CHANNEL_CODE AS CHANNEL_CODE,
  SBRO.SALES_ORG_CODE AS SALES_ORG_CODE,
  SBRO.RESIDENT_DESIGNER_NAME AS RESIDENT_DESIGNER_NAME,
  SBRO.RESIDENT_DESIGNER AS RESIDENT_DESIGNER,
  SBRO.ORDER_TYPE AS ORDER_TYPE,
  SBRO.PURCHASE_CODE AS PURCHASE_CODE,
  SBRO.EXTRACT_METHOD AS EXTRACT_METHOD,
  SBRO.RETURN_CAUSE AS RETURN_CAUSE,
  SBRO.EDITABLE AS EDITABLE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_RETURN_ORDER_1008 SBRO 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.RETURN_ORDER_CODE AS RETURN_ORDER_CODE,
  DWI.SAP_RETURN_CODE AS SAP_RETURN_CODE,
  DWI.SAP_INVOICE_NUM AS SAP_INVOICE_NUM,
  DWI.APPLICANT_DATE AS APPLICANT_DATE,
  DWI.ORDER_ID AS ORDER_ID,
  DWI.SAP_CODE AS SAP_CODE,
  DWI.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  DWI.DELIVERY_ADDRESS AS DELIVERY_ADDRESS,
  DWI.CONSIGNEE_PHONE AS CONSIGNEE_PHONE,
  DWI.UN_CAR_INFO AS UN_CAR_INFO,
  DWI.DELIVERY_DATE AS DELIVERY_DATE,
  DWI.SALES_MAN_NAME AS SALES_MAN_NAME,
  DWI.SALES_MAN_CODE AS SALES_MAN_CODE,
  DWI.APPLICANT_NAME AS APPLICANT_NAME,
  DWI.APPLICANT AS APPLICANT,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.ACCOUNT_NAME AS ACCOUNT_NAME,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.ACCOUNT_CODE AS ACCOUNT_CODE,
  DWI.PAYER_CODE AS PAYER_CODE,
  DWI.DESIGNER AS DESIGNER,
  DWI.OTHER_REMARK AS OTHER_REMARK,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.CONTRACT_CODE AS CONTRACT_CODE,
  DWI.CHANNEL_CODE AS CHANNEL_CODE,
  DWI.SALES_ORG_CODE AS SALES_ORG_CODE,
  DWI.RESIDENT_DESIGNER_NAME AS RESIDENT_DESIGNER_NAME,
  DWI.RESIDENT_DESIGNER AS RESIDENT_DESIGNER,
  DWI.ORDER_TYPE AS ORDER_TYPE,
  DWI.PURCHASE_CODE AS PURCHASE_CODE,
  DWI.EXTRACT_METHOD AS EXTRACT_METHOD,
  DWI.RETURN_CAUSE AS RETURN_CAUSE,
  DWI.EDITABLE AS EDITABLE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RETURN_ORDER_HEAD_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_RETURN_ORDER_1008 SBRO ON DWI.ID = SBRO.ID
WHERE
  SBRO.ID IS NULL;
