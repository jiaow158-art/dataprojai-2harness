  SRC.ROW_ID,
  SRC.CREATED,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.PROD_ID,
  SRC.SALES_ORG_ID,
  SRC.MAT_GRP1,
  SRC.SALES_UNIT,

  SRC.MAT_GRP2,
  SRC.MAT_GRP3,

  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.PROD_ID,
  DWI.SALES_ORG_ID,
  DWI.MAT_GRP1,
  DWI.SALES_UNIT,
  DWI.DELIVERY_UNIT,
  DWI.MAT_GRP2,
  DWI.MAT_GRP3,
  dwi.MODIFY_QTY,
  DWI.SALES_CHAN_ID,
  DWI.MAT_GRP4,
  DWI.ITEM_CAT_GRP,
  DWI.MAT_GRP5,
  DWI.DD_CHAIN_STAT,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.CD_CHAIN_STAT,
  DWI.CREATED_BY,
  DWI.LAST_UPDATE_BY,
  DWI.PROD_ATT1,
  DWI.PROD_ATT2,
  DWI.PROD_ATT4,
  DWI.PROD_ATT6,
  DWI.PROD_ATT3,
  DWI.PROD_ATT8,
  DWI.PROD_ATT10,
  DWI.PROD_ATT9,
  DWI.PROD_ATT5,
  DWI.PROD_ATT7,
  DWI.CONFLICT_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PROD_SALES_OSS_T DWI

WHERE
  SDI.ROW_ID IS NULL;
