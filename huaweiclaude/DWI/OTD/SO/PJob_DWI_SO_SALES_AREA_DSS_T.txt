--DSS销售区域
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SALES_AREA_DSS_T
SELECT
  SBSA.ID AS ID,
  SBSA.CUSTOMER_ID AS CUST_ID,
  SBSA.NAME1 AS NAME1,
  SBSA.DESCRIPTION AS DESCRIPTION,
  SBSA.SALESMAN_CODE AS SALESMAN_CODE,
  SBSA.SALES_GROUP_CODE AS SALES_GROUP_CODE,
  SBSA.SALES_GROUP AS SALES_GROUP,
  SBSA.SALES_DEPT_CODE AS SALES_DEPT_CODE,
  SBSA.SALES_DEPT AS SALES_DEPT,
  SBSA.PRICE_GROUP_CODE AS PRICE_GROUP_CODE,
  SBSA.PRICE_GROUP AS PRICE_GROUP,
  SBSA.SALES_ORGANIZATION_CODE AS SALES_ORG_CODE,
  SBSA.SALES_ORGANIZATION AS SALES_ORGANIZATION,
  CASE
    WHEN SBSA.COMPANY_CODE IS NULL OR SBSA.COMPANY_CODE = '' THEN -999999
    ELSE NVL(COMP.COMPANY_ID, -999998)
  END AS COMPANY_ID,--公司ID
  CASE
    WHEN SBSA.COMPANY_CODE IS NULL OR SBSA.COMPANY_CODE = '' THEN 'SNULL'
    ELSE NVL(COMP.COMPANY_CODE, 'NOT EXISTS')
  END AS COMPANY_CODE,--公司代码
  SBSA.CUSTOMER_GROUP_CODE AS CUST_GROUP_CODE,
  SBSA.CUSTOMER_GROUP AS CUST_GROUP,
  SBSA.SALES_REGION AS SALES_REGION,
  SBSA.CUSTOMER_GROUP1_CODE AS CUST_GROUP1_CODE,
  SBSA.CUSTOMER_GROUP3_CODE AS CUST_GROUP3_CODE,
  SBSA.CUSTOMER_GROUP1 AS CUST_GROUP1,
  SBSA.CUSTOMER_GROUP3 AS CUST_GROUP3,
  SBSA.UPDATE_TIME AS UPDATE_TIME,
  SBSA.DISTRIBUTION_CHANNEL AS DISTRIBUTION_CHANNEL,
  SBSA.DISTRIBUTION_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  SBSA.PARTNER_FUNCTION AS PARTNER_FUNCTION,
  SBSA.CUSTOMER_GROUP2_CODE AS CUST_GROUP2_CODE,
  SBSA.PARTNER_FUNCTION_NAME AS PARTNER_FUNCTION_NAME,
  SBSA.CUSTOMER_GROUP2 AS CUST_GROUP2,
  SBSA.CUSTOMER_GROUP4_CODE AS CUST_GROUP4_CODE,
  SBSA.CUSTOMER_GROUP4 AS CUST_GROUP4,
  SBSA.CURRENCY_CODE AS CURRENCY_CODE,
  SBSA.CURRENCY AS CURRENCY,
  SBSA.ORDER_FREEZE AS ORDER_FREEZE,
  SBSA.SHIPPING_CONDITION_CODE AS SHIPPING_CONDITION_CODE,
  SBSA.CUSTOMER_GROUP5_CODE AS CUST_GROUP5_CODE,
  SBSA.PRODUCT_GROUP_CODE AS PRODUCT_GROUP_CODE,
  SBSA.PRODUCT_GROUP AS PRODUCT_GROUP,
  SBSA.SHIPPING_CONDITION AS SHIPPING_CONDITION,
  SBSA.CUSTOMER_GROUP5 AS CUST_GROUP5,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_SALES_AREA_1008 SBSA 
  --关联公司主数据表取公司ID、编码
  LEFT JOIN DWIMD.DWI_MD_COMPANY_T COMP --公司主数据表
  ON SBSA.COMPANY_CODE = COMP.COMPANY_CODE
  AND COMP.END_DATE = '4712-12-31'
  AND COMP.DEL_FLAG = 'N'
  AND COMP.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.CUST_ID AS CUST_ID,
  DWI.NAME1 AS NAME1,
  DWI.DESCRIPTION AS DESCRIPTION,
  DWI.SALESMAN_CODE AS SALESMAN_CODE,
  DWI.SALES_GROUP_CODE AS SALES_GROUP_CODE,
  DWI.SALES_GROUP AS SALES_GROUP,
  DWI.SALES_DEPT_CODE AS SALES_DEPT_CODE,
  DWI.SALES_DEPT AS SALES_DEPT,
  DWI.PRICE_GROUP_CODE AS PRICE_GROUP_CODE,
  DWI.PRICE_GROUP AS PRICE_GROUP,
  DWI.SALES_ORG_CODE AS SALES_ORG_CODE,
  DWI.SALES_ORGANIZATION AS SALES_ORGANIZATION,
  DWI.COMPANY_ID AS COMPANY_ID,
  DWI.COMPANY_CODE AS COMPANY_CODE,
  DWI.CUST_GROUP_CODE AS CUST_GROUP_CODE,
  DWI.CUST_GROUP AS CUST_GROUP,
  DWI.SALES_REGION AS SALES_REGION,
  DWI.CUST_GROUP1_CODE AS CUST_GROUP1_CODE,
  DWI.CUST_GROUP3_CODE AS CUST_GROUP3_CODE,
  DWI.CUST_GROUP1 AS CUST_GROUP1,
  DWI.CUST_GROUP3 AS CUST_GROUP3,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.DISTRIBUTION_CHANNEL AS DISTRIBUTION_CHANNEL,
  DWI.DISTR_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  DWI.PARTNER_FUNCTION AS PARTNER_FUNCTION,
  DWI.CUST_GROUP2_CODE AS CUST_GROUP2_CODE,
  DWI.PARTNER_FUNCTION_NAME AS PARTNER_FUNCTION_NAME,
  DWI.CUST_GROUP2 AS CUST_GROUP2,
  DWI.CUST_GROUP4_CODE AS CUST_GROUP4_CODE,
  DWI.CUST_GROUP4 AS CUST_GROUP4,
  DWI.CURRENCY_CODE AS CURRENCY_CODE,
  DWI.CURRENCY AS CURRENCY,
  DWI.ORDER_FREEZE AS ORDER_FREEZE,
  DWI.SHIPPING_CONDITION_CODE AS SHIPPING_CONDITION_CODE,
  DWI.CUST_GROUP5_CODE AS CUST_GROUP5_CODE,
  DWI.PRODUCT_GROUP_CODE AS PRODUCT_GROUP_CODE,
  DWI.PRODUCT_GROUP AS PRODUCT_GROUP,
  DWI.SHIPPING_CONDITION AS SHIPPING_CONDITION,
  DWI.CUST_GROUP5 AS CUST_GROUP5,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SALES_AREA_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_SALES_AREA_1008 SBSA ON DWI.ID = SBSA.ID
WHERE
  SBSA.ID IS NULL;
