 SELECT 
    D1.ID,									 				    
    D1.DO_NO AS DO_NUM,	 			    
    D1.DO_LINENO AS DO_DETAIL_NUM,	 		    
    D1.ORDER_NO AS ORDER_NUM,	 			    
    D1.ORDER_LINENO AS ORDER_DETAIL_NUM,	 	    
    D1.SKU_CODE AS MATERIAL_NUM,

    D1.QUANTITY AS QUANTITY,

    D1.LOTNO AS BATCH_NUM,

    D1.SALEORDER_NO AS SAP_SALE_ORDER_NUM,

    D1.DRIVER_NAME  AS DRIVER_NAME,			
    D1.LICENSE_PLATE_NO AS LICENSE_PLATE_NUM,

    D1.DRIVER_PHONE AS DRIVER_PHONE,

    D1.STORAGE_LOCATION AS INVENTORY_LOC,	 			
    D1.WEIGHTUNIT AS WEIGHT_UNIT_CODE,	 		
    D1.CONTAINER_NUMBER AS CONTAINER_NUM,	 			
    D1.OBLIGATE_NO  AS SAP_RESERVED_NUM,	 		
    D1.DUMP_NO  AS SAP_DUMP_NUM,	 			
    D1.RECEIPT_STATUS AS RECEIPT_STATUS_FLAG,	 	
    D1.AREA  AS ACREAGE,	 					
    D1.BOXQTY AS BOXES2,	 				
    D1.LICENSENO AS DRIVER_LICENSE_ID,	 		
    D1.PROJECT_PLAN AS SAP_ITEM_PLAN_NUM,	 		
    D1.PRINTED_NUM  AS PRINTED_NUM,	 			
    CASE
         WHEN D1.SKU_CODE IS NULL THEN
          -999999
         ELSE
          NVL(MATL.MATERIAL_ID, -999998)
       END AS MATERIAL_ID,
	   CASE
         WHEN D1.FACTORY_CODE IS NULL THEN
          -999999
         ELSE
          NVL(FACTORY.FACTORY_ID, -999998)
       END AS FACTORY_ID,
	  FACTORY.FACTORY_NAME,
       'N' AS DEL_FLAG,   
   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
 FROM SDI.SDI_TMS_DO_DETAIL_1005 D1
 LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MATL
    ON D1.SKU_CODE = MATL.MATERIAL_NUM
   AND MATL.END_DATE = '4712-12-31'
   AND MATL.DEL_FLAG = 'N'
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FACTORY
   ON D1.FACTORY_CODE = FACTORY.FACTORY_CODE
   AND FACTORY.END_DATE = '4712-12-31'
 UNION 
--更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
