--SDI_ORDER_TYPE_SETTING_1007 到 DWI_PS_ORDER_TYPE_SETTING_T
--订单类型设置
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_ORDER_TYPE_SETTING_T
SELECT
  SRC.ID AS ID,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.NAME AS NAME,
  SRC.NO AS NO,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.WRITE_OFF AS WRITE_OFF,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_ORDER_TYPE_SETTING_1007 SRC
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.INV_ORG_ID,
  DWI.IS_PHANTOM,
  DWI.NAME,
  DWI.NO,
  DWI.SYNC_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.WRITE_OFF,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_ORDER_TYPE_SETTING_T DWI
  LEFT JOIN SDI.SDI_ORDER_TYPE_SETTING_1007 SRC ON DWI.ID = SRC.ID
WHERE
  SRC.ID IS NULL;
