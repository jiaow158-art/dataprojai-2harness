--SDI_RES_WIP_SCHE_1007 到 DWI_PS_RES_WIP_SCHE_T
--生产资源
set hive.cbo.enable=true;
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_RES_WIP_SCHE_T
SELECT
  T.ID AS ID,
  T.ALGORITHM_MARKING AS ALGORITHM_MARKING,
  T.AUTOMATION_TYPE AS AUTOMATION_TYPE,
  T.CODE AS CODE,
  T.CREATE_BY AS CREATE_BY,
  T.CREATE_DATE AS CREATE_DATE,
  T.FACTORY_ID AS SRC_BELONG_FACT_ID,
  --源系统所属工厂ID
  CASE
    WHEN T.WORK_SHOP_ID IS NULL
    OR T.WORK_SHOP_ID = '' THEN -999999
    ELSE NVL(TMP.SRC_FACTORY_ID, -999998)
  END AS SRC_FACTORY_ID,
  --源系统工厂ID
  CASE
    WHEN T.WORK_SHOP_ID IS NULL
    OR T.WORK_SHOP_ID = '' THEN -999999
    ELSE NVL(TMP.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN T.WORK_SHOP_ID IS NULL
    OR T.WORK_SHOP_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  T.INV_ORG_ID AS INV_ORG_ID,
  T.IS_OUT_MADE AS IS_OUT_MADE,
  T.IS_PHANTOM AS IS_PHANTOM,
  T.IS_SMALL_SINGLE AS IS_SMALL_SINGLE,
  T.NAME AS NAME,
  T.PROCESS_TECH_TYPE_ID AS PROCESS_TECH_TYPE_ID,
  T.QTY AS QTY,
  T.RESOURCE_STATE AS RESOURCE_STATE,
  T.SCHEME_ID AS SCHEME_ID,
  T.SEQUENCE AS SEQUENCE,
  T.SOURCE_ID AS SOURCE_ID,
  T.SOURCE_TYPE AS SOURCE_TYPE,
  T.SYNC_ID AS SYNC_ID,
  T.TAKT_TIME AS TAKT_TIME,
  T.UPDATE_BY AS UPDATE_BY,
  T.UPDATE_DATE AS UPDATE_DATE,
  T.WORK_SHOP_ID AS WORK_SHOP_ID,
  T.IS_PROCESSING_BRICK AS IS_PROCESSING_BRICK,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_RES_WIP_SCHE_1007 T
  LEFT JOIN (
    SELECT
      M1.ID AS SRC_FACTORY_ID,
      --源系统工厂ID
      FAC.FACTORY_ID,
      --工厂ID
      FAC.FACTORY_CODE --工厂编码
    FROM
      SDI.SDI_RES_ENTERPRISE_1007 M1 --工厂编码表
      LEFT JOIN SDI.SDI_RES_ENTERPRISE_1007 M2 --工厂编码表
      ON M1.TREE_PID = M2.ID
      AND M1.IS_PHANTOM = 0 --取有效属性数据
      INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
      ON M2.CODE = FAC.FACTORY_CODE
      AND FAC.END_DATE = '4712-12-31'
      AND FAC.DEL_FLAG = 'N'
      AND FAC.GROUP_NUM = '800'
  ) TMP ON T.WORK_SHOP_ID = TMP.SRC_FACTORY_ID
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.ALGORITHM_MARKING,
  DWI.AUTOMATION_TYPE,
  DWI.CODE,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.SRC_BELONG_FACT_ID,
  DWI.SRC_FACTORY_ID,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.INV_ORG_ID,
  DWI.IS_OUT_MADE,
  DWI.IS_PHANTOM,
  DWI.IS_SMALL_SINGLE,
  DWI.NAME,
  DWI.PROCESS_TECH_TYPE_ID,
  DWI.QTY,
  DWI.RESOURCE_STATE,
  DWI.SCHEME_ID,
  DWI.SEQUENCE,
  DWI.SOURCE_ID,
  DWI.SOURCE_TYPE,
  DWI.SYNC_ID,
  DWI.TAKT_TIME,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.WORK_SHOP_ID,
  DWI.IS_PROCESSING_BRICK,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDAE_DTE
FROM
  DWIOTD.DWI_PS_RES_WIP_SCHE_T DWI
  LEFT JOIN SDI.SDI_RES_WIP_SCHE_1007 SRC ON DWI.ID = SRC.ID
WHERE
  SRC.ID IS NULL;
