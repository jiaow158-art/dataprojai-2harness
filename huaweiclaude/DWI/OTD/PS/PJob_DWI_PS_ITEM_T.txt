--SDI_ITEM_1007 到 DWI_PS_ITEM_T
--物料
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_ITEM_T
SELECT
  SRC.ID AS ID,
  SRC.CODE AS MATERIAL_NUM,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.DESCRIPTION AS DESCRIPTION,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.NAME AS NAME,
  SRC.STATE AS STATE,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.TEMPLATE_ID AS TEMPLATE_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.AREA AS ACREAGE,
  SRC.AREA_UNIT_ID AS AREA_UNIT_ID,
  SRC.CATEGORY AS CATEGORY,
  SRC.CONSUME_MODE AS CONSUME_MODE,
  SRC.EMBRYO AS EMBRYO,
  SRC.ENABLE_EXTEND_PROPERTY AS ENABLE_EXTEND_PROPERTY,
  SRC.GLAZE AS GLAZE,
  SRC.GRAYSCALE AS GRAYSCALE,
  SRC.IS_VIRTUAL_PART AS IS_VIRTUAL_PART,
  SRC.ITEM_LABEL_TYPE AS ITEM_LABEL_TYPE,
  SRC.ITEM_SOURCE_TYPE AS ITEM_SOURCE_TYPE,
  SRC.LENGTH AS LENGTH,
  SRC.LOWER_WEIGHT AS LOWER_WEIGHT,
  SRC.MOULD AS MOULD,
  SRC.PROCESS_TECH_ROUTE_ID AS PROCESS_TECH_ROUTE_ID,
  SRC.PRODUCT_LEVEL_TWO AS PRODUCT_LEVEL_TWO,
  SRC.PRODUCT_PROPERTY AS MATERIAL_GROUP_CODE,
  SRC.SERIE_CODE AS SERIE_CODE,
  SRC.SERIE_NAME AS SERIE_NAME,
  SRC.SOURCE_KEY AS SOURCE_KEY,
  SRC.SOURCE_TYPE AS SOURCE_TYPE,
  SRC.SPECIFICATION_MODEL AS SPECIFICATION_MODEL,
  SRC.TECHNOLOGY AS TECHNOLOGY,
  SRC.TYPE_ AS TYPE_,
  SRC.UNIT_ID AS UNIT_ID,
  SRC.UPPER_WEIGHT AS UPPER_WEIGHT,
  SRC.WIDTH AS WIDTH,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_ITEM_1007 SRC
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.MATERIAL_NUM,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.DESCRIPTION,
  DWI.INV_ORG_ID,
  DWI.IS_PHANTOM,
  DWI.NAME,
  DWI.STATE,
  DWI.SYNC_ID,
  DWI.TEMPLATE_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.ACREAGE,
  DWI.AREA_UNIT_ID,
  DWI.CATEGORY,
  DWI.CONSUME_MODE,
  DWI.EMBRYO,
  DWI.ENABLE_EXTEND_PROPERTY,
  DWI.GLAZE,
  DWI.GRAYSCALE,
  DWI.IS_VIRTUAL_PART,
  DWI.ITEM_LABEL_TYPE,
  DWI.ITEM_SOURCE_TYPE,
  DWI.LENGTH,
  DWI.LOWER_WEIGHT,
  DWI.MOULD,
  DWI.PROCESS_TECH_ROUTE_ID,
  DWI.PRODUCT_LEVEL_TWO,
  DWI.MATERIAL_GROUP_CODE,
  DWI.SERIE_CODE,
  DWI.SERIE_NAME,
  DWI.SOURCE_KEY,
  DWI.SOURCE_TYPE,
  DWI.SPECIFICATION_MODEL,
  DWI.TECHNOLOGY,
  DWI.TYPE_,
  DWI.UNIT_ID,
  DWI.UPPER_WEIGHT,
  DWI.WIDTH,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_ITEM_T DWI
  LEFT JOIN SDI.SDI_ITEM_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
