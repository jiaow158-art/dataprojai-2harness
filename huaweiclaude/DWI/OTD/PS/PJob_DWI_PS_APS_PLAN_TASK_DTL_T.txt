--SDI_APS_PLAN_TASK_DTL_1007 到 DWI_PS_APS_PLAN_TASK_DTL_T
--任务计划明细
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_APS_PLAN_TASK_DTL_T
SELECT
  SRC.ID AS ID,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.EFF AS EFF,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_MAIN_ITEM AS IS_MAIN_ITEM,
  SRC.ORDER_QTY AS ORDER_QTY,
  SRC.PLAN_AMOUNT AS PLAN_AMOUNT,
  SRC.PLAN_TASK_ID AS PLAN_TASK_ID,
  SRC.PROCESS_TECH_TASK_ID AS PROCESS_TECH_TASK_ID,
  SRC.RELEASE_END_TIME AS RELEASE_END_TIME,
  SRC.RELEASE_START_TIME AS RELEASE_START_TIME,
  SRC.SAM AS SAM,
  SRC.UNIT_QTY AS UNIT_QTY,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.WORK_ORDER AS WORK_ORDER,
  SRC.WORK_ORDER_TYPE AS WORK_ORDER_TYPE,
  SRC.OEM_ORDER_DATE AS OEM_ORDER_DATE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_APS_PLAN_TASK_DTL_1007 SRC
UNION ALL
    --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.EFF,
  DWI.INV_ORG_ID,
  DWI.IS_MAIN_ITEM,
  DWI.ORDER_QTY,
  DWI.PLAN_AMOUNT,
  DWI.PLAN_TASK_ID,
  DWI.PROCESS_TECH_TASK_ID,
  DWI.RELEASE_END_TIME,
  DWI.RELEASE_START_TIME,
  DWI.SAM,
  DWI.UNIT_QTY,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.WORK_ORDER,
  DWI.WORK_ORDER_TYPE,
  DWI.OEM_ORDER_DATE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_APS_PLAN_TASK_DTL_T DWI
  LEFT JOIN SDI.SDI_APS_PLAN_TASK_DTL_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
