--SDI_APS_PROTECH_ORDER_1007 到 DWI_PS_APS_PROTECH_ORDER_T
--制程工艺单
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_APS_PROTECH_ORDER_T
SELECT
  SRC.ID AS ID,
  SRC.BATCH_RATIO AS BATCH_RATIO,
  SRC.CODE AS PROTECH_NUM,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_END_NODE AS IS_END_NODE,
  SRC.IS_LOCKED AS IS_LOCKED,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.IS_PROGRESS_COMPLETE AS IS_PROGRESS_COMPLETE,
  SRC.IS_START_NODE AS IS_START_NODE,
  SRC.NODE_NUMBER AS NODE_NUMBER,
  SRC.ORDER_STATE AS ORDER_STATE,
  SRC.PRO_TECH_NAME AS PRO_TECH_NAME,
  SRC.PROCESS_TECH_ID AS PROCESS_TECH_ID,
  SRC.PROCESS_TECH_ITEM_ID AS SRC_MATERIAL_ID,
  --源系统物料ID
  --取主数据取ID,源表关联字段为NULL,赋值-999999,源表关联字段有数据关联不上主数据表,赋值-999998
  CASE
    WHEN SRC.PROCESS_TECH_ITEM_ID IS NULL
    OR SRC.PROCESS_TECH_ITEM_ID = '' THEN -999999
    ELSE NVL(TMP.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.PROCESS_TECH_ITEM_ID IS NULL
    OR SRC.PROCESS_TECH_ITEM_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.PROCESS_TECH_ROUTE_ID AS PROCESS_TECH_ROUTE_ID,
  SRC.PRODUCT_ORDER_ID AS PRODUCT_ORDER_ID,
  SRC.QTY AS QTY,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.TRANSFER_BATCH AS TRANSFER_BATCH,
  SRC.TRANSPORT_TIME AS TRANSPORT_TIME,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_APS_PROTECH_ORDER_1007 SRC --源系统ID LEFT JOIN (编码表和主数据表用编码关联取出ID,编码)
  LEFT JOIN (
    SELECT
      M1.ID,
      M2.MATERIAL_ID,
      --物料ID
      M2.MATERIAL_NUM --物料编码
    FROM
      SDI.SDI_ITEM_1007 M1 --物料编码表
      INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T M2 --物料主数据表
      ON M1.CODE = M2.MATERIAL_NUM
      AND M2.END_DATE = '4712-12-31'
      AND M2.DEL_FLAG = 'N'
      AND M1.IS_PHANTOM = 0 --取有效属性数据
      AND M2.GROUP_NUM = '800'
  ) TMP ON SRC.PROCESS_TECH_ITEM_ID = TMP.ID
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.BATCH_RATIO,
  DWI.PROTECH_NUM,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.INV_ORG_ID,
  DWI.IS_END_NODE,
  DWI.IS_LOCKED,
  DWI.IS_PHANTOM,
  DWI.IS_PROGRESS_COMPLETE,
  DWI.IS_START_NODE,
  DWI.NODE_NUMBER,
  DWI.ORDER_STATE,
  DWI.PRO_TECH_NAME,
  DWI.PROCESS_TECH_ID,
  DWI.SRC_MATERIAL_ID,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.PROCESS_TECH_ROUTE_ID,
  DWI.PRODUCT_ORDER_ID,
  DWI.QTY,
  DWI.SYNC_ID,
  DWI.TRANSFER_BATCH,
  DWI.TRANSPORT_TIME,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_APS_PROTECH_ORDER_T DWI
  LEFT JOIN SDI.SDI_APS_PROTECH_ORDER_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
