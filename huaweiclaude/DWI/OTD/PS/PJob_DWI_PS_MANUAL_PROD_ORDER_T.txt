--SDI_APS_MANUAL_PRO_ORDER_1007 到 DWI_PS_MANUAL_PROD_ORDER_T
--手工生产订单
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_MANUAL_PROD_ORDER_T
SELECT
  SDI.ID AS ID,
  SDI.CODE AS ORDER_NUM,
  SDI.CREATE_BY AS CREATE_BY,
  SDI.CREATE_DATE AS CREATE_DATE,
  SDI.CUSTOMER_ID AS SRC_CUST_ID,
  --源系统客户ID
  --取主数据取ID,源表关联字段为NULL,赋值-999999,源表关联字段有数据关联不上主数据表,赋值-999998
  CASE
    WHEN SDI.CUSTOMER_ID IS NULL
    OR SDI.CUSTOMER_ID = '' THEN -999999
    ELSE NVL(TMP.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SDI.CUSTOMER_ID IS NULL
    OR SDI.CUSTOMER_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编码
  SDI.DATA_SOURCE AS DATA_SOURCE,
  SDI.FACTORY_ID AS FACTORY_ID,
  SDI.INV_ORG_ID AS INV_ORG_ID,
  SDI.IS_CUSTOM AS IS_CUSTOM,
  SDI.IS_PHANTOM AS IS_PHANTOM,
  SDI.IS_SPLIT AS IS_SPLIT,
  SDI.IS_WHOLE AS IS_WHOLE,
  SDI.ITEM_ID AS SRC_MATERIAL_ID,
  --源系统物料ID
  CASE
    WHEN SDI.ITEM_ID IS NULL
    OR SDI.ITEM_ID = '' THEN -999999
    ELSE NVL(TMP1.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SDI.ITEM_ID IS NULL
    OR SDI.ITEM_ID = '' THEN 'SNULL'
    ELSE NVL(TMP1.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SDI.ORDER_DATE AS ORDER_DATE,
  SDI.ORDER_STATE AS ORDER_STATE,
  SDI.ORDER_TYPE_SET_ID AS ORDER_TYPE_SET_ID,
  SDI.PROMISE_DELIVERY AS PROMISE_DELIVERY,
  SDI.QTY AS ORDER_QTY,
  SDI.RELEVANT_CODE AS RELEVANT_CODE,
  SDI.REQUIRE_DELIVERY AS REQUIRE_DELIVERY,
  SDI.SPLIT_PRODUCT_ORDER_ID AS SPLIT_PROD_ORDER_ID,
  SDI.SYNC_ID AS SYNC_ID,
  SDI.UPDATE_BY AS UPDATE_BY,
  SDI.UPDATE_DATE AS UPDATE_DATE,
  SDI.UPLOAD_STATE AS UPLOAD_STATE,
  SDI.REMARK AS REMARK,
  SDI.RELEVANT_ORDER_NO AS RELEVANT_ORDER_NO,
  SDI.IS_SPECIAL_STOCK AS IS_SPECIAL_STOCK,
  'N' AS DEL_FL,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_APS_MANUAL_PRO_ORDER_1007 SDI --源系统ID LEFT JOIN (编码表和主数据表用编码关联取出ID,编码)
  LEFT JOIN (
    SELECT
      T1.ID,
      T2.CUST_ID,
      --客户ID
      T2.CUST_NUM --客户编码
    FROM
      SDI.SDI_CSM_CUSTOMER_1007 T1 --客户编码表
      INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T T2 --客户主数据表
      ON T1.CODE = T2.CUST_NUM
      AND T2.END_DATE = '4712-12-31'
      AND T2.DEL_FLAG = 'N'
      AND T1.IS_PHANTOM = 0 --取有效属性数据
      AND T2.GROUP_NUM = '800'
  ) TMP ON SDI.CUSTOMER_ID = TMP.ID
  LEFT JOIN (
    SELECT
      M1.ID,
      M2.MATERIAL_ID,
      --物料ID
      M2.MATERIAL_NUM --物料编码
    FROM
      SDI.SDI_ITEM_1007 M1 --物料编码表
      INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T M2 --物料主数据表
      ON M1.CODE = M2.MATERIAL_NUM
      AND M2.END_DATE = '4712-12-31'
      AND M2.DEL_FLAG = 'N'
      AND M1.IS_PHANTOM = 0 --取有效属性数据
      AND M2.GROUP_NUM = '800'
  ) TMP1 ON SDI.ITEM_ID = TMP1.ID
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.ORDER_NUM,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.SRC_CUST_ID,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.DATA_SOURCE,
  DWI.FACTORY_ID,
  DWI.INV_ORG_ID,
  DWI.IS_CUSTOM,
  DWI.IS_PHANTOM,
  DWI.IS_SPLIT,
  DWI.IS_WHOLE,
  DWI.SRC_MATERIAL_ID,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.ORDER_DATE,
  DWI.ORDER_STATE,
  DWI.ORDER_TYPE_SET_ID,
  DWI.PROMISE_DELIVERY,
  DWI.ORDER_QTY,
  DWI.RELEVANT_CODE,
  DWI.REQUIRE_DELIVERY,
  DWI.SPLIT_PROD_ORDER_ID,
  DWI.SYNC_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.UPLOAD_STATE,
  DWI.REMARK,
  DWI.RELEVANT_ORDER_NO,
  DWI.IS_SPECIAL_STOCK,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_MANUAL_PROD_ORDER_T DWI
  LEFT JOIN SDI.SDI_APS_MANUAL_PRO_ORDER_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
