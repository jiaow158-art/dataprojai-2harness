--SDI_PLAN_REVIEW_1007 到 DWI_PS_PLAN_REVIEW_T
--计划评审
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_PLAN_REVIEW_T
SELECT
  SRC.ID AS ID,
  SRC.COLORS AS COLORS,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.CUSTOM_FLAG AS CUSTOM_FLAG,
  CASE
    WHEN SRC.CUSTOMER_CODE IS NULL
    OR SRC.CUSTOMER_CODE = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SRC.CUSTOMER_CODE IS NULL
    OR SRC.CUSTOMER_CODE = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编号
  SRC.CUSTOMER_NAME AS CUST_NAME,
  SRC.CUSTOMIZED_REQUIREMENTS AS CUSTOMIZED_REQUIREMENTS,
  SRC.DATA_RESOURCE AS DATA_RESOURCE,
  SRC.DELIVERY_MODE AS DELIVERY_MODE,
  SRC.DELIVERY_TIME AS DELIVERY_TIME,
  SRC.DEMAND_DELIVERY_DATE AS DEMAND_DELIVERY_DATE,
  SRC.END_TIME AS END_TIME,
  SRC.ESTIMATED_DELIVERY_TIME AS ESTIMATED_DELIVERY_TIME,
  SRC.FACTORY_NAME AS FACTORY_NAME,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.ITEM_CATEGORY AS ITEM_CATEGORY,
  CASE
    WHEN SRC.ITEM_CODE IS NULL
    OR SRC.ITEM_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.ITEM_CODE IS NULL
    OR SRC.ITEM_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.ITEM_NAME AS SRC_MATERIAL_NAME,
  SRC.LINE_NO AS LINE_NO,
  SRC.NET_DEMAND_QTY AS NET_DEMAND_QTY,
  SRC.OUTPUT_TIME AS OUTPUT_TIME,
  SRC.OVERDUE_DATE AS OVERDUE_DATE,
  SRC.PACKAGING_CODE AS PACKAGING_CODE,
  SRC.PLAN_QTY AS PLAN_QTY,
  SRC.PLAN_TASK_ID AS PLAN_TASK_ID,
  SRC.PROCESSED_PRODUCT AS PROCESSED_PRODUCT,
  SRC.PROCESSED_QTY AS PROCESSED_QTY,
  SRC.PROCESSING_TIME AS PROCESSING_TIME,
  SRC.PRODUCT_ORDER_CODE AS PRODUCT_ORDER_CODE,
  SRC.PRODUCT_ORDER_ID AS PRODUCT_ORDER_ID,
  SRC.PRODUCT_ORDER_QTY AS PRODUCT_ORDER_QTY,
  SRC.REVIEW_RESULT AS REVIEW_RESULT,
  SRC.REVIEW_STATUS AS REVIEW_STATUS,
  SRC.SALE_ORDER_CODE AS SALE_ORDER_CODE,
  SRC.SALE_ORDER_DETAIL_ID AS SALE_ORDER_DETAIL_ID,
  SRC.SHIPPING_MODE AS SHIPPING_MODE,
  SRC.SPECIAL_PACKAGING AS SPECIAL_PACKAGING,
  SRC.SPECIAL_REQUIREMENT AS SPECIAL_REQUIREMENT,
  SRC.SPECIAL_THICKNESS AS SPECIAL_THICKNESS,
  SRC.START_TIME AS START_TIME,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.UPLOAD_OSS AS UPLOAD_OSS,
  SRC.UPLOAD_OSS_DATE AS UPLOAD_OSS_DATE,
  SRC.WIP_RESOURCE_ID AS WIP_RESOURCE_ID,
  SRC.REVIEW_TIME AS REVIEW_TIME,
  SRC.REVIEWER_ID AS REVIEWER_ID,
  SRC.SALE_ORG AS SALE_ORG,
  SRC.SALE_DEPT AS SALE_DEPT,
  SRC.SALE_DEPT_NAME AS SALE_DEPT_NAME,
  SRC.NOT_MERGE_FLAG AS NOT_MERGE_FLAG,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_PLAN_REVIEW_1007 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.CUSTOMER_CODE = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.ITEM_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.COLORS,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.CUSTOM_FLAG,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.CUST_NAME,
  DWI.CUSTOMIZED_REQUIREMENTS,
  DWI.DATA_RESOURCE,
  DWI.DELIVERY_MODE,
  DWI.DELIVERY_TIME,
  DWI.DEMAND_DELIVERY_DATE,
  DWI.END_TIME,
  DWI.ESTIMATED_DELIVERY_TIME,
  DWI.FACTORY_NAME,
  DWI.INV_ORG_ID,
  DWI.IS_PHANTOM,
  DWI.ITEM_CATEGORY,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.MATERIAL_NAME,
  DWI.LINE_NO,
  DWI.NET_DEMAND_QTY,
  DWI.OUTPUT_TIME,
  DWI.OVERDUE_DATE,
  DWI.PACKAGING_CODE,
  DWI.PLAN_QTY,
  DWI.PLAN_TASK_ID,
  DWI.PROCESSED_PRODUCT,
  DWI.PROCESSED_QTY,
  DWI.PROCESSING_TIME,
  DWI.PRODUCT_ORDER_CODE,
  DWI.PRODUCT_ORDER_ID,
  DWI.PRODUCT_ORDER_QTY,
  DWI.REVIEW_RESULT,
  DWI.REVIEW_STATUS,
  DWI.SALE_ORDER_CODE,
  DWI.SALE_ORDER_DETAIL_ID,
  DWI.SHIPPING_MODE,
  DWI.SPECIAL_PACKAGING,
  DWI.SPECIAL_REQUIREMENT,
  DWI.SPECIAL_THICKNESS,
  DWI.START_TIME,
  DWI.SYNC_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.UPLOAD_OSS,
  DWI.UPLOAD_OSS_DATE,
  DWI.WIP_RESOURCE_ID,
  DWI.REVIEW_TIME,
  DWI.REVIEWER_ID,
  DWI.SALE_ORG,
  DWI.SALE_DEPT,
  DWI.SALE_DEPT_NAME,
  DWI.NOT_MERGE_FLAG,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_PLAN_REVIEW_T DWI
  LEFT JOIN SDI.SDI_PLAN_REVIEW_1007 SRC ON DWI.ID = SRC.ID
WHERE
  SRC.ID IS NULL;
