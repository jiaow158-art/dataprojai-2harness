with BCS_DETAIL_01 as(   -- 条码系统明细数据(收坯数&综合合格数)
	SELECT NVL(T1.prod_date,T2.prod_date) AS prod_date, -- 生产时间
		   NVL(T1.factory_code,T2.factory_code) AS factory_code, -- 工厂编码
		   NVL(T1.material_num,T2.material_num) AS material_num, -- 物料编码
		   NVL(T1.actual_grouting_qty,0) AS ZJ_QTY, -- 注浆数
		   NVL(T1.in_stock_qty,0) AS SS_QTY, -- 实收数
		   NVL(T2.ysfj_qty,0) AS ysfj_qty, -- 一烧分级数
		   NVL(T2.ysyd_qty,0) AS ysyd_qty,  -- 一烧优等数
		   NVL(T2.csyd_qty,0) AS csyd_qty, -- 重烧优等数
		   NVL(T2.sy_jj_qty,0) AS sy_jj_qty -- 升级级数
	FROM DWIOTD.DWI_PAM_MAT_ACTUAL_ACCEPT_T T1 -- 收坯数
	full outer join DWIOTD.DWI_PAM_BASE_TWO_FJYD_T T2    -- 综合合格数
	on NVL(T1.prod_date,'NULL') = NVL(T2.prod_date,'NULL')
	AND NVL(T1.factory_code,'NULL') = NVL(T2.factory_code,'NULL')
	AND NVL(T1.material_num,'NULL') = NVL(T2.material_num,'NULL')
),
BCS_DETAIL_02 as(
    SELECT NVL(T1.prod_date,T2.prod_date) AS prod_date, -- 生产时间
		   NVL(T1.factory_code,T2.factory_code) AS factory_code, -- 工厂编码
		   NVL(T1.material_num,T2.material_num) AS material_num, -- 物料编码
		   NVL(T1.ZJ_QTY,0) AS ZJ_QTY, -- 注浆数
		   NVL(T1.SS_QTY,0) AS SS_QTY, -- 实收数
		   NVL(T1.ysfj_qty,0) AS ysfj_qty, -- 一烧分级数
		   NVL(T1.ysyd_qty,0) AS ysyd_qty,  -- 一烧优等数
		   NVL(T1.csyd_qty,0) AS csyd_qty, -- 重烧优等数
		   NVL(T1.sy_jj_qty,0) AS sy_jj_qty, -- 升级级数
		   NVL(T2.py_qty,0) AS py_qty -- 施釉数量
	FROM BCS_DETAIL_01 T1 -- 条码系统明细数据(收坯数&综合合格数)
	full outer join DWIOTD.DWI_PAM_BASE_PY_DATA_T T2    -- 施釉数据
	on NVL(T1.prod_date,'NULL') = NVL(T2.prod_date,'NULL')
	AND NVL(T1.factory_code,'NULL') = NVL(T2.factory_code,'NULL')
	AND NVL(T1.material_num,'NULL') = NVL(T2.material_num,'NULL')
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_COST_REDUCTION_BCS_T
select T2.prod_date AS prod_date, -- 生产时间
	   T2.factory_code AS factory_code, -- 工厂编码
	   T2.material_num AS material_num, -- 物料编码
	   NVL(T1.line_name,'普通线') as line_name, -- 普通线&高压线
	   T2.zj_qty AS ZJ_QTY, -- 注浆数
	   T2.ss_qty AS SS_QTY, -- 实收数
	   T2.ysfj_qty AS ysfj_qty, -- 一烧分级数
	   T2.ysyd_qty AS ysyd_qty,  -- 一烧优等数
	   T2.csyd_qty AS csyd_qty, -- 重烧优等数
	   T2.sy_jj_qty AS sy_jj_qty, -- 升级级数
	   T2.py_qty AS py_qty,-- 施釉数量
	   CURRENT_TIMESTAMP() AS dw_last_update_date -- 数据同步时间
from BCS_DETAIL_02 T2  -- 与线关联不需要全外关联
LEFT JOIN DWIOTD.DWI_PAM_BASE_LINE_DATA_T T1 --普通线&高压线
on T1.prod_date = T2.prod_date
AND T1.factory_code = T2.factory_code
AND T1.material_num = T2.material_num
where T2.prod_date <= substring('${PERIOD_ID_D}',1,7);
