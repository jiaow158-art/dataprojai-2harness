WITH JX_GY_PT_DATA as( -- 江西 普通线&高压线 逻辑  
select t.prod_date, 
       t.factory_code, 
       t.material_num,
       case when t.monitorname like '%高压%' then '高压线' else '普通线' end as line_name  -- 注浆班组
from(
select t.prod_date,
       t.factory_code,
	   t.material_num,
	   t.monitorname,
	   row_number() over(PARTITION by prod_date, factory_code, material_num order by 
			 case when monitorname like '%高压%' then 1 else 2 end asc) rn
	from(
		select substring(gp.proddate,1,7) as prod_date,    -- 生产日期
		       gp.FACTORY as factory_code,
		       --gpd.moldsid as mould_code,   -- 模具编号
		       --c1.modelid as material_num,
		       t1.material_num,
			   gp.monitorname
		      -- case when gp.monitorname like '%高压%' then '高压线' else '普通线' end as line_name -- --带班班长名称
		from SDI.SDI_qis_groutingplan_1011 gp  -- 注浆计划
		left join SDI.SDI_qis_groutingplandel_1011 gpd -- 注浆计划子表
		on gp.uuid = gpd.planno  
		left join SDI.SDI_qis_mud_1011 b  -- 泥坯
		on gpd.moldsid = b.mudid 
		and b.status<>'9'
		left join SDI.SDI_PRODUCTMODEL_1011 c1
		on b.mudid = c1.mudid
		left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t1 -- 生产BOM关系数据
        on gp.FACTORY = t1.factory_code
        and c1.modelid = t1.small_material_num
		where gp.status <> '9' 
		and gp.Prodstatus='0' 
		AND gp.FACTORY IN ('3200','3202')   
		AND gpd.attribute = '正常'  
		AND b.CATEGORY NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')  
		AND gp.proddate between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		group by substring(gp.proddate,1,7),
				 gp.FACTORY,
				-- gpd.moldsid ,
				-- c1.modelid,
				t1.material_num,
				 gp.monitorname
		        -- case when gp.monitorname like '%高压%' then '高压线' else '普通线' end	
		) t
	)t
	where rn = 1
),
jm_gy_pt_data as( -- 江门高压线及普通线逻辑
select t.prod_date, 
       t.factory_code, 
       t.material_num,
       case when t.jm_zj_class_name like '%高压%' then '高压线' else '普通线' end as line_name  -- 注浆班组
from(
	select t.prod_date,
		   t.factory_code,
		   t.material_num,
		   t.jm_zj_class_name,
		   row_number() over(PARTITION by prod_date, factory_code, material_num order by 
			 case when jm_zj_class_name like '%高压%' then 1 else 2 end asc) rn
	from(
		select substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date,  -- 生产日期
			   a.jm_factory_code as factory_code,
			   --a.jm_np_code,   -- 泥坯编码
			   --c.jm_code as material_num,     -- 物料
			   t1.material_num,
			   b.jm_zj_class_name
			  -- case when b.jm_zj_class_name like '%高压%' then '高压线' else '普通线' end as line_name  -- 注浆班组
		 from SDI.SDI_jm_t_wy_zj_plan_detail_1038 a -- 注浆计划产品明细
		join SDI.SDI_jm_t_wy_zj_plan_1038 b -- 注浆计划
		on b.jm_uuid=a.jm_main_id   
		left join SDI.sdi_jm_t_wy_base_gc_1038 c
		on a.jm_np_code = c.jm_np_code
		left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t1 -- 生产BOM关系数据
        on a.jm_factory_code = t1.factory_code
        and c.jm_code = t1.small_material_num
		where a.jm_is_del=0 
		and b.jm_is_del=0
		and c.jm_is_del=0
		and substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传当天
		group by substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) ,
				 a.jm_factory_code,
				 --a.jm_np_code,
				 --c.jm_code,
				 t1.material_num,
				 b.jm_zj_class_name
				 --case when b.jm_zj_class_name like '%高压%' then '高压线' else '普通线' end	
		)t
	)t
	where rn = 1
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_BASE_LINE_DATA_T
SELECT NVL(T1.prod_date,T2.prod_date) AS prod_date,
	   NVL(T1.factory_code,T2.factory_code) AS factory_code,
	   NVL(T1.material_num,T2.material_num) AS material_num,
	   NVL(T1.line_name,T2.line_name) AS line_name,
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM JX_GY_PT_DATA T1
FULL OUTER JOIN JM_GY_PT_DATA T2
on NVL(T1.prod_date,'NULL') = NVL(T2.prod_date,'NULL')
AND NVL(T1.factory_code,'NULL') = NVL(T2.factory_code,'NULL')
AND NVL(T1.material_num,'NULL') = NVL(T2.material_num,'NULL');
