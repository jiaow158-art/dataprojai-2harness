-- 实际成本
with dim_data_01 as(
	select distinct t1.kalnr,
	                t1.jahrper,
	                t2.matnr,
	                t2.bwkey
	from SDI.SDI_MLDOC_1001 t1 --STG.stg_mldoc_1001 t1
	join SDI.SDI_MBEW_1001 t2 -- STG.stg_mbew_1001 t2
	on t1.kalnr = t2.kaln1 
	where concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) = substring('${PERIOD_ID_D}',1,7) -- 传昨天所在的月份
	and t2.bwkey like '3%'
),
dim_dim_data_01 as(
	SELECT t.txele,t.elemt
	FROM SDI.SDI_TCKH1_1001 T --STG.stg_tckh1_1001 T 
	WHERE T.spras = '1' 
	AND T.elehk = 'M1'
	AND T.elemt in ('100','110','120','130','190','200','300','410',
	                 '421','422','430','440','450','461','462','463','464','465','466')
),
dim_dim_data_02 as(
	select t1.kalnr,
	       t1.jahrper,
	       t1.matnr,
	       t1.bwkey,
	       t2.elemt,
	       t2.txele
	from dim_data_01 t1
	left join dim_dim_data_01 t2
	on 1 = 1
),
sj_tmp_data as(
	SELECT t1.quant ,
	       t2.elesmhk,
	       t2.element,
	       t1.kalnr,
	       t1.jahrper,
	       t2.tot
	FROM SDI.SDI_MLDOC_1001 t1 --STG.stg_mldoc_1001 t1
	left join SDI.SDI_MLDOCCCS_1001 t2 -- stg.stg_mldocccs_1001 t2
	on t1.docref = t2.docref 
	WHERE 1 = 1
	and t2.curtp = 10
	AND t1.CATEG = 'ZU'
	AND t1.PTYP  = 'BF'
	and t2.elesmhk = 'M1'
	and concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) = substring('${PERIOD_ID_D}',1,7) -- 传昨天所在的月份
),
sj_tmp_data_02 as(
	select --t1.*,t2.txele
	     -- t2.kalnr,
	      t2.jahrper,
	      t2.matnr,
	      t2.bwkey,
	      t2.txele,
	      nvl(sum(t1.tot),0) as tot
	from dim_dim_data_02 t2
	JOIN sj_tmp_data t1  -- 去掉left join，过滤掉没数据的那部分维度，
	ON t1.kalnr =  t2.kalnr
	and t1.jahrper = t2.jahrper
	and t1.element = t2.elemt
	group by --t2.kalnr,
	         t2.jahrper,
	         t2.matnr,
	         t2.bwkey,
	         t2.txele
),
sj_tmp_data_03 as(
select --kalnr,
       jahrper,
       matnr,
       bwkey,
       sum(case when txele = '材料－沙泥' then tot else 0 end) sj_cl_ns_amt,
       sum(case when txele = '材料－化工' then tot else 0 end) sj_cl_hg_amt,
       sum(case when txele = '材料－配件' then tot else 0 end) sj_cl_pj_amt,
       sum(case when txele = '材料－包装' then tot else 0 end) sj_cl_bz_amt,
       sum(case when txele = '材料－其他材料' then tot else 0  end) sj_cl_qtcl_amt,
       sum(case when txele = '半成品' then tot else 0 end) sj_bcp_amt,
       sum(case when txele = '产成品' then tot else 0 end) sj_ccp_amt,
       sum(case when txele = '人工费' then tot else 0 end) sj_rgf_amt,
       sum(case when txele = '燃料-煤' then tot else 0 end) sj_rl_m_amt,
       sum(case when txele = '燃料-气' then tot else 0 end) sj_rl_q_amt,
       sum(case when txele = '电费' then tot else 0 end) sj_df_amt,
       sum(case when txele = '折旧及摊销' then tot else 0 end) sj_zjtx_amt,
       sum(case when txele = '委外加工/抛光费' then tot else 0 end) sj_wyjg_amt,
       sum(case when txele = '水费' then tot else 0 end) sj_sf_amt,
       sum(case when txele = '物料消耗' then tot else 0 end) sj_wlxh_amt,
       sum(case when txele = '维修费' then tot else 0 end) sj_wyf_amt,
       sum(case when txele = '柴油' then tot else 0 end) sj_cy_amt,
       sum(case when txele = '污水处理' then tot else 0 end) sj_wscl_amt,
       sum(case when txele = '其他费用' then tot else 0 end) sj_qtfy_amt,
       sum(case when txele in('材料－沙泥','材料－化工','材料－配件','材料－包装','材料－其他材料'
                     ,'半成品','产成品','人工费','燃料-煤','燃料-气','电费','折旧及摊销'
                     ,'委外加工/抛光费','水费','物料消耗','维修费','柴油','污水处理','其他费用') 
           then tot else 0 end) as sj_total_amt
from sj_tmp_data_02 t
group by --kalnr,
         jahrper,
         matnr,
         bwkey
),
sj_cost_qty_data as(
	SELECT --t1.kalnr,  -- 没有数量结构的成本估算的成本估算编号
	       t1.jahrper, -- 过账年份/期间 YYYYPPP
	       t2.matnr,  -- 物料编码
	       t2.bwkey,  -- 估价范围 工厂
	       sum(t1.quant) as quant  -- 以评估数量计量单位计的交易数量
	FROM SDI.SDI_MLDOC_1001 t1 --STG.stg_mldoc_1001 t1
	join SDI.SDI_MBEW_1001 t2 -- STG.stg_mbew_1001 t2
	on t1.kalnr = t2.kaln1 
	AND t1.CATEG = 'ZU'
	AND t1.PTYP  = 'BF'
	group by --t1.kalnr,
	         t1.jahrper,
	         t2.matnr,
	         t2.bwkey
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_ACTUAL_COST_SAP_T PARTITION(STAT_MONTH)
select t1.bwkey as factory_code, -- 工厂编码
       t1.matnr as material_num , -- 物料编码
       t3.material_name, -- 物料名称
       t1.jahrper as period_data, -- 期间
       '实际' as cost_type, -- 类型
       t2.quant as cost_qty , -- 成本数量
       t1.sj_total_amt / nvl(t2.quant,1) as cost_price, -- 成本单价
       t1.sj_total_amt as total_amt, -- 合计
       t1.sj_cl_ns_amt as cl_ns_amt , -- 材料 - 沙泥 
       t1.sj_cl_hg_amt as cl_hg_amt, -- 材料 - 化工
       t1.sj_cl_pj_amt as cl_pj_amt, -- 材料－配件
       t1.sj_cl_bz_amt as cl_bz_amt,  -- 材料 -- 包装
       t1.sj_cl_qtcl_amt as cl_qtcl_amt,  -- -- 材料－其他材料
       t1.sj_bcp_amt as bcp_amt,  -- 半成品
       t1.sj_ccp_amt as ccp_amt,  -- 产成品
       t1.sj_rgf_amt as rgf_amt,  -- 人工费
       t1.sj_rl_m_amt as lq_m_amt,   -- 燃料-煤
       t1.sj_rl_q_amt as lq_q_amt,   -- 燃料-气
       t1.sj_df_amt as df_amt,  -- 电费
       t1.sj_zjtx_amt as zj_tx_amt,  -- 折旧及摊销
       t1.sj_wyjg_amt as wwjg_pgf_amt, -- 委外加工/抛光费
       t1.sj_sf_amt as sf_amt,  -- 水费
       t1.sj_wlxh_amt as wlxh_amt,  -- 物料消耗
       t1.sj_wyf_amt as wxf_amt, -- 维修费
       t1.sj_cy_amt as cy_amt,   -- 柴油
       t1.sj_wscl_amt as wscl_amt, -- 污水处理
       t1.sj_qtfy_amt as qtfy_amt,   -- 其他费用
       current_timestamp() as dw_last_update_date,
       concat(substring(t1.jahrper,1,4),'-',substring(t1.jahrper,6,2)) as stat_month 
from sj_tmp_data_03 t1
left join sj_cost_qty_data t2
--on t1.kalnr    = t2.kalnr
on t1.jahrper = t2.jahrper
and t1.matnr   = t2.matnr
and t1.bwkey   = t2.bwkey
left join dwimd.dwi_md_data_material_general_t t3
on t1.matnr = t3.material_num;
