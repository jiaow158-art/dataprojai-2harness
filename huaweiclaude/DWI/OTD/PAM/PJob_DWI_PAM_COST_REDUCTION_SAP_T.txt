set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with tmp_data as(
SELECT NVL(T1.factory_code, T2.factory_code) AS factory_code, --工厂编码
	   NVL(T1.material_num,T2.material_num) AS material_num,  --物料编码
	   NVL(T1.material_name,T2.material_name) AS material_name, -- 物料名称
       NVL(T1.period_data,T2.period_data) AS period_data, -- 期间
       NVL(t1.cost_qty,0) as bz_cost_qty, --标准成本数量
       NVL(t1.bz_cost_price,0) AS bz_cost_price,     -- 标准成本单价
	   NVL(t1.bz_total_amt,0) AS bz_total_amt,	 -- 标准成本合计
	   NVL(t1.bz_cl_ns_amt,0) AS bz_cl_ns_amt,	-- 材料－沙泥
	   NVL(t1.bz_cl_hg_amt,0) AS bz_cl_hg_amt,	-- 材料－化工
	   NVL(t1.bz_cl_pj_amt,0) AS bz_cl_pj_amt,	-- 材料－配件
	   NVL(t1.bz_cl_bz_amt,0) AS bz_cl_bz_amt,	-- 材料－包装
	   NVL(t1.bz_cl_qtcl_amt,0) AS bz_cl_qtcl_amt, 	-- 材料－其他材料
	   NVL(t1.bz_bcp_amt,0) AS bz_bcp_amt,		-- 半成品
	   NVL(t1.bz_ccp_amt,0) AS bz_ccp_amt,		-- 产成品
	   NVL(t1.bz_rgf_amt,0) AS bz_rgf_amt,		-- 人工费
	   NVL(t1.bz_lq_m_amt,0) AS bz_lq_m_amt,		-- 燃料-煤
	   NVL(t1.bz_lq_q_amt,0) AS bz_lq_q_amt,		-- 燃料-气
	   NVL(t1.bz_df_amt,0) AS bz_df_amt,		-- 电费
	   NVL(t1.bz_zj_tx_amt,0) AS bz_zj_tx_amt,	-- 折旧及摊销
	   NVL(t1.bz_wwjg_pgf_amt,0) AS bz_wwjg_pgf_amt, -- 委外加工/抛光费
	   NVL(t1.bz_sf_amt,0) AS bz_sf_amt,	-- 水费
	   NVL(t1.bz_wlxh_amt,0) AS bz_wlxh_amt, -- 物料消耗
	   NVL(t1.bz_wxf_amt,0) AS bz_wxf_amt,  -- 维修费
	   NVL(t1.bz_cy_amt,0) AS bz_cy_amt,	-- 柴油
	   NVL(t1.bz_wscl_amt,0) AS bz_wscl_amt, -- 污水处理
	   NVL(t1.bz_qtfy_amt,0) AS bz_qtfy_amt, -- 其他费用
	   NVL(t2.cost_qty,0) as sj_cost_qty,			-- 实际成本数量
	   NVL(t2.cost_price,0) as sj_cost_price,		-- 实际单价
	   NVL(t2.total_amt,0) as sj_total_amt,		-- 实际合计
	   NVL(t2.cl_ns_amt,0) as sj_cl_ns_amt,		-- 材料－沙泥
	   NVL(t2.cl_hg_amt,0) as sj_cl_hg_amt,		-- 材料－化工
	   NVL(t2.cl_pj_amt,0) as sj_cl_pj_amt,		-- 材料－配件
	   NVL(t2.cl_bz_amt,0) as sj_cl_bz_amt,	   -- 材料－包装
	   NVL(t2.cl_qtcl_amt,0) as sj_cl_qtcl_amt, -- 材料－其他材料
	   NVL(t2.bcp_amt,0) as sj_bcp_amt,		-- 半成品
	   NVL(t2.ccp_amt,0) as sj_ccp_amt,		-- 产成品
	   NVL(t2.rgf_amt,0) as sj_rgf_amt,		-- 人工费
	   NVL(t2.lq_m_amt,0)	as sj_rq_m_amt,		-- 燃料-煤
	   NVL(t2.lq_q_amt,0) as sj_rq_q_amt,		-- 燃料-气
	   NVL(t2.df_amt,0) as sj_df_amt,  		-- 电费
	   NVL(t2.zj_tx_amt,0) as sj_zjtx_amt, 	-- 折旧及摊销
	   NVL(t2.wwjg_pgf_amt,0) as sj_wyjg_amt, -- 委外加工/抛光费
	   NVL(t2.sf_amt,0) as sj_sf_amt, 	-- 水费
	   NVL(t2.wlxh_amt,0) as sj_wlxh_amt, -- 物料消耗
	   NVL(t2.wxf_amt,0) as sj_wyf_amt,   -- 维修费
	   NVL(t2.cy_amt,0) as sj_cy_amt, 	-- 柴油
	   NVL(t2.wscl_amt,0) as sj_wscl_amt, -- 污水处理
	   NVL(t2.qtfy_amt,0) as sj_qtfy_amt, -- 其他费用
	   current_timestamp() as dw_last_update_date , -- 数据同步时间,
	   NVL(T1.stat_month,t2.stat_month) as stat_month
FROM DWIOTD.DWI_PAM_STANDARD_COST_SAP_T T1  -- 标准成本
FULL OUTER JOIN DWIOTD.DWI_PAM_ACTUAL_COST_SAP_T T2 -- 实际成本
ON NVL(T1.factory_code,'NULL') =  NVL(T2.factory_code,'NULL')
AND NVL(T1.material_num,'NULL') =  NVL(T2.material_num,'NULL')
AND NVL(T1.stat_month,'NULL') =  NVL(T2.stat_month,'NULL')
where NVL(T1.stat_month,t2.stat_month) = SUBSTRING('${PERIOD_ID_D}',1,7)
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_COST_REDUCTION_SAP_T PARTITION(STAT_MONTH)
select *
from tmp_data t;
