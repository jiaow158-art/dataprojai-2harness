set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
INSERT OVERWRITE TABLE DWIOTD.DWI_PLAN_PRODUCT_SPECTRUM_T PARTITION(stat_month)
select t2.code AS material_num,
  	   current_timestamp() as dw_last_update_date,
  	   t3.code as old_material_num,
  	   substring(add_months('${PERIOD_ID_D}',2),1,7) stat_month
from sdi.SDI_WYAPS_Product_Spectrum_1007 t1
LEFT JOIN sdi.sdi_item_1007 t2
on t1.item_id = t2.id 
LEFT JOIN sdi.sdi_item_1007 t3
on t1.item_id = t3.id
where SUBSTR(t1.verson_number,3,6)=REPLACE(substring('${PERIOD_ID_D}',1,7),'-','')
and t1.inv_org_id = '2'
and t1.is_phantom = '0';
