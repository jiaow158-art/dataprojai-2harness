set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIOTD.DWI_WM_GOODS_STOCK_SITE_T		--DWI.SAP货物库位表
SELECT 
		SL1.MANDT	   AS  GROUP_NUM,
		SL1.LGNUM      AS  BLEND_WAREHOUSE,
		SL1.LQNUM      AS  QTY,
		SL1.AZEIT      AS  LAST_STOCK_OUT_TIME,
		SL1.BZEIT      AS  LAST_GOODS_TRANSF_TIME,
		SL1.EZEIT      AS  LAST_INTO_DATE,
		SL1.POSNR      AS  DELIVERY_PROJECT,
		CASE WHEN SL1.MATNR IS NULL OR SL1.MATNR = ''	--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)		--关联不上主数据时赋值-999998
        END 	   AS  MATERIAL_ID,		--物料ID
		CASE
		WHEN SL1.MATNR IS NULL OR SL1.MATNR = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END   		   AS  MATERIAL_NUM,	--物料编号
		CASE WHEN SL1.WERKS IS NULL OR SL1.WERKS = ''	--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)		--关联不上主数据时赋值-999998
        END 	   AS  FACTORY_ID, 
		CASE
		WHEN SL1.WERKS IS NULL OR SL1.WERKS = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 		   AS  FACTORY_CODE,		--工厂编码
		SL1.MEINS      AS  BASE_MEASURE_UNIT,
		SL1.BESTQ      AS  STOCK_CATG,
		T.CONFIG_CODE      AS  SPECIAL_INV_FLAG,
		SL1.BETYP      AS  REQ_TYPE,
		SL1.LGTYP      AS  STORAGE_TYPE,
		SL1.WEPOS      AS  RECEIVE_PROJECT,
		SL1.MGEWI      AS  MATERIAL_WEGT,
		SL1.VBELN      AS  MATERIAL,
		SL1.CHARG      AS  BATCH_NUM,
		CASE WHEN SL1.WDATU = '00000000' 	--判断SDI源表字段收货日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SL1.WDATU, 1, 4) = '9999' 	--SDI源表字段收货日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SL1.WDATU, 'yyyyMMdd'), 'yyyy-MM-dd')
		END  		   AS RECEIVE_DATE,
		SL1.LGPLA      AS  STORE_PLACE,
		SL1.LGORT      AS  INV_LOCATION,
		SL1.VERME      AS  AVAIL_STOCK,
		SL1.BTANR      AS  LAST_STOCK_UP_CREDENCE,
		SL1.LETYP      AS  STORAGE_UNIT_TYPE,
		SL1.ZEUGN      AS  USER_NO,
		SL1.SONUM      AS  SPECIAL_INV_CODE,
		SL1.VFDAT      AS  GOODS_SHELVES_DUEDATE,
		SL1.GESME      AS  TOTAL_QTY,
		SL1.TBNUM      AS  TRANSF_REQUIREMENTS,
		CASE WHEN SL1.BDATU = '00000000' 	--判断SDI源表字段上一次货物转储日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SL1.BDATU, 1, 4) = '9999' --SDI源表字段上一次货物转储日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SL1.BDATU, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 		   AS LAST_GOODS_TRANSF_DATE,
		SL1.WENUM      AS  RECEIVE_CODE,
		SL1.BENUM      AS  REQUEST_CODE,
		CASE WHEN SL1.ADATU = '00000000' 	--判断SDI源表字段最近一次出库日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SL1.ADATU, 1, 4) = '9999' 	--SDI源表字段最近一次出库日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SL1.ADATU, 'yyyyMMdd'), 'yyyy-MM-dd')
		END			   AS RECENTLY_OUT_DATE,
		CASE WHEN SL1.EDATU = '00000000' 	--判断SDI源表字段最近一次入库日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SL1.EDATU, 1, 4) = '9999' 	--SDI源表字段最近一次入库日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SL1.EDATU, 'yyyyMMdd'), 'yyyy-MM-dd')
		END            AS RECENTLY_INTO_DATE,
		CASE WHEN SL1.ZDATU = '00000000' 	--判断SDI源表字段上次加进现有库存的日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SL1.ZDATU, 1, 4) = '9999' 	--SDI源表字段上次加进现有库存的日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SL1.ZDATU, 'yyyyMMdd'), 'yyyy-MM-dd')
		END  		   AS LAST_CURRENT_STOCK_DATE,
		SL1.BTAPS      AS  LAST_TRANSF_PROJECT,
		SL1.IVPOS      AS  STOCK_FILE_PROJECT_NO,
		SL1.EINME      AS  INTO_WAREHOUSE_QTY,
		SL1.AUSME      AS  DELETE_QTY,
		SL1.GEWEI      AS  WEIGHT_UNIT,
		SL1.IDATU      AS  LAST_CHECK_QTY_DATE,
		SL1.TRAME      AS  NO_CLEAR_TRANSF_QTY,
		SL1.QPLOS      AS  CHECK_BATCH_CODE,
		SL1.QKAPV      AS  QTY_ABILITY_USE,
		'N'		   AS  DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS  DW_LAST_UPDATE_DATE
		FROM SDI.SDI_LQUA_1001 SL1		--SDI.SAP货物库位表
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据,取物料ID
			 ON  SL1.MANDT = DMMG.GROUP_NUM			--集团
			 AND SL1.MATNR = DMMG.MATERIAL_NUM		--物料编号
			 AND DMMG.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF		--关联工厂主数据,取工厂ID
			 ON  SL1.MANDT = DMF.GROUP_NUM			--集团
			 AND SL1.WERKS = DMF.FACTORY_CODE		--工厂编码
			 AND DMF.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T	--关联通用配置表	
			 ON T.SRC_SYS_CONFIG_CODE = SL1.SOBKZ		--取特殊库存标识中文名称
			 AND T.GROUP_NUM = SL1.MANDT				--集团
			 AND T.SOURCE_CODE = 'T148T'
		UNION ALL		--SDI_LQUA_1001.SAP货物库位表 UNION ALL 拼接 DWI_WM_GOODS_STOCK_SITE_T.SAP货物库位表
		SELECT 
	        SC.GROUP_NUM
           ,SC.BLEND_WAREHOUSE
	       ,SC.QTY
	       ,SC.LAST_STOCK_OUT_TIME
	       ,SC.LAST_GOODS_TRANSF_TIME
	       ,SC.LAST_INTO_DATE
	       ,SC.DELIVERY_PROJECT
	       ,SC.MATERIAL_ID
		   ,SC.MATERIAL_NUM
	       ,SC.FACTORY_ID
	       ,SC.FACTORY_CODE
		   ,SC.BASE_MEASURE_UNIT
	       ,SC.STOCK_CATG
	       ,SC.SPECIAL_INV_FLAG
	       ,SC.REQ_TYPE
	       ,SC.STORAGE_TYPE
	       ,SC.RECEIVE_PROJECT
	       ,SC.MATERIAL_WEGT
	       ,SC.MATERIAL
	       ,SC.BATCH_NUM
	       ,SC.RECEIVE_DATE
	       ,SC.STORE_PLACE
	       ,SC.INV_LOCATION
	       ,SC.AVAIL_STOCK
	       ,SC.LAST_STOCK_UP_CREDENCE
	       ,SC.STORAGE_UNIT_TYPE
	       ,SC.USER_NO
	       ,SC.SPECIAL_INV_CODE
	       ,SC.GOODS_SHELVES_DUEDATE
	       ,SC.TOTAL_QTY
	       ,SC.TRANSF_REQUIREMENTS
	       ,SC.LAST_GOODS_TRANSF_DATE
	       ,SC.RECEIVE_CODE
	       ,SC.REQUEST_CODE
	       ,SC.RECENTLY_OUT_DATE
	       ,SC.RECENTLY_INTO_DATE
	       ,SC.LAST_CURRENT_STOCK_DATE
	       ,SC.LAST_TRANSF_PROJECT
	       ,SC.STOCK_FILE_PROJECT_NO
	       ,SC.INTO_WAREHOUSE_QTY
	       ,SC.DELETE_QTY
	       ,SC.WEIGHT_UNIT
	       ,SC.LAST_CHECK_QTY_DATE
	       ,SC.NO_CLEAR_TRANSF_QTY
	       ,SC.CHECK_BATCH_CODE
	       ,SC.QTY_ABILITY_USE
		   ,'Y' AS DEL_FLAG	 					
           ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
      FROM DWIOTD.DWI_WM_GOODS_STOCK_SITE_T SC		--DWI.SAP货物库位表
	 LEFT JOIN SDI.SDI_LQUA_1001 SE		--SDI.SAP货物库位表
	  ON SC.GROUP_NUM = SE.MANDT 								
     AND SC.BLEND_WAREHOUSE = SE.LGNUM 
	 AND SC.QTY = SE.LQNUM
   WHERE SE.MANDT  IS NULL
     AND SE.LGNUM  IS NULL
	 AND SE.LQNUM  IS NULL		
	 AND SC.DEL_FLAG = 'N'
