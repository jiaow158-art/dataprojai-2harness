SELECT							
    'CT' AS CUSTOMER_TYPE, -- 客户类型：CZ:瓷砖 SW：洁具							
	custcode AS customer_code, -- 客户编码						
	TransSeq, -- 交易序列号						
	LayerID, -- 层标识						
	CalcPrice, -- 计算价格						
	Balance, -- 库存余额						
	TransValue, -- 交易值						
	LayerInQty, -- 收货数量						
	LayerOutQ, -- 发货数量						
	RevalTotal, -- 库存重估总计						
	CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE, -- 同步数据时间	
    'new' as new_flag --版本标识，区分数据归档后的数据来源	
FROM SDI.sdi_ivl1_1010
