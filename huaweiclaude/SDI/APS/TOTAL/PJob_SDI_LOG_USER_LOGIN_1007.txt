insert overwrite table sdi.sdi_log_user_login_1007
select 
id,
cpu_model,
device_name,
display_metrics,
ip,
login_date,
login_id,
login_name,
login_operation,
mac,
model,
operating_system,
ram,
remarks,
soft_version,
system_type,
system_version,
'N' AS DEL_FLAG,
current_timestamp() as DW_LAST_UPDATE_DATE
from stg.stg_log_user_login_1007

 union all 

select t1.id
      ,t1.cpu_model
      ,t1.device_name
      ,t1.display_metrics
      ,t1.ip
      ,t1.login_date
      ,t1.login_id
      ,t1.login_name
      ,t1.login_operation
      ,t1.mac
      ,t1.model
      ,t1.operating_system
      ,t1.ram
      ,t1.remarks
      ,t1.soft_version
      ,t1.system_type
      ,t1.system_version
      ,t1.del_flag
      ,t1.dw_last_update_date
  from sdi.sdi_log_user_login_1007 t1
  left join stg.stg_log_user_login_1007 t2
    on t1.id = t2.id
 where t2.id is null
