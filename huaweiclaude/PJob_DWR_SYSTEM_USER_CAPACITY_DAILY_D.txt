INSERT OVERWRITE TABLE dwrltc.dwr_system_user_capacity_daily_d PARTITION(PERIOD_ID_D)
select  
system_flag,
count(distinct case when datediff('${PERIOD_ID_D}',
substring(login_date,1,10)) <=365 
AND IS_FROZE = 'N' then  user_id else null end ) as daily_active_users,--(有效用户数)
current_timestamp as dw_lastupdate_date,
count(DISTINCT user_id) as daily_active_users_all, --总访问用户数
count(DISTINCT case when   IS_FROZE = 'N' then user_id else null end ) as daily_active_users_kt, --开通用户数
SUM(CASE WHEN  IS_FROZE = 'N' then  login_lng ELSE 0 END ) AS login_lng,
count(distinct  case when substr(p_system_flag,1,3) = 'OMS' THEN 
    case when login_date = '${PERIOD_ID_D}' and IS_FROZE = 'N' then user_id else null  end 
    ELSE 
    case when login_date = '${PERIOD_ID_D}' then user_id else null  end 
    END 
    )as daily_active_users_login ,--当日登录人数
sum (case when login_date = '${PERIOD_ID_D}' and IS_FROZE = 'N' then login_lng else 0 end  )as daily_login_lng,--当日登录次数
role_name as role_name,
'${PERIOD_ID_D}' as  PERIOD_ID_D
from 
dwrdim.DWR_SYSTEM_ROLE_CAPACITY_D
where  login_date <='${PERIOD_ID_D}'
-- dwrltc.dwr_system_user_capacity_d
group by system_flag,role_name
