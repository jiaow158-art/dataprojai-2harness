INSERT OVERWRITE TABLE DWRHR.DWR_HR_EMP_LEAVE_INFO_F
SELECT A.PERNR AS EMP_NUM --员工工号
      ,A.NACHN AS EMP_NAME --员工姓名
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(SUBSTR(A.BEGDA,1,4)||'-'||SUBSTR(A.BEGDA,5,2)||'-'||SUBSTR(A.BEGDA,7,2) ||
       CASE WHEN COALESCE(A.BEGUZ,'') = '' THEN ' 00:00:00'
	        ELSE ' '||SUBSTR(A.BEGUZ,1,2)||':'||SUBSTR(A.BEGUZ,3,2)||':'||SUBSTR(A.BEGUZ,5,2)
			 END,'yyyy-MM-dd HH:mm:ss')) AS START_TIME --开始时间
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(SUBSTR(A.ENDDA,1,4)||'-'||SUBSTR(A.ENDDA,5,2)||'-'||SUBSTR(A.ENDDA,7,2) ||
       CASE WHEN COALESCE(A.ENDUZ,'') = '' THEN ' 00:00:00'
	        ELSE ' '||SUBSTR(A.ENDUZ,1,2)||':'||SUBSTR(A.ENDUZ,3,2)||':'||SUBSTR(A.ENDUZ,5,2)
			 END,'yyyy-MM-dd HH:mm:ss')) AS END_TIME --结束日期
      ,A.ORGEH_T AS DEPT_NAME --单位名称
      ,A.PLANS_T AS POST_NAME --岗位名称
      ,A.ATEXT AS ABSEN_TYPE_NAME --缺勤/出勤类型文本
      ,A.STDAZ AS ABSEN_TIMES --缺勤时数
      ,A.ABWTG AS ABSEN_DAYS --出勤和缺勤天数
      ,A.KALTG AS CAL_DAYS --日历天数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_ZTHRPT006_1001 A
 WHERE A.MANDT = 800
