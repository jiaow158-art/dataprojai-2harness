INSERT OVERWRITE TABLE DWRHR.DWR_HR_EMP_TRMNT_INFO_F PARTITION(STAT_MONTH)
SELECT
       A.BEGDA AS STAT_DATE --统计日期
      ,A.PERNR AS EMP_NUM --员工工号
      ,A.NACHN AS EMP_NAME --姓名
      ,A.ZZ_XXL AS ED_NAME --学历
      ,A.SEX AS GENDER --性别
      ,A.ZZ_AGE	AS AGE --年龄
      ,A.ZZ_NLD	AS AGE_SECTN --年龄段
      ,A.STAT1_T AS EMP_STATUS_NAME --岗位状态
      ,A.PBTXT AS PERS_AREA_NAME --人事范围
      ,A.BTEXT AS PERS_SUB_AREA_NAME --人事子范围
      ,A.PTEXT AS EMP_GROUP_NAME --员工组
      ,A.PTEXT1	AS EMP_SUB_GROUP_NAME --员工子组
      ,A.ZZ_ZWJB_TEXT AS EMP_POST_LVL_NAME --职务级别
	  ,COALESCE(B.LVL1_DEPT_SHORT_NAME,C.LVL1_DEPT_SHORT_NAME) AS LVL1_DEPT_NAME --一级部门名称
	  ,COALESCE(B.LVL2_DEPT_SHORT_NAME,C.LVL2_DEPT_SHORT_NAME) AS LVL2_DEPT_NAME --二级部门名称
      ,A.ORGEHL1_TXT AS LVL3_DEPT_NAME --三级部门名称
      ,A.ORGEHL2_TXT AS LVL4_DEPT_NAME --四级部门名称
      ,A.ORGEHL3_TXT AS LVL5_DEPT_NAME --五级部门名称
      ,A.ORGEHL4_TXT AS LVL6_DEPT_NAME --六级部门名称
      ,A.ORGEHL5_TXT AS LVL7_DEPT_NAME --七级部门名称
      ,A.ORGEHL6_TXT AS LVL8_DEPT_NAME --八级部门名称
      ,A.ORGEH_TXT AS DEPT_NAME --部门
      ,A.PLANS_TXT AS EMP_POST_NAME --岗位
      ,A.BEGDA AS TRMNT_DATE --离职日期
      ,A.ENDDA AS ENTRY_DATE --入职日期
      ,A.YGGL AS DIV_AGE --员工司龄
      ,A.ZZ_SLD	AS DIV_SECTN --司龄段
      ,A.ZZ_LZLX AS TRMNT_TYPE_NAME --离职类型
      ,A.MNTXT AS OPER_TYPE_NAME --操作类型
      ,A.MGTXT AS OPER_REASON --操作原因
      ,A.ZZ_JTYY AS TRMNT_REASON1 --离职原因1
      ,A.ZZ_JTYY1 AS TRMNT_REASON2 --离职原因2
      ,A.ZZ_JTYY2 AS TRMNT_REASON3 --离职原因3
      ,CASE WHEN A.ZZ_GJGW = '' THEN '否'
            ELSE A.ZZ_GJGW
	         END AS IS_KEY_POST --是否关键岗位
      ,CASE WHEN SUBSTR(A.BEGDA,1,7) <= '2023-05' THEN D.MPO_CAT_TYPE
	        ELSE A.ZZ_ZWLBWB
			 END AS EMP_POST_TYPE_NAME --职族
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.ZZ_GWSX_TEXT AS EMP_POST_ATTR_NAME --岗位属性
	  ,SUBSTR(A.BEGDA,1,7) AS STAT_MONTH --统计月份
  FROM SDI.SDI_ZHRTPA001_02_1001 A
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B
    ON A.ORGEHL1_TXT = B.DEPT_SHORT_NAME
   AND B.DEPT_LEVEL = 3
   AND SUBSTR(A.BEGDA,1,7) = B.STAT_MONTH
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F C
    ON A.ORGEHL1_TXT = C.DEPT_SHORT_NAME
   AND C.DEPT_LEVEL = 3
   AND SUBSTR(A.BEGDA,1,7) = SUBSTR(ADD_MONTHS(CAST(CONCAT(C.STAT_MONTH,'-01') AS DATE),1),1,7)   
  LEFT JOIN SDI.SDI_EHR_EMP_MPO_1037 D
    ON SUBSTR(A.BEGDA,1,4) = D.STAT_YEAR
   AND A.PERNR = D.EMP_NUM
   AND SUBSTR(A.BEGDA,1,7) <= '2023-05' --2023年5月及以前的职族用填报的 
 WHERE SUBSTR(A.BEGDA,1,7) IN(
      SELECT SUBSTR(A.BEGDA,1,7)
	    FROM SDI.SDI_ZHRTPA001_02_1001 A
	   WHERE A.ZUDATE >= REPLACE('${PERIOD_ID_D}','-','')
     )
  AND A.MANDT = 800
