WITH T1 AS(
SELECT DISTINCT STAT_DATE FROM SDI.SDI_WM_YHYH_MATERIAL_1037
),T2 AS(
SELECT STAT_DATE AS START_DATE
      ,LEAD(DATE_ADD(STAT_DATE,-1),1,'4712-12-31') OVER(ORDER BY STAT_DATE ASC) AS END_DATE
  FROM T1
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_VM_YHYH_MATERIAL_D
SELECT B.START_DATE
      ,B.END_DATE
	  ,A.PRODUCT_TYPE
	  ,A.PRODUCT_BOM_CODE
	  ,A.MATERIAL_CODE
	  ,A.MATERIAL_DESCRIPTION AS MATERIAL_NAME
	  ,A.MAIN_PUSH
	  ,A.DW_LAST_UPDATE_DATE AS UPDATE_TIME
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_WM_YHYH_MATERIAL_1037 A
  LEFT JOIN T2 B
    ON A.STAT_DATE = B.START_DATE
;
