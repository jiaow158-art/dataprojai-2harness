with tmposs as (
SELECT 
        T1.X_PROD_APP AS prodNudePriceAppr --审批产品裸价
       ,T1.X_EXT_PRICE as serviceCharge  --服务费率
       ,T10.X_APV_CPM as approvalComQuotation --Added by lyh 20191030 审批综合报价
       ,T1.X_FINAN_COST
       ,REP.DP_PROJECT_WBS_NUM  as wbs_num 
       ,t2.name  AS material_num   
       ,t1.LAST_UPD
       ,(T1.X_PROD_APP*(1- coalesce(T1.X_EXT_PRICE/100,0)) - T10.X_APV_CPM*coalesce(T1.X_FINAN_COST/100,0))/(m3.length_1 * m3.width/1000000) as yj_unit_price
       ,row_number() over(partition by  REP.DP_PROJECT_WBS_NUM order by t1.LAST_UPD desc ) as rn
 FROM sdi.sdi_S_ORDER_ITEM_1002 T1
 JOIN sdi.sdi_S_PROD_INT_1002   T2 ON  t1.prod_id = t2.row_id 
 JOIN sdi.sdi_S_ORDER_ITEM_X_1002 T10 ON  T1.ROW_ID =T10.ROW_ID 
 JOIN sdi.sdi_s_order_1002  T9  On  T1.ORDER_ID = T9.ROW_ID
 JOIN sdi.sdi_cx_proj_rep_1002 REP On t9.X_PROJ_REP_ID = REP.ROW_ID
 left join dwimd.dwi_md_data_material_general_t m3 on t2.name = m3.material_num
WHERE    1=1   
       AND t9.X_OD_TYPE = 'Quote'
       AND (T1.STATUS_CD != '无效' OR T1.STATUS_CD IS NULL)
       AND REP.DP_PROJECT_WBS_NUM IS NOT NULL
       AND  T10.X_APV_CPM>0
      AND t9.STATUS_CD in('已完成','合同审核','签订合同','全部转单','审批成功')
),
tmprtm as (
select b.wbs , a.material , a.review_quotation,a.contract_price,a.service_rate,
       a.contract_price*(1- coalesce(a.service_rate/100,0)) as yj_unit_price,
       row_number() over (partition by b.wbs , a.material order by b.contract_version desc ) rn 
from  sdi.sdi_contract_material_1033 a    
join   sdi.sdi_contract_1033  b   On  a.contract_id  = b.contract_id
where b.RECEIPTS_STATUS = 'PASS'
)
insert overwrite table   dwrltc.dwr_bo_wbs_material_price 
select  wbs as wbs_num ,material as material_num,yj_unit_price,'rtm' as data_source,current_timestamp() as  dw_last_update_date
from  tmprtm 
where rn = 1 
union  all  
select t1.wbs_num,t1.material_num,t1.yj_unit_price,'oss' as data_source,current_timestamp() as  dw_last_update_date
from tmposs t1 
left join tmprtm t2  On  t1.wbs_num = t2.wbs and t1.material_num = t2.material 
where t1.rn = 1 
and t2.wbs is null
