INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_BOM_HEAD_D
 SELECT 
     GT.MATL_TYPE
     ,GT.MATL_LIST
     ,GT.INNER_COUNTER
     ,GT.OPTIONAL_BOM
     ,GT.GROUP_NUM
     ,GT.LAST_CHG_DATE
     ,GT.VALID_DATE_BEGIN
     ,GT.PROD_BOM_FACTORY_CODE
     ,GT.PROD_BOM_FACTORY_NAME
     ,GT.CRT_DATE
     ,GT.DATA_CHG_FLAG
     ,GT.OPTIONAL_BOM_TEXT
     ,GT.BASIC_AMT
     ,GT.CHG_OBJECT_NAME
     ,GT.CRT_REC_USER
     , T.CONFIG_NAME AS PROD_CLASSIFY_NAME
     , T1.CONFIG_NAME AS BOM_STATUS_NAME
     ,GT.BOM_H_CHG_G_STATUS_FLAG
     ,GT.BOM_BASIC_UNIT_CODE
     ,GT.MATERIAL_ID
     ,GT.MATERIAL_NUM
     ,GT.MATERIAL_NAME
     ,GT.DEL_FLAG
     ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
     ,GT.START_DATE
     ,GT.END_DATE
FROM DWIMD.DWI_MD_BOM_HEAD_T GT
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
ON  GT.PROD_CLASSIFY_CODE = T.CONFIG_CODE 
AND GT.GROUP_NUM = T.GROUP_NUM
AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1 
ON  GT.BOM_STATUS_CODE = T1.CONFIG_CODE 
AND GT.GROUP_NUM = T1.GROUP_NUM
AND T1.LANG_CODE = '1';
