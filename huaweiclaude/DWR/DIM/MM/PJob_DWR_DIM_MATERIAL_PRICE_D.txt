INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MATERIAL_PRICE_D
SELECT  
       A.MATNR AS MATERIAL_NUM --物料编码 
      ,A.VTWEG AS SALE_CHNL_CODE --分销渠道
      ,A.VKORG AS SALE_ORG_CODE --销售组织
      ,A.KONDA AS PRICE_GRP_CODE --价格组
      ,A.KNUMH AS REC_NUM --条件记录号
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE --开始日期
      ,FROM_UNIXTIME(UNIX_TIMESTAMP(A.DATBI,'yyyyMMdd'),'yyyy-MM-dd') AS END_DATE --结束日期
      ,B.KBETR / B.KPEIN AS PRICE_AMT --价格
      ,B.KONWA AS CURRENCY_UNIT --货币单位
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE 
  FROM SDI.SDI_A941_1001 A
  LEFT JOIN SDI.SDI_KONP_1001 B
    ON A.KNUMH = B.KNUMH
   AND A.MANDT = B.MANDT
 WHERE A.MANDT = '800'
;
