INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MATERIAL_SALES_D
SELECT
     MD.MATERIAL_ID AS MATERIAL_ID
    ,MD.SALES_ORG_CODE AS SALES_ORG_CODE
    ,MD.MATERIAL_NUM AS MATERIAL_NUM
    ,MD.GROUP_NUM AS GROUP_NUM
    ,MD.DISTRIBUTION_CHANNEL_CODE AS DISTRIBUTION_CHANNEL_CODE
    ,MD.DISTRIBUTION_CHNL_NAME AS DISTRIBUTION_CHNL_NAME
    ,MD.SALES_ORG_NAME AS SALES_ORG_NAME
    ,MD.SPEC_D_CHAIN_MM_STATUS_CODE AS SPEC_D_CHAIN_MM_STATUS_CODE
    ,MD.CASH_DISCOUNT_FLAG AS CASH_DISCOUNT_FLAG
    ,T.CONFIG_NAME AS MATL_GROUP5_CODE
    ,T1.CONFIG_NAME AS MATL_GROUP4_CODE
    ,T2.CONFIG_NAME AS MATL_GROUP3_CODE
    ,T3.CONFIG_NAME AS MATL_GROUP2_CODE
    ,T4.CONFIG_NAME AS MATL_GROUP1_CODE
    ,T5.CONFIG_NAME AS MATL_STATISTICS_GROUP_CODE
    ,T6.CONFIG_NAME AS MATL_PRICE_GROUP_CODE
    ,MD.SPEC_D_C_MATL_S_VALID_DATE AS SPEC_D_C_MATL_S_VALID_DATE
    ,MD.MATL_M_FILE_ITEM_C_GROUP_CODE AS MATL_M_FILE_ITEM_C_GROUP_CODE
    ,MD.DELIVERY_FACTORY AS DELIVERY_FACTORY
    ,MD.MATL_SUBJ_SET_GROUP_CODE AS MATL_SUBJ_SET_GROUP_CODE
    ,MD.PROD_ATTRIBUTE_ID_6 AS PROD_ATTRIBUTE_ID_6
    ,MD.PROD_ATTRIBUTE_ID_5 AS PROD_ATTRIBUTE_ID_5
    ,MD.PROD_ATTRIBUTE_ID_4 AS PROD_ATTRIBUTE_ID_4
    ,MD.PROD_ATTRIBUTE_ID_3 AS PROD_ATTRIBUTE_ID_3
    ,MD.PROD_ATTRIBUTE_ID_2 AS PROD_ATTRIBUTE_ID_2
    ,MD.PROD_ATTRIBUTE_ID_10 AS PROD_ATTRIBUTE_ID_10
    ,MD.PROD_ATTRIBUTE_ID_1 AS PROD_ATTRIBUTE_ID_1
    ,MD.PROD_ATTRIBUTE_ID_9 AS PROD_ATTRIBUTE_ID_9
    ,MD.PROD_ATTRIBUTE_ID_8 AS PROD_ATTRIBUTE_ID_8
    ,MD.PROD_ATTRIBUTE_ID_7 AS PROD_ATTRIBUTE_ID_7
    ,T7.CONFIG_NAME AS PRODUCT_LEVEL_CODE
    ,MD.MATL_DEL_FLAG AS MATL_DEL_FLAG
    ,MD.DEL_FLAG
	,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
    ,MD.START_DATE
    ,MD.END_DATE
    ,MD.SRC_MATERIAL_NUM
FROM DWIMD.DWI_MD_MATERIAL_SALES_T MD
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
       ON MD.MATL_GROUP5_CODE = T.CONFIG_CODE
      AND MD.GROUP_NUM = T.GROUP_NUM
      AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1 
       ON MD.MATL_GROUP4_CODE = T1.CONFIG_CODE
      AND MD.GROUP_NUM = T1.GROUP_NUM
      AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2 
       ON MD.MATL_GROUP3_CODE = T2.CONFIG_CODE
      AND MD.GROUP_NUM = T2.GROUP_NUM
      AND T2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T3 
       ON MD.MATL_GROUP2_CODE = T3.CONFIG_CODE
      AND MD.GROUP_NUM = T3.GROUP_NUM
      AND T3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4 
       ON MD.MATL_GROUP1_CODE = T4.CONFIG_CODE
      AND MD.GROUP_NUM = T4.GROUP_NUM
      AND T4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T5 
       ON MD.MATL_STATISTICS_GROUP_CODE = T5.CONFIG_CODE
      AND MD.GROUP_NUM = T5.GROUP_NUM
      AND T5.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T6 
       ON MD.MATL_PRICE_GROUP_CODE = T6.CONFIG_CODE
      AND MD.GROUP_NUM = T6.GROUP_NUM
      AND T6.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T7 
       ON MD.PRODUCT_LEVEL_CODE = T7.CONFIG_CODE
      AND MD.GROUP_NUM = T7.GROUP_NUM
      AND T7.LANG_CODE = '1';
