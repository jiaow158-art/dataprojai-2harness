insert overwrite table dwrdim.dwr_dim_material_url_d
select b.MATNR as material_num --物料编码
      ,a.item_number as src_material_num --源系统物料编码
	  ,replace(replace(replace(a.img_list,'"',''),'[',''),']','') as url_text --URL路径
	  ,a.create_date as create_time --创建时间
	  ,a.update_date as update_time --更新时间
	  ,a.is_hide as is_hide_code --是否上架，true 上架、false 下架
	  ,case when a.is_hide = 'true' then '上架'
	        when a.is_hide = 'false' then '下架'
			 end as is_hide_name --是否上架名称
	  ,current_timestamp as dw_last_update_date --DW数据同步时间		 
  from sdi.SDI_PRODUCT_1047 a
 inner join SDI.SDI_MARA_1001 b
    on(true)
 where locate(a.item_number,b.MATNR) > 0
   and b.MANDT = '800'
   and a.is_delete = 0
