INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MATERIAL_ACCT_HIS_D
SELECT
MBEWH.MANDT AS GROUP_NUM,
MBEWH.MATNR AS MATERIAL_NUM,
MBEWH.BWKEY AS VALUE_RANGE,
MBEWH.BWTAR AS ASSESS_TYPE,
MBEWH.LFGJA AS CUR_YEAR,
MBEWH.LFMON AS CUR_MON,
CONCAT(MBEWH.LFGJA,'-',MBEWH.LFMON) AS YEAR_MON,
MBEWH.LBKUM AS TOTAL_ESTIM_INVENT,
MBEWH.SALK3 AS ESTIM_TOTAL_INVENT_VALUE,
MBEWH.VPRSV AS PRICE_CTR_FLAG,
MBEWH.VERPR AS MOV_AVG_PRICE,
MBEWH.STPRS AS STDRD_PRICE,
MBEWH.PEINH AS PRICE_UNIT,
MBEWH.BKLAS AS ASSESS_CLASS_CODE,
MBEWH.SALKV AS BASE_AVG_MOV_VALUE,
MBEWH.VKSAL AS SALES_TOTAL_ESTIM_INVENT,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM sdi.sdi_mbewh_1001 MBEWH
;
