      AND R.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R R1
       ON MD.CONSUM_PATTERNS = R1.DOMAIN_VALUE_CODE 
      AND R1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R R2
       ON MD.NO_COST_ACCT = R2.DOMAIN_VALUE_CODE 
      AND R2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R R3
       ON MD.POST_TO_INSPECT_STOCK = R3.DOMAIN_VALUE_CODE 
      AND R3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R R4
       ON MD.PUR_TYPE = R4.DOMAIN_VALUE_CODE 
      AND R4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
       ON MD.LOAD_GROUP_CODE = T.CONFIG_CODE
      AND MD.GROUP_NUM = T.GROUP_NUM
      AND T.LANG_CODE = '1'

       ON MD.REPE_PARAMETER_FILE_CODE = T1.CONFIG_CODE
      AND MD.GROUP_NUM = T1.GROUP_NUM
      AND T1.LANG_CODE = '1'

       ON MD.MATL_SHIP_GROUP_CODE = T2.CONFIG_CODE
      AND MD.GROUP_NUM = T2.GROUP_NUM
      AND T2.LANG_CODE = '1'

       ON MD.MATL_REQUIRE_PLAN_TYPE_CODE = T3.CONFIG_CODE
      AND MD.GROUP_NUM = T3.GROUP_NUM
      AND T3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4 
       ON MD.MATL_P_BATCH_PROCEDURE_CODE = T4.CONFIG_CODE
      AND MD.GROUP_NUM = T4.GROUP_NUM
      AND T4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T5 
       ON MD.ASSESS_CATEGORY_CODE = T5.CONFIG_CODE
      AND MD.GROUP_NUM = T5.GROUP_NUM
      AND T5.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T6 
       ON MD.PLAN_STRATEGY_GROUP_CODE = T6.CONFIG_CODE
      AND MD.GROUP_NUM = T6.GROUP_NUM
      AND T6.LANG_CODE = '1'

       ON MD.FACTORY_SPEC_MM_STATUS_CODE = T7.CONFIG_CODE
      AND MD.GROUP_NUM = T7.GROUP_NUM
      AND T7.LANG_CODE = '1'

       ON MD.PP_PLAN_PROCEDURE_CODE = T8.CONFIG_CODE
      AND MD.GROUP_NUM = T8.GROUP_NUM
      AND T8.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_MATL_REQ_PLAN_CONTROLLER_T SA
       ON MD.MATL_REQ_PLAN_CTRER_CODE = SA.MATL_REQ_PLAN_CTRER_CODE
	  AND MD.GROUP_NUM = SA.GROUP_NUM
	  AND MD.FACTORY_CODE = SA.FACTORY_CODE
LEFT JOIN DWIMD.DWI_MD_INVENT_LOC_T TL
      ON MD.OUT_PUR_WH_LOC_CODE = TL.INVENT_LOC_CODE
     AND MD.GROUP_NUM = TL.GROUP_NUM
     AND MD.FACTORY_CODE = TL.FACTORY_CODE
LEFT JOIN DWIMD.DWI_MD_PROFIT_CENTER_T CP
      ON MD.PROFIT_CENTER_CODE = CP.PROFIT_CENTER_CODE
     AND MD.GROUP_NUM = CP.GROUP_NUM
     AND CP.LANGUAGE_CODE = '1'
LEFT JOIN  DWIMD.DWI_MD_INVENT_LOC_T SP
      ON MD.DELIVERY_LOC_CODE = SP.INVENT_LOC_CODE
     AND MD.GROUP_NUM = SP.GROUP_NUM
     AND MD.FACTORY_CODE = SP.FACTORY_CODE;
