insert overwrite table DWRDIM.DWR_DIM_MAT_QUALITY_DETAIL_T_03
SELECT 
    g.material_num, -- 物料编号
    g.material_name, -- 物料名称
    c.assess_type, -- 评估类型
    c.value_range, -- 价值范围
    g.matl_type_code, --物料类型编码
    g.matl_type_name, --物料类型编码
    substr(c.matl_price_ctr,1,1) as matl_price_ctr, -- 物料价格确定
    c.price_ctr_flag,--物料价格控制
    g.matl_group_code, --物料组
    g.product_group_name, --物料组名称
    g.prod_property, -- 产品所有权  
    g.prod_property_name, --产品所有权名称
    g.product_level_code,--产品层次编码
    c.ASSESS_CLASS_CODE, -- 评估类代码
    -- 归类标识：根据产品属性和物料类型对产品进行分类
    CASE 
        WHEN g.prod_property IN ('A0001', 'A0002', 'C0001', 'C0002', 'C0003', 'C0004', 'C0006') THEN '瓷砖(产成品)'
        WHEN g.prod_property = 'B0001' THEN '卫浴(产成品)'
        WHEN g.prod_property = 'D0001' THEN '生态新材(产成品)'
        WHEN g.prod_property = 'E0001' THEN '国贸(产成品)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND c.VALUE_RANGE IN ('1200','1300','1350','1500','1550','1600','1650','1700','1750','1800','1850','1900','1950','7220','7320','7920','7A20','7D00','7D50','8120','8820') THEN '瓷砖(原材料)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND c.VALUE_RANGE IN ('3200','3320','3500','3602','3603','3606','3620','3700','3820','3920','3A20') 
        THEN '卫浴(原材料)'
        ELSE '其他' 
    END AS flag,    
    -- 评估类与产品层次匹配总行数：参考表 PRDHA 与产品层次代码匹配的记录数
    CASE WHEN e.PRDHA = g.product_level2_code OR e.PRDHA = g.product_level3_code THEN 1 ELSE 0 END AS acc_sum_count,
    -- 评估类与产品层次匹配错误行数：参考表 BKLAS 不为空且评估类代码不一致的记录数
    CASE WHEN e.BKLAS IS NOT NULL AND c.ASSESS_CLASS_CODE <> e.BKLAS THEN 1 ELSE 0 END AS acc_error_count,    
    1 AS PRI_SUM_COUNT,
    -- 价格控制中心相关错误行数：价格控制中心代码第一位与价格控制标志不匹配的记录数
    CASE WHEN substr(c.matl_price_ctr, 1, 1) <> 
    			case when c.price_ctr_flag = 'S' THEN '3'
                when c.price_ctr_flag = 'V' THEN '2'
                ELSE null END 
        THEN 1 ELSE 0 END AS PRI_ERROR_COUNT,
    --评估类在各个工厂视图的一致性 
    1 as acct_sum_count,
    case when count(distinct c.ASSESS_CLASS_CODE) over(partition by c.material_num) > 1 THEN 1 ELSE 0 END AS acct_error_count,
    CURRENT_TIMESTAMP()	AS DW_LAST_UPDATE_DATE			-- 数据同步时间
FROM (
    SELECT 
        material_num,
        assess_type,
        value_range,
        ASSESS_CLASS_CODE,
        matl_price_ctr,
        price_ctr_flag
    FROM 
        dwimd.dwi_md_material_acct_t
    WHERE 
        end_date = '4712-12-31' AND del_flag = 'N'
) c 
-- 关联物料评估数据，条件为物料编号相同且数据有效
LEFT JOIN dwimd.dwi_md_data_material_general_t g  
ON g.material_num = c.material_num
-- 关联评估类与产品层次参考表，根据参考表 PRDHA 长度匹配产品层次代码
LEFT JOIN sdi.sdi_zmmt016_1001 e ON 
    CASE 
        WHEN LENGTH(e.PRDHA) = 4 THEN g.product_level2_code
        WHEN LENGTH(e.PRDHA) = 7 THEN g.product_level3_code end = e.PRDHA
where g.x_dstrb_c_mm_status_code <> 'GG';
