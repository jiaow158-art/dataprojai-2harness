INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_COMPANY_D
SELECT
     MD.COMPANY_ID AS COMPANY_ID
    ,MD.COMPANY_CODE AS COMPANY_CODE
    ,MD.GROUP_NUM AS GROUP_NUM
    ,MD.LC_NO_ERD_FLAG AS LC_NO_ERD_FLAG
    ,MD.ERD_ENTRY_FLAG AS ERD_ENTRY_FLAG
    ,MD.DUTY_FREE_BUSI_TAX_CODE AS DUTY_FREE_BUSI_TAX_CODE
    ,MD.CASH_MANAGE_ACTIVE_FLAG AS CASH_MANAGE_ACTIVE_FLAG
    ,MD.PRODUCTION_COMPANY_FLAG AS PRODUCTION_COMPANY_FLAG
    ,MD.FM_ACTIVE_FLAG AS FM_ACTIVE_FLAG
    ,MD.OUTPUT_TAX_CODE AS OUTPUT_TAX_CODE
    ,MD.RECOMMEND_FISCAL_YEAR_FLAG AS RECOMMEND_FISCAL_YEAR_FLAG
    ,MD.NEED_BD_BALANCE_SHEET_FLAG AS NEED_BD_BALANCE_SHEET_FLAG
    ,MD.RECOMMEND_CURR_DATE_FLAG AS RECOMMEND_CURR_DATE_FLAG
    ,MD.FOREIGN_CURRENCY_CONVERSION AS FOREIGN_CURRENCY_CONVERSION
    ,MD.EXTEND_WITHHOLD_TAX_ACT_FLAG AS EXTEND_WITHHOLD_TAX_ACT_FLAG
    ,MD.REVERSE_SUBJECT_WAY_CODE AS REVERSE_SUBJECT_WAY_CODE
    ,MD.CURRENCY_CODE AS CURRENCY_CODE
    ,SA.ADDRESS_NAME AS ADDRESS_NAME
    ,MD.ANTI_ACCOUNTING_FLAG AS ANTI_ACCOUNTING_FLAG
    ,MD.INFLATION_WAY_CODE AS INFLATION_WAY_CODE
    ,MD.CREDIT_CTL_SCOPE_REWRITE_FLAG AS CREDIT_CTL_SCOPE_REWRITE_FLAG
    ,MD.COMPANY_NAME AS COMPANY_NAME
    ,MD.ACCT_PERIOD_VARIANT AS ACCT_PERIOD_VARIANT
    ,MD.LANG_CODE AS LANG_CODE
    ,MD.FUND_MANGT_ACT_UPDATE_FLAG AS FUND_MANGT_ACT_UPDATE_FLAG
    ,MD.HIDE_ITEM_FLAG AS HIDE_ITEM_FLAG
    ,MD.ASSIGN_FLAG AS ASSIGN_FLAG
    ,MD.CITY_NAME AS CITY_NAME
    ,MD.ORIGION_COMPANY_CODE AS ORIGION_COMPANY_CODE
    ,MD.COMPANY_VARIANT_CODE AS COMPANY_VARIANT_CODE
    ,MD.FIELD_VARIANT_STATUS AS FIELD_VARIANT_STATUS
    ,TS.CNTRY_REGION_NAME AS CNTRY_REGION_NAME
    ,MD.FISCAL_YEAR_VARIANT AS FISCAL_YEAR_VARIANT
    ,MD.SALE_PUR_TAX_GROUP_CODE AS SALE_PUR_TAX_GROUP_CODE
    ,MD.VAT_REGISTRATION_NUM AS VAT_REGISTRATION_NUM
    ,MD.FINANCIAL_MNGNT_SCOPE AS FINANCIAL_MNGNT_SCOPE
    ,MD.VOUCHER_ACT_TAX_DATE_FLAG AS VOUCHER_ACT_TAX_DATE_FLAG
    ,MD.PUR_SUBJECT_ACT_FLAG AS PUR_SUBJECT_ACT_FLAG
    ,MD.MAX_RATE_DEVIATION_PCT AS MAX_RATE_DEVIATION_PCT
    ,T.DOMAIN_VALUE_NAME AS MM_PUR_ORDER_CTL_NAME
    ,MD.POST_PERIOD_VARIATION_FLAG AS POST_PERIOD_VARIATION_FLAG
    ,T1.DOMAIN_VALUE_NAME AS SALE_ACCT_STATUS_COST_NAME
    ,MD.FLOWWORK_VARIANT_CODE AS FLOWWORK_VARIANT_CODE
    ,MD.CAL_TAX_NET_DISCOUNT_FLAG AS CAL_TAX_NET_DISCOUNT_FLAG
    ,MD.NET_AMT_DISCOUNT_FLAG AS NET_AMT_DISCOUNT_FLAG
    ,MD.RECEIV_ACCT_ACT_FLAG AS RECEIV_ACCT_ACT_FLAG
    ,MD.ACT_ACCT_ALLOCAT_FLAG AS ACT_ACCT_ALLOCAT_FLAG
    ,R.CONFIG_NAME AS CREDIT_CONTROL_NAME
    ,R1.CONFIG_NAME AS CAPITAL_RESERVE_STATUS_NAME
    ,R2.CONFIG_NAME AS ACCOUNT_TABLE_NAME
    ,MD.AMOUNT_CORRECT_CODE AS AMOUNT_CORRECT_CODE
    ,MD.TAX_RECEIV_DOC_TYPE_CODE AS TAX_RECEIV_DOC_TYPE_CODE
    ,MD.TAX_CODE_CORRECT_CODE AS TAX_CODE_CORRECT_CODE
    ,MD.REMIT_CHALLAN_DOC_TYPE_CODE AS REMIT_CHALLAN_DOC_TYPE_CODE
    ,MD.TEMPLATE_FLAG AS TEMPLATE_FLAG
    ,MD.DEL_FLAG
	,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
    ,MD.START_DATE
    ,MD.END_DATE
FROM DWIMD.DWI_MD_COMPANY_T MD
LEFT JOIN DWIMD.DWI_MD_ADDRESS_SERVICE_T SA
       ON MD.GROUP_NUM = SA.GROUP_NUM 
	  AND MD.ADDRESS_CODE = SA.ADDRESS_CODE
	  AND SA.LANGUAGE_CODE = '1' 
LEFT JOIN DWIMD.DWI_MD_COUNTRY_NAME_T TS
       ON MD.GROUP_NUM = TS.GROUP_NUM 
	  AND MD.CNTRY_REGION_CODE = TS.CNTRY_REGION_CODE
	  AND TS.LANGUAGE_CODE = '1' 
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T
       ON  MD.MM_PUR_ORDER_CTL_CODE = T.DOMAIN_VALUE_CODE  
      AND T.LANG_CODE = '1'
LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T1
       ON  MD.SALE_ACCT_STATUS_COST_CODE = T1.DOMAIN_VALUE_CODE  
      AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R
      ON  MD.CREDIT_CONTROL_SCOPE_CODE = R.CONFIG_CODE
      AND MD.GROUP_NUM = R.GROUP_NUM
      AND R.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R1
      ON  MD.CAPITAL_RESERVE_STATUS_CODE = R1.CONFIG_CODE
      AND MD.GROUP_NUM = R1.GROUP_NUM
      AND R1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R2
      ON  MD.ACCOUNT_TABLE_CODE = R2.CONFIG_CODE
      AND MD.GROUP_NUM = R2.GROUP_NUM
      AND R2.LANG_CODE = '1';
