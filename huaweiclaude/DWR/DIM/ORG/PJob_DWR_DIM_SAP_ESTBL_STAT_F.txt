WITH T1 AS(
SELECT *
      ,CASE WHEN COALESCE(A.ORGEH6,'') <> '' THEN '-'||A.ORGEH6 ELSE '' END AS LVL8_DEPT_NAME1
      ,CASE WHEN COALESCE(A.ORGEH5,'') <> '' THEN '-'||A.ORGEH5 ELSE '' END AS LVL7_DEPT_NAME1
      ,CASE WHEN COALESCE(A.ORGEH4,'') <> '' THEN '-'||A.ORGEH4 ELSE '' END AS LVL6_DEPT_NAME1
      ,CASE WHEN COALESCE(A.ORGEH3,'') <> '' THEN '-'||A.ORGEH3 ELSE '' END AS LVL5_DEPT_NAME1
      ,CASE WHEN COALESCE(A.ORGEH2,'') <> '' THEN '-'||A.ORGEH2 ELSE '' END AS LVL4_DEPT_NAME1
      ,CASE WHEN COALESCE(A.ORGEH1,'') <> '' THEN '-'||A.ORGEH1 ELSE '' END AS LVL3_DEPT_NAME1
  FROM SDI.SDI_ZHRTOM005A_1001 A
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_SAP_ESTBL_STAT_F PARTITION(STAT_MONTH)
SELECT 
       SUBSTR(A.UPDAT,1,4) || '-' || SUBSTR(A.UPDAT,5,2) || '-' || SUBSTR(A.UPDAT,7,2) AS STAT_DATE --统计日期
      ,COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股') LVL1_DEPT_SHORT_NAME --一级部门
	  ,B.LVL2_DEPT_SHORT_NAME --二级部门
	  ,A.ORGEH1 AS LVL3_DEPT_SHORT_NAME	--三级部门
      ,A.ORGEH2 AS LVL4_DEPT_SHORT_NAME	--四级部门
      ,A.ORGEH3 AS LVL5_DEPT_SHORT_NAME	--五级部门
      ,A.ORGEH4 AS LVL6_DEPT_SHORT_NAME	--六级部门
      ,A.ORGEH5 AS LVL7_DEPT_SHORT_NAME	--七级部门
      ,A.ORGEH6 AS LVL8_DEPT_SHORT_NAME	--八级部门
      ,A.JOB_TEXT AS EMP_POST_SUB_NAME --职务名称
      ,A.ZZ_ZWJB_TEXT AS EMP_POST_LVL_NAME --职务级别
      ,A.Z_NCBZ AS CONTRAST_ESTBL_NUM --对比编制
      ,A.ZZ_DYSL AS CUR_ESTBL_NUM --现有编制数
      ,A.ZZ_BZTZ AS PRCS_ESTBL_NUM --过程编制调整
      ,A.ZZ_SJBZ AS UND_PREP_ESTBL_NUM --在编人数
      ,A.SYRYS AS ACTUAL_NUM --实有人数
      ,A.CHAE AS OVER_ESTBL_NUM --缺编人数
      ,A.CBSJ AS OVER_TIME --超编时间
      ,A.ENAME AS UND_EMP_NAME --在职人员
	  ,CASE WHEN A.ZZ_ZWJB_TEXT = '员工级' THEN '员工级'
	        WHEN A.ZZ_ZWJB_TEXT = '主管级' THEN '主管级'
			WHEN A.ZZ_ZWJB_TEXT = '经理级' THEN '经理级'
			WHEN A.ZZ_ZWJB_TEXT IN('总监级','总经理级','总裁级','董事长级') THEN '总监及以上'
			 END AS EMP_LVL_TYPE_NAME --职级分类名称
	  ,CASE WHEN A.ZZ_ZWJB_TEXT = '员工级' THEN 1
	        WHEN A.ZZ_ZWJB_TEXT = '主管级' THEN 2
			WHEN A.ZZ_ZWJB_TEXT = '经理级' THEN 3
			WHEN A.ZZ_ZWJB_TEXT IN('总监级','总经理级','总裁级','董事长级') THEN 4
			 END AS EMP_LVL_TYPE_RANK --职级分类排序		 
	  ,CASE WHEN A.ZZ_ZWJB_TEXT IN('总监级','总裁级','总经理级') THEN '是'
	        ELSE '否'
			 END AS IS_MGMT_CADRES --是否管理干部
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.ZZ_GWSX_TEXT AS EMP_POST_ATTR_NAME --岗位属性
	  ,A.ZZ_ZWLB_TEXT AS EMP_POST_TYPE_NAME --职族
	  ,CASE WHEN A.ZZ_ZWLB_TEXT IN('管理职族','专业职族') THEN 'MP'
            WHEN A.ZZ_ZWLB_TEXT = '操作职族' AND A.ZZ_GWSX_TEXT = '间接产出' THEN 'O辅'
            WHEN A.ZZ_ZWLB_TEXT = '操作职族' AND A.ZZ_GWSX_TEXT = '直接产出' THEN 'OO'
             END AS EMP_POST_BIG_TYPE_NAME --职族分类
	  ,CASE WHEN COALESCE(A.ORGEH6,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1||A.LVL8_DEPT_NAME1
	        WHEN COALESCE(A.ORGEH6,'') <> ''
			THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1||A.LVL8_DEPT_NAME1			
			WHEN COALESCE(A.ORGEH5,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1
			WHEN COALESCE(A.ORGEH5,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1
	        WHEN COALESCE(A.ORGEH4,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1
			WHEN COALESCE(A.ORGEH4,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1
			WHEN COALESCE(A.ORGEH3,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1
			WHEN COALESCE(A.ORGEH3,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1
			WHEN COALESCE(A.ORGEH2,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1
			WHEN COALESCE(A.ORGEH2,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1
			WHEN COALESCE(A.ORGEH1,'') <> '' AND COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME||A.LVL3_DEPT_NAME1
			WHEN COALESCE(A.ORGEH1,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||A.LVL3_DEPT_NAME1
			WHEN COALESCE(B.LVL2_DEPT_SHORT_NAME,'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')||'-'||B.LVL2_DEPT_SHORT_NAME
			WHEN COALESCE(COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股'),'') <> ''
            THEN COALESCE(B.LVL1_DEPT_SHORT_NAME,'东鹏控股')
	         END AS DEPT_JG_NAME --部门结构名称
	  ,A.ZZ_SFGJGW AS IS_IMPT_NAME --是否关键岗位
	  ,SUBSTR(A.UPDAT,1,4) || '-' || SUBSTR(A.UPDAT,5,2) AS STAT_MONTH --统计月份
  FROM T1 AS A
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B
    ON A.ORGEH1 = B.DEPT_SHORT_NAME
   AND B.DEPT_LEVEL = 3
   AND REPLACE(B.STAT_DATE,'-','') = A.UPDAT
 WHERE A.MANDT = 800
   AND CASE WHEN A.ORGEH1 = '资本运营部' THEN 1
            WHEN A.ORGEH3 IN('陈村和园','鹏云','星信厂管理组') THEN 1
			ELSE 0
			 END = 0
