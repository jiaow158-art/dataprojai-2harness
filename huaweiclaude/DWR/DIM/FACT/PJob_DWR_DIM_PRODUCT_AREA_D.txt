--产区关系主数据
WITH T1 AS(
SELECT 
       A.ID
      ,A.BATCH_START
      ,A.DISTRIBUTION_FLAG
      ,A.INVENTORY_LOCATION
      ,A.IS_PHANTOM
      ,A.PLANT_CLASS
      ,A.PRODUCT_AREA_ID
	  ,A.PRODUCT_AREA_CODE
      ,A.PRODUCT_AREA_NAME
      ,A.SALE_FACTORY_CODE
      ,A.PRODUCT_FACTORY_CODE
	  ,B.SUPPLIER_OR_CREDITOR_ID
      ,B.SUPPLIER_OR_CREDITOR_NUM
	  ,B.SUPPLIER_OR_CREDITOR_NAME
      ,CASE WHEN A.FACTORY_TYPE = '0' THEN '自制'
	        WHEN A.FACTORY_TYPE = '1' THEN '外协'
			ELSE 'NA'
			 END AS FACTORY_TYPE
      ,A.CREATE_BY
      ,A.CREATE_DATE	  
      ,A.UPDATE_BY
      ,A.UPDATE_DATE
FROM DWIMD.DWI_MD_PRODUCT_AREA_T A
LEFT JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T B
ON CAST(B.SUPPLIER_OR_CREDITOR_NUM as decimal(30)) = cast(A.SUPPLIER_OR_CREDITOR_ACCT as decimal(30))
AND B.END_DATE ='4712-12-31'
AND B.GROUP_NUM = '800'
),T2 AS(
SELECT BATCH_START,COUNT(distinct PRODUCT_AREA_CODE) PRODUCT_AREA_CODE_CNT
  FROM T1
 GROUP BY BATCH_START
 HAVING COUNT(distinct PRODUCT_AREA_CODE)>1
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_PRODUCT_AREA_D
SELECT 
       A.ID
      ,A.BATCH_START
      ,A.DISTRIBUTION_FLAG
      ,A.INVENTORY_LOCATION
      ,A.IS_PHANTOM
      ,A.PLANT_CLASS
      ,A.PRODUCT_AREA_ID
	  ,A.PRODUCT_AREA_CODE
      ,A.PRODUCT_AREA_NAME
      ,A.SALE_FACTORY_CODE
      ,A.PRODUCT_FACTORY_CODE
	  ,A.SUPPLIER_OR_CREDITOR_ID
      ,A.SUPPLIER_OR_CREDITOR_NUM
	  ,A.SUPPLIER_OR_CREDITOR_NAME
      ,A.FACTORY_TYPE
      ,A.CREATE_BY
      ,A.CREATE_DATE	  
      ,A.UPDATE_BY
      ,A.UPDATE_DATE
	  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
	  ,CASE WHEN B.BATCH_START IS NOT NULL THEN 'GD'
 	        ELSE A.PRODUCT_AREA_CODE
			 END AS BELONG_AREA_CODE
	  ,CASE WHEN B.BATCH_START IS NOT NULL THEN '广东基地'
 	        ELSE A.PRODUCT_AREA_NAME
			 END AS BELONG_AREA_NAME
	  ,CASE WHEN A.PRODUCT_AREA_NAME IN ('广东产区','福建产区','广西产区') THEN '华南产区'
	        WHEN A.PRODUCT_AREA_NAME IN ('江西产区','湖南产区') THEN '华东产区'
	        WHEN A.PRODUCT_AREA_NAME = '西南产区' THEN '西南产区'
	        WHEN A.PRODUCT_AREA_NAME = '山东产区' THEN '华北产区'
	        WHEN A.PRODUCT_AREA_NAME = '山西产区' THEN '西北产区'
	   		END AS final_area_name
  FROM T1 A
  LEFT JOIN T2 B
    ON A.BATCH_START = B.BATCH_START
UNION ALL 
SELECT 
       null,
       a.BATCH_START,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       null,
       '自制' as FACTORY_TYPE,
       null,
       null,
       null,
       null,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
       null as BELONG_AREA_code,
       a.product_area_name as BELONG_AREA_NAME,
       CASE WHEN A.PRODUCT_AREA_NAME IN ('广东产区','福建产区','广西产区') THEN '华南产区'
	        WHEN A.PRODUCT_AREA_NAME IN ('江西产区','湖南产区') THEN '华东产区'
	        WHEN A.PRODUCT_AREA_NAME = '西南产区' THEN '西南产区'
	        WHEN A.PRODUCT_AREA_NAME = '山东产区' THEN '华北产区'
	        WHEN A.PRODUCT_AREA_NAME = '山西产区' THEN '西北产区'
	   		END AS final_area_name
 FROM (
SELECT
       batch_start,
       product_area_name, row_number() over(partition by batch_start order by case when product_area_name='广东基地' then 1 else 2 end) as rn
 from DWIOTD.DWI_PS_SELF_MADE_PRODUCTION_AREA_T 
 ) A
 left join T1 B ON A.BATCH_START = B.BATCH_START    
 where A.rn=1
 and b.BATCH_START is null
