INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_FACTORY_D
SELECT 
    MD.FACTORY_ID
   ,MD.GROUP_NUM
   ,MD.FACTORY_CODE
   ,MD.COST_OBJECT_CONTROL_FLAG
   ,MD.ACT_COST_UPDATE_ACTIVE_FLAG
   ,SA.ADDRESS_NAME
   ,T.REGION_DECS AS REGION_NAME
   ,MD.LANG_CODE
   ,MD.FACTORY_CALENDAR_CODE
   ,MD.POSTAL_CODE
   ,MD.CITY_NAME
   ,R.CONFIG_NAME AS PROD_GROUP_NAME
   ,TS.CNTRY_REGION_NAME
   ,MD.NAME_2
   ,MD.FACTORY_NAME
   ,PUR.PUR_ORG_DESC
   ,MD.RECEIVE_POINTS
   ,R1.CONFIG_NAME AS SALE_ORG_CODE
   ,MD.ACTIVE_PLAN_FLAG
   ,MD.STREET_HOUSE_NUM
   ,MD.FIRST_REMINDER_DAYS
   ,MD.SECOND_REMINDER_DAYS
   ,MD.THIRD_REMINDER_DAYS
   ,MD.CUST_NUM
   ,MD.VALUE_RANGE_CODE
   ,MD.FIRST_REMINDER_TEXT_NAME
   ,MD.SECOND_REMINDER_TEXT_NAME
   ,MD.THIRD_REMINDER_TEXT_NAME
   ,R2.CONFIG_NAME AS DISTRIBUTION_CHANNEL_CODE
   ,MD.WORK_COST
   ,R3.CONFIG_NAME AS NAME_COMPOSITION
   ,MD.PO_TOLERANCE_DAYS
   ,MD.DIFF_CODE
   ,MD.SRC_SUPPLIER_NUM
   ,MD.SUPPLIER_OR_CREDITOR_ID
   ,MD.CUST_ID
   ,MD.SUPPLIER_OR_CREDITOR_NAME
   ,MD.CUST_NAME
  ,MD.DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
  ,MD.START_DATE
  ,MD.END_DATE
  ,MD.SUPPLIER_OR_CREDITOR_NUM
  ,MD.SRC_CUST_NUM
FROM DWIMD.DWI_MD_FACTORY_T MD
LEFT JOIN DWIMD.DWI_MD_ADDRESS_SERVICE_T SA
       ON MD.GROUP_NUM = SA.GROUP_NUM 
	  AND MD.ADDRESS_ID = SA.ADDRESS_CODE
	  AND MD.LANG_CODE = SA.LANGUAGE_CODE 
LEFT JOIN DWIMD.DWI_MD_TAX_REGION_TEXT_T T
	  ON MD.REGION_CODE = T.REGION_CODE 
     AND MD.GROUP_NUM = T.GROUP_NUM 
     AND MD.LANG_CODE = T.LANGUAGE_CODE 
     AND MD.CNTRY_REGION_CODE = T.CNTRY_REGION_CODE 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R
      ON MD.PROD_GROUP = R.CONFIG_CODE 
     AND MD.GROUP_NUM = R.GROUP_NUM
     AND R.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R1
      ON MD.SALE_ORG_CODE = R1.CONFIG_CODE 
     AND MD.GROUP_NUM = R1.GROUP_NUM
     AND R1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R2
      ON MD.DISTRIBUTION_CHANNEL_CODE = R2.CONFIG_CODE 
     AND MD.GROUP_NUM = R2.GROUP_NUM
     AND R2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R3
      ON MD.NAME_COMPOSITION = R3.CONFIG_CODE 
     AND MD.GROUP_NUM = R3.GROUP_NUM
     AND R3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_COUNTRY_NAME_T TS
     ON MD.GROUP_NUM = TS.GROUP_NUM 
     AND MD.LANG_CODE = TS.LANGUAGE_CODE 
     AND MD.CNTRY_REGION_CODE = TS.CNTRY_REGION_CODE 
LEFT JOIN DWIMD.DWI_MD_PUR_ORG_TEXT_T PUR 
     ON MD.GROUP_NUM = PUR.GROUP_NUM 
	 AND MD.PUR_ORG = PUR.PUR_ORG;
