WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_TMS_CARR_D
SELECT A.ID AS ID --承运商ID
      ,A.PARTNER_KEY AS CARR_CODE --承运商编码
	  ,A.DESCR AS CARR_NAME --承运商名称
	  ,A.IS_ENABLE AS IS_ENABL_CODE --是否有效编码
	  ,T1.DESCR AS IS_ENABL_NAME --是否有效描述
	  ,A.GRADE AS PRIR_CODE --优先级编码
	  ,T2.NAME AS PRIR_NAME --优先级描述
	  ,A.ENTRY_DT AS CREATE_TIME --创建时间
	  ,A.ENTRY_ID AS CREATE_EMP_ID --创建人ID
	  ,T3.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T3.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATE_DT AS UPDATE_TIME --更新时间
	  ,A.UPDATE_ID AS UPDATE_EMP_ID --更新人ID
	  ,T4.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T4.NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间	  
  FROM SDI.SDI_PLAT_PARTNER_1005 A 
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.IS_ENABLE = T1.CODE
   AND T1.TYPENAME = 'YESNO'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.GRADE = T2.CODE
   AND T2.TYPENAME = 'TMS_PRIORITY'
  LEFT JOIN T AS T3
    ON A.ENTRY_ID = T3.ID
  LEFT JOIN T AS T4
    ON A.UPDATE_ID = T4.ID
 WHERE A.TENANT_ID = 1
   AND A.TRADE_TYPE = 'CAR'
