INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_PAM_WORK_NO_D
SELECT T1.site AS SITE_CODE, -- 基地编码
	   T2.site_desc AS SITE_NAME, -- 基地名称
	   T1.work_shop AS work_shop_code,      -- 车间编码 
	   T3.work_shop_desc AS work_shop_NAME, -- 车间名称
	   T1.product_line AS PRODUCT_LINE_CODE, -- 产线编码
	   T4.product_line_desc AS PRODUCT_LINE_NAME, -- 产线名称
	   T1.work_no_code , -- 工序编码
	   T1.work_no_desc , -- 工序名称
	   'N' AS del_flag, -- 删除数据标识
	   current_timestamp() AS dw_last_update_date, -- 最后更新日期
	   CASE WHEN t1.site = '1300' THEN '清远'
	   	    WHEN t1.site = '1500' THEN '丰城'
	   	    WHEN t1.site = '1600' THEN '澧县'
	   	    WHEN t1.site = '1800' THEN '重庆'
	   	    WHEN t1.site = '1900' THEN '山西'
	   ELSE '' END SITE_SHORT_NAME, -- 基地简称
	   T3.work_shop_type -- 车间类型
FROM SDI.SDI_DP_WORK_NO_1006 T1  -- 工序
LEFT JOIN SDI.SDI_M_SITE_1006 T2 -- 基地
ON T1.site = T2.site 
AND T2.enable_flag = '1'
LEFT JOIN SDI.SDI_M_WORK_SHOP_1006 T3 -- 车间
ON T1.site = T3.site 
AND T1.work_shop = T3.work_shop 
AND T3.state = '1'
LEFT JOIN SDI.SDI_M_PRODUCT_LINE_1006 T4 -- 产线
ON T1.site = T4.site 
AND T1.work_shop = regexp_extract(T4.work_shop_bo, '[^,]+:?\\d+?,(.*)', 1)
AND T1.product_line = T4.product_line 
AND T4.state = '1'
WHERE T1.delete_flag = 1;
