WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_TMS_SHR_PER_D
SELECT A.ID AS SHR_DLR_ID --共享仓ID
	  ,A.LOADING_PROVINCE AS DELV_PROV_CODE --提货省编码
	  ,T1.STREET_NAME AS DELV_PROV_NAME --提货省名称
	  ,A.LOADING_CITY AS DELV_CITY_CODE --提货市编码
	  ,T2.STREET_NAME AS DELV_CITY_NAME --提货市名称
	  ,A.LOADING_COUNTY AS DELV_DIST_CODE --提货区编码
	  ,T3.STREET_NAME AS DELV_DIST_NAME --提货区名称
	  ,A.UNLOADING_PROVINCE AS RECV_PROV_CODE --收货省编码
	  ,T4.STREET_NAME AS RECV_PROV_NAME --收货省名称
	  ,A.UNLOADING_CITY AS RECV_CITY_CODE --收货市编码
	  ,T5.STREET_NAME AS RECV_CITY_NAME --收货市名称
	  ,A.UNLOADING_COUNTY AS RECV_DIST_CODE --收货区编码
	  ,T6.STREET_NAME AS RECV_DIST_NAME --收货区名称
	  ,A.SHIP_METHOD AS SHIP_METH_CODE --运输方式编码
	  ,T7.DESCR AS SHIP_METH_NAME --运输方式名称
	  ,A.BMS_SKUTYPE AS BMS_MATERIAL_TYPE_CODE --BMS物料类型编码
	  ,T8.DESCR AS BMS_MATERIAL_TYPE_NAME --BMS物料类型名称
	  ,A.BUSINESS_TYPE AS BUSI_TYPE_CODE --业务类型编码
	  ,T9.DESCR AS BUSI_TYPE_NAME --业务类型名称
	  ,A.SHIP_DAYS AS SHIP_DAY --运输天数
	  ,B.RANGE_BEGIN AS RNG_BEGIN --区间开始
	  ,CASE WHEN B.RANGE_BEGIN_SYMBOL = '&gt;' THEN '>' END AS RNG_BEGIN_SYM --区间开始符号
	  ,CASE WHEN B.RANGE_END_SYMBOL = '&lt;=' THEN '<=' END AS RNG_END_SYM --区间结束符号
	  ,B.RANGE_END AS RNG_END --区间结束
	  ,A.STATUS AS ST_CODE --状态编码
	  ,T10.DESCR AS ST_NAME --状态描述
	  ,A.EFFECTIVE_DATE AS EFF_DATE --生效日期
	  ,A.INACTIVE_DATE AS INVAL_DATE --失效日期
	  ,B.NOTES AS REMARK --备注
	  ,A.ENTRYDT AS CREATE_TIME --创建时间
	  ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	  ,T11.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T11.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATEDT AS UPDATE_TIME --更新时间
	  ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	  ,T12.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T12.NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_TRANSPORT_AGING_SHARE_1005 A
  LEFT JOIN SDI.SDI_TMS_TRANSPORT_AGING_SHARE_D_1005 B
    ON A.ID = B.AGING_SHARE_ID
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T1
    ON A.LOADING_PROVINCE = T1.STREET_CODE
   AND T1.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T2
    ON A.LOADING_CITY = T2.STREET_CODE
   AND T2.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T3
    ON A.LOADING_COUNTY = T3.STREET_CODE
   AND T3.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T4
    ON A.UNLOADING_PROVINCE = T4.STREET_CODE
   AND T4.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T5
    ON A.UNLOADING_CITY = T5.STREET_CODE
   AND T5.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T6
    ON A.UNLOADING_COUNTY = T6.STREET_CODE
   AND T6.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T7
    ON A.SHIP_METHOD = T7.CODE
   AND T7.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T8
    ON A.BMS_SKUTYPE = T8.CODE
   AND T8.TYPENAME = 'BMS_SKU_TYPE' 
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T9
    ON A.BUSINESS_TYPE = T9.CODE
   AND T9.TYPENAME = 'BMS_QUOTE_ORDER_TYPE' 
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T10
    ON A.STATUS = T10.CODE
   AND T10.TYPENAME = 'EFFECTIVE_STATUS' 
  LEFT JOIN T AS T11
    ON A.ENTRYID = T11.ID
  LEFT JOIN T AS T12
    ON A.UPDATEID = T12.ID
