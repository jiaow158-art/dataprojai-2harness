WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_TMS_SITE_D
SELECT A.ID AS SITE_ID --站点ID
      ,A.CODE AS SITE_CODE --站点编码
	  ,A.NAME AS SITE_NAME --站点名称
	  ,A.DESCRIPTION AS SITE_DESC --站点描述
	  ,A.CONTACT_PERSONNAME1 AS CNTCT_NAME1 --联系人1名称
	  ,A.CONTACT_TELEPHONE1 AS CNTCT_PHONE_NUM1 --联系人1电话号码
	  ,A.CONTACT_PERSONNAME2 AS CNTCT_NAME2 --联系人2名称
	  ,A.CONTACT_TELEPHONE2 AS CNTCT_PHONE_NUM2 --联系人2电话号码
	  ,A.ADDRESS AS ADDR --详细地址
	  ,A.STATUS AS SITE_ST_CODE --站点状态编码
	  ,T7.DESCR AS SITE_ST_NAME --站点状态描述
	  ,A.SITE_LEVEL AS SITE_LVL_CODE --站点等级编码
	  ,T6.DESCR AS SITE_LVL_NAME --站点等级描述
	  ,A.SITETYPE AS SITE_TYPE_CODE --站点类型编码
	  ,T1.DESCR AS SITE_TYPE_NAME --站点类型描述
	  ,A.STREET_COD1 AS PROV_CODE --省份编码
	  ,T2.STREET_NAME AS PROV_NAME --省份名称
	  ,A.STREET_COD2 AS CITY_CODE --城市编码
	  ,T3.STREET_NAME AS CITY_NAME --城市名称
	  ,A.STREET_COD3 AS DIST_CODE --区编码
	  ,T4.STREET_NAME AS DIST_NAME --区名称
	  ,A.STREET_COD4 AS CNTY_CODE --镇编码
	  ,T5.STREET_NAME AS CNTY_NAME --镇名称
	  ,A.ENTRYDT AS CREATE_TIME --创建时间
	  ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	  ,T8.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T8.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATEDT AS UPDATE_TIME --更新时间
	  ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	  ,T9.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T9.NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_SITE_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.SITETYPE = T1.CODE
   AND T1.TYPENAME = 'TMS_SITE_SITETYPE'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T2
    ON A.STREET_COD1 = T2.STREET_CODE
   AND T2.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T3
    ON A.STREET_COD2 = T3.STREET_CODE
   AND T3.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T4
    ON A.STREET_COD3 = T4.STREET_CODE
   AND T4.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T5
    ON A.STREET_COD4 = T5.STREET_CODE
   AND T5.STREET_LEVEL = '4'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T6
    ON A.SITE_LEVEL = T6.CODE
   AND T6.TYPENAME = 'TMS_SITE_LEVEL'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T7
    ON A.STATUS = T7.CODE
   AND T7.TYPENAME = 'TMS_SITE_STATUS'
  LEFT JOIN T AS T8
    ON A.ENTRYID = T8.ID
  LEFT JOIN T AS T9
    ON A.UPDATEID = T9.ID
 WHERE A.TENANT_ID = 1
