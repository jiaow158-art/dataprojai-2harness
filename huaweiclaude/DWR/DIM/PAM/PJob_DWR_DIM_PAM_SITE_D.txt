INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_PAM_SITE_D
SELECT site AS SITE_CODE,  -- 基地编码
	   site_desc AS SITE_NAME, -- 基地名称
	   'N' AS del_flag, -- 删除数据标识
	   current_timestamp() AS dw_last_update_date, -- 最后更新日期
	   CASE WHEN site = '1300' THEN '清远'
	   		WHEN site = '1500' THEN '丰城'
	   		WHEN site = '1600' THEN '澧县'
	   		WHEN site = '1800' THEN '重庆'
	   		WHEN site = '1900' THEN '山西'
	   		ELSE '' END SITE_SHORT_NAME -- 基地简称
FROM SDI.SDI_M_SITE_1006  -- 基地
WHERE enable_flag = '1';
