WITH T1 AS(
SELECT A.DICT_TYPE
      ,A.DICT_VALUE
	  ,A.DICT_NAME
	  ,B.DICT_TYPE_NAME
  FROM SDI.SDI_BS_DICT_1030 A
 INNER JOIN SDI.SDI_BS_DICT_TYPE_1030 B
    ON A.DICT_TYPE = B.DICT_TYPE
   AND A.EP_ID = B.EP_ID
   AND A.APP_ID = B.APP_ID	
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND B.DICT_TYPE_NAME = '服务商星级别'
   AND B.ABLE_STATUS = 1
),T2 AS(
SELECT PROVIDER_ID
      ,CONCAT_WS(',',COLLECT_SET(CONCAT(CONCAT(PROVINCE,CITY),DISTRICT))) AS PRVDR_DIST --服务区域
  FROM SDI.SDI_HD_PROVIDER_AREA_REL_1030
 WHERE EP_ID = 3044
   AND APP_ID = 1831
 GROUP BY PROVIDER_ID
),T3 AS(
/*
服务城市计算逻辑：
1、如果服务商的服务城市只有一个，且服务城市为县时，算一个服务城市；
2、除去1中的服务商，统计服务城市为市一级城市数量（市一级不为空）；
3、统计市一级城市为空的省级城市数量
4、三者数量求和
*/
SELECT PROVIDER_ID
      ,CONCAT(CONCAT(PROVINCE,CITY),DISTRICT) AS PRVDR_DIST --服务区域省市区
	  ,CONCAT(PROVINCE,CITY) AS PRVDR_DIST1 --服务区域省市
	  ,CASE WHEN DISTRICT LIKE '%县' THEN 1 ELSE 0 END AS IS_DIS --是否县
  FROM SDI.SDI_HD_PROVIDER_AREA_REL_1030
 WHERE EP_ID = 3044
   AND APP_ID = 1831
   AND DISTRICT IS NOT NULL
),T4 AS(
SELECT A.PROVIDER_ID
      ,COUNT(DISTINCT CASE WHEN IS_DIS = 1 THEN PRVDR_DIST END) AS DIST_CNT --服务区域数省市区
	  ,COUNT(DISTINCT PRVDR_DIST1) AS DIST_CNT1 --服务区域数省市
  FROM T3 AS A
 GROUP BY A.PROVIDER_ID
HAVING COUNT(DISTINCT CASE WHEN IS_DIS = 1 THEN PRVDR_DIST END) = 1
),T5 AS(
SELECT A.PROVIDER_ID
      ,COUNT(DISTINCT CONCAT(PROVINCE,CITY)) AS DIST_CNT --服务区域数省市
  FROM SDI.SDI_HD_PROVIDER_AREA_REL_1030 A
  LEFT JOIN T4 AS B
    ON A.PROVIDER_ID = B.PROVIDER_ID
   AND B.DIST_CNT1 = B.DIST_CNT
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.CITY IS NOT NULL
   AND B.PROVIDER_ID IS NULL
 GROUP BY A.PROVIDER_ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_ZDJ_PRVDR_F
SELECT
       A.ID AS PRVDR_ID	--服务商ID
      ,A.PROVIDER_NAME AS PRVDR_NAME --服务商名称
      ,A.PROVIDER_CODE AS PRVDR_CODE --服务商编码
      ,A.DEPT_ID --部门ID
      ,A.CONTACT_PERSON	--联系人
      ,A.CONTACT_PHONE AS CONTACT_PHONE_NUM --联系人手机号
      ,A.COUNTRY AS CTRY_NAME --国家
      ,A.PROVINCE_ID AS PROV_ID --省ID
      ,A.PROVINCE_CODE AS PROV_CODE --省编码
      ,A.PROVINCE AS PROV_NAME --省
      ,A.CITY_ID --市ID
      ,A.CITY_CODE --市编码
      ,A.CITY --市
      ,A.DISTRICT_ID AS DIST_ID --区ID
      ,A.DISTRICT_CODE AS DIST_CODE --区编码
      ,A.DISTRICT AS DIST_NAME --区
      ,A.STREET	--街道
      ,A.ADDRESS AS ADDR --详细地址
      ,A.POSTAL_CODE AS POST_CODE --邮政编码
      ,A.BRAND_NAME	--品牌名称
      ,A.LONGITUDE AS ADDR_LON --地理经度
      ,A.LATITUDE AS ADDR_LAT --地理纬度
      ,A.PROVIDER_LEVEL AS PRVDR_LVL_ID	--服务商等级
	  ,B.DICT_NAME AS PRVDR_LVL_NAME --服务商等级名称
      ,A.ACCUMULATIVE_SIGNATURE	AS ORDER_CNT --累计签单
      ,A.ACCUMULATIVE_SALES	AS ORDER_AMT --累计销售
      ,A.AVAILABLE_POINT AS AVAIL_POINT_AMT --可用积分
      ,A.AUTHORIZATION_START_TIME AS START_TIME --授权开始时间
      ,A.AUTHORIZATION_END_TIME	AS END_TIME --授权结束时间
      ,A.DEFAULT_STATUS	AS IS_DFLT --默认服务商,默认不是默认服务商
      ,A.STATUS	--启用禁用状态0-禁用,1-启用
	  ,CASE WHEN A.STATUS = 0 THEN '禁用'
	        WHEN A.STATUS = 1 THEN '启用'
			 END AS STATUS_NAME --启用禁用状态名称
      ,A.AUTHORIZATION_STATUS AS AUTH_STATUS --授权状态,默认已授权,1-授权，0-未授权
	  ,CASE WHEN A.AUTHORIZATION_STATUS = 1 THEN '授权'
	        WHEN A.AUTHORIZATION_STATUS = 0 THEN '未授权'
			 END AS AUTH_STATUS_NAME --授权状态名称
      ,A.INTEGRAL AS POINT_AMT --积分
	  ,A.CREATE_BY AS CREATE_ID --创建人
      ,A.CREATE_TIME --创建时间
      ,A.CREATE_NAME --创建人名称
	  ,A.UPDATE_BY AS UPDATE_ID --更新人
      ,A.UPDATE_TIME --更新时间
      ,A.UPDATE_NAME --更新人名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,C.PRVDR_DIST --服务区域
	  ,COALESCE(D.DIST_CNT,E.DIST_CNT) AS DIST_CNT --服务区域数
  FROM SDI.SDI_HD_PROVIDER_1030 A
  LEFT JOIN T1 AS B
    ON A.PROVIDER_LEVEL = B.DICT_VALUE
  LEFT JOIN T2 AS C
    ON A.ID = C.PROVIDER_ID
  LEFT JOIN T4 AS D
    ON A.ID = D.PROVIDER_ID
   AND D.DIST_CNT1 = D.DIST_CNT
  LEFT JOIN T5 AS E
    ON A.ID = E.PROVIDER_ID
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.IS_DELETED = 0
   AND COALESCE(A.PROVINCE,'') NOT LIKE '%台湾%'
   AND COALESCE(A.PROVINCE,'') NOT LIKE '%澳门%'
