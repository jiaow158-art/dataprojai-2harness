insert overwrite table DWRDIM.DWR_DIM_CUST_ORD_GRP_T
SELECT 
 T.kunnr AS CUST_NUM 
,T0.cust_name AS CUST_NAME
,T.so_group AS ORD_GRP_CODE --订单组编码
,T.sg_describe AS ORD_GRP_NAME --订单组名称
,T.bukrs AS COMP_CODE 
,T1.BUTXT AS COMP_NAME
,FROM_UNIXTIME(UNIX_TIMESTAMP(T.create_date ,'yyyyMMdd'),'yyyy-MM-dd') AS CREATE_DATE
,FROM_UNIXTIME(UNIX_TIMESTAMP(T.change_date ,'yyyyMMdd'),'yyyy-MM-dd') AS UPDATE_DATE  
,T.undertaker AS UNDERTAKER --担保人姓名
,T.xblnr  AS XBLNR --参考凭证编号
,T.bktxt  AS BKTXT 
,T.sg_status  AS SG_STATUS
,T.sg_old_status AS SG_OLD_STATUS 
,T.clienttype  AS CUST_TYPE
,T.discount  AS DISCOUNT
,CASE WHEN T.sales IS NULL OR T.sales = '' THEN NULL ELSE 'DDZ' END AS EMP_CODE
,CASE WHEN T.sales IS NULL OR T.sales = '' THEN NULL ELSE T.sales END AS EMP_NAME
,CURRENT_TIMESTAMP() AS dw_last_update_date 
FROM SDI.SDI_ZGATHERING_SG_1001 T
LEFT JOIN DWRDIM.DWR_DIM_CUST_GENERAL_D T0 
ON T.kunnr = T0.CUST_NUM 
AND T0.end_DATE = '4712-12-31'
LEFT JOIN SDI.SDI_T001_1001 T1
ON T.bukrs = T1.BUKRS AND T1.MANDT = '800'
WHERE T.MANDT = '800'
