INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_OMS_CUST_INFO_F
SELECT 
       DI.DISTRIBUTOR_CODE AS CUSTOMER_CODE --经销商编码
      ,DI.DISTRIBUTOR_NAME AS CUSTOMER_NAME --经销商名称
      ,CI.CUSTOMER_CODE AS CUST_CODE --客户编码
      ,CI.CUSTOMER_NAME AS CUST_NAME  --客户名称
      ,CI.CUSTOMER_STATUS AS CUST_STATUS_CODE --客户状态编码
      ,B.MEANING AS CUST_STATUS_NAME --客户状态名称	  
      ,CI.ORGANIZATION_TYPE AS CUST_TYPE --客户类型编码
	  ,C.MEANING AS CUST_TYPE_NAME --客户类型名称	
      ,CI.CUSTOMER_CONTACTS AS CONTACT_NAME --网点联系人
      ,CI.CUSTOMER_CONTACTS_PHONE AS CONTACT_PHONE_NUM --网点联系电话
      ,D.EMPLOYEE_NUMBER AS SALE_EMP_NUM --订单员编码
	  ,CI.ORDERMAN_NAME AS SALE_EMP_NAME --订单员名称
      ,CI.WAREHOUSE_CODE --仓库编码
	  ,CI.WAREHOUSE_NAME --仓库名称
      ,CI.LOCATION_CODE AS INV_LOCATION_CODE --库存地点编码
	  ,CI.LOCATION_NAME AS INV_LOCATION_NAME --库存地点名称
      ,CI.PROVINCE AS PROVINCE_NAME --省 
      ,CI.CITY AS CITY_NAME --市
      ,CI.COUNTY AS COUNTY_NAME --县
      ,CI.ADDRESS --详细地址
      ,CASE WHEN CC.CONTACTS_TYPE = 'N' THEN '否'
 	        WHEN CC.CONTACTS_TYPE = 'Y' THEN '是'
			 END AS IS_XCX_MAIN --是否小程序主账号
      ,CC.CONTACTS AS CONTACT_NAME1 --姓名
      ,CC.CONTACTS_PHONE AS XCX_NUM --小程序账号
      ,CC.POSITION AS POSITION_NAME --职位
      ,CASE WHEN CCI.DEFAULT_FLAG = 'N' THEN '否'
            WHEN CCI.DEFAULT_FLAG = 'Y' THEN '是'
	         END AS IS_DEFAULT --是否默认
      ,CCI.CONSIGNEE AS CNSGN_NAME --收货人
      ,CCI.CONSIGNEE_PHONE AS CNSGN_PAHONE_NUM --收货人电话
      ,CCI.PROVINCE AS CNSGN_PROVINCE_NAME --收货省
      ,CCI.CITY AS CNSGN_CITY_NAME --收货市
      ,CCI.COUNTY AS CNSGN_COUNTY_NAME --收货区/县
      ,CCI.ADDRESS AS CNSGN_ADDRESS --详细地址
      ,CI.CREATION_DATE AS CREATE_DATE --制单日期
      ,A.FILE_COUNT --附件数量
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,CASE WHEN CI.HEAD_ENTERPRISE = 'Y' THEN '是'
            WHEN CI.HEAD_ENTERPRISE = 'N' THEN '否'
			 END AS IS_HEAD_ENTRP --是否头部装企
	  ,E.CUSTOMER_CODE AS HEAD_ENTRP_CODE --所属头部装企编码
      ,CI.HEAD_ENTERPRISE_NAME AS HEAD_ENTRP_NAME --所属头部装企名称
      ,CI.TO_EXAMINE AS EXAM_STATUS_CODE --审核编码
	  ,F.MEANING AS EXAM_STATUS_NAME --审核名称	  
  FROM SDI.SDI_DTCUST_CUSTOMER_INFO_1029 CI 
 INNER JOIN SDI.SDI_DT_DISTRIBUTOR_INFO_1029 DI 
    ON CI.DISTRIBUTOR_ID = DI.DISTRIBUTOR_ID
  LEFT JOIN  SDI.SDI_DTCUST_CUSTOMER_CONTACTS_1031 CC 
    ON CI.CUSTOMER_ID = CC.CUSTOMER_ID
  LEFT JOIN SDI.SDI_DTCUST_CUSTOMER_CONSIGNEE_1031 CCI 
    ON CI.CUSTOMER_ID = CCI.CUSTOMER_ID
  LEFT JOIN (SELECT BUSINESS_ID
                   ,COUNT(1) AS FILE_COUNT
               FROM SDI.SDI_BASE_FILE_1029 
              WHERE BUSINESS_KEY = 'CUSTOMERIDINFO'
              GROUP BY BUSINESS_ID
			 ) A
    ON A.BUSINESS_ID = CI.CUSTOMER_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B
    ON CI.CUSTOMER_STATUS = B.LOOKUP_CODE
   AND B.LOOKUP_TYPE = 'CUSTOMER_STATUS'
   AND B.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 C
    ON CI.ORGANIZATION_TYPE = C.LOOKUP_CODE
   AND C.LOOKUP_TYPE = 'CUSTOMER_STATUS'
   AND C.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_PERSON_1029 D
    ON D.PERSON_ID = CI.SALESMAN_ID
  LEFT JOIN SDI.SDI_DTCUST_CUSTOMER_INFO_1029 E
    ON E.CUSTOMER_ID = CI.HEAD_ENTERPRISE_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 F
    ON CI.TO_EXAMINE = F.LOOKUP_CODE
   AND F.LOOKUP_TYPE = 'DB_CUSTOMER_REVIEWED'
   AND F.ENABLED_FLAG = 'Y'	
WHERE 1 = 1
