INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_CUST_CREDIT_MNGMT_D
SELECT 
   CT.CUST_ID,
   CT.SRC_CUST_NUM,
   CT.GROUP_NUM,
   T.CONFIG_NAME  AS CALC_SCORE_CREDIT_L_RULE_NAME,
   CT.SCORE_CHG_DATE,
   CT.RISK_CLASS_CHG_DATE,
   CT.EFFEC_END_DATE,
   T1.CONFIG_NAME AS RISK_CHASS_NAME,
   CT.SCORE,
   CT.CUST_CREDIT_GROUP,
   T2.CONFIG_NAME AS CREDIT_CHECK_RULES_NAME,
   CT.DEL_FLAG,
   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
   CT.START_DATE,
   CT.END_DATE,
   CT.CUST_NUM,
   CT.CUST_CHECK_RULE,
   CT.CUST_LIMIT_RULE,
   CT.CUST_RISK_CLASS
FROM DWIMD.DWI_MD_CUST_CREDIT_MNGMT_T CT
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
     ON CT.GROUP_NUM = T.GROUP_NUM
	 AND CT.CALC_SCORE_CREDIT_L_RULE_CODE = T.CONFIG_CODE
     AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1 
     ON CT.GROUP_NUM = T1.GROUP_NUM
	 AND CT.RISK_CHASS_CODE = T1.CONFIG_CODE
     AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2
     ON CT.GROUP_NUM = T2.GROUP_NUM
	 AND CT.CREDIT_CHECK_RULES = T2.CONFIG_CODE
     AND T2.LANG_CODE = '1'
