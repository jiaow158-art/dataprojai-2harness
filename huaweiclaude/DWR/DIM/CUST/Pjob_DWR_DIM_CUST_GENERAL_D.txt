FROM
  DWIMD.DWI_MD_CUST_GENERAL_T GT
  LEFT JOIN(
    SELECT
      ADRP.EMP_NAME,
      USR21.CTR_OBJECT_PERSON_NAME
    FROM
      DWIMD.DWI_MD_PERSON_OFFI_ADDR_T ADRP
      INNER JOIN DWIMD.DWI_MD_USER_NAME_T USR21 ON ADRP.EMP_NUM = USR21.EMP_NUM
  ) D ON GT.CTR_OBJECT_PERSON_CODE = D.CTR_OBJECT_PERSON_NAME
  LEFT JOIN DWIMD.DWI_MD_ADDRESS_SERVICE_T TS 
  ON GT.ADDRESS_CODE = TS.ADDRESS_CODE
  AND GT.GROUP_NUM = TS.GROUP_NUM
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T ON GT.CUST_CLASS_CODE = T.CONFIG_CODE
  AND GT.GROUP_NUM = T.GROUP_NUM
  AND T.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1 ON GT.CUST_ACCT_GROUP_CODE = T1.CONFIG_CODE
  AND GT.GROUP_NUM = T1.GROUP_NUM
  AND T1.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2 ON GT.ATTRI_CODE_5 = T2.CONFIG_CODE
  AND GT.GROUP_NUM = T2.GROUP_NUM
  AND T2.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T3 ON GT.ATTRI_CODE_6 = T3.CONFIG_CODE
  AND GT.GROUP_NUM = T3.GROUP_NUM
  AND T3.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4 ON GT.ATTRI_CODE_7 = T4.CONFIG_CODE
  AND GT.GROUP_NUM = T4.GROUP_NUM
  AND T4.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T5 ON GT.CUST_CRITERIA_GROUP1_CODE = T5.CONFIG_CODE
  AND GT.GROUP_NUM = T5.GROUP_NUM
  AND T5.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T6 ON GT.CUST_ORDER_BLOCK_CODE = T6.CONFIG_CODE
  AND GT.GROUP_NUM = T6.GROUP_NUM
  AND T6.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T7 ON GT.ANNUAL_SALES_LEVEL_CODE = T7.CONFIG_CODE
  AND GT.GROUP_NUM = T7.GROUP_NUM
  AND T7.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T8 ON GT.CUST_MKT_SEGMENTATION_CODE = T8.CONFIG_CODE
  AND GT.GROUP_NUM = T8.GROUP_NUM
  AND T8.LANG_CODE = '1'
  left join DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T9 ON GT.ATTRI_CODE_2 = T9.CONFIG_CODE
  AND GT.GROUP_NUM = T9.GROUP_NUM
  AND T9.LANG_CODE = '1'
  LEFT JOIN SDI.SDI_T005T_1001 T10
  ON GT.CNTRY_REGION_CODE = T10.LAND1 AND T10.SPRAS = '1' AND T10.MANDT = '800';
