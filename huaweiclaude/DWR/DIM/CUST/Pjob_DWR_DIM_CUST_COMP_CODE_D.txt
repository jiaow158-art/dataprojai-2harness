INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_CUST_COMP_CODE_D
SELECT 
   ST.CUST_ID AS CUST_ID,
   ST.CUST_NUM AS CUST_NUM ,
   ST.COMPANY_CODE AS COMPANY_CODE ,
   ST.GROUP_NUM AS GROUP_NUM ,
   ST.MAJOR_FIN_ACT AS MAJOR_FIN_ACT ,
   ST.REC_CRT_DATE AS REC_CRT_DATE ,
   D.EMP_NAME AS CTR_OBJECT_PERSON_NAME ,
   ST.INSURANCE_DATE AS INSURANCE_DATE ,
    T.CONFIG_NAME AS PERMISSION_GROUP_NAME ,
   ST.FROZEN_COMP_CODE_POST AS FROZEN_COMP_CODE_POST ,
   ST.ASSIGN_NUM_SEQ_CODE AS ASSIGN_NUM_SEQ_CODE ,
   ST.CHG_CONFIR_DATE AS CHG_CONFIR_DATE ,
   ST.PERSON_NUM AS PERSON_NUM ,
   ST.INSURANCE_TERM AS INSURANCE_TERM ,
   ST.LAST_CHG_CONFIR_DATE AS LAST_CHG_CONFIR_DATE ,
   S.GENERAL_LEDGER_CTR_NAME AS GENERAL_LEDGER_CTR_ACCT_NAME,
   T1.CONFIG_NAME  AS PLAN_GROUP_NAME,
   ST.INSUE_AMT AS INSUE_AMT ,
   ST.PAY_TERM_CODE AS PAY_TERM_CODE ,
   ST.CASH_CHECK_TIME AS CASH_CHECK_TIME ,
   ST.DEDUCTION_PCT AS DEDUCTION_PCT ,
   ST.DRAFT_LIMIT_AMT AS DRAFT_LIMIT_AMT ,
   ST.MTH_INTEREST_CALC_FRQCY AS MTH_INTEREST_CALC_FRQCY ,
   ST.LAST_INTEREST_CALC_R_DATE AS LAST_INTEREST_CALC_R_DATE ,
   ST.MASTER_REC_DEL_FROZEN AS MASTER_REC_DEL_FROZEN ,
   ST.LAST_INTEREST_CALC_K_DATE AS LAST_INTEREST_CALC_K_DATE ,
   ST.COMPANY_ID AS COMPANY_ID ,
   ST.COMPANY_NAME AS COMPANY_NAME,
   ST.DEL_FLAG AS DEL_FLAG,
   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
   ST.START_DATE,
   ST.END_DATE,
   ST.SRC_COMPANY_CODE,
   ST.SRC_CUST_NUM
   ,ST.PLAN_GROUP_CODE      -- 现金组       20230904新增
   ,ST.PLAN_GROUP_DESC      -- 现金组描述   20230904新增
FROM DWIMD.DWI_MD_CUST_COMP_CODE_T	ST
LEFT JOIN(SELECT ADRP.EMP_NAME,USR21.CTR_OBJECT_PERSON_NAME 
			FROM DWIMD.DWI_MD_PERSON_OFFI_ADDR_T ADRP
	 INNER JOIN DWIMD.DWI_MD_USER_NAME_T USR21
            ON ADRP.EMP_NUM = USR21.EMP_NUM) D
	 ON ST.CTR_OBJECT_PERSON_CODE = D.CTR_OBJECT_PERSON_NAME 	
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
     ON ST.GROUP_NUM = T.GROUP_NUM
	 AND ST.PERMISSION_GROUP_CODE = T.CONFIG_CODE
     AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1 
     ON ST.GROUP_NUM = T1.GROUP_NUM
	 AND ST.PLAN_GROUP = T1.CONFIG_CODE
     AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_LEDGER_TEXT_T S	
      ON  ST.GROUP_NUM = S.GROUP_NUM
     AND ST.GENERAL_LEDGER_CTR_ACCT_CODE = S.GENERAL_LEDGER_CTR_ACCT 
     AND S.LANGUAGE_CODE = '1';
