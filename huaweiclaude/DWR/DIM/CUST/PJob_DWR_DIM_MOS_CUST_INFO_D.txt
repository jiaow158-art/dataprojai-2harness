WITH T AS(
SELECT CLIENT_INFO_ID
      ,CONCAT_WS(',', COLLECT_SET(CAST(PHONE AS STRING))) AS PHONE_NUM
  FROM SDI.SDI_SAL_CLIENT_INFO_PHONE_1012
 WHERE DELETE_FLAG = 0
 GROUP BY CLIENT_INFO_ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MOS_CUST_INFO_D
SELECT  A.ID AS CUST_ID --ID
       ,A.CODE AS CUST_NUM --客户编号
	   ,B.PHONE_NUM --客户电话
       ,A.CREATER_ID AS CREATE_EMP_ID --创建人ID
       ,A.CREATER_NAME AS CREATE_EMP_NAME --创建人名称
       ,A.CREATE_TIME --创建时间
       ,A.MODIFIER_ID AS UPDATE_EMP_ID --修改人ID
       ,A.MODIFIER_NAME AS UPDATE_EMP_NAME --修改人名称
       ,A.MODIFY_TIME AS UPDATE_TIME --修改时间
       ,A.VERSION AS VERS_NUM --数据版本号
       ,A.TENANT_ID AS TENAT_ID --租户ID
       ,A.FIRST_FLOW_SOURCE AS FST_SRC_ID --客户首次来源编码
       ,A.FIRST_STORE_ID AS FST_STORE_ID --首次门店ID
       ,A.USABLE_FLAG AS IS_USAB_CODE --是否有效编码
       ,A.SYNC_DMT_CLIENT_INFO_ID AS SRC_CUST_ID --从哪个DMT客户同步过来
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SAL_CLIENT_INFO_1012 A
  LEFT JOIN T AS B
    ON A.ID = B.CLIENT_INFO_ID
 WHERE A.DELETE_FLAG = 0
