set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_SAP_EMP_INFO_F PARTITION(STAT_DATE)
SELECT
       SUBSTR(A.UPDAT,1,4) || '-' || SUBSTR(A.UPDAT,5,2) AS STAT_MONTH --统计月份
      ,A.PERNR AS EMP_NUM --人员编号
      ,A.NACHN AS EMP_NAME --姓名
	  ,'东鹏控股' AS LVL1_DEPT_NAME --一级部门
      ,A.ZL0 AS LVL2_DEPT_NAME --二级部门
      ,A.ZL1 AS LVL3_DEPT_NAME --三级部门
      ,A.ZL2 AS LVL4_DEPT_NAME --四级部门
      ,A.ZL3 AS LVL5_DEPT_NAME --五级部门
      ,A.ZL4 AS LVL6_DEPT_NAME --六级部门
      ,A.ZL5 AS LVL7_DEPT_NAME --七级部门
      ,A.ZL6 AS LVL8_DEPT_NAME --八级部门
      ,CASE WHEN SUBSTR(A.UPDAT,1,6) <= '202305' THEN C.MPO_CAT_TYPE
	        ELSE A.ZZ_ZWLBWB
			 END AS EMP_POST_TYPE_NAME --职族
      ,A.PLANS AS EMP_POST_ID --岗位ID	  
      ,A.PLANS_TEXT AS EMP_POST_NAME --岗位
	  ,A.STELL AS EMP_POST_SUB_ID --职务ID
      ,A.STELL_TEXT AS EMP_POST_SUB_NAME --职务
      ,A.ZZ_ZJ AS EMP_POST_LVL_NAME --职级
      ,A.ZZ_ZWJBWB AS EMP_LVL_NAME --岗位层级
      ,A.GESCH AS GENDER --性别
      ,REPLACE(A.GBDAT_TEXT,'.','-') AS BIRTH_DATE --出生日期
      ,A.NL AS AGE --年龄
      ,A.STAT1_TEXT AS EMP_STATUS_NAME --员工状态
      ,REPLACE(A.RZRQ,'.','-') AS ENTRY_DATE --入职日期
      ,REPLACE(A.ZZRQ,'.','-') AS TFR_INTO_DATE	--转正日期
      ,A.SN_TEXT AS DIV_AGE --司龄
      ,A.GN_TEXT AS WORK_AGE --工龄
      ,A.SN_SECTION AS DIV_SECTN --司龄段
      ,A.NN_SECTION AS AGE_SECTN --年龄段
      ,A.NAME1 AS PERS_AREA_NAME --人事范围
      ,A.BTEXT AS PERS_SUB_AREA_NAME --人事子范围
      ,A.PTEXT AS EMP_GROUP_NAME --员工组
      ,A.PKTXT AS EMP_SUB_GROUP_NAME --员工子组
      ,A.ATEXT AS WAGES_BELONG_NAME --工资核算范围
      ,A.ZZ_RACE_TEXT AS NATION_NAME --民族
      ,A.ZZ_ZZMM_TEXT AS POLIT_NAME --政治面貌
      ,A.FAMST_TEXT AS MRTL_STATUS_NAME	--婚姻状况
      ,A.ZZ_SYZK_TEXT AS BIRTH_STATUS_NAME --生育状况
      ,A.GBORT AS BIRTH_PLACE_NAME --籍贯
      ,A.ZZ_HKXZ_TEXT AS RSDN_TYPE_NAME --户口性质
      ,A.STRAS_JT AS HOME_ADDRESS --家庭地址
      ,A.STRAS_XJ AS CUR_HOME_ADDRESS --现居住地地址
      ,A.ZZ_HKDZ AS IC_ADDRESS --身份证地址
      ,REPLACE(A.BYRQ,'.','-') AS GRAD_DATE --毕业时间
      ,A.INSTI AS GRAD_SCHL_NAME --毕业院校
      ,A.ZZ_SXZY AS PRFSN_NAME --所学专业
      ,A.SLART_TEXT AS ED_NAME --学历
      ,A.USRID_SJ AS PHONE_NUM --手机
      ,A.USRID_JT AS TEL_PHONE_NUM --家庭电话
      ,A.ZZ_XM AS URGENT_CNTCT_NAME --紧急联系人
      ,A.USRID AS URGENT_PHONE_NUM --紧急联系电话
      ,A.PERNR_ZJSJ AS SUPER_EMP_NUM --直接上级工号
      ,A.NACHN_ZJSJ	AS SUPER_EMP_NAME --直接上级姓名
      ,A.PLANS_ZJSJ	AS SUPER_EMP_POST_NAME --直接上级岗位
      ,A.ZZ_GWSX_TEXT AS EMP_POST_ATTR_NAME --岗位属性
      ,CASE WHEN A.ZGJGW = '' THEN '否'
   	        ELSE A.ZGJGW
			 END AS IS_KEY_POST --是否关键岗位
      ,A.ZZ_HRBP_REN AS HRBP_NAME --HRBP
      ,CASE WHEN A.ZZ_XRB = '' THEN '否'
	        ELSE A.ZZ_XRB
			 END AS IS_XRB --是否新锐班
      ,A.ZZ_XRBJS AS XRB_NUM_NAME --新锐班届数     
      ,A.BUKRS AS COMP_CODE --公司代码
      ,A.ZZ_HTJF AS CNTC_PARTY_A_CODE --合同甲方
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN A.ZZ_SFBZB = 'X' THEN '否'
            ELSE '是'
			 END AS IS_ZB --是否占编
	  ,A.ZKSTTEXT AS FEE_TYPE_NAME --费用分类		 
	  ,SUBSTR(A.UPDAT,1,4) || '-' || SUBSTR(A.UPDAT,5,2) || '-' || SUBSTR(A.UPDAT,7,2) AS STAT_DATE --统计日期
  FROM SDI.SDI_ZHRTPA004_1001 A
 INNER JOIN (SELECT DISTINCT DAY_ID_MEND
               FROM DM.DM_DIM_DATE_D
              WHERE DAY_ID >= '20220101'
			) B
	ON B.DAY_ID_MEND = A.UPDAT
  LEFT JOIN SDI.SDI_EHR_EMP_MPO_1037 C
    ON SUBSTR(A.UPDAT,1,4) = C.STAT_YEAR
   AND A.PERNR = C.EMP_NUM
   AND SUBSTR(A.UPDAT,1,6) <= '202305'   --2023年5月及以前的职族用填报的
 WHERE A.MANDT = 800
;
