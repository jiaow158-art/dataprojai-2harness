with tmp1 as (
	SELECT
		 t1.PSPHI
		,T1.POSNR level_1_posnr
		,COALESCE(T2.POSNR,T1.POSNR)  level_2_posnr
		,COALESCE(T3.POSNR,T2.POSNR,T1.POSNR)  level_3_posnr
		,COALESCE(T3.POSNR,T2.POSNR,T1.POSNR) key
	FROM (
		SELECT t1.PSPHI,
			   t1.UP ,
			   t1.POSNR ,
			   t1.DOWN
		FROM sdi.sdi_prhi_1001  T1
		WHERE 1=1
	    AND T1.UP ='00000000'
	) t1 LEFT JOIN sdi.sdi_prhi_1001 t2
	 ON t1.POSNR = t2.UP
	 AND t1.PSPHI = t2.PSPHI
	 LEFT JOIN sdi.sdi_prhi_1001 t3
	 ON t2.POSNR = t3.UP
	 AND t2.PSPHI = t3.PSPHI
	 
 union all
  
SELECT
		 t1.PSPHI
		,T1.POSNR level_1_posnr
		,COALESCE(T2.POSNR,T1.POSNR)  level_2_posnr
		,COALESCE(T2.POSNR,T1.POSNR)  level_3_posnr
		,COALESCE(T2.POSNR,T1.POSNR) key
	FROM (
		SELECT t1.PSPHI,
			   t1.UP ,
			   t1.POSNR ,
			   t1.DOWN
		FROM sdi.sdi_prhi_1001  T1
		WHERE 1=1
	    AND T1.UP ='00000000'
	) t1 LEFT JOIN sdi.sdi_prhi_1001 t2
	 ON t1.POSNR = t2.UP
	 AND t1.PSPHI = t2.PSPHI

 union all
  
SELECT
		 t1.PSPHI
		,T1.POSNR level_1_posnr
		,T1.POSNR  level_2_posnr
		,T1.POSNR  level_3_posnr
		,T1.POSNR key
	FROM (
		SELECT t1.PSPHI,
			   t1.UP ,
			   t1.POSNR ,
			   t1.DOWN
		FROM sdi.sdi_prhi_1001  T1
		WHERE 1=1
	    AND T1.UP ='00000000'
	) t1 
),tmp as(
select distinct PSPHI
		,T1.level_1_posnr
		,T1.level_2_posnr
		,T1.level_3_posnr
		,T1.key
  from tmp1 as t1
)
,system_station as (

     SELECT
 objnr
,max(case when stat ='I0001' THEN B.txt04 END)  system_station_creator
,max(case when stat ='I0001' THEN B.TXT30 END)  system_station_desc_creator
,max(case when stat ='I0002' THEN B.txt04 END)  system_station_release
,max(case when stat ='I0002' THEN B.TXT30 END ) system_station_desc_release
,max(case when stat ='I0028' THEN B.txt04 END ) system_station_maintain
,max(case when stat ='I0028' THEN B.TXT30 END ) system_station_desc_maintain
,max(case when stat ='I0042' THEN B.txt04 END ) system_station_partial
,max(case when stat ='I0042' THEN B.TXT30 END ) system_station_Desc_partial
,max(case when stat ='I0065' THEN B.txt04 END ) system_station_lock
,max(case when stat ='I0065' THEN B.TXT30 END ) system_station_desc_lock
FROM  SDI.SDI_JEST_1001  A
JOIN SDI.SDI_TJ02T_1001 B ON A.STAT  = B.ISTAT AND B.SPRAS ='1'
where A.inact ='' group by objnr
)
