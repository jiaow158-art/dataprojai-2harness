--DWI_SO_USER_OSS_T 到 DWR_SO_USER_OSS_F
--OSS人员表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_USER_OSS_F
SELECT
  DWI.PAR_ROW_ID AS PAR_ROW_ID,
  DWI.ROW_ID AS ROW_ID,
  DWI.LOGIN AS LOGIN,
  DWI.WORKCODE AS WORKCODE,
  DWI.LAST_UPDATE_DATE AS LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE AS DB_LAST_UPDATE_DATE,
  DWI.ADDR_TEXT AS ADDR_TEXT,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.USER_PASSWORD AS USER_PASSWORD,
  DWI.LAST_LOGIN_TS AS LAST_LOGIN_TS,
  DWI.LAST_UPDATE_BY AS LAST_UPDATE_BY,
  DWI.MODIFY_QTY AS MODIFY_QTY,
  DWI.USER_STATUS AS USER_STATUS,
  DWI.DB_LAST_UPDATE_DATE_SRC AS DB_LAST_UPDATE_DATE_SRC,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.INTEG_USER_FLAG AS INTEG_USER_FLAG,
  DWI.USER_FLAG AS USER_FLAG,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.CHALLENGE_ANSWER AS CHALLENGE_ANSWER,
  DWI.TRNG_DL_ADMIN_FLAG AS TRNG_DL_ADMIN_FLAG,
  DWI.CHALLENGE_QUESTION AS CHALLENGE_QUESTION,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_USER_OSS_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
