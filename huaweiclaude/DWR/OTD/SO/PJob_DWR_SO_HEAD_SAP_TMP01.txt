--SAP销售订单头临时表(关联通用配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_HEAD_SAP_TMP01
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  RCD1.CONFIG_NAME AS SAL_VCH_TYPE_NAME,
  RCD2.CONFIG_NAME AS ORDER_REASON_NAME,
  RCD3.CONFIG_NAME AS SAL_ORG_ID_NAME,
  RCD4.CONFIG_NAME AS DIS_CHANNEL_NAME,
  RCD5.CONFIG_NAME AS SAL_GROUP_NAME,
  RCD6.CONFIG_NAME AS SAL_OFFICE_NAME,
  RCD7.CONFIG_NAME AS ADVICE_INVOICE_TYPE_NAME,
  RCD8.CONFIG_NAME AS CUST_GROUP1_NAME,
  RCD9.CONFIG_NAME AS CUST_GROUP2_NAME,
  RCD10.CONFIG_NAME AS CUST_GROUP3_NAME,
  RCD11.CONFIG_NAME AS BUSINESS_PLATE_NAME,
  RCD12.CONFIG_NAME AS CREDIT_CONTROL_SCOPE_NAME
FROM
  DWIOTD.DWI_SO_HEAD_SAP_T DWI --SAP销售订单头表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD1 --销售凭证类型中文名称
  ON DWI.GROUP_NUM = RCD1.GROUP_NUM
  AND DWI.SAL_VCH_TYPE = RCD1.CONFIG_CODE
  AND RCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD2 --订单原因（业务交易原因）中文名称
  ON DWI.GROUP_NUM = RCD2.GROUP_NUM
  AND DWI.ORDER_REASON = RCD2.CONFIG_CODE
  AND RCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD3 --销售组织中文名称
  ON DWI.GROUP_NUM = RCD3.GROUP_NUM
  AND DWI.SAL_ORG_ID = RCD3.CONFIG_CODE
  AND RCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD4 --分销渠道中文名称
  ON DWI.GROUP_NUM = RCD4.GROUP_NUM
  AND DWI.DIS_CHANNEL = RCD4.CONFIG_CODE
  AND RCD4.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD5 --销售组中文名称
  ON DWI.GROUP_NUM = RCD5.GROUP_NUM
  AND DWI.SAL_GROUP = RCD5.CONFIG_CODE
  AND RCD5.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD6 --销售办事处中文名称
  ON DWI.GROUP_NUM = RCD6.GROUP_NUM
  AND DWI.SAL_OFFICE = RCD6.CONFIG_CODE
  AND RCD6.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD7 --与订单有关的出具发票凭证的建议出具发票类型中文名称
  ON DWI.GROUP_NUM = RCD7.GROUP_NUM
  AND DWI.ADVICE_INVOICE_TYPE = RCD7.CONFIG_CODE
  AND RCD7.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD8 --客户组 1中文名称
  ON DWI.GROUP_NUM = RCD8.GROUP_NUM
  AND DWI.CUST_GROUP1 = RCD8.CONFIG_CODE
  AND RCD8.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD9 --客户组 2中文名称
  ON DWI.GROUP_NUM = RCD9.GROUP_NUM
  AND DWI.CUST_GROUP2 = RCD9.CONFIG_CODE
  AND RCD9.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD10 --客户组 3中文名称
  ON DWI.GROUP_NUM = RCD10.GROUP_NUM
  AND DWI.CUST_GROUP3 = RCD10.CONFIG_CODE
  AND RCD10.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD11 --业务版块中文名称
  ON DWI.GROUP_NUM = RCD11.GROUP_NUM
  AND DWI.BUSINESS_PLATE = RCD11.CONFIG_CODE
  AND RCD11.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD12 --信用控制范围中文名称
  ON DWI.GROUP_NUM = RCD12.GROUP_NUM
  AND DWI.CREDIT_CONTROL_SCOPE = RCD12.CONFIG_CODE
  AND RCD12.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
