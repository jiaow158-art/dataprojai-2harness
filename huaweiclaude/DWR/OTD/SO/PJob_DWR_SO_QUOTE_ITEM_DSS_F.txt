--DSS报价单行
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_QUOTE_ITEM_DSS_F
SELECT
  DWI.ID AS ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.OPERATE_LOG AS OPERATE_LOG,
  DWI.QUOTATION_ID AS QUOTATION_ID,
  DWI.DISCOUNT AS DISCOUNT,
  DWI.APPROVAL_DISCOUNT AS APPROVAL_DISCOUNT,
  DWI.APPLY_TOTAL_AMT AS APPLY_TOTAL_AMT,
  DWI.APPROVAL_TOTAL_AMT AS APPROVAL_TOTAL_AMT,
  DWI.APPROVAL_PRICE AS APPROVAL_PRICE,
  DWI.APPLY_PRICE AS APPLY_PRICE,
  DWI.PRICE AS PRICE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  DWI.SELLING_PRICE AS SELLING_PRICE,
  DWI.PRODUCT_NAME AS PRODUCT_NAME,
  DWI.DIRECT_SALES_COST AS DIRECT_SALES_COST,
  DWI.QUANTITY AS QUANTITY,
  DWI.TRANSFORM_QTY AS TRANSFORM_QTY,
  DWI.APPROVAL_QUANTITY AS APPROVAL_QUANTITY,
  DWI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  DWI.BATCH_CODE AS BATCH_CODE,
  DWI.SALES_AREA AS SALES_AREA,
  DWI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  DWI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  DWI.LEVEL AS LEVEL,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.WAREHOUSE_AGE AS WAREHOUSE_AGE,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.DESIGNER_FD_POINTS AS DESIGNER_FD_POINTS,
  DWI.ACREAGE AS ACREAGE,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.LAST_PRICE AS LAST_PRICE,
  DWI.UNIT AS UNIT,
  DWI.QCYY AS QCYY,
  DWI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  DWI.PROCESSING_UNIT AS PROCESSING_UNIT,
  DWI.STATUS AS STATUS,
  DWI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  DWI.COMMISSION_POINTS AS COMMISSION_POINTS,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_QUOTE_ITEM_DSS_T DWI --DSS报价单行
  --关联工厂主数据表取工厂名称
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800' 
  --关联物料主数据表取物料名称
  INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
