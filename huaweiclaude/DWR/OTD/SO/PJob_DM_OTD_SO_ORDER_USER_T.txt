insert overwrite table dm.dm_otd_so_order_user_t
select
 t1.order_num					order_num   	-- 订单
,t1.plan_type_name				order_type  	-- 订单类型
,t1.sap_order_num				sap_order_num 	-- SAP订单
,t1.sap_order_line_num 		    sap_order_item  -- SAP行号
,t1.material_code				material_num 	-- 物料编码
,t1.material_name 				material_name	-- 物料名称
,t1.spld_area_name				SPLD_AREA_NAME  -- 供货产区
,t1.factory_code 				factory_code	-- 工厂编码
,t1.WAREHOUSE_CODE 				WAREHOUSE_CODE	-- 库存地点
,t1.qty 						qty				-- 数量
,t3.batch_num 						batch 			-- 批次
,t3.rmn_qty 					CONFIRM_QTY		-- 已确定数量
,case when COALESCE(t1.denial_reason_name,'') = '取消留货' then null
     when t1.qty - t3.rmn_qty  > 0 then '待留货'
	 when t1.qty  - t3.rmn_qty  = 0 then '已留货'  end  RMN_ST_NAME  -- 留货状态
,case when t3.rmn_qty <> 0 then t3.rmn_date   end    	RMN_DATE			-- 留货时间
-- ,sum( t3.rmn_qty ) over(partition by t3.sales_num,t3.batch_num  )
,case when t3.rmn_qty  =  0 then null
       when t3.rmn_qty - sum(nvl(t4.sales_qty,0)) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num  )   > 0 then round(datediff(current_date,t3.rmn_date),2)
       when t3.rmn_qty - sum(nvl(t4.sales_qty,0))  over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num  )   =  0 then null
       end RMN_DAYS	-- 留货天数
-- ,round(datediff(current_date,t3.rmn_date),2)  	RMN_DAYS	-- 留货天数
,t13.yqts 							OVER_DAYS			-- 逾期天数
,case when t13.zoverdue_mark = 'Y' THEN '已逾期'
      when t13.zoverdue_mark = 'N' THEN '未逾期' END 	RMN_WARN			-- 留货预警
,t1.order_line_st_name 			ORDER_ITEM_STATUS	-- 订单行状态
,t2.order_row_status_desc		REVIEW_ST			-- 评审状态
,t2.req_delivery_date			DEMAND_DELIVERY_DATE	-- 需求交货日期
,t14.PLAN_DELV_TIME 			PLAN_DELIVERY_DATE	-- 计划交货日期
,t1.sale_org_code 				SALE_ORG			-- 销售组织
,t1.sale_chnl_code 				SALE_CHANNEL_CODE		-- 分销渠道
,t1.cust_code 					CUST_NUM				-- 客户编码
,NULL 				        PROJECT_CODE				-- 工程报备号
,NULL 						PROJECT_NAME				-- 工程部报备名称
,t4.delv_num                DELV_NUM								-- 	提货单号
,t4.sales_qty               INV_PREPARATION_QTY					-- 	备货数量
--备货天数：如果备货数量=0则备货天数为空，如果备货数量-出库数>0则当前时间-提货单创建时间，如果备货数量-库存数=0则出库时间-提货单创建时间
,case when COALESCE(t4.sales_qty,0) = 0 then null
      when t4.sales_qty - nvl(t11.OUT_STOCK_QTY,0) > 0 then round(datediff(current_date,t15.create_date),2)
      when t4.sales_qty - nvl(t11.OUT_STOCK_QTY,0) = 0 then round(datediff(t11.voucher_post_date,t15.create_date),2)   end  INV_PREPARATION_DAYS					-- 	备货天数
-- ,round(datediff(current_date,t3.rmn_date),2)      INV_PREPARATION_DAYS					-- 	备货天数
,case when  t3.rmn_qty = 0 then '待申请提货'
      when t3.rmn_qty -  sum(nvl(t4.sales_qty,0) ) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num )   > 0 then '待申请提货'
      when t3.rmn_qty - sum(nvl(t4.sales_qty,0)) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num )   = 0 then '已申请提货' end     DELIVERY_APPLY_ST							-- 	提货申请状态
,t5.reservation_date      RESV_DATE						-- 	预约时间
,t5.sign_time     SIGN_TIME								-- 	司机到场时间
,t5.car_loading_status      LOAD_ST_NAME				-- 	装车状态
,t5.load_end_time     LEAVE_TIME							-- 	离场时间
,t11.OUT_STOCK_QTY       ACTL_DELV_QTY					-- 	已出库数量
,sum(t11.OUT_STOCK_QTY) over(partition by t3.sales_num,t3.batch_num ,t1.sap_order_line_num )       ACTL_DELV_QTY_TOTAL			-- 	出库数量汇总
,case when sum(nvl(t4.sales_qty,0) ) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num )
         - sum(t11.OUT_STOCK_QTY) over(partition by t3.sales_num,t3.batch_num ,t1.sap_order_line_num ) > 0 then '待出库'
      when sum(nvl(t4.sales_qty,0) ) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num )
	     - sum(t11.OUT_STOCK_QTY) over(partition by t3.sales_num,t3.batch_num ,t1.sap_order_line_num )  = 0 then '已出库' end  ACTL_DELV_ST									-- 	出库状态
,t6.loading_method_name      ACTL_DELV_TYPE					-- 	提货方式
,t6.dspt_num      DSPT_NUM								-- 	发运单号
,t6.drvr_name      DRVR_NAME			-- 	提货司机
,t6.drvr_phone_num      DRVR_PHONE_NUM		-- 	司机电话
,t6.drvr_card_num      DRVR_CARD_NUM		-- 	证件号码
,t6.drvr_car_num        DRVR_CAR_NUM			-- 	车牌号
,t6.delv_prov_name      DELV_PROV_NAME		-- 	收货地址省
,t6.delv_city_name      DELV_CITY_NAME		-- 	市
,t6.delv_dist_name      DELV_DIST_NAME		-- 	区
,t6.delv_addr       DELV_ADDR			-- 	详细地址
,t6.sign_time      DELV_RECEIPT_TIME	-- 	物流签收时间
,t12.sales_group_code      SAELS_GRP			-- 	线组编码
,substr(t1.create_date,1,10)      CREATE_DATE			-- 	订单创建日期
,current_timestamp()      			DW_LAST_UPDATE_DATE	-- 	最后更新时间
,sum(t4.sales_qty) over(PARTITION by t3.sales_num,t3.batch_num ,t1.sap_order_line_num) RMN_TOT_QTY  -- 备货汇总
,t6.delv_cnty_name      DELV_CNTY_NAME      -- 县
,t1.denial_reason_name      denial_reason  -- 拒绝原因
,t3.rmn_qty - sum(nvl(t4.sales_qty,0)) over(PARTITION by t3.sales_num,t3.batch_num,t1.sap_order_line_num ) not_deliver_qty -- 未提货申请数
,t4.plan_item_num as proj_plan_num --计划项目号
,t1.md_sale_emp_num string --销售雇员工号（客户主档）
,t14.BEF_PLAN_DELV_TIME --上一次计划交付时间
from DWROTD.DWR_OTD_SO_SALE_OFR_ORDER_F t1 -- 销售订单
 left join DWROTD.DWR_OTD_PLAN_ORDER_APPROVAL_F t2 on t1.order_num = t2.order_number and t1.order_line_num=t2.order_item_id and
                                                      t1.sap_order_num = t2.sap_number and t1.sap_order_line_num = t2.sap_order_line_num and
                                                      t2.order_row_status in(0,1,2)
 left join (select  sales_num,sales_item,batch_num,sales_qty,SUM(rmn_qty) rmn_qty ,MAX(rmn_date) rmn_date
from DWROTD.DWR_OTD_SO_REMAIN_GOODS_T GROUP BY  sales_num,sales_item,batch_num,sales_qty) t3
on t1.sap_order_num = t3.sales_num and t1.sap_order_line_num = t3.sales_item  -- 留货单
left join DWROTD.DWR_OTD_LOADING_PLAN_T t4 on t1.sap_order_num = t4.sales_num  and t1.sap_order_line_num = t4.sales_item_num
and  t4.delv_item_num<> '000000'
 left join DWROTD.DWR_TM_RESERVATION_RECORD_DETAIL_F t5 on t4.delv_num =t5.delivery_order_number
 and COALESCE(t5.reservation_state,'') not in('CANCELED','EXIT')
 left join DWROTD.DWR_TRA_TMS_SHIP_DTL_F t6 on t4.delv_num = t6.delv_num  and t4.delv_item_num  = t6.delv_line_num
 left join (select do_num from DWROTD.DWR_TRA_RCPT_ORDER_F group by do_num) t8  on t6.do_num = t8.do_num
 left join (SELECT A.DELIVERY_IM AS DELV_NUM
      ,A.DELIVERY_PROJECT AS DELV_LINE_NUM
      ,substr(voucher_post_date,1,10)  voucher_post_date
      ,SUM(CASE WHEN DR_CR_ID = 'H_SHKZG' THEN QUANTITY ELSE 0 END) AS OUT_STOCK_QTY --出库数量
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
 GROUP BY A.DELIVERY_IM
      ,A.DELIVERY_PROJECT,substr(voucher_post_date,1,10)  ) t11 on t6.delv_num = t11.DELV_NUM and t6.delv_line_num = t11.DELV_LINE_NUM
 join (select sales_org_code,cust_num,distribution_chnl_code,sales_group_code from  DWRDIM.DWR_DIM_CUST_SALES_AREA_D where end_date='4712-12-31'
group by sales_org_code,cust_num,distribution_chnl_code,sales_group_code) t12 
on t1.sale_org_code   = t12.sales_org_code   and t1.cust_code =t12.cust_num
 and t1.sale_chnl_code  = t12.distribution_chnl_code
left join SDI.SDI_ZTSD014_1001 t13 on t1.sap_order_num = t13.vbeln and t1.sap_order_line_num = t13.posnr
left join (select sale_no,line_no,PLAN_DELV_TIME,BEF_PLAN_DELV_TIME from DWROTD.DWR_PS_SALE_ORDER_DTL_F where dtl_st_name = '正常' group by sale_no,line_no,PLAN_DELV_TIME,BEF_PLAN_DELV_TIME
) t14 on t1.order_num = t14.sale_no and t1.order_line_num = t14.line_no
left join dwiotd.dwi_wm_ladingbill_header_t t15 on t4.delv_num = t15.delivery and t15.group_num = 800
