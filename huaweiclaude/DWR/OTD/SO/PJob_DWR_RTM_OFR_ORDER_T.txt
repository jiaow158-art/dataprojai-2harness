DROP TABLE IF EXISTS DWROTD.DWR_RTM_OFR_ORDER_T_TEMP01;
CREATE TABLE DWROTD.DWR_RTM_OFR_ORDER_T_TEMP01
AS
SELECT ZUONR --报价单号
      --,SGTXT 
      ,SUM(TSL) AS TSL
  FROM SDI.SDI_ACDOCA_1001
 WHERE PERIOD_ID_M >= '202303'
   AND DRCRK = 'S'
   AND (SGTXT = 'OMS订货会提货定金回款' --退定金
    OR SGTXT LIKE '%释放OMS订货会报价单定金%')
   AND ZUONR IN(SELECT QUOTATION_CODE
                  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F
			   )
 GROUP BY ZUONR
;

DROP TABLE IF EXISTS DWROTD.DWR_RTM_OFR_ORDER_T_TEMP02;
CREATE TABLE DWROTD.DWR_RTM_OFR_ORDER_T_TEMP02
AS
SELECT ZUONR --报价单号
      --,SGTXT 
      ,SUM(TSL) AS TSL
  FROM SDI.SDI_ACDOCA_1001
 WHERE PERIOD_ID_M >= '202303'
   AND DRCRK = 'S'
   AND (SUBSTR(SGTXT,1,13) IN('OMS订货会报价单定金扣款')
    OR SGTXT = 'OMS订货会提货定金回款')
   AND ZUONR IN(SELECT QUOTATION_CODE
                  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F
			   )
 GROUP BY ZUONR
;

INSERT OVERWRITE TABLE DWROTD.DWR_RTM_OFR_ORDER_T
SELECT A.CREATION_DATE AS CREATE_DATE --报价单创建时间
      ,A.QUOTATION_TYPE AS OFR_TYPE_CODE --报价单类型编码
	  ,J.MEANING AS OFR_TYPE_NAME --报价单类型名称
      ,A.QUOTATION_CODE AS OFR_ORDER_NUM --报价单号
      ,A.SALE_ORG_CODE --销售组织编码
	  ,C.SALES_ORGANIZATION AS SALE_ORG_NAME --销售组织名称
      ,A.SALE_CHANNEL_CODE AS SALE_CHNL_CODE --分销渠道编码
	  ,G.VTEXT AS SALE_CHNL_NAME --分销渠道名称
      ,A.ACCOUNT_NUMBER AS CUST_NUM --客户编码
      ,A.ACCOUNT_NAME AS CUST_NAME --客户名称
      ,A.AMOUNT_BALANCE AS BAL_AMT --余额
      ,A.INTEGRAL_BALANCE AS INGR_BAL_AMT --积分余额
      ,A.SUBMIT_TIME --提交时间
      ,A.STATUS AS STATUS_CODE --报价单状态编码
	  ,H.MEANING AS STATUS_NAME --报价单状态名称
      ,A.QUOTATION_ID AS OFR_ORDER_ID --报价单ID
      ,B.SAP_DEPOSIT_NUMBER AS SAP_DEP_NUM --SAP扣款凭证号
      ,A.DEPOSIT_PROPORTION AS DEP_PROP --整单定金比例
	  ,COALESCE(F.TSL,0) AS DEPOSIT_AMT --定金金额（初始定金）
	  ,COALESCE(E.TSL,0) AS RETURN_AMT --退回定金	  
      ,A.SALESMAN_NUMBER AS SALE_EMP_NUM --销售员工编码
	  ,D.EMP_NAME AS SALE_EMP_NAME --销售员工名称
      ,A.VALID_TIME --有效时间
      ,A.VALID_END_TIME --有效结束时间
      ,A.POLICY_NAME AS PLCY_NAME --活动政策名称
      ,B.PRODUCT_CODE AS MATERIAL_CODE --物料编码
	  ,B.PRODUCT_NAME AS MATERIAL_NAME --物料名称
	  ,B.AGENT_PRICE --标准单价
	  ,B.UNIT_PRICE --折后单价
	  ,B.UNIT_AREA --单位面积
	  ,B.POLICY_DISCOUNT AS PLCY_DISC --折扣数
	  ,B.QUANTITY AS OFR_QTY --报价单数量
	  ,B.PRODUCT_UNIT AS MATERIAL_UNIT --计量单位
	  ,B.QUANTITY * B.UNIT_PRICE AS OFR_ORDER_AMT --单据金额
	  ,B.TOTAL_AMOUNT AS TOTAL_AMT --行总计金额
	  ,B.SALES_AREA AS SALE_AREA --销售面积
	  ,B.AVAILABLE_QTY AS AVAIL_QTY --可用数量
	  ,B.APPROVED_DISCOUNT AS APPR_DISC_AMT --审批折扣
	  ,B.APPROVED_QTY AS APPR_QTY --审批数量
	  ,B.TRANSFERRED_QUANTITY AS TFR_QTY --已转数量
	  ,B.REMAINING_TRANSFERRED_QUANTITY AS RMN_TFR_QTY --剩余可转数量
	  ,B.FACTORY_CODE --工厂编码
	  ,B.WAREHOUSE_CODE --库存地点编码
	  ,B.WAREHOUSE_AGE --库龄
	  ,B.STATUS AS LINE_STATUS_CODE --报价单行状态编码
	  ,I.MEANING AS LINE_STATUS_NAME --报价单行状态名称
	  ,B.CLEARANCE_REASONS AS CLRNC_RSN_CODE --清仓原因编码
	  ,B.PROMOTION_REASON_CODE AS PROM_RSN_CODE --促销原因编码
	  ,B.APPROVED_UNIT_PRICE AS APPR_UNIT_PRICE --审批单价
	  ,B.CLEARANCE_RESERVE_PRICE AS CLRNC_RESV_PRICE --清仓保底价
	  ,C.SALES_AREA AS YZ_DEPT_NAME --运营中心
	  ,C.SALES_DEPARTMENT_CODE AS XQ_DEPT_CODE --销区编码
	  ,C.SALES_DEPARTMENT AS XQ_DEPT_NAME --销区名称
	  ,C.SALES_GROUP_CODE AS SALE_GRP_CODE --销售组编码
	  ,C.SALES_GROUP AS SALE_GRP_NAME --销售组名称
	  ,C.COMPANY_CODE AS COM_CODE --公司编码
	  ,C.SHARED_WAREHOUSE_NAME --共享仓客户
	  ,C.CUST_MAK_SEG_CODE --客户城市等级编码	  
	  ,C.CUST_MAK_SEG_NAME --客户城市等级名称
	  ,C.CUST_QUALIFICATION_CODE AS CUST_QUAL_CODE --客户资质编码
	  ,C.CUST_QUALIFICATION_NAME AS CUST_QUAL_NAME --客户资质名称
	  ,C.LAND1 AS COUNTRY_CODE --国家编码
	  ,C.COUNTRY_NAME --国家名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,B.ID AS OFR_ORDER_LINE_ID --报价单行ID
  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F A
  LEFT JOIN DWRCS.DWR_CS_QUOTATION_ITEM_F B
    ON A.ID = B.QUOTATION_ID
  LEFT JOIN SDI.SDI_ZMASTER_CUS_1001 C
    ON A.ACCOUNT_NUMBER = C.ACCOUNT_NUMBER
   AND A.SALE_ORG_CODE = C.SALES_ORGANIZATION_CODE
   AND A.SALE_CHANNEL_CODE = C.SALES_CHANNEL_CODE
   AND C.PARVW = 'AG'
  LEFT JOIN DM.DM_WF_EMP_INFO_T D
    ON A.SALESMAN_NUMBER = D.EMP_NUM
  LEFT JOIN DWROTD.DWR_RTM_OFR_ORDER_T_TEMP01 E
    ON E.ZUONR = A.QUOTATION_CODE
  LEFT JOIN DWROTD.DWR_RTM_OFR_ORDER_T_TEMP02 F
    ON F.ZUONR = A.QUOTATION_CODE
  LEFT JOIN SDI.SDI_TVTWT_1001 G
    ON A.SALE_CHANNEL_CODE = G.VTWEG
   AND G.SPRAS = '1'
   AND G.MANDT = '800'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 H
    ON A.STATUS = H.LOOKUP_CODE
   AND H.LOOKUP_TYPE = 'OMS_QUOTATION_ORDER_HEAD_STATUS'
   AND H.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 I
    ON B.STATUS = I.LOOKUP_CODE
   AND I.LOOKUP_TYPE = 'OMS_QUOTATION_ORDER_LINE_STATUS'
   AND I.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 J
    ON A.QUOTATION_TYPE = J.LOOKUP_CODE
   AND J.LOOKUP_TYPE = 'OMS_CERAMIC_QUOTATION_ORDER_TYPE'
   AND J.ENABLED_FLAG = 'Y'
 WHERE A.STATUS NOT IN('DRAFT','CANCEL')  
;
