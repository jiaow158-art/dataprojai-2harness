INSERT OVERWRITE TABLE DWROTD.DWR_PRICE_CUST_OFR_ORDER_F
SELECT 
       PH.PRICE_CODE AS OFR_ORDER_NUM --申请单号
      ,PH.PRICE_NAME AS OFR_ORDER_NAME --价目表名称
      ,CI.CUSTOMER_CODE AS CUST_NUM --客户编码
      ,CI.CUSTOMER_NAME AS CUST_NAME --客户名称
      ,PH.CHANNEL_CODE --渠道编码
      ,PH.CHANNEL_NAME --渠道名称
      ,DI.DISTRIBUTOR_CODE AS CUSTOMER_CODE --经销商编码
      ,DI.DISTRIBUTOR_NAME AS CUSTOMER_NAME --经销商名称
      ,PH.PRICE_STATUS AS OFR_ORDER_STATUS_CODE --状态编码
	  ,C.MEANING AS OFR_ORDER_STATUS_NAME --状态
      ,PL.PRODUCT_CODE AS MATERIAL_CODE --商品编码
      ,PL.PRODUCT_NAME AS MATERIAL_NAME --商品名称
	  ,B.LENGTH_1 --长度
	  ,B.WIDTH --宽度
	  ,B.PRODUCT_LEVEL_CODE --产品层次编码
	  ,B.PRODUCT_LEVEL_NAME --产品层次名称
	  ,B.PRODUCT_LEVEL1_CODE --产品层次1编码
	  ,B.PRODUCT_LEVEL1_NAME --产品层次1名称
	  ,B.PRODUCT_LEVEL2_CODE --产品层次2编码
	  ,B.PRODUCT_LEVEL2_NAME --产品层次2名称
	  ,B.PRODUCT_LEVEL3_CODE --产品层次3编码
	  ,B.PRODUCT_LEVEL3_NAME --产品层次3名称
	  ,B.PRODUCT_LEVEL4_CODE --产品层次4编码
	  ,B.PRODUCT_LEVEL4_NAME --产品层次4名称
	  ,B.PRODUCT_LEVEL5_CODE --产品层次5编码
	  ,B.PRODUCT_LEVEL5_NAME --产品层次5名称
	  ,B.PRODUCT_LEVEL6_CODE --产品层次6编码
	  ,B.PRODUCT_LEVEL6_NAME --产品层次6名称
	  ,B.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道
	  ,B.RESERVED_FIELD1_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称	  
      ,PL.AUXILIARY_NAME AS AUX_NAME --辅助名称
      ,PL.COST_PRICE --成本价
      ,PL.SALE_PRICE --销售价
      ,PL.RETAIL_PRICE --零售价
      ,PL.FREIGHT AS FRT_PRICE --运费
      ,PL.DESCRIBED AS REMARK --备注
      ,PL.IS_ASSEMBLY AS IS_ASBL --是否套件
      ,PH.CREATION_DATE AS CREATE_DATE --制单时间
	  ,CI.ORGANIZATION_TYPE AS ORG_TYPE --机构类型
	  ,A.MEANING AS ORG_TYPE_NAME --机构类型名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_DTCUST_PRICE_HEADER_1031 PH 
 INNER JOIN SDI.SDI_DTCUST_PRICE_LINE_1031 PL 
    ON PH.HEADER_ID = PL.HEADER_ID 
 INNER JOIN SDI.SDI_DTCUST_CUSTOMER_INFO_1029 CI 
    ON PH.CUSTOMER_ID = CI.CUSTOMER_ID 
 INNER JOIN SDI.SDI_DT_DISTRIBUTOR_INFO_1029 DI 
    ON DI.DISTRIBUTOR_ID = PH.DISTRIBUTOR_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 A
    ON CI.ORGANIZATION_TYPE = A.LOOKUP_CODE
   AND A.LOOKUP_TYPE = 'ORGANIZATION_TYPE'
   AND A.ENABLED_FLAG = 'Y'
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
    ON PL.PRODUCT_CODE = B.MATERIAL_NUM 
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 C
    ON PH.PRICE_STATUS = C.LOOKUP_CODE
   AND C.LOOKUP_TYPE = 'SALES_PRICING_STATUS'
   AND C.ENABLED_FLAG = 'Y'	
 WHERE 1 = 1
;
