--DWI_SO_SALES_GROUP_CODE_OSS_T 到 DWR_SO_SALES_GROUP_CODE_OSS_F
--销售组编码转换表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_SALES_GROUP_CODE_OSS_F
SELECT
  DWI.SALES_CODE AS SALES_CODE,
  DWI.SALES_CODE_NAME AS SALES_CODE_NAME,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SALES_GROUP_CODE_OSS_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
