set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售订单行临时表TMP01(关联通用配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_ITEM_SAP_TMP01
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  RCD1.CONFIG_NAME AS SO_ITEM_TYPE_NAME,
  RCD2.CONFIG_NAME AS ROUTE_NAME,
  RCD3.CONFIG_NAME AS SPECIAL_INV_FLAG_NAME,
  RCD4.CONFIG_NAME AS SUBJECT_DISTR_TYPE_NAME,
  RCD5.CONFIG_NAME AS PROD_GROUP_NAME,
  RCD6.CONFIG_NAME AS COMMISSION_GROUP_NAME,
  RCD7.CONFIG_NAME AS PROD_LEVEL_NAME,
  RCD8.CONFIG_NAME AS REQ_TYPE_NAME,
  RCD9.CONFIG_NAME AS RECEIVE_POINTS_NAME,
  RCD10.CONFIG_NAME AS MATERIAL_SHIP_GROUP_NAME,
  RCD11.CONFIG_NAME AS DELIVERY_PRIORITY_NAME
FROM
  DWIOTD.DWI_SO_ITEM_SAP_T DWI --SAP销售订单行表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD1 --销售订单行类别中文名称
  ON DWI.GROUP_NUM = RCD1.GROUP_NUM
  AND DWI.SO_ITEM_TYPE = RCD1.CONFIG_CODE
  AND RCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD2 --路线中文名称
  ON DWI.GROUP_NUM = RCD2.GROUP_NUM
  AND DWI.ROUTE = RCD2.CONFIG_CODE
  AND RCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD3 --特殊库存标识中文名称
  ON DWI.GROUP_NUM = RCD3.GROUP_NUM
  AND DWI.SPECIAL_INV_FLAG = RCD3.CONFIG_CODE
  AND RCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD4 --科目分配类别中文名称
  ON DWI.GROUP_NUM = RCD4.GROUP_NUM
  AND DWI.SUBJECT_DISTR_TYPE = RCD4.CONFIG_CODE
  AND RCD4.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD5 --产品组中文名称
  ON DWI.GROUP_NUM = RCD5.GROUP_NUM
  AND DWI.PROD_GROUP = RCD5.CONFIG_CODE
  AND RCD5.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD6 --佣金组中文名称
  ON DWI.GROUP_NUM = RCD6.GROUP_NUM
  AND DWI.COMMISSION_GROUP = RCD6.CONFIG_CODE
  AND RCD6.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD7 --产品层次中文名称
  ON DWI.GROUP_NUM = RCD7.GROUP_NUM
  AND DWI.PROD_LEVEL = RCD7.CONFIG_CODE
  AND RCD7.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD8 --需求类型中文名称
  ON DWI.GROUP_NUM = RCD8.GROUP_NUM
  AND DWI.REQ_TYPE = RCD8.CONFIG_CODE
  AND RCD8.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD9 --装运点/接收点中文名称
  ON DWI.GROUP_NUM = RCD9.GROUP_NUM
  AND DWI.RECEIVE_POINTS = RCD9.CONFIG_CODE
  AND RCD9.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD10 --物料运输组中文名称
  ON DWI.GROUP_NUM = RCD10.GROUP_NUM
  AND DWI.MATERIAL_SHIP_GROUP = RCD10.CONFIG_CODE
  AND RCD10.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD11 --交货优先权中文名称
  ON DWI.GROUP_NUM = RCD11.GROUP_NUM
  AND DWI.DELIVERY_PRIORITY = RCD11.CONFIG_CODE
  AND RCD11.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
