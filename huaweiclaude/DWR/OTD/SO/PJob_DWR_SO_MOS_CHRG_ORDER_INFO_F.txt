INSERT OVERWRITE TABLE DWROTD.DWR_SO_MOS_CHRG_ORDER_INFO_F
SELECT  A.ID AS CHRG_ORDER_ID --ID
       ,A.SN AS CHRG_ORDER_NUM --收款单号
       ,A.CREATER_ID AS CREATE_EMP_ID --创建人ID
       ,A.CREATER_NAME AS CREATE_EMP_NAME --创建人名称
       ,A.CREATE_TIME --创建时间
       ,A.MODIFIER_ID AS UPDATE_EMP_ID --修改人ID
       ,A.MODIFIER_NAME AS UPDATE_EMP_NAME --修改人名称
       ,A.MODIFY_TIME AS UPDATE_TIME --修改时间
       ,A.INTENTION_ORDER_ID AS ORDER_ID --意向订单ID
       ,A.ORDER_SN AS ORDER_NUM --意向订单单号
       ,A.CLIENT_INFO_NUM AS CUST_NUM --客户代号
       ,A.COLLECTION_TYPE AS CHRG_TYPE_NAME --收款类型
       ,COALESCE(A.CHARGE_AMOUNT,0) AS CHRG_AMT --收款金额
       ,COALESCE(A.APPLIED_REFUND_AMOUNT,0) AS APLY_RTN_AMT --申请退款金额
       ,COALESCE(A.REFUND_AMOUNT,0) AS RTN_AMT --已退款金额
       ,A.CHARGE_STATUS AS CHRG_ST_CODE --20240910：\R\N-1.收款已驳回 1.收款审核中 2.退款中 3.已退款 4.退款已驳回 5.已收款
	   ,CASE WHEN A.CHARGE_STATUS = '-1' THEN '收款已驳回'
	         WHEN A.CHARGE_STATUS = '1' THEN '收款审核中'
	         WHEN A.CHARGE_STATUS = '2' THEN '退款中'
			 WHEN A.CHARGE_STATUS = '3' THEN '已退款'
			 WHEN A.CHARGE_STATUS = '4' THEN '退款已驳回'
			 WHEN A.CHARGE_STATUS = '5' THEN '已收款'
			  END AS CHRG_ST_NAME --退款状态描述
       ,A.CHARGE_TIME AS CHRG_TIME --收款时间
       ,A.APPLY_REFUND_TIME AS APLY_RTN_TIME --申请退款时间
       ,A.PASS_REFUND_TIME AS RTN_TIME --退款通过时间
       ,A.REJECT_REFUND_TIME AS REJ_RTN_TIME --退款驳回时间
       ,A.CLIENT_NAME AS CUST_NAME --客户名称
       ,A.CLIENT_TEL AS CUST_PHONE_NUM --客户电话
       ,A.STORE_ID --门店ID
	   ,B.CUST_NUM AS DEALER_NUM  --经销商编码
       ,B.DEALER_NAME --经销商名称
       ,COALESCE(B.OA_STORE_CODE,B.STORE_CODE) AS STORE_CODE --门店编码
       ,B.STORE_NAME --门店名称
       ,B.MKT_NAME --运营中心
	   ,B.SALE_GRP_NAME --线组名称	   
       ,A.ACCEPT_USER_ID AS FOLOW_EMP_ID --跟进用户ID
       ,A.PARENT_CLIENT_ID AS PARNT_CUST_ID --上级客户ID
       ,A.EXAMINE_CHARGE_TIME AS EXAM_CHRG_TIME --收款审核时间
       ,A.B1_ORDER_PAY_ID --B1收款单ID
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SAL_INTENTION_ORDER_CLIENT_CHARGE_1012 A
  LEFT JOIN DWRDIM.DWR_DIM_MOS_STORE_INFO_T B
    ON A.STORE_ID = B.ID
 WHERE CAST(DELETE_FLAG AS STRING) = 'FALSE'
