INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_DS_ENG_QUO_OA_APR_DTL_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.MAIN_ID AS MAIN_ID,
  DWI.EP_PRICE_GROSS_RATE AS EP_PRICE_GROSS_RATE,
  DWI.REFE_COST_GROSS_RATE AS REFE_COST_GROSS_RATE,
  DWI.APPRV_MATERIAL_PRICE AS APPRV_MATERIAL_PRICE,
  DWI.APPRV_COMPOSITE_PRICE AS APPRV_COMPOSITE_PRICE,
  DWI.APPLY_QTY AS APPLY_QTY,
  DWI.EP_DISCOUNT AS EP_DISCOUNT,
  DWI.APPROVAL_QUOTATION AS APPROVAL_QUOTATION,
  DWI.B_COST_GROSS_RATE AS B_COST_GROSS_RATE,
  DWI.APPLICATION_QUOTATION AS APPLICATION_QUOTATION,
  DWI.GUARANTEED_COST AS GUARANTEED_COST,
  DWI.BASE_COST_A_RATE AS BASE_COST_A_RATE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.MATERIAL_DESC AS MATERIAL_DESC,
  DWI.A_COST AS A_COST,
  DWI.B_COST AS B_COST,
  DWI.REFE_COST AS REFE_COST,
  DWI.AMOEBA_GROSS_RATE AS AMOEBA_GROSS_RATE,
  DWI.SQUARE_QTY AS SQUARE_QTY,
  DWI.EP_PRICE AS EP_PRICE,
  DWI.FREIGHT AS FREIGHT,
  DWI.SHOWROOM_RETAIL_PRICE AS SHOWROOM_RETAIL_PRICE,
  DWI.APP_COMPOSITE_SRV_FEE AS APP_COMPOSITE_SRV_FEE,
  DWI.FREIGHT_CITY AS FREIGHT_CITY,
  DWI.PROD_LEVEL AS PROD_LEVEL,
  DWI.AMOEBA_COST_AMT AS AMOEBA_COST_AMT,
  DWI.FINAN_COST AS FINAN_COST,
  DWI.SPECIFICATIONS AS SPECIFICATIONS,
  DWI.SEQUENCE AS SEQUENCE,
  DWI.BATCH AS BATCH,
  DWI.PROCESS_COST AS PROCESS_COST,
  DWI.INV_LOCATION AS INV_LOCATION,
  DWI.IS_HISTY_QUOTE AS IS_HISTY_QUOTE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DS_ENG_QUO_OA_APR_DTL_T DWI
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
