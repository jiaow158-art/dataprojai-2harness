INSERT OVERWRITE TABLE DWROTD.DWR_OMS_SALE_ORDER_F
SELECT HEAD.ORDER_NUMBER AS ORDER_NUM --订单编号
      ,ITEM.SAP_ORDER_NUMBER AS SAP_ORDER_NUM --SAP单号
      ,ITEM.POLICY_NAME --政策信息
      ,ITEM.STATUS AS ITEM_STATUS --订单行状态
      ,COALESCE(HEAD.STATE_SUBSIDY_CUSTOMER_CODE,HEAD.ACCOUNT_NUMBER) AS CUST_NUM --客户编码
      ,COALESCE(HEAD.STATE_SUBSIDY_CUSTOMER_NAME,HEAD.ACCOUNT_NAME) AS CUST_NAME --客户名称
      ,HEAD.CREATION_DATE AS CREATE_DATE --创建日期
      ,ITEM.MATERIAL_NUMBER AS MATERIAL_CODE --物料编码
      ,ITEM.MATERIAL_NAME --物料名称
      ,ITEM.MATERIAL_SPECIFICATIONS AS DIMENSION --规格
      ,ITEM.AGENT_PRICE --代理基准价 
      ,ITEM.POLICY_DISCOUNT AS DISCOUNT --折扣
      ,ITEM.UNIT_PRICE --单价
      ,ITEM.SUPPLY_AREA_CODE AS PRODUCT_AREA_CODE --产区编码
      ,ITEM.BATCH AS BATCH_NUM --批次
      ,ITEM.QUANTITY AS ORDER_QTY --下单数量
      ,ITEM.CONFIRM_QUANTITY AS CONFIRM_QTY --确认数
      ,A.PICK_QTY --提货数
      ,ITEM.UNIT_PRICE * ITEM.QUANTITY AS LINE_TOTAL_AMT --行合计
      ,ITEM.FACTORY_CODE AS FACTORY_WERKS_CODE --工厂编码
      ,ITEM.STOCK_LOCATION_CODE AS INV_LOCATION --库存地点
      ,CASE WHEN INSTR(ITEM.SAP_ORDER_ITEM_NUMBER,ITEM.SAP_ORDER_NUMBER) > 0
            THEN REPLACE(ITEM.SAP_ORDER_ITEM_NUMBER,ITEM.SAP_ORDER_NUMBER,'')
			ELSE ITEM.SAP_ORDER_ITEM_NUMBER
			 END AS SAP_ORDER_LINE_NUM --SAP行号
      ,HEAD.ORDER_TYPE --订单类型
	  ,HEAD.STATUS AS ORDER_STATUS --订单状态
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,ITEM.IS_ONE_PRICE --是否一口价政策
	  ,HEAD.SALE_ORG_CODE --销售组织
	  ,HEAD.SALE_CHANNEL_CODE AS SALE_CHNL_CODE --分销渠道
	  ,B.SALES_GROUP_CODE AS SALE_GRP_CODE --销售组编码
	  ,B.SALES_GROUP AS SALE_GRP_NAME --销售组名称
	  ,B.SALES_DEPARTMENT_CODE AS SALE_DEPT_CODE --销售部门编码 销区编码
	  ,B.SALES_DEPARTMENT AS SALE_DEPT_NAME --销售部门名称 销区名称
	  ,B.SALES_AREA AS SALE_AREA_NAME --运营中心名称
	  ,LPAD(B.PARENT_ACCNT_NUMBER,10,'0') AS MTH_CUST_NUM --母客户编码
	  ,C.DESCRIPTION AS MTH_CUST_NAME --母客户名称
	  ,CASE WHEN HEAD.STATE_SUBSIDY_CUSTOMER_CODE IS NOT NULL THEN HEAD.ACCOUNT_NUMBER END AS SBSDZ_CUST_NUM --国补客户编码
	  ,CASE WHEN HEAD.STATE_SUBSIDY_CUSTOMER_CODE IS NOT NULL THEN HEAD.ACCOUNT_NAME END AS SBSDZ_CUST_NAME --国补客户名称
  FROM SDI.SDI_S_ORDER_ALL_BUSINESS_1029 HEAD
  LEFT JOIN SDI.SDI_S_ENGINEERING_PLAN_ORDER_LINE_1029 ITEM
    ON HEAD.ID = ITEM.HEAD_ID
   AND ITEM.DEL_FLAG = 'N'
  LEFT JOIN (SELECT VBELN
                   ,POSNR
                   ,SUM(MENGE) AS PICK_QTY
               FROM DM.DM_OTD_SALES_ORDER_DET_T
              GROUP BY
                    VBELN
                   ,POSNR
            ) A
	ON A.VBELN = ITEM.SAP_ORDER_NUMBER
   AND A.POSNR = CASE WHEN INSTR(ITEM.SAP_ORDER_ITEM_NUMBER,ITEM.SAP_ORDER_NUMBER) > 0
                      THEN REPLACE(ITEM.SAP_ORDER_ITEM_NUMBER,ITEM.SAP_ORDER_NUMBER,'')
					  ELSE ITEM.SAP_ORDER_ITEM_NUMBER
					   END
  LEFT JOIN SDI.SDI_ZMASTER_CUS_1001 B
    ON COALESCE(HEAD.STATE_SUBSIDY_CUSTOMER_CODE,HEAD.ACCOUNT_NUMBER) = B.ACCOUNT_NUMBER
   AND HEAD.SALE_ORG_CODE = B.SALES_ORGANIZATION_CODE
   AND HEAD.SALE_CHANNEL_CODE = B.SALES_CHANNEL_CODE
   AND B.PARVW = 'AG'
  LEFT JOIN(SELECT DISTINCT ACCOUNT_NUMBER
                           ,DESCRIPTION
              FROM SDI.SDI_ZMASTER_CUS_1001
			 WHERE PARVW = 'AG'
		   ) C
    ON LPAD(B.PARENT_ACCNT_NUMBER,10,'0') = C.ACCOUNT_NUMBER
 WHERE HEAD.DEL_FLAG = 'N'
;
