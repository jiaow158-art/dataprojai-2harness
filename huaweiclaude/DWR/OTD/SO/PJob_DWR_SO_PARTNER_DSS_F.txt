  DWI.DESCRIPTION AS DESCRIPTION,

  DWI.DISTR_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  DWI.PARTNER_FUNCTION AS PARTNER_FUNCTION,
  DWI.NAME AS NAME,

  DWI.DEL_FLAG = 'N';
