with 
mara as(
select 
t1.material_num
,LENGTH_1--长度
,WIDTH--宽度
,DIMENSION--规格
,WIDTH*LENGTH_1/1000/1000 as product_area ,-- 面积
'' matl_group1_code ,  -- 产品主次（物料组1）
'' prodoct_zc --  -- 产品主次（物料组1）
,case when t1.RESERVED_FIELD_1  in('G004',
'G102',
'G112',
'G115'
)  then '工程' else '非工程' end  matl_channel -- 产品渠道  
,X6.VTEXT  as matl_cate-- 产品品类
,t1.RESERVED_FIELD_1
,t1.RESERVED_FIELD_3
from DWRDIM.DWR_DIM_MATERIAL_GENERAL_D t1
LEFT JOIN SDI.SDI_T179T_1001 X6 --产品品类
    ON X6.MANDT =t1.GROUP_NUM
   AND X6.SPRAS = '1'
   AND SUBSTR(t1.PRODUCT_LEVEL_CODE,1,4) = X6.PRODH

 WHERE t1.END_DATE = '4712-12-31'
   AND t1.DEL_FLAG = 'N'
   and t1.group_num='800'

           WHEN  LEV3_CODE IN('国内营销系统') AND LEV4_NAME IN('工程市场中心') THEN LEV4_NAME 
           WHEN  LEV3_CODE IN('国内营销系统')  THEN concat(LEV3_NAME,'其他') 
         ELSE LEV3_NAME END ,
        t8.JHDDLX, -- 计划订单类型
        t4.matl_cate,    -- 品类
        t4.DIMENSION,    -- 规格
        t4.prodoct_zc,    -- 产品主次
        t4.matl_channel,    -- 产品渠道
        t1.DISTR_CHAN ,     -- 销售渠道
        t6.vtext ,
        t1.plant,     -- 工厂  werks
        t7.name1 ,     -- 
        t4.product_area,
        t4.RESERVED_FIELD_1,
        t4.RESERVED_FIELD_3

        t1.calmonth,
        t1.bill_num ,t1.bill_item ,t1.doc_number ,t1.s_ord_item,

        t4.DIMENSION,    -- 规格

             WHEN t4.RESERVED_FIELD_1 = 'G105' THEN '设计师'
             WHEN t1.DISTR_CHAN = '08' THEN '家装大包'
             WHEN t4.RESERVED_FIELD_1 IN( 'G103','G104','G102') AND t4.RESERVED_FIELD_3 = 'A9' THEN '家装大包' 
             ELSE '零售' END as VTWEG_name,
        t1.plant as werks,     -- 工厂  werks
        t7.name1 as werks_name,     -- 
        t4.product_area
,0 as sales_amount  --货物价值_不含税   
,0 as  sales_tax-- 销售税金
,0 as rebate_amount --返点金额_含税
,0 as adjust_amount
,sum(inv_qty) as sales_qty

ON  t1.DISTR_CHAN =t2.DISTR_CHAN and t1.`/bic/zcustsale` = t2.`/bic/zcustsale`  and t1.SALESORG =t2.SALESORG and t1.DIVISION  = t2.DIVISION 
join DWRDIM.DWR_DIM_ORG_SALE_GRP_F  t3 on t2.sales_grp  =t3.SALE_GRP 
left join mara t4 ON t1.material  = t4.material_num
left join(
select vtweg,max(vtext) as vtext from sdi.sdi_tvtwt_1001 where spras='1' and mandt='800'
group by vtweg
) t6 On t1.DISTR_CHAN = t6.VTWEG
left join(
 select werks, max(name1) as name1 from sdi.sdi_t001w_1001 where mandt='800' and del_flag='N'
 group by werks
) t7 On t1.plant  = t7.werks

ON t1.DOC_NUMBER  = t8.VBELN
where  t1.BILL_TYPE in ('ZP1','ZF2','ZF21','ZRE1','ZRE2','ZS1','ZS11','ZS2','ZS21','ZL2','ZLS1','ZG2','ZGS2','ZT1','ZTS1')

and (t1.`/bic/zkpzt` <> 'E' or t1.`/bic/zkpzt`  is null )

and (t1.MATL_GROUP not in ('9000010','8000010','7000010','7000011') or t1.MATL_GROUP is null)

--and t1.calday>=concat(substr('20220430',1,6),1,2) and t1.calday<='20220430'
group by  t1.calday,
        t1.calmonth,
        t1.bill_num ,t1.bill_item ,t1.doc_number ,t1.s_ord_item,
        t2.sales_grp , -- 销售组织
        CASE WHEN  LEV3_CODE IN('国内营销系统') AND LEV4_NAME IN('区域运营中心','直营经营管理板块') THEN LEV5_NAME

        t8.JHDDLX, -- 计划订单类型
        t4.matl_cate,    -- 品类
        t4.DIMENSION,    -- 规格
        t4.prodoct_zc,    -- 产品主次
        t4.matl_channel,    -- 产品渠道
        t1.DISTR_CHAN ,     -- 销售渠道
        t6.vtext ,
        t1.plant,     -- 工厂  werks
        t7.name1 ,     -- 
        t4.product_area,
        t4.RESERVED_FIELD_1,
        t4.RESERVED_FIELD_3

 -- 财务调整数
 select '' as calday,

        `/bic/zkdpos` as s_ord_item,
        t1.sales_grp as vkgrp, -- 销售组织
        CASE WHEN  LEV3_CODE IN('国内营销系统') AND LEV4_NAME IN('区域运营中心','直营经营管理板块') THEN LEV5_NAME 
           WHEN  LEV3_CODE IN('国内营销系统') AND LEV4_NAME IN('工程市场中心') THEN LEV4_NAME 
            WHEN  LEV3_CODE IN('国内营销系统')  THEN concat(LEV3_NAME,'其他') 
       ELSE LEV3_NAME END ,
        t8.jhddlx as JHDDLX, -- 计划订单类型
        t4.matl_cate,    -- 品类

             WHEN t1.DISTR_CHAN = '08' THEN '家装大包'
