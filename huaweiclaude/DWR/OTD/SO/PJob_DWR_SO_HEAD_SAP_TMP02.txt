set hive.mapjoin.smalltable.filesize=100000000;

--SAP销售订单头临时表(关联DD07T配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_HEAD_SAP_TMP02
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  MCD1.DOMAIN_VALUE_NAME AS SAL_DIS_VCH_TYPE_NAME,
  MCD2.DOMAIN_VALUE_NAME AS AFFAIRS_GROUP_NAME,
  MCD3.DOMAIN_VALUE_NAME AS REFERENCE_BILL_NUM_NAME,
  MCD4.DOMAIN_VALUE_NAME AS PREVIOUSLY_VCH_TYPE_NAME,
  MCD5.DOMAIN_VALUE_NAME AS ALL_PROJ_REFUSE_STATUS_NAME,
  MCD6.DOMAIN_VALUE_NAME AS CONFIRM_STATUS_NAME,
  MCD7.DOMAIN_VALUE_NAME AS CREDIT_SAP_MNGMT_STATUS_NAME,
  MCD8.DOMAIN_VALUE_NAME AS CREDIT_TECH_ERR_STATUS_NAME,
  MCD9.DOMAIN_VALUE_NAME AS CREDIT_CHECK_STATUS_NAME,
  MCD10.DOMAIN_VALUE_NAME AS ALE_CONFIRM_STATUS_NAME,
  MCD11.DOMAIN_VALUE_NAME AS ISSUE_INVOICE_STATUS_NAME,
  MCD12.DOMAIN_VALUE_NAME AS STATUS_FUND_MAGT_NAME,
  MCD13.DOMAIN_VALUE_NAME AS BILL_FROZEN_STATUS_NAME,
  MCD14.DOMAIN_VALUE_NAME AS WHOLE_VCH_DEAL_STATUS_NAME
FROM
  DWIOTD.DWI_SO_HEAD_SAP_T DWI --SAP销售订单头表
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD1 --销售和分销凭证类别中文名称
  ON DWI.SAL_DIS_VCH_TYPE = MCD1.DOMAIN_VALUE_CODE
  AND MCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD2 --事务组中文名称
  ON DWI.AFFAIRS_GROUP = MCD2.DOMAIN_VALUE_CODE
  AND MCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD3 --参考单据的单据编号中文名称
  ON DWI.REFERENCE_BILL_NUM = MCD3.DOMAIN_VALUE_CODE
  AND MCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD4 --先前销售和分销凭证的凭证类别中文名称
  ON DWI.PREVIOUSLY_VCH_TYPE = MCD4.DOMAIN_VALUE_CODE
  AND MCD4.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD5 --所有凭证项目的总体拒绝状态中文名称
  ON DWI.ALL_PROJ_REFUSE_STATUS = MCD5.DOMAIN_VALUE_CODE
  AND MCD5.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD6 --确认状态中文名称
  ON DWI.CONFIRM_STATUS = MCD6.DOMAIN_VALUE_CODE
  AND MCD6.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD7 --信用检查 SAP 信用管理的状态中文名称
  ON DWI.CREDIT_SAP_MNGMT_STATUS = MCD7.DOMAIN_VALUE_CODE
  AND MCD7.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD8 --SAP Credit Management 的技术错误状态中文名称
  ON DWI.CREDIT_TECH_ERR_STATUS = MCD8.DOMAIN_VALUE_CODE
  AND MCD8.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD9 --信用检查的总体状态中文名称
  ON DWI.CREDIT_CHECK_STATUS = MCD9.DOMAIN_VALUE_CODE
  AND MCD9.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD10 --ALE 的确认状态中文名称
  ON DWI.ALE_CONFIRM_STATUS = MCD10.DOMAIN_VALUE_CODE
  AND MCD10.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD11 --出具发票状态( 订单相关的出具发票凭证)中文名称
  ON DWI.ISSUE_INVOICE_STATUS = MCD11.DOMAIN_VALUE_CODE
  AND MCD11.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD12 --状态基金管理中文名称
  ON DWI.STATUS_FUND_MANAGEMENT = MCD12.DOMAIN_VALUE_CODE
  AND MCD12.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD13 --总体开票冻结状态中文名称
  ON DWI.BILL_FROZEN_STATUS = MCD13.DOMAIN_VALUE_CODE
  AND MCD13.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD14 --凭证的总体处理状态中文名称
  ON DWI.WHOLE_VCH_DEAL_STATUS = MCD14.DOMAIN_VALUE_CODE
  AND MCD14.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
