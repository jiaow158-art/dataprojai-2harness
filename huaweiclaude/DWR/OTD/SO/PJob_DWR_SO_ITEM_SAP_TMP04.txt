--SAP销售订单行临时表TMP04(关联DD07配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_ITEM_SAP_TMP04
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  MCD11.DOMAIN_VALUE_NAME AS PROJ_PRICE_INCOMP_TYPE_NAME,
  MCD12.DOMAIN_VALUE_NAME AS SAL_ITEM_DEAL_STATUS_NAME,
  MCD13.DOMAIN_VALUE_NAME AS ALE_CONFIRM_STATUS_NAME,
  MCD14.DOMAIN_VALUE_NAME AS PROJ_INCOMP_STATUS_NAME,
  MCD15.DOMAIN_VALUE_NAME AS SAL_ITEM_REJECT_STATUS_NAME,
  MCD16.DOMAIN_VALUE_NAME AS BOM_TYPE_NAME,
  MCD17.DOMAIN_VALUE_NAME AS CHANGE_STATUS_NAME,
  MCD18.DOMAIN_VALUE_NAME AS CONSUMPTION_ENTRY_NAME,
  MCD19.DOMAIN_VALUE_NAME AS QUOTE_CON_COMP_RULES_NAME,
  MCD20.DOMAIN_VALUE_NAME AS PREVIOUS_SAL_VCH_TYPE_NAME
FROM
  DWIOTD.DWI_SO_ITEM_SAP_T DWI --SAP销售订单行表
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD11 --项目定价不完备类型中文名称
  ON DWI.PROJ_PRICE_INCOMPLETE_TYPE = MCD11.DOMAIN_VALUE_CODE
  AND MCD11.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD12 --销售与分销凭证项目的总体处理状态中文名称
  ON DWI.SAL_ITEM_DEAL_STATUS = MCD12.DOMAIN_VALUE_CODE
  AND MCD12.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD13 --ALE 的确认状态中文名称
  ON DWI.ALE_CONFIRM_STATUS = MCD13.DOMAIN_VALUE_CODE
  AND MCD13.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD14 --关于开票的项目未完成状态中文名称
  ON DWI.PROJ_INCOMPLETE_STATUS = MCD14.DOMAIN_VALUE_CODE
  AND MCD14.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD15 --销售凭证项目拒绝状态中文名称
  ON DWI.SAL_ITEM_REJECT_STATUS = MCD15.DOMAIN_VALUE_CODE
  AND MCD15.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD16 --物料清单类别中文名称
  ON DWI.BOM_TYPE = MCD16.DOMAIN_VALUE_CODE
  AND MCD16.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD17 --人工价格更改状态中文名称
  ON DWI.CHANGE_STATUS = MCD17.DOMAIN_VALUE_CODE
  AND MCD17.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD18 --消耗记帐中文名称
  ON DWI.CONSUMPTION_ENTRY = MCD18.DOMAIN_VALUE_CODE
  AND MCD18.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD19 --报价/合同的完成规则中文名称
  ON DWI.QUOTE_CON_COMPLETE_RULES = MCD19.DOMAIN_VALUE_CODE
  AND MCD19.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD20 --先前销售和分销凭证的凭证类别中文名称
  ON DWI.PREVIOUS_SAL_VCH_TYPE = MCD20.DOMAIN_VALUE_CODE
  AND MCD20.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
