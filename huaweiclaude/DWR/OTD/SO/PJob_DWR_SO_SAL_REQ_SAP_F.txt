--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售需求表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_SAL_REQ_SAP_F
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  DWI.SAL_PLAN_ITEM_NUM AS SAL_PLAN_ITEM_NUM,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  RCD1.CONFIG_NAME AS SAL_VCH_TYPE_NAME,
  DWI.PROJ_SO_PROBABILITY AS PROJ_SO_PROBABILITY,
  DWI.REQ_TYPE AS REQ_TYPE,
  DWI.MRP_SCOPE AS MRP_SCOPE,
  DWI.BATCH_NUM AS BATCH_NUM,
  DWI.REQ_DISTR_QTY AS REQ_DISTR_QTY,
  DWI.SUBJECT_DISTR_TYPE AS SUBJECT_DISTR_TYPE,
  RCD2.CONFIG_NAME AS SUBJECT_DISTR_TYPE_NAME,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.GIVEN_INV_REVALUED AS GIVEN_INV_REVALUED,
  MCD1.DOMAIN_VALUE_NAME AS GIVEN_INV_REVALUED_NAME,
  DWI.CONSUMPTION_ENTRY AS CONSUMPTION_ENTRY,
  MCD2.DOMAIN_VALUE_NAME AS CONSUMPTION_ENTRY_NAME,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  DWI.INV_LOCATION AS INV_LOCATION,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  DWI.AVAIL_DATE AS AVAIL_DATE,
  DWI.BASE_MEASURE_UNIT AS BASE_MEASURE_UNIT,
  DWI.MRP_NC_REQ_RECORD AS MRP_NC_REQ_RECORD,
  DWI.OPEN_QTY AS OPEN_QTY,
  DWI.PLAN_TYPE AS PLAN_TYPE,
  DWI.BUSINESS_PROJ_NUM AS BUSINESS_PROJ_NUM,
  DWI.WBS_ELEMENT AS WBS_ELEMENT,
  DWI.REQ_PLAN_DISTR_QTY AS REQ_PLAN_DISTR_QTY,
  DWI.SPECIAL_INV_FLAG AS SPECIAL_INV_FLAG,
  RCD3.CONFIG_NAME AS SPECIAL_INV_FLAG_NAME,
  DWI.CONVER_QTY AS CONVER_QTY,
  DWI.BUSINESS_VCH_NUM AS BUSINESS_VCH_NUM,
  DWI.SALES_DIS_VCH_TYPE AS SALES_DIS_VCH_TYPE,
  MCD3.DOMAIN_VALUE_NAME AS SALES_DIS_VCH_TYPE_NAME,
  DWI.SKU_CONFIRM_QTY AS SKU_CONFIRM_QTY,
  DWI.DISTRIBUTION_FLAG AS DISTRIBUTION_FLAG,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  T5.CUST_ID,
  T5.CUST_NAME
FROM
  DWIOTD.DWI_SO_SAL_REQ_SAP_T DWI --SAP销售需求表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD1 --销售凭证类型中文名称
  ON DWI.GROUP_NUM = RCD1.GROUP_NUM
  AND DWI.SAL_VCH_TYPE = RCD1.CONFIG_CODE
  AND RCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD2 --科目分配类别中文名称
  ON DWI.GROUP_NUM = RCD2.GROUP_NUM
  AND DWI.SUBJECT_DISTR_TYPE = RCD2.CONFIG_CODE
  AND RCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD3 --特殊库存标识中文名称
  ON DWI.GROUP_NUM = RCD3.GROUP_NUM
  AND DWI.SPECIAL_INV_FLAG = RCD3.CONFIG_CODE
  AND RCD3.LANG_CODE = '1' 
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD1 --特定库存的评估中文名称
  ON DWI.GIVEN_INV_REVALUED = MCD1.DOMAIN_VALUE_CODE
  AND MCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD2 --消耗记帐中文名称
  ON DWI.CONSUMPTION_ENTRY = MCD2.DOMAIN_VALUE_CODE
  AND MCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD3 --销售和分销凭证类别中文名称
  ON DWI.SALES_DIS_VCH_TYPE = MCD3.DOMAIN_VALUE_CODE
  AND MCD3.LANG_CODE = '1' 
  --关联物料主数据表取物料名称
  INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON DWI.GROUP_NUM = MAT.GROUP_NUM
  AND DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N' 
  --关联工厂主数据表取工厂名称
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC ON DWI.GROUP_NUM = FAC.GROUP_NUM
  AND DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  --关联客户主数据取客户名称 
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T T5		--关联客户主数据表,取CUST_NAME
  ON T5.CUST_NUM = DWI.CUST_NUM				--客户ID
  AND T5.GROUP_NUM = DWI.GROUP_NUM			--集团
  AND T5.END_DATE ='4712-12-31'				--结束时间'4712-12-31'
  AND T5.DEL_FLAG = 'N'					--删除标识'N'  
WHERE
  DWI.DEL_FLAG = 'N';
