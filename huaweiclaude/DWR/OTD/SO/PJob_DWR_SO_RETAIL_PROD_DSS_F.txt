  DWI.ID AS ID,
  DWI.PRODUCT_MODEL AS PRODUCT_MODEL,
  DWI.NAME AS NAME,
  DWI.PRICE AS PRICE,
  DWI.DISCOUNTED_PRICES AS DISCOUNTED_PRICES,

  DWI.SPECIFICATION AS SPECIFICATION,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.REMARK AS REMARK,
  DWI.END_DATE AS END_DATE,
  DWI.START_DATE AS START_DATE,
  DWI.SERIES AS SERIES,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  DWI.DEL_FLAG AS DEL_FLAG,

  DWIOTD.DWI_SO_RETAIL_PROD_DSS_T DWI --DSS零售产品

  DWI.DEL_FLAG = 'N';
