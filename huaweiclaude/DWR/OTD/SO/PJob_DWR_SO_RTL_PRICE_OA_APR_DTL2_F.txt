--DWI_SO_RTL_PRICE_OA_APR_DTL2_T 到 DWR_SO_RTL_PRICE_OA_APR_DTL2_F
--SBC-DCM-A5-001-直营价格审批申请流程明细表2(零售)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_RTL_PRICE_OA_APR_DTL2_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.SUBTOTAL_AMT AS SUBTOTAL_AMT,
  DWI.MAIN_ID AS MAIN_ID,
  DWI.APPLY_DISCOUNT AS APPLY_DISCOUNT,
  DWI.APPROVE_DISCOUNT AS APPROVE_DISCOUNT,
  DWI.APP_PRICE AS APP_PRICE,
  DWI.APPLY_PRICE AS APPLY_PRICE,
  DWI.SALE_GROSS_PROFIT_MARGIN AS SALE_GROSS_PROFIT_MARGIN,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.REFER_COST AS REFER_COST,
  DWI.MATERIAL_ABBREVIATION AS MATERIAL_ABBREVIATION,
  DWI.SERVICE_FEE AS SERVICE_FEE,
  DWI.ITEM_ID AS ITEM_ID,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL2_T DWI
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
