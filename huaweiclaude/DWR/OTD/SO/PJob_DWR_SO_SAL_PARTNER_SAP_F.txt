--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售凭证合作伙伴表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_SAL_PARTNER_SAP_F
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  DWI.PARTNER_FUNC AS PARTNER_FUNC,
  DWI.CUST_ID AS CUST_ID,
  DWI.CUST_NUM AS CUST_NUM,
  CUST.CUST_NAME AS CUST_NAME,
  DWI.ADDRESS_ID AS ADDRESS_ID,
  DWI.ADDRESS_FLAG AS ADDRESS_FLAG,
  DWI.BUSI_PARTNER_CUST_DESC AS BUSI_PARTNER_CUST_DESC,
  DWI.PERSON_NUM AS PERSON_NUM,
  DWI.CNTRY_REGION_CODE AS CNTRY_REGION_CODE,
  DWI.GOODS_SEND_DESTINATION_CODE AS GOODS_SEND_DESTINATION_CODE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  B.NACHN AS SALE_EMP_NAME --销售雇员名称
FROM DWIOTD.DWI_SO_SAL_PARTNER_SAP_T DWI --SAP销售凭证合作伙伴表
  --关联客户主数据表取客户名称
  INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST ON DWI.GROUP_NUM = CUST.GROUP_NUM
  AND DWI.CUST_ID = CUST.CUST_ID
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_HR_OA_PERNR_INFO_T B 
ON DWI.PERSON_NUM = B.PERNR
WHERE
  DWI.DEL_FLAG = 'N';
