set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置成非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
INSERT OVERWRITE TABLE  DWROTD.DWR_TRA_TMS_BILL_STM_F PARTITION (PERIOD_ID_M)  
select
a.entrydt as create_date,--创建日期
a.statement_of_account_no as SUBL_NO,--对账单号
bbrt.BILL_ORDER_NO AS WB_NO,--, --TMS运单号
bbrt.fee_item_name AS FEE_ITEM_NAME,--,--费用项目
bbrt.bill_order_table AS FEE_DTL_CODE,--费用说明编码
p2.name AS FEE_DTL_NAME,--费用说明描述
bbrt.sku_code as material_code, --物料编码
bbrt.sku_name  as  material_name, --物料名称
bbrt.COSTCENTER_NO as COST_CENT_CODE,--成本中心编码
CUST.COSTCENTER_NAME AS COST_CENT_NAME,--成本中心描述
a.status, --状态
p.name as statu_name,--状态描述
a.udf5 AS CARG_ANME,--承运商/结算对象
a.statement_date AS statement_date,--结算周期
a.company_id ,--公司ID
substr(regexp_extract(A.udf2,'[0-9]*[A-Z]*[0-9]*[0-9]',0),1,10) as INDTY_ID, --实际赔偿凭证号
bbrt.quote_type AS quote_type_CODE, -- 合同种类编码
p1.name AS quote_type_NAME ,  --合同种类描述
case when d.ordertype = '30' THEN pcus.descr 
else pcom.descr 
end as companyname  ,  --开票公司
bbrt.bill_total as bill_total_line,  --计费金额
bbrt.charge_unit_name as charge_unit_name,--单位
bbrt.cal_qty as cal_qty,--计费系数
bbrt.bill_unit_price  as bill_unit_price ,--计费单价
a.after_tax_amount, -- 含税金额
A.pretax_amount,--不含税金额
a.entryid AS CREAT_id,--创建人ID
e1.name  as CREAT_NAME,--创建人名称
bbrt.source_order_no,--计费业务单据关联单号
bbrt.SOURCE_ORDER_LINENO, --计费业务单据关联单行号
CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE,
SUBSTR(a.entrydt,1,7) AS PERIOD_ID_M
from  sdi.sdi_bms_bill_order_1005  a
left join  
 SDI.SDI_plat_code_1005 p ON a.status = p.code 
AND p.typename = 'ACCOUNT_CHECK_STATUS' 
AND p.tenant_id = a.tenant_id
LEFT  JOIN sdi.sdi_bms_bill_result_tms_1005 bbrt 
ON bbrt.statement_no = a.statement_of_account_no
LEFT  JOIN  SDI.SDI_plat_code_1005  p1 
ON 
p1.code = bbrt.quote_type
AND p1.typename = 'BMS_ARAP'
left join sdi.sdi_tms_do_1005  d
on  d.do_no = bbrt.bill_order_no
LEFT  JOIN sdi.sdi_plat_partner_1005 pcom ON
(	
((a.udf1 = pcom.partner_key)AND (pcom.trade_type = 'COM'))
AND (a.tenant_id = pcom.tenant_id)
)
LEFT  JOIN sdi.sdi_plat_partner_1005 pcus
ON (
(pcus.partner_key = d.customer_code)
AND (pcus.trade_type = 'CUS')
				)
LEFT  JOIN 
sdi.sdi_plat_login_user_1005 u1 
ON (a.entryid = u1.id)
LEFT  JOIN sdi.sdi_plat_employee_1005 e1
ON (e1.id = u1.employee_id)
LEFT  JOIN  SDI.SDI_plat_code_1005  p2
ON 
p2.code = bbrt.bill_order_table
AND p2.typename = 'BILL_ORDER_TABLE'
LEFT JOIN   SDI.SDI_BMS_COSTCENTER_1005 CUST 
ON bbrt.COSTCENTER_NO = CUST.COSTCENTER_CODE
