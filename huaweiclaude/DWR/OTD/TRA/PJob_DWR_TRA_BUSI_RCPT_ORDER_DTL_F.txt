INSERT OVERWRITE TABLE DWROTD.DWR_TRA_BUSI_RCPT_ORDER_DTL_F
SELECT A.ID
      ,B.ID AS RCPT_ORDER_DTL_ID
      ,A.RCPT_NUM --回单号
      ,A.DO_NUM --运单号
	  ,B.DO_LINENO AS DO_LINE_NUM --运单行号
      ,A.OP_TIME --操作时间
      ,A.OP_EMP_NAME --操作人
      ,A.DELV_NUM --提货单号
      ,A.OBL_NUM --预留单号
      ,A.PROJ_PLAN_NUM --项目计划号
      ,A.RCPT_ST_CODE --回单状态编码
      ,A.RCPT_ST_NAME --回单状态描述
	  ,B.QUANTITY AS QTY --数量
	  ,B.WEIGHT AS WT --重量
	  ,B.SKUCODE AS MATERIAL_NUM --物料编码
	  ,B.CARRIER_CODE AS DRVR_CAR_NUM --司机车牌号
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM DWROTD.DWR_TRA_BUSI_RCPT_ORDER_F A
  LEFT JOIN SDI.SDI_TMS_RECEIPT_BUSINESS_DETAIL_1005 B
    ON A.RCPT_NUM = B.RECEIPT_NO
