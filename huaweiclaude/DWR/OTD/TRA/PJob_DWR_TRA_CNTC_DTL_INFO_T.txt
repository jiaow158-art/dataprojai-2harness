WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWROTD.DWR_TRA_CNTC_DTL_INFO_T
SELECT A.ID AS CNTC_DTL_ID --合同行ID
      ,B.CNTC_ID --合同ID
      ,A.CONTRACT_NO AS CNTC_NUM --合同编号
	  ,B.CNTC_NAME --合同名称
	  ,B.EFF_TIME --生效时间
	  ,B.INVAL_TIME --失效时间
	  ,B.OA_RQST_ID --OA流水号
	  ,A.PARAM4 AS FEE_TYPE_CODE --计费类型编码
	  ,T1.DESCR AS FEE_TYPE_NAME --计费类型描述
	  ,A.PARAM15 AS TAX_PRC --含税单价
	  ,A.PRICE AS NO_TAX_PRC --不含税单价
	  ,A.PARAM2 AS SHIP_METH_CODE --运输方式编码
	  ,T2.DESCR AS SHIP_METH_NAME --运输方式描述
	  ,A.BILL_METHOD AS BILL_METH_CODE --价格体系编码
	  ,T3.DESCR AS BILL_METH_NAME --价格体系描述
	  ,A.LOG_ID AS LGSTCS_WAY_CODE --物流方向编码
	  ,CASE WHEN A.LOG_ID = '0' THEN '反向物流' ELSE '正向物流' END AS LGSTCS_WAY_NAME --物流方向描述
	  ,A.PARAM1 AS MATERIAL_TYPE_CODE --物料类型编码
	  ,T4.DESCR AS MATERIAL_TYPE_NAME --物料类型描述
	  ,A.LOADING_SITE_CODE AS DELV_SITE_CODE --提货站点编码
	  ,T5.NAME AS DELV_SITE_NAME --提货站点名称
	  ,A.PARAM6 AS DELV_PROV_CODE --提货省编码
	  ,T6.STREET_NAME AS DELV_PROV_NAME --提货省名称
	  ,A.PARAM7 AS DELV_CITY_CODE --提货市编码
	  ,T7.STREET_NAME AS DELV_CITY_NAME --提货市名称
	  ,A.PARAM8 AS DELV_DIST_CODE --提货区编码
	  ,T8.STREET_NAME AS DELV_DIST_NAME --提货区名称
	  ,A.PARAM9 AS DELV_TOWN_CODE --提货乡镇编码
	  ,T9.STREET_NAME AS DELV_TOWN_NAME --提货乡镇名称
	  ,A.UNLOADING_SITE_CODE AS RECV_SITE_CODE --收货站点编码
	  ,T10.NAME AS RECV_SITE_NAME --收货站点名称
	  ,A.PARAM10 AS RECV_PROV_CODE --收货省编码
	  ,T11.STREET_NAME AS RECV_PROV_NAME --收货省名称
	  ,A.PARAM11 AS RECV_CITY_CODE --收货市编码
	  ,T12.STREET_NAME AS RECV_CITY_NAME --收货市名称
	  ,A.PARAM12 AS RECV_DIST_CODE --收货区编码
	  ,T13.STREET_NAME AS RECV_DIST_NAME --收货区名称
	  ,A.PARAM13 AS RECV_TOWN_CODE --收货乡镇编码
	  ,T14.STREET_NAME AS RECV_TOWN_NAME --收货乡镇名称
	  ,A.PARAM21 AS SRVC_TYPE_CODE --服务类型编码
	  ,T15.NAME AS SRVC_TYPE_NAME --服务类型描述
	  ,A.CREATE_DT AS CREATE_TIME --创建时间
	  ,A.CREATE_ID AS CREATE_EMP_ID --创建人ID
	  ,T16.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T16.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_BMS_CONTRACT_QUOTE_1005 A
 INNER JOIN DWROTD.DWR_TRA_CNTC_INFO_T B
    ON A.CONTRACT_NO = B.CNTC_NUM
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.PARAM4 = T1.CODE
   AND T1.TYPENAME = 'BMS_CHARGE_TYPE'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.PARAM2 = T2.CODE
   AND T2.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T3
    ON A.BILL_METHOD = T3.CODE
   AND T3.TYPENAME = 'PRICE_SYSTEM'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T4
    ON A.PARAM1 = T4.CODE
   AND T4.TYPENAME = 'BMS_SKU_TYPE'
  LEFT JOIN SDI.SDI_TMS_SITE_1005 T5
    ON A.LOADING_SITE_CODE = T5.CODE
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T6
    ON A.PARAM6 = T6.STREET_CODE
   AND T6.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T7
    ON A.PARAM7 = T7.STREET_CODE
   AND T7.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T8
    ON A.PARAM8 = T8.STREET_CODE
   AND T8.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T9
    ON A.PARAM9 = T9.STREET_CODE
   AND T9.STREET_LEVEL = '4'
  LEFT JOIN SDI.SDI_TMS_SITE_1005 T10
    ON A.UNLOADING_SITE_CODE = T10.CODE
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T11
    ON A.PARAM10 = T11.STREET_CODE
   AND T11.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T12
    ON A.PARAM11 = T12.STREET_CODE
   AND T12.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T13
    ON A.PARAM12 = T13.STREET_CODE
   AND T13.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T14
    ON A.PARAM13 = T14.STREET_CODE
   AND T14.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T15
    ON A.PARAM21 = T15.CODE
   AND T15.TYPENAME = 'SERVICE_WAY'
   AND T15.STATUS <> '-1'
  LEFT JOIN T AS T16
    ON A.CREATE_ID = T16.ID
 WHERE A.TENANT_ID = 1
