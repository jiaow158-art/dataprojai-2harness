INSERT OVERWRITE TABLE DWROTD.DWR_TRA_TMS_ORDER_INFO_T
SELECT A.ID AS ORDER_ID --订单ID
      ,A.CODE AS ORDER_NUM --TMS订单号
	  ,A.ORDER_SOURCE AS ORDER_SRC_NAME --订单来源
	  ,A.LOADING_METHOD AS PICK_METH_CODE --提货方式编码
	  ,T1.NAME AS PICK_METH_NAME --提货方式描述
	  ,A.SHIP_METHOD AS SHIP_METH_CODE --运输方式编码
	  ,T2.NAME AS SHIP_METH_NAME --运输方式描述
	  ,A.ORDERTYPE AS ORDER_TYPE_CODE --订单类型编码
	  ,T3.NAME AS ORDER_TYPE_NAME --订单类型描述
	  ,A.STATUS AS ORDER_ST_CODE --状态编码
	  ,T4.NAME AS ORDER_ST_NAME --状态描述
	  ,A.EXTERNAL_ORDER_CODE AS EXTR_ORDER_NUM --外部订单号
	  ,A.PRIORITY AS PRIR_CODE --优先级编码
	  ,T5.NAME AS PRIR_NAME --优先级描述
	  ,SUBSTR(A.ENTRYDT,1,10) AS ORDER_DATE --建单日期
	  ,A.ENTRYDT AS ORDER_TIME --建单时间
	  ,A.PLAN_DELIVERY_DATE AS PLAN_DELV_DATE --计划提货时间
      ,A.REQUIRED_SEND_DATE AS RQST_SEND_DATE --计划到货时间
	  ,A.LOADING_SITE_NAME AS DELV_SITE_NAME --提货站点名称
	  ,A.UNLOADING_SITE_NAME AS RECV_SITE_NAME --收货站点名称
	  ,A.LOADING_PROVINCE AS DELV_PROV_CODE --提货省编码
	  ,T6.STREET_NAME AS DELV_PROV_NAME --提货省名称
	  ,A.LOADING_CITY AS DELV_CITY_CODE --提货市编码
	  ,T7.STREET_NAME AS DELV_CITY_NAME --提货市名称
	  ,A.LOADING_COUNTY AS DELV_DIST_CODE --提货区编码
	  ,T8.STREET_NAME AS DELV_DIST_NAME --提货区名称
	  ,A.LOADING_ADDRESS AS DELV_ADDR --提货详细地址
	  ,A.UNLOADING_PROVINCE AS RECV_PROV_CODE --收货省编码
	  ,T9.STREET_NAME AS RECV_PROV_NAME --收货省名称
	  ,A.UNLOADING_CITY AS RECV_CITY_CODE --收货市编码
	  ,T10.STREET_NAME AS RECV_CITY_NAME --收货市名称
	  ,A.UNLOADING_COUNTY AS RECV_DIST_CODE --收货区编码
	  ,T11.STREET_NAME AS RECV_DIST_NAME --收货区名称
	  ,A.UNLOADING_ADDRESS AS RECV_ADDR --收货详细地址
	  ,A.UNLOADINGPERSONNAME AS RECV_NAME --收货联系人
	  ,A.UNLOADINGPERSON_TELEPHONE AS RECV_PHONE_NUM --收货人电话
	  ,A.CUSTOMER_CODE AS CUST_NUM --客户代码
	  ,T12.DESCR AS CUST_NAME --客户名称
	  ,A.TOTAL_VOLUME AS TOT_VOL --总体积(m³)
	  ,A.TOTAL_NUMBER AS TOT_QTY --总数量
	  ,A.TOTAL_GROSS_WEIGHT AS TOT_GRS_WT --总毛重(kg)
	  ,A.TOTAL_NET_WEIGHT AS TOT_NET_WT --总净重(kg)
	  ,A.TOTAL_BOXES AS TOT_BOX_QTY --总箱数
	  ,A.TOTAL_MONEY AS TOT_AMT --总价
	  ,A.ACTUAL_TOTAL_NUMBER AS ACTL_TOT_QTY --实际总数量
	  ,A.ACTUAL_TOTAL_VOLUME AS ACTL_TOT_VOL --实际总体积(m³)
	  ,A.ACTUAL_TOTAL_GROSS_WEIGHT AS ACTL_TOT_GRS_WT --实际总毛重(kg)
	  ,A.ACTUAL_TOTAL_NET_WEIGHT AS ACTL_TOT_NET_WT --实际总净重(kg)
	  ,A.ACTUAL_TOTAL_BOXES AS ACTL_TOT_BOX_QTY --实际总箱数
	  ,A.ACTUAL_TOTAL_MONEY AS ACTL_TOT_AMT --实际总价
	  ,A.SUSR12 AS UPST_METH_CODE --上楼方式编码
	  ,T13.NAME AS UPST_METH_NAME --上楼方式描述
	  ,A.LADDER AS LAD_NUM --楼层数
	  ,A.ELEVATOR AS ELV_NUM --电梯楼层数
	  ,A.PROCESSOR AS OP_EMP_NAME --单证处理人
	  ,A.CANCEL_RESON AS CAN_RSN --取消原因
	  ,A.SUSR14 AS IS_VALUT_CODE --是否保价编码
	  ,T14.NAME AS IS_VALUT_NAME --是否保价描述
	  ,A.RELEASE_FLAG AS IS_RLSE_CODE --是否释放编码
	  ,T15.NAME AS IS_RLSE_NAME --是否释放描述
	  ,A.DESCRIPTION AS ORDER_DESC --订单描述
	  ,A.SUSPENDDESCRIPTION AS SUSP_DESC --暂停描述
	  ,A.SUSPENDREASON AS SUSP_RSN_CODE --暂停原因编码
	  ,T16.NAME AS SUSP_RSN_NAME --暂停原因描述
	  ,A.CLEAN_FLAG AS IS_CLEAN_CODE --是否清洗编码
	  ,T17.NAME AS IS_CLEAN_NAME --是否清洗描述
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.IS_ALL_INCLUSIVE AS IS_ALL_INCLS_CODE --是否大区外包编码
	  ,CASE WHEN A.IS_ALL_INCLUSIVE = '1' THEN '是'
			ELSE '否'
			 END AS IS_ALL_INCLS_NAME --是否大区外包描述
	  ,A.TMS_PAY_METHOD AS PAY_METH_NAME --支付方式描述
	  ,A.ORDERDATE AS TMS_ORDER_TIME --TMS订单时间
	  ,A.IS_UNLOADING AS IS_UNLOAD_CODE --是否卸货编码
	  ,T18.NAME AS IS_UNLOAD_NAME --是否卸货描述
  FROM SDI.SDI_TMS_ORDER_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.LOADING_METHOD = T1.CODE
   AND T1.TYPENAME = 'TMS_LOADING_METHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.SHIP_METHOD = T2.CODE
   AND T2.TYPENAME = 'TMS_SHIPMETHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T3
    ON A.ORDERTYPE = T3.CODE
   AND T3.TYPENAME = 'TMS_ORDER_TYPE'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T4
    ON A.STATUS = T4.CODE
   AND T4.TYPENAME = 'TMS_ORDER_STATUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T5
    ON A.PRIORITY = T5.CODE
   AND T5.TYPENAME = 'TMS_PRIORITY'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T6
    ON A.LOADING_PROVINCE = T6.STREET_CODE
   AND T6.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T7
    ON A.LOADING_CITY = T7.STREET_CODE
   AND T7.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T8
    ON A.LOADING_COUNTY = T8.STREET_CODE
   AND T8.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T9
    ON A.UNLOADING_PROVINCE = T9.STREET_CODE
   AND T9.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T10
    ON A.UNLOADING_CITY = T10.STREET_CODE
   AND T10.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T11
    ON A.UNLOADING_COUNTY = T11.STREET_CODE
   AND T11.STREET_LEVEL = '3'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T12
    ON T12.PARTNER_KEY = A.CUSTOMER_CODE 
   AND T12.TRADE_TYPE = 'CUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T13
    ON A.SUSR12 = T13.CODE
   AND T13.TYPENAME = 'WAY_UPSTAIRS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T14
    ON A.SUSR14 = T14.CODE
   AND T14.TYPENAME = 'IS_VALUATION'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T15
    ON A.RELEASE_FLAG = T15.CODE
   AND T15.TYPENAME = 'IS_FLAG'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T16
    ON A.SUSPENDREASON = T16.CODE
   AND T16.TYPENAME = 'TMS_SUSPEND_REASON'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T17
    ON A.CLEAN_FLAG = T17.CODE
   AND T17.TYPENAME = 'IS_FLAG'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T18
    ON A.IS_UNLOADING = T18.CODE
   AND T18.TYPENAME = 'IS_OR_NOT_PULL_LIST'
 WHERE A.TENANT_ID = 1
