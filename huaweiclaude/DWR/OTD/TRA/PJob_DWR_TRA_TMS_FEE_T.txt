with invoice as ( ---根据提货单号,行号,批次，数量，公司去匹配创建日期最大的发票号
select row_number() over(PARTITION BY reference_bill_num,reference_item_num,batch_num,abs(actual_billed_qty),company_code ORDER BY create_date||inPUT_TIME DESC) AS RN
,row_number() over(PARTITION BY reference_bill_num,reference_item_num,batch_num,company_code ORDER BY create_date||inPUT_TIME DESC) AS RN1,country,
term_setting_group,product_group,cust_group1,cust_group2,cust_group3,refer,order_type,policy_name,wbs_sale_group,abs(actual_billed_qty) as actual_billed_qty,sale_voucher_num,sale_voucher_item_num,BILL_VOUCHER_NUM,acc_voucher_line_item_num,BILL_TYPE,cust_sal,distribution_channel_code,wbs_element,sale_region_code,sale_org_code,sale_staff,sale_office_code,material_num,so_item_text,prod_level,material_group,company_code,batch_num,factory_code,inv_location,voucher_currency_code,engineer_report_code,inter_trade_clause_loc_1,oss_number,oss_order,material_group1,material_group2,material_group3,material_group4,material_group5,cust_price_group,cust_group,sale_group,assessment_type,item_status,pricing_procedure,alternative_tax_type,account_check_status,cust_refer,reference_bill_num,reference_item_num,sale_unit,purchase_order_type,PS_WBS_ELEMENT
    ,export_country
    ,CLEARANCE_REASON --（清仓原因/促销原因）新增字段
    ,CLEARANCE_REASON_T --（清仓原因/促销原因文本）新增字段
from dwrfin.dwr_gl_invoice_conditional_f 
where period_id_m >= REPLACE(SUBSTR(ADD_MONTHS(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('${PERIOD_ID_M}','01'),'yyyyMMdd'),'yyyy-MM-dd'), -6 ), 1,7),'-','')
and period_id_m <= '${PERIOD_ID_M1}'
--出于存在TMS运费在产生发票之后，所以要往前获取发票区间
)
,gsj_invoice as (
select row_number() over(PARTITION BY reference_bill_num,reference_item_num ORDER BY create_date||inPUT_TIME DESC) AS RN,--取公司间发票
reference_bill_num,reference_item_num,BILL_VOUCHER_NUM,acc_voucher_line_item_num
from dwrfin.dwr_gl_invoice_conditional_f 
where period_id_m >= REPLACE(SUBSTR(ADD_MONTHS(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('${PERIOD_ID_M}','01'),'yyyyMMdd'),'yyyy-MM-dd'), -6 ), 1,7),'-','') 
and period_id_m <= '${PERIOD_ID_M1}'
AND BILL_TYPE in ('ZIV','ZIG')
)
,tmp1 as (
SELECT 
 substr(A.SUBMIT_DATE,1,4)||substr(A.SUBMIT_DATE,6,2) AS calmonth
,substr(A.SUBMIT_DATE,1,4)||substr(A.SUBMIT_DATE,6,2)||substr(A.SUBMIT_DATE,9,2) AS calday 
,substr(regexp_extract(A.SUSR1,'[0-9]*[A-Z]*[0-9]*[0-9]',0),1,10) AS AC_DOC_NR --财务凭证号
,substr(regexp_extract(A.SUSR1,'[0-9]*[A-Z]*[0-9]*[0-9]',0),11,4) AS comp_code  --公司代码 
,A.do_no AS ztmswldh --TMS物料单号
,A.sku_name AS zddms --订单描述
,'TMS' AS INFOprov  --信息提供者
,substr(regexp_extract(A.SUSR1,'[0-9]*[A-Z]*[0-9]*[0-9]',0),1,10) AS zasd_p001_zcbpz --成本凭证
,A.EXTERNNO as refer_doc --参考凭证编号
,A.EXTERNLinENO as refer_itm  --参考凭证项目号
,A.COST_CENTER
,A.CONFIRM_AMOUNTS 
,CASE WHEN A.subject_financial IS NOT NULL OR A.subject_financial <> '' THEN A.subject_financial
 ELSE  CASE 
            WHEN a.order_biz_type = '31' THEN   --城配订单类型
            (CASE 
                WHEN c.func_area IN ('4110', '5502') THEN  
                   (CASE 
                        WHEN a.fee_item_code IN ('T001','T019') THEN '0060701010' --运费、点费、提货站点费用
                        ELSE '0060704000'
                    END)
                ELSE 
                    (CASE WHEN a.fee_item_code IN ('T001','T019') THEN '0054010600' 
                          ELSE '0054010700' 
                      END)
                END)
        ELSE 
            CASE 
                WHEN a.ordertype IN ('04','09','10') AND a.fee_item_code = 'T001' THEN '0060702010'
                WHEN c.func_area NOT IN ('5501','5504') AND a.fee_item_code = 'T001' THEN '0060701010'
                WHEN c.func_area IN ('5501','5504') THEN '0054010600'
                ELSE '0060704000'
            END 
        END
 END AS subject_reconciliation --总账科目
,CASE WHEN A.FEE_ITEM_NAME <> '暂无逻辑，TMS装卸费先置为0' and C.FUNC_AREA in ('5504','5501') AND (A.ORDER_BIZ_TYPE <> '31' OR A.ORDER_BIZ_TYPE IS NULL) THEN A.CONFIRM_AMOUNTS ELSE 0 END  
-CASE WHEN A.FEE_ITEM_NAME = '运费' and A.ORDERTYPE in ('04','09','10') and C.FUNC_AREA in ('5504','5501') AND (A.ORDER_BIZ_TYPE <> '31' OR A.ORDER_BIZ_TYPE IS NULL) then A.CONFIRM_AMOUNTS ELSE 0 END  
-CASE WHEN A.FEE_ITEM_NAME = '平移费' and C.FUNC_AREA in ('5504','5501') AND (A.ORDER_BIZ_TYPE <> '31' OR A.ORDER_BIZ_TYPE IS NULL) then A.CONFIRM_AMOUNTS ELSE 0 END  
+CASE WHEN A.FEE_ITEM_NAME in ('点费','运费','提货站点费用') and C.FUNC_AREA not in ('5502','4110') AND A.ORDER_BIZ_TYPE = '31' then A.CONFIRM_AMOUNTS ELSE 0 END AS ztms_yf --  "TMS运费_计费"
,CASE WHEN A.FEE_ITEM_NAME not in ('点费','运费','提货站点费用') and C.FUNC_AREA not in ('5502','4110') AND A.ORDER_BIZ_TYPE = '31' THEN A.CONFIRM_AMOUNTS ELSE 0 END as ztms_zxf --TMS装卸费_计费
from (SELECT DISTINCT FEE_ID,YT_NO,SUBMIT_DATE,FEE_ITEM_CODE,FEE_ITEM_NAME,CONFIRM_AMOUNTS,QUANTITY,ORDERTYPE,COST_CENTER,SUSR1,DO_NO,EXTERNNO,EXTERNLinENO,SKU_NAME,ORDER_BIZ_TYPE,subject_financial
FROM sdi.sdi_v_bi_tms_do_fee_1005 ) A
JOIN sdi.sdi_csks_1001 C
on A.COST_CENTER  = C.KOSTL and C.KOKRS = 'DP00' AND C.DATBI = '99991231' AND C.MANDT = '800'
WHERE substr(A.SUBMIT_DATE,1,4)||substr(A.SUBMIT_DATE,6,2) BETWEEN '${PERIOD_ID_M}' AND '${PERIOD_ID_M1}' 
UNION all --TMS运费结算
SELECT 
 substr(A.M_REVIEW_DATE,1,4)||substr(A.M_REVIEW_DATE,6,2) AS calmonth
,substr(A.M_REVIEW_DATE,1,4)||substr(A.M_REVIEW_DATE,6,2)||substr(A.M_REVIEW_DATE,9,2) AS calday 
,substr(regexp_extract(A.UDF2,'[0-9]*[A-Z]*[0-9]*[0-9]',0),1,10) AS AC_DOC_NR --财务凭证号
,substr(regexp_extract(A.UDF2,'[0-9]*[A-Z]*[0-9]*[0-9]',0),11,4) AS comp_code  --公司代码 
,A.do_no AS ztmswldh --TMS物料单号
,A.sku_name AS zddms --订单描述
,'TMS-结算' AS INFOprov  --信息提供者
,substr(regexp_extract(A.UDF2,'[0-9]*[A-Z]*[0-9]*[0-9]',0),1,10) AS zasd_p001_zcbpz --成本凭证
,A.EXTERNNO as refer_doc --参考凭证编号
,A.EXTERNLinENO as refer_itm  --参考凭证项目号
,A.COST_CENTER
,A.CONFIRM_AMOUNTS 
,A.subject_reconciliation --总账科目
,CASE WHEN subject_reconciliation = '0054010600' THEN A.CONFIRM_AMOUNTS ELSE 0 END AS ztms_yf --  "TMS运费_计费"
,CASE WHEN subject_reconciliation = '0054010700' THEN A.CONFIRM_AMOUNTS ELSE 0 END as ztms_zxf --TMS装卸费_计费
from (SELECT DISTINCT FEE_ID,M_REVIEW_DATE,FEE_ITEM_CODE,FEE_ITEM_NAME,CONFIRM_AMOUNTS,QUANTITY,ORDERTYPE,COST_CENTER,UDF2,DO_NO,EXTERNNO,EXTERNLinENO,SKU_NAME,ORDER_BIZ_TYPE,SUBJECT_RECONCILIATION
FROM SDI.SDI_V_BI_TMS_DO_FEEDZ_1005 ) A
WHERE substr(A.M_REVIEW_DATE,1,4)||substr(A.M_REVIEW_DATE,6,2) BETWEEN '${PERIOD_ID_M}' AND '${PERIOD_ID_M1}' 
)
