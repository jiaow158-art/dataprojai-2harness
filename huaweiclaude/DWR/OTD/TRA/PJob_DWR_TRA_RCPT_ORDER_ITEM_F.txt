INSERT OVERWRITE TABLE DWROTD.DWR_TRA_RCPT_ORDER_ITEM_F
SELECT A.ID
      ,B.ID AS RCPT_LINE_ID
      ,A.DO_NUM --运单号
	  ,B.DO_LINENO AS DO_LINE_NUM --运单行号
	  ,A.RCPT_NUM --回单号
	  ,A.RCPT_ST_CODE --回单状态编码
	  ,A.RCPT_ST_NAME --回单状态描述
	  ,A.OP_DATE --操作日期
	  ,A.OP_NAME --操作者
	  ,B.ERP_NO AS DELV_NUM --提货单号
	  ,A.APPR_TIME --回单审核时间
	  ,B.PROJECT_PLAN AS PLAN_ITEM_NUM --计划项目号
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM DWROTD.DWR_TRA_RCPT_ORDER_F A
  LEFT JOIN SDI.SDI_TMS_RECEIPT_DETAIL_1005 B
    ON A.RCPT_NUM = B.RECEIPT_NO
;
