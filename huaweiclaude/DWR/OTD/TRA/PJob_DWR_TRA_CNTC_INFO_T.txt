WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWROTD.DWR_TRA_CNTC_INFO_T
SELECT A.ID AS CNTC_ID --合同ID
      ,A.CONTRACT_NO AS CNTC_NUM --合同编号
	  ,A.CONTRACT_NAME AS CNTC_NAME --合同名称
	  ,A.CONTRACT_TYPE AS CNTC_TYPE_CODE --合同类型编码
	  ,T1.DESCR AS CNTC_TYPE_NAME --合同类型描述
	  ,A.CONTRACT_TEXT_ATRR AS CNTC_ATRR_CODE --合同属性编码
	  ,T3.DESCR AS CNTC_ATRR_NAME --合同属性描述
	  ,A.CONTRACT_TEXT_NO AS CNTC_TEXT_NUM --合同文本号
	  ,A.STATUS AS CNTC_ST_CODE --合同状态编码
	  ,T2.DESCR AS CNTC_ST_NAME --合同状态描述
	  ,A.ARAP AS CNTC_CAT_CODE --合同种类编码
	  ,T4.DESCR AS CNTC_CAT_NAME --合同种类描述
	  ,A.VENDOR_CODE AS VEND_CODE --签约对象编码
	  ,A.VENDOR_NAME AS VEND_NAME --签约对象描述
	  ,A.EFFECTIVE_DATE AS EFF_TIME --生效时间
	  ,A.INACTIVE_DATE AS INVAL_TIME --失效时间
	  ,A.CREATE_DT AS CREATE_TIME --创建时间
	  ,A.CREATE_ID AS CREATE_EMP_ID --创建人ID
	  ,T5.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T5.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATE_DT AS UPDATE_TIME --修改时间
	  ,A.UPDATE_ID AS UPDATE_EMP_ID --修改人ID
	  ,T6.EMPLOYEE_NO AS UPDATE_EMP_NUM --修改人工号
	  ,T6.NAME AS UPDATE_EMP_NAME --修改人姓名
	  ,A.SUBMIT_DT AS SUBMT_TIME --提交时间
	  ,A.SUBMIT_ID AS SUBMT_EMP_ID --提交人ID
	  ,T7.EMPLOYEE_NO AS SUBMT_EMP_NUM --提交人工号
	  ,T7.NAME AS SUBMT_EMP_NAME --提交人姓名
	  ,A.APPROVAL_DT AS APPR_TIME --审核时间
	  ,A.APPROVAL_ID AS APPR_EMP_ID --审核人ID
	  ,T8.EMPLOYEE_NO AS APPR_EMP_NUM --审核人工号
	  ,T8.NAME AS APPR_EMP_NAME --审核人姓名
	  ,A.TAX_POINT AS TAX_RATE --税点
	  ,A.OA_FLOW_NO AS OA_RQST_ID --OA流水号
	  ,A.OWNER_KEY AS OWN_CODE --货主编码
	  ,A.TRADE_TYPE AS TRD_TYPE_CODE --贸易对象类型编码
	  ,T9.DESCR AS OWN_NAME --货主名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_BMS_CONTRACT_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.CONTRACT_TYPE = T1.CODE
   AND T1.TYPENAME = 'CONTRACT_TYPE1'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.STATUS = T2.CODE
   AND T2.TYPENAME = 'BMS_APPROVE_STATUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T3
    ON A.CONTRACT_TEXT_ATRR = T3.CODE
   AND T3.TYPENAME = 'CONTRACT_ATRR'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T4
    ON A.ARAP = T4.CODE
   AND T4.TYPENAME = 'BMS_SETTLE_TYPE'
  LEFT JOIN T AS T5
    ON A.CREATE_ID = T5.ID
  LEFT JOIN T AS T6
    ON A.UPDATE_ID = T6.ID
  LEFT JOIN T AS T7
    ON A.SUBMIT_ID = T7.ID
  LEFT JOIN T AS T8
    ON A.APPROVAL_ID = T8.ID
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T9
    ON A.OWNER_KEY = T9.PARTNER_KEY
   AND A.TRADE_TYPE = T9.TRADE_TYPE
   AND T9.TRADE_TYPE IN('STOR','CAR','SUP','WAR','MAN')
 WHERE A.TENANT_ID = 1
