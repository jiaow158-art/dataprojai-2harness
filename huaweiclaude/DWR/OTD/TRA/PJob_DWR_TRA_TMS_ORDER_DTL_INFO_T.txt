WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWROTD.DWR_TRA_TMS_ORDER_DTL_INFO_T
SELECT  A.ID AS ORDER_DTL_ID --ID
       ,A.ORDER_CODE AS ORDER_NUM --TMS订单号
       ,A.ORDER_LINENO AS ORDER_LINE_NUM --TMS订单行号
       ,A.SKUCODE AS MATERIAL_NUM --物料编号
       ,A.SKUNAME AS MATERIAL_NAME --物料名称
       ,A.SKUTYPECODE AS MATERIAL_TYPE_CODE --物料类型编码
       ,CASE WHEN A.SKUTYPECODE = 'CZ' THEN '瓷砖'
	         WHEN A.SKUTYPECODE = 'JJ' THEN '洁具'
  			  END AS MATERIAL_TYPE_NAME --物料类型描述
       ,COALESCE(A.ORIGINALQUANTITY,0) AS ORIG_QTY --原始数量
       ,COALESCE(A.ALLOCATIVEQUANTITY,0) AS ALCT_QTY --分配数量
       ,COALESCE(A.QUANTITY,0) AS QTY --数量
       ,COALESCE(A.VOLUME,0) AS VOL --体积
       ,COALESCE(A.PRICE,0) AS PRC --价格
       ,COALESCE(A.AMOUNT,0) AS AMT --金额
       ,A.QUANTITYUNIT AS QTY_UNIT --单位
       ,A.WEIGHTUNIT AS WT_UNIT --重量单位
       ,A.VOLUMEUNIT AS VOL_UNIT --体积单位
       ,COALESCE(A.RECEIPT_QUANTITY,0) AS RCPT_QTY --接收数量
       ,A.SKUSPECIFICATION AS MATERIAL_CAT_CODE --物料分类编码
	   ,CASE WHEN A.SKUSPECIFICATION = 'CZ' THEN '瓷砖'
	         WHEN A.SKUSPECIFICATION = 'JJ' THEN '洁具'
  			  END AS MATERIAL_CAT_NAME --物料分类描述
       ,A.LOADING_SITE_SHORT_NAME AS DELV_SITE_NAME --提货站点名称
       ,A.UNLOADING_SITE_SHORT_NAME AS RECV_SITE_NAME --收货站点名称
       ,A.ENTRYDT AS CREATE_TIME --创建时间
       ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	   ,T2.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	   ,T2.NAME AS CREATE_EMP_NAME --创建人名称
       ,A.UPDATEDT AS UPDATE_TIME --更新时间
       ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	   ,T3.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	   ,T3.NAME AS UPDATE_EMP_NAME --更新人名称
       ,A.STATUS AS ORDER_ST_CODE --订单状态编码
	   ,T1.NAME AS ORDER_ST_NAME --订单状态描述
       ,COALESCE(A.BOXQTY,0) AS BOX_QTY --箱数
       ,A.GROSS_WEIGHT AS GRS_WT --毛重(kg)
       ,A.NET_WEIGHT AS NET_WT --总净重(kg)
       ,COALESCE(A.ACTUAL_QTY,0) AS ACTL_QTY --实际数量
       ,COALESCE(A.WEIGHT_TOTAL,0) AS TOT_WT --总重
       ,A.OWNER_CODE AS OWN_CODE --货主编码
	   ,A.EXTERNAL_ORDER_CODE AS EXTR_ORDER_NUM --外部订单号	   
       ,A.EXTERNAL_ORDER_LINENO AS EXTR_ORDER_LINE_NUM --外部订单行号
       ,A.ORDER_SOURCE AS ORDER_SRC_CODE --订单来源（系统）
       ,A.FACTORY_CODE --工厂
       ,A.STORAGE_LOCATION AS WAREHOUSE_CODE --库存地点编码
       ,A.SAP_PLAN_NO AS LOAD_PLAN_NUM --装载计划号
       ,A.LOTNO AS LOT_NUM --LOT号
       ,A.COST_CENTER_CODE AS COST_CTR_CODE --成本中心编码
       ,A.SALEORDER_NO AS SALE_ORDER_NUM --销售订单号
       ,A.BILL_NOTES AS REMARK --备注
       ,A.DRIVER_NAME AS DRVR_NAME --司机
       ,A.LICENSE_PLATE_NO AS DRVR_CAR_NUM --司机车牌号
       ,A.LICENSENO AS DRVR_PHONE_NUM --司机电话
       ,A.OBLIGATE_NO AS CUST_NUM --客户编码
	   ,T4.DESCR AS CUST_NAME --客户名称
       ,A.DUMP_NO AS SAP_ORDER_NUM --SAP订单号
       ,A.AREA AS UNIT_AREA --单位面积
       ,A.PROJECT_PLAN AS PROJ_PLAN_NUM --项目计划号
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_ORDERDETAIL_1005 A
 INNER JOIN DWROTD.DWR_TRA_TMS_ORDER_INFO_T B
    ON A.ORDER_CODE = B.ORDER_NUM
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.STATUS = T1.CODE
   AND T1.TYPENAME = 'TMS_ORDER_STATUS'
  LEFT JOIN T AS T2
    ON A.ENTRYID = T2.ID
  LEFT JOIN T AS T3
    ON A.UPDATEID = T3.ID
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T4
    ON T4.PARTNER_KEY = A.OBLIGATE_NO 
   AND T4.TRADE_TYPE = 'CUS'
 WHERE A.TENANT_ID = 1
