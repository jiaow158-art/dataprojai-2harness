set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置成非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

INSERT OVERWRITE TABLE DWROTD.DWR_TRA_TMS_DED_DTL_F PARTITION (PERIOD_ID_M)  
SELECT
A.entrydt AS CREATE_DATE ,--创建日期
C.ABNORMAL_DATE AS ABNORMAL_DATE,--异常发生日期
A.DEDUCTION_NO AS DEDUCTION_NO,--异常扣款单号
B.LINE_NO   AS DEDUCTION_LINE_NO, --异常扣款单行号
A.ABNORMAL_NO AS ABNORMAL_NO,--异常单号
B.ABNORMAL_LINE_NO   AS WB_LINE_NO, --(异常/运单)行号
B.DO_NO  as WB_NO,--TMS运单号
B.ERP_NO as LDNG_NO,--SAP提货单号
C.ORDER_NO as MRG_PLAN_NO,--合并发货计划号
A.STATUS as STATUS,--状态编码
P.NAME as STATUS_NAME ,--状态描述
P1.NAME AS ABNML_STATUS_NAME,--异常状态描述
C.RESPONSIBLE_PARTY_CODE AS RESPONSIBLE_PARTY_CODE,--异常责任方编码
C.RESPONSIBLE_PARTY_NAME AS RESPONSIBLE_PARTY_NAME,--异常责任方描述
B.DEDUCTION_SKUTYPE AS DEDUCTION_SKUTYPE_CODE,--货品类型编码
PSKU.NAME AS DEDUCTION_SKUTYPE_NAME, --货品类型描述
B.SKUCODE AS CRTG_CODE,--货物编码
SKU.sku_name AS CRTG_NAME,  --货物描述
B.FEE_ITEM AS FEE_ITEM_CODE,--费用项目代码 
BFI.FEE_ITEM_NAME AS FEE_ITEM_NAME,--费用项目
B.COST_CENTER AS COST_CENTER_CODE,  --成本中心编码
TCUST.COSTCENTER_NAME  AS COST_CENTER_NAME,--成本中心名称
coalesce(B.DEDUCTION_QTY,0) AS DEDUCTION_QTY_AREA,--免赔(面积/数量)
coalesce(B.COMPENSABLE_QTY_AREA,0) AS COMPENSABLE_QTY_AREA,--应赔(数量/面积)
coalesce(B.COMPENSABLE_AMOUNT,0) AS COMPENSABLE_AMOUNT,--应赔金额
B.DEDUCTION_REASON AS DEDUCTION_REASON, --扣款原因
CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE,
SUBSTR(A.entrydt,1,7) AS PERIOD_ID_M
FROM 
 SDI.SDI_TMS_ABNORMAL_DEDUCTION_1005 A
LEFT  JOIN 
 SDI.SDI_TMS_ABNORMAL_DEDUCTION_DETAIL_1005  B 
 ON  A.ABNORMAL_NO = B.ABNORMAL_NO
LEFT JOIN 
SDI.SDI_TMS_ABNORMAL_1005 C 
ON A.ABNORMAL_NO = C.ABNORMAL_NO
LEFT JOIN SDI.SDI_PLAT_CODE_1005 P ON P.CODE = A.STATUS 
AND P.TYPENAME = 'TMS_ABNORMAL_DEDUCTION_STATUS' 
AND P.TENANT_ID = A.TENANT_ID
LEFT JOIN SDI.SDI_PLAT_CODE_1005 P1 ON P1.CODE = A.STATUS 
AND P1.TYPENAME = 'TMS_EXCEPTION_STATUS' 
AND P1.TENANT_ID = A.TENANT_ID
LEFT JOIN SDI.SDI_PLAT_CODE_1005 PSKU ON PSKU.CODE = B.DEDUCTION_SKUTYPE 
AND PSKU.TYPENAME = 'SKU_TYPE' 
AND PSKU.TENANT_ID = A.TENANT_ID  
LEFT  JOIN 
SDI.SDI_BMS_FEE_ITEM_1005 BFI 
ON (BFI.FEE_ITEM_KEY = B.FEE_ITEM) 
LEFT JOIN SDI.SDI_BMS_COSTCENTER_1005 TCUST 
ON TCUST.COSTCENTER_CODE =  B.COST_CENTER 
left JOIN SDI.SDI_tms_sku_1005 SKU
ON B.SKUCODE = SKU.sku_code
where  
C.ABNORMAL_DATE IS NOT NULL  and C.ABNORMAL_DATE <> ''
