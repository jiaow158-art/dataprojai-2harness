WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWROTD.DWR_TRA_TMS_PLAN_INFO_T
SELECT A.ID AS TRA_PLAN_ID --发货计划ID
      ,A.PLAN_NO AS TRA_PLAN_NUM --TMS计划号
	  ,A.PLAN_NO_MERGE AS MRG_TRA_PLAN_NUM --合并计划号
	  ,A.CARRIER_CODE AS CARR_NUM --承运商编码
	  ,T1.DESCR AS CARR_NAME --承运商名称
	  ,A.STATUS AS PLAN_ST_CODE --承运商状态编码
	  ,T2.DESCR AS PLAN_ST_NAME --承运商状态描述
	  ,A.WAR_WORK_PROGRESS AS WHS_ST_CODE --仓库状态编码
	  ,T3.DESCR AS WHS_ST_NAME --仓库状态描述
	  ,A.SHIP_LINE AS SHIP_LINE_CODE --线路编码
	  ,T4.LINE_NAME AS SHIP_LINE_NAME --线路名称
	  ,A.WAREHOUSERKEY AS WH_LOC --库存地点
	  ,A.CAR_TYPE --车型
	  ,A.SPECIAL_CAR AS IS_SPEC_CAR_CODE --是否专车编码
	  ,CASE WHEN A.SPECIAL_CAR = '0' THEN '否' ELSE '是' END AS IS_SPEC_CAR_NAME --是否专车描述
	  ,B.SHIP_METHOD AS SHIP_METH_CODE --运输方式编码
	  ,T5.NAME AS SHIP_METH_NAME --运输方式描述
	  ,A.OPTIMAL_TRANSPORT AS OPTM_SHIP_METH_CODE --最优运输方式编码
	  ,T6.NAME AS OPTM_SHIP_METH_NAME --最优运输方式描述
	  ,A.OPTIMAL_CARRIERS AS OPTM_CARR_NUM --最优承运商编码
	  ,T7.DESCR AS OPTM_CARR_NAME --最优承运商描述
	  ,B.SHIPPER_CODE AS LTL_RSN_CODE --零担发运原因编码
	  ,T8.NAME AS LTL_RSN_NAME --零担发运原因描述
	  ,B.TMS_PAY_METHOD AS PAY_METH_CODE --运费支付方式
	  ,C.DRIVER_NAME AS DRVR_NAME --司机
	  ,C.LICENSE_PLATE_NO AS DRVR_CAR_NUM --司机车牌号
	  ,C.LICENSENO AS DRVR_CARD_NUM --司机证件号
	  ,C.DRIVER_PHONE AS DRVR_PHONE_NUM --司机电话
	  ,D.TRACE_DT AS DSPCH_TIME --派车时间
	  ,B.TOTAL_BOXES AS TOT_BOX_QTY --总箱数
	  ,COALESCE(C.QUANTITY,0) AS TOT_QTY --总数量
	  ,COALESCE(C.VOLUME,0) AS TOT_VOL --总体积
	  ,COALESCE(C.GROSS_WEIGHT,0) AS TOT_GRS_WT --总毛重(kg)
	  ,A.PUBLISH_DATE AS PBLSH_TIME --发布承运商时间
	  ,A.PLAN_DELIVERY_DATE AS PLAN_DELV_TIME --计划提货时间
	  ,A.DELIVERY_DATE AS DELV_DATE --配送日期
	  ,B.LOADING_SITE_CODE AS DELV_SITE_CODE --提货站点编码
	  ,B.LOADING_SITE_NAME AS DELV_SITE_NAME --提货站点名称
	  ,B.LOADING_ADDR AS DELV_ADDR --提货详细地址
	  ,B.UNLOADING_SITE_CODE AS RECV_SITE_CODE --收货站点编码
	  ,B.UNLOADING_SITE_NAME AS RECV_SITE_NAME --收货站点名称
	  ,B.UNLOADING_ADDR AS RECV_ADDR --收货详细地址
	  ,COALESCE(A.CZ_PRICE,0) AS CERM_DELV_PRC --瓷砖运费单价
	  ,COALESCE(A.JJ_PRICE,0) AS SW_DELV_PRC --洁具运费单价
	  ,COALESCE(A.TOTAL_PRICE,0) AS TOT_DELV_AMT --运费总价
	  ,COALESCE(A.THF_PRICE,0) AS DELV_AMT --提货费总价
	  ,COALESCE(A.SHF_PRICE,0) AS DELV1_AMT --送货费总价
	  ,COALESCE(A.XHF_PRICE,0) AS UNLD_DELV_AMT --卸货费总价
	  ,COALESCE(A.DCF_PRICE,0) AS BC_DELV_AMT --倒车费总价
	  ,COALESCE(A.MJF_PRICE,0) AS WF_DELV_AMT --木架费总价
	  ,COALESCE(A.OTHER_PRICE,0) AS OTH_DELV_AMT --其他费总价
	  ,COALESCE(A.MTF_PRICE,0) AS WP_DELV_AMT --木托费总价
	  ,COALESCE(A.PYF_PRICE,0) AS SHFT_DELV_AMT --平移费总价
	  ,COALESCE(A.BTF_PRICE,0) AS LAD_DELV_AMT --步梯楼层费总价
	  ,COALESCE(A.DTF_PRICE,0) AS ELV_DELV_AMT --电梯楼层费总价
	  ,COALESCE(A.LCF_PRICE,0) AS FLOOR_DELV_AMT --楼层费总价
	  ,COALESCE(A.SLJCF_PRICE,0) AS UPST_FDTN_DELV_AMT --上楼基础费总价
	  ,COALESCE(A.SLCZF_PRICE,0) AS UPST_OVER_DELV_AMT --上楼超重费总价
	  ,COALESCE(A.VALUATION_PRICE,0) AS VALUT_DELV_AMT --保价费总价
	  ,COALESCE(A.BYF_PRICE,0) AS TRNSP_DELV_AMT --搬运费总价
	  ,COALESCE(A.HDF_PRICE,0) AS RCPT_DELV_AMT --回单费总价
	  ,COALESCE(A.ZTF_PRICE,0) AS SLF_DELV_AMT --自提费总价
	  ,COALESCE(A.SHDDF_PRICE,0) AS MORE_DELV_AMT --送货多点费
	  ,COALESCE(A.DELIVER_DISTANCE,0) AS DELV_DSTC --两点送货距离(KM)
	  ,COALESCE(A.TOTAL_PRICE,0) 
	  + COALESCE(A.THF_PRICE,0)
	  + COALESCE(A.SHF_PRICE,0)
	  + COALESCE(A.XHF_PRICE,0)
	  + COALESCE(A.DCF_PRICE,0)
	  + COALESCE(A.MJF_PRICE,0)
	  + COALESCE(A.MTF_PRICE,0)
	  + COALESCE(A.HDF_PRICE,0)
	  + COALESCE(A.OTHER_PRICE,0)
	  + COALESCE(A.SLJCF_PRICE,0)
	  + COALESCE(A.SLCZF_PRICE,0)
	  + COALESCE(A.VALUATION_PRICE,0)
	  + COALESCE(A.LCF_PRICE,0)
	  + COALESCE(A.PYF_PRICE,0)
	  + COALESCE(A.BTF_PRICE,0)
	  + COALESCE(A.DTF_PRICE,0)
	  + COALESCE(A.SHDDF_PRICE,0) AS TOT_DELV_FEE --费用总额
	  ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	  ,T10.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T10.NAME AS CREATE_EMP_NAME --创建人名称
	  ,A.ENTRYDT AS CREATE_TIME --创建时间
	  ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	  ,T11.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T11.NAME AS DATE_EMP_NAME --更新人名称
	  ,A.UPDATEDT AS UPDATE_TIME --更新时间
	  ,A.NOTES AS CARR_REMARK --承运商备注
	  ,A.WAREHOUSE_NOTES AS WH_REMARK --仓库备注
	  ,A.SHARED_MANAGER_WH_NOTES AS SHR_MGR_RFUS_RSN --共享仓经理驳回原因
	  ,A.PRICE_SYSTEM AS PRC_SYS --价格体系
	  ,A.CUS_PROJECT AS CUST_PROJ --客户项目
	  ,A.CUSTOMER_CODE AS CUST_NUM --客户编码
	  ,T12.DESCR AS CUST_NAME --客户名称
	  ,A.MOTHER_CUSTOMER_CODE AS MTH_CUST_NUM --母客编
	  ,T13.DESCR AS MTH_CUST_NAME --母客户名称
	  ,A.CONFIRM_DATE AS CNFRM_TIME --确认时间
	  ,A.BIZ_TYPE AS BUSI_TYPE_CODE --业务类型编码
	  ,T14.NAME AS BUSI_TYPE_NAME --业务类型描述
	  ,A.SKU_TYPE AS MATERIAL_TYPE_CODE --物料类型编码
	  ,CASE WHEN A.SKU_TYPE = 'CZ' THEN '瓷砖'
	        WHEN A.SKU_TYPE = 'JJ' THEN '洁具'
  			 END AS MATERIAL_TYPE_NAME --物料类型描述
	  ,A.SHIP_METHOD AS FEE_SHIP_METH_CODE --计费运输方式编码
	  ,T15.NAME AS FEE_SHIP_METH_NAME --计费运输方式描述
	  ,A.IS_BIG AS IS_BIG_CODE --是否干线大板编码
	  ,CASE WHEN A.IS_BIG = 1 THEN '是' ELSE '否' END IS_BIG_NAME --是否干线大板描述
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_TRANSPORT_PLAN_1005 A
  LEFT JOIN SDI.SDI_TMS_DO_1005 B
    ON A.PLAN_NO = B.PLAN_NO
  LEFT JOIN (SELECT  DO_NO
                    ,MIN(TDD.DRIVER_NAME) AS DRIVER_NAME
                    ,MIN(TDD.LICENSE_PLATE_NO) AS LICENSE_PLATE_NO
                    ,MIN(TDD.LICENSENO) AS LICENSENO
                    ,MIN(TDD.DRIVER_PHONE) AS DRIVER_PHONE
					,SUM(COALESCE(QUANTITY,0)) AS QUANTITY
					,SUM(COALESCE(VOLUME,0)) AS VOLUME
					,SUM(COALESCE(GROSS_WEIGHT,0)) AS GROSS_WEIGHT
               FROM SDI.SDI_TMS_DO_DETAIL_1005 TDD
			  GROUP BY DO_NO
            ) C
	ON B.DO_NO = C.DO_NO
  LEFT JOIN (SELECT DO_NO
                   ,TRACE_DT
				   ,ROW_NUMBER() OVER(PARTITION BY DO_NO ORDER BY ID ASC) AS RN
               FROM SDI.SDI_TMS_TRACE_LOG_1005
              WHERE TRANS_NODE = '35'
                AND TENANT_ID = 1
            ) D
	ON B.DO_NO = D.DO_NO
   AND D.RN = 1
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T1
    ON A.CARRIER_CODE = T1.PARTNER_KEY
   AND T1.TRADE_TYPE = 'CAR'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.STATUS = T2.CODE
   AND T2.TYPENAME = 'TRANSPORA.ANSTATUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T3
    ON A.WAR_WORK_PROGRESS = T3.CODE
   AND T3.TYPENAME = 'WAR_WORK_PROGRESS'
  LEFT JOIN SDI.SDI_TMS_LINE_1005 T4
    ON A.SHIP_LINE = T4.LINE_CODE
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T5
    ON B.SHIP_METHOD = T5.CODE
   AND T5.TYPENAME = 'TMS_SHIPMETHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T6
    ON A.OPTIMAL_TRANSPORT = T6.CODE
   AND T6.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T7
    ON A.OPTIMAL_CARRIERS = T7.PARTNER_KEY
   AND T7.TRADE_TYPE = 'CAR'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T8
    ON B.SHIPPER_CODE = T8.CODE
   AND T8.TYPENAME = 'LDFY_CAUSE'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T9
    ON B.TMS_PAY_METHOD = T9.CODE
   AND T9.TYPENAME = 'LDFY_CAUSE'
  LEFT JOIN T AS T10
    ON A.ENTRYID = T10.ID
  LEFT JOIN T AS T11
    ON A.UPDATEID = T11.ID
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T12
    ON T12.PARTNER_KEY = A.CUSTOMER_CODE 
   AND T12.TRADE_TYPE = 'CUS'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T13
    ON T13.PARTNER_KEY = A.MOTHER_CUSTOMER_CODE 
   AND T13.TRADE_TYPE = 'CUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T14
    ON A.BIZ_TYPE = T14.CODE
   AND T14.TYPENAME = 'BMS_QUOTE_ORDER_TYPE'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T15
    ON A.SHIP_METHOD = T15.CODE
   AND T15.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
