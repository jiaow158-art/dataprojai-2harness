INSERT OVERWRITE TABLE DWROTD.DWR_TRA_RCPT_ORDER_F
SELECT A.ID
      ,A.DO_NO AS DO_NUM --运单号
      ,A.TENANT_ID AS TENAT_ID --租户ID
	  ,A.RECEIPT_NO AS RCPT_NUM --回单号
	  ,A.STATUS AS RCPT_ST_CODE --回单状态编码
	  ,B.DESCR AS RCPT_ST_NAME --回单状态描述
	  ,A.OPER_DATETIME AS OP_DATE --操作日期
	  ,A.`OPERATOR` AS OP_NAME --操作者
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.ERP_NO AS DELV_NUM --提货单号
	  ,CASE WHEN A.STATUS <> '01' THEN A.UPDATEDT END AS APPR_TIME --回单审核时间
  FROM SDI.SDI_TMS_RECEIPT_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 B
    ON A.STATUS = B.CODE
   AND B.TYPENAME = 'RECEIPTORDERSTATUS'
 WHERE A.TENANT_ID = 1
