WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWROTD.DWR_TRA_PLLT_INFO_T
SELECT A.ID AS PLLT_TRA_ID --托盘出入库ID
      ,A.PROJECT_PLAN AS PROJ_PLAN_NUM --项目计划号
	  ,A.SAP_NO AS DELV_NUM --提货单号
	  ,A.DO_NO AS DO_NUM --运单号
	  ,C.WAREHOUSERKEY AS WH_CODE --仓库编码
	  ,T1.DESCR AS WH_NAME --仓库名称
	  ,A.PALLET_CODE AS PLLT_CODE --托盘编码
	  ,A.PALLET_NAME AS PLLT_NAME --托盘名称
	  ,A.STOCK_TYPE AS STOCK_TYPE_CODE --出入库类型编码
	  ,CASE WHEN A.STOCK_TYPE = '0' THEN '出库'
            WHEN A.STOCK_TYPE = '1' THEN '入库'
             END AS STOCK_TYPE_NAME --出入库类型名称
	  ,COALESCE(A.QUANTITY,0) AS QTY --托盘数量
	  ,A.VOUCHER AS VCHR_NUM --出库/入库凭证
	  ,A.FEE_VOUCHER AS SALE_VCHR_NUM --上架/销售凭证
	  ,COALESCE(A.MONEY,0) AS AMT --扣费金额
	  ,A.ENTRYDT AS CREATE_TIME --操作时间
	  ,A.ENTRYNAME AS CREATE_EMP_NUM --处理人工号
	  ,T2.NAME AS CREATE_EMP_NAME --处理人名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_DO_DETAIL_PALLET_1005 A
  LEFT JOIN SDI.SDI_TMS_DO_1005 B
    ON A.DO_NO = B.DO_NO
  LEFT JOIN SDI.SDI_TMS_TRANSPORT_PLAN_1005 C
    ON B.PLAN_NO = C.PLAN_NO
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T1
    ON T1.PARTNER_KEY = C.WAREHOUSERKEY
   AND T1.TRADE_TYPE = 'WAR'
  LEFT JOIN T AS T2
    ON A.ENTRYID = T2.ID
 WHERE A.TENANT_ID = 1
