INSERT OVERWRITE TABLE DWROTD.DWR_TRA_BUSI_RCPT_ORDER_F
SELECT A.ID
      ,A.RECEIPT_NO AS RCPT_NUM --回单号
      ,A.DO_NO AS DO_NUM --运单号
      ,A.OPER_DATETIME AS OP_TIME --操作时间
      ,COALESCE(C.DESCR,A.OPERATOR) AS OP_EMP_NAME --操作人
      ,A.ERP_NO AS DELV_NUM --提货单号
      ,A.OBLIGATE_NO AS OBL_NUM --预留单号
      ,A.PROJECT_PLAN AS PROJ_PLAN_NUM --项目计划号
      ,A.STATUS AS RCPT_ST_CODE --回单状态编码
      ,B.DESCR AS RCPT_ST_NAME --回单状态描述
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_RECEIPT_BUSINESS_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 B
    ON A.STATUS = B.CODE
   AND B.TYPENAME = 'RECEIPTORDERSTATUS'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 C
    ON C.ID = A.OPERATOR 
   AND C.TRADE_TYPE = 'CUS'
