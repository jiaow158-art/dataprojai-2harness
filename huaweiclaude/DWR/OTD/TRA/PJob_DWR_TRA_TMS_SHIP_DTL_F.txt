WITH T AS(
SELECT A.DO_NO
      ,MAX(CASE WHEN B.DESCR = '已调度' THEN A.TRACE_DT END) AS HNDL_TIME --处理时间
	  ,MAX(CASE WHEN B.DESCR = '已发布承运商' THEN A.TRACE_DT END) AS PBLSH_TIME --发布时间
	  ,MAX(CASE WHEN B.DESCR = '承运商派车' THEN A.TRACE_DT END) AS DSPCH_TIME --派车时间
	  ,MAX(CASE WHEN B.DESCR = '已送达' THEN A.TRACE_DT END) AS SIGN_TIME --签收时间
	  ,MAX(CASE WHEN B.DESCR = '仓库接单' THEN A.TRACE_DT END) AS RECV_TIME --仓库接单时间
	  ,MAX(CASE WHEN B.DESCR = '已提货' THEN A.TRACE_DT END) AS DELV_TIME --提货时间
  FROM SDI.SDI_TMS_TRACE_LOG_1005 A
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 B
    ON A.TRANS_NODE = B.CODE
   AND B.TYPENAME = 'TRACENODE_TYPE'
 GROUP BY A.DO_NO
), FEE AS(SELECT
				 BILL_ORDER_NO
				,SOURCE_ORDER_LINENO
				,SUM(BILL_TOTAL) AS BILL_TOTAL --总运费
				,SUM(CASE WHEN FEE_NO = 'T001' THEN BILL_TOTAL
						  ELSE 0
						  END
				   ) AS FREIGHT_FEE --运费
				,SUM(CASE WHEN FEE_NO = 'T002' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS UNLOADING_FEE --卸货费
				,SUM(CASE WHEN FEE_NO = 'T003' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS SEND_FEE --送货费
				,SUM(CASE WHEN FEE_NO = 'T004' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS PARKING_FEE --倒车费
				,SUM(CASE WHEN FEE_NO = 'T005' THEN BILL_TOTAL
					      ELSE 0
						   END
					) AS LOADING_FEE --提货费
				,SUM(CASE WHEN FEE_NO = 'T006' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS OTHER_FEE --其他费用
				,SUM(CASE WHEN FEE_NO = 'T016' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS LADDER_FEE --步梯楼层费
				,SUM(CASE WHEN FEE_NO = 'T019' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS SITE_FEE --点费
				,SUM(CASE WHEN FEE_NO = 'T017' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS ELEVATOR_FEE --电梯楼层费
				,SUM(CASE WHEN FEE_NO = 'T018' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS MOVE_FEE --平移费
				,SUM(CASE WHEN FEE_NO = 'T020' THEN BILL_TOTAL
						  ELSE 0
						   END
					) AS HANDLING_FEE --搬运费
		   FROM SDI.SDI_BMS_BILL_RESULT_TMS_1005
		  GROUP BY BILL_ORDER_NO
				  ,SOURCE_ORDER_LINENO
			)
