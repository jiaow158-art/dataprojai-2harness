--SDI_APS_PRO_ORDER_1007 到 DWR_PS_APS_PRO_ORDER_F
--生产订单
WITH T1 AS(
SELECT APO.PRODUCT_ORDER_ID
      ,SUM(COALESCE(APT.CLOSE_QTY,0)) AS CLOSE_QTY
	  ,SUM(CASE WHEN APT.CLOSE_FLAG = 1 THEN COALESCE(APTD.PLAN_AMOUNT,0) ELSE 0 END) AS CANCEL_QTY
  FROM SDI.SDI_APS_PLAN_TASK_DTL_1007 APTD  
 INNER JOIN SDI.SDI_APS_PLAN_TASK_1007 APT
    ON APTD.PLAN_TASK_ID = APT.ID  
 INNER JOIN SDI.SDI_APS_PROTECH_ORDER_1007 APO
    ON APTD.PROCESS_TECH_TASK_ID = APO.ID
   AND APO.IS_PHANTOM = 0
 GROUP BY APO.PRODUCT_ORDER_ID
)

    ON T1.PRODUCT_ORDER_ID = DWI.ID

  DWI.DEL_FLAG = 'N';
