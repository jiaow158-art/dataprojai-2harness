--DWI_PS_APS_PROTECH_SAM_T 到 DWR_PS_APS_PROTECH_SAM_F
--制程工艺定额
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_APS_PROTECH_SAM_F
SELECT
  DWI.ID AS ID,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.SRC_MATERIAL_ID AS SRC_MATERIAL_ID,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.PRIORITY AS PRIORITY,
  DWI.PROCESS_TECH_ID AS PROCESS_TECH_ID,
  DWI.SAM AS SAM,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.TRANSFER_TIME AS TRANSFER_TIME,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.WIP_RESOURCE_ID AS WIP_RESOURCE_ID,
  DWI.STATE AS STATE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_APS_PROTECH_SAM_T DWI
  INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
