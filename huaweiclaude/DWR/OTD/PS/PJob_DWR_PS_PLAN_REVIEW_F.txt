  OVERWRITE TABLE DWROTD.DWR_PS_PLAN_REVIEW_F
SELECT
  DWI.ID AS ID,
  DWI.COLORS AS COLORS,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.CUSTOM_FLAG AS CUSTOM_FLAG,
  DWI.CUST_ID AS CUST_ID,
  DWI.CUST_NUM AS CUST_NUM,
  CUST.CUST_NAME AS CUST_NAME,
  DWI.CUST_NAME AS SRC_CUST_NAME,
  --客户名称
  DWI.CUSTOMIZED_REQUIREMENTS AS CUSTOMIZED_REQUIREMENTS,
  DWI.DATA_RESOURCE AS DATA_RESOURCE,
  DWI.DELIVERY_MODE AS DELIVERY_MODE,
  DWI.DELIVERY_TIME AS DELIVERY_TIME,
  DWI.DEMAND_DELIVERY_DATE AS DEMAND_DELIVERY_DATE,
  DWI.END_TIME AS END_TIME,
  DWI.ESTIMATED_DELIVERY_TIME AS ESTIMATED_DELIVERY_TIME,
  DWI.FACTORY_NAME AS FACTORY_NAME,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.ITEM_CATEGORY AS ITEM_CATEGORY,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.MATERIAL_NAME AS SRC_MATERIAL_NAME,
  DWI.LINE_NO AS LINE_NO,
  DWI.NET_DEMAND_QTY AS NET_DEMAND_QTY,
  DWI.OUTPUT_TIME AS OUTPUT_TIME,
  DWI.OVERDUE_DATE AS OVERDUE_DATE,
  DWI.PACKAGING_CODE AS PACKAGING_CODE,
  DWI.PLAN_QTY AS PLAN_QTY,
  DWI.PLAN_TASK_ID AS PLAN_TASK_ID,
  DWI.PROCESSED_PRODUCT AS PROCESSED_PRODUCT,
  DWI.PROCESSED_QTY AS PROCESSED_QTY,
  DWI.PROCESSING_TIME AS PROCESSING_TIME,
  DWI.PRODUCT_ORDER_CODE AS PRODUCT_ORDER_CODE,
  DWI.PRODUCT_ORDER_ID AS PRODUCT_ORDER_ID,
  DWI.PRODUCT_ORDER_QTY AS PRODUCT_ORDER_QTY,
  DWI.REVIEW_RESULT AS REVIEW_RESULT,
  DWI.REVIEW_STATUS AS REVIEW_STATUS,
  DWI.SALE_ORDER_CODE AS SALE_ORDER_CODE,
  DWI.SALE_ORDER_DETAIL_ID AS SALE_ORDER_DETAIL_ID,
  DWI.SHIPPING_MODE AS SHIPPING_MODE,
  DWI.SPECIAL_PACKAGING AS SPECIAL_PACKAGING,
  DWI.SPECIAL_REQUIREMENT AS SPECIAL_REQUIREMENT,
  DWI.SPECIAL_THICKNESS AS SPECIAL_THICKNESS,
  DWI.START_TIME AS START_TIME,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.UPLOAD_OSS AS UPLOAD_OSS,
  DWI.UPLOAD_OSS_DATE AS UPLOAD_OSS_DATE,
  DWI.WIP_RESOURCE_ID AS WIP_RESOURCE_ID,
  DWI.REVIEW_TIME AS REVIEW_TIME,
  DWI.REVIEWER_ID AS REVIEWER_ID,
  DWI.SALE_ORG AS SALE_ORG,
  DWI.SALE_DEPT AS SALE_DEPT,
  DWI.SALE_DEPT_NAME AS SALE_DEPT_NAME,
  DWI.NOT_MERGE_FLAG AS NOT_MERGE_FLAG,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_PLAN_REVIEW_T DWI
  INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON DWI.CUST_ID = CUST.CUST_ID
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
  INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
