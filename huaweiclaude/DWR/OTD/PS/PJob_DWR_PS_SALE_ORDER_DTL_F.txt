SELECT A.SALE_ORDER_DETAIL_ID
      ,B.BEFORE
	  ,ROW_NUMBER() OVER(PARTITION BY A.SALE_ORDER_DETAIL_ID ORDER BY B.CREATE_DATE DESC) AS RN
  FROM SDI.SDI_DELIVERY_UPDATE_RECORD_1007 A
 INNER JOIN SDI.SDI_DELIVERY_UPDATE_RECORD_DETAIL_1007 B
    ON B.DELIVERY_UPDATE_RECORD_ID = A.ID
)

      ,SRC.SEQUENCE_NO AS SEQUENCE_NO

	        WHEN SRC.EMERGENCY_FLAG = 1 THEN '是'

	        WHEN SRC.SELF_FOREIGN_FLAG = 1 THEN '外协'

	        WHEN SRC.EVALUATION_RESULT = 1 THEN '拒绝'

	  ,CASE WHEN SRC.CUSTOM_FLAG = 0 THEN '否'
	        WHEN SRC.CUSTOM_FLAG = 1 THEN '是'
			 END AS IS_CSTM_NAME --是否定制描述
	  ,CASE WHEN SRC.IS_SUITABLE_PRICE = 0 THEN '否'
	        WHEN SRC.IS_SUITABLE_PRICE = 1 THEN '是'
			 END AS IS_ADD_PRC_NAME --是否提价描述
	  ,SRC.TRADEMARK AS SPEC_LOGO --特殊砖底商标
	  ,SRC.SPECIAL_SIZE AS SPEC_SIZE --特殊尺寸
	  ,SRC.PREPOSITION_DATE AS REP_POS_DATE --前置时间
	  ,A.SALE_DEPT_NAME --组织架构部门
	  ,A.SALE_ORG --销售组织
	  ,A.SALE_DEPT --销售部门
	  ,A.SALE_GROUP AS SALE_GRP_CODE --销售组编码
	  ,A.SALE_GRP_NAME --销售组名称
	  ,A.APPL_EMP_NAME --申请人姓名
      ,A.SALE_EMP_NAME --业务员姓名
	  ,SRC.CATEGORY AS MATERIAL_CAT_NAME --品类
	  ,SRC.CHANGED_REMARK AS CHNG_REMARK --变更备注
	  ,SRC.PLANNED_DELIVERY_TIME_TEXT AS PLAN_DELV_TIME --计划交付时间
	  ,T2.BEFORE AS BEF_PLAN_DELV_TIME --上一次计划交付时间
  FROM DWROTD.DWR_PS_SALE_ORDER_F A
  LEFT JOIN SDI.SDI_SALE_ORDER_DTL_1007 SRC
    ON A.ID = SRC.SALE_ORDER_ID
  LEFT JOIN (
    SELECT
       M.ID
      ,MAT.MATERIAL_ID--物料ID
      ,MAT.MATERIAL_NUM --物料编码
	  ,MAT.MATERIAL_NAME --物料描述
	  ,MAT.DIMENSION --规格
    FROM
      SDI.SDI_ITEM_1007 M --物料编码表
      INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
      ON M.CODE = MAT.MATERIAL_NUM
      AND M.IS_PHANTOM = 0 --取有效属性数据
      AND MAT.END_DATE = '4712-12-31'
      AND MAT.DEL_FLAG = 'N'
      AND MAT.GROUP_NUM = '800'
  ) TMP ON SRC.ITEM_ID = TMP.ID
 LEFT JOIN DWROTD.DWR_PS_RES_WIP_SCHE_F T1 --Y
   ON SRC.PRODUCT_AREA_ID = T1.ID
 LEFT JOIN T1 AS T2
   ON T2.SALE_ORDER_DETAIL_ID = SRC.ID
  AND T2.RN = 1
