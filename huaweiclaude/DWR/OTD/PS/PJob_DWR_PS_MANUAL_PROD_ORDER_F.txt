--DWI_PS_MANUAL_PROD_ORDER_T 到 DWR_PS_MANUAL_PROD_ORDER_F
--手工生产订单
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_MANUAL_PROD_ORDER_F
SELECT
  DWI.ID AS ID,
  DWI.ORDER_NUM AS ORDER_NUM,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.SRC_CUST_ID AS SRC_CUST_ID,
  DWI.CUST_ID AS CUST_ID,
  DWI.CUST_NUM AS CUST_NUM,
  CUST.CUST_NAME AS CUST_NAME,
  --客户名称
  DWI.DATA_SOURCE AS DATA_SOURCE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_CUSTOM AS IS_CUSTOM,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.IS_SPLIT AS IS_SPLIT,
  DWI.IS_WHOLE AS IS_WHOLE,
  DWI.SRC_MATERIAL_ID AS SRC_MATERIAL_ID,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.ORDER_DATE AS ORDER_DATE,
  DWI.ORDER_STATE AS ORDER_STATE,
  DWI.ORDER_TYPE_SET_ID AS ORDER_TYPE_SET_ID,
  DWI.PROMISE_DELIVERY AS PROMISE_DELIVERY,
  DWI.ORDER_QTY AS ORDER_QTY,
  DWI.RELEVANT_CODE AS RELEVANT_CODE,
  DWI.REQUIRE_DELIVERY AS REQUIRE_DELIVERY,
  DWI.SPLIT_PROD_ORDER_ID AS SPLIT_PROD_ORDER_ID,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.UPLOAD_STATE AS UPLOAD_STATE,
  DWI.REMARK AS REMARK,
  DWI.RELEVANT_ORDER_NO AS RELEVANT_ORDER_NO,
  DWI.IS_SPECIAL_STOCK AS IS_SPECIAL_STOCK,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  RE.CODE AS CREATE_EMP_NUM, --创建人工号
  RE.NAME AS CREATE_EMP_NAME, --创建人名称
  MAT.PRODUCT_LEVEL_CODE,--产品层次编码
  MAT.PRODUCT_LEVEL_NAME,--产品层次名称
  MAT.PRODUCT_LEVEL1_CODE,--产品层次1编码
  MAT.PRODUCT_LEVEL1_NAME,--产品层次1名称
  MAT.PRODUCT_LEVEL2_CODE,--产品层次2编码
  MAT.PRODUCT_LEVEL2_NAME,--产品层次2名称
  MAT.PRODUCT_LEVEL3_CODE,--产品层次3编码
  MAT.PRODUCT_LEVEL3_NAME,--产品层次3名称
  MAT.PRODUCT_LEVEL4_CODE,--产品层次4编码
  MAT.PRODUCT_LEVEL4_NAME,--产品层次4名称
  MAT.PRODUCT_LEVEL5_CODE,--产品层次5编码
  MAT.PRODUCT_LEVEL5_NAME,--产品层次5名称
  MAT.PRODUCT_LEVEL6_CODE,--产品层次6编码
  MAT.PRODUCT_LEVEL6_NAME,--产品层次6名称
  MAT.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE,--产品渠道
  MAT.RESERVED_FIELD1_NAME AS PRODUCT_CHANNEL_NAME,--产品渠道名称
  MAT.RESERVED_FIELD_6 AS PRODUCT_GRADE_CODE, --产品等级
  MAT.RESERVED_FIELD6_NAME AS PRODUCT_GRADE_NAME, --产品等级名称
  MAT.LENGTH_1, --长度
  MAT.WIDTH, --宽度
  DWI.ORDER_QTY * MAT.LENGTH_1 * MAT.WIDTH / 1000000 AS ORDER_AREA --订单面积(平方米)
FROM DWIOTD.DWI_PS_MANUAL_PROD_ORDER_T DWI
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON DWI.CUST_ID = CUST.CUST_ID
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_NUM = MAT.MATERIAL_NUM
  AND MAT.GROUP_NUM = '800'
  LEFT JOIN SDI.SDI_RES_EMP_1007 RE --人员表
  ON RE.ID = DWI.CREATE_BY
  AND RE.DEL_FLAG = 'N'
WHERE
  DWI.DEL_FLAG = 'N';
