--DWI_PS_DAILY_SCHEDULE_T 到 DWR_PS_DAILY_SCHEDULE_F
--生产日进度
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_DAILY_SCHEDULE_F
SELECT
  DWI.ID AS ID,
  DWI.BATCH AS BATCH,
  DWI.COST_HOUR AS COST_HOUR,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.DATA_SOURCE AS DATA_SOURCE,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.INVENTORY_LOCATION AS INVENTORY_LOCATION,
  DWI.PLAN_TASK_ID AS PLAN_TASK_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.FACTORY_ID AS FACTORY_ID,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.PROCESS_TECH_ORDER_ID AS PROCESS_TECH_ORDER_ID,
  DWI.PRODUCTION_DATE AS PRODUCTION_DATE,
  DWI.QTY AS QTY,
  DWI.SAM AS SAM,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.WIP_RESOURCE_ID AS WIP_RESOURCE_ID,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_DAILY_SCHEDULE_T DWI
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
