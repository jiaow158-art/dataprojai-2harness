WITH T AS( --累计出库数量
SELECT A.MAT_PSPNR AS WBS_PSPNR_NUM --WBS元素号
	  ,B.POSID AS WBS_NUM --WBS号
	  ,SUM(CASE WHEN A.BWART = '602' THEN 0 - COALESCE(A.MENGE,0)
	            WHEN A.BWART = '653' THEN 0 - COALESCE(A.MENGE,0)
			    WHEN A.BWART = '688' THEN 0 - COALESCE(A.MENGE,0)
			    ELSE COALESCE(A.MENGE,0)
			     END) AS OUT_STOCK_QTY --出库数量
  FROM SDI.SDI_MATDOC_1001 AS A
  LEFT JOIN SDI.SDI_PRPS_1001 B
    ON A.MANDT = B.MANDT
   AND A.MAT_PSPNR = B.PSPNR
 WHERE A.MANDT = 800
   AND A.BWART IN('601','602','654','653','687','688')
   AND A.SOBKZ = 'Q'
 GROUP BY A.MAT_PSPNR --WBS元素号
	  ,B.POSID --WBS号   
)
,T1 AS( --项目库存
SELECT WBS_LEVEL3 AS WBS_NUM
      ,SUM(COALESCE(RMN_STOCK_QTY,0)) AS PROJ_STOCK_QTY
  FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T
 GROUP BY WBS_LEVEL3
),T2 AS(
SELECT ZJHD AS PLAN_NUM --计划订单号
      ,ZJHDID AS PLAN_ITEM_NUM --计划单行号
	  ,ZPOSID AS WBS_NUM --WBS号
	  ,MATNR AS MATERIAL_NUM --物料编码
	  ,MAKTX AS MATERIAL_NAME --物料名称
	  ,WERKS AS FACTORY_CODE --工厂编码
	  ,MENGE AS QTY --需求数量
  FROM SDI.SDI_ZPST005_1001 A
 WHERE A.MANDT = 800
)
INSERT OVERWRITE TABLE DWROTD.DWR_OTD_RSDU_PLAN_F
SELECT T2.WBS_NUM --WBS号
      ,T2.PLAN_NUM --计划订单号
	  ,T2.PLAN_ITEM_NUM --计划单行号
	  ,T2.MATERIAL_NUM --物料编码
	  ,T2.MATERIAL_NAME --物料名称
	  ,T2.FACTORY_CODE --工厂编码
	  ,T2.QTY --需求数量
	  ,T2.QTY * CAST(LENGTH_1 * WIDTH / 1000000 AS DECIMAL(38,10)) AS AREA --需求面积
	  ,COALESCE(T.OUT_STOCK_QTY,0) AS OUT_STOCK_QTY --出库数量
	  ,COALESCE(T.OUT_STOCK_QTY,0) * CAST(LENGTH_1 * WIDTH / 1000000 AS DECIMAL(38,10)) AS OUT_STOCK_AREA --出库面积
	  ,COALESCE(T1.PROJ_STOCK_QTY,0) AS PROJ_STOCK_QTY --项目库存数量
	  ,COALESCE(T1.PROJ_STOCK_QTY,0) * CAST(LENGTH_1 * WIDTH / 1000000 AS DECIMAL(38,10)) AS PROJ_STOCK_AREA --项目库存面积
	  ,COALESCE(T2.QTY,0) - COALESCE(T.OUT_STOCK_QTY,0) - COALESCE(T1.PROJ_STOCK_QTY,0) AS RSDU_PLAN_QTY --项目剩余需求数量
	  ,(COALESCE(T2.QTY,0) - COALESCE(T.OUT_STOCK_QTY,0) - COALESCE(T1.PROJ_STOCK_QTY,0)) * CAST(LENGTH_1 * WIDTH / 1000000 AS DECIMAL(38,10)) AS RSDU_PLAN_AREA --项目剩余需求面积
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T2
  LEFT JOIN T
    ON T2.PLAN_ITEM_NUM = T.WBS_NUM
  LEFT JOIN T1
    ON T2.PLAN_ITEM_NUM = T1.WBS_NUM
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T3
    ON T2.MATERIAL_NUM = T3.MATERIAL_NUM
