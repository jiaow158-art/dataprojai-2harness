
insert overwrite table DWROTD.DWR_OTD_SO_PICKUP_ORDERS_F
select
 t1.pick_header_no   	-- 提货单
,t2.pick_item_number    -- 提货单行
,t3.SAP_ORDER_NUMBER    -- sap订单
,case when length(t3.SAP_ORDER_ITEM_NUMBER)=6 then t3.SAP_ORDER_ITEM_NUMBER else substr(t3.SAP_ORDER_ITEM_NUMBER,11,6) end SAP_ORDER_ITEM_NUMBER    -- sap 行
,t2.load_plan_number    -- 装载计划单
,t2.project_plan_number -- 装载计划行
,t1.reservation_num	  	-- 预约单编码
,t2.pick_type 			-- 提货类型
,t2.creation_date   	-- 提货时间
,t2.apply_personnel 	-- 申请人
,t2.shipping_status 	-- 发货状态
,t2.oa_appro_num 		-- oa流水
,t2.oa_refuse_reason	-- oa拒绝原因
,t2.status 				-- 审批状态
,t2.consignee_name 		-- 收货人
,t2.consignee_phone 	-- 收货人电话
,t2.hope_delivery_date 	-- 期望发货日期
,t2.hope_receiving_date -- 期望收货日期
,t2.delivere_date 		-- 发货日期
,t2.tray_deli_flag 		-- 是否带托
,t2.province 			-- 省
,t2.city 				-- 市
,t2.county 				-- 区
,t2.address 			-- 详细地址
,t2.pick_driver 		-- 提货司机
,t2.driver_phone 		-- 司机电话
,t2.driver_no 			-- 司机证件
,t2.car_no 				-- 车牌号
,t2.cabinet_num 		-- 柜号
,t2.site_code			-- 站点编码
,t2.site_name			-- 站点名称
,t2.site_description 	-- 站点备注
,t2.prebook_type		-- 预约类型
,t2.prebook_date		-- 预约日期
,t2.start_time			-- 预约开始时间
,t2.end_time			-- 预约结束时间
,t2.prebook_name		-- 预约人
,t2.prebook_name_phone	-- 预约电话
,t2.unloading           -- 是否卸货
,s1.meaning             -- 是否卸货描述
,t2.backingup           -- 是否倒车
,s2.meaning             -- 是否倒车描述
,t3.material_number     -- 物料编码
,t3.material_name       -- 物料描述
,t3.batch               -- 批次
,t3.stock_location_code -- 库存地点
,t3.quantity            -- 下单数量
,t2.pick_qty            -- 提货数量
,t2.machining           -- 是否加工
,s3.meaning             -- 是否加工描述
,t2.supplier_code		-- 供应商编码
-- ,t2.tms_code
-- ,t1.tms_flag
-- ,t1.sap_error_msg
,`current_timestamp`()
from sdi.sdi_pickup_order_header_cz_1034 t1
join sdi.sdi_pickup_order_cz_1034  t2 on t1.id  = t2.head_id
left join sdi.sdi_s_engineering_plan_order_line_1029   t3 on t2.order_item_id = t3.id
left join sdi.sdi_base_lookup_values_1029 s1 on t2.unloading = s1.lookup_code and s1.lookup_type='DISCHARGE_FLAG' and s1.system_code='LTC' and s1.enabled_flag='Y'
left join sdi.sdi_base_lookup_values_1029 s2 on t2.backingup = s2.lookup_code and s2.lookup_type='DISCHARGE_FLAG' and s2.system_code='LTC' and s2.enabled_flag='Y'
left join sdi.sdi_base_lookup_values_1029 s3 on t2.machining = s3.lookup_code and s3.lookup_type='DISCHARGE_FLAG' and s3.system_code='LTC' and s3.enabled_flag='Y'
