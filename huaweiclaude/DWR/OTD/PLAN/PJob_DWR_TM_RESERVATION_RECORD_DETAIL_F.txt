with RESERVATION_VENUE_T as (
SELECT t1.reservation_record_id,T1.reservation_code,
T1.reservation_group_code,T2.load_end_time,T2.load_start_time,T2.sign_time,T2.leave_time,T1.is_abnormal
,case when  T2.sign_time IS NOT NULL then '已到厂' else '未到厂' end  as  DRIVER_PRESENCE_STATUS--司机到场状态
,case when  T2.load_end_time IS NOT NULL THEN  '装车完成' 
      when  T2.load_start_time IS NOT NULL then '装车中' 
      ELSE '未装车'
      end  as  CAR_LOADING_STATUS --装车状态
FROM
SDI.SDI_VENUE_INFO_1041  T1 LEFT JOIN 
(SELECT 
T1.venue_id,    
MAX(CASE WHEN T1.venue_status = 'LOAD_END' THEN T1.operation_start_time ELSE NULL END) AS load_end_time, --装车完成时间
MAX(CASE WHEN T1.venue_status = 'BE_LOAD' THEN T1.operation_start_time ELSE NULL END) AS load_start_time, --装车开始时间
MAX(CASE WHEN T1.venue_status = 'LINE_UP' THEN T1.operation_start_time ELSE NULL END) AS sign_time, --到场日期
MAX(CASE WHEN T1.venue_status = 'COMPLETE' THEN T1.operation_start_time ELSE NULL END) AS leave_time --离场时间
FROM 
sdi.SDI_RESERVATION_VENUE_LOG_1041 t1 
where t1.delete_flag <> 1
GROUP BY T1.venue_id) T2  
ON T1.venue_id = T2.venue_id
--and t1.is_abnormal <> 1 异常数据也要加入统计单量
and t1.delete_flag <> 1
)
,RESERVATION_PROJECT_PLAN_MAPPING as (
SELECT  
DISTINCT reservation_record_id,reservation_code,delete_flag,project_plan_number,t2.vbeln_vl as delivery_order_number,reservation_number
from  
SDI.SDI_RESERVATION_PROJECT_PLAN_MAPPING_1041 t1 
left join  sdi.sdi_zdel_plan_d_1001  t2  on t1.project_plan_number = t2.spec
)

INSERT OVERWRITE TABLE dwrotd.DWR_TM_RESERVATION_RECORD_DETAIL_F
select 
T1.driver_timeout as driver_timeout,
T1.customer_timeout as customer_timeout,
A.reservation_record_id as reservation_record_id,
T1.reservation_state as reservation_state,  --预约状态
A.reservation_code as reservation_code, 
T1.reservation_group_code as reservation_group_code, 
T1.sign_in_time as sign_in_time, --签到日期
T1.reservation_end_time as reservation_end_time, 
T1.reservation_start_time as reservation_start_time,
T1.reservation_date as  reservation_date,
T1.weight as weight, 
T1.version_num as  version_num, 
T1.last_update_login as last_update_login, 
T1.last_update_date as last_update_date, 
T1.last_updated_by as last_updated_by, 
T1.creation_date as creation_date,
T1.created_by as created_by,
A.delete_flag as delete_flag, 
A.project_plan_number as project_plan_number, 
T1.loading_plan_number as loading_plan_number, 
T1.customer_code as customer_code, 
T1.customer_name as customer_name, 
T1.departure_time as departure_time,
T1.warehouse_name as warehouse_name, 
T1.warehouse_code as warehouse_code,
T1.warehouse_id as warehouse_id,
T1.reservation_type as reservation_type, 
T1.carrier as carrier, 
T1.driver_phone as driver_phone,
T1.license_plate as license_plate,  
T1.driver_name as driver_name,
T1.reservation_flag as reservation_flag,
A.delivery_order_number as delivery_order_number,
T1.bookable_date_start as bookable_date_start,
T1.bookable_date_end as bookable_date_end, 
T1.axle_number as axle_number, 
T1.bookable_flag as bookable_flag,
T2.sign_time AS sign_time,
T2.leave_time AS leave_time,
T2.load_start_time AS load_start_time,
T2.load_end_time AS load_end_time,
case 
    when  t1.reservation_date = substring(t2.sign_time,1,10) AND T3.staff_id IS NULL  then 'Y' ELSE 'N' END AS is_accurate ,--是否准到
case 
    when t3.staff_id is not null then 'Y' 
    else 'N' end as is_staff, --是否仓管人
T3.staff_id  as staff_id,--仓管ID
case when hour(t2.load_end_time) >=5 then (UNIX_TIMESTAMP(T2.load_end_time) - UNIX_TIMESTAMP(T2.sign_time)) / 3600 end AS wait_time, -- 等待时间(小时),
T2.is_abnormal AS is_abnormal, -- 是否异常
T2.DRIVER_PRESENCE_STATUS AS DRIVER_PRESENCE_STATUS ,--司机到场状态
current_timestamp as  dw_last_update_date,
T2.CAR_LOADING_STATUS AS CAR_LOADING_STATUS, -- 装车状态
TT.MEANING AS reservation_state_desc,--预约状态描述
TT2.MEANING  AS  reservation_type_desc,--预约类型描述
A.reservation_number  as reservation_number --预留单号
from  RESERVATION_PROJECT_PLAN_MAPPING A
LEFT JOIN  
sdi.SDI_RESERVATION_RECORD_1041 T1 
on 
A.reservation_record_id = T1.reservation_record_id
LEFT JOIN RESERVATION_VENUE_T T2
ON T1.reservation_group_code = T2.reservation_group_code
LEFT JOIN (select DISTINCT(staff_id) from SDI.SDI_RS_WAREHOUSE_USER_INFO_1041)  T3
ON  T1.created_by = T3.staff_id
LEFT JOIN sdi.sdi_base_lookup_values_1027 TT 
ON  T1.reservation_state = TT.lookup_code 
and tt.lookup_type = 'RESERVATION_STATE'  
LEFT JOIN sdi.sdi_base_lookup_values_1027 TT2 
ON   T1.reservation_type = TT2.lookup_code 
AND TT2.lookup_type = 'RESERVATION_TYPE'
where 
a.delete_flag <> 1
