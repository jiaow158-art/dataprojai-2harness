--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;

--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true;

INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_ENERGYCONSUME_D_F PARTITION(PERIOD_ID_M)
SELECT START_DATE, -- 日期
	   '4' AS BUSINESS_TYPE_CODE, -- 业务类型编码(1,2,3,4)
	   '物耗' AS BUSINESS_TYPE_NAME, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
	   SITE_CODE, -- 基地编码
	   WORK_SHOP_CODE, -- 车间编码
	   PRODUCT_LINE_CODE, -- 产线编码
	   WORK_NO_CODE, -- 工序编码
	   COST_CENTER_CODE, -- 成本中心
	   MATERIAL_CONSUME_AMT AS USE_QTY, -- 用量(物耗金额)
	   'N' AS DEL_FLAG, -- 删除数据标识
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	   WORK_SHOP_CODE AS S_WORK_SHOP_CODE, --原车间编码
	   PERIOD_ID_M --  分区月
FROM DWROTD.DWR_PAM_BASE_MATERIALCONSUME_D_F -- 瓷砖各基地物耗表(日)
WHERE DEL_FLAG = 'N'
AND PERIOD_ID_M >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)  -- 前3月，调度传昨天,格式：YYYY-MM-DD
UNION ALL
SELECT START_DATE, -- 日期
	   '3' AS BUSINESS_TYPE_CODE, -- 业务类型编码(1,2,3,4)
	   '煤' AS BUSINESS_TYPE_NAME, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
	   SITE_CODE, -- 基地编码
	   WORK_SHOP_CODE, -- 车间编码
	   PRODUCT_LINE_CODE, -- 产线编码
	   WORK_NO_CODE, -- 工序编码
	   COST_CENTER_CODE, -- 成本中心
	   COAL_QTY AS USE_QTY, -- 用量(煤)
	   'N' AS DEL_FLAG, -- 删除数据标识
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	   WORK_SHOP_CODE AS S_WORK_SHOP_CODE, --原车间编码
	   PERIOD_ID_M --  分区月
FROM DWROTD.DWR_PAM_BASE_COAL_F -- 瓷砖各基地用煤表
WHERE DEL_FLAG = 'N'
AND PERIOD_ID_M >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)  -- 前3月，调度传昨天,格式：YYYY-MM-DD
UNION ALL
SELECT  START_DATE, -- 日期
		'2' AS BUSINESS_TYPE_CODE, -- 业务类型编码(1,2,3,4)
	    '气' AS BUSINESS_TYPE_NAME, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
		SITE_CODE,-- 基地编码
		WORK_SHOP_CODE, -- 车间编码
		PRODUCT_LINE_CODE, -- 产线编码
		WORK_NO_CODE,  -- 工序编码
		COST_CENTER_CODE, -- 成本中心
		GAS_QTY AS USE_QTY,-- 用量（气）
		DEL_FLAG, -- 删除数据标识
		DW_LAST_UPDATE_DATE, -- 数据同步时间
		WORK_SHOP_CODE AS S_WORK_SHOP_CODE, --原车间编码
		PERIOD_ID_M --  分区月
FROM DWROTD.DWR_PAM_BASE_NATURALGAS_F -- 瓷砖各基地用气表
WHERE DEL_FLAG = 'N'
AND PERIOD_ID_M >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)  -- 前3月，调度传昨天,格式：YYYY-MM-DD
UNION ALL
SELECT  START_DATE, -- 日期
		'1' AS BUSINESS_TYPE_CODE, -- 业务类型编码(1,2,3,4)
	    '电' AS BUSINESS_TYPE_NAME, -- 业务类型名称(1:电(度) 2:气(立方米) 3:煤(吨) 4:物耗(元))
		SITE_CODE,-- 基地编码
		WORK_SHOP_CODE, -- 车间编码
		PRODUCT_LINE_CODE, -- 产线编码
		WORK_NO_CODE,  -- 工序编码
		COST_CENTER_CODE, -- 成本中心
		ELECTRICITY_QTY AS USE_QTY,-- 用量（电）
		DEL_FLAG, -- 删除数据标识
		DW_LAST_UPDATE_DATE, -- 数据同步时间
		S_WORK_SHOP_CODE AS S_WORK_SHOP_CODE, --原车间编码
		PERIOD_ID_M --  分区月
FROM DWROTD.DWR_PAM_BASE_ELECTRIC_F -- 瓷砖各基地用电表
WHERE DEL_FLAG = 'N'
AND PERIOD_ID_M >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)  -- 前3月，调度传昨天,格式：YYYY-MM-DD
