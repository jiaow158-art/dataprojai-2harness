set hive.cbo.enable=true;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
WITH WORK_CENTER AS (
SELECT 
T.SITE_CODE,
T.SITE_NAME,
T.COST_CENTER_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
T.S_PRODUCT_LINE_CODE,
T.S_PRODUCT_LINE_NAME
FROM ( 
SELECT   
T.SITE_CODE,
T.SITE_NAME,
T.COST_CENTER_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_CODE 
     ELSE '' END AS S_PRODUCT_LINE_CODE,  --服务产线编码
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_NAME 
     ELSE '' END AS S_PRODUCT_LINE_NAME  --服务产线名称
FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 T --基地产线成本表
-- LEFT JOIN SDI.SDI_RADIATION_WORKSHOP_T_1037 T2  --辅助车间辐射车间
-- ON T.SITE_CODE = T2.SITE_CODE 
-- AND T.COST_CENTER_CODE = T2.COST_CENTER_CODE
-- AND T.M_PRODUCT_LINE_CODE = T2.M_PRODUCT_LINE_CODE
-- AND '${PERIOD_ID_M1}' BETWEEN SUBSTR(T2.EFFECT_START_DATE,1,7) AND SUBSTR(T2.EFFECT_END_DATE,1,7) 
-- AND '${PERIOD_ID_M2}' BETWEEN SUBSTR(T2.EFFECT_START_DATE,1,7) AND SUBSTR(T2.EFFECT_END_DATE,1,7) 
) T
GROUP BY  
T.SITE_CODE,
T.SITE_NAME,
T.COST_CENTER_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
T.S_PRODUCT_LINE_CODE,
T.S_PRODUCT_LINE_NAME
)
,SUBJECT AS (
SELECT A.VALFROM,A.VALTO,A.SETNAME,B.DESCRIPT
FROM SDI.SDI_SETLEAF_1001 A 
LEFT JOIN SDI.SDI_SETHEADERT_1001 B 
ON B.MANDT = '800' 
AND B.LANGU = '1'
AND A.SETCLASS = B.SETCLASS
AND A.SUBCLASS = B.SUBCLASS
AND A.SETNAME = B.SETNAME
WHERE A.SETCLASS = '0102' AND A.SUBCLASS = 'CNDP'
AND A.SETNAME IN ('B','C','D','L','M','H','J','K')
)
-- ,VAL_TYPE AS (
-- SELECT MATNR,SUBSTR(BWKEY,1,2) AS PLANT,BKLAS
-- ,ROW_NUMBER() OVER (PARTITION BY MATNR,SUBSTR(BWKEY,1,2)) AS RN
-- FROM SDI.SDI_MBEW_1001
-- )
,LOCKOUT_DATE AS (
SELECT SUM(CASE WHEN LOCKOUT_DAYS >= 15 THEN 1 ELSE 0 END) AS LOCKOUT_LINE_QTY,STAT_MONTH,S_PRODUCT_LINE_CODE 
FROM DWROTD.DWR_PAM_BASE_LOCKOUT_DATE_T A  
WHERE STAT_MONTH BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
GROUP BY STAT_MONTH,S_PRODUCT_LINE_CODE 
)
,TMP AS (
SELECT 
 T.MONTH AS CALMONTH 
,T.COMP_CODE  --公司代码
,T.COST_CENTER_CODE -- 成本中心
,T.COST_CENTER_DESC --成本中心描述
,T.GENERAL_LEDGER_ACCOUNT AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,T.GENERAL_LEDGER_ACCOUNT_DESC AS GENERAL_LEDGER_ACCOUNT_NAME--总账科目描述
,T.LOCAL_CURRENCY_AMT  --金额
,T.ITEM_TEXT AS ITEM_TEXT1 --项目文本
,T.VOUCHER_TYPE_CODE --凭证类型
,T.MATERIAL_CODE AS MATERIAL_NUM --物料编码
,T.MATERIAL_DESC AS MATERIAL_NAME --物料描述
,CASE WHEN T.ITEM_TEXT LIKE '%停工损失%' THEN '停工结转' 
      WHEN T.GENERAL_LEDGER_ACCOUNT = '0060540100' THEN '停工结转' ELSE '' END AS ITEM_TEXT --停工结转
,CASE WHEN T.GENERAL_LEDGER_ACCOUNT = '0061499000' AND SUBSTR(T.MATERIAL_CODE,1,6) IN ('11-A01','12-A01') THEN 'X' 
      WHEN T.ITEM_TEXT LIKE '%球旦%' OR T.ITEM_TEXT LIKE '%球石%' THEN 'X' 
      WHEN T.GENERAL_LEDGER_ACCOUNT = '0061701500' THEN 'X' ELSE '' END AS FEE_FLAG  --费用排除标识
,'SAP实际费用' AS DATA_SOURCE --数据来源
FROM DWRFIN.DWR_FIN_COST_D_COMPRE_SUBJ_T T
-- LEFT JOIN VAL_TYPE T4 
-- ON T.MATERIAL_CODE = T4.MATNR 
-- AND SUBSTR(T.COMP_CODE,1,2) = T4.PLANT AND T4.RN = 1
WHERE T.FUNC_SCOPE_CODE IN ('4101','4105') 
AND T.VOUCHER_TYPE_CODE <> 'CO'
AND T.MONTH BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
UNION ALL 
SELECT 
 T.GJAHR||'-'||T.MONAT AS CALMONTH
,T.WERKS AS SITE_CODE
,T.KOSTL AS COST_CENTER_CODE -- 成本中心
,T1.KTEXT AS COST_CENTER_DESC --成本中心描述
,T.HKONT AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,CASE WHEN T2.TXT20 IS NOT NULL THEN T2.TXT20 ELSE T3.TXT20 END AS GENERAL_LEDGER_ACCOUNT_DESC -- 总账科目描述
,T.ZAMT --金额
,'ZCORP012研发费用调整数' AS ITEM_TEXT1 --项目文本
,'' AS VOUCHER_TYPE_CODE --凭证类型
,'' AS MATERIAL_NUM --物料编码
,'' AS MATERIAL_NAME --物料描述
,'' AS ITEM_TEXT --停工结转
,'' AS FEE_FLAG  --费用排除标识
,'SAP调整研发费用' AS DATA_SOURCE --数据来源
FROM SDI.SDI_ZTCOYF_FY_1001 T
LEFT JOIN SDI.SDI_CSKT_1001 T1
ON T1.SPRAS = '1' AND T1.KOKRS = 'DP00' AND T1.DATBI = '99991231' AND T.KOSTL = T1.KOSTL
left join SDI.SDI_SKAT_1001 T2 -- 95108
on T.HKONT = T2.SAKNR
and T2.KTOPL = 'CNDP'
left join SDI.SDI_SKAT_1001 T3 -- 95108
on T.HKONT = T3.SAKNR
and T3.KTOPL = 'CND1'
WHERE T.MANDT = '800'
AND T.GJAHR||'-'||T.MONAT BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
UNION ALL 
SELECT 
 T.STAT_MONTH 
,T.COMP_CODE
,T.COST_CENTER_CODE -- 成本中心
,T1.KTEXT AS COST_CENTER_DESC --成本中心描述
,T.GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,CASE WHEN T2.TXT20 IS NOT NULL THEN T2.TXT20 ELSE T3.TXT20 END AS GENERAL_LEDGER_ACCOUNT_DESC -- 总账科目描述
,T.SOURCE_AMT --金额
,T.ITEM_TEXT AS ITEM_TEXT1--项目文本
,'' AS VOUCHER_TYPE_CODE --凭证类型
,'' AS MATERIAL_NUM --物料编码
,'' AS MATERIAL_NAME --物料描述
,'' AS ITEM_TEXT --停工结转
,'' AS FEE_FLAG  --费用排除标识
,'中台调整费用' AS DATA_SOURCE --数据来源
FROM SDI.SDI_MANUAL_FEE_T_1037 T 
LEFT JOIN SDI.SDI_CSKT_1001 T1
ON T1.SPRAS = '1' AND T1.KOKRS = 'DP00' AND T1.DATBI = '99991231' AND T.COST_CENTER_CODE = T1.KOSTL
left join SDI.SDI_SKAT_1001 T2 -- 95108
on T.GENERAL_LEDGER_ACCOUNT_CODE = T2.SAKNR
and T2.KTOPL = 'CNDP'
left join SDI.SDI_SKAT_1001 T3 -- 95108
on T.GENERAL_LEDGER_ACCOUNT_CODE = T3.SAKNR
and T3.KTOPL = 'CND1'
WHERE T.STAT_MONTH BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}' 
)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_MFG_COST_T PARTITION(PERIOD_ID_M) 
SELECT  
 T.CALMONTH 
,T.COMP_CODE  --公司代码
,T.COST_CENTER_CODE -- 成本中心
,T.COST_CENTER_DESC --成本中心描述
,T.GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,T.GENERAL_LEDGER_ACCOUNT_NAME--总账科目描述
,T.LOCAL_CURRENCY_AMT  --金额
,T.ITEM_TEXT1 --项目文本
,T.VOUCHER_TYPE_CODE --凭证类型
,T.MATERIAL_NUM --物料编码
,T.MATERIAL_NAME --物料描述
,T.ITEM_TEXT --停工结转
,T3.SETNAME AS COST_ELEMENT_CODE --成本要素 
,T3.DESCRIPT AS COST_ELEMENT_NAME --成本要素描述
,T2.PCS_SEG_CODE --'工序段编码'
,T2.PCS_SEG_NAME --'工序段名称'
,T2.SITE_CODE  -- '基地编码'
,T2.SITE_NAME  -- '基地名称'
,T2.M_PRODUCT_LINE_CODE  --'修正产线编码'
,T2.M_PRODUCT_LINE_NAME  --'修正产线名称'
,T2.S_PRODUCT_LINE_CODE  --'服务产线编码'
,T2.S_PRODUCT_LINE_NAME  --'服务产线名称'
,CASE WHEN T.GENERAL_LEDGER_ACCOUNT_CODE = '0061499000' AND SUBSTR(T.MATERIAL_NUM,1,6) IN ('11-A01','12-A01') AND T2.M_PRODUCT_LINE_NAME IN ('分级车间','辅助车间','釉料车间','原料车间') THEN 'X' 
      WHEN (T.ITEM_TEXT1 LIKE '%球旦%' OR T.ITEM_TEXT1 LIKE '%球石%') AND T2.M_PRODUCT_LINE_NAME IN ('分级车间','辅助车间','釉料车间','原料车间') THEN 'X' 
      WHEN T.GENERAL_LEDGER_ACCOUNT_CODE = '0061701500' THEN 'X'
      WHEN T3.SETNAME = 'H' AND SUBSTR(T.CALMONTH,1,4) > '2025' AND (SUBSTR(T.MATERIAL_NUM,1,3) IN ('1WB','1BW') OR T.ITEM_TEXT1 LIKE '%承包%' OR T.ITEM_TEXT1 LIKE '%外包%') THEN 'X'
      WHEN T3.SETNAME IN ('H','J') AND T2.M_PRODUCT_LINE_NAME IN ('分级车间','辅助车间','釉料车间','原料车间') AND ((T4.COST_CENTER_CODE IS NOT NULL 
      AND T4.RN1 <> T4.RN2) OR (T4.COST_CENTER_CODE IS NULL AND T5.RN3 <> T5.RN4)) 
      THEN 'X' ELSE '' END AS FEE_FLAG--费用排除标识
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
,DATA_SOURCE --数据来源
,T.CALMONTH AS PERIOD_ID_M
FROM TMP T 
LEFT JOIN WORK_CENTER T2 
ON T.COST_CENTER_CODE = T2.COST_CENTER_CODE
JOIN SUBJECT T3 
ON T.GENERAL_LEDGER_ACCOUNT_CODE BETWEEN T3.VALFROM AND T3.VALTO
LEFT JOIN (SELECT A.COST_CENTER_CODE,C.STAT_MONTH,COUNT(DISTINCT A.S_PRODUCT_LINE_CODE) AS RN1,SUM(NVL(B.LOCKOUT_LINE_QTY,0)) AS RN2
FROM SDI.SDI_RADIATION_WORKSHOP_T_1037 A
LEFT JOIN (SELECT DISTINCT STAT_MONTH FROM LOCKOUT_DATE) C 
ON 1 = 1
LEFT JOIN LOCKOUT_DATE B 
ON A.S_PRODUCT_LINE_CODE = B.S_PRODUCT_LINE_CODE
AND C.STAT_MONTH = B.STAT_MONTH
GROUP BY A.COST_CENTER_CODE,C.STAT_MONTH ) T4 
ON T.COST_CENTER_CODE = T4.COST_CENTER_CODE 
AND T.CALMONTH = T4.STAT_MONTH 
LEFT JOIN (SELECT A.SITE_CODE,C.STAT_MONTH,COUNT(1) AS RN3, SUM(NVL(B.LOCKOUT_LINE_QTY,0)) AS RN4
FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 A --基地产线成本表 
LEFT JOIN (SELECT DISTINCT STAT_MONTH FROM LOCKOUT_DATE) C 
ON C.STAT_MONTH BETWEEN SUBSTR(A.START_DATE,1,7) AND SUBSTR(A.END_DATE,1,7)
LEFT JOIN LOCKOUT_DATE B 
ON A.M_PRODUCT_LINE_CODE = B.S_PRODUCT_LINE_CODE
AND C.STAT_MONTH = B.STAT_MONTH
WHERE A.M_PRODUCT_LINE_NAME NOT IN ('分级车间','辅助车间','釉料车间','原料车间')
GROUP BY A.SITE_CODE,C.STAT_MONTH ) T5
ON T.COMP_CODE = T5.SITE_CODE
AND T.CALMONTH = T5.STAT_MONTH
