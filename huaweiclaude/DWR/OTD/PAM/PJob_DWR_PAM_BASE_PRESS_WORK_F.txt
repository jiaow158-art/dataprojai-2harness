,t1.production_num --'产量'
,t1.stamping_frequency --'冲压频率'
,t1.molding_pressure --'成型压力'
,t1.oil_temperature --'油温'
,t1.pressure --'气压'
,t1.creater --'创建人姓名'
,t1.create_time --'创建时间'
,t1.updater --'修改人姓名'
,t1.update_time --'修改时间'
,'N' AS DEL_FLAG 
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
,( CASE WHEN regexp_extract(UPPER(t1.PRODUCT_DIM),UPPER('X'),0) = UPPER('X') THEN 
						   split(UPPER(t1.PRODUCT_DIM), UPPER('X'))[0]
						WHEN regexp_extract(UPPER(t1.PRODUCT_DIM),UPPER('\\*'),0) = UPPER('*') THEN
					       split(UPPER(t1.PRODUCT_DIM), UPPER('\\*'))[0]
					ELSE split(UPPER(t1.PRODUCT_DIM), UPPER('X'))[0] END) *
			 -- 长 600X600 按X拆分字段,拆前部分，可能会有*
			( CASE WHEN regexp_extract(UPPER(t1.PRODUCT_DIM),UPPER('X'),0) = UPPER('X') THEN 
						   split(UPPER(t1.PRODUCT_DIM), UPPER('X'))[1]
					WHEN regexp_extract(UPPER(t1.PRODUCT_DIM),UPPER('\\*'),0) = UPPER('*') THEN
					       split(UPPER(t1.PRODUCT_DIM), UPPER('\\*'))[1]
					ELSE split(UPPER(t1.PRODUCT_DIM), UPPER('X'))[1] 
				END) / 1000000 as unit_area --宽
,substr(SHIFT_DATE,1,7) as period_id_m
from tmp t1
