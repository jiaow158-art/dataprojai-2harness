,a.SUBSTANDARD_QTY	--次品数量
,a.WASTE_QTY --废品数量

,a.LAST_UPDATE_DATE  -- 修改时间

on a.PRODUCT_CODE = b.material_num

ON a.SITE_CODE = c.SITE
