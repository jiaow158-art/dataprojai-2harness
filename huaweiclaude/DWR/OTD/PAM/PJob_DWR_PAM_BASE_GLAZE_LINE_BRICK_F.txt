 T.site as site_code -- '基地'
,T.work_shop as work_shop_code --'车间'

,T.work_no_code -- '工序代码'
,SUBSTR(T.shift_date,1,10) AS PRODUCT_DATE -- '生产日期'
,T.shift_code as SHIFT_CODE -- '班次代码'
,T.SHIFT_START_TIME -- '班次开始时间'
,T.SHIFT_END_TIME -- '班次结束时间'

,0 AS SHIFT_INKJET_COUNTER_NUM -- '喷墨机计数器-交班'
,0 AS INKJET_COUNTER_BRICK_NUM --'喷墨机计数器-实际入砖数'
,T.SUCCESSION_STOCK_NUM  -- '储坯器储坯数-接班'
,T.SHIFT_STOCK_NUM --	'储坯器储坯数-交班'
,0 AS SUCCESSION_PICK_NUM    --'捡砖机计数器-接班'
,0 AS SHIFT_PICK_NUM	   --'捡砖机计数器-交班'
,0 AS BRICK_PICK_NUM	   --'捡砖机计数器-实际入砖数'
,T.creater -- '创建人姓名'
,T.create_time --'创建时间'
,T.updater -- '修改人姓名'
,T.update_time -- '修改时间'
,0 as shift_change_stock_num_jiaob --储坯器储坯数-交班
,0 as shift_join_stock_num_jieb --储坯器储坯数-接班
,0 as  shift_change_green_track_num_jiaob --储坯轨道储坯数-交班
,0 as shift_join_green_track_num_jieb --储坯轨道储坯数-接班
from sdi.sdi_dj_glaze_line_brick_record_1006 T  --釉线入砖数记录表
where T.site in ('1900') and T.delete_flag = '1'
and T.shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')

select 
 T.site as site_code -- '基地'
,T.work_shop as work_shop_code --'车间'
,T.PRODUCT_LINE as PRODUCT_LINE_CODE -- '产线代码'

insert overwrite table DWROTD.DWR_PAM_BASE_GLAZE_LINE_BRICK_F partition(period_id_m)
select 
 T.site_code -- '基地'
,T.work_shop_code --'车间'
,T.PRODUCT_LINE_CODE -- '产线代码'
,T.work_no_code -- '工序代码'
,T.PRODUCT_DATE -- '生产日期'
,T.SHIFT_CODE -- '班次代码'
,T.SHIFT_START_TIME -- '班次开始时间'
,T.SHIFT_END_TIME -- '班次结束时间'
,T.PRODUCT_CODE -- '产品编号'
,T.PRODUCT_DIM -- '规格'
,NVL(T1.UNIT_AREA,0) AS UNIT_AREA--单位面积
,NVL(T.SUCCESSION_INKJET_COUNTER_NUM,0) AS SUCCESSION_INKJET_COUNTER_NUM -- '喷墨机计数器-接班'
,NVL(T.SHIFT_INKJET_COUNTER_NUM,0) AS SHIFT_INKJET_COUNTER_NUM -- '喷墨机计数器-交班'
,NVL(T.INKJET_COUNTER_BRICK_NUM,0) AS INKJET_COUNTER_BRICK_NUM --'喷墨机计数器-实际入砖数'
,NVL(T.SUCCESSION_STOCK_NUM,0) AS SUCCESSION_STOCK_NUM -- '储坯器储坯数-接班'
,NVL(T.SHIFT_STOCK_NUM,0) AS SHIFT_STOCK_NUM--	'储坯器储坯数-交班'
,NVL(T.SUCCESSION_PICK_NUM,0) AS SUCCESSION_PICK_NUM    --'捡砖机计数器-接班'
,NVL(T.SHIFT_PICK_NUM,0) AS SHIFT_PICK_NUM	   --'捡砖机计数器-交班'
,NVL(T.BRICK_PICK_NUM,0) AS BRICK_PICK_NUM	   --'捡砖机计数器-实际入砖数'
,T.creater -- '创建人姓名'
,T.create_time --'创建时间'
,T.updater -- '修改人姓名'
,T.update_time -- '修改时间'
,'N' AS DEL_FLAG  --  '删除标识'
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,shift_change_stock_num_jiaob --储坯器储坯数-交班
,shift_join_stock_num_jieb --储坯器储坯数-接班
,shift_change_green_track_num_jiaob --储坯轨道储坯数-交班
,shift_join_green_track_num_jieb --储坯轨道储坯数-接班

left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T1
on T.PRODUCT_CODE = T1.material_num
