--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH TMP AS (
SELECT 
 SITE AS SITE_CODE 
,WORK_SHOP AS WORK_SHOP_CODE
,PRODUCT_LINE AS PRODUCT_LINE_CODE
,SHIFT AS SHIFT_CODE
,WORK_NO_CODE 
,RESERVE1 AS GROUP_CODE
,RESERVE4
,RESERVE5
,PRODUCT_DATE AS SHIFT_DATE
,TRANSFER_PRODUCT_BEFORE_MATERIAL_NO AS BEFORE_MATERIAL_NUM 
,TRANSFER_PRODUCT_AFTER_MATERIAL_NO AS AFTER_MATERIAL_NUM 
,GROUP_SPECIFICATION AS PROD_SIZE
,TRANSFER_PRODUCT_DATE||':00' AS TRANSFER_PRODUCT_DATE
,LAG(TRANSFER_PRODUCT_DATE,1) OVER(PARTITION BY PRODUCT_DATE,SITE,WORK_SHOP ORDER BY TRANSFER_PRODUCT_DATE)||':00' AS START_TIME
,CREATE_BY AS CREATER
,CREATE_DATE 
,LAST_UPDATE_BY AS UPDATER
,LAST_UPDATE_DATE AS UPDATE_DATE
,ROW_NUMBER() OVER(PARTITION BY PRODUCT_DATE,SITE,WORK_SHOP ORDER BY TRANSFER_PRODUCT_DATE DESC) RN
FROM SDI.SDI_D_INKJET_TRANSFER_PRODUCT_1006
WHERE DELETE_FLAG = 1
AND PRODUCT_DATE >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
)
,TMP1 AS (
SELECT 
 SITE_CODE 
,WORK_SHOP_CODE
,PRODUCT_LINE_CODE
,SHIFT_CODE
,WORK_NO_CODE 
,GROUP_CODE
,SHIFT_DATE
,BEFORE_MATERIAL_NUM AS MATERIAL_NUM 
,PROD_SIZE
,COALESCE(START_TIME,CASE WHEN RESERVE4 < RESERVE5 THEN CONCAT(SUBSTR(SHIFT_DATE,1,10),' ',RESERVE4) ELSE CONCAT(SUBSTR(SHIFT_DATE,1,10),' ',RESERVE4) END) AS START_TIME -- 开始生产时间，如果START_TIME为空(即RN1=1，则是第一个生产的物料；第一个生产的物料取班次开始时间)
,TRANSFER_PRODUCT_DATE AS END_TIME
,CREATER
,CREATE_DATE 
,UPDATER
,UPDATE_DATE
FROM TMP 
UNION ALL 
SELECT 
 SITE_CODE 
,WORK_SHOP_CODE
,PRODUCT_LINE_CODE
,SHIFT_CODE
,WORK_NO_CODE 
,GROUP_CODE
,SHIFT_DATE
,AFTER_MATERIAL_NUM AS MATERIAL_NUM 
,PROD_SIZE
,TRANSFER_PRODUCT_DATE AS START_TIME
,CASE WHEN RESERVE4 < RESERVE5 THEN  CONCAT(SUBSTR(SHIFT_DATE,1,10),' ',RESERVE5) ELSE CONCAT(DATE_ADD(SUBSTR(SHIFT_DATE,1,10),1),' ',RESERVE5) END AS END_TIME
,CREATER
,CREATE_DATE 
,UPDATER
,UPDATE_DATE
FROM TMP 
WHERE RN = 1 
)

INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_INKJET_TRANSFER_S PARTITION(PERIOD_ID_M)
SELECT  
 SITE_CODE 
,WORK_SHOP_CODE
,PRODUCT_LINE_CODE
,SHIFT_CODE
,WORK_NO_CODE 
,GROUP_CODE
,SHIFT_DATE
,MATERIAL_NUM 
,PROD_SIZE
,START_TIME
,END_TIME
,ABS(UNIX_TIMESTAMP(END_TIME) - UNIX_TIMESTAMP(START_TIME)) AS TIME_DIFF
,CREATER
,CREATE_DATE 
,UPDATER
,UPDATE_DATE
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
,SUBSTR(SHIFT_DATE,1,7) AS PERIOD_ID_M
FROM TMP1
