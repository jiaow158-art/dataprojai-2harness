--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
WITH DETAIL_DATA AS( -- 煤的明细数据
	SELECT SUBSTRING(BLDAT,1,6) AS period_id_m,  -- 分区月
	       BLDAT AS START_DATE,  -- 日期
	       WERKS AS SITE_CODE, -- 基地
	       KOSTL AS COST_CENTER_CODE,-- 成本中心
	       SUM ( CASE WHEN SHKZG = 'H' THEN ERFMG ELSE -1 * ERFMG END) * 1000 AS COAL_QTY  -- 数量
	FROM SDI.SDI_MATDOC_1001 T
	WHERE del_flag = 'N'
	AND T.BLDAT >= REPLACE(CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01'),'-','') -- 前3月，调度传昨天,格式：YYYY-MM-DD
	AND BWART IN ( 'Z13', 'Z14' )  -- 移动类型
	AND WERKS IN ( '1300', '1500', '1600', '1900' )  -- 基地
	AND MATNR IN ( '13-A01-000003', '13-A01-000005', '13-A01-000019', '13-A01-000020', '13-A01-000022', '13-A01-000021' )  -- 物料
	GROUP BY SUBSTRING(BLDAT,1,6), 
	         BLDAT,
	         WERKS,
	         KOSTL
),
DWR_PAM_BASE_COAL_DATA AS( -- 煤数据
	SELECT CONCAT(SUBSTRING(T1.START_DATE,1,4),'-',SUBSTRING(T1.START_DATE,5,2),'-',SUBSTRING(T1.START_DATE,7,2)) AS START_DATE, -- 日期
		   T1.SITE_CODE,  --基地编码
		   T2.work_shop AS WORK_SHOP_CODE,  -- 车间编码
		   T2.product_line AS PRODUCT_LINE_CODE, -- 产线编码
		   T2.work_no_code , -- 工序编码
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.COAL_QTY , -- 用煤量(数量)
		   CONCAT(SUBSTRING(T1.PERIOD_ID_M,1,4),'-',SUBSTRING(T1.PERIOD_ID_M,5,2)) AS PERIOD_ID_M -- 分区月
	FROM DETAIL_DATA T1
	LEFT JOIN SDI.sdi_dj_powerconsumption_costcenter_maintenance_1006 T2  -- 清远-用电与成本中心维护表
	ON T1.SITE_CODE = T2.site 
	AND T1.COST_CENTER_CODE = T2.cost_center 
	AND T2.delete_flag = 1
),
dim_date as (    -- 日期维度,发散维度用
	select day_dt 
	     from dm.dm_dim_date_d 
	  where day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	    and day_dt >=  CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01') -- 前3月，调度传昨天,格式：YYYY-MM-DD 
),
ACTICAL_DETAIL_DATA AS( -- 实际数据
	SELECT period_id_m,
		   START_date,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code,
		   SUM(COAL_QTY) AS COAL_QTY
	FROM DWR_PAM_BASE_COAL_DATA T 
	WHERE period_id_m >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)
	GROUP BY period_id_m,
			 START_date,
			 site_code,
			 work_shop_code,
			 product_line_code,
			 work_no_code,
			 cost_center_code
),
ACTICAL_DETAIL_DIM_DATA AS( -- 维度数据
	SELECT period_id_m,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code
	FROM ACTICAL_DETAIL_DATA
	GROUP BY period_id_m,
			 site_code,
			 work_shop_code,
			 product_line_code,
			 work_no_code,
			 cost_center_code
),
result_data as(
	SELECT t1.day_dt AS start_date,
		   t2.period_id_m ,
		   t2.site_code ,
		   t2.work_shop_code,
		   t2.product_line_code ,
		   t2.work_no_code ,
		   t2.cost_center_code ,
		   nvl(t3.COAL_QTY,0) AS COAL_QTY
	FROM dim_date T1
	LEFT JOIN ACTICAL_DETAIL_DIM_DATA t2
	ON SUBSTRING(T1.day_dt,1,7) = T2.period_id_m  -- 按月发散
	LEFT JOIN ACTICAL_DETAIL_DATA t3
	ON t1.day_dt = t3.START_date
	AND t2.period_id_m = t3.period_id_m
	AND t2.site_code = t3.site_code
	AND t2.work_shop_code = t3.work_shop_code
	AND t2.product_line_code = t3.product_line_code
	AND t2.work_no_code = t3.work_no_code
	AND t2.cost_center_code = t3.cost_center_code
),
CQ_GAS_RESULT_DATA AS( -- 重庆气的结果数据
	SELECT t1.START_DATE, -- 使用日期
		   t1.SITE_CODE, -- 基地编码
		   t1.work_shop_code, -- 车间编码
		   t1.product_line_code, -- 产线编码
		   t1.work_no_code, -- 工序编码
		   t1.COST_CENTER_CODE, -- 成本中心编码
		   SUM(t1.gas_qty) AS gas_qty -- 用气量
	FROM DWROTD.DWR_PAM_BASE_GAS_CQ_F T1 -- 瓷砖天然气中间表(重庆)
    WHERE UPPER(t1.gas_code) LIKE '%GAS-PW%' -- 只取 GAS-PW 的数据
    AND t1.START_DATE >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
    GROUP BY start_date,
             site_code,
             work_shop_code,
             product_line_code,
             work_no_code,
             cost_center_code
)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_COAL_F PARTITION (PERIOD_ID_M)
SELECT START_DATE, -- 日期
	   SITE_CODE,  --基地编码
	   WORK_SHOP_CODE,  -- 车间编码
	   PRODUCT_LINE_CODE, -- 产线编码
	   work_no_code , -- 工序编码
	   COST_CENTER_CODE, -- 成本中心
	   COAL_QTY , -- 用煤量(数量)
	   'N' AS DEL_FLAG, -- 删除数据标识
	   CURRENT_TIMESTAMP() AS dw_last_update_date , -- 最后同步数据时间
	   NVL(PERIOD_ID_M,SUBSTRING(START_DATE,1,7)) AS PERIOD_ID_M -- 分区月
FROM result_data
UNION ALL
SELECT START_DATE, -- 使用日期
	   SITE_CODE, -- 基地编码
	   work_shop_code, -- 车间编码
	   product_line_code, -- 产线编码
	   work_no_code, -- 工序编码
	   COST_CENTER_CODE, -- 成本中心编码
	   gas_qty AS COAL_QTY, -- 用煤量(用气量)
	   'N' AS DEL_FLAG, -- 删除数据标识
	   CURRENT_TIMESTAMP() AS dw_last_update_date , -- 最后同步数据时间
	   SUBSTRING(START_DATE,1,7) AS PERIOD_ID_M -- 分区月
FROM CQ_GAS_RESULT_DATA;
