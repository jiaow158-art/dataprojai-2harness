WITH repeat_grade_data_01 as(  -- 重烧明细数据1
 SELECT fs.fjdate
        ,pm.category
          ,pm.MUDID
        ,fs.promodel
        ,fs.factory
        ,COUNT(DISTINCT fs.uuid) cnt   -- 分级数
        ,COUNT(CASE WHEN fs.prograde in('优等','合格') THEN fs.code END )  yds -- 优等数
   FROM sdi.sdi_productmodel_1011 pm    -- 光瓷产品信息表
       ,SDI.SDI_qis_fsfjinfo_1011 fs    -- 返烧信息
  WHERE 1 = 1
    AND fs.status <> '9'
    AND fs.promodel = pm.modelid
    AND fs.zjproperty in ('正常','试制') -- = '正常'
    AND fs.qxclass = '普通缺陷'
    AND fs.factory IN ('3200','3202')
  --  AND fs.fjdate >= '2022-11-10'   AND fs.fjdate <= '2022-11-10'
    AND PM.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')
  GROUP BY fs.fjdate, 
           pm.category, 
           pm.MUDID,
           fs.promodel,
           fs.factory
 UNION ALL 
 SELECT fs.fjdate
        ,pm.category
        ,pm.MUDID
        ,fs.promodel
        ,fs.factory
        ,SUM(fs.fjnum) cnt
        ,SUM(fs.ydnum) yds
   FROM sdi.sdi_productmodel_1011 pm  -- 光瓷产品信息表
       ,SDI.SDI_qis_xjfsinfo_1011 fs  -- 小件产品返烧分级
  WHERE fs.promodel = pm.modelid
    AND fs.qxclass = '普通缺陷'
    AND fs.factory IN ('3200','3202')
  --  AND fs.fjdate >= '2022-11-10'  AND fs.fjdate <= '2022-11-10'
    AND PM.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')
  GROUP BY fs.fjdate,
      pm.category, 
      pm.MUDID,
      fs.promodel,
      fs.factory
)
,repeat_grade_data_02 as(  -- 重烧明细数据2
 SELECT substring(t.fjdate,1,7) AS grand_date, -- 分级日期
        t.factory as factory_code, -- 工厂编码
	     --t.category,  -- 产品类别
	     --t.MUDID AS MUDID_MODEL,   -- 泥坯型号
	     --t.promodel AS material_num, -- 产品型号(物料编码)
	     t1.material_num AS material_num, -- 产品型号(物料编码)
	     sum(t.cnt) AS csfj_qty,  -- 分级数
	     sum(t.yds) AS csyd_qty  -- 优等数
 FROM repeat_grade_data_01 T
 left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t1 -- 生产BOM关系数据
 on t.factory = t1.factory_code
 and t.promodel = t1.small_material_num
 WHERE t.fjdate  between '2022-06-01' and '${PERIOD_ID_D}'  -- 传当天
 GROUP BY substring(t.fjdate,1,7),
          t.factory,
          --t.category, 
          --t.MUDID,
          --t.promodel
          t1.material_num
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_BASE_CSFJ_JX_T
SELECT T.*,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM repeat_grade_data_02 T;
