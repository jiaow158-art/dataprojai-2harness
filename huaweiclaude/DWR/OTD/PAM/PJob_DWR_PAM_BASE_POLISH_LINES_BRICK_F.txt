--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with tmp as (
select 
 SITE as site_code 
,WORK_SHOP as work_shop_code
,PRODUCT_LINE as product_line_code
,PRODUCT_DATE as shift_date
,SHIFT as shift_code 
,GROUP_CODE 
,WORK_NO_CODE
,GROUP_CODE as material_num 
,REAL_ON_BRICK_NUM as ON_BRICK_NUM
,CREATE_BY as creater
,CREATE_DATE as create_time
,LAST_UPDATE_BY as updater
,LAST_UPDATE_DATE as update_time
from sdi.sdi_d_polish_lines_brick_1006
where site = '1800' and delete_flag = '1'
and PRODUCT_DATE >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select 
 site as site_code 
,work_shop as work_shop_code
,product_line as product_line_code
,shift_date
,shift_code
,group_code
,work_no_code
,product_code as material_num 
,polish_lines_num as ON_BRICK_NUM
,creater
,create_time
,updater
,update_time
from sdi.sdi_dj_molding_section_production_data_1006
where site = '1500' and delete_flag = '1'
and shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select 
 site as site_code 
,work_shop as work_shop_code
,product_line as product_line_code
,shift_date
,shift_code
,group_code
,work_no_code
,product_code as material_num 
,number_bricks_stored as ON_BRICK_NUM
,creater
,create_time
,updater
,update_time
from sdi.sdi_dp_polishing_line_data_tracking_info_1006
where site = '1300' and delete_flag = '1'
and shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
)
insert overwrite table dwrotd.dwr_pam_base_polish_lines_brick_f partition(period_id_m)
select 
 T.site_code 
,T.work_shop_code
,T.product_line_code
,T.shift_date
,T.shift_code
,T.group_code
,T.work_no_code
,T.material_num 
,NVL(T1.UNIT_AREA,0) AS UNIT_AREA--单位面积
,NVL(T.ON_BRICK_NUM,0) AS ON_BRICK_NUM -- 上砖数
,T.creater
,T.create_time
,T.updater
,T.update_time
,'N' AS DEL_FLAG  --  '删除标识'
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,SUBSTR(T.shift_date,1,7) AS period_id_m
from tmp T
left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T1
on T.material_num = T1.material_num
