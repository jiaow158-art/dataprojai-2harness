with dim_date as (
	select day_dt 
	     from dm.dm_dim_date_d 
	  where  day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	        and day_dt >= add_months(concat(substring('${PERIOD_ID_D}',1,7),'-01'),-2) -- 昨天所在1号的前2个月
	    --and day_dt >= date_add(SUBSTR('${PERIOD_ID_D}',1,10),-34)
	  --and day_dt >= '2022-05-01'
),
SDI_B1_CUSTOMER_DATA_1037_data as(
    select case when length(rtm_cust_code) = 10 then rtm_cust_code  else concat('0000',rtm_cust_code) end as rtm_cust_code,
           t.abbreviation,
		   SUBSTRING(CAST(t.rtm_online_date AS STRING),1,10) AS  rtm_online_date
    from sdi.SDI_B1_CUSTOMER_DATA_1037 t
    where t.is_main ='是'
),
BOM_DETAIL_DATA AS(   -- BOM数据
	select material_num
	   from(
			select t1.MATNR as material_num,  -- 物料编码
				   t2.IDNRK as small_material_num, -- 子项物料编码
			   	   t2.MENGE as small_material_qty  -- 子项物料数量
			from(
				select MATNR,
					   STLNR,
					   row_number() over(partition by MATNR order by ANDAT desc ) as RN
				from SDI.SDI_MAST_1001 
				WHERE STLAN = '5'
			) t1
			left join sdi.sdi_STPO_1001 t2
			on t1.STLNR = t2.STLNR
			where t1.RN = 1
		  )t
		  group by material_num
),
tmp1 as  (
	select 
		 substr(creation_date,1,10) day_dt 
		,creation_date
		,t2.rtm_cust_code AS customer_code  -- 客户别名
		,product_code AS itemcode -- 物料编码
-- 		,product_name AS itemname -- 物料描述
		,transaction_price -- 价格
		,sum(nvl(before_qty,0)) `期初数量`  -- 期初数量
		,sum(CASE WHEN inv_type = 'IN' THEN NVL(transaction_quantity,0) ELSE 0 END ) AS `本期入库` -- 本期入库 '调拨入库',
		,sum(CASE WHEN transaction_type_id in (1,4) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS `期间采购收货数量` -- 期间采购收货数量
		,sum(CASE WHEN transaction_type_id in (6,7) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS `期间销售退货数量` -- 期间销售退货数量
		,sum(CASE WHEN transaction_type_id = 17  THEN NVL(transaction_quantity,0) WHEN  transaction_type_id = 18  THEN NVL(transaction_quantity,0) * -1 ELSE 0 END) AS `期间其他入库数量` -- 期间其他入库数量
		,sum(CASE WHEN transaction_type_id in (2,3) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS `期间采购退货数量` -- 期间采购退货数量
		,sum(CASE WHEN inv_type = 'OUT' THEN NVL(transaction_quantity,0) ELSE 0 END ) AS `本期出库` -- 本期出库 '调拨出库',
		,sum(CASE WHEN transaction_type_id in (5,8) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS `期间销售出库数量` -- 期间销售出库数量
		,sum(CASE WHEN transaction_type_id = 13	THEN NVL(transaction_quantity,0) WHEN transaction_type_id = 14	THEN NVL(transaction_quantity,0) * -1 ELSE 0 END) AS `期间其他出库数量` -- 期间其他出库数量
		,t1.warehouse_code --库存地点编码
	from SDI.SDI_dtwms_mtl_material_transactions_1031 t1 
	JOIN SDI_B1_CUSTOMER_DATA_1037_data   t2 
	on t1.distributor_code = t2.rtm_cust_code
	where 1=1 
	and t1.product_code not in (select material_num from BOM_DETAIL_DATA)
--	and t1.distributor_code = t2.rtm_cust_code
-- 	and SUBSTRING(t1.creation_date,1,10) >= t2.rtm_upload_date
	group by 
	 substr(creation_date,1,10)  
	,t2.rtm_cust_code
	,product_code 
-- 	,product_name 
	,creation_date
	,transaction_price
	,t1.warehouse_code
)
,tmp2 as(
	select
	    t2.day_dt 
		,customer_code
		,itemcode
-- 		,itemname
		,'' as `最末入库日期` -- 最末入库日期
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(`本期入库`,0) - nvl(`本期出库`,0) else 0  end)     `期初数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end )  `本期入库`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间采购收货数量`,0) else 0 end )  `期间采购收货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间销售退货数量`,0) else 0 end )  `期间销售退货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间其他入库数量`,0) else 0 end )  `期间其他入库数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间采购退货数量`,0) else 0 end )  `期间采购退货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end )  `本期出库`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间销售出库数量`,0) else 0 end )  `期间销售出库数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`期间其他出库数量`,0) else 0 end )  `期间其他出库数量`
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(`本期入库`,0) - nvl(`本期出库`,0) else 0  end)  + 
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end ) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end )  `期末数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end ) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end )  as `期间发生`
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(`本期入库`,0) - nvl(`本期出库`,0) else 0  end * t1.transaction_price ) as `期初金额` -- 期初金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end * t1.transaction_price) as `本期入库金额`  -- 本期入库金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end * t1.transaction_price) as `本期出库金额` -- 本期出库金额
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(`本期入库`,0) - nvl(`本期出库`,0) else 0  end * t1.transaction_price)  + 
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end * t1.transaction_price) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end * t1.transaction_price)  as  `期末金额` -- 期末金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期入库`,0) else 0 end * t1.transaction_price) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(`本期出库`,0) else 0 end * t1.transaction_price) as `本期发生金额`  -- 本期发生金额
		,t1.warehouse_code
	from dim_date t2 join tmp1 t1 on 1=1 
	where t1.day_dt <= t2.day_dt  
	group by 
	customer_code
	,itemcode
-- 	,itemname
	,t2.day_dt  
	,t1.warehouse_code
)
