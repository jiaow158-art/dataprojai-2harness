---SAP提货单行表临时表TMP002
set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_LADINGBILL_ITEM_TMP002
SELECT 
		DWLI.GROUP_NUM			AS	GROUP_NUM
		,DWLI.DELIVERY			AS	DELIVERY
		,DWLI.DELIVERY_PROJECT	AS	DELIVERY_PROJECT
		,R.CONFIG_NAME			AS	SO_ITEM_NUM_TYPE_NAME
		,R1.CONFIG_NAME			AS	SPECIAL_INV_FLAG_NAME
		,R2.CONFIG_NAME			AS	SUBJECT_DISTR_TYPE_NAME
		,R3.CONFIG_NAME			AS	PROD_GROUP_NAME
		,R4.CONFIG_NAME			AS	BUSINESS_SCOPE_NAME
		,R5.CONFIG_NAME			AS	MATERIAL_GROUP_NAME
		,R6.CONFIG_NAME			AS	CONTROL_SCOPE_NAME
		,R7.CONFIG_NAME			AS	PROFIT_CENTER_NAME
		,R8.CONFIG_NAME			AS	MATERIAL_TYPE_NAME
		,R9.CONFIG_NAME			AS	PROD_LEVEL_NAME
		,R10.CONFIG_NAME		AS	DIS_CHANNEL_NAME
		,R11.CONFIG_NAME		AS	MATERIAL_SHIP_GROUP_NAME
		,R12.CONFIG_NAME		AS	SHIP_GROUP_NAME
		,R13.CONFIG_NAME		AS	SAL_GROUP_NAME
FROM DWIOTD.DWI_WM_LADINGBILL_ITEM_T DWLI
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R			---交货项目类别中文名称
	 ON R.CONFIG_CODE = DWLI.SO_ITEM_NUM_TYPE
	 AND R.GROUP_NUM = DWLI.GROUP_NUM
	 AND R.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R1			---特殊库存标识中文名称
	 ON R1.CONFIG_CODE = DWLI.SPECIAL_INV_FLAG
	 AND R1.GROUP_NUM = DWLI.GROUP_NUM
	 AND R1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R2			---科目分配类别中文名称
	 ON R2.CONFIG_CODE = DWLI.SUBJECT_DISTR_TYPE
	 AND R2.GROUP_NUM = DWLI.GROUP_NUM
	 AND R2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R3			---产品组中文名称
	 ON R3.CONFIG_CODE = DWLI.PROD_GROUP
	 AND R3.GROUP_NUM = DWLI.GROUP_NUM
	 AND R3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R4			---业务范围中文名称
	 ON R4.CONFIG_CODE = DWLI.BUSINESS_SCOPE
	 AND R4.GROUP_NUM = DWLI.GROUP_NUM
	 AND R4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R5			---物料组中文名称
	 ON R5.CONFIG_CODE = DWLI.MATERIAL_GROUP
	 AND R5.GROUP_NUM = DWLI.GROUP_NUM
	 AND R5.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R6			---控制范围中文名称
	 ON R6.CONFIG_CODE = DWLI.CONTROL_SCOPE
	 AND R6.GROUP_NUM = DWLI.GROUP_NUM
	 AND R6.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R7			---利润中心中文名称
	 ON R7.CONFIG_CODE = DWLI.PROFIT_CENTER
	 AND R7.GROUP_NUM = DWLI.GROUP_NUM
	 AND R7.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R8			---物料类型中文名称
	 ON R8.CONFIG_CODE = DWLI.MATERIAL_TYPE
	 AND R8.GROUP_NUM = DWLI.GROUP_NUM
	 AND R8.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R9			---产品层次中文名称
	 ON R9.CONFIG_CODE = DWLI.PROD_LEVEL
	 AND R9.GROUP_NUM = DWLI.GROUP_NUM
	 AND R9.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R10		---分销渠道中文名称
	 ON R10.CONFIG_CODE = DWLI.DIS_CHANNEL
	 AND R10.GROUP_NUM = DWLI.GROUP_NUM
	 AND R10.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R11		---物料运输组中文名称
	 ON R11.CONFIG_CODE = DWLI.MATERIAL_SHIP_GROUP
	 AND R11.GROUP_NUM = DWLI.GROUP_NUM
	 AND R11.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R12		---运输组中文名称
	 ON R12.CONFIG_CODE = DWLI.SHIP_GROUP
	 AND R12.GROUP_NUM = DWLI.GROUP_NUM
	 AND R12.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R13		---销售组名称中文名称
	 ON R13.CONFIG_CODE = DWLI.SAL_GROUP
	 AND R13.GROUP_NUM = DWLI.GROUP_NUM
	 AND R13.LANG_CODE = '1'
WHERE DWLI.DEL_FLAG = 'N'
