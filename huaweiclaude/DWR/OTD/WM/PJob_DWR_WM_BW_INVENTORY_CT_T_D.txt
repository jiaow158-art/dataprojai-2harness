--CREATE TABLE DWROTD.DWR_WM_BW_INVENTORY_CT_T_D AS
WITH BW_INVENTORY_DETAIL_DATA AS(  -- 来源BW的库存明细数据
	SELECT CONCAT(SUBSTRING(T1.DOCDATE,1,4),'-',SUBSTRING(T1.DOCDATE,5,2),'-',SUBSTRING(T1.DOCDATE,7,2)) as  doc_date, -- 单据日期
		   CASE WHEN LENGTH(T1.distid) < 10 THEN CONCAT('0000',T1.distid) ELSE T1.distid END AS customer_code ,    -- 经销商编码
		   T1.itemcode as material_num, -- 物料编码
		   T1.objtype , -- 单据类型
		   T1.quantity  -- 数量
	FROM SDI.SDI_ZIM_CUS_DATA_1009 T1 
	JOIN DWIMD.dwi_md_data_material_general_t T2
	ON T1.itemcode = T2.material_num 
	AND T2.prod_property = 'A0001' -- 瓷砖客户
	WHERE T1.distid in ('0000316827','352516') -- 0000316827 福建石狮市三跃建材贸易有限公司 , 352516 杭州东箭完整家居有限公司
	AND T1.objtype IN ('13','14','18','19','59','60') -- 13 销售出库  14 销售退货 18 采购收货 19 采购退货 59 其他入库 60 其他出库
),
dim_date as (    -- 日期维度,按月分区
	select day_dt 
	     from dm.dm_dim_date_d 
	  where  day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	    and day_dt >= DATE_SUB('${PERIOD_ID_D}',18) -- 传昨天,昨天往前18天
),
BW_INVENTORY_DETAIL_DATA_01 as(
	SELECT dt.day_dt,
		   T1.customer_code,
		   T1.material_num,
		   case when t1.doc_date < dt.day_dt then 'qc' 
				when t1.doc_date  >=  dt.day_dt and  t1.doc_date< date_sub(dt.day_dt,-1) then 'qj' 
				else 'qm'  end as dttype ,
			CASE WHEN T1.objtype = '14' then T1.quantity ELSE 0 END AS IN_QTY,     -- 入库
			CASE WHEN T1.objtype = '13' then T1.quantity ELSE 0 END AS OUT_QTY,    -- 出库
			CASE WHEN T1.customer_code = '0000352516' THEN -- 杭州东箭18正数为入库，负数为出库，福建的18是入库，19是出库
			    CASE WHEN T1.objtype = '18' AND T1.quantity > 0 THEN T1.quantity ELSE 0 END 
			ELSE
			    CASE WHEN T1.objtype = '18' THEN T1.quantity ELSE 0 END 
			END AS PUCH_IN_QTY,   -- 采购入库
			CASE WHEN T1.customer_code = '0000352516' THEN
			   CASE WHEN T1.objtype = '18' AND T1.quantity < 0 THEN T1.quantity ELSE 0 END 
			ELSE
			   CASE WHEN T1.objtype = '19' THEN T1.quantity ELSE 0 END 
			END AS PUCH_OUT_QTY, -- 采购出库
			CASE WHEN T1.objtype = '59' THEN T1.quantity ELSE 0 END AS OTHER_IN_QTY, -- 其它入库
			CASE WHEN T1.objtype = '60' THEN T1.quantity ELSE 0 END AS OTHER_OUT_QTY -- 其它出库
	FROM BW_INVENTORY_DETAIL_DATA T1
	JOIN dim_date dt
	on 1 = 1
),
BW_INVENTORY_DETAIL_DATA_02 as(
	select 
	     t.day_dt,
	     t.dttype,
	     t.customer_code,
	     t.material_num,
	     sum(case when t.dttype='qc' then nvl(t.IN_QTY,0.0) else 0.0 end) as qc_in_qty,
	     sum(case when t.dttype='qj' then nvl(t.IN_QTY,0.0) else 0.0 end) as qj_in_qty,
	     sum(case when t.dttype='qm' then nvl(t.IN_QTY,0.0) else 0.0 end) as qm_in_qty,
	     sum(case when t.dttype='qc' then nvl(t.OUT_QTY,0.0) else 0.0 end) as qc_out_qty,
	     sum(case when t.dttype='qj' then nvl(t.OUT_QTY,0.0) else 0.0 end) as qj_out_qty,
	     sum(case when t.dttype='qm' then nvl(t.OUT_QTY,0.0) else 0.0 end) as qm_out_qty,
	     sum(case when t.dttype='qc' then nvl(t.PUCH_IN_QTY,0.0) else 0.0 end) as qc_puch_in_qty,
	     sum(case when t.dttype='qj' then nvl(t.PUCH_IN_QTY,0.0) else 0.0 end) as qj_puch_in_qty,
	     sum(case when t.dttype='qc' then nvl(t.OTHER_IN_QTY,0.0) else 0.0 end) as qc_other_in_qty,
	     sum(case when t.dttype='qj' then nvl(t.OTHER_IN_QTY,0.0) else 0.0 end) as qj_other_in_qty,
	     sum(case when t.dttype='qc' then nvl(t.PUCH_OUT_QTY,0.0) else 0.0 end)as qc_puch_out_qty, -- 值存的是负数
	     sum(case when t.dttype='qj' then nvl(t.PUCH_OUT_QTY,0.0) else 0.0 end)as qj_puch_out_qty,
	     sum(case when t.dttype='qc' then nvl(t.OTHER_OUT_QTY,0.0) else 0.0 end) as qc_other_out_qty,
	     sum(case when t.dttype='qj' then nvl(t.OTHER_OUT_QTY,0.0) else 0.0 end) as qj_other_out_qty
	from BW_INVENTORY_DETAIL_DATA_01 t  
	group by t.day_dt,
			 t.dttype,
			 t.customer_code,
	         t.material_num
),
BW_INVENTORY_DETAIL_DATA_03 as(
	select  
		 tt.day_dt, -- "日期"
		 tt.customer_code, -- 客户编码
		 tt.material_num, --  "物料编码"
		 (sum(nvl(tt.qc_in_qty,0.0))-sum(nvl(tt.qc_out_qty,0.0))) + (sum(nvl(tt.qc_puch_in_qty,0.0)) - abs(sum(nvl(tt.qc_puch_out_qty,0.0)))) + (sum(nvl(tt.qc_other_in_qty,0.0)) - sum(nvl(tt.qc_other_out_qty,0.0))) as qc_qty, -- "期初数量"
		 sum(nvl(tt.qj_in_qty,0.0)) as qj_instock_qty, -- "本期入库"
		 sum(nvl(tt.qj_puch_in_qty,0.0)) as qj_pp_del_qty, -- "期间采购收货数量"
		 sum(nvl(tt.qj_other_in_qty,0.0)) as qj_other_instock_qty, -- "期间其他入库数量"
		 sum(nvl(tt.qj_puch_out_qty,0.0)) as qj_pp_return_qty, -- "期间采购退货数量"
		 sum(nvl(tt.qj_out_qty,0.0)) as qj_outstock_qty, -- "本期出库"
		 sum(nvl(tt.qj_other_out_qty,0.0)) as qj_other_outstock_qty, -- "期间其他出库数量"
		 ((sum(nvl(tt.qc_in_qty,0.0))-sum(nvl(tt.qc_out_qty,0.0))) + (sum(nvl(tt.qc_puch_in_qty,0.0)) - abs(sum(nvl(tt.qc_puch_out_qty,0.0)))) + (sum(nvl(tt.qc_other_in_qty,0.0)) - sum(nvl(tt.qc_other_out_qty,0.0))))  + (sum(nvl(tt.qj_in_qty,0.0))-sum(nvl(tt.qj_out_qty,0.0))) as qm_qty, -- "期末数量"
		 sum(nvl(tt.qj_in_qty,0.0))-sum(nvl(tt.qj_out_qty,0.0)) as qj_happen_qty -- "期间发生"
	from BW_INVENTORY_DETAIL_DATA_02  tt  
	group by tt.day_dt ,
			 tt.customer_code,
			 tt.material_num
),
BW_CUSTOMER_DATA AS(  -- BW的客户数据
	select customer_code,
		   customer_name,
		   data_source_falg,
		   operation_center_name
	from SDI.SDI_CT_CUSTOMER_DATA_1037 T 
	where T.data_source_falg = 'BW'
	and NVL(stop_flag,'NULL') <> 'Y'
)
INSERT OVERWRITE TABLE DWROTD.DWR_WM_BW_INVENTORY_CT_T_D PARTITION(period_id_d,p_customer_code)
select t2.data_source_falg, -- 数据来源标识
	   t1.day_dt, -- 日期,
	   t1.customer_code, -- 客户编码
	   t2.customer_name, -- 客户名称
	   t2.operation_center_name as sales_org_name, -- 销售组织名称(待定)
	   t1.material_num,  -- 物料编码
	   t3.material_name,-- `SAP物料名称`,
	   t3.menu_desc_name,  -- 是否主菜单
       t3.dimension , -- 规格
       nvl((t3.length_1 * t3.width) / 1000000,0) as unit_area_qty, -- 单位面积
       t1.qc_qty, -- 期初数量,
       nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qc_qty as qc_area_qty, -- 期初面积
       0 as qc_amt, -- 期初金额,
       t1.qm_qty, -- 期末数量,
       nvl((t3.length_1 * t3.width) / 1000000,0) * t1.qm_qty as qm_area_qty, -- 期末面积
       0 as qm_amt, -- 期末金额,
       t1.qj_instock_qty, -- 本期入库,
       nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qj_instock_qty as qj_instock_area_qty, -- 期间入库面积
       0 as bq_instock_amt, -- 本期入库金额,
       t1.qj_other_instock_qty, -- 期间其他入库数量,
	   nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qj_other_instock_qty as qj_other_instock_area_qty, -- 期间其他入库面积
       0 as qj_other_instock_amt, -- 期间其他入库金额,
       t1.qj_pp_del_qty, -- 期间采购收货数量,
       nvl((t3.length_1 * t3.width) / 1000000,0)  * t1.qj_pp_del_qty as qj_pp_del_area_qty, -- 期间采购收货面积
       0 as qj_pp_del_area_amt, -- 期间采购收货金额
       0 as qj_sale_return_qty, -- 期间销售退货数量,
       0 as qj_sale_return_area_qty, -- 期间销售退货面积
       0 as qj_sale_return_amt,-- 期间销售退货金额,
       t1.qj_outstock_qty, -- 本期出库,
       nvl((t3.length_1 * t3.width) / 1000000,0)  *  t1.qj_outstock_qty as qj_outstock_area_qty, -- 本期出库面积, 
       0 as bq_outstock_amt, -- 本期出库金额,
       0 as qj_sale_outstock_qty, -- 期间销售出库数量,
       0 as qj_sale_outstock_area_qty , -- 期间销售出库面积
       0 as qj_sale_outstock_amt, -- "期间销售出库金额" 
       t1.qj_other_outstock_qty, -- 期间其他出库数量,
       nvl((t3.length_1 * t3.width) / 1000000,0)  * t1.qj_other_outstock_qty as qj_other_outstock_area_qty, -- 期间其他出库面积,
       0 as qj_other_outstock_amt, -- "期间其他出库金额"  
       t1.qj_happen_qty, -- 期间发生,
       nvl((t3.length_1 * t3.width) / 1000000,0) * t1.qj_happen_qty as qj_happen_area_qty, -- 期间发生面积
       0 as bq_happen_amt, -- 本期发生金额,
       '' as last_instock_date, -- 最末入库日期,
	   current_timestamp() as dw_last_update_date, -- 数据同步时间
	   t1.day_dt as PERIOD_ID_D,  -- 分区字段
	   t1.customer_code AS P_CUSTOMER_CODE -- 客户编码
from BW_INVENTORY_DETAIL_DATA_03 t1
join BW_CUSTOMER_DATA t2
on t1.customer_code  = t2.customer_code
join dwimd.dwi_md_data_material_general_t t3
on t1.material_num = t3.material_num
and t3.prod_property = 'A0001' -- 瓷砖客户
--where t1.customer_code = '0000352516'
--and t1.material_num = 'CFG802105_A';
;
