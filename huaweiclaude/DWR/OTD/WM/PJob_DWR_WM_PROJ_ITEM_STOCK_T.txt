set hive.cbo.enable=true;				--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
with tmp as (
select 
 a.matnr as material_num
,a.werks as plant_code
,a.lgort as loc_code
,a.charg as batch_num
,b.posid as wbs_level3
,a.MBLNR as mat_voucher --物料凭证
,a.MJAHR as mat_year --物料凭证年份
,a.ZEILE as mat_voucher_item  --物料凭证项目
,a.VBELN_IM as delivery_num --交货单号
,a.VBELP_IM as delivery_item --交货行号
,a.EBELN as pur_num --采购单号
,a.EBELP as pur_item --采购单行号
,a.cpudt as rmn_date --留货时间
,a.bwart as move_type --移动类型
,c.prlab as non_rstk_stock_qty --非限制库存数量(分析力度只去到wbs)
,c.prins as check_stock_qty -- 在检库存数量(分析力度只去到wbs)
,c.prspe as frz_stock_qty -- 冻结库存数量(分析力度只去到wbs)
,case when a.shkzg  = 'S' then a.menge else 0 end as in_stock_qty --入库数量
,case when a.shkzg  = 'H' then a.menge else 0 end as out_stock_qty --出库数量  
,a.stock_qty as rmn_stock_qty--库存数量(留货数)
,case when a.bwart = '415' AND a.shkzg = 'S' then a.menge 
      when (a.bwart in ('411','415') and a.shkzg = 'H') or (a.bwart = '412' and a.shkzg = 'S') then -a.menge else 0 end as in_stock_zt_qty --累计入库数量
,case when a.bwart in ('601','654','687') then a.menge 
      when a.bwart in ('602','653','688') then -a.menge else 0 end as out_stock_zt_qty --累计出库数量
,case when a.bwart = '951' then -a.menge 
      when a.bwart = '952' then a.menge else 0 end loss_stock_qty --破损数量
,case when a.bwart = '991' then a.menge 
      when a.bwart = '992' then -a.menge else 0 end alt_batch_stock_qty -- 物料转批次数量
from sdi.sdi_matdoc_1001 as a 
inner join sdi.sdi_prps_1001 as b 
on a.mat_pspnr = b.pspnr
inner join sdi.sdi_mspr_1001 as c
on a.matnr = c.matnr 
AND a.werks = c.werks 
AND a.lgort = c.lgort 
AND a.charg = c.charg 
AND a.mat_pspnr = c.pspnr 
WHERE a.sobkz  = 'Q'
AND  a.bwart  <> '313'
)
insert overwrite table DWROTD.DWR_WM_PROJ_ITEM_STOCK_T
select 
 a.material_num
,a.plant_code
,a.loc_code
,a.batch_num
,a.wbs_level3
,b.wbs_level2
,b.wbs_level1
,a.mat_voucher --物料凭证
,a.mat_year --物料凭证年份
,a.mat_voucher_item  --物料凭证项目
,a.delivery_num --交货单号
,a.delivery_item --交货行号
,a.pur_num --采购单号
,a.pur_item --采购单行号
,FROM_UNIXTIME(UNIX_TIMESTAMP(a.rmn_date,'yyyyMMdd'),'yyyy-MM-dd') as rmn_date --留货时间
,a.move_type --移动类型
,a.non_rstk_stock_qty --非限制库存数量
,a.check_stock_qty -- 在检库存数量
,a.frz_stock_qty -- 冻结库存数量
,a.in_stock_qty --入库数量
,a.out_stock_qty --出库数量  
,a.rmn_stock_qty--库存数量(留货数)
,a.in_stock_zt_qty --累计入库数量
,a.out_stock_zt_qty --累计出库数量
,a.loss_stock_qty --破损数量
,a.alt_batch_stock_qty -- 物料转批次数量
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
from tmp a
left join DWRDIM.DWR_DIM_WBS_BASIS_INFO_F b
on a.wbs_level3 = b.wbs_level3
