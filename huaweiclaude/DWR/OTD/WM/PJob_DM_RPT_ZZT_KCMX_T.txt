with mat as (  --存在批次为空的物料读不到入库时间
SELECT 
      MATNR AS MATERIAL_NUM,
      MIN(CASE WHEN ((BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG <> 'K') 
                or (BSTTYP_SG = 'K' and BWART in ('411','101','991'))) and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30'  THEN BUDAT 
               WHEN ((BWART in ('101','911','913','Z01','131','531','561','653') and BSTTYP_SG <> 'K') or (BSTTYP_SG = 'K' and BWART = '101')) 
               THEN BUDAT END) AS EARLIEST_VOUCHER_POST_DATE_BW, ---批次最早入库日期(BW通用)
      MIN(CASE WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and WERKS = '7C20' THEN BUDAT 
               WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and BSTTYP_SG <> 'K' and EBELN not like '47%'
               THEN BUDAT END) AS earliest_voucher_post_date_fc --批次最早入库日期(辅材专用)      
  FROM SDI.SDI_MATDOC_1001
  where charg = '' and charg_cid = ''
  GROUP BY
  MATNR
)
,mat0 as (
select MATNR,CHARG,LIFNR,WERKS,min(BUDAT) as BUDAT
from SDI.SDI_MATDOC_1001
where BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG = 'K'
and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30'
group by MATNR,CHARG,LIFNR,WERKS
)
,MC AS (  ---物料凭证与采购信息记录表关联获取生产日期期间的日期记录
select * from (
select A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG,substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2) as HSDAT,
c.MATNR,c.LIFNR,c.WERKS,c.EKORG,c.ESOKZ,c.KNUMH,c.KSCHL,c.DATAB,c.DATBI
,row_number() over(partition by c.MATNR,c.LIFNR,c.WERKS,B.PRODUCE_DATE,A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG order by c.DATBI,c.EKORG desc ) AS RN
from SDI.SDI_MATDOC_1001 A
inner join DWIMD.DWI_MD_MATL_BATCH_INFO_T B
on A.MATNR = B.MATERIAL_NUM and A.CHARG = B.BATCH_NUM
inner join SDI.SDI_A017_1001 c 
on A.WERKS = c.WERKS and A.LIFNR = c.LIFNR and A.MATNR = c.MATNR and ESOKZ = '2' and KSCHL = 'PB00'
and c.DATBI >= substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2) 
and c.DATAB <= substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2)
where BSTTYP_SG = 'K' AND B.PRODUCE_DATE IS NOT NULL
) a where RN = 1
)

,MC1 AS (  --最小过账日期与采购信息记录表关联记录
select * from (
select A.MATNR,A.CHARG,A.LIFNR,A.WERKS,A.BUDAT
,B.EKORG,B.ESOKZ,B.KNUMH,B.KSCHL,B.DATAB,B.DATBI
,row_number() over(partition by A.MATNR,A.LIFNR,A.WERKS,A.BUDAT,A.MATNR,A.CHARG,A.LIFNR,A.WERKS order by B.DATBI desc ) AS RN
from MAT0 A
inner join SDI.SDI_A017_1001 B 
on A.MATNR = B.MATNR and A.LIFNR = B.LIFNR and A.WERKS = B.WERKS and ESOKZ = '2' and KSCHL = 'PB00'
and B.DATBI >= A.BUDAT and B.DATAB <= A.BUDAT
-- where A.BSTTYP_SG = 'K'
) a where RN = 1
)
,MC2 AS (  ---物料凭证与物料账目表关联获取过账日期期间的日期记录
select * from (
select A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG,A.BUDAT,
B.MATNR,B.BWKEY,B.BWTAR,B.KALNR,C.VPRSV,C.PEINH,C.STPRS,C.PVPRS,C.BDATJ,C.POPER
,row_number() over(partition by B.MATNR,B.BWKEY,B.BWTAR,A.BUDAT,A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG order by C.BDATJ||substr(C.POPER,2,2) desc) AS RN 
from SDI.SDI_MATDOC_1001 A
inner join SDI.SDI_CKMLHD_1001 B 
on A.MATNR = B.MATNR and A.WERKS = B.BWKEY and A.BWTAR = B.BWTAR 
inner join SDI.SDI_CKMLCR_1001 C 
on B.KALNR = C.KALNR and C.BDATJ = substr(A.BUDAT,1,4) and substr(C.POPER,2,2) = substr(A.BUDAT,5,2) 
where A.BSTTYP_SG = 'K'
)
a where RN = 1
)

,sum(CASE WHEN A.BSTTYP_SG = 'K' AND W.BWTTY <> 'X'   --k库存且评估类别<>X (非批次成本)
	     THEN COALESCE(y1.KBETR,y3.KBETR,yy.KBETR) / COALESCE(y1.KPEIN,y3.KPEIN,yy.KPEIN) * A.STOCK_QTY --取当期采购价，取不到再取小于入库/过账日期的最新日期采购价
	     WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'S' --为标准价
	     THEN COALESCE(y5.STPRS,y7.STPRS) / COALESCE(y5.PEINH,y7.PEINH) * A.STOCK_QTY  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期标准价，取不到拿小于入库/过账日期往期最大标准价
		 WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'V' --为移动平均价
		 THEN COALESCE(y5.PVPRS,y7.PVPRS) / COALESCE(y5.PEINH,y7.PEINH) * A.STOCK_QTY  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期移动平均价，取不到拿小于入库/过账日期往期最大移动平均价 
		 ELSE 0 END) as zsjkcje --实际库存金额

	     THEN COALESCE(y1.KBETR,y3.KBETR,yy.KBETR) / COALESCE(y1.KPEIN,y3.KPEIN,yy.KPEIN)  --取当期入库/过账日期采购价，保底期初数取所在信息记录最小一条采购价
	     WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'S' --为标准价
	     THEN COALESCE(y5.STPRS,y7.STPRS) / COALESCE(y5.PEINH,y7.PEINH)  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期入库/过账日期标准价
		 WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'V' --为移动平均价
		 THEN COALESCE(y5.PVPRS,y7.PVPRS) / COALESCE(y5.PEINH,y7.PEINH)   --k库存逻辑  正常按成本估算号找不到，按工厂物料评估类取当期入库/过账日期移动平均价
		 ELSE 0 END as zmmm_o017_zzqj--周期价
,A.BWTAR as val_type --"评估类型"
,A.KDAUF as doc_number --"销售凭证"
,A.KDPOS as s_ord_item --"项目",
,A.kalnr --成本估算号
,A.SAKTO --科目号
-- ,A.FIPOS  --承诺项目
,'${PERIOD_ID_M}' as period_id_m
from SDI.SDI_MATDOC_1001 A
LEFT JOIN DWIMD.DWI_MD_MATL_BATCH_INFO_T BB
ON A.MATNR = BB.MATERIAL_NUM AND (case when A.CHARG = '' then A.CHARG_CID ELSE A.CHARG END) = BB.BATCH_NUM
left join sdi.sdi_mbew_1001 W
on A.MATNR = W.MATNR and A.WERKS = W.BWKEY and coalesce(A.BWTAR,'#') = coalesce(W.BWTAR,'#')
----------------------------------------------取物料批次的生产日期所在的信息记录--------------------------------------

on A.MBLNR = x1.MBLNR and A.ZEILE = x1.ZEILE and A.MJAHR = x1.MJAHR and A.SHKZG = x1.SHKZG  
----------------------------------------------取所在月过账日期所在的信息记录------------------------------------------
left join MC1 x3
ON A.MATNR = x3.MATNR AND A.CHARG = x3.CHARG AND A.LIFNR = x3.LIFNR AND A.WERKS = x3.WERKS
----------------------------------------------保底期初数取所在信息记录最小一条---------------------------------------
left join (select MATNR,LIFNR,WERKS,EKORG,ESOKZ,KNUMH,KSCHL,DATAB,DATBI
,row_number() over(partition by MATNR,LIFNR,WERKS,KSCHL order by DATBI ) AS RN from SDI.SDI_A017_1001 
where ESOKZ = '2' ) xx
on A.WERKS = xx.WERKS and A.LIFNR = xx.LIFNR and A.MATNR = xx.MATNR and xx.RN = 1 
-------------------------------------------------------------------------------------------------------------
left join SDI.SDI_KONP_1001 y1
on x1.KNUMH = y1.KNUMH and x1.KSCHL = y1.KSCHL and y1.LOEVM_KO <> 'X'--|获
left join SDI.SDI_KONP_1001 y3
on x3.KNUMH = y3.KNUMH and x3.KSCHL = y3.KSCHL and y3.LOEVM_KO <> 'X'--|采
left join SDI.SDI_KONP_1001 yy
on xx.KNUMH = yy.KNUMH and xx.KSCHL = yy.KSCHL and yy.LOEVM_KO <> 'X'--|价
----------------------------------------------取物料批次的生产日期所在的标准价--------------------------------------
left join (select MATNR,BWKEY,BWTAR,KALNR,row_number() over(partition by MATNR,BWKEY,BWTAR ) AS RN from SDI.SDI_CKMLHD_1001) x5
on A.MATNR = x5.MATNR and A.WERKS = x5.BWKEY and A.BWTAR = x5.BWTAR and x5.RN = 1
left join sdi.sdi_ckmlcr_1001 y5 
on x5.KALNR = y5.KALNR and y5.BDATJ = substr(BB.PRODUCE_DATE,1,4) and substr(y5.POPER,2,2) = substr(BB.PRODUCE_DATE,6,2)
-------------------------------------------------------------------------------------------------------------
left join MC2 y7
on A.MBLNR = y7.MBLNR and A.ZEILE = y7.ZEILE and A.MJAHR = y7.MJAHR and A.SHKZG = y7.SHKZG 
-------------------------------------------------------------------------------------------------------------
where A.period_id_m = '${PERIOD_ID_M}'  ---获取当期变化值
GROUP BY 
 case when A.CHARG = '' then A.CHARG_CID ELSE A.CHARG END  --"批次"
,A.BUKRS -- "公司代码"
,A.WAERS  --"货币",
,A.MATNR --"物料编码",
,A.WERKS  --"工厂",
,substr(A.BSTTYP_SG,1,1)  --"库存类别",
,A.BSTAUS_SG  --"库存类型",
,A.LGORT  --"库存地点",	
,A.MEINS --"计量单位",
,A.MAT_PSPNR  -- "WBS",
,case when A.CHARG = '' then LPAD(A.CHARG_CID,10,'0') ELSE LPAD(A.CHARG,10,'0') END  --"物料批次主数据",
,A.LIFNR --"供应商编码",
,CASE WHEN A.BSTTYP_SG = 'K' AND W.BWTTY <> 'X'   --k库存且评估类别<>X (非批次成本)
	     THEN COALESCE(y1.KBETR,y3.KBETR,yy.KBETR) / COALESCE(y1.KPEIN,y3.KPEIN,yy.KPEIN)  --取当期入库/过账日期采购价，保底期初数取所在信息记录最小一条采购价
	     WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'S' --为标准价
	     THEN COALESCE(y5.STPRS,y7.STPRS) / COALESCE(y5.PEINH,y7.PEINH)  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期入库/过账日期标准价
		 WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'V' --为移动平均价
		 THEN COALESCE(y5.PVPRS,y7.PVPRS) / COALESCE(y5.PEINH,y7.PEINH)   --k库存逻辑  正常按成本估算号找不到，按工厂物料评估类取当期入库/过账日期移动平均价
		 ELSE 0 END --周期价
,A.BWTAR  --"评估类型"
,A.KDAUF  --"销售凭证"
,A.KDPOS  --"项目",
,A.kalnr --成本估算号
,A.SAKTO --科目号
