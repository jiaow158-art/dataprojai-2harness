--CREATE TABLE DWROTD.DWR_WM_RTM_INVENTORY_CT_T_D AS
with dim_date as (    -- 日期维度,按月分区
	select day_dt 
	     from dm.dm_dim_date_d 
	  where  day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	    and day_dt >= add_months(concat(substring('${PERIOD_ID_D}',1,7),'-01'),-1) -- 传昨天所在月份1号的数据
		--and day_dt >='2022-12-04' 
),
SDI_CT_CUSTOMER_DATA_1037_DATA as(    -- RTM B1的客户数据
    select case when length(CUSTOMER_CODE) = 10 then CUSTOMER_CODE  else concat('0000',CUSTOMER_CODE) end as CUSTOMER_CODE,
           t.CUSTOMER_NAME,
           t.data_source_falg ,
           t.operation_center_name 
    from SDI.SDI_CT_CUSTOMER_DATA_1037 t
    where DATA_SOURCE_FALG = 'RTM_B1'
    and NVL(stop_flag,'NULL') <> 'Y'
),
BOM_DETAIL_DATA AS(   -- BOM数据
	select material_num
	   from(
			select t1.MATNR as material_num,  -- 物料编码
				   t2.IDNRK as small_material_num, -- 子项物料编码
			   	   t2.MENGE as small_material_qty  -- 子项物料数量
			from(
				select MATNR,
					   STLNR,
					   row_number() over(partition by MATNR order by ANDAT desc ) as RN
				from SDI.SDI_MAST_1001 
				WHERE STLAN = '5'
			) t1
			left join sdi.sdi_STPO_1001 t2
			on t1.STLNR = t2.STLNR
			where t1.RN = 1
		  )t
		  group by material_num
),
inv_detail_data_01 as  (
	select 
		 substr(creation_date,1,10) day_dt 
		--,transaction_batch_code  -- 批次
		,t2.customer_code AS customer_code  -- 客户别名
		,product_code AS itemcode -- 物料编码
		,transaction_price -- 价格
		,sum(nvl(before_qty,0)) as qc_qty --`期初数量`  -- 期初数量
		,sum(CASE WHEN inv_type = 'IN' AND transaction_type_name <> '调拨入库' THEN NVL(transaction_quantity,0) ELSE 0 END ) AS qj_instock_qty --`本期入库` -- 本期入库
		,sum(CASE WHEN transaction_type_id in (1,4) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_pp_del_qty --`期间采购收货数量` -- 期间采购收货数量
		,sum(CASE WHEN transaction_type_id in (6,7) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_sale_return_qty -- `期间销售退货数量` -- 期间销售退货数量
		,sum(CASE WHEN transaction_type_id in (14,17)   THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_other_instock_qty -- `期间其他入库数量` -- 期间其他入库数量
		,sum(CASE WHEN transaction_type_id in (2,3) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_pp_return_qty -- `期间采购退货数量` -- 期间采购退货数量
		,sum(CASE WHEN inv_type = 'OUT' AND transaction_type_name <> '调拨出库' THEN NVL(transaction_quantity,0) ELSE 0 END ) AS qj_outstock_qty --`本期出库` -- 本期入库
		,sum(CASE WHEN transaction_type_id in (5,8) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_sale_outstock_qty -- `期间销售出库数量` -- 期间销售出库数量
		,sum(CASE WHEN transaction_type_id in (13,18) 	THEN NVL(transaction_quantity,0) ELSE 0 END) AS qj_other_outstock_qty -- `期间其他出库数量` -- 期间其他出库数量
	from SDI.SDI_dtwms_mtl_material_transactions_1031 t1 
	JOIN SDI_CT_CUSTOMER_DATA_1037_DATA   t2 
	on t1.distributor_code = t2.customer_code
	where 1=1 
	and t1.product_code not in (select material_num from BOM_DETAIL_DATA)
	group by  substr(creation_date,1,10)  
			,t2.customer_code
			,product_code 
			--,transaction_batch_code
			,transaction_price
),
inv_detail_data_02 as(
	select
	    t2.day_dt 
		,customer_code
		,itemcode
		--,t1.transaction_batch_code
		,'' as last_instock_date -- `最末入库日期` -- 最末入库日期
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(qj_instock_qty,0) - nvl(qj_outstock_qty,0) else 0  end) as qc_qty --  `期初数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end ) as qj_instock_qty -- `本期入库`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_pp_del_qty,0) else 0 end )  as qj_pp_del_qty -- `期间采购收货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_sale_return_qty,0) else 0 end ) as qj_sale_return_qty -- `期间销售退货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_other_instock_qty,0) else 0 end ) as qj_other_instock_qty --  `期间其他入库数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_pp_return_qty,0) else 0 end ) as qj_pp_return_qty -- `期间采购退货数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end )  as qj_outstock_qty -- `本期出库`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_sale_outstock_qty,0) else 0 end ) as qj_sale_outstock_qty --  `期间销售出库数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_other_outstock_qty,0) else 0 end ) as qj_other_outstock_qty -- `期间其他出库数量`
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(qj_instock_qty,0) - nvl(qj_outstock_qty,0) else 0  end)  + 
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end ) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end ) as qm_qty -- `期末数量`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end ) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end )  as qj_happen_qty -- `期间发生`
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(qj_instock_qty,0) - nvl(qj_outstock_qty,0) else 0  end * t1.transaction_price ) as qc_amt --`期初金额` -- 期初金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end * t1.transaction_price) as bq_instock_amt --`本期入库金额`  -- 本期入库金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end * t1.transaction_price) as bq_outstock_amt --`本期出库金额` -- 本期出库金额
		,sum(case when t1.day_dt< t2.day_dt   then  nvl(qj_instock_qty,0) - nvl(qj_outstock_qty,0) else 0  end * t1.transaction_price)  + 
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end * t1.transaction_price) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end * t1.transaction_price)  as  qm_amt -- `期末金额` -- 期末金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_instock_qty,0) else 0 end * t1.transaction_price) -
			sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_outstock_qty,0) else 0 end * t1.transaction_price) as bq_happen_amt --`本期发生金额`  -- 本期发生金额
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_other_instock_qty,0) else 0 end * t1.transaction_price) as qj_other_instock_amt --  `期间其他入库金额`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_pp_del_qty,0) else 0 end * t1.transaction_price)  as qj_pp_del_amt -- `期间采购收货金额`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_sale_return_qty,0) else 0 end * t1.transaction_price) as qj_sale_return_amt -- `期间销售退货金额`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_sale_outstock_qty,0) else 0 end * t1.transaction_price) as qj_sale_outstock_amt --  `期间销售出库金额`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_other_outstock_qty,0) else 0 end * t1.transaction_price) as qj_other_outstock_amt -- `期间其他出库金额`
		,sum(case when t1.day_dt = t2.day_dt   then  nvl(qj_pp_return_qty,0) else 0 end * t1.transaction_price) as qj_pp_return_amt -- `期间采购退货金额`
	from dim_date t2 
	join inv_detail_data_01 t1 
	on 1=1 
	where t1.day_dt <= t2.day_dt  
	group by 
	customer_code
	,itemcode
	,t2.day_dt 
	--,t1.transaction_batch_code
),
inv_detail_data_03 as(
select t2.data_source_falg,  -- 数据来源标识
	   T1.day_dt,   -- 日期
	   T2.CUSTOMER_CODE,  -- 客户编码
	   T2.CUSTOMER_NAME,  -- 客户名称
	   T2.operation_center_name as sales_org_name, -- 销售组织名称(取数逻辑待定)
	   T1.itemcode as material_num,  -- 物料编码
	   t4.material_name ,-- `SAP物料名称`,
	   t4.menu_desc_name,  -- 是否主菜单
       t4.dimension , -- 规格
	   nvl((t4.length_1 * t4.width) / 1000000,0) as unit_area_qty, -- 单位面积
	   T1.qc_qty ,--`期初数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qc_qty as qc_area_qty, -- 期初面积
	   T1.qc_amt , -- `期初金额`,
	   T1.qm_qty , -- `期末数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qm_qty as qm_area_qty, -- 期末面积
	   T1.qm_amt , -- `期末金额`,
	   T1.qj_instock_qty , -- `本期入库`, 期间入库数量
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_instock_qty as qj_instock_area_qty, -- 期间入库面积
	   T1.bq_instock_amt , --`本期入库金额`,
	   T1.qj_other_instock_qty , -- `期间其他入库数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_other_instock_qty as qj_other_instock_area_qty, -- 期间其他入库面积
	   t1.qj_other_instock_amt, -- 期间其他入库金额
	   T1.qj_pp_del_qty , -- `期间采购收货数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_pp_del_qty as qj_pp_del_area_qty, -- 期间采购收货面积
	   t1.qj_pp_del_amt, -- 期间采购收货金额
	   T1.qj_sale_return_qty , -- `期间销售退货数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_sale_return_qty as qj_sale_return_area_qty , -- 期间销售退货面积
	   t1.qj_sale_return_amt, -- 期间退货金额
	   T1.qj_outstock_qty , -- `本期出库`, 本期出库数量
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_outstock_qty as qj_outstock_area_qty, -- 本期出库面积
	   T1.bq_outstock_amt , -- `本期出库金额`,
	   T1.qj_sale_outstock_qty , -- `期间销售出库数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_sale_outstock_qty as qj_sale_outstock_area_qty, -- 期间销售出库面积
	   t1.qj_sale_outstock_amt , -- 期间销售出库金额
	   T1.qj_other_outstock_qty , -- `期间其他出库数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) *  T1.qj_other_outstock_qty as qj_other_outstock_area_qty, -- 期间其他出库面积
	   t1.qj_other_outstock_amt, -- 期间其他出库金额
	   T1.qj_happen_qty , -- `期间发生`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_happen_qty as qj_happen_area_qty , -- 本期发生面积
	   T1.bq_happen_amt , -- `本期发生金额`,
	   T1.qj_pp_return_qty , -- `期间采购退货数量`,
	   nvl((t4.length_1 * t4.width) / 1000000,0) * T1.qj_pp_return_qty as qj_pp_return_area_qty, -- 期间采购退货面积
	   t1.qj_pp_return_amt, -- 期间采购退货金额
	   T1.last_instock_date , -- `最末入库日期`,
	   '' as user_flag, -- 使用标识
	   t4.material_id ,--`物料ID`,
       t4.matl_type_code ,--  `SAP物料类型`,
       t4.matl_type_name ,-- `SAP物料类型名称`,
       t4.product_group_code ,-- `SAP产品组`,
       t4.product_group_name ,-- `SAP产品组名称`,
       t4.product_classify_code  ,--   `SAP产品分级`,
       t4.product_classify_name   ,--  `SAP产品分级名称`,
       t4.product_level_code    ,--    `SAP层次代码`,
       t4.product_level_name    ,--    `SAP层次名称`,
       t4.product_level1_code   ,--    `SAP层次1`,
       t4.product_level1_name   ,--    `SAP层次1名称`,
       t4.product_level2_code   ,--    `SAP层次2`,
       t4.product_level2_name   ,--    `SAP层次2名称`,
       t4.product_level3_code    ,--   `SAP层次3`,
       t4.product_level3_name    ,--   `SAP层次3名称`,
       t4.product_level4_code    ,--   `SAP层次4`,
       t4.product_level4_name    ,--   `SAP层次4名称`,
   	  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
	  --t1.transaction_batch_code as batch_code, -- 批次
   	  substring(T1.day_dt,1,7) as STAT_MONTH
from inv_detail_data_02 t1
join SDI_CT_CUSTOMER_DATA_1037_DATA  t2  
on t1.customer_code =t2.customer_code  
join dwimd.dwi_md_data_material_general_t t4 
on t1.itemcode = t4.material_num
and t4.prod_property_name = '瓷砖产品'
)
