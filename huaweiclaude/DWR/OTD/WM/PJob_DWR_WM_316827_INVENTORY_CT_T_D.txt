set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
WITH
	TMP AS (
		SELECT
		 DATA_SOURCE_FALG
		,'${PERIOD_ID_D}' as DAY_DT
		,CUSTOMER_CODE
		,CUSTOMER_NAME
		,sales_org_name
		,material_num 
		,material_name
		,menu_desc_name   -- 是否主菜单
		,DIMENSION -- 规格
		,unit_area_qty -- 单位面积
		,qm_qty as qc_qty  --起初数量
		,qm_area_qty as qc_area_qty  --期初面积
		,qm_amt as qc_amt --期初金额
		,qm_qty--期末数量
		,qm_area_qty--期末面积
		,qm_amt --期末金额
		,0 AS QJ_INSTOCK_QTY
		,0 AS QJ_INSTOCK_AREA_QTY
		,0 AS BQ_INSTOCK_AMT
		,0 AS QJ_OTHER_INSTOCK_QTY
		,0 as QJ_OTHER_INSTOCK_AREA_QTY --期间其他入库面积
		,0 AS QJ_OTHER_INSTOCK_AMT --期间其他入库金额
		,0 AS QJ_PP_DEL_QTY  --期间采购收货数量
		,0 AS QJ_PP_DEL_AREA_QTY --期间采购收货面积
        ,0 AS qj_pp_del_area_amt --期间采购收货金额
        ,0 AS QJ_SALE_RETURN_QTY  --期间销售退货数量
        ,0 AS QJ_SALE_RETURN_AREA_QTY --期间销售退货面积
        ,0 as QJ_SALE_RETURN_AMT --期间退货金额
        ,0 AS QJ_OUTSTOCK_QTY  --本期出库数量
        ,0 AS QJ_OUTSTOCK_AREA_QTY --本期出库面积
        ,0 as bq_outstock_amt --本期出库金额
        ,0 as qj_sale_outstock_qty -- 期间销售出库数量
        ,0 as qj_sale_outstock_area_qty  -- 期间销售出库面积
        ,0 as qj_sale_outstock_amt-- "期间销售出库金额" 
        ,0 AS qj_other_outstock_qty -- 期间其他出库数量
        ,0 AS qj_other_outstock_area_qty-- 期间其他出库面积
        ,0 as qj_other_outstock_amt -- "期间其他出库金额"
        ,0 AS qj_happen_qty -- 期间发生,
        ,0 AS qj_happen_area_qty -- 期间发生面积
        ,0 as bq_happen_amt -- 本期发生金额
        ,'' as last_instock_date -- 最末入库日期
	FROM
		DWROTD.DWR_WM_316827_INVENTORY_CT_T_D	t1	
    where period_id_d = date_sub('${PERIOD_ID_D}',1)
UNION ALL
		select
		'SAP_B1' AS DATA_SOURCE_FALG		
		,t1.cdate as  DAY_DT
		,t1.distid as CUSTOMER_CODE
		,t1.distname AS CUSTOMER_NAME
		,'' as sales_org_name
		,t1.itemcode as material_num
		,t1.itemname as material_name
		,T2.menu_desc_name   -- 是否主菜单
		,t2.dimension as DIMENSION
		,nvl((t2.length_1 * t2.width) / 1000000,0) as unit_area_qty -- 单位面积
		,0 as qc_qty  --起初数量
		,0 as qc_area_qty  --期初面积
		,0 as qc_amt --期初金额
		,case
				when t1.objtype in  ('13','60','19')   then t1.quantity * -1
				else t1.quantity
			end qm_qty --期末数量
        ,case
				when t1.objtype in  ('13','60','19')  then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity *-1 
            else nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity end  AS qm_area_qty --期末面积
        ,0 as qm_amt --期末金额
        ,CASE WHEN objtype IN ('19','18','59') then t1.quantity  else 0  end  as  QJ_INSTOCK_QTY  --期间入库数量
        ,CASE WHEN objtype IN ('19','18','59') then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity  end as QJ_INSTOCK_AREA_QTY --期间入库面积
        ,0 AS BQ_INSTOCK_AMT --期间入库金额
        ,CASE WHEN objtype IN ('59') then t1.quantity  else 0  end  as  QJ_OTHER_INSTOCK_QTY --期间其他入库数量
        ,CASE WHEN objtype IN ('59') then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity  end 
        as QJ_OTHER_INSTOCK_AREA_QTY --期间其他入库面积
        ,0 AS QJ_OTHER_INSTOCK_AMT --期间其他入库金额
        ,CASE WHEN objtype IN ('18') then t1.quantity  else 0  end  as  QJ_PP_DEL_QTY  --期间采购收货数量
        ,CASE WHEN objtype IN ('18')then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity  end 
         as QJ_PP_DEL_AREA_QTY --期间采购收货面积
        ,0 AS qj_pp_del_area_amt --期间采购收货金额
        ,CASE WHEN objtype IN ('19') then t1.quantity  else 0  end  as  QJ_SALE_RETURN_QTY  --期间销售退货数量
        ,CASE WHEN objtype IN ('19')then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity  end
        as QJ_SALE_RETURN_AREA_QTY --期间销售退货面积
        ,0  as QJ_SALE_RETURN_AMT --期间退货金额
        ,CASE WHEN objtype IN ('13','60') then t1.quantity  else 0  end  as  QJ_OUTSTOCK_QTY  --本期出库数量
        ,CASE WHEN objtype IN ('13','60')then nvl((t2.length_1 * t2.width) / 1000000,0) * t1.quantity  
        end as QJ_OUTSTOCK_AREA_QTY --本期出库面积
      ,0 as  bq_outstock_amt--本期出库金额
      , 0 as qj_sale_outstock_qty -- 期间销售出库数量,
       ,0 as qj_sale_outstock_area_qty  -- 期间销售出库面积
      , 0 as qj_sale_outstock_amt -- "期间销售出库金额" 
       ,0 AS qj_other_outstock_qty -- 期间其他出库数量,
       ,0 AS qj_other_outstock_area_qty-- 期间其他出库面积
       ,0 as qj_other_outstock_amt -- "期间其他出库金额"
       ,0 AS qj_happen_qty -- 期间发生,
       ,0 AS qj_happen_area_qty -- 期间发生面积
       ,0 as bq_happen_amt -- 本期发生金额
       ,'' as last_instock_date -- 最末入库日期
from
		SDI.SDI_KCDATA_316827_NEW_1037 t1
		left join 
		dwimd.dwi_md_data_material_general_t t2
        on t1.itemcode = t2.material_num
        where  cdate = '${PERIOD_ID_D}'
 )
 INSERT OVERWRITE TABLE DWROTD.DWR_WM_316827_INVENTORY_CT_T_D PARTITION(PERIOD_ID_D)
select 
data_source_falg,
day_dt,
customer_code,
customer_name,
sales_org_name,
material_num,
material_name,
menu_desc_name,
dimension,
SUM(unit_area_qty),
SUM(qc_qty),
SUM(qc_area_qty),
SUM(qc_amt),
SUM(qm_qty),
SUM(qm_area_qty),
SUM(qm_amt),
SUM(qj_instock_qty),
SUM(qj_instock_area_qty),
SUM(bq_instock_amt),
SUM(qj_other_instock_qty),
SUM(qj_other_instock_area_qty),
SUM(qj_other_instock_amt),
SUM(qj_pp_del_qty),
SUM(qj_pp_del_area_qty),
SUM(qj_pp_del_area_amt),
SUM(qj_sale_return_qty),
SUM(qj_sale_return_area_qty),
SUM(qj_sale_return_amt),
SUM(qj_outstock_qty),
SUM(qj_outstock_area_qty),
SUM(bq_outstock_amt),
SUM(qj_sale_outstock_qty),
SUM(qj_sale_outstock_area_qty),
SUM(qj_sale_outstock_amt),
SUM(qj_other_outstock_qty),
SUM(qj_other_outstock_area_qty),
SUM(qj_other_outstock_amt),
SUM(qj_happen_qty),
SUM(qj_happen_area_qty),
SUM(bq_happen_amt),
last_instock_date,
current_timestamp as  dw_last_update_date,
'${PERIOD_ID_D}'  as   period_id_d
from  
tmp 
group by 
data_source_falg,
day_dt,
customer_code,
customer_name,
sales_org_name,
material_num,
material_name,
menu_desc_name,
dimension,
last_instock_date
