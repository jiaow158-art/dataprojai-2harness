---SAP供应商特殊库存(O)
set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_VENDOR_STOCK_O_F		--DWR.SAP供应商特殊库存(O)
SELECT
	 DWVSO.GROUP_NUM							AS	GROUP_NUM
	,DWVSO.MATERIAL_ID                        AS	MATERIAL_ID
	,DWVSO.MATERIAL_NUM                       AS	MATERIAL_NUM
	,T.MATERIAL_NAME                      AS	MATERIAL_NAME
	,DWVSO.FACTORY_ID                         AS	FACTORY_ID
	,DWVSO.FACTORY_CODE                       AS	FACTORY_CODE
	,T1.FACTORY_NAME                       AS	FACTORY_NAME
	,DWVSO.BATCH_NUM                          AS	BATCH_NUM
	,DWVSO.SPECIAL_STOCK_FLAG                 AS	SPECIAL_STOCK_FLAG
	,T2.CONFIG_NAME                        AS	SPECIAL_STOCK_FLAG_NAME
	,DWVSO.SUPPLIER_OR_CREDITOR_ID            AS	SUPPLIER_OR_CREDITOR_ID
	,DWVSO.SUPPLIER_OR_CREDITOR_NUM           AS	SUPPLIER_OR_CREDITOR_ACCT
	,T3.SUPPLIER_OR_CREDITOR_NAME          AS	SUPPLIER_OR_CREDITOR_NAME
	,DWVSO.CURRENT_STOCK_FIN_YEAR             AS	CURRENT_STOCK_FIN_YEAR
	,DWVSO.CREATE_DATE                        AS	CREATE_DATE
	,DWVSO.CURRENT_PERIOD                     AS	CURRENT_PERIOD
	,DWVSO.CURRENT_PERIOD_FISCAL_YEAR         AS	CURRENT_PERIOD_FISCAL_YEAR
	,DWVSO.LIMIT_BATCH_TOTOAL_STOCK           AS	LIMIT_BATCH_TOTOAL_STOCK
	,DWVSO.CHECK_STOCK                        AS	CHECK_STOCK
	,DWVSO.WX_WORK_TRANSFER_STOCK             AS	WX_WORK_TRANSFER_STOCK
	,DWVSO.LAST_QUALITY_CHECK_STOCK           AS	LAST_QUALITY_CHECK_STOCK
	,DWVSO.LAST_PASS_CHECK_DATE               AS	LAST_PASS_CHECK_DATE
	,DWVSO.LAST_LIMIT_USE_STOCK               AS	LAST_LIMIT_USE_STOCK
	,DWVSO.DEL_FLAG                           AS	DEL_FLAG
	,CURRENT_TIMESTAMP()                AS	DW_LAST_UPDATE_DATE
FROM DWIOTD.DWI_WM_VENDOR_STOCK_O_T DWVSO		--DWI.SAP供应商特殊库存(O)
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T		--关联物料主数据取物料名称
	 ON T.MATERIAL_ID = DWVSO.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DWVSO.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWVSO.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T1				--关联工厂主数据取工厂名称
	 ON  T1.FACTORY_ID = DWVSO.FACTORY_ID		--工厂ID
	 AND T1.GROUP_NUM = DWVSO.GROUP_NUM			--集团
	 AND T1.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWVSO.DEL_FLAG = 'N'					--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2	--关联通用配置表
	 ON T2.CONFIG_CODE = DWVSO.SPECIAL_STOCK_FLAG	--取特殊库存标识名称中文名称
	 AND T2.GROUP_NUM = DWVSO.GROUP_NUM			--集团
	 AND T2.LANG_CODE = '1'
INNER JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T T3	--关联供应商主数据取供应商名称
	 ON T3.SUPPLIER_OR_CREDITOR_ID = DWVSO.SUPPLIER_OR_CREDITOR_ID	--供应商ID
	 AND T3.GROUP_NUM = DWVSO.GROUP_NUM			--集团
	 AND T3.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWVSO.DEL_FLAG = 'N'					--删除标识为'N'
WHERE DWVSO.DEL_FLAG = 'N'
