with tem as 
(
select 
	bdt.`month` as  start_month,
	dp.sales_center_code,-- 运营中心  
	dp.sales_center_name, -- 运营中心   
	dp.sales_dep_code,  -- 部门
	dp.sales_dep_name,  -- 部门
	dp.sales_region_code, -- 销区编码
	dp.sales_region_name, -- 销区名称
	dp.dp_obj_code,   -- 预测对象编码
	dp.dp_obj_name,   -- 预测对象名称
	bdsm.dp_cate,
	bdsm.dp_dimension, 
	bdsm.integrate_channel,
	bdsm.integrate_channel_name,
	bdsm.amount ,    -- 预测金额（万元）（不含1+N） + amount_n1
	bdsm.amount_n1,
	bdsm.amount_n2,
	bdsm.area,     -- 预测面积（万方）（不含1+N） + area_n1
	bdsm.area_n1,
	bdsm.area_n2,
	CURRENT_TIMESTAMP() as dw_last_update_date,
	0 as n_amount,
	0 as n_amount_n1,
	0 as n_amount_n2
from SDI.SDI_CZDP_biz_dp_cate_1045 bdsm
inner join SDI.SDI_CZDP_BIZ_DP_1045 dp on dp.id = bdsm.dp_id 
inner join SDI.SDI_CZDP_BIZ_DP_TASK_1045 bdt on bdt.id = dp.task_id 
where 1=1 
and dp.delete_flag = 0
and bdsm.delete_flag = 0
union ALL 
select 
	bdt.`month` as start_month,
	dp.sales_center_code,
	dp.sales_center_name ,
	dp.sales_dep_code,
	dp.sales_dep_name ,
	dp.sales_region_code,
	dp.sales_region_name,
	dp.dp_obj_code,
	dp.dp_obj_name ,
	'' as dp_cate,
	'' as dp_dimension, 
	dp.integrate_channel,
	dp.integrate_channel_name,
	0 as amount ,    -- 预测金额（万元）（不含1+N） + amount_n1
	0 as amount_n1,
	0 as amount_n2,
	0 as area,     -- 预测面积（万方）（不含1+N） + area_n1
	0 as area_n1,
	0 as area_n2,
	CURRENT_TIMESTAMP() as dw_last_update_date,
	bdsm.amount  as n_amount,--N金额
	bdsm.amount_n1 as n_amount_n1,
	bdsm.amount_n2 as n_amount_n2
from SDI.SDI_CZDP_biz_dp_1N_1045 bdsm
join SDI.SDI_CZDP_BIZ_DP_1045 dp on bdsm.dp_id = dp.id
join SDI.SDI_CZDP_BIZ_DP_TASK_1045 bdt on bdt.id=dp.task_id
where 1=1
and dp.delete_flag =0
and bdsm.dp_item='1+N预测'
)
INSERT OVERWRITE TABLE dwrotd.dwr_wm_sales_predict_t
select 
	start_month,
	sales_center_code,
	sales_center_name ,
	sales_dep_code,
	sales_dep_name ,
	sales_region_code,
	sales_region_name,
	case 
	    when start_month = '2025-09' and dp_obj_code = 'H07250801' then 'Z122'
	    when start_month = '2025-09' and dp_obj_code = 'H06230640' then 'Z121'
	    else dp_obj_code
	end dp_obj_code,
	dp_obj_name ,
	dp_cate,
	dp_dimension, 
	integrate_channel,
	integrate_channel_name,
	amount ,    -- 预测金额（万元）（不含1+N） + amount_n1
	amount_n1,
	amount_n2,
	area,     -- 预测面积（万方）（不含1+N） + area_n1
	area_n1,
	area_n2,
	CURRENT_TIMESTAMP() as dw_last_update_date,
	n_amount,--N金额
	n_amount_n1,
	n_amount_n2
from tem
where replace(start_month,'-','') not in (select distinct substring(replace(start_month,'-',''),1,6) from sdi.sdi_wm_sales_predict_t_1037)
