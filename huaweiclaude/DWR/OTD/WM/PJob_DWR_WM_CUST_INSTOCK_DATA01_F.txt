set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;

--CREATE TABLE DWROTD.DWR_WM_CUST_INSTOCK_DATA01_F AS -- 湖南江西广西福建客户入库数
INSERT OVERWRITE TABLE DWROTD.DWR_WM_CUST_INSTOCK_DATA01_F PARTITION(STAT_MONTH)
select day_dt, 
	   customer_code,
	   material_num,
	   sum(PUR_QTY) as PUR_QTY,
	   current_timestamp() as dw_last_update_date,
	   sum(qj_pp_return_qty) as qj_pp_return_qty,
	   substring(day_dt,1,7) as stat_month
from(
	SELECT 
		a.period_id_d as day_dt,
		a.customer_code,
		a.b1_material_num as material_num,
		sum(a.qj_pp_del_qty - qj_pp_return_qty) as PUR_QTY,  -- 期间采购收货数量 - 期间采购退货数量
		current_timestamp() as dw_last_update_date ,
		sum(qj_pp_return_qty) as qj_pp_return_qty
	FROM dwrcs.dwr_dea_b1_inventory_sw_t a
	where (a.analyze_product_class	<> '齐套' or a.analyze_product_class is null) 
	--and a.period_id_d >= '2023-01-01'
	and a.period_id_d >= CASE WHEN substring('${PERIOD_ID_D}',-2) IN ('01','02','03') THEN 
                                   ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-1) -- 传昨天，所在月份1号的前1个月
                               ELSE
                        		   CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01')  -- 传昨天，当月
                           END
	and a.customer_code in ('0000310635','0000353879','0000321104','0000318683')
	group by a.period_id_d,
			 a.customer_code,
			 a.b1_material_num
	union all
	SELECT 
		 a.period_id_d as day_dt,
		 a.customer_code,
		 b.bom_material_num  as material_num,
		 --b.material_num,
		 sum((a.qj_pp_del_qty - a.qj_pp_return_qty) * b.component_num ) as PUR_QTY,
		 current_timestamp() as dw_last_update_date ,
		 sum(a.qj_pp_return_qty*b.component_num) as qj_pp_return_qty
	FROM dwrcs.dwr_dea_b1_inventory_sw_t a
	left join DWRDIM.DWR_DIM_MATERIAL_SALES_BOM_F b 
	on a.b1_material_num = b.material_num 
	--and b.rn = 1
	where 1 = 1 
	--and a.analyze_product_class = '齐套' 
	--and a.period_id_d >= '2023-01-01'
	and a.period_id_d >= CASE WHEN substring('${PERIOD_ID_D}',-2) IN ('01','02','03') THEN 
                                   ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-1) -- 传昨天，所在月份1号的前1个月
                               ELSE
                        		   CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01')  -- 传昨天，当月
                           END
	and a.customer_code in ('0000310635','0000353879','0000321104','0000318683')
	group by a.period_id_d,
			 a.customer_code,
			 b.bom_material_num
			 --b.material_num
)T
group by day_dt, 
		 customer_code,
		 material_num;
