---SAP项目库存表
INSERT OVERWRITE TABLE DWROTD.DWR_WM_PROJECT_STOCK_F  --DWR.SAP项目库存表
SELECT
	 DWPS.GROUP_NUM						 AS	GROUP_NUM
	,DWPS.MATERIAL_ID                    AS	MATERIAL_ID
	,DWPS.MATERIAL_NUM                   AS	MATERIAL_NUM
	,T.MATERIAL_NAME                  	 AS	MATERIAL_NAME
	,DWPS.FACTORY_ID                     AS	FACTORY_ID
	,DWPS.FACTORY_CODE                   AS	FACTORY_CODE
	,T1.FACTORY_NAME                     AS	FACTORY_NAME
	,DWPS.STOCK_PLACE                    AS	STOCK_PLACE
	,DWPS.BATCH_NUM                      AS	BATCH_NUM
	,DWPS.SPECIAL_STOCK_FLAG             AS	SPECIAL_STOCK_FLAG
	,T2.CONFIG_NAME                      AS	SPECIAL_STOCK_FLAG_NAME
	,DWPS.WBS_ELEMENT                    AS	WBS_ELEMENT
	,DWPS.LAST_MXXX_PERIOD               AS	LAST_MXXX_PERIOD
	,DWPS.CURRENT_STOCK_FIN_YEAR         AS	CURRENT_STOCK_FIN_YEAR
	,DWPS.CREATE_DATE                    AS	CREATE_DATE
	,DWPS.CURRENT_PERIOD                 AS	CURRENT_PERIOD
	,DWPS.CURRENT_PERIOD_FISCAL_YEAR     AS	CURRENT_PERIOD_FISCAL_YEAR
	,DWPS.FROZEN_STOCK                   AS	FROZEN_STOCK
	,DWPS.LAST_FROZEN_STOCK              AS	LAST_FROZEN_STOCK
	,DWPS.LIMIT_BATCH_TOTOAL_STOCK       AS	LIMIT_BATCH_TOTOAL_STOCK
	,DWPS.CHECK_STOCK                    AS	CHECK_STOCK
	,DWPS.LAST_QUALITY_CHECK_STOCK       AS	LAST_QUALITY_CHECK_STOCK
	,DWPS.UNRESTRICTED_EVALUATED_STOCK   AS UNRESTRICTED_EVALUATED_STOCK
	,DWPS.LAST_PASS_CHECK_DATE           AS	LAST_PASS_CHECK_DATE
	,DWPS.LAST_LIMIT_USE_STOCK           AS	LAST_LIMIT_USE_STOCK
    ,DWPS.DEL_FLAG                       AS DEL_FLAG
    ,CURRENT_TIMESTAMP()                 AS DW_LAST_UPDATE_DATE
FROM DWIOTD.DWI_WM_PROJECT_STOCK_T DWPS			--DWI.SAP项目库存表
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T		--关联物料主数据取物料名称
	 ON T.MATERIAL_ID = DWPS.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DWPS.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWPS.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T1				--关联工厂主数据取工厂名称
	 ON  T1.FACTORY_ID = DWPS.FACTORY_ID		--工厂ID 
	 AND T1.GROUP_NUM = DWPS.GROUP_NUM			--集团
	 AND T1.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWPS.DEL_FLAG = 'N'					--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表
	 ON T2.CONFIG_CODE = DWPS.SPECIAL_STOCK_FLAG	--取特殊库存标识名称中文名称	
	 AND T2.GROUP_NUM = DWPS.GROUP_NUM				--集团
	 AND T2.LANG_CODE = '1'
WHERE DWPS.DEL_FLAG = 'N'
