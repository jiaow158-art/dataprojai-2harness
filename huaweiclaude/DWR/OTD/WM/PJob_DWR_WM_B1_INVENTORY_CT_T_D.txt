--CREATE TABLE DWROTD.DWR_WM_B1_INVENTORY_CT_T_D AS
with SAP_B1_CUSTOMER_DATA AS(  -- SAP B1的客户数据
	select customer_code,
		   customer_name,
		   data_source_falg,
		   operation_center_name
	from SDI.SDI_CT_CUSTOMER_DATA_1037 T 
	where T.data_source_falg = 'SAP_B1'
	and NVL(stop_flag,'NULL') <> 'Y'
),
OIVL_DATA as(  -- 库存交易账
	select 
		 docdate
		,transseq
		,loccode
		,transtype
		,itemcode
		,messageid
		,t3.customer_code
		,sum(inqty) inqty
		,sum(outqty) outqty 
		,sum(inqty)-sum(outqty) bqqty
		,createdate 
	from dwiotd.dwi_wm_b1_oivl_t t1 
	join SAP_B1_CUSTOMER_DATA t3 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t3.customer_code 
	join dwimd.dwi_md_data_material_general_t t4 on t1.itemcode = t4.material_num
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	group by docdate,
			 transseq,
			 loccode,
			 transtype,
			 itemcode,
			 messageid,
			 t3.customer_code,
			 createdate 
),
IVL1_DATA as ( -- 库存交易账行
	select transseq,
		   t2.customer_code,
		  sum(transvalue) transvalue --,
		 -- sum(CalcPrice) cost_price -- 成本单价 
	from dwiotd.dwi_wm_b1_ivl1_t  t1 
	join SAP_B1_CUSTOMER_DATA t2 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code 
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	group by transseq,
			 t2.customer_code
),
OITM_DATA as (   -- 物料
	select itemcode,
		   itemname,
		   codebars,
		   frgnname,
		   suppcatnum,
		   invntryuom,
		   cardcode,
		   itmsgrpcod,
		   firmcode,
		   slength1,
		   swidth1,
		   t2.customer_code 
	  from dwiotd.dwi_wm_b1_oitm_t t1 
	  join SAP_B1_CUSTOMER_DATA t2 
	  on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	  where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	  group by itemcode,
			   itemname,
			   codebars,
			   frgnname,
			   suppcatnum,
			   invntryuom,
			   cardcode,
			   itmsgrpcod,
			   firmcode,
			   slength1,
			   swidth1,
			   t2.customer_code 
), 
OITB_DATA as ( -- 物料组
	select itmsgrpcod,
		  itmsgrpnam,
		  t2.customer_code
	from dwiotd.dwi_wm_b1_oitb_t t1 
	join SAP_B1_CUSTOMER_DATA t2 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	group by itmsgrpcod,
			 itmsgrpnam,
			 t2.customer_code
),
OMRC_DATA as (  -- 制造商
	select firmcode,
		   firmname,
		   t2.customer_code
	from dwiotd.dwi_wm_b1_omrc_t  t1 
	join SAP_B1_CUSTOMER_DATA t2 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	group by  firmcode,
			  firmname,
			  t2.customer_code
), 
OILM_DATA as ( -- 库存日志表
	select messageid,
		   effectqty,
		   bpcardcode,
		   t2.customer_code
	from dwiotd.dwi_wm_b1_oilm_t t1 
	join SAP_B1_CUSTOMER_DATA t2 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	 group by messageid,
			  effectqty,
			  bpcardcode,
			  t2.customer_code
),
OCRD_DATA as (  -- 客户
	select cardcode,
		   cardname,
		   t2.customer_code 
	from dwiotd.dwi_wm_b1_ocrd_t t1 
	join SAP_B1_CUSTOMER_DATA t2 
	on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	where (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	 group by cardcode,
			  cardname,
			  t2.customer_code 
), 
DIM_DATE as (  -- 日期维度
  select day_dt 
	from dm.dm_dim_date_d 
	where  day_dt >= date_sub('${PERIOD_ID_D}',3)  -- 昨天的前3天
	and day_dt <= '${PERIOD_ID_D}'  -- 传昨天
	-- day_dt >= '2023-06-01'
	--and day_dt <= '2023-06-30'
	/*
	 * select CASE WHEN SUBSTRING('${PERIOD_ID_D}',9,2) <= '03' THEN ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-1) 
       ELSE CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01') END  -- 传昨天，每月1,2,3 跑上月1号到昨天，4号以后跑当月
	 * */
),
PDN1_DATA AS ( -- 销售退货单行
	 select   
		t2.customer_code
		,t3.itemcode
		,t3.dscription
		,substr(dt.day_dt,1,10) day_dt
		,t1.docdate
		,row_number() over(partition by  t2.customer_code,t3.itemcode,dt.day_dt order by substr(t1.docdate,1,10)  desc ) rk 
	 from DWIOTD.DWI_WM_b1_OPDN_T t1  
	 join SAP_B1_CUSTOMER_DATA t2 
	 on case when length(t1.customer_code) < 10 then concat('0000',t1.customer_code) else t1.customer_code end = t2.customer_code
	 and (t1.customer_type ='CT' OR t1.customer_code IN('310597','321991') )
	 join DWIOTD.DWI_WM_b1_PDN1_T t3 on t1.customer_code =t3.customer_code and t1.docentry =t3.docentry
	 join DIM_DATE dt on 1=1 
	 where 1=1  
	 and t1.docdate<=dt.day_dt
),
SAP_B1_DETAIL_DATA_01 as (
	  select 
	       dt.day_dt,
		   case when t0.docdate < dt.day_dt then 'qc' 
				when t0.docdate  >=  dt.day_dt and  t0.docdate< date_sub(dt.day_dt,-1) then 'qj' 
				else 'qm'  end as dttype,       
			t0.itemcode,
			t2.codebars,
			t2.frgnname,
			t2.suppcatnum,
			t2.invntryuom,
			t3.itmsgrpnam,
			t4.firmname,
			t7.cardcode,
			t7.cardname,
			t0.inqty,
			t0.outqty,       
			case when nvl(t1.transvalue,0.0)>0.0 then t1.transvalue 
				 else 0.0 
			end as inlinebalance,       
			case when nvl(t1.transvalue,0.0)<0.0 then -t1.transvalue 
				 else 0.0 
			end as outlinebalance,	       
			case when (t0.transtype=18 or t0.transtype=20) then t0.inqty 
				 else 0.0 
			end as puch_inqty,  
			case when (t0.transtype=18 or t0.transtype=20) then t1.transvalue 
				 else 0.0 
			end as puch_in_amt,  
			case when (t0.transtype=14 or t0.transtype=16) then t0.inqty 
				else 0.0
			end as sale_inqty,     
			case when (t0.transtype=14 or t0.transtype=16) then t1.transvalue
				else 0.0
			end as sale_in_amt, 
			case when nvl(t1.transvalue,0.0)>=0.0 and t0.transtype<>18 and t0.transtype<>20 and t0.transtype<>14 and t0.transtype<>16 then t0.inqty 
				else 0.0 
			end as other_inqty,       
			--
			case when nvl(t1.transvalue,0.0)>=0.0 and t0.transtype<>18 and t0.transtype<>20 and t0.transtype<>14 and t0.transtype<>16 then t1.transvalue
				else 0.0 
			end as other_in_amt, 
			--
			case when (t0.transtype=19 or t0.transtype=21) then t0.outqty 
				else 0.0 
			end as puch_outqty,        
			case when (t0.transtype=13 or t0.transtype=15) then t0.outqty 
				else 0.0 
			end as sale_outqty,      
			case when (t0.transtype=13 or t0.transtype=15) then t1.transvalue 
				else 0.0 
			end as sale_outamt,  
			case when nvl(t1.transvalue,0.0)<=0.0 and t0.transtype<>19 and t0.transtype<>21 and t0.transtype<>13 and t0.transtype<>15 then t0.outqty 
				else 0.0 
			end as other_outqty,   
			case when nvl(t1.transvalue,0.0)<=0.0 and t0.transtype<>19 and t0.transtype<>21 and t0.transtype<>13 and t0.transtype<>15 then t1.transvalue 
				else 0.0 
			end as other_outamt,
			t1.transvalue as linebalance,
			--t1.cost_price,
			t0.loccode as loccode ,
			t0.transseq,
			t0.customer_code
	from oivl_data t0  
	inner join ivl1_data t1 on t0.transseq = t1.transseq  and t0.customer_code = t1.customer_code         
	inner join oitm_data t2  on t0.itemcode = t2.itemcode  and t0.customer_code = t2.customer_code      
	inner join oitb_data t3  on t2.itmsgrpcod=t3.itmsgrpcod  and t2.customer_code = t3.customer_code
	inner join omrc_data t4  on t2.firmcode=t4.firmcode    and t2.customer_code = t4.customer_code
	left join oilm_data t5 on nvl(t0.messageid,'') = nvl(t5.messageid,'') and nvl(t5.effectqty,0.0)<>0.0  and t0.customer_code = t5.customer_code
	left join ocrd_data t6 on nvl(t5.bpcardcode,'') = nvl(t6.cardcode,'')    and t5.customer_code = t6.customer_code   
	left join ocrd_data t7 on nvl(t2.cardcode,'')=nvl(t7.cardcode,'')        and t2.customer_code = t7.customer_code 
    join DIM_DATE dt on 1=1 
)

DM_WM_B1_INVENTORY_T_D_DATA AS(
select  
	 tt.day_dt as  day_dt -- "日期"
	,tt.customer_code
	,tt.itemcode   as b1_material_num --  "物料编码"
	,sum(nvl(tt.qcinqty,0.0))-sum(nvl(tt.qcoutqty,0.0)) as qc_qty -- "期初数量"
	,sum(nvl(tt.qjinqty,0.0)) as qj_instock_qty -- "本期入库"
	,sum(nvl(tt.qjpuch_inqty,0.0)) as qj_pp_del_qty -- "期间采购收货数量"
	,sum(nvl(tt.qjsale_inqty,0.0)) as qj_sale_return_qty -- "期间销售退货数量"
	,sum(nvl(tt.qjother_inqty,0.0)) as qj_other_instock_qty -- "期间其他入库数量"
	,sum(nvl(tt.qjpuch_outqty,0.0)) as qj_pp_return_qty -- "期间采购退货数量"
	,sum(nvl(tt.qjoutqty,0.0)) as qj_outstock_qty -- "本期出库"
	,sum(nvl(tt.qjsale_outqty,0.0)) as qj_sale_outstock_qty -- "期间销售出库数量" 
	,sum(nvl(tt.qjother_outqty,0.0)) as qj_other_outstock_qty -- "期间其他出库数量" 
	,(sum(nvl(tt.qcinqty,0.0))-sum(nvl(tt.qcoutqty,0.0)))+(sum(nvl(tt.qjinqty,0.0))-sum(nvl(tt.qjoutqty,0.0))) as qm_qty -- "期末数量"
	,sum(nvl(tt.qjinqty,0.0))-sum(nvl(tt.qjoutqty,0.0)) as qj_happen_qty -- "期间发生"
	,sum(nvl(tt.qclinebalance,0)) as qc_amt -- "期初金额"
	,sum(nvl(tt.qjinlinebalance,0.0)) as bq_instock_amt -- "本期入库金额"
	,sum(nvl(tt.qjoutlinebalance,0.0)) as bq_outstock_amt -- "本期出库金额"
	,sum(nvl(tt.qclinebalance,0.0))+sum(nvl(tt.qjlinebalance,0.0)) as qm_amt -- "期末金额"
	,sum(nvl(tt.qjlinebalance,0.0)) as bq_happen_amt -- "本期发生金额"
	,sum(nvl(tt.qjother_amt,0.0)) as qj_other_instock_amt -- 期间其它入库金额
	,sum(nvl(tt.qjpuch_inamt,0.0)) as qj_pp_del_area_amt -- 期间采购收货金额
	,sum(nvl(tt.qjsale_inamt,0.0)) as qj_sale_return_amt -- "期间销售退货金额"
	,sum(nvl(tt.qjsale_outamt,0.0)) as qj_sale_outstock_amt -- "期间销售出库金额" 
	,sum(nvl(tt.qjother_outamt,0.0)) as qj_other_outstock_amt -- "期间其他出库金额"  
	,t8.docdate as last_instock_date -- "最末入库日期"
from  SAP_B1_DETAIL_DATA_02  tt 
join oitm_data t4  on tt.itemcode=t4.itemcode AND tt.customer_code  =t4.customer_code 
left join pdn1_data t8 on t4.itemcode=t8.itemcode AND t4.customer_code  =t8.customer_code and tt.day_dt = t8.day_dt and rk = 1 
group by  tt.customer_code,
		  tt.itemcode,
		  t4.itemname,
		  t8.docdate,
          tt.day_dt 
)

INSERT OVERWRITE TABLE DWROTD.DWR_WM_B1_INVENTORY_CT_T_D PARTITION(PERIOD_ID_D)
select t2.data_source_falg, -- 数据来源标识
	   t1.day_dt, -- 日期,
	   t1.customer_code, -- 客户编码
	   t2.customer_name, -- 客户名称
	   t2.operation_center_name as sales_org_name, -- 销售组织名称(待定)
	   t1.b1_material_num as material_num,  -- 物料编码
	   t3.material_name ,-- `SAP物料名称`,
	   t3.menu_desc_name,  -- 是否主菜单
       t3.dimension , -- 规格
       nvl((t3.length_1 * t3.width) / 1000000,0) as unit_area_qty, -- 单位面积
       t1.qc_qty, -- 期初数量,
       nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qc_qty as qc_area_qty, -- 期初面积
       t1.qc_amt, -- 期初金额,
       t1.qm_qty, -- 期末数量,
       nvl((t3.length_1 * t3.width) / 1000000,0) * t1.qm_qty as qm_area_qty, -- 期末面积
       t1.qm_amt, -- 期末金额,
       t1.qj_instock_qty, -- 本期入库,
       nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qj_instock_qty as qj_instock_area_qty, -- 期间入库面积
       t1.bq_instock_amt, -- 本期入库金额,
       t1.qj_other_instock_qty, -- 期间其他入库数量,
	   nvl((t3.length_1 * t3.width) / 1000000,0) * T1.qj_other_instock_qty as qj_other_instock_area_qty, -- 期间其他入库面积
       t1.qj_other_instock_amt, -- 期间其他入库金额,
       t1.qj_pp_del_qty, -- 期间采购收货数量,
       nvl((t3.length_1 * t3.width) / 1000000,0)  * t1.qj_pp_del_qty as qj_pp_del_area_qty, -- 期间采购收货面积
       t1.qj_pp_del_area_amt, -- 期间采购收货金额
       t1.qj_sale_return_qty, -- 期间销售退货数量,
       nvl((t3.length_1 * t3.width) / 1000000,0)  * t1.qj_sale_return_qty as qj_sale_return_area_qty, -- 期间销售退货面积
       t1.qj_sale_return_amt,-- 期间销售退货金额,
       t1.qj_outstock_qty, -- 本期出库,
       nvl((t3.length_1 * t3.width) / 1000000,0)  *  t1.qj_outstock_qty as qj_outstock_area_qty, -- 本期出库面积,
