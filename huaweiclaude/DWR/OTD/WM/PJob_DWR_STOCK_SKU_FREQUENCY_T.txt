with DWR_WM_PRODSTOCK_OUT_IN_F as (
select 
t1.VOUCHER_POST_DATE
,t1.factory_werks_code --工厂代码
,t1.material_num --物料编码
,t1.inv_location --库存地点
,t2.matl_group1_code MVGR1 --物料组1
,t4.reserved_field_5 yl05 --型号
,t4.reserved_field_6 yl06
,case 
	when inv_location in  ('SA','SC','GCSA') then '江西基地'
	when inv_location in  ('CA','CD','GOCA') then '清远基地'
	when inv_location in  ('TA','GCTA') then '湖南基地'
	when inv_location in  ('QA','GCQA') then '重庆基地'
end as jd_name
,case when t1.move_type in ('541','542','601','602','641','642','643','644','687','688','901',
			'902','927','928','931','932','933','934')  
	then 
		-1*t1.stock_qty
	else 0 end out_stock_qty--出库数量
from dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join (select material_num,max(matl_group1_code)  matl_group1_code from dwimd.dwi_md_data_material_sales_t group by material_num) t2 on t1.material_num = t2.material_num
left join dwimd.dwi_md_data_material_general_t  t4 on t1.material_num = t4.material_num
where 1=1
and t4.prod_property in ('A0001','A0002')
and t1.VOUCHER_POST_DATE >= date_add('${PERIOD_ID_D}',-91)
and t1.VOUCHER_POST_DATE < date_add('${PERIOD_ID_D}',1)
and t1.inv_location in  ('SA','SC','GCSA','CA','CD','GOCA','TA','GCTA','QA','GCQA')
)

--条件2：
,tj2_1 as 
(
select 
	t1.jd_name,
	t1.material_num,
	sum(out_stock_qty) as out_stock_qty 
from DWR_WM_PRODSTOCK_OUT_IN_F t1
where 1=1
and t1.MVGR1 in ('D','T')
and t1.yl06 in ('A')
group by 
t1.jd_name,
t1.material_num
)
,tj2_2 as 
(
select 
	t1.jd_name,
	t1.material_num,
	ROW_NUMBER() OVER (partition by t1.jd_name ORDER BY out_stock_qty DESC) as rnk,
	COUNT(1) OVER (partition by t1.jd_name) AS total_sku_count
from tj2_1 t1
)

--条件3：
,tj3_1 as 
(
select 
	t1.jd_name,
	t1.material_num,
	sum(out_stock_qty) as out_stock_qty 
from DWR_WM_PRODSTOCK_OUT_IN_F t1
where 1=1
and t1.MVGR1 in ('H')
group by 
t1.jd_name,
t1.material_num
)
,tj3_2 as 
(
select 
	t1.jd_name,
	t1.material_num,
	ROW_NUMBER() OVER (partition by t1.jd_name ORDER BY out_stock_qty DESC) as rnk,
	COUNT(1) OVER (partition by t1.jd_name) AS total_sku_count
from tj3_1 t1
)

--条件4：
,tj4_1 as 
(
select 
	t1.jd_name,
	t1.material_num,
	sum(out_stock_qty) as out_stock_qty 
from DWR_WM_PRODSTOCK_OUT_IN_F t1
where 1=1
and t1.MVGR1 in ('A','B','C')
group by 
t1.jd_name,
t1.material_num
)
,tj4_2 as 
(
select 
	t1.jd_name,
	t1.material_num,
	ROW_NUMBER() OVER (partition by t1.jd_name ORDER BY out_stock_qty DESC) as rnk,
	COUNT(1) OVER (partition by t1.jd_name) AS total_sku_count
from tj4_1 t1
)


,detail_data as 
(
--条件1
select 
t1.jd_name,
t1.material_num,
'快' as fre_flag
from DWR_WM_PRODSTOCK_OUT_IN_F t1
left join sdi.sdi_matertial_product_flag_t_1037 t2 on t1.material_num = t2.material_num
where 1=1
and t2.new_material_flag = '是'
-- and t1.yl05 in ('H001')
group by 
t1.jd_name,
t1.material_num

union all

--条件2
select 
t1.jd_name,
t1.material_num,
case 
	when rnk <= total_sku_count*0.2 then '快'
	else '慢'
end as fre_flag
from tj2_2 t1

union all

--条件3
select 
t1.jd_name,
t1.material_num,
case 
	when rnk <= total_sku_count*0.01 then '快'
	when rnk <= total_sku_count*0.04 then '中'
	else '慢'
end as fre_flag
from tj3_2 t1


union all

--条件4
select 
t1.jd_name,
t1.material_num,
case 
	when rnk <= total_sku_count*0.15 then '快'
	when rnk <= total_sku_count*0.45 then '中'
	else '慢'
end as fre_flag
from tj4_2 t1
)

INSERT OVERWRITE TABLE DWROTD.DWR_STOCK_SKU_FREQUENCY_T
select 
jd_name,
material_num,
fre_flag,
CURRENT_TIMESTAMP AS LAST_UPDATE_DATE
from detail_data
group by 
jd_name,
material_num,
fre_flag
