with ekbe as (
  select t1.ebeln ,t1.ebelp ,
		   sum(case when t1.bewtp='L' then t1.menge else 0 end) as lmeng,
		   sum(case when t1.bewtp='E' AND t1.bwart = '101' then t1.menge 
					when t1.bewtp='E' AND t1.bwart = '102' then 0-t1.menge else 0 end) as emeng,
		   max(case when t1.bewtp='L' then t1.budat end) res_stock_date
  from sdi.sdi_ekbe_1001 t1
  where 1=1
   and t1.menge >0
  group by t1.ebeln ,ebelp 
  )
,yt_date as 
(
select vbeln,posnr,replace(date_sub(from_unixtime(unix_timestamp(min(createdate), 'yyyyMMdd'), 'yyyy-MM-dd'),1),'-','') as min_createdate
from SDI.SDI_ZTCOSTBHLH_NEW_1001   -- 23年之前的表不读了
where substr(createdate,1,6) <=  replace(substring(current_timestamp(),1,7),'-','')
--and datatypetxt = '留货'
group by vbeln,posnr
)
,tmp as(
select 
	CHARG_SID
	,mat_pspnr--已估价的销售订单库存工作分解结构要素
    ,matnr--物料编码
    ,werks--工厂
    ,min(BUDAT) as BUDAT--过账日期 取第一个
from sdi.sdi_MATDOC_1001 i
where bwart IN ('415','411')--转入的项目库存，多批次取最后的时间
and sobkz = 'Q'
and DEL_FLAG = 'N'
group by CHARG_SID,mat_pspnr,matnr,werks
)
,tem as(
SELECT
    -- MATERIAL_ID
	 A.MATERIAL_NUM -- matnr
	--,MATERIAL_NAME
	--,FACTORY_ID
	,A.FACTORY_CODE -- werks
	--,FACTORY_NAME
	,A.INV_LOCATION -- lgort
	,A.BATCH_NUM -- CHARG
	,A.SAL_VCH_NUM as stockout_order_no
	,A.SAL_VCH_ITEM_NUM as stockout_order_item_no--销售凭证项目  POSNR
	,A.WBS_ELEMENT --WBS 要素 PSPEL
	--,A.CUST_ID
	,C.CUST_DEL_NUM as cust_num --客户 as CUSTOMER  KUNNR
	,COALESCE(E.lgobe, cust.cust_name) as cust_name
	,COALESCE(F1.VKORG ,CONCAT(SUBSTR(d2.reswk,1,2),'00')) AS SALES_ORG_CODE  
	,A.SUBJECT_DISTR_TYPE
	,case when COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'Q_T163I' then COALESCE(SKU_CONFIRM_QTY,0) else 0 end as PRO_SKU_PREPARE_QTY --项目备货数
    ,case when (COALESCE(A.SUBJECT_DISTR_TYPE,'') = '' OR  (COALESCE(A.SUBJECT_DISTR_TYPE,'') = 'M_T163I' AND COALESCE(A.REQ_TYPE,'') = '01'))
	then COALESCE(SKU_CONFIRM_QTY,0) else 0 end as SKU_PREPARE_QTY, -- VMENG 备货数
	COALESCE(SKU_CONFIRM_QTY,0) as TOTAL_PREPARE_QTY,
	CASE WHEN B.REFERENCE_BILL_NUM LIKE '4%' THEN 'PURCHASE' WHEN  B.REFERENCE_BILL_NUM LIKE '1%' THEN 'SALES' END AS  ORDER_SOURCE,
	B.REFERENCE_BILL_NUM,
	B.REFERENCE_PROJ_NUM,
	a.avail_date as prepare_stock_date,
	DATEDIFF(CURRENT_DATE(),a.avail_date) AS prepare_stock_days,
	F1.VTWEG  AS distribution_channel_code,
	COALESCE(D2.ernam,F1.ERNAM) as creator_id,       
	coalesce(concat(substr(m011.mbdat,1,4),'-',substr(m011.mbdat,5,2),'-',substr(m011.mbdat,7,2)),
			concat(substr(F1.MAX_CONFIRM_DATE,1,4),'-',substr(F1.MAX_CONFIRM_DATE,5,2),'-',substr(F1.MAX_CONFIRM_DATE,7,2)),
			concat(substr(g.res_stock_date,1,4),'-',substr(g.res_stock_date,5,2),'-',substr(g.res_stock_date,7,2))
			)  as res_stock_date,
	current_timestamp() as dw_last_update_date,
	case when F1.BSTDK<>'00000000' then concat(substr(F1.BSTDK,1,4),'-',substr(F1.BSTDK,5,2),'-',substr(F1.BSTDK,7,2)) end  as require_date,
	case when F1.CNJQ<>'00000000' then concat(substr(F1.CNJQ,1,4),'-',substr(F1.CNJQ,5,2),'-',substr(F1.CNJQ,7,2)) end   as promise_date,
	t3.lev1_name,
	t3.lev2_name,
	t3.lev3_name,
	t3.lev4_name,
	t3.lev5_name,
	t3.lev6_name,
	d.lgort,--公司代码
	case when d.lgort = '0028' then '西南运营中心'
	when d.lgort = '0020' then '粤东运营中心'
	when d.lgort = 'GFZ' then '粤西运营中心'
	when d.lgort = '0030' then '华东运营中心'
	when d.lgort = '0018' then '西北运营中心'
	when d.lgort = '0026' then '西南运营中心'
	when d.lgort='7F21' then '闽琼运营中心' end as butxt --公司名称
	,F1.erdat as create_time
	,t3.sale_grp --线组
	,F1.JHDDLX as plan_order_type
	,D2.reswk  as  stockout_FACTORY_CODE     --出库的工厂
    ,D.reslo as stockout_LOCATION -- sto 出库的库存地点
    ,F1.SUBMI --备货类型
    ,case 
            when F1.VTWEG = '06' then 'GD03'
            -- when F1.werks like '%80' and C1.AUART = 'ZA02' then 'GD04' -- 取消特惠品渠道
            when F1.VTWEG = '08' then 'GD02'
            else 'GD01' end as zh_channel_code4
    ,D.lgort as stockin_LOCATION-- sto 入库的库存地点
    ,F1.VRKME --单位
    ,D.werks as stockin_factory_code --sto 入库的工厂
    ,COALESCE(F1.VOLUM/F1.KWMENG,D.volum,mat.busi_volume) as unit_VOLUM
    ,t3.lev7_name
	,coalesce(F1.BSTDK,'') as BSTDK
	,F1.FIRST_CONFIRM_DATE
	,F1.CZJHDH
	,F1.BSARK
	,F1.kz_date
	,F1.is_xh
FROM DWIOTD.DWI_SO_SAL_REQ_SAP_T a -- VBBE
LEFT JOIN DWIOTD.dwi_wm_ladingbill_item_t B oN A.SAL_VCH_NUM = B.delivery AND A.SAL_VCH_ITEM_NUM = B.delivery_project 
LEFT JOIN DWIOTD.dwi_wm_ladingbill_header_t C oN A.SAL_VCH_NUM = C.delivery 
LEFT JOIN SDI.sdi_ekpo_1001 D on B.REFERENCE_BILL_NUM = D.ebeln  AND substr(B.REFERENCE_PROJ_NUM,-5,5) = D.ebelp 
LEFT JOIN SDI.sdi_ekko_1001 D2 on B.REFERENCE_BILL_NUM = D2.ebeln  
LEFT JOIN SDI.sdi_t001l_1001 E oN A.FACTORY_CODE  = E.werks  AND D.lgort  = E.lgort 
LEFT JOIN DM.DM_OTD_SALES_ORDER_DET_T  F1 on B.REFERENCE_BILL_NUM = F1.vbeln and B.reference_proj_num = F1.posnr
LEFT JOIN ekbe g ON  B.reference_bill_num  = g.ebeln  AND substr(B.REFERENCE_PROJ_NUM,-5,5)  = g.ebelp
LEFT JOIN DWRDIM.dwr_dim_cust_general_d cust ON cust.cust_num =C.CUST_DEL_NUM  and cust.end_date ='4712-12-31'
--LEFT JOIN SDI.sdi_vbak_1001 C1 ON B.reference_bill_num = C1.vbeln
--left join vbep as vbep on F1.vbeln = vbep.vbeln and F1.posnr = vbep.posnr 
left join sdi.sdi_ZTMM011_1001 m011 on m011.vbeln = F1.VBELN and m011.mbdat = F1.POSNR
LEFT JOIN dm.dm_rpt_sale_grp_t t3 on F1.vkgrp = t3.sale_grp
LEFT JOIN dwimd.dwi_md_data_material_general_t mat on A.MATERIAL_NUM = mat.material_num
WHERE COALESCE(A.SALES_DIS_VCH_TYPE,'') = 'J_VBTYPL'  -- VBTYP
AND COALESCE(A.SKU_CONFIRM_QTY,0) > 0
AND A.DEL_FLAG = 'N'
)
