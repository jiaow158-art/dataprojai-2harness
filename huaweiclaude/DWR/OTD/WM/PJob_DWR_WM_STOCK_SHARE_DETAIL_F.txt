with tmp_data as( -- 明细数据
        select
            t0.customer_type,
            t0.customer_CODE,
            case when t0.customer_CODE = '310277' and t0.loccode = '01' then '郑州共享仓'
				 when t0.customer_CODE = '310277' and t0.loccode = '03' then '驻马店共享仓'
            else t3.customer_name_cn 
			end as customer_name_cn,
            substring(cast(t0.DocDate as string),1,10) as DocDate,
            t0.ItemCode,
            T4.material_name AS itemname ,
            t2.DistNumber,
            t1.Quantity,
            T4.area_m2  -- 面积
        from DWIOTD.dwi_wm_b1_oitl_t t0 
        inner join DWIOTD.dwi_wm_b1_ITL1_t t1 
        on t0.LogEntry=t1.LogEntry
        and t0.customer_type = t1.customer_type 
        and t0.customer_code = t1.customer_code 
        inner join  dwiotd.dwi_wm_b1_OBTN_t t2 
        on t1.ItemCode=t2.ItemCode 
        and t1.SysNumber=t2.SysNumber 
        and t1.customer_type = t2.customer_type 
        and t1.customer_code = t2.customer_code 
        join (select distinct t.customer_type,
                              t.customer_code ,
                              t.customer_name_cn 
              from dm.dm_b1_customer_data_t t
              where t.customer_type = 'CT'
              AND t.share_stock_flag = 'Y') t3
        on t0.customer_type = t3.customer_type 
        and t0.customer_code = t3.customer_code 
        JOIN (
        select material_num,
                material_name,
                (width * length_1) / 1000000 area_m2
        from DWIMD.dwi_md_data_material_general_t
        where matl_group_code = '1000010'
        ) T4
        ON T0.ITEMCODE = T4.material_num 
        where t0.customer_type = 'CT'
        AND t1.customer_type = 'CT'
        AND t2.customer_type = 'CT'
        AND t1.Quantity <> 0
		UNION ALL
		-- 特殊场景,321991(佛山百存建材有限公司) 本身属于卫浴的经销商，也需要统计
		select
			t0.customer_type,
			t0.customer_CODE,
			t3.customer_name_cn,
			substring(cast(t0.DocDate as string),1,10) as DocDate,
			t0.ItemCode,
			T4.material_name AS itemname ,
			t2.DistNumber,
			t1.Quantity ,
			T4.area_m2  -- 面积
		from DWIOTD.dwi_wm_b1_oitl_t t0 
		inner join DWIOTD.dwi_wm_b1_ITL1_t t1 
		on t0.LogEntry=t1.LogEntry
		and t0.customer_type = t1.customer_type 
		and t0.customer_code = t1.customer_code 
		inner join  dwiotd.dwi_wm_b1_OBTN_t t2 
		on t1.ItemCode=t2.ItemCode 
		and t1.SysNumber=t2.SysNumber 
		and t1.customer_type = t2.customer_type 
		and t1.customer_code = t2.customer_code 
		join (select distinct t.customer_type,
						  t.customer_code ,
						  t.customer_name_cn 
		  from dm.dm_b1_customer_data_t t
		  where t.customer_type = 'SW'
		AND t.share_stock_flag = 'Y') t3
		on t0.customer_type = t3.customer_type 
		and t0.customer_code = t3.customer_code 
        JOIN (
        select material_num,
                material_name,
                (width * length_1) / 1000000 area_m2
        from DWIMD.dwi_md_data_material_general_t
        where matl_group_code = '1000010'
        ) T4
		ON T0.ITEMCODE = T4.material_num 
		where t0.customer_type = 'SW'
		AND t1.customer_type = 'SW'
		AND t2.customer_type = 'SW'
		AND t0.customer_code = '321991'
		AND t1.Quantity <> 0
)
,tmp_data1 as(   -- 算出入库数据
SELECT customer_type,
         customer_code,
         customer_name_cn,
         docdate, 
         itemcode, 
         itemname,
         distnumber,
         area_m2,   -- 面积
         sum(case when Quantity>0 then Quantity else 0 end) INSTOCK_QTY,  -- 入库数量
         sum(case when Quantity<0 then Quantity else 0 end) OUTSTOCK_QTY  -- 出库数量
FROM tmp_data T
group by customer_type,
         customer_code,
         customer_name_cn,
         docdate, 
         itemcode, 
         itemname,
         distnumber,
         area_m2    -- 面积
)
,tmp_data2 as(  -- 算累计入库、在库数量
SELECT T.customer_type,
       T.customer_code,
       T.customer_name_cn,
       T.itemcode AS material_num,
       T.itemname AS material_name,
       T.distnumber AS BATCH_NUM,
       T.docdate AS DOC_DATE,
       T.INSTOCK_QTY,  -- 入库数量
       T.OUTSTOCK_QTY,  -- 出库数量
       T.area_m2,   -- 面积
       SUM(T.INSTOCK_QTY) OVER(PARTITION by customer_type,customer_code,customer_name_cn,itemcode,distnumber ORDER BY docdate desc) TOTAL_INSTOCK_QTY, -- 累计入库数量
       SUM(T.INSTOCK_QTY) OVER(PARTITION by customer_type,customer_code,customer_name_cn,itemcode,distnumber ) SUM_INSTOCK_QTY,  -- 入库总数量
       SUM(T.OUTSTOCK_QTY) OVER(PARTITION by customer_type,customer_code,customer_name_cn,itemcode,distnumber ) SUM_OUTSTOCK_QTY, -- 出库总数量
       SUM(T.INSTOCK_QTY) OVER(PARTITION by customer_type,customer_code,customer_name_cn,itemcode,distnumber ) - 
       ABS(SUM(T.OUTSTOCK_QTY) OVER(PARTITION by customer_type,customer_code,customer_name_cn,itemcode,distnumber )) AS NOW_STOCK_QTY -- 在库数量
FROM tmp_data1 T
)
,
tmp_data3 as(
select T.customer_type, -- 客户类型
       T.customer_code, -- 客户编码
       T.customer_name_cn, -- 客户中文名称
       T.material_num,  -- 物料编码
       T.material_name,  -- 物料名称
       T.BATCH_NUM,  -- 批次号
       T.DOC_DATE, -- 单据日期
       T.INSTOCK_QTY,  -- 入库数量
       T.OUTSTOCK_QTY,  -- 出库数量
       T.TOTAL_INSTOCK_QTY, -- 累计入库数量
       T.NOW_STOCK_QTY, -- 在库数量
       case when t.now_stock_qty >= t.total_instock_qty then t.instock_qty
            when t.now_stock_qty < t.total_instock_qty then
                 case when t.now_stock_qty >= t.total_instock_qty - t.instock_qty then
                           t.now_stock_qty - (t.total_instock_qty - t.instock_qty)
                       else 0 end
       else 0 end  now_fare_stock_qty, -- 在库分摊数量
	   /* 
	   如果在库数量大于等于累计入库数量，那么在库分摊等于入库数量;如果在库数量小于累计入库数量，继续判断：如果在库数量 大于等于 累计入库数量 - 入库数量，那么在库分摊为：在库数量 - （累计入库数量 - 入库数量），否则为0;
	   */
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,  -- 数据同步时间
       T.area_m2  -- 面积
from tmp_data2 T
)
INSERT OVERWRITE TABLE DWROTD.DWR_WM_STOCK_SHARE_DETAIL_F
select T.customer_type, -- 客户类型
       T.customer_code, -- 客户编码
       T.customer_name_cn, -- 客户中文名称
       T.material_num,  -- 物料编码
       T.material_name,  -- 物料名称
       T.BATCH_NUM,  -- 批次号
       T.DOC_DATE, -- 单据日期
       T.INSTOCK_QTY,  -- 入库数量
       T.OUTSTOCK_QTY,  -- 出库数量
       T.TOTAL_INSTOCK_QTY, -- 累计入库数量
       T.NOW_STOCK_QTY, -- 在库数量
       T.now_fare_stock_qty, -- 在库分摊数量
       T.DW_LAST_UPDATE_DATE,  -- 数据同步时间
       T.area_m2 * T.now_fare_stock_qty   area_m2  -- 面积    /*2023/04/19 guzhengfeng 面积计算乘以在库分摊数量*/
from tmp_data3 T
;
