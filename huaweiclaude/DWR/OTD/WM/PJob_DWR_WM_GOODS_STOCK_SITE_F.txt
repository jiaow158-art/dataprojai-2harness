	,CURRENT_TIMESTAMP()				  AS  DW_LAST_UPDATE_DATE
FROM DWIOTD.DWI_WM_GOODS_STOCK_SITE_T DWGSS		--DWI.SAP货物库位表
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T		--关联物料主数据取物料名称
	 ON T.MATERIAL_ID = DWGSS.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DWGSS.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWGSS.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T1				--关联工厂主数据取工厂名称
	 ON  T1.FACTORY_ID = DWGSS.FACTORY_ID		--工厂ID
	 AND T1.GROUP_NUM = DWGSS.GROUP_NUM			--集团
	 AND T1.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWGSS.DEL_FLAG = 'N'					--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2	--关联通用配置表
	 ON T2.CONFIG_CODE = DWGSS.SPECIAL_INV_FLAG	--取特殊库存标识名称中文名称
	 AND T2.GROUP_NUM = DWGSS.GROUP_NUM			--集团
	 AND T2.LANG_CODE = '1'
WHERE DWGSS.DEL_FLAG = 'N'
