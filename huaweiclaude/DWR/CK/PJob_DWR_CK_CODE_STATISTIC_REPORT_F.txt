INSERT OVERWRITE TABLE DWROTD.DWR_CK_CODE_STATISTIC_REPORT_F
SELECT
       ID
      ,TYPE
      ,CASE WHEN TYPE = '1' THEN '隐形码'
	        WHEN TYPE = '2' THEN '箱码'
			 END AS TYPE_NAME
      ,LINE_NAME
      ,PRODUCTION_NUM AS PRODUCT_QTY
      ,IDENTIFY_NUM AS IDENTIFY_QTY
      ,IDENTIFY_RATE
      ,STATISTIC_TIME
      ,STATISTIC_TIME_RANGE
      ,CREATE_TIME
      ,UPDATE_TIME
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_CODE_STATISTIC_REPORT_1040
 WHERE FLAG = '1'
;
