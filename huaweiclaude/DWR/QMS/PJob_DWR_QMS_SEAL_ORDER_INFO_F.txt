WITH T0 AS(
SELECT  HANDLE
       ,SPECIAL_MAKE_USER
       ,TABLE_VIEW.EMP_NUM
  FROM SDI.SDI_M_QMS_PRODUCT_SEAL_ORDER_1053
LATERAL VIEW EXPLODE(SPLIT(SPECIAL_MAKE_USER,',')) TABLE_VIEW AS EMP_NUM
),T1 AS(
SELECT T0.HANDLE
      ,T0.EMP_NUM
      ,T.EMP_NAME
  FROM T0
  LEFT JOIN DM.DM_WF_EMP_INFO_T T
    ON T0.EMP_NUM = T.EMP_NUM
),T2 AS(
SELECT HANDLE
      ,CONCAT_WS(',',COLLECT_SET(CAST(EMP_NAME AS STRING))) AS EMP_NAME
  FROM T1
 GROUP BY HANDLE
)
INSERT OVERWRITE TABLE DWRQMS.DWR_QMS_SEAL_ORDER_INFO_F
SELECT
        A.HANDLE AS ID --主键
       ,A.SITE AS FACTORY_CODE --工厂编码
	   ,D.SITE_NAME AS FACTORY_NAME --工厂名称
	   ,D.SITE_SHORT_NAME AS FACTORY_SHORT_NAME --工厂简称
       ,A.SEAL_ORDER_NO AS SEAL_NUM --封仓单号
       ,A.SEAL_DATE --封仓日期
       ,A.MOVE_TYPE AS MOVE_TYPE_CODE --移动类型编码
       ,T1.NAME AS MOVE_TYPE_NAME --移动类型描述
       ,A.SEAL_REASON AS SEAL_RSN --封仓原因
       ,A.ORDER_STATUS AS SEAL_ST_CODE --单据状态编码
       ,T2.NAME AS SEAL_ST_NAME --单据状态描述
       ,A.INSPECT_ORDER_NO AS CHK_NUM --检验单号
       ,A.RECEIPT_TYPE AS CHK_TYPE_CODE --检验类型编码
       ,T3.NAME AS CHK_TYPE_NAME --检验类型描述
       ,A.WAREHOUSE_CODE AS WH_CODE --仓库
       ,SPLIT(A.ITEM_HANDLE,':')[1] AS MATERIAL_NUM --物料编码
       ,A.LOT_CODES AS BATCH_NUM --批号(多个)
       ,A.SEAL_QTY --封仓数量
       ,A.REWORK_QTY AS RWRK_QTY --已开返工数量
       ,A.REWORK_SEAL_QTY AS NOT_RWRK_QTY --返工未解封数量
       ,A.REWORK_RELIEVE_QTY AS RWRK_RELV_QTY --返工已解封数量
       ,A.DIRECT_RELIEVE_QTY AS DIR_RELV_QTY --直接解封数量
       ,A.TOTAL_RELIEVE_QTY AS TOT_RELV_QTY --累计解封数量
       ,A.TODO_QTY AS TBD_QTY --待处理数量
       ,A.AUTH_BY AS APR_EMP_NUM --审核人工号
       ,E.EMP_NAME AS APR_EMP_NAME --审核人姓名
       ,A.AUTH_DATE AS APR_TIME --审核时间
       ,A.TACKLE_DATE AS PLAN_TACK_DATE --计划处理日期
       ,A.SAP_UNIQUE_ID --SAP回传唯一值
       ,A.OA_UNIQUE_ID --OA回传唯一值
       ,A.OA_FLOW_NO AS OA_FLOW_NUM --OA流程号
       ,CAST(A.IS_PASSIVE AS STRING) AS IS_PAS_CODE --是否无源编码： 0是 1否
       ,CASE WHEN CAST(A.IS_PASSIVE AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_PASSIVE AS STRING) = 'TRUE' THEN '否'
              END AS IS_PAS_NAME --是否无源描述
       ,REPLACE(SPLIT(A.WORK_SHOP_HANDLE,':')[1],A.SITE||',','') AS WORK_SHOP_CODE --车间编码
       ,REPLACE(SPLIT(A.PRODUCT_LINE_HANDLE,':')[2],SPLIT(A.WORK_SHOP_HANDLE,':')[1]||',','') AS PROD_LINE_CODE --产线编码
       ,SPLIT(A.TEAM_HANDLE,',')[LENGTH(A.TEAM_HANDLE)-LENGTH(REPLACE(A.TEAM_HANDLE,',',''))] AS TEAM_CODE --班组编码
       ,A.TACKLE_TYPE AS TACK_TYPE_CODE --处理类型编码
       ,T4.NAME AS TACK_TYPE_NAME --处理类型描述
       ,A.SPECIAL_MAKE_USER AS SPEC_EMP_NUM --特殊抄送人工号(多个)
       ,F.EMP_NAME AS SPEC_EMP_NAME --特殊抄送人姓名(多个)
       ,A.FEEDBACK_NUM AS FBACK_CNT --反馈次数
       ,A.REMARK --备注
       ,A.STATE AS ST_CODE --是否启用编码：0是 1否
       ,CASE WHEN A.STATE = 0 THEN '是'
             WHEN A.STATE = 1 THEN '否'
              END AS ST_NAME --是否启用描述
       ,A.CREATE_BY AS CREATE_EMP_NUM --创建人工号
       ,COALESCE(B.EMP_NAME,A.CREATE_BY) AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATE_DATE AS CREATE_TIME --创建时间
       ,A.UPDATE_BY AS UPDATE_EMP_NUM --修改人工号
       ,COALESCE(C.EMP_NAME,A.UPDATE_BY) AS UPDATE_EMP_NAME --修改人姓名
       ,A.UPDATE_DATE AS UPDATE_TIME --修改时间
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_M_QMS_PRODUCT_SEAL_ORDER_1053 A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON A.CREATE_BY = B.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T C
    ON A.UPDATE_BY = C.EMP_NUM
  LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D D
    ON A.SITE = D.SITE_CODE
  LEFT JOIN DM.DM_WF_EMP_INFO_T E   
    ON A.AUTH_BY = E.EMP_NUM
  LEFT JOIN T2 AS F
    ON A.HANDLE = F.HANDLE
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T1
    ON A.MOVE_TYPE = T1.CODE
   AND T1.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,productMoveType'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T2
    ON A.ORDER_STATUS = T2.CODE
   AND T2.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,productSealOrderStatus'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T3
    ON A.RECEIPT_TYPE = T3.CODE
   AND T3.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,dpReceiptType'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T4
    ON A.TACKLE_TYPE = T4.CODE
   AND T4.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,productSealOrderTackleType'
