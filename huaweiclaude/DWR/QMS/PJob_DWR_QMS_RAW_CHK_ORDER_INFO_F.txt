INSERT OVERWRITE TABLE DWRQMS.DWR_QMS_RAW_CHK_ORDER_INFO_F
SELECT  A.HANDLE AS ID --主键(INSPECTIONORDERBO:SITE,ORDER_NO)
       ,A.SITE AS FACTORY_CODE --工厂编码
	   ,G.SITE_NAME AS FACTORY_NAME --工厂名称
	   ,G.SITE_SHORT_NAME AS FACTORY_SHORT_NAME --工厂简称
       ,A.ORDER_NO AS CHK_NUM --检验单号
       ,A.REPORT_NO AS RPT_NUM --报检单号
       ,SPLIT(A.ITEM_HANDLE,':')[1] AS MATERIAL_NUM --物料编码
       ,SPLIT(A.ITEM_GROUP_HANDLE,':')[1] AS SUB_TYPE_CODE --小类编码
       ,SPLIT(A.VEND_HANDLE,':')[1] AS PRVDR_CODE --供应商主键
       ,A.VEND_LOT_NO AS PRVDR_BATCH_NUM --供应商批次
       ,A.INNER_LOT_NO AS BATCH_NUM --内部批次
       ,A.TAKE_DATE AS CNSGN_DATE --收货日期
       ,A.PRODUCE_DATE AS PROD_DATE --生产日期
       ,A.EXPIRE_DATE AS EXP_DATE --有效期至
       ,CAST(A.IS_SEDIMENT AS STRING) AS IS_SEDI_CODE --是否沙泥编码：0是  1否
       ,CASE WHEN CAST(A.IS_SEDIMENT AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_SEDIMENT AS STRING) = 'TRUE' THEN '否'
              END AS IS_SEDI_NAME --是否沙泥描述
       ,CAST(A.IS_URGENT AS STRING) AS IS_URGNT_CODE --是否急料编码：0是 1否
       ,CASE WHEN CAST(A.IS_URGENT AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_URGENT AS STRING) = 'TRUE' THEN '否'
              END AS IS_URGNT_NAME --是否急料描述
       ,A.ORDER_QTY AS CHK_QTY --检验数量
       ,A.RECEIPT_TYPE AS CHK_TYPE_CODE --检验类型编码
	   ,T6.NAME AS CHK_TYPE_NAME --检验类型描述
       ,A.RECEIPT_CLASS AS CHK_CAT_CODE --检验类别编码
	   ,T1.NAME AS CHK_CAT_NAME --检验类别描述
       ,A.ORDER_STATUS AS CHK_ST_CODE --检验单状态编码
       ,CASE WHEN A.ORDER_STATUS = 'COMPLETE' THEN '已完成'
             WHEN A.ORDER_STATUS = 'RETURN_SAP' THEN '已退回'
             WHEN A.ORDER_STATUS = 'WAIT' THEN '待检验'
             WHEN A.ORDER_STATUS = 'CHECKING' THEN '检验中'
             WHEN A.ORDER_STATUS = 'CREATE_RECHECK' THEN '重测'
             WHEN A.ORDER_STATUS = 'COUNTERSIGN' THEN '会签中'
             WHEN A.ORDER_STATUS = 'TO_AUDIT' THEN '待审核'
             WHEN A.ORDER_STATUS = 'TO_CONFIRM' THEN '待确认'
              END AS CHK_ST_NAME --检验单状态描述
       ,A.ORDER_REMARK AS CHK_REMARK --检验单备注
       ,A.INSPECT_RESULT AS CHK_RSULT_CODE --检验结果编码
	   ,T5.NAME AS CHK_RSULT_NAME --检验结果描述
       ,A.WAREHOUSE_CODE AS WH_CODE --仓库编码
       ,A.UNQUALIFIED_COUNT AS UNQUAL_CNT --不合格项数
       ,A.LOT_NO AS LOT_NUM --填写批号
       ,A.SAMPLING_QTY AS SAMP_QTY --抽样数
       ,REPLACE(SPLIT(A.INSPECTION_STANDARD_HANDLE,':')[1],SITE||',','') AS CHK_STD_ID --检验标准主键
       ,A.INSPECT_AMOUNT_FINE_RATIO_SUGGEST AS CHK_AMT_RATE --检验金额扣罚比例建议
       ,A.INSPECT_NUMBER_FINE_RATIO_SUGGEST AS CHK_NUM_RATE --检验数量扣罚比例建议
       ,A.INSPECT_DISPOSE_SUGGEST AS CHK_DISE_CODE --检验处置建议编码
	   ,T2.NAME AS CHK_DISE_NAME --检验处置建议描述
       ,A.INSPECT_BY AS CHK_EMP_NUM --检验人工号
       ,B.EMP_NAME AS CHK_EMP_NAME --检验人姓名
       ,A.INSPECT_DATE AS CHK_TIME --检验时间
       ,A.AUTH_AMOUNT_FINE_RATIO_SUGGEST AS APR_AMT_RATE --审核金额扣罚比例建议
       ,A.AUTH_NUMBER_FINE_RATIO_SUGGEST AS APR_NUM_RATE --审核数量扣罚比例建议
       ,A.AUTH_DISPOSE_SUGGEST AS APR_DISE_CODE --审核处置建议编码
	   ,T3.NAME AS APR_DISE_NAME --审核处置建议描述
       ,A.AUTH_BY AS APR_EMP_NUM --审核人工号
       ,C.EMP_NAME AS APR_EMP_NAME --审核人姓名
       ,A.AUTH_DATE AS APR_TIME --审核时间
       ,A.AUTH_REMARK AS APR_REMARK --审核备注
       ,A.OA_COUNTERSIGN_VERDICT AS OA_CNTRSIGN_CODE --OA会签结论编码
	   ,T4.NAME AS OA_CNTRSIGN_NAME --OA会签结论描述
       ,A.OA_AMOUNT_FINE_RATIO_SUGGEST AS OA_AMT_RATE --OA金额扣罚比例建议
       ,A.OA_NUMBER_FINE_RATIO_SUGGEST AS OA_NUM_RATE --OA数量扣罚比例建议
       ,A.OA_REMARK --OA备注
       ,A.OA_OVER_DATE AS OA_APR_TIME --OA会签结束时间
       ,A.CONFIRM_BY AS CNFRM_EMP_NUM --确认人工号
       ,D.EMP_NAME AS CNFRM_EMP_NAME --确认人姓名
       ,A.CONFIRM_DATE AS CNFRM_TIME --确认时间
       ,A.RETEST_NUM AS RETST_CNT --重测次数
       ,A.RETEST_SORT AS RETST_SORT_NUM --重测序号
       ,A.RECHECK_NUM AS RECHK_CNT --复测次数
       ,A.RECHECK_SORT AS RECHK_NUM --复测序号
       ,A.PARENT_HANDLE AS PARNT_ID --复测上级主键
       ,A.RETURN_HANDLE AS RTN_ID --退货原单主键
       ,A.DATA_RESOURCE AS DATA_SRC_CODE --数据来源
       ,CAST(A.IS_SAP AS STRING) AS IS_SAP_CODE --是否SAP回传编码
       ,CASE WHEN CAST(A.IS_SAP AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_SAP AS STRING) = 'TRUE' THEN '否'
              END IS_SAP_NAME --是否SAP回传描述
       ,A.SAP_UNIQUE_ID --SAP回传唯一值
       ,A.OA_UNIQUE_ID --OA回传唯一值
       ,A.OA_FLOW_NO AS OA_FLOW_NUM --OA流程编号
       ,A.STATE AS ST_CODE --是否启用编码：0是 1否
       ,CASE WHEN A.STATE = 0 THEN '是'
             WHEN A.STATE = 1 THEN '否'
              END AS ST_NAME --是否启用描述
       ,A.CREATE_BY AS CREATE_EMP_NUM --创建人工号
       ,COALESCE(E.EMP_NAME,A.CREATE_BY) AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATE_DATE AS CREATE_TIME --创建时间
       ,A.UPDATE_BY AS UPDATE_EMP_NUM --修改人工号
       ,COALESCE(F.EMP_NAME,A.UPDATE_BY) AS UPDATE_EMP_NAME --修改人姓名
       ,A.UPDATE_DATE AS UPDATE_TIME --修改时间
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_M_QMS_MATERIAL_INSPECTION_ORDER_1053 A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON A.INSPECT_BY = B.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T C
    ON A.AUTH_BY = C.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T D
    ON A.CONFIRM_BY = D.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T E
    ON A.CREATE_BY = E.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T F
    ON A.UPDATE_BY = F.EMP_NUM
  LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D G
    ON A.SITE = G.SITE_CODE
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T1
    ON A.RECEIPT_CLASS = T1.CODE
   AND T1.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,receiptClass'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T2
    ON A.INSPECT_DISPOSE_SUGGEST = T2.CODE
   AND T2.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectDisposeSuggest'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T3
    ON A.AUTH_DISPOSE_SUGGEST = T3.CODE
   AND T3.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectDisposeSuggest'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T4
    ON A.OA_COUNTERSIGN_VERDICT = T4.CODE
   AND T4.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectDisposeSuggest'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T5
    ON A.INSPECT_RESULT = T5.CODE
   AND T5.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectionOrderResult'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T6
    ON A.RECEIPT_TYPE = T6.CODE
   AND T6.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,dpReceiptType'
