INSERT OVERWRITE TABLE DWRQMS.DWR_QMS_PROD_CHK_ORDER_INFO_F
SELECT
       A.HANDLE AS ID --主键(INSPECTIONORDERBO:SITE,ORDER_NO)
      ,A.SITE AS FACTORY_CODE --工厂编码
	  ,B.SITE_NAME AS FACTORY_NAME --工厂名称
	  ,B.SITE_SHORT_NAME AS FACTORY_SHORT_NAME --工厂简称
      ,A.ORDER_NO AS CHK_NUM --检验单号
      ,A.REPORT_NO AS RPT_NUM --报检单号
      ,SPLIT(A.WORK_SHOP_HANDLE,':')[1] AS WORK_SHOP_CODE --车间编码
      ,REPLACE(SPLIT(A.PRODUCT_LINE_HANDLE,':')[2],SPLIT(A.WORK_SHOP_HANDLE,':')[1]||',','') AS PROD_LINE_CODE --产线编码
      ,SPLIT(A.ITEM_HANDLE,':')[1] AS MATERIAL_NUM --物料编码
      ,A.STORAGE_DATE AS INPUT_DATE --入库日期
      ,A.ORDER_QTY AS CHK_QTY --检验数量
      ,A.LOT_NOS AS BATCH_NUM --批次(多个)
      ,A.COLOR_NOS AS COLOR_NUM --色号(多个)
      ,A.PRODUCT_DATES AS PROD_DATES --生成日期(多个)
      ,A.WAREHOUSE_CODE AS WH_CODE --仓库
      ,A.LOCATION_CODES AS LOCAT_CODES --区位(多个)
      ,A.RECEIPT_TYPE AS CHK_TYPE_CODE --检验类型编码
	  ,T4.NAME AS CHK_TYPE_NAME --检验类型描述
      ,A.RECEIPT_CLASS AS CHK_CAT_CODE --检验类别编码
	  ,T5.NAME AS CHK_CAT_NAME --检验类别描述
      ,A.WATER_UPTAKE AS WATER_UPTAKE_CODE --吸水率编码
      ,A.WATER_UPTAKE_DESC AS WATER_UPTAKE_NAME --吸水率描述
      ,CAST(A.IS_C AS STRING) AS IS_C_CODE --是否C标编码
      ,CASE WHEN CAST(A.IS_C AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_C AS STRING) = 'TRUE' THEN '否'
              END AS IS_C_NAME --是否C标描述
      ,A.ORDER_STATUS AS CHK_ST_CODE --检验单状态编码
	  ,T6.NAME AS CHK_ST_NAME --检验单状态描述
      ,A.ORDER_REMARK AS CHK_REMARK --检验单备注
      ,REPLACE(SPLIT(A.MATERIALIZATION_INSPECTION_STANDARD_HANDLE,':')[1],SITE||',','') AS MATL_CHK_STD_ID --物化检验标准主键
      ,REPLACE(SPLIT(A.FUNCTION_INSPECTION_STANDARD_HANDLE,':')[1],SITE||',','') AS FUNC_CHK_STD_ID --功能检验标准主键
      ,REPLACE(SPLIT(A.EXTERIOR_INSPECTION_STANDARD_HANDLE,':')[1],SITE||',','') AS EXTRR_CHK_STD_ID --外观检验标准主键
      ,A.INSPECT_RESULT AS CHK_RSULT_CODE --检验结果
      ,T1.NAME AS CHK_RSULT_NAME --检验结果描述
      ,A.INSPECT_BY AS CHK_EMP_NUM --检验人工号
      ,C.EMP_NAME AS CHK_EMP_NAME --检验人姓名
      ,A.INSPECT_DATE AS CHK_TIME --检验时间
      ,A.SEAL_ORDER_NO AS SEAL_ORDER_NUM --封仓单号
      ,A.CORRECT_PREVENT AS COR_PRVNT --纠正预防
      ,A.AUTH_DISPOSE_SUGGEST AS APR_DISE_CODE --审核处置建议
      ,T2.NAME AS APR_DISE_NAME --审核处置建议描述
      ,A.AUTH_BY AS APR_EMP_NUM --审核人工号
      ,D.EMP_NAME AS APR_EMP_NAME --审核人姓名
      ,A.AUTH_DATE AS APR_TIME --审核时间
      ,A.AUTH_REMARK AS APR_REMARK --审核备注
      ,A.OA_COUNTERSIGN_VERDICT AS OA_CNTRSIGN_CODE --OA会签结论编码
	  ,T3.NAME AS OA_CNTRSIGN_NAME --OA会签结论描述
      ,A.OA_AMOUNT_FINE_RATIO_SUGGEST AS OA_AMT_RATE --OA金额扣罚比例建议
      ,A.OA_NUMBER_FINE_RATIO_SUGGEST AS OA_NUM_RATE --OA数量扣罚比例建议
      ,A.OA_REMARK --OA备注
      ,A.OA_OVER_DATE AS OA_APR_TIME --OA会签结束时间
      ,A.CONFIRM_BY AS CNFRM_EMP_NUM --确认人工号
      ,E.EMP_NAME AS CNFRM_EMP_NAME --确认人姓名
      ,A.CONFIRM_DATE AS CNFRM_TIME --确认时间
      ,A.EXTRACT_NUM AS EXTC_CNT --抽检次数
      ,A.EXTRACT_SORT AS EXTC_NUM --抽检序号
      ,A.PARENT_HANDLE AS PARNT_ID --抽检上级主键
      ,A.DATA_RESOURCE AS DATA_SRC_CODE --数据来源
      ,A.SAP_UNIQUE_ID --SAP回传唯一值
      ,A.OA_UNIQUE_ID --OA回传唯一值
      ,A.OA_FLOW_NO AS OA_FLOW_NUM --OA流程编号
      ,A.AUTO_FLG AS IS_AUTO_CODE --自动标识编码：0未处理 1处理中 2已处理
      ,CASE WHEN A.AUTO_FLG = 0 THEN '未处理'
            WHEN A.AUTO_FLG = 1 THEN '处理中'
            WHEN A.AUTO_FLG = 2 THEN '已处理'
             END AS IS_AUTO_NAME --自动标识描述
      ,A.SAMPLING_QTY AS SAMP_QTY --抽样数
      ,A.OMISSION_QTY AS OM_QTY --漏检数
      ,A.STATE AS ST_CODE --是否启用：0是 1否
      ,A.CREATE_BY AS CREATE_EMP_NUM --创建人工号
      ,COALESCE(F.EMP_NAME,A.CREATE_BY) AS CREATE_EMP_NAME --创建人姓名
      ,A.CREATE_DATE AS CREATE_TIME --创建时间
      ,A.UPDATE_BY AS UPDATE_EMP_NUM --修改人工号
      ,COALESCE(G.EMP_NAME,A.UPDATE_BY) AS UPDATE_EMP_NAME --修改人姓名
      ,A.UPDATE_DATE AS UPDATE_TIME --修改时间
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_M_QMS_PRODUCT_INSPECTION_ORDER_1053 A
  LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D B
    ON A.SITE = B.SITE_CODE
  LEFT JOIN DM.DM_WF_EMP_INFO_T C
    ON A.INSPECT_BY = C.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T D
    ON A.AUTH_BY = D.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T E
    ON A.CONFIRM_BY = E.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T F
    ON A.CREATE_BY = F.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T G
    ON A.UPDATE_BY = G.EMP_NUM
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T1
    ON A.INSPECT_RESULT = T1.CODE
   AND T1.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectionOrderResult'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T2
    ON A.AUTH_DISPOSE_SUGGEST = T2.CODE
   AND T2.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectDisposeSuggest'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T3
    ON A.OA_COUNTERSIGN_VERDICT = T3.CODE
   AND T3.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectDisposeSuggest'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T4
    ON A.RECEIPT_TYPE = T4.CODE
   AND T4.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,dpReceiptType'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T5
    ON A.RECEIPT_CLASS = T5.CODE
   AND T5.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,receiptClass'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T6
    ON A.ORDER_STATUS = T6.CODE
   AND T6.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,inspectionOrderStatus'
