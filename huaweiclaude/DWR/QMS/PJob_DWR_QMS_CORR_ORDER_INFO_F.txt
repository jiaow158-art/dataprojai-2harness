WITH T0 AS(
SELECT  HANDLE
       ,CURRENT_NODE_BY
       ,TABLE_VIEW.EMP_NUM
  FROM SDI.SDI_M_QMS_CORRECT_PREVENT_ORDER_1053
LATERAL VIEW EXPLODE(SPLIT(CURRENT_NODE_BY,',')) TABLE_VIEW AS EMP_NUM
),T1 AS(
SELECT T0.HANDLE
      ,T0.EMP_NUM
      ,T.EMP_NAME
  FROM T0
  LEFT JOIN DM.DM_WF_EMP_INFO_T T
    ON T0.EMP_NUM = T.EMP_NUM
),T2 AS(
SELECT HANDLE
      ,CONCAT_WS(',',COLLECT_SET(CAST(EMP_NAME AS STRING))) AS EMP_NAME
  FROM T1
 GROUP BY HANDLE
),T3 AS(
SELECT  HANDLE
       ,MAKE_COPY_BY
       ,TABLE_VIEW.EMP_NUM
  FROM SDI.SDI_M_QMS_CORRECT_PREVENT_ORDER_1053
LATERAL VIEW EXPLODE(SPLIT(MAKE_COPY_BY,',')) TABLE_VIEW AS EMP_NUM
),T4 AS(
SELECT T3.HANDLE
      ,T3.EMP_NUM
      ,T.EMP_NAME
  FROM T3
  LEFT JOIN DM.DM_WF_EMP_INFO_T T
    ON T3.EMP_NUM = T.EMP_NUM
),T5 AS(
SELECT HANDLE
      ,CONCAT_WS(',',COLLECT_SET(CAST(EMP_NAME AS STRING))) AS EMP_NAME
  FROM T4
 GROUP BY HANDLE
),T6 AS(
SELECT  HANDLE
       ,MANAGER_BY
       ,TABLE_VIEW.EMP_NUM
  FROM SDI.SDI_M_QMS_CORRECT_PREVENT_ORDER_1053
LATERAL VIEW EXPLODE(SPLIT(MANAGER_BY,',')) TABLE_VIEW AS EMP_NUM
),T7 AS(
SELECT T6.HANDLE
      ,T6.EMP_NUM
      ,T.EMP_NAME
  FROM T6
  LEFT JOIN DM.DM_WF_EMP_INFO_T T
    ON T6.EMP_NUM = T.EMP_NUM
),T8 AS(
SELECT HANDLE
      ,CONCAT_WS(',',COLLECT_SET(CAST(EMP_NAME AS STRING))) AS EMP_NAME
  FROM T7
 GROUP BY HANDLE
)
INSERT OVERWRITE TABLE DWRQMS.DWR_QMS_CORR_ORDER_INFO_F
SELECT  A.HANDLE AS ID --主键
       ,A.SITE AS FACTORY_CODE --工厂编码
	   ,B.SITE_NAME AS FACTORY_NAME --工厂名称
	   ,B.SITE_SHORT_NAME AS FACTORY_SHORT_NAME --工厂简称
       ,A.ORDER_NO AS CORR_NUM --单号
       ,A.ORDER_TYPE AS CORR_TYPE_CODE --单据类型编码
       ,T1.NAME AS CORR_TYPE_NAME --单据类型描述
       ,A.ORDER_DATE AS SEND_DATE --订单日期(发文日期)
       ,A.FREQUENCY_DATE AS FREQ_DATE --班次日期
       ,A.ORDER_STATUS AS CORR_ST_CODE --单据状态编码
       ,T2.NAME AS CORR_ST_NAME --单据状态描述
       ,A.EVENT_RESOURCE AS EV_RSRS_CODE --事件来源编码
       ,T3.NAME AS EV_RSRS_NAME --事件来源描述
       ,A.CURRENT_NODE_BY AS CURR_APR_EMP_NUM --当前节点人工号(多个)
       ,C.EMP_NAME AS CURR_APR_EMP_NAME --当前节点人姓名(多个)
       ,A.PLAN_RECTIFICATION_DATE AS PLAN_REC_DATE --计划整改日期
       ,A.PROBLEM_DESC AS PROB_DESC --问题/缺陷描述
       ,A.EVENT_DESC AS EV_DESC --事件及问题描述
       ,A.SEND_BY AS SEND_EMP_NUM --发文人工号
       ,D.EMP_NAME AS SEND_EMP_NAME --发文人姓名
       ,A.SEND_ORG_NAME AS SEND_DEPT_NAME --发文部门
       ,A.RECTIFICATION_BY AS REC_EMP_NUM --整改人工号
       ,E.EMP_NAME AS REC_EMP_NAME --整改人姓名
       ,A.RECTIFICATION_ORG_NAME AS REC_DEPT_NAME --整改部门
       ,A.WORK_SHOP_NAME --车间
       ,A.PRODUCT_LINE_NAME AS PROD_LINE_NAME --产线
       ,A.PROCESS_NAME AS PROC_NAME --工序
       ,A.TEAM_NAME --班组
       ,A.MAKE_COPY_BY AS COPY_EMP_NUM --抄送人工号(多个)
       ,F.EMP_NAME AS COPY_EMP_NAME --抄送人姓名(多个)
       ,A.MANAGER_BY AS MGR_EMP_NUM --主管人工号(多个)
       ,G.EMP_NAME AS MGR_EMP_NAME --主管人姓名(多个)
       ,A.REASON_ANALYSIS AS RSN_ANAL --原因分析
       ,A.MEASURE_DATE AS MEAS_DATE --处理措施日期
       ,A.ORG_BY AS ORG_EMP_NUM --部门负责人工号
       ,H.EMP_NAME AS ORG_EMP_NAME --部门负责人姓名
       ,CAST(A.IS_SITE AS STRING) AS IS_SITE_CODE --厂部确认编码：0是 1否
       ,CASE WHEN CAST(A.IS_SITE AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_SITE AS STRING) = 'TRUE' THEN '否'
              END AS IS_SITE_NAME --厂部确认描述
       ,A.SITE_BY AS SITE_EMP_NUM --厂部确认人工号
       ,I.EMP_NAME AS SITE_EMP_NAME --厂部确认人姓名
       ,CAST(A.IS_ANALYSIS AS STRING) AS IS_ANAL_CODE --是否分析到根本原因编码：0是 1否
       ,CASE WHEN CAST(A.IS_ANALYSIS AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_ANALYSIS AS STRING) = 'TRUE' THEN '否'
              END AS IS_ANAL_NAME --是否分析到根本原因描述
       ,CAST(A.IS_EXECUTE AS STRING) AS IS_EXEQ_CODE --措施是否有效执行编码：0是 1否
       ,CASE WHEN CAST(A.IS_EXECUTE AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_EXECUTE AS STRING) = 'TRUE' THEN '否'
              END AS IS_EXEQ_NAME --措施是否有效执行描述
       ,A.SITE_ADVICE AS SITE_ADVC --厂部意见
       ,A.SITE_DATE --厂部确认日期
       ,CAST(A.IS_IMPLEMENT AS STRING) AS IS_IMP_CODE --整改措施有效实施编码：0是 1否
       ,CASE WHEN CAST(A.IS_IMPLEMENT AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_IMPLEMENT AS STRING) = 'TRUE' THEN '否'
              END AS IS_IMP_NAME --整改措施有效实施描述
       ,CAST(A.IS_IMPROVE AS STRING) AS IS_IMPRV_CODE --改善效果到位编码：0是 1否
       ,CASE WHEN CAST(A.IS_IMPROVE AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_IMPROVE AS STRING) = 'TRUE' THEN '否'
              END AS IS_IMPRV_NAME --改善效果到位描述
       ,CAST(A.IS_TRAIN AS STRING) AS IS_TRAIN_CODE --对新的文件/要求/措施进行培训编码：0是 1否
       ,CASE WHEN CAST(A.IS_TRAIN AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_TRAIN AS STRING) = 'TRUE' THEN '否'
              END AS IS_TRAIN_NAME --对新的文件/要求/措施进行培训描述
       ,CAST(A.IS_CLOSED_LOOP AS STRING) AS IS_CLSE_CODE --措施无法执行闭环(强制闭环)编码：0是 1否
       ,CASE WHEN CAST(A.IS_CLOSED_LOOP AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_CLOSED_LOOP AS STRING) = 'TRUE' THEN '否'
              END AS IS_CLSE_NAME --措施无法执行闭环(强制闭环)描述
       ,A.TRACK_ADVICE AS TRK_ADVC --跟踪意见
       ,A.TRACK_BY AS TRK_EMP_NUM --跟踪确认人工号
       ,J.EMP_NAME AS TRK_EMP_NAME --跟踪确认人姓名
       ,A.TRACK_DATE AS TRK_DATE --跟踪确认日期
       ,CAST(A.IS_EXPERIENCE_LIBRARY AS STRING) AS IS_EXPNC_CODE --是否纳入经验教训库编码：0是 1否
       ,CASE WHEN CAST(A.IS_EXPERIENCE_LIBRARY AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.IS_EXPERIENCE_LIBRARY AS STRING) = 'TRUE' THEN '否'
              END AS IS_EXPNC_NAME --是否纳入经验教训库描述
       ,A.CLOSE_BY AS CLSE_EMP_NUM --结案确认人工号
       ,K.EMP_NAME AS CLSE_EMP_NAME --结案确认人姓名 
       ,A.CLOSE_REMARK AS CLSE_REMARK --结案备注
       ,A.CLOSE_DATE AS CLSE_DATE --结案确认日期
       ,A.RELEVANCE_HANDLE AS REL_ID --关联信息主键
       ,A.RELEVANCE_DETAIL_HANDLE AS REL_DTL_ID --关联信息明细主键
       ,A.RELEVANCE_TYPE AS REL_TYPE_CODE --关联信息类型
       ,CAST(A.STATE AS STRING) AS ST_CODE --是否启用编码：0是 1否
       ,CASE WHEN CAST(A.STATE AS STRING) = 'FALSE' THEN '是'
             WHEN CAST(A.STATE AS STRING) = 'TRUE' THEN '否'
              END AS ST_NAME --是否启用描述
       ,A.CREATE_BY AS CREATE_EMP_NUM --创建人工号
       ,COALESCE(L.EMP_NAME,A.CREATE_BY) AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATE_DATE AS CREATE_TIME --创建时间
       ,A.UPDATE_BY AS UPDATE_EMP_NUM --修改人工号
       ,COALESCE(M.EMP_NAME,A.UPDATE_BY) AS UPDATE_EMP_NAME --修改人姓名
       ,A.UPDATE_DATE AS UPDATE_TIME --修改时间
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_M_QMS_CORRECT_PREVENT_ORDER_1053 A
  LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D B
    ON A.SITE = B.SITE_CODE
  LEFT JOIN T2 AS C
    ON A.HANDLE = C.HANDLE
  LEFT JOIN DM.DM_WF_EMP_INFO_T D
    ON A.SEND_BY = D.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T E
    ON A.RECTIFICATION_BY = E.EMP_NUM
  LEFT JOIN T5 AS F
    ON A.HANDLE = F.HANDLE
  LEFT JOIN T8 AS G
    ON A.HANDLE = G.HANDLE
  LEFT JOIN DM.DM_WF_EMP_INFO_T H
    ON A.ORG_BY = H.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T I
    ON A.SITE_BY = I.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T J
    ON A.TRACK_BY = J.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T K
    ON A.CLOSE_BY = K.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T L
    ON A.CREATE_BY = L.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T M
    ON A.UPDATE_BY = M.EMP_NUM
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T1
    ON A.ORDER_TYPE = T1.CODE
   AND T1.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,correctPreventOrderType'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T2
    ON A.ORDER_STATUS = T2.CODE
   AND T2.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,correctPreventOrderStatus'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T3
    ON A.EVENT_RESOURCE = T3.CODE
   AND T3.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,correctPreventOrderResource'
