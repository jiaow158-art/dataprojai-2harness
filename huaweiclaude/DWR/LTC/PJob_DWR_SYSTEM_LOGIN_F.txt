,ltc_create_date as 
(
select 
'LTC' as system_flag
,bu.user_name user_id
,bu.user_full_name user_name
,substr(min(bu.creation_date),1,10) create_date
FROM SDI.SDI_BASE_MENU_1029 bm
inner JOIN SDI.SDI_BASE_ROLE_MENU_1029 brm ON bm.menu_id = brm.menu_id
inner JOIN SDI.SDI_BASE_ROLE_1029 br ON brm.role_id = br.role_id
inner JOIN SDI.SDI_BASE_RESPONSIBILITY_ROLE_1029 brr ON br.role_id = brr.role_id --无数据
inner JOIN SDI.SDI_BASE_RESPONSIBILITY_1029 bresp ON brr.responsibility_id = bresp.responsibility_id
inner JOIN SDI.SDI_BASE_USER_RESPONSIBILITY_1029 bur ON bresp.responsibility_id = bur.responsibility_id
inner JOIN SDI.SDI_BASE_USERS_1029 bu ON bur.user_id = bu.user_id 
INNER JOIN SDI.SDI_BASE_SAP_ORGANIZATIONAL_STRUCTURE_1029  t on t.employee =bu.user_name
where 1 = 1
and bm.system_code = 'LTC'
and bm.menu_parent_id = 0 
and bu.delete_flag = 0
group by 
bu.user_name
,bu.user_full_name
UNION ALL 
select 
'LTC' as system_flag
,bu.user_name user_id
,bu.user_full_name user_name
,substr(min(bu.creation_date),1,10) create_date
FROM
	SDI.SDI_base_menu_1029 bm
	    inner JOIN SDI.SDI_base_role_menu_1029 brm ON bm.menu_id = brm.menu_id
        inner JOIN SDI.SDI_base_role_1029 br ON brm.role_id = br.role_id
        inner JOIN SDI.SDI_base_responsibility_role_1029 brr ON br.role_id = brr.role_id
        inner JOIN SDI.SDI_base_responsibility_1029 bresp ON brr.responsibility_id = bresp.responsibility_id
        inner JOIN SDI.SDI_base_user_responsibility_1029 bur ON bresp.responsibility_id = bur.responsibility_id
        inner JOIN SDI.SDI_base_users_1029 bu ON bur.user_id = bu.user_id 
		INNER JOIN SDI.SDI_base_organization_grant_1029 g on g.user_id = bu.user_id 
		INNER JOIN SDI.SDI_dt_distributor_info_1029 ddi on ddi.org_id = g.org_id and ddi.DEL_flag =0 and  (ddi.end_date is null or ddi.end_date >current_timestamp)
		inner Join SDI.SDI_sap_distributor_sale_area_1029 sdsa on sdsa.distributor_code = ddi.distributor_code
	where
	 bm.system_code = 'LTC'
	 and  bm.menu_parent_id = 0 
				and bu.delete_flag = 0
				and bu.user_type='OUT'
group by 
bu.user_name
,bu.user_full_name				
)
