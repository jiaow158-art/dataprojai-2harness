INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_PROJ_CTRCT_REL_T
SELECT 
 T.project_class AS busi_class_code	--商机类别
,T2.MEANING  AS busi_class_name	--商机类别
,T.business_code AS busi_code	--商机编号
,T.business_name AS busi_name	--商机名称
,T.contract_code AS cont_code	--合同号
,T.contract_name AS cont_name	--合同名称
,T.customer_code AS cust_code	--客户编码
,T.customer_name AS cust_name	--客户名称
,T.effective_date AS effect_date	--生效日期
,T.invalid_date	--失效日期
,T.sale_org	--销售组织
,T.apply AS employee	--业务员
,T.CONTRACT_STATUS AS cont_status_code --合同状态
,T3.MEANING AS cont_status_name --合同状态名称
,T.wbs	--WBS号
,T.delivery_period	--预计供货周期
,nvl(T.order_discount_rate,0) AS ORD_DISCOUNT_RATE	--整单折扣率
,T.is_allograph	--是否代签
,T.creation_date AS CREATE_DATE	--创建日期
,nvl(T1.contract_amount,0)	AS CONT_AMT--合同金额
,nvl(T1.product_income,0) AS PROD_INCOME_AMT--产品收入
,nvl(T1.freight_revenue,0)	AS FRE_REV_AMT  --运费收入
,nvl(T1.cost_of_sales_amb,0) AS SALES_COST_AMB_AMT	--销售成本阿米巴
,nvl(T1.cost_of_sales_moving,0) AS SALES_COS_AVG_AMT	--销售成本移动平均A
,nvl(T1.cost_of_sales_predict,0) AS PRE_SALES_COST_AMT	--销售成本预估未来
,nvl(T1.product_cost_amb,0) AS PROD_COST_AMB_AMT	--产品成本阿米巴
,nvl(T1.product_cost_moving,0)	AS PROD_COST_MOV_AMT--产品成本移动平均A
,nvl(T1.product_cost_predict,0) AS PRE_PROD_COST_AMT--	产品成本预估未来
,nvl(T1.freight,0) AS FRE_AMT--运费
,nvl(T1.landing_charge,0) AS ASLAND_AMT--卸货费
,nvl(T1.installation_costs,0)	AS INSTALL_AMT--安装费
,nvl(T1.service_charge,0) AS	SER_AMT--服务费
,nvl(T1.excl_tax,0) AS EXCL_TAX_AMT	--不含税销售净额
,nvl(T1.gross_profit_amb,0) AS GROSS_PRO_AMB_AMT	--毛利额阿米巴
,nvl(T1.gross_profit_moving,0)	AS GROSS_PRO_MOV_AMT--毛利额移动平均A
,nvl(T1.gross_profit_predict,0) AS PRE_GROSS_PRO_AMT	--毛利额预估未来
,nvl(T1.gross_margin_amb,0) AS GROSS_MAR_AMB_AMT	--毛利率阿米巴
,nvl(T1.gross_margin_moving,0)	AS GROSS_MAR_MOV_AMT --毛利率移动平均A
,nvl(T1.gross_margin_predict,0) AS	PRE_GROSS_MAR_AMT--毛利率预估未来
,nvl(T1.damage_amb,0)	AS LOSS_AMB_AMT --破损阿米巴
,nvl(T1.damage_moving,0) AS LOSS_MOV_AVG_AMT	--破损移动平均A
,nvl(T1.damage_predict,0)	AS PRE_LOSS_AMT--破损预估未来
,nvl(T1.capital_cost,0) AS CAP_COST_AMT	--资金成本
,nvl(T1.labor_cost,0)	AS LAB_COST_AMT--人工成本
,nvl(T1.make_compensation,0) AS MAKE_COMP_AMT	--售后赔偿
,nvl(T1.storage_cost,0) AS STO_COST_AMT	--仓储费用
,nvl(T1.tray_fee,0) AS TRAY_FEE_AMT	--托盘费
,nvl(T1.bad_debt_loss,0) AS BAD_DEBT_LOSS_AMT 	--坏账损失
,nvl(T1.inventory_falling_price_loss,0) AS INV_FALL_PRI_AMT	--存货跌价损失
,nvl(T1.taxes_and_surcharges_amb,0) AS TAX_ADD_AMB_AMT	--税金及附加阿米巴
,nvl(T1.taxes_and_surcharges_moving,0)	AS TAX_ADD_MOV_AMT--税金及附加移动平均A
,nvl(T1.taxes_and_surcharges_predict,0) AS TAX_ADD_PRE_AMT	--税金及附加预估未来
,nvl(T1.total_expenses_amb,0)	AS TOTAL_EXP_AMB_AMT --费用合计阿米巴
,nvl(T1.total_expenses_moving,0) AS TOTAL_EXP_MOV_AMT	--费用合计移动平均A
,nvl(T1.total_expenses_predict,0)	AS TOTAL_EXP_PRE_AMT --费用合计预估未来
,nvl(T1.cost_rate_amb,0) AS COST_RATE_AMB_AMT	--费用率阿米巴
,nvl(T1.cost_rate_moving,0) AS COST_RATE_MOV_AMT	--费用率移动平均A
,nvl(T1.cost_rate_predict,0) AS COST_RATE_PRE_AMT	--费用率预估未来
,nvl(T1.total_profit_amb,0) AS	TOTAL_PRO_AMB_AMT--利润总额阿米巴
,nvl(T1.total_profit_moving,0)	AS TOTAL_PRO_MOV_AMT--利润总额移动平均A
,nvl(T1.total_profit_predict,0) AS TOTAL_PRO_PRE_AMT	--利润总额预估未来
,nvl(T1.profit_margin_amb,0) AS PRO_MAR_AMB_AMT	--利润率阿米巴
,nvl(T1.profit_margin_moving,0) AS PRO_MAR_MOV_AMT	--利润率移动平均A
,nvl(T1.profit_margin_predict,0) AS PRO_MAR_PRE_AMT	--利润率预估未来
,T4.service_supplier AS SER_SUPLY_NAME -- 服务商名称
,T4.service_content AS SER_CON_NAME  --服务内容
,T.delete_flag AS DEL_FLAG -- 删除标识 
,current_date() AS DW_LAST_UPDATE_DATE 
,T5.MEANING AS DEPT_NAME --事业部
,T.NEW_FLAG --合同版本标识
FROM SDI.SDI_CONTRACT_1033 T
LEFT JOIN SDI.SDI_CONTRACT_PROFIT_1033 T1 
ON T.CONTRACT_ID = T1.CONTRACT_ID
LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 T2 
ON T2.LOOKUP_TYPE = 'BUSINESS_CATEGORY'
AND T2.LOOKUP_CODE = T.project_class
LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 T3 
ON T3.LOOKUP_TYPE = 'CONTRACT_SEAL_STATUS'
AND T3.LOOKUP_CODE = T.CONTRACT_STATUS
LEFT JOIN (SELECT A.CONTRACT_ID,CONCAT_WS(';', COLLECT_SET(A.SERVICE_SUPPLIER)) AS SERVICE_SUPPLIER,CONCAT_WS(';', COLLECT_SET(B.MEANING)) AS SERVICE_CONTENT 
FROM SDI.SDI_CONTRACT_FEE_1033 A 
LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 B 
ON B.LOOKUP_TYPE = 'SERVICE_CONTENT'
AND B.LOOKUP_CODE = A.SERVICE_CONTENT
GROUP BY A.CONTRACT_ID) T4 
ON T.CONTRACT_ID = T4.CONTRACT_ID
LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T5
ON T.delivery = T5.LOOKUP_CODE
AND T5.LOOKUP_TYPE = 'DIVISION'
AND T5.DELETE_FLAG = 0
