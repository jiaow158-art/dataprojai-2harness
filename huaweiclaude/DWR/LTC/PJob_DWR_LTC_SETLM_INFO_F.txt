WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
--LTC结算单
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_SETLM_INFO_F
SELECT  A.SETTLEMENT_BILL_ID AS SETLM_BILL_ID --主键
       ,A.SETTLEMENT_BILL_CODE AS SETLM_BILL_NUM --结算流程号
       ,B.WBS_NUM --WBS号
       ,A.BUSINESS_CODE AS BUSI_CODE --商机编码
       ,A.BUSINESS_NAME AS BUSI_NAME --商机名称
       ,A.SALES_UNIT_CODE --销售单位编码
       ,A.SALES_UNIT AS SALES_UNIT_NAME --销售单位名称
       ,CASE WHEN COALESCE(A.SALES_DEPARTMENT_CODE,'') <> '' THEN A.SALES_DEPARTMENT_CODE ELSE NULL END AS SALE_DEPT_CODE --销售部门编码
       ,CASE WHEN COALESCE(A.SALES_DEPARTMENT,'') <> '' THEN A.SALES_DEPARTMENT ELSE NULL END AS SALE_DEPT_NAME --销售部门名称
       ,A.BIG_ACCOUNT AS BIG_CUST_NAME --大客户名称
       ,A.SALES_MAN_CODE AS SALE_EMP_NUM --业务员编码
       ,A.SALES_MAN_NAME AS SALE_EMP_NAME --业务员名称
       ,CASE WHEN COALESCE(A.ACCOUNT_CODE,'') <> '' THEN A.ACCOUNT_CODE ELSE NULL END AS CUST_NUM --客户编码
       ,CASE WHEN COALESCE(A.ACCOUNT_NAME,'') <> '' THEN A.ACCOUNT_NAME ELSE NULL END AS CUST_NAME --客户名称
       ,A.PROJECT_FINISH_TIME  AS PROJ_FIN_TIME --项目竣工时间
       ,A.SETTLEMENT_DUE_DATE AS SETLM_DUE_DATE --结算回单到期日
       ,A.BILL_STATUS AS BILL_ST_CODE --单据状态编码
	   ,T1.MEANING AS BILL_ST_NAME --单据状态描述
       ,A.SETTLEMENT_AMOUNT AS SETLM_AMT --结算金额
       ,A.REGISTER_STATUS AS REGS_ST_CODE --登记状态编码
	   ,T2.MEANING AS REGS_ST_NAME --登记状态描述
       ,A.RECIPIENT_AMOUNT AS RCPNT_AMT --应收结算金额
       ,A.BACK_DATE --回单日期
       ,A.VERSION_NUM AS VERS_NUM --版本号
       ,A.CREATION_DATE AS CREATE_DATE --创建日期
       ,A.LAST_UPDATE_DATE AS UPDATE_DATE --更新日期
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_BILL_1033 A
  LEFT JOIN T AS T1
    ON A.BILL_STATUS = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'SETTLEMENT_BILL_STATUS'
  LEFT JOIN T AS T2
    ON A.REGISTER_STATUS = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'REGISTER_STATUS'
  LEFT JOIN DWRLTC.DWR_BO_BUSI_INFO_F B
    ON A.BUSINESS_CODE = B.BUSI_CODE
 WHERE A.DELETE_FLAG = 0
