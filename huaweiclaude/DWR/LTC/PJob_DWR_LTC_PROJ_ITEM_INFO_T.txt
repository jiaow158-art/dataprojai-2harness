WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
),T2 AS(
SELECT B.PROJECT_ID AS PROJ_ID
      ,B.MATERIAL_PRICE_ID
      ,B.PRODUCT_ATTRIBUTE AS MATERIAL_SRC_CODE --产品来源编码
      ,T1.MEANING AS MATERIAL_SRC_NAME --产品来源描述
	  ,B.MATERIAL AS MATERIAL_NUM --物料编码
	  ,B.MATERIAL_DESCRIPTION AS MATERIAL_NAME --物料名称
	  ,B.category_code AS MATERIAL_CAT_CODE --品类编码
	  ,B.CATEGORY AS MATERIAL_CAT_NAME --品类描述
	  ,B.product_function_code AS MATERIAL_FUNC_CODE --产品功能编码
	  ,B.product_function AS MATERIAL_FUNC_NAME --产品功能名称
	  ,B.product_place_code AS MATERIAL_PLACE_CODE --产品工艺编码
	  ,B.product_place_code AS MATERIAL_PLACE_NAME --产品工艺名称
	  ,B.white_code --胚体白度编码
	  ,B.white AS WHITE_NAME-- 胚体白度名称
	  ,B.color_code --色调编码
	  ,B.color AS COLOR_NAME -- 色调名称
	  ,B.high_value_desc_code as high_value_code --高值说明编码
	  ,B.high_value_desc AS high_value_name --高值说明名称
	  ,B.PRODUCT_SPECIFICATION AS DIMENSION --规格
	  ,B.kit_height AS HEIGHT --高度
	  ,B.bottom_slab_code --胚体编码
	  ,B.bottom_slab  AS bottom_slab_name -- 胚体名称
	  ,B.product_channel_code AS MATERIAL_CHNL_CODE --产品渠道编码
	  ,B.PRODUCT_CHANNEL AS MATERIAL_CHNL_NAME --产品渠道描述
	  ,COALESCE(B.ENGINEERING_SIDE_PRICE,0) AS GCF_PRC --工程方价格
	  ,COALESCE(B.APPLY_QUOTATION,0) AS APLY_PRC --申请价格
	  ,COALESCE(B.REVIEW_DOSAGE,0) AS APLY_QTY --申请数量
	  ,COALESCE(B.PRODUCT_NAKED_PRICE,0) AS APPR_NAK_PRC --审批裸价
	  ,COALESCE(B.REVIEW_QUOTATION,0) AS APPR_PRC --审批价格
	  ,COALESCE(B.AMOEBA_COST,0) AS AMB_COST --阿米巴成本
	  ,COALESCE(B.AMOEBA_PROFIT,0) AS AMT_PRFT_AMT --阿米巴毛利额
	  ,COALESCE(B.AMOEBA_MARGIN,0) AS AMT_PRFT_RATE --阿米巴毛利率

	  ,COALESCE(B.ACOST_PROFIT,0) AS STD_A_AMT --标A成本毛利额
	  ,COALESCE(B.ACOST_MARGIN,0) AS STD_A_RATE --标A成本毛利率

      ,COALESCE(B.EP_PRICE,0) AS EP_PRC	--EP牌价
      ,COALESCE(B.DISCOUNT_RATE,0) AS DISC_RATE --折扣率

	  ,COALESCE(B.SERVICE_RATE,0) AS SRVC_RATE --综合费率

      ,B.LAST_UPDATE_DATE AS ITEM_UPDATE_DATE --行更新日期

	  ,C.UNIT_AREA * COALESCE(B.REVIEW_DOSAGE,0) AS APLY_AREA --申请面积
	  ,'LTC' AS DATA_SRC_CODE --数据来源编码
	  ,T3.MEANING as price_type --申请价格类型 空代表手动填写
	  ,buyout_price --一口价
	  ,E.price_directory_name  AS PRI_POLICY_NAME--价格政策名称
	  ,D.category||D.product_specification||D.product_function||D.product_place_code||D.embryoid_body||D.white||D.kit_height||D.color||D.product_channel||D.height_define AS MAT_PRI_GAR --物料价格组合
  FROM SDI.SDI_PROJECT_REVIEW_MATERIAL_PRICE_1033 B
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T C
    ON B.MATERIAL = C.MATERIAL_NUM
  LEFT JOIN T AS T1
    ON B.PRODUCT_ATTRIBUTE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'LTC_PRODUCT_ATTRIBUTE'
  LEFT JOIN T AS T2
    ON B.IS_INSTALL = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'YES_OR_NO'
   LEFT JOIN T AS T3
    ON B.PRICE_TYPE = T3.LOOKUP_CODE
   AND T3.LOOKUP_TYPE = 'LTC_PRICE_TYPE'
   LEFT JOIN SDI.SDI_PRICE_MANAGE_MATERIAL_CZJXS_1033 D 
    ON B.price_manage_line_id = D.price_manage_line_id
   LEFT JOIN SDI.SDI_PRICE_MANAGE_CZJXS_1033 E
    ON D.price_manage_id = E.price_manage_id
 WHERE B.DELETE_FLAG = 0
 
 UNION ALL
 
SELECT B.PROJECT_ID AS PROJ_ID
      ,B.MATERIAL_PRICE_ID

	  ,COALESCE(B.ACOST,0) AS STD_A_COST --标A成本
	  ,COALESCE(B.ACOST_PROFIT,0) AS STD_A_AMT --标A成本毛利额
	  ,COALESCE(B.ACOST_MARGIN,0) AS STD_A_RATE --标A成本毛利率
      ,COALESCE(B.FORECAST_FUTURE_COSTS,0) AS FCST_COST	--预测未来成本
      ,COALESCE(B.FORECAST_FUTURE_COSTS_PROFIT,0) AS FCST_AMT --预测未来成本毛利额
      ,COALESCE(B.FORECAST_FUTURE_COSTS_MARGIN,0) AS FCST_RATE --预测未来成本毛利率
      ,0 AS DSTR_GRS_RATE --经销商毛利率
      ,0 AS TRNSP_COST --运输成本
	  --（审批价格-工程方价格*集采服务费率/100-工程方价格*重包费率*0.13/100）*合同用量或申请数量
	  ,COALESCE(B.REVIEW_QUOTATION,0) * COALESCE(B.REVIEW_DOSAGE,0) AS SALE_AMT --销售额
	  ,B.UNIT --单位
      ,COALESCE(B.WEIGHT_VOLUME,0) AS WT_VOL --重量体积
      ,COALESCE(B.EP_PRICE,0) AS EP_PRC	--EP牌价
      ,COALESCE(B.DISCOUNT_RATE,0) AS DISC_RATE --折扣率
	  ,B.IS_INSTALL AS IS_INSTL_CODE --是否包安装编码
	  ,T2.MEANING AS IS_INSTL_NAME --是否包安装描述
	  ,COALESCE(B.INSTALLATION_COST,0) AS INSTL_PRC	--安装单价
	  ,COALESCE(B.CENTRALIZED_SERVICE_RATE,0) AS CTRL_SRVC_RATE	--集采服务费率
	  ,COALESCE(B.ORDER_SERVICE_RATE,0) AS ORDER_SRVC_RATE --抢单服务费率
	  ,COALESCE(B.GROUND_SERVICE_RATE,0) AS GRND_SRVC_RATE --落地服务费率
	  ,COALESCE(B.ADVANCE_SERVICE_RATE,0) AS ADVN_SRVC_RATE --垫资服务费率
	  ,COALESCE(B.SERVICE_RATE,0) AS SRVC_RATE --综合费率
	  ,COALESCE(B.BUSINESS_COMMISSION_RATE,0) AS BUSI_COMS_RATE --业务提成率
      ,B.CREATION_DATE AS ITEM_CREATE_DATE --行创建日期
      ,B.LAST_UPDATE_DATE AS ITEM_UPDATE_DATE --行更新日期
      ,C.UNIT_AREA --单片面积
	  ,C.UNIT_AREA * COALESCE(B.REVIEW_DOSAGE,0) AS APLY_AREA --申请面积
	  ,'LTC卫浴经销商' AS DATA_SRC_CODE --数据来源编码
	  ,null as price_type --申请价格类型 空代表手动填写
	  ,null as buyout_price --一口价
	  ,null AS PRI_POLICY_NAME--价格政策名称
	  ,null AS MAT_PRI_GAR --物料价格组合
  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_MATERIAL_PRICE_1033 B
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T C
    ON B.MATERIAL = C.MATERIAL_NUM
  LEFT JOIN T AS T1
    ON B.PRODUCT_ATTRIBUTE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'LTC_PRODUCT_ATTRIBUTE'
  LEFT JOIN T AS T2
    ON B.IS_INSTALL = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'YES_OR_NO'
 WHERE B.DELETE_FLAG = 0
)
