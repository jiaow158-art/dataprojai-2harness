),TT AS(
SELECT MAX(VOUCHER_POST_DATE) AS VOUCHER_POST_DATE --过账日期
      ,DELIVERY_IM --提货单号
      ,DELIVERY_PROJECT --提货单行号
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F
 WHERE MOVE_TYPE IN('601','653')
 GROUP BY DELIVERY_IM --提货单号
      ,DELIVERY_PROJECT --提货单行号
),T1 AS(
SELECT NAME
      ,TYPE
	  ,VAL
	  ,ROW_NUMBER() OVER(PARTITION BY TYPE,NAME ORDER BY ORDER_BY ASC) AS RN
  FROM SDI.SDI_S_LST_OF_VAL_1002
 WHERE ACTIVE_FLG = 'Y'
   AND LANG_ID = 'CHS'
   AND COALESCE(TRANSLATE_FLG,'Y') = 'Y'
),T2 AS(
SELECT NAME
      ,TYPE
	  ,VAL
  FROM T1
 WHERE RN = 1
),TTT AS(
SELECT A.VBELN

      ,A.ID_OSS
  FROM SDI.SDI_VBAK_1001 A
 WHERE A.OSS_VBELN <> ''
   AND A.PS_PSP_PNR <> '00000000'
