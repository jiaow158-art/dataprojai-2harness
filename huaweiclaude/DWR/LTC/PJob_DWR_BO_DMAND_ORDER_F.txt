WITH TT AS(
SELECT DISTINCT A.WBS
               ,ORDERS_PUSH_METHOD
			   ,B.MEANING
  FROM SDI.SDI_CONTRACT_1033 A
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B
    ON A.ORDERS_PUSH_METHOD = B.LOOKUP_CODE
   AND B.LOOKUP_TYPE = 'ORDER_PUSH_METHOD'
   AND B.ENABLED_FLAG = 'Y'
 WHERE COALESCE(A.ORDERS_PUSH_METHOD,'') <> ''
   AND A.DELETE_FLAG = 0
),TX AS(
SELECT NAME
      ,TYPE
	  ,VAL
	  ,ROW_NUMBER() OVER(PARTITION BY TYPE,NAME ORDER BY ORDER_BY ASC) AS RN
  FROM SDI.SDI_S_LST_OF_VAL_1002
 WHERE ACTIVE_FLG = 'Y'
   AND LANG_ID = 'CHS'
   AND COALESCE(TRANSLATE_FLG,'Y') = 'Y'
),T1 AS(
SELECT NAME
      ,TYPE
	  ,VAL
  FROM TX
 WHERE RN = 1 
),T2 AS(
SELECT A.PARTY_UID
      ,B.LAST_NAME
      ,C.LOGIN
  FROM SDI.SDI_S_PARTY_1002 A
  LEFT JOIN SDI.SDI_S_CONTACT_1002 B
    ON A.ROW_ID = B.PAR_ROW_ID
  LEFT JOIN SDI.SDI_S_USER_1002 C
    ON A.ROW_ID = C.PAR_ROW_ID
),T3 AS(
SELECT I1.DP_REQUITE_ORDE_ITEMR_ID									
      ,SUM(QTI1.X_QTY_REQ) NUMBER1									
  FROM SDI.SDI_CX_PURCH_ITM_1002 I1									
 INNER JOIN SDI.SDI_S_ORDER_ITEM_1002 QTI1
    ON I1.ROW_ID = QTI1.X_DP_SHIPMENT_PLAN_ORDER_ITEM  
 INNER JOIN SDI.SDI_S_ORDER_1002 OD1
    ON QTI1.ORDER_ID = OD1.ROW_ID
 INNER JOIN SDI.SDI_CX_PURCH_1002 C1
    ON I1.HEAD_ID = C1.ROW_ID
   AND C1.ROW_ID = OD1.X_DP_SHIPMENT_PLAN_ORDER_ID	
 WHERE QTI1.X_DP_BATCH_CHILD_ID IS NOT NULL
   AND OD1.X_SAP_NUM IS NOT NULL									
   AND OD1.X_OD_SUBTYPE = 'Project Order'									
   AND OD1.STATUS_CD = '已完成'									
   AND (QTI1.X_CANCEL_REASON <> '52' OR QTI1.X_CANCEL_REASON IS NULL)
   AND QTI1.X_OPERA_FLAG <> 'D'									
   AND QTI1.STATUS_CD <> '无效'									
--AND C1.DP_REQUITE_ORDER_ID = '1-GOGR57'									
 GROUP BY I1.DP_REQUITE_ORDE_ITEMR_ID
),T4 AS(
SELECT I2.DP_REQUITE_ORDE_ITEMR_ID
      ,SUM(COALESCE(I2.TRAN_NUM,0)) NUMBER2
  FROM SDI.SDI_CX_PURCH_ITM_1002 I2
  LEFT JOIN SDI.SDI_CX_PURCH_1002 PURCHASE2
    ON I2.HEAD_ID = PURCHASE2.ROW_ID  
 INNER JOIN SDI.SDI_CX_PURCH_ITM_1002 II2
    ON II2.ROW_ID = I2.DP_REQUITE_ORDE_ITEMR_ID
 WHERE PURCHASE2.STATUS IN('Purchase Processing','New')
   --AND II2.HEAD_ID = '1-GOGR57'
 GROUP BY I2.DP_REQUITE_ORDE_ITEMR_ID
)
