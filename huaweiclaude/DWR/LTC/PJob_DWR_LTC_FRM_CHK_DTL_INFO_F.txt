INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_FRM_CHK_DTL_INFO_F
SELECT  A.FORMAL_CHECK_DETAIL_ID AS FRM_CHK_DTL_ID --正式细表ID
       ,A.FORMAL_CHECK_ID AS FRM_CHK_ID --正式主表ID
       ,A.VIRTUAL_CHECK_DETAIL_ID AS VRTUL_CHK_DTL_ID --虚拟细表ID
       ,A.LINE_ID AS DSPT_LINE_ID --发运单行ID
	   ,T.WBS_NUM --WBS号
	   ,T.SETLM_MODE_CODE --结算方式编码
	   ,T.SETLM_MODE_NAME --结算方式描述
	   ,T.CHK_TYPE_CODE --验收单方式编码
	   ,T.CHK_TYPE_NAME --验收单方式描述
       ,T.PRDCT_RQST_TIME --预计请款日期
       ,T.RQST_ST_CODE --请款状态编码
	   ,T.RQST_ST_NAME --请款状态描述
       ,T.AFFR_RSULT_CODE --确认结果编码
	   ,T.AFFR_RSULT_NAME --确认结果描述	   
	   ,B.WBS AS PLAN_ITEM_NUM --计划单行号
       ,COALESCE(A.CURRENT_FORMAL_QTY,0) AS CUR_CHK_QTY --本次验收数
       ,COALESCE(A.CURRENT_FORMAL_AMOUNT,0) AS CUR_CHK_AMT --本次验收金额
       ,COALESCE(A.CURRENT_REQUEST_QTY,0) AS CUR_RQST_QTY --本次请款数
       ,COALESCE(A.CURRENT_REQUEST_AMOUNT,0) AS CUR_RQST_AMT --本次请款金额
       ,A.DEMAND_CODE AS DMAND_NUM --要货单号
       ,A.MATERIAL_CODE AS MATERIAL_NUM --物料编码
       ,A.MATERIAL_NAME --物料名称
       ,A.MATERIAL_SPEC AS DIMENSION --规格型号	   
       ,A.UNIT --单位
       ,COALESCE(A.DEMAND_QTY,0) AS DMAND_QTY --要货单需求数
       ,A.PRICE AS PRC --单价
       ,A.PICK_CODE AS DELV_NUM --提货单号
	   ,B.DELIVERY_LINE_NO AS DELV_LINE_NUM --提货单行号
       ,A.TRANSPORT_ENTRUST_CODE AS DSPT_NUM --运输委托号
	   ,B.PROJECT_PLAN_NO AS PROJ_PLAN_NUM --项目计划号
	   ,B.LOADING_PLAN_NO AS LOAD_PLAN_NUM --装载计划号
       ,A.PICK_DATE AS DELV_DATE --提货日期
       ,COALESCE(A.PICK_QTY,0) AS DELV_QTY --提货数量
       ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	   ,T1.USER_NAME AS CREATE_EMP_NUM --创建人工号
	   ,T1.USER_FULL_NAME AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATION_DATE AS CREATE_TIME --创建时间
       ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	   ,T2.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	   ,T2.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名
       ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
       ,A.VERSION_NUM AS VERS_NUM --版本号
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_FORMAL_CHECK_DETAIL_1033 A
 INNER JOIN DWRLTC.DWR_LTC_FRM_CHK_INFO_F T
    ON A.FORMAL_CHECK_ID = T.FRM_CHK_ID
  LEFT JOIN SDI.SDI_DELIVER_DESPATCH_LINE_1033 B
    ON A.LINE_ID = B.LINE_ID
   AND B.DELETE_FLAG = 0
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T1
    ON A.CREATED_BY = T1.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T2
    ON A.LAST_UPDATED_BY = T2.USER_ID
 WHERE A.DELETE_FLAG = 0
