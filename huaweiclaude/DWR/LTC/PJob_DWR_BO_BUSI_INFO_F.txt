WITH T0 AS(
SELECT O.BUSINESS_ID
      ,O.PRODUCT_CATEGORY
      ,TABLE_VIEW.NEW_PRODUCT_CATEGORY
  FROM SDI.SDI_BUSINESS_OPPORTUNITY_1033 O
LATERAL VIEW EXPLODE(SPLIT(PRODUCT_CATEGORY,',')) TABLE_VIEW AS NEW_PRODUCT_CATEGORY
),TTT AS(
SELECT T0.BUSINESS_ID
      ,A.MEANING
  FROM T0
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 A
    ON T0.NEW_PRODUCT_CATEGORY = A.LOOKUP_CODE
   AND A.LOOKUP_TYPE = 'PRODUCT_CATEGORY'
   AND A.DELETE_FLAG = 0
),TT AS(
SELECT BUSINESS_ID
      ,CONCAT_WS(',',COLLECT_SET(CAST(MEANING AS STRING))) AS MEANING
  FROM TTT
 GROUP BY BUSINESS_ID
)
,TC AS(
SELECT B.WBS_SPIT_ELEMENT AS WBS
      ,A.PARTNER AS CUST_NUM
	  ,SUM(A.CREDIT_LIMIT) AS CREDIT_AMT
  FROM SDI.SDI_ZTSDXD01_1001 A
  LEFT JOIN (SELECT DISTINCT WBS_SPIT_ELEMENT,WBS_ELEMENT FROM DWROTD.DWR_SO_PROJECT_WBS_F) B
    ON A.PS_PSP_PNR = B.WBS_ELEMENT
 WHERE A.MANDT = 800
   AND A.CREDIT_SGMNT <> 0000
 GROUP BY B.WBS_SPIT_ELEMENT
         ,A.PARTNER
)
