WITH XSSR AS (
SELECT 
material
,sum(dept_inner_cost + zcbtzz + cross_cost + standard_price_a + cost_1n) as standard_price_a
,sum(zxssl) AS SALE_QTY
,period_id_m
from  DM.DM_AMBV2_PROFIT_T a
 left join dm.dm_rpt_sales_group_t d on
a.sales_grp = d.node_name9
where 
A.cust_group <> 'Z0'
AND A.doc_type  <> 'ZT50'
and d.node_name2 = 'H11000001'
and substring(period_id_m,1,4) >= '2023'
and  period_id_m = '${period_id_m}'
group by  material ,period_id_m
)
--算年累计值
,XSSR_S AS (
SELECT   
material
,period_id_m
,sum(standard_price_a)  over  (partition by substring(period_id_m,1,4),material  order by period_id_m ROWS BETWEEN UNBOUNDED PRECEDING  AND CURRENT ROW) as standard_price_a_s
,sum( SALE_QTY)   over  (partition by  substring(period_id_m,1,4),material order by period_id_m ROWS BETWEEN UNBOUNDED PRECEDING  AND CURRENT ROW) as SALE_QTY_s
from  
xssr  
)

AND substring(t2.approval_date,1,7) =  t4.period_id_m
where t2.approval_date is not null and substring(t2.approval_date,1,4) >= '2023'
