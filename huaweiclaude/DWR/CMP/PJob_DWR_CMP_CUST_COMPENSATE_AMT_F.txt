--赔偿金额信息表
-- DWRCMP.DWR_CMP_CUST_COMPENSATE_AMT_F
INSERT OVERWRITE TABLE DWRCMP.DWR_CMP_CUST_COMPENSATE_AMT_F
SELECT 
SUBSTR(T2.APPROVAL_DATE,1,7) AS  APPROVAL_MONTH--批复月份 
,T2.APPROVAL_DATE   --批复日期
, coalesce (T2.CUST_COMP_QTY,0) --客诉宗数
,T2.cc_class  --客诉分类
,T2.appeal_num --客诉流水号
,T2.CUST_CODE   --客户编码
,T2.CUST_NAME   --客户名称
,T2.CUST_COMP_channel_name   --客诉渠道
,T2.requestid   --请求id
,T2.Appeal_type_name   --申诉类型
,T2.SUB_Appeal_time   --提出申诉时间
,T2.contacts   --联系人
,T2.Telephone   --联系电话
,T2.is_SAP_ORDER   --是否有SAP订单号
,T2.consumer_NAME   --顾客名称
,T2.consumer_tel   --顾客电话
,T2.consumer_add   --顾客地址
,T2.DTD_SRVC_NAME   --上门服务人员
,T2.service_tel   --服务人员电话
,T2.consumer_purch_date   --顾客购买日期
,T2.PROD_INSTALL_DATE   --产品安装日期
,T2.is_quality_issue_name   --是否属于质量问题范围
,T2.is_returned_name   --是否退回
,T2.compensation_type_name   --补偿类型
,T2.compensation_method   --赔偿方式
,T2.quality_lel_name   --质量级别
,T2.CUST_COMP_expectant   --客户期望值
,T2.SALES_ORG_FINANCIAL_NODES   --销售组织财务节点
,T2.RESP_NAME   --责任方名称主表
,T2.RESP_CODE   --责任方编码主表
,T2.IS_QC_CHECK   --是否需要品管核查判定
,T2.DEPARTMENT   --所属部门
,T2.Payment_meth_name   --入账方式
,T2.doc_date   --凭证日期
,T2.Point_voucher     --积分凭证
,T2.CURR_NODENAME   --当前节点名称
,T2.INDTY_amount --赔偿金额
,T2.cust_serv_rep_feedback    --客户服务专员意见
,T2.cust_serv_responsible_feedback  --客户服务责任人意见
,T2.sys_manager_feedback    --系统总经理意见
,T2.warehouse_recipient   --仓库收件人
,T2.cust_serv_rep_handler  --客户服务专员处理人员 
,T2.cust_serv_rep_time  --客户服务专员处理时间
,T2.total_comp_amount   --赔付产品总金额
,T2.total_comp_area   --赔偿产品总面积
,T2.complainant_name   --申诉人名称
,T2.complainant_id  --申诉人ID号
,T2.process_date  --流程处理日期
,T2.process_name  --流程名称
,T2.complaint_review_result_name  --申诉资料审核结果
,T2.prod_category_name    --产品分类
,T2.company  --公司
,T2.mkt_department  --营销部
,T2.current_node  --当前节点
,T2.is_return_completed  --是否完成退货入库手续
,T1.id
,T1.mainid
,T1.YQFM    AS PUB_SENT --舆情
,T1.PKGS    AS COMP  --公司
,T1.PKXSZZ  AS compensation_batch --赔偿批次
,T1.PKFKF   AS payer --付款方
,T1.PKPFJE  as compensation_amt  --赔付金额
,T1.PKRZPZ  as acct_voucher --入账凭证
,T1.PKRZRQ  as post_date --入账日期
,T1.PKKHBM  as cust_code --客户编码
,T1.PKKHMC  as cust_name --客户名称
,T1.PKZRFMC as resp_name --责任方名称
,T1.PKZRFDM as resp_code --责任方代码
,T1.tssccyyfl_mx3 as complaint_reason --投诉深层次原因分类
,CASE WHEN substring(T1.pkzrfdm,-4) = '1000' THEN '总部'  
     WHEN substring(T1.pkzrfdm,-4) = '1200' THEN '淄博'  
     WHEN substring(T1.pkzrfdm,-4) = '1300' THEN '清远'  
     WHEN substring(T1.pkzrfdm,-4) = '1500' THEN '丰城'  
     WHEN substring(T1.pkzrfdm,-4) = '1600' THEN '澧县'  
     WHEN substring(T1.pkzrfdm,-4) = '1800' THEN '重庆'  
     WHEN substring(T1.pkzrfdm,-4) = '1900' THEN '山西'  
     ELSE '外协' END AS RESP_CONV --责任方
,current_timestamp   AS  dw_last_update_date
,T2.RESP_CONV_QTY   --责任方数量
FROM 
SDI.SDI_FORMTABLE_MAIN_565_dt3_1004 T1
LEFT JOIN  
DWRCMP.DWR_CMP_CUST_COMPLAINT_INFO_F T2 
ON T1.mainid = T2.id
where 
SUBSTR(T2.APPROVAL_DATE,1,4) >= '2023'
