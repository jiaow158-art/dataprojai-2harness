-- DWRCMP.DWR_CMP_CUST_COMPLAINT_INFO_F 
with  tmp as (
select 
T1.requestid  AS requestid  --请求id
, case when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  0  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 5000 then 1
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  5000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 20000 then 2
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  20000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 50000 then 3
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0)  >= 50000 then 4
            END  AS A1 ,
             case when coalesce(PFCPZMJ,0) >  0  
             and coalesce(PFCPZMJ,0) <= 5000 then 1
            when coalesce(PFCPZMJ,0) >  5000 
            and  coalesce(PFCPZMJ,0) <= 1000 then 2
            when coalesce(PFCPZMJ,0) >  1000 
            and  coalesce(PFCPZMJ,0) <= 15000 then 3
            when coalesce(PFCPZMJ,0)  >= 50000 then 4
        end AS A2            
  ,  GREATEST(
 case when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  0  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 5000 then 1
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  5000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 20000 then 2
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  20000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 50000 then 3
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0)  >= 50000 then 4
            else 0 
            END ,
             case when coalesce(PFCPZMJ,0) >  0  
             and coalesce(PFCPZMJ,0) <= 5000 then 1
            when coalesce(PFCPZMJ,0) >  5000 
            and  coalesce(PFCPZMJ,0) <= 1000 then 2
            when coalesce(PFCPZMJ,0) >  1000 
            and  coalesce(PFCPZMJ,0) <= 15000 then 3
            when coalesce(PFCPZMJ,0)  >= 50000 then 4
            else 0
        end
)
AS max_level
from SDI.SDI_FORMTABLE_MAIN_565_1004 t1
where BCFS = 2  
)
,tmp2 as (
select 
T1.requestid  AS requestid  --请求id
--赔款-不赔物
,case when   BCFS = 0 
 then  case when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  0  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 5000 then '16748' 
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  5000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 20000 then '16750' 
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0) >  20000  
            and coalesce(PCJE,0) + coalesce(PFCPZJE,0) <= 50000 then '269206' 
            when coalesce(PCJE,0) + coalesce(PFCPZJE,0)  >= 50000 then '16769' 
        end 
     --赔物不赔款   
      WHEN  BCFS = 1  
 then  case when coalesce(PFCPZMJ,0) >  0  
             and coalesce(PFCPZMJ,0) <= 5000 then '16748' 
            when coalesce(PFCPZMJ,0) >  5000 
            and  coalesce(PFCPZMJ,0) <= 1000 then '16750' 
            when coalesce(PFCPZMJ,0) >  1000 
            and  coalesce(PFCPZMJ,0) <= 15000 then '269206' 
            when coalesce(PFCPZMJ,0)  >= 50000 then '16769'    
        end
      WHEN  BCFS in ('4','5')
 then  '16748'         
             end as nodeid

from SDI.SDI_FORMTABLE_MAIN_565_1004  t1
where BCFS is not null  and BCFS <> 2 
union all  
SELECT 
requestid  --请求id
,case when max_level = 1 then '16748' when max_level = 2 then '16750' when max_level = 3 then '269206' when max_level = 4  then '16769' end as  nodeid 
from  
tmp t1 
)
, TEX AS (
select 
t2.selectvalue,t2.selectname,T1.fieldname
from SDI.SDI_WORKFLOW_BILLFIELD_1004 T1 
JOIN  SDI.SDI_workflow_selectitem_1004 T2   
ON  T1.id = T2.fieldid 
where 
billid = '-565'  
)
, LC1 AS (
select 
T1.LSH AS appeal_num,
CASE WHEN substring(T2.PCZRFDM,-4) = '1000' THEN '总部'  
     WHEN substring(T2.PCZRFDM,-4) = '1200' THEN '淄博'  
     WHEN substring(T2.PCZRFDM,-4) = '1300' THEN '清远'  
     WHEN substring(T2.PCZRFDM,-4) = '1500' THEN '丰城'  
     WHEN substring(T2.PCZRFDM,-4) = '1600' THEN '澧县'  
     WHEN substring(T2.PCZRFDM,-4) = '1800' THEN '重庆'  
     WHEN substring(T2.PCZRFDM,-4) = '1900' THEN '山西'  
     ELSE '外协' END AS RESP_CONV --责任方
from  
SDI.SDI_FORMTABLE_MAIN_565_dt2_1004  t2
LEFT JOIN 
SDI.SDI_FORMTABLE_MAIN_565_1004   T1 
ON 
T2.MAINID =  T1.ID  
union all  
select 
T1.LSH AS appeal_num,
CASE WHEN substring(T2.pkzrfdm,-4) = '1000' THEN '总部'  
     WHEN substring(T2.pkzrfdm,-4) = '1200' THEN '淄博'  
     WHEN substring(T2.pkzrfdm,-4) = '1300' THEN '清远'  
     WHEN substring(T2.pkzrfdm,-4) = '1500' THEN '丰城'  
     WHEN substring(T2.pkzrfdm,-4) = '1600' THEN '澧县'  
     WHEN substring(T2.pkzrfdm,-4) = '1800' THEN '重庆'  
     WHEN substring(T2.pkzrfdm,-4) = '1900' THEN '山西'  
     ELSE '外协' END AS RESP_CONV --责任方
from  
SDI.SDI_FORMTABLE_MAIN_565_dt3_1004  t2
LEFT JOIN
SDI.SDI_FORMTABLE_MAIN_565_1004   T1 
ON 
T2.MAINID =  T1.ID   
)
,LC2 AS (
select
appeal_num,
COUNT(DISTINCT RESP_CONV) AS RESP_CONV_QTY
from 
LC1 
GROUP BY appeal_num
)

INSERT OVERWRITE TABLE   DWRCMP.DWR_CMP_CUST_COMPLAINT_INFO_F 
select   
T1.requestid  AS requestid  --请求id
,T1.id    AS ID 
,coalesce(T1.KSZS,0)  AS CUST_COMP_QTY  --客诉宗数
,T1.FXQDCW   AS dis_channel  --分销渠道
,T1.KSQD   as  CUST_COMP_channel  --客诉渠道
,T1.KHQWZ  as      CUST_COMP_expectant  --客诉期望值 
,T1.XSZZCW  as  SALES_ORG_FINANCIAL_NODES      --销售组织财务节点
,T1.ZRFMCZB   as  RESP_NAME   --责任方名称主表
,T1.ZRFBMZB  AS  RESP_CODE    --责任方编码主表
,T1.SFXYPGJLSH  AS IS_QC_CHECK   --是否需要品管核查判定
,T1.SSBM  as  DEPARTMENT     --所属部门
,T1.RZFS  as Payment_meth    --入账方式
,T1.PZRQ   as doc_date     --凭证日期
,T1.JFPZ  as Point_voucher   --积分凭证
,T1.DQJDMC as CURR_NODENAME   --当前节点名称
,T1.SFLE  AS Appeal_type      --申诉类型
,T1.LSH    as  appeal_num     --申诉流水号
,T1.KHBM   as   CUST_CODE     --客户编码
,T1.KHMC   as   CUST_NAME   --客户名称
,T1.TCSSSJ  as  SUB_Appeal_time   --提出申诉时间
,T1.LXR    AS    contacts    --联系人
,T1.LXDH    as   Telephone  --联系电话
,T1.SFYSAPDDH  as  is_SAP_ORDER  --是否有SAP订单号
,T1.GKMC     AS   consumer_NAME   --顾客名称
,T1.GKDH    as    consumer_tel   --顾客电话
,T1.GKDZ   as     consumer_add   --顾客地址
,T1.SMFWRY  as   DTD_SRVC_NAME    --上门服务人员
,T1.FWRYDH    as    service_tel   --服务人员电话
,T1.GKGMRQ   as   consumer_purch_date   --顾客购买日期
,T1.CPAZRQ   as   PROD_INSTALL_DATE   --产品安装日期
,regexp_replace(T1.ssnrms, '<[^>]+>', '')   as  appeal_desc    --申诉内容描述
,T1.SFSYZLWTFW     as   is_quality_issue   --是否属于质量问题范围
,T1.SFTH   as  is_returned   --是否退回
,T1.BCLX   as   compensation_type   --补偿类型
,T1.BCFS  as  compensation_method  --补偿方式  
,T1.ZLJB  as   quality_lel    --质量级别
,T1.PCJE  as  INDTY_amount    --赔偿金额
,regexp_replace(T1.khfwzyyj, '<[^>]+>', '')       as  cust_serv_rep_feedback    --客户服务专员意见
,regexp_replace(T1.KHFUZRRYJ, '<[^>]+>', '')      as   cust_serv_responsible_feedback  --客户服务责任人意见
,regexp_replace(T1.TXZJLYJ, '<[^>]+>', '')      as   sys_manager_feedback    --系统总经理意见
,T1.CKSJR    as   warehouse_recipient   --仓库收件人
,T1.KHFWZYCLRY   as  cust_serv_rep_handler  --客户服务专员处理人员 
,T1.KHFWZYCLRYCLSJ1  as cust_serv_rep_time  --客户服务专员处理时间
,T1.PFCPZJE  as total_comp_amount   --赔付产品总金额
,T1.PFCPZMJ  as  total_comp_area   --赔偿产品总面积
,T1.SSRMC  as complainant_name   --申诉人名称
,T1.SSRIDH  as complainant_id  --申诉人ID号
,T1.LCCLRQ  as process_date  --流程处理日期
,T1.LCMC  as  process_name  --流程名称
,T1.SSZLSHJG  as complaint_review_result  --申诉资料审核结果
,T1.CPFL  as prod_category    --产品分类
,T1.GS  as company  --公司
,T1.YXB  as mkt_department  --营销部
,T1.ACT  as current_node  --当前节点
,T1.SFYWCTHRKSX  as  is_return_completed  --是否完成退货入库手续
,T2.APPROVAL_DATE AS APPROVAL_DATE
,CASE when t1.bcfs = '5' then '解析归档'  WHEN T1.SFSYZLWTFW = '0' then '质量问题' when T1.SFSYZLWTFW = '1' then '非质量问题'  end as cc_class
,current_timestamp AS  dw_last_update_date   
,t3.selectname as dis_channel_name
,t4.selectname as Appeal_type_name 
,t5.selectname as is_quality_issue_name
,t6.selectname as is_returned_name 
,t7.selectname as compensation_type_name
,t8.selectname as quality_lel_name
,t9.selectname as Payment_meth_name
,t10.selectname as complaint_review_result_name
,t11.selectname as prod_category_name
,T12.selectname AS CUST_COMP_channel_name
,L2.RESP_CONV_QTY   --责任方数量 
from
SDI.SDI_FORMTABLE_MAIN_565_1004  t1    
left join  (
select  
t1.requestid   
, min(operatedate) as APPROVAL_DATE
from  tmp2 t1 
left join  
SDI.SDI_WORKFLOW_REQUESTLOG_1004 t2 
on t1.requestid = t2.requestid 
and t1.nodeid = t2.nodeid
where  t2.logtype ='0'
group by t1.requestid  
) t2
ON  
t1.requestid = t2.requestid 
--渠道
LEFT JOIN TEX T3 
ON T1.FXQDCW = T3.selectvalue
AND T3.fieldname = 'FXQDCW'
--客诉类型
LEFT JOIN  TEX T4 
ON T1.SFLE = T4.selectvalue
AND T4.fieldname = 'SFLE'
--是否属于质量问题范围
LEFT JOIN TEX T5 
ON  T1.SFSYZLWTFW = T5.selectvalue
AND T5.fieldname = 'SFSYZLWTFW'
--是否退回
left join TEX  T6
ON  T1.SFTH = T6.selectvalue
AND T6.fieldname = 'SFTH'
--补偿类型
left join  TEX  T7
ON  T1.BCLX = T7.selectvalue
and T7.fieldname = 'BCLX'
-- 质量级别
left join TEX T8
ON  T1.ZLJB = T8.selectvalue
AND T8.fieldname = 'ZLJB'
--RZFS 入账方式
left join TEX T9
ON  T1.RZFS = T9.selectvalue
AND T9.fieldname = 'RZFS'
--申诉资料审核结果
left join  tex t10 
ON  T1.SSZLSHJG = t10.selectvalue
AND t10.fieldname = 'SSZLSHJG'
--  CPFL 产品分类
left join tex t11 
ON  T1.CPFL = t11.selectvalue
AND t11.fieldname = 'CPFL'
-- 客诉渠道
left join tex t12 
ON  T1.KSQD = t12.selectvalue
AND t12.fieldname = 'KSQD'
LEFT JOIN  LC2 L2
ON 
T1.LSH  = L2.appeal_num
