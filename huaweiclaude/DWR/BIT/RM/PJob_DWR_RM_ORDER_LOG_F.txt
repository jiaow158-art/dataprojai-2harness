WITH  AA_FLOW_LOG   AS(
SELECT
	ROW_NUMBER() OVER (
		PARTITION BY
			FLOW_ID
		ORDER BY
			g.create_time DESC
	) AS rn,
	g.FLOW_ID,
	g.LOG_TEXT
FROM
	SDI.SDI_AA_FLOW_LOG_1013 g
WHERE
     g.LOG_TEMPLATE_ID = 91  
)
INSERT OVERWRITE TABLE  DWRBIT.DWR_RM_ORDER_LOG_F
SELECT
	b.flow_id,
	g.LOG_TEXT,
   regexp_replace(substr(g.LOG_TEXT, instr(g.LOG_TEXT, '解决方案：') ), '<[^<]+>', '') AS SOLUTION,
   'N' AS del_flag,
    CURRENT_TIMESTAMP AS dw_last_update_date
FROM
	SDI.SDI_AA_FLOW_1013 b
	left join AA_FLOW_LOG  g on g.FLOW_ID=b.FLOW_ID and g.RN=1
