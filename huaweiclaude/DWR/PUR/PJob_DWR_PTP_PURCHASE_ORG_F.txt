	,DPPO.PURCHASE_INFO_RECORD_NO            AS	PURCHASE_INFO_RECORD_NO
	,DPPO.PURCHASE_ORG                       AS	PURCHASE_ORG
	,DPPO.PURCHASE_INFO_RECORD_CODE          AS	PURCHASE_INFO_RECORD_CODE
	,DPPO.FACTORY_ID                         AS	FACTORY_ID
	,DPPO.FACTORY_CODE                       AS	FACTORY_CODE
	,T.FACTORY_NAME                       AS	FACTORY_NAME
	,DPPO.CURRENCY_CODE                      AS	CURRENCY_CODE
	,DPPO.RECORD_CREATE_DATE                 AS	RECORD_CREATE_DATE
	,DPPO.PURCHASE_VCH_ITEM_NUM              AS	PURCHASE_VCH_ITEM_NUM
	,DPPO.CREATE_BY                          AS	CREATE_BY
	,DPPO.PURCHASE_VCH_TYPE                  AS	PURCHASE_VCH_TYPE
	,DPPO.RECENTLY_PO_DATE                   AS	RECENTLY_PO_DATE
	,DPPO.PURCHASE_ORG_INFO_FLAG             AS	PURCHASE_ORG_INFO_FLAG
	,DPPO.REVALUED_RECEIPT_FLAG              AS	REVALUED_RECEIPT_FLAG
	,DPPO.PRICE_VALID_DATE                   AS	PRICE_VALID_DATE
	,DPPO.PRICE_UNIT                         AS	PRICE_UNIT
	,DPPO.PURCHASE_TAX_CODE                  AS	PURCHASE_TAX_CODE

	,DPPO.ASSESSMENT_TYPE                    AS	ASSESSMENT_TYPE
	,DPPO.PURCHASE_VOUCHER_NUM               AS	PURCHASE_VOUCHER_NUM
	,DPPO.PURCHASE_GROUP                     AS	PURCHASE_GROUP
	,DPPO.PLAN_DELIVERY_TIME_DAYS            AS	PLAN_DELIVERY_TIME_DAYS
	,DPPO.PO_PRICE_UNIT                      AS	PO_PRICE_UNIT
	,DPPO.PURCHASE_INFO_VALID_PRICE          AS	PURCHASE_INFO_VALID_PRICE
	,DPPO.RECEIVING_BILL_FLAG                AS	RECEIVING_BILL_FLAG
	,DPPO.OVERDELIVERY_FLAG                  AS	OVERDELIVERY_FLAG

	,DPPO.OVERDELIVERY_LIMIT                 AS	OVERDELIVERY_LIMIT
	,DPPO.SHORTEST_GOODS_SHELVES_QTY         AS	SHORTEST_GOODS_SHELVES_QTY
	,DPPO.DISCOUNT_NO_ITEM_FLAG              AS	DISCOUNT_NO_ITEM_FLAG
	,DPPO.MATERIAL_PO_FLAG                   AS	MATERIAL_PO_FLAG
	,DPPO.TRANS_PKG_COMPONENT_FLAG           AS	TRANS_PKG_COMPONENT_FLAG
	,DPPO.RFQ_ITEM_NO                        AS	RFQ_ITEM_NO
	,DPPO.ORDER_UNIT_DENOMINATOR             AS	ORDER_UNIT_DENOMINATOR
	,DPPO.ORDER_UNIT_NUMERATOR               AS	ORDER_UNIT_NUMERATOR
	,DPPO.QUOTE_VALID_DATE                   AS	QUOTE_VALID_DATE
	,DPPO.MAX_PO_QTY                         AS	MAX_PO_QTY
	,DPPO.MIN_PO_QTY                         AS	MIN_PO_QTY
	,DPPO.STOCKUP_TIME_DAYS                  AS	STOCKUP_TIME_DAYS
    ,DPPO.DEL_FLAG                           AS  DEL_FLAG
    ,CURRENT_TIMESTAMP()                AS  DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_PURCHASE_ORG_T DPPO	--DWI.SAP采购组织数据表
INNER JOIN DWIMD.DWI_MD_FACTORY_T T		--关联工厂主数据取工厂名称
	 ON  T.FACTORY_ID = DPPO.FACTORY_ID	--工厂ID
	 AND T.GROUP_NUM = DPPO.GROUP_NUM	--集团
	 AND T.END_DATE ='4712-12-31'		--结束时间为'4712-12-31'
	 AND DPPO.DEL_FLAG = 'N'			--删除标识为'N'
WHERE DPPO.DEL_FLAG = 'N'
