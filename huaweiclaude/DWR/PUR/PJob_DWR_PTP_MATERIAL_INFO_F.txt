set hive.mapjoin.smalltable.filesize=100000000;

INSERT OVERWRITE TABLE DWRPUR.DWR_PTP_MATERIAL_INFO_F	--DWR.SAP物料信息记录表
SELECT 
	 DPMI.GROUP_NUM						AS	GROUP_NUM
	,DPMI.APPLICATION_PROGRAM_CODE       AS	APPLICATION_PROGRAM_CODE
	,DPMI.CONDITION_TYPE                 AS	CONDITION_TYPE
	,NULL                    AS	CONDITION_TYPE_NAME
	,DPMI.VENDOR_ACCOUNT                 AS	VENDOR_ACCOUNT
	,DPMI.MATERIAL_ID                    AS	MATERIAL_ID
	,DPMI.MATERIAL_NUM                   AS	MATERIAL_NUM
	,T1.MATERIAL_NAME                  AS	MATERIAL_NAME
	,DPMI.PURCHASE_ORG                   AS	PURCHASE_ORG
	,DPMI.FACTORY_ID                     AS	FACTORY_ID
	,DPMI.FACTORY_CODE                   AS	FACTORY_CODE
	,T2.FACTORY_NAME                   AS	FACTORY_NAME
	,DPMI.PURCHASE_INFO_RECORD           AS	PURCHASE_INFO_RECORD
	,DPMI.CONDITION_END_DATE             AS	CONDITION_END_DATE
	,DPMI.CONDITION_START_DATE           AS	CONDITION_START_DATE
	,DPMI.CONDITION_RECORD_NO            AS	CONDITION_RECORD_NO
	,DPMI.DEL_FLAG                       AS	DEL_FLAG
	,CURRENT_TIMESTAMP()            AS	DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_MATERIAL_INFO_T DPMI	--DWI.SAP物料信息记录表
--LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T		--关联通用配置表
--	 ON T.CONFIG_CODE = DPMI.CONDITION_TYPE			--取条件类型中文名称
--	 AND T.GROUP_NUM = DPMI.GROUP_NUM				--集团
--	 AND T.LANG_CODE = '1'
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T1		--关联物料主数据取物料名称
	 ON T1.MATERIAL_ID = DPMI.MATERIAL_ID			--物料ID
	 AND T1.GROUP_NUM = DPMI.GROUP_NUM				--集团
	 AND T1.END_DATE ='4712-12-31'					--结束时间为'4712-12-31'
	 AND DPMI.DEL_FLAG = 'N'						--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T2					--关联工厂主数据工厂名称
	 ON T2.FACTORY_ID = DPMI.FACTORY_ID				--工厂ID
	 AND T2.GROUP_NUM = DPMI.GROUP_NUM				--集团
	 AND T2.END_DATE ='4712-12-31'					--结束时间为'4712-12-31'
	 AND DPMI.DEL_FLAG = 'N'						--删除标识为'N'
WHERE DPMI.DEL_FLAG = 'N'
