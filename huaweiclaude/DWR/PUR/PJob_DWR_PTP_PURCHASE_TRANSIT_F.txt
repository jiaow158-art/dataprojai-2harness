set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置成非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
WITH EKBE AS (
SELECT 
ebeln,
EBELP,
SUM(CASE WHEN bewtp='L' then menge else 0 end) as lmeng, --提货
sum(case when bewtp='U' AND shkzg = 'H' then menge 
                            when bewtp='U' AND shkzg = 'S' then 0-menge 
                            else 0 end) as out_emeng,   -- 发货数量
           sum(case when bewtp='E' AND shkzg = 'S' then menge 
                            when bewtp='E' AND shkzg = 'H' then 0-menge 
                            else 0 end) as in_emeng   -- 收货数量
FROM SDI.sdi_ekbe_1001
GROUP BY 
ebeln,
EBELP
),
eket as (
select  ebeln,ebelp,min(eindt) as min_delv_date,max(eindt) max_delv_date 
from sdi.sdi_eket_1001
group by ebeln,ebelp
)
insert overwrite table dwrpur.dwr_ptp_purchase_transit_f partition(stat_month)
SELECT 
T2.ebeln as purchase_order_num,  -- 采购单号
T2.EBELP as purchase_item_num,  -- 采购行号
T1.bsart as purchase_order_type,  -- 采购单类型
T1.BUKRS as comp_code, -- 公司代码
T1.EKORG as purchase_org_code, -- 采购组织编码
T1.EKGRP as purchase_group_code, -- 采购组
T1.STATU as purchase_order_status, -- 采购组状态
concat(substr(T1.bedat,1,4),'-',substr(t1.bedat,5,2),'-',substr(t1.bedat,7,2)) as purchase_voucher_date,   -- 凭证日期
concat(substr(T1.AEDAT,1,4),'-',substr(T1.AEDAT,5,2),'-',substr(T1.AEDAT,7,2)) as order_create_date,  -- 创建日期
T1.ERNAM,  -- 创建人
-- T1.reswk,  -- 发货工厂
-- T2.Reslo,  -- 发货库存地点
T1.LIFNR as supplier_num, -- 供应商编码
T2.MATNR as material_num, -- 物料编码
T2.lgort as inv_location,   -- 收货库存地点
T2.werks as FACTORY_CODE,   -- 收货工厂
T2.TXZ01 as item_txt,  -- 行短文本
T2.MENGE as purchase_qty,  -- 采购下单数量
T3.in_emeng as stockin_qty, -- 采购入库数量
T2.MENGE - COALESCE (T3.in_emeng,0) as transit_qty, -- 在途数量
T1.unsez  as purchaser, -- 采购员
T2.AFNAM as apply_obj, -- 申请部门/申请人
T1.FRGKE as approved, -- 审批标识
concat(substr(T4.MIN_DELV_DATE,1,4),'-',substr(T4.MIN_DELV_DATE,5,2),'-',substr(T4.MIN_DELV_DATE,7,2))  as MIN_DELV_DATE , -- 最小交货日期
concat(substr( T4.MAX_DELV_DATE ,1,4),'-',substr( T4.MAX_DELV_DATE ,5,2),'-',substr( T4.MAX_DELV_DATE ,7,2))  as MAX_DELV_DATE , -- 最大交货日期
T1.bstyp AS purchase_order_type0,
T5.oktext AS purchase_order_txt,
substr(date_sub(CURRENT_date,1),1,7) as stat_month
FROM  
SDI.SDI_EKKO_1001  T1
INNER JOIN SDI.SDI_EKPO_1001  T2 ON  T1.ebeln = T2.ebeln
LEFT JOIN EKBE T3 ON T2.ebeln = T3.ebeln AND T2.EBELP = T3.EBELP
LEFT JOIN EKET T4 ON T2.ebeln = T4.ebeln AND T2.EBELP = T4.EBELP
LEFT JOIN SDI.SDI_ZOKDATE_1001 T5 on T2.ebeln = T5.VBELN AND T2.EBELP = SUBSTR(T5.POSNR,2,5)
WHERE T1.bsart IN('NB','ZNB1','ZNB2')
AND t1.loekz=''  -- 正常状态
and t2.loekz=''  -- 正常状态
AND T2.RETPO =''  -- 退货
AND T2.ELIKZ=''   -- 已交货
and T2.MENGE - COALESCE (T3.in_emeng,0) > 0
and COALESCE (t1.memory,' ')<>'X';
