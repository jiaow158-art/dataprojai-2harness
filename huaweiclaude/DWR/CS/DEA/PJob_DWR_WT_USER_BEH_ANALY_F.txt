set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;

--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true;

INSERT OVERWRITE TABLE DWRCS.DWR_WT_USER_BEH_ANALY_F PARTITION (STAT_MONTH)
SELECT A.ID
      ,B.DEALER_ID --经销商ID
	  ,E.NAME AS DEALER_NAME --经销商名称
      ,A.STORE_ID --店铺ID
	  ,B.STORE_NAME --门店名称
      ,B.STORE_BYNAME --门店简称
      ,A.BEH_TYPE --类型 0:浏览 1:收藏
      ,A.USER_ID --用户ID
      ,A.USER_SOURCE --客户来源
	  ,F.NAME AS USER_SOURCE_NAME --客户来源名称
      ,A.ITEM_NO --商品唯一码
	  ,G.MODEL AS ITEM_CODE --商品编码
	  ,G.NAME AS ITEM_NAME --商品名称
	  --,C.CATEGORY_ID --规格ID
	  --,D.CATEGORY AS CATEGORY_NAME --规格名称
      ,A.CREATE_TIME --创建时间
      ,A.UPDATE_TIME --更新时间
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,REPLACE(SUBSTR(A.CREATE_TIME,1,7),'-','') AS STAT_MONTH
  FROM DWICS.DWI_WT_USER_BEH_ANALY_T A
  LEFT JOIN DWIMD.DWI_MD_DMT_STORE_INFO_T B
    ON A.STORE_ID = B.STORE_ID
   AND SUBSTR(A.CREATE_TIME,1,10) >= B.START_DATE
   AND SUBSTR(A.CREATE_TIME,1,10) <= B.END_DATE 
/*  LEFT JOIN SDI.SDI_ITEM_CATEGORY_RELATION_1030 C
    ON A.ITEM_NO = C.ITEM_NO
   AND C.IS_DELETED = 0
   AND C.DEL_FLAG = 'N'
  LEFT JOIN SDI.SDI_ITEM_CATEGORY_1030 D
    ON C.CATEGORY_ID = D.ID
   AND D.DEL_FLAG = 'N'
   AND D.APP_ID = 1831*/
  LEFT JOIN SDI.SDI_BUZ_DEPT_INFO_1030 E
    ON B.DEALER_ID = E.DEALER_ID
   AND E.DELETED = 0
   AND E.TYPE = 2
   AND E.EP_ID = 3044
   AND E.APP_ID = 1831
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 F
    ON A.USER_SOURCE = F.ID
   AND F.EP_ID = 3044
   AND F.DELETED = 0
   AND F.DEL_FLAG = 'N'
  LEFT JOIN SDI.SDI_ITEM_1030 G
    ON A.ITEM_NO = G.ITEM_NO
   AND G.DEL_FLAG = 'N'
   AND G.IS_DELETED = 0
   AND G.APP_ID = 1831
   AND G.STATUS = 1
 WHERE SUBSTR(A.CREATE_TIME,1,7) = SUBSTR('${PERIOD_ID_D}',1,7)
;
