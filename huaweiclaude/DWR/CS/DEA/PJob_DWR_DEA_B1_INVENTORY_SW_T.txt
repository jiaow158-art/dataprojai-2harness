(

  		 abbreviation customer_short_name ,

  where  (is_main ='是'

	select 
		 docdate
		,transseq
		,loccode
		,transtype
		,itemcode
		,messageid
		--,customer_short_name 
		,t3.customer_code
		,sum(inqty) inqty
		,sum(outqty) outqty 
		,sum(inqty)-sum(outqty)
		--,t1.customer_code
		,createdate
		,new_flag
	from dwiotd.dwi_wm_b1_oivl_t t1 
	join dm_b1_customer_data t3 
	on case when t1.customer_code = '1000005469' then '1000006985' -- 安徽
		 	when t1.customer_code = '321540' then '1000004847'  -- 苏沪
		 else t1.customer_code end =t3.customer_code 
	and customer_type ='SW'
	where 1=1 
	group by docdate,
			 transseq,
			 loccode,
			 transtype,
			 itemcode,
			 messageid,
			 --customer_short_name ,
			 t3.customer_code,
			 --t1.customer_code,
			 createdate,
			 new_flag
),
ivl1 as (
	select transseq,
		   t2.customer_code,
		  --customer_short_name ,
		  sum(transvalue) transvalue,
		  new_flag		  
	from dwiotd.dwi_wm_b1_ivl1_t  t1 
	join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
	and customer_type ='SW'
	group by transseq,
			 t2.customer_code,
			 new_flag
			--customer_short_name 
),
oitm as (
	select itemcode,
		   itemname,
		   codebars,
		   frgnname,
		   suppcatnum,
		   invntryuom,
		   cardcode,
		   itmsgrpcod,
		   firmcode,
		   slength1,
		   swidth1,
		   --customer_short_name
		   t2.customer_code 
	  from dwiotd.dwi_wm_b1_oitm_t t1 
	  join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
	and customer_type ='SW'
	  group by itemcode,
			   itemname,
			   codebars,
			   frgnname,
			   suppcatnum,
			   invntryuom,
			   cardcode,
			   itmsgrpcod,
			   firmcode,
			   slength1,
			   swidth1,
			   --customer_short_name
			   t2.customer_code 
), 
oitb as (
	select itmsgrpcod,
		  itmsgrpnam,
		  t2.customer_code
		  --customer_short_name  
	from dwiotd.dwi_wm_b1_oitb_t t1 
	join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
	and customer_type ='SW'
	group by itmsgrpcod,
			 itmsgrpnam,
			 --customer_short_name
			 t2.customer_code
),
omrc as (
	select firmcode,
		   firmname,
		   --customer_short_name  
		   t2.customer_code
	from dwiotd.dwi_wm_b1_omrc_t  t1 
	join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
	and customer_type ='SW'
	group by  firmcode,
			  firmname,
			  --customer_short_name 
			  t2.customer_code
), 
oilm as (
	select messageid,
		   effectqty,
		   bpcardcode,
		   --customer_short_name  
		   t2.customer_code
	from dwiotd.dwi_wm_b1_oilm_t t1 
	join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'

	 group by messageid,
			  effectqty,
			  bpcardcode,
			  --customer_short_name

ocrd as (
	select cardcode,
		   cardname,
		   --customer_short_name  
		   t2.customer_code 
	from dwiotd.dwi_wm_b1_ocrd_t t1 
	join dm_b1_customer_data t2 
	on case when t1.customer_code = '1000005469' then '1000006985'
		 	when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
	and customer_type ='SW'
	 group by cardcode,
			  cardname,
			  --customer_short_name 
			  t2.customer_code 
), dt as (
  select day_dt 
	from dm.dm_dim_date_d 
	where  day_dt <= '${PERIOD_ID_D}'  -- 传昨天
	and day_dt >= date_sub('${PERIOD_ID_D}',34)  -- 昨天的前34天
) ,
pdn1 AS (
	 select   
		t2.customer_code
		-- t2.customer_short_name
		,t3.itemcode
		,t3.dscription
		,substr(dt.day_dt,1,10) day_dt
		,t1.docdate
		--,row_number() over(partition by  t2.customer_short_name,t3.itemcode,dt.day_dt order by substr(t1.docdate,1,10)  desc ) rk 
		,row_number() over(partition by  t2.customer_code,t3.itemcode,dt.day_dt order by substr(t1.docdate,1,10)  desc ) rk 
	 from DWIOTD.DWI_WM_b1_OPDN_T t1  
	 join dm_b1_customer_data t2 
	 on  case when t1.customer_code = '1000005469' then '1000006985'
		 	  when t1.customer_code = '321540' then '1000004847'
		 else t1.customer_code end =t2.customer_code 
		 and t1.customer_type ='SW'
	 join DWIOTD.DWI_WM_b1_PDN1_T t3 on t1.customer_code =t3.customer_code and t1.docentry =t3.docentry
	 join  dt on 1=1 
	 where 1=1  
	 and t1.docdate<=dt.day_dt
),
tmp1 as (
		  select 
		       dt.day_dt,
			   case when t0.docdate < dt.day_dt then 'qc' 
					when t0.docdate  >=  dt.day_dt and  t0.docdate< date_sub(dt.day_dt,-1) then 'qj' 
					else 'qm'  end as dttype,       
				t0.itemcode,
				t2.codebars,
				t2.frgnname,
				t2.suppcatnum,
				t2.invntryuom,
				t3.itmsgrpnam,
				t4.firmname,
				t7.cardcode,
				t7.cardname,
				t0.inqty,
				t0.outqty,       
				case when nvl(t1.transvalue,0.0)>0.0 then t1.transvalue 
					 else 0.0 
				end as inlinebalance,       
				case when nvl(t1.transvalue,0.0)<0.0 then -t1.transvalue 
					 else 0.0 
				end as outlinebalance,	       
				case when (t0.transtype=18 or t0.transtype=20) then t0.inqty 
					 else 0.0 
				end as puch_inqty,        
				case when (t0.transtype=14 or t0.transtype=16) then t0.inqty 
					else 0.0
				end as sale_inqty,       
				case when nvl(t1.transvalue,0.0)>=0.0 and t0.transtype = 59 then t0.inqty 
					else 0.0 
				end as other_inqty,        -- t0.transtype<>18 and t0.transtype<>20 and t0.transtype<>14 and t0.transtype<>16
				case when (t0.transtype=19 or t0.transtype=21) then t0.outqty 
					else 0.0 
				end as puch_outqty,        
				case when (t0.transtype=13 or t0.transtype=15) then t0.outqty 
					else 0.0 
				end as sale_outqty,       
				case when nvl(t1.transvalue,0.0)<=0.0 and t0.transtype = 60 then t0.outqty 
					else 0.0 
				end as other_outqty,     -- t0.transtype<>19 and t0.transtype<>21 and t0.transtype<>13 and t0.transtype<>15
				t1.transvalue as linebalance,
				t0.loccode as loccode ,
				--t0.customer_short_name      
				t0.customer_code
		from oivl t0  
		inner join ivl1 t1 on t0.transseq = t1.transseq  and t0.customer_code = t1.customer_code and t0.new_flag = t1.new_flag
		inner join oitm t2  on t0.itemcode = t2.itemcode  and t0.customer_code = t2.customer_code      
		inner join oitb t3  on t2.itmsgrpcod=t3.itmsgrpcod  and t2.customer_code = t3.customer_code
		inner join omrc t4  on t2.firmcode=t4.firmcode    and t2.customer_code = t4.customer_code
		left join oilm t5 on nvl(t0.messageid,'') = nvl(t5.messageid,'') and nvl(t5.effectqty,0.0)<>0.0  and t0.customer_code = t5.customer_code
		left join ocrd t6 on nvl(t5.bpcardcode,'') = nvl(t6.cardcode,'')    and t5.customer_code = t6.customer_code   
		left join ocrd t7 on nvl(t2.cardcode,'')=nvl(t7.cardcode,'')        and t2.customer_code = t7.customer_code 
	    join dt on 1=1 
),
tmp as (
    select 
     t.day_dt
    ,t.dttype
    ,t.itemcode
    ,t.codebars
    ,t.frgnname
    ,t.suppcatnum
    ,t.invntryuom
    ,t.itmsgrpnam
    ,t.firmname
    ,t.loccode
    ,t.cardcode
    ,t.cardname
    --, t.customer_short_name
	,t.customer_code
    ,sum(case when t.dttype='qc' then nvl(t.inqty,0.0) else 0.0 end) as qcinqty
    ,sum(case when t.dttype='qj' then nvl(t.inqty,0.0) else 0.0 end) as qjinqty
    ,sum(case when t.dttype='qm' then nvl(t.inqty,0.0) else 0.0 end) as qminqty
    ,sum(case when t.dttype='qc' then nvl(t.outqty,0.0) else 0.0 end) as qcoutqty
    ,sum(case when t.dttype='qj' then nvl(t.outqty,0.0) else 0.0 end) as qjoutqty
    ,sum(case when t.dttype='qm' then nvl(t.outqty,0.0) else 0.0 end) as qmoutqty
    ,sum(case when t.dttype='qj' then nvl(t.inlinebalance,0.0) else 0.0 end )as qjinlinebalance
    ,sum(case when t.dttype='qj' then nvl(t.outlinebalance,0.0) else 0.0 end) as qjoutlinebalance
    ,sum(case when t.dttype='qc' then nvl(t.linebalance,0.0) else 0.0 end) as qclinebalance
    ,sum(case when t.dttype='qj' then nvl(t.linebalance,0.0) else 0.0 end) as qjlinebalance
    ,sum(case when t.dttype='qm' then nvl(t.linebalance,0.0) else 0.0 end) as qmlinebalance
    ,sum(case when t.dttype='qj' then nvl(t.puch_inqty,0.0) else 0.0 end) as qjpuch_inqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_inqty,0.0) else 0.0 end) as qjsale_inqty
    ,sum(case when t.dttype='qj' then nvl(t.other_inqty,0.0) else 0.0 end) as qjother_inqty
    ,sum(case when t.dttype='qj' then nvl(t.puch_outqty,0.0) else 0.0 end)as qjpuch_outqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_outqty,0.0) else 0.0 end) as qjsale_outqty
    ,sum(case when t.dttype='qj' then nvl(t.other_outqty,0.0) else 0.0 end) as qjother_outqty
      from tmp1 t  
	   group by t.itemcode,
				t.codebars,
				t.frgnname,
				t.suppcatnum,
				t.invntryuom,
				t.itmsgrpnam,
				t.firmname,
				t.loccode,
				t.cardcode,
				t.cardname,
				--t.customer_short_name,
				t.customer_code,
				t.dttype,
				t.day_dt
),
DM_WM_B1_INVENTORY_T_D_DATA AS(
select  
	 tt.day_dt as  day_dt -- "日期"
	--,tt.customer_short_name 
	,tt.customer_code
	,tt.itemcode   as b1_material_num --  "物料编码"
	--,t4.itemname   as b1_material_name --  "物料描述"
	,t9.material_name as b1_material_name --  "物料描述"
	,t5.material_type -- 分类
	,t5.product_class -- 品类
	,t5.analyze_product_class -- 分析品类
	,sum(nvl(tt.qcinqty,0.0))-sum(nvl(tt.qcoutqty,0.0)) as qc_qty -- "期初数量"
	,sum(nvl(tt.qjinqty,0.0)) as qj_instock_qty -- "本期入库"
	,sum(nvl(tt.qjpuch_inqty,0.0)) as qj_pp_del_qty -- "期间采购收货数量"
	,sum(nvl(tt.qjsale_inqty,0.0)) as qj_sale_return_qty -- "期间销售退货数量"
	,sum(nvl(tt.qjother_inqty,0.0)) as qj_other_instock_qty -- "期间其他入库数量"
	,sum(nvl(tt.qjpuch_outqty,0.0)) as qj_pp_return_qty -- "期间采购退货数量"
	,sum(nvl(tt.qjoutqty,0.0)) as qj_outstock_qty -- "本期出库"
	,sum(nvl(tt.qjsale_outqty,0.0)) as qj_sale_outstock_qty -- "期间销售出库数量"
	,sum(nvl(tt.qjother_outqty,0.0)) as qj_other_outstock_qty -- "期间其他出库数量"

	,sum(nvl(tt.qjoutlinebalance,0.0)) as bq_outstock_amt -- "本期出库金额"
	,sum(nvl(tt.qclinebalance,0.0))+sum(nvl(tt.qjlinebalance,0.0)) as qm_amt -- "期末金额"
	,sum(nvl(tt.qjlinebalance,0.0)) as bq_happen_amt -- "本期发生金额"
	,t8.docdate as last_instock_date -- "最末入库日期"
	,t9.material_id as sap_material_id --"物料ID"
    ,t9.material_name as sap_material_name --  "SAP物料名称"
    ,t9.matl_type_code as sap_matl_type_code -- "SAP物料类型"
    ,t9.matl_type_name as sap_matl_type_name -- "SAP物料类型名称"
    ,t9.product_group_code as sap_product_group_code -- "SAP产品组"
    ,t9.product_group_name as sap_product_group_name -- "SAP产品组名称"
    ,t9.product_classify_code as sap_product_classify_code --   "SAP产品分级"
    ,t9.product_classify_name  as sap_product_classify_name --  "SAP产品分级名称"

    ,t9.product_level4_code as sap_product_level4_code --      "SAP层次4"
    ,t9.product_level4_name as sap_product_level4_name --     "SAP层次4名称"
	,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	,'' as volume -- "体积"
	, tt.loccode as warehouse_code
from  tmp  tt 
    join oitm t4  on tt.itemcode=t4.itemcode AND tt.customer_code  =t4.customer_code 
    left join  SDI.SDI_VM_B1_INV_CATE_MATCH_1037  t5 on tt.itemcode = t5.material_num
   -- left join  dm.dm_rpt_bi0_pmaterial_t t6 on tt.itemcode = t6."MATERIAL"
