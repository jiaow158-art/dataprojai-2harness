INSERT OVERWRITE TABLE DWRCS.DWR_CS_PROTECT_STAT_F
SELECT 
       SUBSTR(A.STATIS_DATE,1,10) AS STAT_DATE --统计日期
      ,A.EP_ID --企业ID
      ,A.REFERENCE_ID AS REF_ID --关联ID
      ,A.SOURCE	AS CUST_SRC_CODE --客户来源
	  ,B.NAME AS CUST_SRC_NAME --客户来源名称
	  ,COALESCE(C.NAME,B.NAME) AS CUST_SRC_CAT_NAME --客户来源类别名称
      ,A.PROTECT_COUNT AS PROTECT_CNT --报备数
	  ,D.LVL1_DEPT_ID --一级部门ID
	  ,D.LVL1_DEPT_SYC_ID --一级部门原ID
	  ,D.LVL1_DEPT_NAME --一级部门名称
	  ,D.LVL2_DEPT_ID --二级部门ID
	  ,D.LVL2_DEPT_SYC_ID --二级部门原ID
	  ,D.LVL2_DEPT_NAME --二级部门名称
	  ,D.LVL3_DEPT_ID --三级部门ID
	  ,D.LVL3_DEPT_SYC_ID --三级部门原ID
	  ,D.LVL3_DEPT_NAME --三级部门名称
	  ,D.LVL4_DEPT_ID --四级部门ID
	  ,D.LVL4_DEPT_SYC_ID --四级部门原ID
	  ,D.LVL4_DEPT_NAME --四级部门名称
	  ,D.LVL5_DEPT_ID --五级部门ID
	  ,D.LVL5_DEPT_SYC_ID --五级部门原ID
	  ,D.LVL5_DEPT_NAME --五级部门名称
	  ,D.LVL6_DEPT_ID --六级部门ID
	  ,D.LVL6_DEPT_SYC_ID --六级部门原ID
	  ,D.LVL6_DEPT_NAME --六级部门名称
	  ,D.LVL7_DEPT_ID --七级部门ID
	  ,D.LVL7_DEPT_SYC_ID --七级部门原ID
	  ,D.LVL7_DEPT_NAME --七级部门名称
	  ,D.LVL8_DEPT_ID --八级部门ID
	  ,D.LVL8_DEPT_SYC_ID --八级部门原ID
	  ,D.LVL8_DEPT_NAME --八级部门名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_CLIENT_PROTECT_STATIS_1030 A
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 B
    ON A.SOURCE = B.ID
   AND A.EP_ID = B.EP_ID	
   AND B.DEL_FLAG = 'N'
   AND B.DELETED = 0   
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 C
    ON B.PARENT_ID = C.ID
   AND A.EP_ID = C.EP_ID	
   AND C.DEL_FLAG = 'N'
   AND C.DELETED = 0
  LEFT JOIN DWRDIM.DWR_DIM_DMT_DEPT_INFO_D D
    ON A.EP_ID = D.EP_ID
   AND A.REFERENCE_ID = D.DEPT_ID 
 WHERE A.EP_ID IN(3044,3045)
