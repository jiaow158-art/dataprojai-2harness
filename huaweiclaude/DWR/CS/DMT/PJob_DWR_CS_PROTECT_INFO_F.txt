INSERT OVERWRITE TABLE DWRCS.DWR_CS_PROTECT_INFO_F
SELECT
      A.ID --主键ID
     ,A.EMPLOYEE_ID AS EMP_ID --员工ID
	 ,A.EMPLOYEE_NAME AS EMP_NAME --员工名称
     ,A.SEARCH_WORD	--搜索内容
     ,A.SEARCH_TIME	--搜索时间
     ,A.SEARCH_NO --搜索编号
     ,A.RESULT_NO --结果编号
     ,NULL AS CUST_ID --结果客户ID
	 ,NULL AS CUST_NAME --客户名称
	 ,NULL AS CUST_NICK_NAME --客户昵称
     ,A.HISTORY_PHONE AS HIS_PHONE_NUM --历史电话
     ,A.HISTORY_NAME AS HIS_CUST_NAME --历史客户名
     ,A.HISTORY_PROVINCE AS HIS_PROVINCE_NAME --历史省
     ,A.HISTORY_CITY AS HIS_CITY_NAME --历史市
     ,A.HISTORY_DISTRICT AS HIS_DISTRICT_NAME --历史区
     ,A.HISTORY_ADDRESS	AS HIS_ADDR --历史地址
     ,A.HISTORY_PROTECT_EMPLOYEE_ID AS HIS_PROTECT_EMP_ID --历史报备人ID
     ,A.HISTORY_PROTECT_EMPLOYEE_NAME AS HIS_PROTECT_EMP_NAME --历史报备人
     ,A.HISTORY_PROTECT_STORE_ID AS HIS_PROTECT_STORE_ID --历史报备门店ID
     ,A.HISTORY_PROTECT_STORE_NAME AS HIS_PROTECT_STORE_NAME --历史报备门店
     ,A.HISTORY_PROTECT_TIME AS HIS_PROTECT_TIME --历史报备时间
     ,A.HISTORY_PROTECT_END_TIME AS HIS_PROTECT_END_TIME --历史报备结束时间
     ,A.CREATE_TIME	--创建时间
     ,A.CREATER_ID --创建人ID
     ,A.CREATER_NAME --创建人名称
     ,A.MODIFY_TIME	--更新时间
     ,A.MODIFIER_ID --更新人ID
     ,A.MODIFIER_NAME	--更新人名称
	 ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_CLIENT_SEARCH_PROTECTING_LOG_1012 A
 WHERE CAST(DELETE_FLAG AS STRING) = 'FALSE'
