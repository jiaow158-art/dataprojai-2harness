set hive.cbo.enable=true;

INSERT OVERWRITE TABLE DWRFIN.DWR_FIN_EXPENSE_DETAIL_F   PARTITION(PERIOD_ID_M)
SELECT
  b.corporation_code AS CO_CODE,--公司代码
  b.corporation_name AS CO_NAME,--公司名称
  c.erp_voucher_no AS  erp_voucher_no,--会计凭证号
  a.erp_item_number AS erp_item_number,--ERP行项目号
  a.bl_sign AS bl_sign,--借贷方标识 
  a.ledger_account_code AS  account_code,  -- 总账科目代码,
  a.ledger_account_name AS  account_name,--总账科目名称,
  a.amount AS acc_amount,--科目金额,
  b.voucher_date AS voucher_date,--凭证日期,
  b.posting_date AS posting_date,--`过账日期`,
  d.NAME AS bus_name,--`业务来源类型名称`,
  e.order_no AS SHARED_ORDER_NUM, --`共享服务单号`,
  e.source_order_no AS source_order_no,--`共享服务单号对应的业务来源单号`,
  COALESCE( j.NAME, r.NAME ) AS EXPENSE_CATEGORY,--`费用大类`,
  COALESCE( h.item_number, p.item_number ) AS EXPENSE_SUBCLASS_LINE,--`小类行项目号`,
  COALESCE( i.NAME, q.NAME ) AS EXPENSE_SUBCLASS,--`费用小类`,
  COALESCE(
    o.amount,
    COALESCE( h.amount - h.travel_amount, 0 ) - COALESCE( apportionment_total.sum_apportionment, 0 )
  ) AS EXPENSE_SUBCLASS_AMOUNT,--`费用小类金额（含税不含商旅代垫）`,
  CASE
    WHEN COALESCE( subtotal.expense_detail_type_id, subtotal2.expense_detail_type_id )= "0F587B92-28D7-11ED-8DCC-0242AC120002" THEN
      "" ELSE COALESCE( l.NAME, t.NAME )
  END AS COST_BREAKDOWN, -- `费用细类`,
ROUND(
  CASE 
    WHEN COALESCE(o.amount, 0) = 0 THEN 
      (COALESCE(subtotal.amount_subtotal, 0) - COALESCE(subtotal.deduction_tax_subtotal, 0)) * 
      (1 - COALESCE(apportionment_total.sum_apportionment, 0) / 
      CASE 
        WHEN COALESCE(h.amount - h.travel_amount, 0) = 0 THEN 1 
        ELSE h.amount - h.travel_amount 
      END)
    ELSE 
      (COALESCE(subtotal2.amount_subtotal2, 0) - COALESCE(subtotal2.deduction_tax_subtotal2, 0)) * 
      (o.amount / 
      CASE 
        WHEN COALESCE(p.amount - p.travel_amount, 0) = 0 THEN 1 
        ELSE p.amount - p.travel_amount 
      END)
  END,
  2
) AS COST_BREAKDOWN_ENTRY,--`费用细类入账金额`
CASE 
  WHEN h.creator_account IS NOT NULL THEN h.creator_account  
  ELSE p.creator_account
END AS CREATER,--`制单人工号`,

CASE 
  WHEN h.creator_name IS NOT NULL THEN h.creator_name 
  ELSE p.creator_name 
END AS CREATNAME,--`制单人姓名`,

CASE 
  WHEN h.personnel_code IS NOT NULL THEN h.personnel_code
  ELSE p.personnel_code
END AS REIMBURSER_NUM, --`实际报销人工号`,

CASE 
  WHEN h.personnel_name IS NOT NULL THEN h.personnel_name
  ELSE p.personnel_name
END AS REIMBURSER_NAME,--`实际报销人姓名`
  g.bear_organization_code AS bear_organization_code,--`承担部门代码`
  g.bear_organization_name AS bear_organization_name,--`承担部门名称`
  COALESCE(h.supplier_code, p.supplier_code) AS supplier_code,--`供应商代码`
  COALESCE(h.supplier_name, p.supplier_name) AS supplier_name,--`供应商名称`
  COALESCE(get_json_object(h.budget_info, '$[0].poolCode'), get_json_object(p.budget_info, '$[0].poolCode')) AS budget_pool_act,--`实际占用预算池`
  m.contract_no AS contract_no,--`合同编号`
  m.contract_name AS contract_name,--`合同名称`
  n.oa_serial_number AS oa_serial_number,--`OA合同流水号`
  current_timestamp() as dw_last_update_date, -- 最后更新时间
  substring(a.created_date,1,7) as period_id_m
FROM
SDI.SDI_VOUCHER_ITEM_1039 a
	LEFT JOIN SDI.SDI_VOUCHER_HEAD_1039 b ON a.head_id = b.id 
	LEFT JOIN SDI.SDI_VOUCHER_STATE_1039 c ON c.head_id = b.id 
	LEFT JOIN SDI.SDI_ACCOUNTING_SOURCE_TYPE_1039 d ON b.accounting_source_type_id = d.id 
	LEFT JOIN SDI.SDI_SHARE_ORDER_HEAD_1039 e ON b.share_order_id = e.id 
	LEFT JOIN SDI.SDI_SHARE_ORDER_ITEM_1039 f ON f.share_order_head_id = e.id 
	LEFT JOIN SDI.SDI_REIMBURSEMENT_HEAD_1039 g ON g.order_no = e.source_order_no 
	LEFT JOIN SDI.SDI_REIMBURSEMENT_ACTIVITY_1039 h ON h.id = a.business_source_item_id 
	LEFT JOIN SDI.SDI_REIM_ACTIVITY_APPORTIONMENT_1039 u ON h.id = u.reimbursement_activity_id 
	LEFT JOIN (
		SELECT reimbursement_activity_id, SUM( amount ) AS sum_apportionment 
		FROM SDI.SDI_REIM_ACTIVITY_APPORTIONMENT_1039 
		GROUP BY reimbursement_activity_id
	) AS apportionment_total ON h.id = apportionment_total.reimbursement_activity_id
	LEFT JOIN SDI.SDI_REIM_ACTIVITY_APPORTIONMENT_1039 o ON o.id = a.business_source_item_id 
	LEFT JOIN SDI.SDI_REIMBURSEMENT_ACTIVITY_1039 p ON o.reimbursement_activity_id = p.id
	LEFT JOIN SDI.SDI_EXPENSE_TYPE_1039 i ON i.id = h.expense_type_id 
	LEFT JOIN SDI.SDI_EXPENSE_TYPE_1039 q ON q.id = p.expense_type_id 
	LEFT JOIN SDI.SDI_EXPENSE_TYPE_CATEGORY_1039 j ON j.id = i.expense_type_category_id 
	LEFT JOIN SDI.SDI_EXPENSE_TYPE_CATEGORY_1039 r ON r.id = q.expense_type_category_id 
	LEFT JOIN SDI.SDI_ACTIVITY_CONTRACT_INFO_1039 m ON m.source_id = h.id
	LEFT JOIN SDI.SDI_CSM_CONTRACT_HEAD_1039 n ON m.contract_id = n.id 
	LEFT JOIN (
		SELECT
			reimbursement_activity_id,
			expense_detail_type_id,
			SUM(amount) AS amount_subtotal,
			SUM(deduction_tax) AS deduction_tax_subtotal
		FROM
			SDI.SDI_REIM_ACTIVITY_EXPENSE_DETAIL_1039 
		WHERE
			travel_detail_type IS NULL 
		GROUP BY
			reimbursement_activity_id,
			expense_detail_type_id 
	) AS subtotal ON h.id = subtotal.reimbursement_activity_id 
	LEFT JOIN (
		SELECT
			reimbursement_activity_id,
			expense_detail_type_id,
			SUM(amount) AS amount_subtotal2,
			SUM(deduction_tax) AS deduction_tax_subtotal2
		FROM
			SDI.SDI_REIM_ACTIVITY_EXPENSE_DETAIL_1039 
		WHERE
			travel_detail_type IS NULL 
		GROUP BY
			reimbursement_activity_id,
			expense_detail_type_id 
	) AS subtotal2 ON p.id = subtotal2.reimbursement_activity_id 
	LEFT JOIN SDI.SDI_EXPENSE_DETAIL_TYPE_1039 l ON l.id = subtotal.expense_detail_type_id 
	LEFT JOIN SDI.SDI_EXPENSE_DETAIL_TYPE_1039 t ON t.id = subtotal2.expense_detail_type_id
WHERE d.CODE = 'ERMS-ReimSimulator'
and  SUBSTRING(A.created_date,1,7) = '${PERIOD_ID_M}'
