,asset_limits_tmp as (

            select
     t1.company_code
   , t1.asset_code
   , t1.asset_sub_code
   , t1.asset_cat
   , substr(t1.asset_inactive_date,1,7)         asset_inactive_date
   , substr(t1.depreciation_start_date,1,7)     depreciation_start_date
   ,sum(sum(net_fixed_asset_value)) over (partition by t1.company_code,t1.asset_code,t1.asset_sub_code order by t1.period_id_m)
   ,case when sum(sum(net_fixed_asset_value)) over (partition by t1.company_code,t1.asset_code,t1.asset_sub_code order by t1.period_id_m) = 0
       then  coalesce(case when substr(asset_inactive_date,1,7) = '9999-12' then null else substr(asset_inactive_date,1,7) end ,t1.period_id_m) else coalesce(substr(t1.depreciation_start_date,1,7),'1900-01')   end period_id_m
    from   asset_data t1
    where asset_original_value_all > 0
    and flag_2020 = 1
    group by
     t1.company_code
   , t1.asset_code
   , t1.asset_sub_code
    ,t1.period_id_m
   , coalesce(case when substr(asset_inactive_date,1,7) = '9999-12' then null else substr(asset_inactive_date,1,7) end ,t1.period_id_m)
   , coalesce(substr(t1.depreciation_start_date,1,7),'1900-01')
   , substr(t1.asset_inactive_date,1,7)
   , substr(t1.depreciation_start_date,1,7)
   , t1.asset_cat
)

,asset_limits_0 as (

       else '9999-12' end   limits_end
   ,count( t1.period_id_m )   limits_cnt
   ,case when asset_cat = '固定资产' then cast((rand()*100000000) as int ) else 1 end  random

select
    t1.company_code
   , t1.asset_code
   , t1.asset_sub_code
   , t1.asset_cat
   , t1.asset_inactive_date
   , t1.depreciation_start_date
   , t1.period_id_m

        t1.company_code

   , t1.asset_sub_code
   , t1.asset_cat
   , t1.asset_inactive_date
   , t1.depreciation_start_date
   , t1.period_id_m

group by
     company_code
   , asset_code
   , asset_sub_code
   ,asset_inactive_date
   ,depreciation_start_date
   ,t1.period_id_m
   ,asset_cat
)
