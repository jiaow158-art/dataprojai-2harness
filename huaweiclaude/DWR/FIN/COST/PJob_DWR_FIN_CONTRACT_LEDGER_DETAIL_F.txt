LEFT JOIN SDI.SDI_BAF_CONTRACT_CLASSIFY_1039 g ON g.id = a.contract_classify_id
LEFT JOIN SDI.SDI_BAF_CONTRACT_TYPE_1039 h ON h.id = a.contract_type_id
LEFT JOIN SDI.SDI_BAF_PAYMENT_NATURE_1039 i ON i.id = b.payment_nature_id
