而且：客户编码（customer）存在共享仓客户范围内（表：SDI.SDI_CT_SHARE_CUSTOMER_REL_1037，字段：共享仓客户编码（dealer_code））

select
	d.cust_code														-- 客户编码
	,d.material_code												-- 物料编码
	,max(case when calyear = substr('${PERIOD_ID_D}',1,4) then  d.revenue_unit_price_amt end) as revenue_unit_price_amt_0								-- 今年收入单价
	,max(case when  calyear = substr('${PERIOD_ID_D}',1,4) then coalesce(d.actual_cost_unit_price_amt_A,d.actual_cost_unit_price_amt_B,d.actual_cost_unit_price_amt_C) end) as actual_cost_unit_price_amt_0						-- 今年实际成本单价 优先取模式A再取模式B最后取模式C
	,max(case when calyear =substr('${PERIOD_ID_D}',1,4)- 1 then  d.revenue_unit_price_amt end) as revenue_unit_price_amt_1									-- 去年收入单价
	,max(case when  calyear = substr('${PERIOD_ID_D}',1,4)- 1 then coalesce(d.actual_cost_unit_price_amt_A,d.actual_cost_unit_price_amt_B,d.actual_cost_unit_price_amt_C) end)	as actual_cost_unit_price_amt_1							-- 去年实际成本单价
from
(
	select
		t.customer				as cust_code						-- 客户编码
		,t.material				as material_code					-- 物料编码
		,substr(t.period_id_m,1,4)  as calyear
		,case when sum(t.zxssl) = 0 then null -- 数量为0置空
			  else sum(t.notax_sales_net_amt)/sum(t.zxssl) end as revenue_unit_price_amt			-- 收入单价
		,case when sum(case when t.trading_mode = 'A'  then t.zxssl end )=0 -- 数量为0置空

		,case when sum(case when t.trading_mode = 'B'  then t.zxssl end )=0  -- 数量为0置空
		      then null else sum(case when t.trading_mode = 'B'  then t.notax_sales_net_amt - t.profit_1n end )/sum(case when t.trading_mode = 'B'  then t.zxssl end ) end  as actual_cost_unit_price_amt_B
		,case when sum(case when t.trading_mode = 'C'  then t.zxssl end)=0 then null  -- 数量为0置空
		      else sum(case when t.trading_mode = 'C'  then t.notax_sales_net_amt - t.cross_profit end )/sum(case when t.trading_mode = 'C'  then t.zxssl end ) end  as actual_cost_unit_price_amt_C
	from
		dm.dm_ambv2_profit_t t
	where t.data_source = 'S'
	and t.prod_property is not null and t.prod_property <> ''
	and t.customer in (select dealer_code from SDI.SDI_CT_SHARE_CUSTOMER_REL_1037 group by dealer_code)
	and (substr(t.period_id_m,1,7) between substr('${PERIOD_ID_D}',1,4)||'-01' and substr('${PERIOD_ID_D}',1,7)
	or substr(t.period_id_m,1,4) = substr('${PERIOD_ID_D}',1,4) - 1)
	and (t.bus_area not in('G000','G001')  or t.bus_area is null or bus_area = '')
	and (t.distr_chan <> '06' or t.distr_chan is null or t.distr_chan = '' )
	and t.trading_mode is not null and t.trading_mode <> ''
	group by t.customer,t.material, substr(t.period_id_m,1,4) 
) d
group by d.cust_code								
	,d.material_code

/*
