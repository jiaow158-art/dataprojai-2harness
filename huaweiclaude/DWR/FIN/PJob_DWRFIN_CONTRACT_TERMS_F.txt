-- CREATE TABLE DWRFIN.DWRFIN_CONTRACT_TERMS_F (
-- customer_code VARCHAR(356) COMMENT '客户编码',
-- wbs_number VARCHAR(356) COMMENT 'WBS号',
-- company_code VARCHAR(356) COMMENT '公司代码',
-- contract_id VARCHAR(356) COMMENT '合同ID',
-- contract_amount DECIMAL(38,10) COMMENT '合同金额',
-- advance_payment_amount DECIMAL(38,10) COMMENT '预付款金额',
-- advance_payment_ratio DECIMAL(38,10) COMMENT '预付款比例',
-- progress_payment_ratio DECIMAL(38,10) COMMENT '进度款比例',
-- settlement_payment_ratio DECIMAL(38,10) COMMENT '结算款比例',
-- retention_money_ratio DECIMAL(38,10) COMMENT '质保金比例',
-- contract_source VARCHAR(356) COMMENT '合同来源',
-- dw_last_update_date  TIMESTAMP
-- )
with zsd_c_ps as (
select row_number()over(PARTITION  by PARTNER,credit_sgmnt,PS_PSP_PNR order by ZCONT_SEQ desc) as rn  ,
 PARTNER,credit_sgmnt,PS_PSP_PNR,ZPROZ
from  sdi.sdi_ZSD_CONTRACT_PS_1001
)
, zsd_c_ps_lzh as (
SELECT  
PARTNER,credit_sgmnt,PS_PSP_PNR,
SUM(CASE WHEN RN = 1 THEN zproz  else  0 end ) as zproz_1,
SUM(CASE WHEN RN = 2 THEN zproz  else  0 end ) as zproz_2,
SUM(CASE WHEN RN = 3 THEN zproz  else  0 end ) as zproz_3,
SUM(CASE WHEN RN = 4 THEN zproz  else  0 end ) as zproz_4
from zsd_c_ps
group by  
PARTNER,credit_sgmnt,PS_PSP_PNR
)
,tmp as (
select 
t1.PARTNER,
t1.PS_PSP_PNR,
t1.CREDIT_SGMNT,
'' as  contract_id,
t1.ZZHTE,
t1.ZZYFBL,
count(t2.ZPROZ) as zproz_cnt
from  sdi.SDI_ZTSDXD01_1001 t1 
left join 
sdi.sdi_ZSD_CONTRACT_PS_1001  t2 
on t1.PARTNER = t2.PARTNER
and t1.CREDIT_SGMNT = t2.credit_sgmnt
and t1.PS_PSP_PNR = t2.PS_PSP_PNR
group by  
t1.PARTNER,
t1.PS_PSP_PNR,
t1.CREDIT_SGMNT,
t1.ZZHTE,
t1.ZZYFBL
)
INSERT OVERWRITE TABLE  DWRFIN.DWRFIN_CONTRACT_TERMS_F
select  
A.customer_code
,A.WBS
,concat(substr(A.engineering_sale_org,1,2),'00') as  engineering_sale_org
,A.contract_id
,B.contract_amount2
,0 AS YFKJE
,B.payment_scale2
,B.progress_scale2
,B.settlement_scale2
,B.retention_scale2
,'LTC' AS HTLY
,current_timestamp  
from  
SDI.SDI_CONTRACT_1033 A 
LEFT JOIN 
SDI.SDI_CONTRACT_SETTLEMENT_HEAD_1033 B 
ON A.contract_id = B.contract_id
WHERE new_flag = 'NEW'
and A.delete_flag = 0
union all 
select  
t1.PARTNER,
t4.posid,
t1.CREDIT_SGMNT,
'' as contract_id,
t1.ZZHTE,
t1.ZZYFBL,
case when  t1.zproz_cnt = 1 then 0 else ZZYFBL/ZZHTE*100  end as bili1,
case when  t1.zproz_cnt <= 2  then 0 else  t2.ZPROZ -  (t1.ZZYFBL/t1.ZZHTE)*100 end as bili2 ,
case when  t1.zproz_cnt = 1 then t3.ZPROZ_1  WHEN T1.zproz_cnt = 2  THEN T3.ZPROZ_2  - (t1.ZZYFBL/t1.ZZHTE)*100  else  T3.ZPROZ_2  end as bili3 ,
case when  t1.zproz_cnt = 1 then 0 else  t3.ZPROZ_1 end  as  bili4,
'sap' as HTLY,
current_timestamp  
from 
tmp t1 left join 
(select    
 PARTNER,credit_sgmnt,PS_PSP_PNR,
sum(ZPROZ) as ZPROZ
from 
zsd_c_ps 
where rn >= 3 
group by  PARTNER,credit_sgmnt,PS_PSP_PNR
) t2  on 
t1.PARTNER = t2.PARTNER
and t1.PS_PSP_PNR = t2.PS_PSP_PNR
and t1.CREDIT_SGMNT = t2.CREDIT_SGMNT
left join 
zsd_c_ps_lzh t3
on t1.PARTNER = t3.PARTNER
and t1.CREDIT_SGMNT = t3.credit_sgmnt
and t1.PS_PSP_PNR = t3.PS_PSP_PNR
LEFT JOIN SDI.SDI_PRPS_1001  T4
ON T1.PS_PSP_PNR = T4.pspnr
