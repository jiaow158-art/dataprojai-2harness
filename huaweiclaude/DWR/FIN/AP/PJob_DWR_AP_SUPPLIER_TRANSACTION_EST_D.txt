INSERT OVERWRITE TABLE DWRFIN.DWR_AP_SUPPLIER_TRANSACTION_EST_D
SELECT 
    T.FISAL_YEAR                        AS FISAL_YEAR                       -- 财年
    ,T.COMP_CODE                        AS COMP_CODE                        -- 公司代码
    ,T.GENERAL_LEDGER_ACCOUNT           AS GENERAL_LEDGER_ACCOUNT           -- 总账科目
    ,T5.TXT20                           AS GENERAL_LEDGER_ACCOUNT_DESC      -- 总账科目描述
    ,T.SPECIAL_GENERAL_LEDGER           AS SPECIAL_GENERAL_LEDGER           -- 特别总账
    ,T3.BPKIND                          AS SUPPLIER_ATTRIBUTE_CODE          -- 供应商属性代码
    ,T4.TEXT40                          AS SUPPLIER_ATTRIBUTE_DESC          -- 供应商属性描述
    ,T.SUPPLIER_CODE                    AS SUPPLIER_CODE                    -- 供应商代码
    ,CASE WHEN T3.NAME_ORG1 <> '' THEN T3.NAME_ORG1 ELSE T3.BU_SORT1 END
                                        AS SUPPLIER_NAME                    -- 供应商全称
    ,T1.ZTERM                           AS PAY_TERM_CODE                    -- 付款条款
    ,T2.TEXT1                           AS PAY_TERM_DESC                    -- 付款条款描述
    ,T.VOUCHER_TYPE                     AS VOUCHER_TYPE                     -- 凭证类型
    ,T.VOUCHER_CODE                     AS VOUCHER_CODE                     -- 凭证编码
    ,T.VOUCHER_LINE_CODE                AS VOUCHER_LINE_CODE                -- 凭证行编码
    ,T.VOUCHER_DATE                     AS VOUCHER_DATE                     -- 凭证日期
    ,T.POTING_DATE                      AS POTING_DATE                      -- 过账日期
    ,T.POTING_DATE                      AS DUE_DATE                         -- 到期日
    ,T.CLEARED_DATE                     AS CLEARED_DATE                     -- 清账日期
    ,T.ORIGIN_CU_END_BALANCE_AMT        AS ORIGIN_CU_END_BALANCE_AMT        -- 原币期末余额
    ,T.VOUCHER_CURRENCY                 AS VOUCHER_CURRENCY                 -- 凭证货币
    ,T.LOCAL_CU_END_BALANCE_AMT         AS LOCAL_CU_END_BALANCE_AMT         -- 本币期末余额
    ,T.LOCAL_CURRENCY                   AS LOCAL_CURRENCY                   -- 本币货币
    ,T6.KURSF                           AS ACTUAL_EXCHANGE_RATE             -- 实际汇率
    ,CURRENT_TIMESTAMP()	            AS DW_LAST_UPDATE_DATE				-- 数据同步时间
FROM (
SELECT 
    T.RBUKRS                			AS COMP_CODE                        -- 公司代码
    ,T.RACCT                			AS GENERAL_LEDGER_ACCOUNT           -- 总账科目
    ,T.UMSKZ                			AS SPECIAL_GENERAL_LEDGER           -- 特别总账
    ,CASE WHEN T.LIFNR <> '' THEN T.LIFNR
        ELSE CASE WHEN T.GKOAR = 'K' THEN T.GKONT
                ELSE CASE WHEN T.EBELN <> '' THEN T1.LIFNR
                        ELSE T2.LIFNR
                    END
            END
    END                     			AS SUPPLIER_CODE                    -- 供应商代码
    ,T.BLART                			AS VOUCHER_TYPE                     -- 凭证类型
    ,T.BELNR                			AS VOUCHER_CODE                     -- 凭证编码
    ,T.DOCLN                            AS VOUCHER_LINE_CODE                -- 凭证行编码
    ,T.BLDAT                            AS VOUCHER_DATE                     -- 凭证日期
    ,T.BUDAT                			AS POTING_DATE                      -- 过账日期
    ,T.AUGDT                			AS CLEARED_DATE                     -- 清账日期
    ,T.WSL                  			AS ORIGIN_CU_END_BALANCE_AMT        -- 原币期末余额
    ,T.RWCUR                			AS VOUCHER_CURRENCY                 -- 凭证货币
    ,T.HSL                  			AS LOCAL_CU_END_BALANCE_AMT         -- 本币期末余额
    ,T.RHCUR                            AS LOCAL_CURRENCY                   -- 本币货币
    ,T.GJAHR                			AS FISAL_YEAR                       -- 财年
    ,T.BUZEI                			AS ITEM                             -- 项目
FROM 
    SDI.SDI_ACDOCA_1001 T
    LEFT JOIN SDI.SDI_EKKO_1001 T1 ON T.EBELN = T1.EBELN
    LEFT JOIN SDI.SDI_MATDOC_1001 T2 ON SUBSTR(T.SGTXT,6,10) = T2.MBLNR AND SUBSTR(T.SGTXT,18,4) = T2.ZEILE
WHERE T.RACCT IN ('0021210910','0021210920') AND T.BUDAT <> '00000000' AND T.BUDAT <> ''
) T
    LEFT JOIN SDI.SDI_LFB1_1001 T1 ON T.SUPPLIER_CODE = T1.LIFNR AND T.COMP_CODE = T1.BUKRS
    LEFT JOIN SDI.SDI_T052U_1001 T2 ON T1.ZTERM = T2.ZTERM AND T2.SPRAS = '1'
    LEFT JOIN SDI.SDI_BUT000_1001 T3 ON T.SUPPLIER_CODE = T3.PARTNER
    LEFT JOIN SDI.SDI_TB004T_1001 T4 ON T3.BPKIND = T4.BPKIND AND T4.SPRAS = '1'
    LEFT JOIN SDI.SDI_SKAT_1001 T5 ON T.GENERAL_LEDGER_ACCOUNT = T5.SAKNR AND T5.KTOPL = 'CNDP'
    LEFT JOIN SDI.SDI_BKPF_1001 T6 ON T.COMP_CODE = T6.BUKRS AND T.FISAL_YEAR = T6.GJAHR AND T.VOUCHER_CODE = T6.BELNR
