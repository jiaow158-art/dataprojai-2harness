INSERT OVERWRITE TABLE DWRFIN.DWR_CRE_CUST_CREDIT_F       --客户授信
SELECT 
S.GROUP_NUM AS GROUP_NUM, 
S.CUST_ID AS CUST_ID,
S.CUST_NUM AS CUST_NUM, 
CUS.NAME_1 AS CUST_NAME,
S.CREDIT_SGMNT AS CREDIT_SGMNT,
T.CONFIG_NAME AS CREDIT_SGMNT_NAME,
S.CREDIT_LIMIT_AMT AS CREDIT_LIMIT_AMT,
S.LIMIT_CHANGE_DATE AS LIMIT_CHANGE_DATE,
S.LOG_HNDL AS LOG_HNDL,
S.LIMIT_VALID_DATE AS LIMIT_VALID_DATE,
S.CRITICAL_FLAG AS CRITICAL_FLAG,
S.LIMIT_ZERO_FLAG AS LIMIT_ZERO_FLAG,
S.DEL_FLAG AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE 
FROM DWIFIN.DWI_CRE_CUST_CREDIT_T S
INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUS     --关联客户主数据
ON S.CUST_ID = CUS.CUST_ID 
AND CUS.END_DATE ='4712-12-31' 
AND CUS.GROUP_NUM = S.GROUP_NUM 
AND CUS.DEL_FLAG = 'N' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T   --关联配置表
ON T.CONFIG_CODE = S.CREDIT_SGMNT
AND T.GROUP_NUM = S.GROUP_NUM 
AND T.LANG_CODE = '1'
