/* 补齐月份数据 */
set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nostrick;
 
with
	tmp_grp as (

		SELECT
			G.KUNNR,
			G.VKORG,
			G.VKGRP,
			G.KDGRP,
			G.PERNR
		FROM(SELECT
					G.KUNNR,
					SUBSTR(G.VKORG, 1, 2) AS VKORG,
					G.VKGRP,
					G.KDGRP,
					G.VTWEG,
					G2.PERNR,
					ROW_NUMBER() OVER (PARTITION BY G.KUNNR,SUBSTR(G.VKORG, 1, 2) ORDER BY
							G.VKORG,G.KUNNR,G.VTWEG ASC  ) RN
				FROM
					SDI.SDI_KNVV_1001 G
					LEFT JOIN sdi.sdi_knvp_1001 G2 ON G2.KUNNR = G.KUNNR
					AND SUBSTR(G.VKORG, 1, 2) = SUBSTR(G2.VKORG, 1, 2)
					AND G.VTWEG = G2.VTWEG
					and G2.parvw = 'VE'
				WHERE
					G.VKGRP <> ''
					AND G.FAKSD = '') G   WHERE  G.RN = 1
	)

, dim_date as ( select substr(day_dt,1,7) month
                  from dm.dm_dim_date_d
                  where substr(day_dt,1,7) <= substr('${period_id_m}', 1, 7)
                  group by substr(day_dt,1,7) )

, tmp as (
 SELECT
	T.YEAR							AS	YEAR							-- 年度
	,T.MONTH						AS	MONTH							-- 日历月
	,T.COMP_CODE					AS	COMP_CODE						-- 公司代码
	,T.CUST_CODE					AS	CUST_CODE						-- 客户编码
-- 	,T.CUST_NAME					AS	CUST_NAME						-- 客户全称
	,coalesce(T2.saels_grp, T1.VKGRP) 		AS 	SALES_GROUP_MASTER_FILE			-- 销售组-主档
	,T.SPECIAL_GENERAL_LEDGER		AS	SPECIAL_GENERAL_LEDGER			-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		AS	GENERAL_LEDGER_ACCOUNT			-- 总账科目
	,T.WBS							AS	WBS								-- WBS
	,T2.saels_grp				AS	WBS_SALES_GROUP					-- WBS-销售组
	,coalesce(T.CUR_CURRENCY,'RMB')		AS CURRENCY							-- 货币
	,sum(T.CUR_CU_CLOS_BALANCE_AMT)		AS CURRENT_AMT						-- 本币金额
	,T1.KDGRP              AS CUST_GROUP_CODE                  -- 客户组编码
	FROM  DWRFIN.DWR_AR_CUST_DETAIL_F T
	    left join tmp_grp t1 on t.cust_code = t1.kunnr and substr(COMP_CODE, 1, 2) = substr(t1.VKORG, 1, 2)
		left join DWRDIM.DWR_DIM_WBS_BASIS_INFO_F t2 on t.WBS = t2.wbs_level3
	where 1=1
-- 	t.GENERAL_LEDGER_ACCOUNT ='0011310100'
	and month <=substr('${period_id_m}', 1, 7)
	group  by
	T.YEAR
	,T.MONTH
	,T.COMP_CODE
	,T.CUST_CODE
-- 	,T.CUST_NAME
	,coalesce(T2.saels_grp, T1.VKGRP)
	,T.SPECIAL_GENERAL_LEDGER
	,T.GENERAL_LEDGER_ACCOUNT
	,T.WBS
	,T2.saels_grp
	,coalesce(T.CUR_CURRENCY,'RMB')
	,T1.KDGRP
 )

 ,tmp1 as (

 select
 	 substr(t2.month,1,4)	year						-- 年度
	,t2.month							-- 日历月
	,COMP_CODE						-- 公司代码
	,CUST_CODE						-- 客户编码
-- 	,CUST_NAME						-- 客户全称
	,SALES_GROUP_MASTER_FILE			-- 销售组-主档
	,SPECIAL_GENERAL_LEDGER			-- 特别总账
	,GENERAL_LEDGER_ACCOUNT			-- 总账科目
	,WBS								-- WBS
	,WBS_SALES_GROUP					-- WBS-销售组
    ,CURRENCY							-- 货币
    ,CUST_GROUP_CODE                  -- 客户组编码
    ,sum(case when t1.month = t2.month then t1.CURRENT_AMT else 0 end ) CURRENT_AMT 	-- 本币金额
 from tmp t1 join dim_date t2 on 1=1
 	where t2.month >= t1.MONTH
 group by
 	 substr(t2.month,1,4)							-- 年度
	,t2.month							-- 日历月
	,COMP_CODE						-- 公司代码
	,CUST_CODE						-- 客户编码
-- 	,CUST_NAME						-- 客户全称
	,SALES_GROUP_MASTER_FILE		-- 销售组-主档
	,SPECIAL_GENERAL_LEDGER			-- 特别总账
	,GENERAL_LEDGER_ACCOUNT			-- 总账科目
	,WBS								-- WBS
	,WBS_SALES_GROUP					-- WBS-销售组
    ,CURRENCY							-- 货币
    ,CUST_GROUP_CODE                  -- 客户组编码

 )

,tmp2 as (

      select
     YEAR							-- 年度
	,month							-- 日历月
	,COMP_CODE						-- 公司代码
	,CUST_CODE						-- 客户编码
    ,t3.match_code_srh_criteria										-- 客户全称
	,SALES_GROUP_MASTER_FILE		-- 销售组-主档
	,WBS							-- WBS
	,WBS_SALES_GROUP				-- WBS-销售组
	,SPECIAL_GENERAL_LEDGER			-- 特别总账
	,GENERAL_LEDGER_ACCOUNT			-- 总账科目
    ,sum(CURRENT_AMT) CURRENT_AMT   -- 本期货币余额
    ,CURRENCY						-- 货币
	,sum(SUM(CURRENT_AMT)) OVER(PARTITION BY COMP_CODE,CUST_CODE,SALES_GROUP_MASTER_FILE,WBS,SPECIAL_GENERAL_LEDGER,GENERAL_LEDGER_ACCOUNT ORDER BY MONTH )
							AS LOCAL_CURRENCY_BALANCE_SUM		-- 本期本币累计余额
	,CASE WHEN T.WBS_SALES_GROUP IS NULL OR T.WBS_SALES_GROUP = '' THEN T6.NODE_NAME2
		ELSE T5.NODE_NAME2
	END    					AS 	BUSINESS_DEPARTMENT				-- 事业部
-- 	,CURRENT_TIMESTAMP() 	AS	DW_LAST_UPDATE_DATE				--	数据同步时间
    ,CUST_GROUP_CODE                -- 客户组编码
    ,month period_id_m
from tmp1 T
	LEFT JOIN DM.DM_RPT_SALES_GROUP_T T5
		   ON T.WBS_SALES_GROUP = T5.NODE_NAME9
	LEFT JOIN DM.DM_RPT_SALES_GROUP_T T6
		   ON T.SALES_GROUP_MASTER_FILE = T6.NODE_NAME9
   left join DWRDIM.DWR_DIM_CUST_GENERAL_D  t3 on T.cust_code = t3.cust_num and substr(t3.end_date,1,10) = '4712-12-31'
 group by
     YEAR							-- 年度
	,month							-- 日历月
	,COMP_CODE						-- 公司代码
	,CUST_CODE						-- 客户编码
	,t3.match_code_srh_criteria						-- 客户全称
	,SALES_GROUP_MASTER_FILE		-- 销售组-主档
	,WBS							-- WBS
	,WBS_SALES_GROUP				-- WBS-销售组
	,SPECIAL_GENERAL_LEDGER			-- 特别总账
	,GENERAL_LEDGER_ACCOUNT			-- 总账科目
    ,CURRENCY						-- 货币
	,CASE WHEN T.WBS_SALES_GROUP IS NULL OR T.WBS_SALES_GROUP = '' THEN T6.NODE_NAME2
		ELSE T5.NODE_NAME2
	END    								-- 事业部
    ,cust_group_code                -- 客户组编码
)

INSERT OVERWRITE TABLE DWRFIN.dwr_ar_receivable_balance_f partition(period_id_m)
select
YEAR							-- 年度
,month							-- 日历月
,COMP_CODE						-- 公司代码
,CUST_CODE						-- 客户编码
,match_code_srh_criteria		-- 客户全称
,SALES_GROUP_MASTER_FILE		-- 销售组-主档
,WBS							-- WBS
,WBS_SALES_GROUP				-- WBS-销售组
,SPECIAL_GENERAL_LEDGER			-- 特别总账
,GENERAL_LEDGER_ACCOUNT			-- 总账科目
,CURRENT_AMT   -- 本期货币余额
,CURRENCY       
,LOCAL_CURRENCY_BALANCE_SUM		-- 本期本币累计余额
,BUSINESS_DEPARTMENT				-- 事业部
,CURRENT_TIMESTAMP() 	AS	DW_LAST_UPDATE_DATE				--	数据同步时间
,CUST_GROUP_CODE                -- 客户组编码
,month period_id_m
    from tmp2 where month = substr('${period_id_m}',1,7)
