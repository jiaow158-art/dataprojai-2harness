INSERT OVERWRITE TABLE DWRFIN.DWR_AR_COMMERCIAL_BILL_CAPITAL_COST_F partition (period_id_m) 
SELECT 
     CASE WHEN T2.sales_grp IS NULL OR T2.sales_grp = '' THEN T3.NODE_NAME2 ELSE T4.NODE_NAME2 END AS BUSINESS_DEPARTMENT						-- 事业部		
    ,T.CAPITAL_COST_TYPE                            AS CAPITAL_COST_TYPE					    -- 资金成本类型
    ,T.MONTH										AS MONTH							        -- 日历月
    ,T.COMP_CODE									AS COMP_CODE						        -- 公司代码
    ,T.CUST_CODE									AS CUST_CODE							    -- 客户编码
    ,T1.CUST_NAME									AS CUST_NAME						        -- 客户全称
    ,CASE WHEN T2.sales_grp IS NULL OR T2.sales_grp = '' THEN T1.sales_grp ELSE T2.sales_grp END AS SALES_GROUP_MASTER_FILE 	        -- 销售组-主档
	,T.WBS									        AS WBS                                      -- WBS
    ,T2.sales_grp                  	                AS WBS_SALES_GROUP                          -- WBS-销售组
	,NVL(T.CLOSING_BALANCE_202012_P,0)              AS CLOSING_BALANCE_202012_P                 -- 2020年12月账面余额
    ,NVL(T.CLOSING_BALANCE_202012_C,0)              AS CLOSING_BALANCE_202012_C                 -- 2020年12月计算余额
    ,NVL(T.CLOSING_BALANCE_LAST_MONTH,0)            AS CLOSING_BALANCE_LAST_MONTH               -- 期初余额
    ,NVL(T.CLOSING_BALANCE,0)			            AS CLOSING_BALANCE                          -- 期末余额
    ,NVL(T.CAPITAL_COST,0)							AS CAPITAL_COST					            -- 资金成本
    ,NVL(T.CAPITAL_COST_SUM,0)					    AS CAPITAL_COST_SUM							-- 累计资金成本
    ,CURRENT_TIMESTAMP()							AS DW_LAST_UPDATE_DATE						-- 数据同步时间	
	,T1.cust_grp                                     AS CUST_GROUP_CODE                          -- 客户组编码
	,CASE WHEN T2.sales_emp IS NULL OR T2.sales_emp = '' THEN T1.sales_man ELSE T2.sales_emp END AS sales_emp_code  --销售人员
	,T.MONTH                                        AS PERIOD_ID_M 
FROM 
    AR_COMMERCIAL_BILL4 T    
    LEFT JOIN DWRDIM.DWR_CUST_LATEST_DIMENSION_T T1
    ON T.CUST_CODE = T1.CUST_NUM 
    AND SUBSTR(T.COMP_CODE,1,2) = T1.SALES_ORG
    left join DWR_DIM_WBS_BASIS_INFO_F T2 
    on T.WBS = T2.WBS_NUM
    LEFT JOIN DM.DM_RPT_SALES_GROUP_T T3
	ON T1.sales_grp = T3.NODE_NAME9
	LEFT JOIN DM.DM_RPT_SALES_GROUP_T T4
	ON T2.sales_grp = T4.NODE_NAME9
    where T.MONTH = '${PERIOD_ID_M}'
GROUP BY
     CASE WHEN T2.sales_grp IS NULL OR T2.sales_grp = '' THEN T3.NODE_NAME2 ELSE T4.NODE_NAME2 END	-- 事业部
    ,T.CAPITAL_COST_TYPE								                    -- 资金成本类型
    ,T.MONTH																-- 日历月
    ,T.COMP_CODE															-- 公司代码
    ,T.CUST_CODE														    -- 客户编码
    ,T1.CUST_NAME															-- 客户全称
    ,CASE WHEN T2.sales_grp IS NULL OR T2.sales_grp = '' THEN T1.sales_grp ELSE T2.sales_grp END	-- 销售组-主档
	,T.WBS									                                -- WBS
    ,T2.sales_grp                  	                                        -- WBS-销售组
	,NVL(T.CLOSING_BALANCE_202012_P,0)                                      -- 2020年12月账面余额
    ,NVL(T.CLOSING_BALANCE_202012_C,0)                                      -- 2020年12月计算余额
    ,NVL(T.CLOSING_BALANCE_LAST_MONTH,0)                                    -- 期初余额
    ,NVL(T.CLOSING_BALANCE,0)			                                    -- 期末余额
    ,NVL(T.CAPITAL_COST,0)										            -- 资金成本
