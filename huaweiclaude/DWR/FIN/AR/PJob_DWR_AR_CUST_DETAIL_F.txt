WITH ACDOCA AS (
SELECT
	T.GJAHR								-- 年度
	,T.FISCYEARPER						-- 月度
	,T.RBUKRS		    				-- 公司代码
	,T.RACCT		    				-- 总账科目
	,D.TXT20							-- 总账科目描述
	,T.UMSKZ		    				-- 特别总账
	,T.KUNNR		    				-- 客户编码
	,D1.NAME_ORG1						-- 客户全称
	,D3.ZTERM							-- 付款条款
	,T.BLART							-- 凭证类型
	,T.BELNR		    				-- 凭证编码
	,T.BUZEI							-- 凭证项目
	,to_date(from_unixtime(unix_timestamp(T.BLDAT,'yyyyMMdd'))) BLDAT   -- 凭证日期
	,to_date(from_unixtime(unix_timestamp(T.BUDAT,'yyyyMMdd'))) BUDAT   -- 过账日期
	,to_date(from_unixtime(unix_timestamp(CASE WHEN T.AUGDT = '00000000' THEN '47121231' ELSE T.AUGDT END,'yyyyMMdd'))) AUGDT   -- 清账日期
	,to_date(from_unixtime(unix_timestamp(CASE WHEN D3.ZFBDT = '' OR D3.ZFBDT IS NULL OR D3.ZFBDT = '00000000' THEN '47121231' ELSE D3.ZFBDT END,'yyyyMMdd'))) ZFBDT		-- 付款起算日
	,CAST(NVL(D3.ZBD1T,0) AS INT) ZBD1T	-- 天数1
	,CAST(NVL(D3.ZBD2T,0) AS INT) ZBD2T	-- 天数2
	,T.WSL								-- 原币期末余额
	,T.RWCUR							-- 凭证货币
	,T.HSL								-- 本币期末余额	
	,T.RHCUR		    				-- 本币货币
	,D2.KURSF							-- 实际汇率
	,T.ZUONR							-- 分配
	,T.SGTXT							-- 项目文本
	,T.AWREF							-- 参考凭证
	,NVL(D3.XREF3,'') 	AS XREF3		-- WBS
	,T.KDAUF							-- 销售凭证
	,D3.ZZOSSDH							-- OSS计划单号
	,D10.SO_GROUP						-- 订单组
	,D11.VBELV							-- 销售订单凭证
	,M1.AUBEL							-- 销售订单
	,T.XTRUEREV
	,T.AUGBL  -- 
FROM 
	SDI.SDI_ACDOCA_1001 T
	LEFT JOIN SDI.SDI_SKAT_1001 D
		   ON T.RACCT = D.SAKNR
		  AND D.KTOPL = 'CNDP'
	LEFT JOIN SDI.SDI_BUT000_1001 D1
		   ON T.KUNNR = D1.PARTNER
	LEFT JOIN SDI.SDI_ZGATHERING_SG_1001 D10
		   ON T.KUNNR = D10.KUNNR
		  AND T.RBUKRS = D10.BUKRS
		  AND T.ZUONR = D10.SO_GROUP
	LEFT JOIN (SELECT VBELN,AUBEL,ROW_NUMBER()OVER(PARTITION BY VBELN ORDER BY POSNR ASC) RN 
				FROM 
					SDI.SDI_VBRP_1001
				WHERE MANDT = '800'
				) M1
		   ON T.AWREF = M1.VBELN
		  AND M1.RN = 1
	LEFT JOIN (SELECT VBELN,VBELV,ROW_NUMBER()OVER(PARTITION BY VBELN ORDER BY POSNN ASC) RN 
                FROM	
                    SDI.SDI_VBFA_1001
                WHERE VBTYP_V = 'C'
	            ) D11
		   ON T.AWREF = D11.VBELN
		  AND D11.RN = 1		  
	LEFT JOIN SDI.SDI_BKPF_1001 D2
		   ON T.GJAHR = D2.GJAHR
		  AND T.RBUKRS = D2.BUKRS
		  AND T.BELNR = D2.BELNR
	LEFT JOIN SDI.SDI_BSEG_1001 D3
		   ON T.GJAHR = D3.GJAHR
		  AND T.RBUKRS = D3.BUKRS
		  AND T.BELNR = D3.BELNR
		  AND T.BUZEI = D3.BUZEI	  
WHERE T.KOART = 'D' -- 科目类型客户
  AND SUBSTR(T.DW_LAST_UPDATE_DATE,1,10) >= '${PERIOD_ID_D}'
-- and T.FISCYEARPER ='2023012'
-- AND CONCAT(SUBSTRING(T.BUDAT,1,4),'-',SUBSTRING(T.BUDAT,5,2)) = SUBSTRING(DATE_SUB('${PERIOD_ID_D}',1),1,7) -- 传昨天，用昨天所在的月份过滤主表数据

GROUP BY 
	T.GJAHR								-- 年度
	,T.FISCYEARPER						-- 月度
	,T.RBUKRS		    				-- 公司代码
	,T.RACCT		    				-- 总账科目
	,D.TXT20							-- 总账科目描述
	,T.UMSKZ		    				-- 特别总账
	,T.KUNNR		    				-- 客户编码
	,D1.NAME_ORG1						-- 客户全称
	,D3.ZTERM							-- 付款条款
	,T.BLART							-- 凭证类型
	,T.BELNR		    				-- 凭证编码
	,T.BUZEI							-- 凭证项目
	,to_date(from_unixtime(unix_timestamp(T.BLDAT,'yyyyMMdd')))   -- 凭证日期
	,to_date(from_unixtime(unix_timestamp(T.BUDAT,'yyyyMMdd')))   -- 过账日期
	,to_date(from_unixtime(unix_timestamp(CASE WHEN T.AUGDT = '00000000' THEN '47121231' ELSE T.AUGDT END,'yyyyMMdd')))   -- 清账日期
	,to_date(from_unixtime(unix_timestamp(CASE WHEN D3.ZFBDT = '' OR D3.ZFBDT IS NULL OR D3.ZFBDT = '00000000' THEN '47121231' ELSE D3.ZFBDT END,'yyyyMMdd')))		-- 付款起算日
	,CAST(NVL(D3.ZBD1T,0) AS INT) 		-- 天数1
	,CAST(NVL(D3.ZBD2T,0) AS INT) 		-- 天数2
	,T.WSL								-- 原币期末余额
	,T.RWCUR							-- 凭证货币
	,T.HSL								-- 本币期末余额	
	,T.RHCUR		    				-- 本币货币
	,D2.KURSF							-- 实际汇率
	,T.ZUONR							-- 分配
	,T.SGTXT							-- 项目文本
	,T.AWREF							-- 参考凭证
	,NVL(D3.XREF3,'') 					-- WBS
	,T.KDAUF							-- 销售凭证
	,D3.ZZOSSDH							-- OSS计划单号
	,D10.SO_GROUP						-- 订单组
	,D11.VBELV							-- 销售订单凭证
	,M1.AUBEL							-- 销售订单
	,T.XTRUEREV  
	,T.AUGBL
)

ACDOCA1 AS (
SELECT
	T1.GJAHR							-- 年度
	,T1.FISCYEARPER						-- 月度
	,T1.RBUKRS		    				-- 公司代码
	,T1.RACCT		    				-- 总账科目
	,T1.TXT20							-- 总账科目描述
	,T1.UMSKZ		    				-- 特别总账
	,T1.KUNNR		    				-- 客户编码
	,T1.NAME_ORG1						-- 客户全称
	,T1.ZTERM							-- 付款条款
	,D4.VTEXT							-- 付款条款描述
	,T1.BLART							-- 凭证类型
	,T1.BELNR		    				-- 凭证编码
	,T1.BUZEI							-- 凭证项目
	,T1.BLDAT   						-- 凭证日期
	,T1.BUDAT   						-- 过账日期
	,date_add(T1.ZFBDT,T1.ZBD1T+T1.ZBD2T ) 
		AS DUE_DATE						-- 到期日
	,T1.AUGDT   						-- 清账日期	
	,T1.WSL								-- 原币期末余额
	,T1.RWCUR							-- 凭证货币
	,T1.HSL								-- 本币期末余额	
	,T1.RHCUR		    				-- 本币货币
	,T1.KURSF							-- 实际汇率
	,T1.ZUONR							-- 分配
	,T1.SGTXT							-- 项目文本
	,CASE WHEN T1.SO_GROUP <> '' OR T1.SO_GROUP IS NOT NULL THEN T1.SO_GROUP
	    ELSE CASE WHEN M2.ZSO_GROUP <> '' OR M2.ZSO_GROUP IS NOT NULL THEN M2.ZSO_GROUP
	            ELSE CASE WHEN D12.ZSO_GROUP <> '' OR D12.ZSO_GROUP IS NOT NULL THEN D12.ZSO_GROUP
	                    ELSE ''
	                   END
	           END
	END 			AS SO_GROUP_CODE	-- 订单组编码
	,T1.SO_GROUP AS SO_GROUP1
	,M2.ZSO_GROUP AS SO_GROUP2
	,D12.ZSO_GROUP AS SO_GROUP3
	--,D14.ZSO_GROUP AS SO_GROUP4
	,T1.AWREF							-- 参考凭证
	,T1.XREF3							-- WBS
	,T1.KDAUF							-- 销售凭证
	,T1.ZZOSSDH							-- OSS计划单号
	,D5.PSPHI							-- 项目号
	,T1.XTRUEREV
	,T1.AUGBL
FROM 
	ACDOCA T1
	LEFT JOIN SDI.SDI_TVZBT_1001 D4
		   ON T1.ZTERM = D4.ZTERM		-- 付款条款
		  AND D4.SPRAS = '1'
	LEFT JOIN SDI.SDI_PRPS_1001 D5
		   ON NVL(T1.XREF3,'') = NVL(D5.POSID,'')		-- WBS
	LEFT JOIN SDI.SDI_ZVBAK_1001 M2
		   ON T1.AUBEL = M2.VBELN		-- 销售订单凭证
		  AND T1.KUNNR = M2.KUNNR		   		   
	LEFT JOIN SDI.SDI_ZVBAK_1001 D12
		   ON T1.VBELV = D12.VBELN		-- 销售订单凭证
		  AND T1.KUNNR = D12.KUNNR
/*
	LEFT JOIN SDI.SDI_ZVBAK_1001 D14
		   ON T1.SGTXT = D14.VBELN		-- 凭证上的文本
		  AND T1.KUNNR = D14.KUNNR
*/
	LEFT JOIN SDI.SDI_ZGATHERING_SG_1001 D13
		   ON T1.KUNNR = D13.KUNNR
		  AND T1.RBUKRS = D13.BUKRS
		  AND T1.SGTXT = D13.SO_GROUP
GROUP BY 
	T1.GJAHR							-- 年度
	,T1.FISCYEARPER						-- 月度
	,T1.RBUKRS		    				-- 公司代码
	,T1.RACCT		    				-- 总账科目
	,T1.TXT20							-- 总账科目描述
	,T1.UMSKZ		    				-- 特别总账
	,T1.KUNNR		    				-- 客户编码
	,T1.NAME_ORG1						-- 客户全称
	,T1.ZTERM							-- 付款条款
	,D4.VTEXT							-- 付款条款描述
	,T1.BLART							-- 凭证类型
	,T1.BELNR		    				-- 凭证编码
	,T1.BUZEI							-- 凭证项目
	,T1.BLDAT   						-- 凭证日期
	,T1.BUDAT   						-- 过账日期
	,date_add(T1.ZFBDT,T1.ZBD1T+T1.ZBD2T ) 	-- 到期日
	,T1.AUGDT   						-- 清账日期	
	,T1.WSL								-- 原币期末余额
	,T1.RWCUR							-- 凭证货币
	,T1.HSL								-- 本币期末余额	
	,T1.RHCUR		    				-- 本币货币
	,T1.KURSF							-- 实际汇率
	,T1.ZUONR							-- 分配
	,T1.SGTXT							-- 项目文本
	,CASE WHEN T1.SO_GROUP <> '' OR T1.SO_GROUP IS NOT NULL THEN T1.SO_GROUP
	    ELSE CASE WHEN M2.ZSO_GROUP <> '' OR M2.ZSO_GROUP IS NOT NULL THEN M2.ZSO_GROUP
	            ELSE CASE WHEN D12.ZSO_GROUP <> '' OR D12.ZSO_GROUP IS NOT NULL THEN D12.ZSO_GROUP
	                    ELSE ''
	                   END
	           END
	END 				                -- 订单组编码
	,T1.SO_GROUP 
	,M2.ZSO_GROUP 
	,D12.ZSO_GROUP 
	--,D14.ZSO_GROUP AS SO_GROUP4
	,T1.AWREF							-- 参考凭证
	,T1.XREF3							-- WBS
	,T1.KDAUF							-- 销售凭证
	,T1.ZZOSSDH							-- OSS计划单号
	,D5.PSPHI							-- 项目号
	,T1.XTRUEREV
	,T1.AUGBL
)
INSERT OVERWRITE TABLE DWRFIN.DWR_AR_CUST_DETAIL_F_01 
-- PARTITION (PERIOD_ID_M)
SELECT
	T2.GJAHR				AS YEAR								-- 年度
	,substr(T2.FISCYEARPER,1,4)||'-'||substr(T2.FISCYEARPER,6,2)
			                AS MONTH							-- 月度
	,T2.RBUKRS				AS COMP_CODE		    			-- 公司代码
	,T2.RACCT				AS GENERAL_LEDGER_ACCOUNT    		-- 总账科目
	,T2.TXT20				AS GENERAL_LEDGER_ACCOUNT_DESC		-- 总账科目描述
	,T2.UMSKZ				AS SPECIAL_GENERAL_LEDGER    		-- 特别总账
	,T2.KUNNR				AS CUST_CODE    					-- 客户编码
	,T2.NAME_ORG1			AS CUST_NAME						-- 客户全称
	,T2.ZTERM				AS PAY_TERM_CODE					-- 付款条款
	,T2.VTEXT				AS PAY_TERM_DESC					-- 付款条款描述
	,T2.BLART				AS VOUCHER_TYPE						-- 凭证类型
	,T2.BELNR				AS VOUCHER_CODE    					-- 凭证编码
	,T2.BUZEI				AS VOUCHER_ITEM						-- 凭证项目
	,T2.BLDAT   			AS VOUCHER_DATE						-- 凭证日期
	,T2.BUDAT   			AS POSTING_DATE						-- 过账日期
	,T2.DUE_DATE			AS DUE_DATE							-- 到期日
	,T2.AUGDT   			AS CLEAR_DATE						-- 清账日期	
	,T2.WSL					AS ORI_CU_CLOS_BALANCE_AMT			-- 原币期末余额
	,T2.RWCUR				AS VOUCHER_CURRENCY					-- 凭证货币
	,T2.HSL					AS CUR_CU_CLOS_BALANCE_AMT			-- 本币期末余额	
	,T2.RHCUR				AS CUR_CURRENCY    					-- 本币货币
	,T2.KURSF				AS REAL_EXCHANGE_RATE				-- 实际汇率
	,T2.ZUONR				AS DIVIDE							-- 分配
	,T2.SGTXT				AS NOTES							-- 项目文本
	,T2.SO_GROUP_CODE		AS SO_GROUP_CODE					-- 订单组编码
	,T2.AWREF				AS REFERENCE_VOUCHER				-- 参考凭证
	,CASE WHEN D6.ZVKGRP <> '' THEN D6.ZVKGRP 
		ELSE D9.VKGRP 
	 END            		AS SALES_GROUP_MASTER_FILE		    -- 销售组-主档
	,T2.XREF3				AS WBS								-- WBS
	,D6.ZVKGRP				AS WBS_SALES_GROUP					-- WBS-销售组
	,T2.KDAUF				AS SALES_VOUCHER					-- 销售凭证
	,T2.ZZOSSDH				AS OSS_PLAN_ORDER					-- OSS计划单号
	,CASE WHEN COALESCE (D7.DEL_FLAG,D8.DEL_FLAG) IS NOT NULL THEN COALESCE (D7.DEL_FLAG,D8.DEL_FLAG)
		ELSE 'N'
	END 					AS IS_OVERDUE_CUST					-- 暴雷客户
	,D9.KDGRP				AS CUST_GROUP_CODE					-- 客户组编码
	,CURRENT_TIMESTAMP()	AS DW_LAST_UPDATE_DATE				-- 数据同步时间
	,CASE WHEN D6.ZXSRY IS NULL OR D6.ZXSRY = '' THEN D9.PERNR ELSE D6.ZXSRY END  AS business_person
	,D9.VKGRP AS sales_group_cust_master--销售组（客户主档）
	,T2.XTRUEREV AS cover_flag --冲销标识
	,T2.AUGBL  as cover_voucher
	,D10.WBANK    as  REMIT_LONGNUMBER   --251014新增字段汇款长号
--	,substr(T2.FISCYEARPER,1,4)||substr(T2.FISCYEARPER,6,2)
--			                AS PERIOD_ID_M						-- 分区（月）
FROM 
	ACDOCA1 T2
	LEFT JOIN SDI.SDI_PROJ_1001 D6
		   ON T2.PSPHI = D6.PSPNR						-- 项目号
	LEFT JOIN SDI.SDI_AMB_CREDIT_CUSTCODE_1037 D7
		   ON T2.KUNNR = D7.CUST_CODE 
		  AND T2.RBUKRS = D7.COMP_CODE 
		  AND T2.BUDAT BETWEEN D7.STAT_DATE AND D7.END_DATE
	LEFT JOIN SDI.SDI_AMB_CREDIT_CUSTCODE_1037 D8
		   ON T2.KUNNR = D8.CUST_CODE 
		  AND T2.RBUKRS IS NULL
		  AND T2.BUDAT BETWEEN D8.STAT_DATE AND D8.END_DATE
	LEFT JOIN (
			SELECT
				G.KUNNR,
				G.VKORG,
				G.VKGRP,
				G.KDGRP,
				G.PERNR
			FROM
				(
					SELECT
						G.KUNNR,
						SUBSTR(G.VKORG, 1, 2) AS VKORG,
						G.VKGRP,
						G.KDGRP,
						G.VTWEG,
						G2.PERNR,
						ROW_NUMBER() OVER(
							PARTITION BY G.KUNNR,
							SUBSTR(G.VKORG, 1, 2)
							ORDER BY
								G.VKORG,G.KUNNR,G.VTWEG ASC
						) RN
					FROM
						SDI.SDI_KNVV_1001 G
					LEFT JOIN sdi.sdi_knvp_1001 G2 ON G2.KUNNR=G.KUNNR
			           AND SUBSTR(G.VKORG, 1, 2)=SUBSTR(G2.VKORG, 1, 2)
					   AND G.VTWEG=G2.VTWEG
					   and G2.parvw='VE'
					WHERE G.VKGRP <> '' AND G.FAKSD = ''
				) G
			WHERE
				G.RN = 1
		) D9 
			  ON T2.KUNNR = D9.KUNNR
			 AND SUBSTR(T2.RBUKRS, 1, 2) = D9.VKORG	
			 LEFT JOIN SDI.SDI_BSED_1001  D10 
			 ON  T2.GJAHR = D10.gjahr
			 AND T2.RBUKRS = D10.BUKRS 
			 AND T2.BELNR = D10.BELNR
			 and T2.BUZEI = D10.buzei
