	,SUM(CASE WHEN T.DUE_DAYS between 211 AND 240 THEN NVL(T.AMT,0) 
	          ELSE 0 END)                           AS OVERDUE_RECEIVABLES_211_240_DAY		-- 211-240天逾期账龄

	          ELSE 0 END)                           AS OVERDUE_RECEIVABLES_271_300_DAY		-- 271-300天逾期账龄

	          ELSE 0 END)                           AS OVERDUE_RECEIVABLES_301_330_DAY		-- 301-330天逾期账龄

	DWRFIN.DWR_AR_CUST_EXTEND_F T
	where T.SPECIAL_GENERAL_LEDGER IN('W','V') 
GROUP BY T.COMP_CODE ,T.CUST_CODE ,T.CUST_NAME ,T.SALES_GROUP_MASTER_FILE ,T.WBS ,T.WBS_SALES_GROUP ,T.SPECIAL_GENERAL_LEDGER ,T.GENERAL_LEDGER_ACCOUNT ,T.CURRENCY,T.SO_GROUP_CODE

LEFT JOIN DM.DM_RPT_SALES_GROUP_T T5
ON T.WBS_SALES_GROUP = T5.NODE_NAME9
LEFT JOIN DM.DM_RPT_SALES_GROUP_T T6
ON T.SALES_GROUP_MASTER_FILE = T6.NODE_NAME9
