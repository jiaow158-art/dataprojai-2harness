INSERT OVERWRITE TABLE DWRFIN.DWR_PRC_PROD_PRICE_F          --产品定价
SELECT 
S.GROUP_NUM AS GROUP_NUM,
S.CONDITION_RECORD_NUM AS CONDITION_RECORD_NUM,
S.SRC_SYS_START_DATE AS SRC_SYS_START_DATE,
S.START_DATE AS START_DATE,
S.END_DATE AS END_DATE,
S.APPLICATION_PROGRAM_CODE AS APPLICATION_PROGRAM_CODE,
S.CONDITION_TYPE AS CONDITION_TYPE,
S.APPROVAL_STATUS_CODE AS APPROVAL_STATUS_CODE,
S.PRICE_TREATMENT_STATUS AS PRICE_TREATMENT_STATUS,
S.SALE_OFFICE_CODE  AS SALE_OFFICE_CODE,
T.CONFIG_NAME AS SALE_OFFICE_NAME,
S.SALE_ORG_CODE AS SALE_ORG_CODE,
T1.CONFIG_NAME AS SALE_ORG_NAME,
S.DIS_CHANNEL_CODE AS DIS_CHANNEL_CODE,
T2.CONFIG_NAME AS DIS_CHANNEL_NAME,
S.FACTORY_ID AS FACTORY_ID,
S.FACTORY_CODE AS FACTORY_CODE,
FAC.FACTORY_NAME AS FACTORY_NAME,
S.PAYER_ID AS PAYER_ID,
S.CUST_ID AS CUST_ID,
S.CUST_NUM AS CUST_NUM,
CUS.NAME_1 AS CUST_NAME,
S.MATERIAL_ID AS MATERIAL_ID,
S.MATERIAL_NUM AS MATERIAL_NUM,
MAT.MATERIAL_NAME AS MATERIAL_NAME,
S.CUST_PRICE_GROUP AS CUST_PRICE_GROUP,
S.SOURCE_TYPE AS SOURCE_TYPE,
S.SOURCE_DESC AS SOURCE_DESC,
S.DEL_FLAG AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T S
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T      --关联配置表
ON T.CONFIG_CODE = S.SALE_OFFICE_CODE 
AND T.GROUP_NUM = S.GROUP_NUM 
AND T.LANG_CODE = '1' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1     --关联配置表
ON T1.CONFIG_CODE = S.SALE_ORG_CODE 
AND T1.GROUP_NUM = S.GROUP_NUM 
AND T1.LANG_CODE = '1' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2     --关联配置表
ON T2.CONFIG_CODE = S.DIS_CHANNEL_CODE 
AND T2.GROUP_NUM = S.GROUP_NUM 
AND T2.LANG_CODE = '1'
AND T2.DEL_FLAG = 'N' 
INNER JOIN  DWIMD.DWI_MD_FACTORY_T FAC          --关联工厂主数据
ON FAC.FACTORY_ID  = S.FACTORY_ID 
AND FAC.GROUP_NUM = S.GROUP_NUM 
AND FAC.END_DATE = '4712-12-31'
AND FAC.DEL_FLAG = 'N' 
INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUS       --关联客户主数据
ON S.CUST_ID = CUS.CUST_ID 
AND CUS.END_DATE ='4712-12-31' 
AND CUS.GROUP_NUM = S.GROUP_NUM 
AND CUS.DEL_FLAG = 'N' 
INNER JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT      --关联物料主数据
ON MAT.MATERIAL_ID  = S.MATERIAL_ID 
AND MAT.GROUP_NUM = S.GROUP_NUM 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG = 'N'
