WITH 
ACDOCA AS(
SELECT
    T.ZUONR            AS   DRAFT_SHORT_NUM                 -- 汇票短号（分配）
    ,T.BELNR		   AS   VOUCHER_CODE 				    -- 凭证编号
    ,T.BLART           AS   VOUCHER_TYPE                    -- 凭证类型
    ,T.BLDAT           AS   VOUCHER_DATE                    -- 凭证日期
    ,T.BSCHL           AS   POSTING_CODE                    -- 过账码
    ,T.RHCUR           AS   VOUCHER_CURRENCY                -- 凭证货币
    ,T.SGTXT           AS   TXT                             -- 文本
    ,T.GKONT           AS   CHARGE_ACCOUNT                  -- 冲销科目记帐
    ,T.RBUKRS		   AS  	COMP_CODE			            -- 公司代码
    ,T.BUDAT           AS   POSTING_DATE                    -- 过帐日期
    ,T.FISCYEARPER     AS   YEAR_MONTH                      -- 年度/月份
    ,T.RACCT           AS   ACCOUNT                         -- 科目
    ,T.HSL             AS   AMT                             -- 金额
    ,T.LIFNR           AS   SUPPLIER_CODE                   -- 供应商编码
    ,T.KUNNR           AS   CUST_CODE                       -- 客户编码
    ,B.XBLNR           AS   REF_DOC                         -- 参照
    ,CASE WHEN T.PS_POSID IS NULL OR T.PS_POSID = '' THEN C.XREF3 ELSE RPAD(T.PS_POSID,16,0) END AS WBS_NUM  -- WBS编码
    ,CASE WHEN day(to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd'))) - to_date(from_unixtime(unix_timestamp(T.BUDAT,'yyyyMMdd')))) >= 0 THEN day(to_date(from_unixtime    (unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd'))) - to_date(from_unixtime(unix_timestamp(T.BUDAT,'yyyyMMdd'))))
    		ELSE 0
    	END DUE_DAYS					                        -- 到期天数
    ,B.CPUDT AS  INPUT_DATE --输入日期（新增于6月13日）
FROM SDI.SDI_ACDOCA_1001 T
LEFT JOIN SDI.SDI_BKPF_1001 B
ON T.BELNR = B.BELNR AND T.RBUKRS = B.BUKRS AND T.GJAHR = B.GJAHR
LEFT JOIN SDI.SDI_BSEG_1001 C
ON T.GJAHR = C.GJAHR AND T.RBUKRS = C.BUKRS AND T.BELNR = C.BELNR AND T.BUZEI = C.BUZEI
WHERE T.RACCT IN ('0011330120','0011330901','0011330140','0011330110','0011330130','0011330900','0011330100')
AND to_date(from_unixtime(unix_timestamp(T.BUDAT,'yyyyMMdd'))) <= CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END 
AND T.BUDAT <> '00000000'
AND (T.AUGDT = '00000000' OR to_date(from_unixtime(unix_timestamp(T.AUGDT,'yyyyMMdd'))) > CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END)
--GROUP BY 
--    T.ZUONR                                                 -- 汇票短号（分配）
--    ,T.BELNR		    				                    -- 凭证编号

--    ,T.RBUKRS		    				                    -- 公司代码

--    ,T.HSL                                                  -- 金额
)
INSERT OVERWRITE TABLE DWRFIN.DWR_AR_OTHER_RECEIVABLE_AGING_F
SELECT
    CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END AS QUERY_DATE		                    -- 查询日期
    ,T.DRAFT_SHORT_NUM                 	                    -- 汇票短号（分配）
    ,T.VOUCHER_CODE 				                        -- 凭证编号
    ,T.VOUCHER_TYPE                    	                    -- 凭证类型
    ,T.VOUCHER_DATE                    	                    -- 凭证日期
    ,T.POSTING_CODE                    	                    -- 过账码
    ,T.VOUCHER_CURRENCY                	                    -- 凭证货币
    ,T.TXT                             	                    -- 文本
    ,T.CHARGE_ACCOUNT                  	                    -- 冲销科目记帐
    ,T.COMP_CODE			                                -- 公司代码
    ,T.POSTING_DATE                    	                    -- 过帐日期
    ,T.YEAR_MONTH                      	                    -- 年度/月份
    ,T.ACCOUNT                         	                    -- 科目
    ,T.AMT                             	                    -- 金额
    ,T.DUE_DAYS					                            -- 到期天数
    ,CASE WHEN T.DUE_DAYS >= 0 AND T.DUE_DAYS <= 365 THEN T.AMT 
        ELSE 0 
     END WITHIN_1YEAR_AMT                                   -- 1年以内金额
    ,CASE WHEN T.DUE_DAYS >= 366 AND T.DUE_DAYS <= 730 THEN T.AMT 
        ELSE 0 
     END 1_2YEAR_AMT                                        -- 1-2年金额
    ,CASE WHEN T.DUE_DAYS >= 731 AND T.DUE_DAYS <= 1095 THEN T.AMT 
        ELSE 0 
     END 2_3YEAR_AMT                                        -- 2-3年金额
    ,CASE WHEN T.DUE_DAYS >= 1096 AND T.DUE_DAYS <= 1460 THEN T.AMT
        ELSE 0 
     END 3_4YEAR_AMT                                        -- 3-4年金额
    ,CASE WHEN T.DUE_DAYS >= 1461 THEN T.AMT
        ELSE 0 
     END OVER_4YEAR_AMT                                     -- 4年以上金额
    ,CASE WHEN T.DUE_DAYS >= 0 AND T.DUE_DAYS <= 365 THEN T.AMT * (${P_NUM1}/100)
        ELSE 0 
     END WITHIN_1YEAR_AGING                                 -- 1年以内账龄
    ,CASE WHEN T.DUE_DAYS >= 366 AND T.DUE_DAYS <= 730 THEN T.AMT * (${P_NUM2}/100)
        ELSE 0 
     END 1_2YEAR_AGING                                      -- 1-2年账龄
    ,CASE WHEN T.DUE_DAYS >= 731 AND T.DUE_DAYS <= 1095 THEN T.AMT * (${P_NUM3}/100)
        ELSE 0 
     END 2_3YEAR_AGING                                      -- 2-3年账龄
    ,CASE WHEN T.DUE_DAYS >= 1096 AND T.DUE_DAYS <= 1460 THEN T.AMT * (${P_NUM4}/100)
        ELSE 0 
     END 3_4YEAR_AGING                                      -- 3-4年账龄
    ,CASE WHEN T.DUE_DAYS >= 1461 THEN T.AMT * (${P_NUM5}/100)
        ELSE 0 
     END OVER_4YEAR_AGING                                   -- 4年以上账龄
    ,T1.DIVISION_CODE	 AS BUSINESS_DEPARTMENT				-- 事业部
    ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE				-- 数据同步时间
    ,T.SUPPLIER_CODE                                        -- 供应商编码
    ,T.CUST_CODE                                            -- 客户编码
    ,T.REF_DOC                                              -- 参照
    ,T2.CUST_GRP  AS CUST_GRP_CODE                          -- 客户组
    ,T.WBS_NUM                                              -- WBS编码
    ,CASE WHEN T3.ZVKGRP IS NULL OR T3.ZVKGRP = '' THEN T2.SALES_GRP ELSE T3.ZVKGRP END AS SALES_GRP_CODE --销售组编码
    ,T.INPUT_DATE --输入日期（新增于6月13日）
FROM ACDOCA T
LEFT JOIN SDI.SDI_DIVISION_COMP_T_1037 T1
 ON T.COMP_CODE = T1.COMP_CODE	                     -- 公司代码
LEFT JOIN DWRDIM.DWR_CUST_LATEST_DIMENSION_T T2
 ON T.CUST_CODE = T2.CUST_NUM AND SUBSTR(T.COMP_CODE,1,2) = T2.SALES_ORG
LEFT JOIN SDI.SDI_PROJ_1001 T3
ON T3.PSPID = SUBSTR(T.WBS_NUM,1,10) 
 where substr(t1.end_date,1,10)='9999-12-31'
