-- 取本币累计余额、回款金额、逾期应收、1年内未逾期应收、1-2年内未逾期应收、2-3年内未逾期应收、3-4年内未逾期应收、4年以上未逾期应收
WITH ACDOCA4 AS(
SELECT 
	T.COMP_CODE		    			            	AS COMP_CODE                            -- 公司代码
	,T.CUST_CODE		    				        AS CUST_CODE                            -- 客户编码
	,T.CUST_NAME						            AS CUST_NAME                            -- 客户全称
	,T.SALES_GROUP_MASTER_FILE		                AS SALES_GROUP_MASTER_FILE              -- 销售组-主档
	,T.WBS							                AS WBS                                  -- WBS
	,T.WBS_SALES_GROUP						        AS WBS_SALES_GROUP                      -- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    	        AS SPECIAL_GENERAL_LEDGER               -- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    	        AS GENERAL_LEDGER_ACCOUNT               -- 总账科目
	,T.CURRENCY		    				        	AS CURRENCY                             -- 货币
	,T.SO_GROUP_CODE								AS SO_GROUP_CODE						-- 订单组编码
	,T.LOCAL_CURRENCY_BALANCE_SUM           		AS LOCAL_CURRENCY_BALANCE_SUM			-- 本币累计余额
	,T.REPAY_AMT                            		AS REPAY_AMT							-- 回款金额
	,greatest(T.OVERDUE_RECEIVABLES,0)              AS OVERDUE_RECEIVABLES					-- 逾期应收
	,T.N_OVERDUE_RECEIVABLES                		AS N_OVERDUE_RECEIVABLES				-- 未逾期应收余额
	,CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 <= 0 THEN T.N_OVERDUE_RECEIVABLES
		ELSE T.N_OVERDUE_RECEIVABLES_0_365
	END 											AS N_OVERDUE_RECEIVABLES_0_365        	-- 1年内未逾期应收余额
	,CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 <= 0 THEN 0
		ELSE 
			CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 <= 0 
				THEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365
			ELSE T.N_OVERDUE_RECEIVABLES_366_730
			END
	END 											AS N_OVERDUE_RECEIVABLES_366_730       	-- 1-2年未逾期应收余额
	,CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 <= 0 THEN 0
		ELSE 
			CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 <= 0
				THEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730
			ELSE T.N_OVERDUE_RECEIVABLES_731_1095
			END
	END 											AS N_OVERDUE_RECEIVABLES_731_1095      	-- 2-3年未逾期应收余额
	,CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 <= 0 THEN 0
		ELSE
			CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 - T.N_OVERDUE_RECEIVABLES_1096_1460 <= 0
				THEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095
			ELSE T.N_OVERDUE_RECEIVABLES_1096_1460
			END
	END 											AS N_OVERDUE_RECEIVABLES_1096_1460     	-- 3-4年未逾期应收余额
	,CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 - T.N_OVERDUE_RECEIVABLES_1096_1460 <= 0 THEN 0 
		ELSE 
			CASE WHEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 - T.N_OVERDUE_RECEIVABLES_1096_1460 - T.N_OVERDUE_RECEIVABLES_1461 <= 0
				THEN T.N_OVERDUE_RECEIVABLES - T.N_OVERDUE_RECEIVABLES_0_365 - T.N_OVERDUE_RECEIVABLES_366_730 - T.N_OVERDUE_RECEIVABLES_731_1095 - T.N_OVERDUE_RECEIVABLES_1096_1460
			ELSE T.N_OVERDUE_RECEIVABLES_1461
			END
	END 											AS N_OVERDUE_RECEIVABLES_1461          	-- 4年以上未逾期应收余额
FROM
(SELECT
	T.COMP_CODE		    			            	AS COMP_CODE                            -- 公司代码
	,T.CUST_CODE		    				        AS CUST_CODE                            -- 客户编码
	,T.CUST_NAME						            AS CUST_NAME                            -- 客户全称
	,T.SALES_GROUP_MASTER_FILE		                AS SALES_GROUP_MASTER_FILE              -- 销售组-主档
	,T.WBS							                AS WBS                                  -- WBS
	,T.WBS_SALES_GROUP						        AS WBS_SALES_GROUP                      -- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    	        AS SPECIAL_GENERAL_LEDGER               -- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    	        AS GENERAL_LEDGER_ACCOUNT               -- 总账科目
	,T.CURRENCY		    				        	AS CURRENCY                             -- 货币
	,T.SO_GROUP_CODE								AS SO_GROUP_CODE						-- 订单组编码
	,SUM(NVL(T.AMT,0)) 							    AS LOCAL_CURRENCY_BALANCE_SUM          	-- 本币累计余额
	,SUM(REPAY_AMT) 				            	AS REPAY_AMT                			-- 回款金额
	,greatest(SUM(NVL(T.AMT,0)) -greatest(SUM(CASE WHEN T.DUE_DATE <T.DATE1 THEN OVERDUE_AMT ELSE 0 END)
	                   +SUM(REPAY_AMT),0),0)              AS N_OVERDUE_RECEIVABLES                -- 未逾期应收余额
	,SUM(CASE WHEN T.DUE_DATE <T.DATE1 THEN OVERDUE_AMT ELSE 0 END)
	                   +SUM(REPAY_AMT) 	AS OVERDUE_RECEIVABLES                  -- 逾期应收
	,SUM(CASE WHEN  T.DUE_DATE >=T.DATE1 AND T.N_OVERDUE_DUE_DAYS BETWEEN 0 AND 365 
	               THEN OVERDUE_AMT ELSE 0 END)    AS N_OVERDUE_RECEIVABLES_0_365			-- 1年内未逾期应收余额 
	,SUM(CASE WHEN  T.DUE_DATE >=T.DATE1 AND T.N_OVERDUE_DUE_DAYS BETWEEN 366 AND 730 
	               THEN OVERDUE_AMT ELSE 0 END)    AS N_OVERDUE_RECEIVABLES_366_730		-- 2年内未逾期应收余额
	,SUM(CASE WHEN  T.DUE_DATE >=T.DATE1 AND T.N_OVERDUE_DUE_DAYS BETWEEN 731 AND 1095 
	               THEN OVERDUE_AMT ELSE 0 END)    AS N_OVERDUE_RECEIVABLES_731_1095		-- 2-3年未逾期应收余额
	,SUM(CASE WHEN  T.DUE_DATE >=T.DATE1 AND T.N_OVERDUE_DUE_DAYS BETWEEN 1096 AND 1460 
	               THEN OVERDUE_AMT ELSE 0 END)    AS N_OVERDUE_RECEIVABLES_1096_1460		-- 3-4年未逾期应收余额
	,SUM(CASE WHEN  T.DUE_DATE >=T.DATE1 AND T.N_OVERDUE_DUE_DAYS >1460 
	               THEN OVERDUE_AMT ELSE 0 END)    AS N_OVERDUE_RECEIVABLES_1461			-- 4年以上未逾期应收余额
FROM 
	DWRFIN.DWR_AR_CUST_EXTEND_F T
	WHERE T.SPECIAL_GENERAL_LEDGER IN('','U','Q','M')
GROUP BY T.COMP_CODE ,T.CUST_CODE ,T.CUST_NAME ,T.SALES_GROUP_MASTER_FILE ,T.WBS ,T.WBS_SALES_GROUP ,T.SPECIAL_GENERAL_LEDGER ,T.GENERAL_LEDGER_ACCOUNT ,T.CURRENCY,T.SO_GROUP_CODE
) T

)
INSERT OVERWRITE TABLE DWRFIN.DWR_AR_N_RECEIVABLE_AGING_F2 partition(period_id_m)  
SELECT 
	 '${PERIOD_ID_D}' 								AS QUERY_DATE							-- 查询日期
	,T.COMP_CODE		    						AS COMP_CODE							-- 公司代码
	,T.CUST_CODE		    						AS CUST_CODE							-- 客户编码
	,T.CUST_NAME									AS CUST_NAME							-- 客户全称
	,T.SALES_GROUP_MASTER_FILE						AS SALES_GROUP_MASTER_FILE				-- 销售组-主档
	,T.WBS											AS WBS									-- WBS
	,T.WBS_SALES_GROUP								AS WBS_SALES_GROUP						-- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    			AS SPECIAL_GENERAL_LEDGER				-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    			AS GENERAL_LEDGER_ACCOUNT				-- 总账科目
	,T.CURRENCY		    							AS CURRENCY								-- 货币
	,T.LOCAL_CURRENCY_BALANCE_SUM					AS LOCAL_CURRENCY_BALANCE_SUM			-- 本币累计余额
	,T.REPAY_AMT                            		AS REPAY_AMT                        	-- 回款金额
	,T.OVERDUE_RECEIVABLES 							AS OVERDUE_RECEIVABLES					-- 逾期应收
	,T.N_OVERDUE_RECEIVABLES                		AS N_OVERDUE_RECEIVABLES            	-- 未逾期应收余额
	,T.N_OVERDUE_RECEIVABLES_0_365        			AS N_OVERDUE_RECEIVABLES_0_365			-- 1年内未逾期应收余额
	,T.N_OVERDUE_RECEIVABLES_366_730       			AS N_OVERDUE_RECEIVABLES_366_730		-- 1-2年未逾期应收余额
	,T.N_OVERDUE_RECEIVABLES_731_1095      			AS N_OVERDUE_RECEIVABLES_731_1095		-- 2-3年未逾期应收余额
	,T.N_OVERDUE_RECEIVABLES_1096_1460     			AS N_OVERDUE_RECEIVABLES_1096_1460		-- 3-4年未逾期应收余额
	,T.N_OVERDUE_RECEIVABLES_1461          			AS N_OVERDUE_RECEIVABLES_1461			-- 4年以上未逾期应收余额
	,CASE WHEN T.WBS_SALES_GROUP IS NULL OR T.WBS_SALES_GROUP = '' THEN T6.NODE_NAME2
		ELSE T5.NODE_NAME2 
	END                                    			AS BUSINESS_DEPARTMENT			    	-- 事业部
	,CURRENT_TIMESTAMP()							AS DW_LAST_UPDATE_DATE					-- 数据同步时间
	,T.SO_GROUP_CODE								AS SO_GROUP_CODE						-- 订单组编码
	,SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) AS period_id_m
FROM 
	ACDOCA4 T
LEFT JOIN DM.DM_RPT_SALES_GROUP_T T5
ON T.WBS_SALES_GROUP = T5.NODE_NAME9
LEFT JOIN DM.DM_RPT_SALES_GROUP_T T6
ON T.SALES_GROUP_MASTER_FILE = T6.NODE_NAME9
 ;
