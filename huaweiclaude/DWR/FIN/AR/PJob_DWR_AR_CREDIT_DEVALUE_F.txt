with CONTRACT_TERMS  as (
select  
customer_code, wbs_number, company_code, contract_id, contract_amount, advance_payment_amount, advance_payment_ratio, progress_payment_ratio, settlement_payment_ratio, retention_money_ratio, contract_source,
row_number()over  (partition by customer_code,wbs_number,company_code ORDER BY CASE WHEN contract_source = 'LTC' THEN 1 ELSE 2 END )  AS RN 
from
DWRFIN.DWRFIN_CONTRACT_TERMS_F 
)
, cust_detail_f as (
select 
 to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd')))												        AS QUERY_DATE							-- 查询日期
	,MAX(T3.cust_class_name) AS    cust_class_name      --客户分类
	,MAX(coalesce(T4.contract_amount,0)) as  contract_amount        --合同总金额
	,MAX(coalesce(T4.progress_payment_ratio,0)) as  progress_payment_ratio --进度款比例
	,MAX(coalesce(T4.settlement_payment_ratio,0)) as settlement_payment_ratio --结算款比例
	,MAX(coalesce(T4.retention_money_ratio,0)) as retention_money_ratio   --质保金比例
    ,SUM(case when 	voucher_type = 'RV' and special_general_ledger in ('U','Q','M','','&') THEN cur_cu_clos_balance_amt ELSE 0 END)  AS SHIP_VAL  --发货额
    ,SUM(CASE WHEN coalesce(voucher_type,'#')  NOT IN ('RV') and special_general_ledger in ('U','Q','M','','&') THEN cur_cu_clos_balance_amt ELSE 0 END)  AS Collections_VAL--回款额
    ,SUM(case when 	voucher_type = 'RV' and special_general_ledger in ('U','Q','M','','&')  THEN cur_cu_clos_balance_amt * 
    case when coalesce(T4.progress_payment_ratio,0) + coalesce(T4.settlement_payment_ratio,0) + coalesce(T4.retention_money_ratio,0) = 0 then 100 else  coalesce(T4.progress_payment_ratio,0) end 
    *0.01
    ELSE 0 END)  AS SHIP_Progress--发货进度款
    ,SUM(case when 	voucher_type = 'RV' and special_general_ledger in ('U','Q','M','','&')  THEN cur_cu_clos_balance_amt * coalesce(T4.settlement_payment_ratio,0)*0.01 ELSE 0 END)  AS SHIP_settlement--发货结算款
    ,SUM(case when 	voucher_type = 'RV' and special_general_ledger in ('U','Q','M','','&')  THEN cur_cu_clos_balance_amt * coalesce(T4.retention_money_ratio,0)*0.01 ELSE 0 END)  AS SHIP_retention--发货质保金
    ,SUM(T.cur_cu_clos_balance_amt) AS cur_cu_clos_balance_amt
    ,SUM(CASE WHEN  GENERAL_LEDGER_ACCOUNT = '0021310200' then cur_cu_clos_balance_amt else 0 end) AS balance_amt_cl
    ,T.COMP_CODE                                                                 -- 公司代码
	,T.CUST_CODE                                         	                      -- 客户编码
	,max(T.CUST_NAME) as     CUST_NAME                                                           -- 客户全称
-- 	,T.sales_group_cust_master                                      -- 销售组-主档
	,T.WBS                                                                     -- WBS
-- 	,T.WBS_SALES_GROUP                                                   -- WBS-销售组
	
	,MAX(T.cur_currency) AS       cur_currency                                                 -- 货币
	,MAX(T5.`/bic/za_syamb`)       as BUSINESS_DEPARTMENT                                         -- 事业部
	,T.SO_GROUP_CODE                					   	          		-- 订单组编码	
	,T.general_ledger_account                                               --总账科目
from 
DWRFIN.DWR_AR_CUST_DETAIL_F T 
LEFT JOIN DWRDIM.DWR_DIM_CUST_GENERAL_D  T3 
            ON T.CUST_CODE = T3.cust_num
            and T3.end_date = '4712-12-31' 
LEFT JOIN  CONTRACT_TERMS  T4
            ON T.CUST_CODE = T4.customer_code
            and   T.WBS = T4.wbs_number
            and  T.comp_code =T4.company_code
            AND T4.RN = 1 
LEFT JOIN SDI.SDI_BIC_AZMAM_O0062_1009 T5
            ON T.CUST_CODE = T5.comp_code
where posting_date  <=  to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd')))		 and special_general_ledger in ('','U','Q','M','O','&')
