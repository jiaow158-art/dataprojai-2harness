WITH
	DWR_FIN_COST_D_COMPRE_SUBJ_T_T AS (
		SELECT
	    	*
		FROM
			DWRFIN.DWR_FIN_COST_D_COMPRE_SUBJ_T
		WHERE
			comp_code = 'Z000'
			AND (
				general_ledger_account = '0030092000'
				OR general_ledger_account = '0051011000'
			)
			AND MONTH >= '${PERIOD_ID_M}'--YYYY-MM格式
	)
,COSTCENTER_T AS (
	SELECT
		sales_grp,
		costcenter,
		ROW_NUMBER() OVER (PARTITION BY costcenter ORDER BY	DATETO DESC	) RN
	FROM
		SDI.SDI_BI0_QCOSTCENTER_1009
	WHERE
		CO_AREA = 'DP00'
		AND SALES_GRP IS NOT NULL
		OR SALES_GRP <> ''
)
INSERT
	OVERWRITE TABLE DWRFIN.DWR_GL_Accounting_Voucher_F  PARTITION (period_id_m)
SELECT
	T1.comp_code AS COMP_CODE, --公司代码
	T1.fin_voucher_code AS ZJFGGBH, --更改记录编号/凭证号
	T1.fin_voucher_line_code AS ZJFGGIT, -- 行项目
	T1.month AS CALMONTH, --年月
	T1.cust_code AS CUSTOMER, --客户编码
	T1.sales_org_code AS SALESORG, --销售组织
	T1.channel_code AS DISTR_CHAN, --分销渠道
	'' AS ZJFLX, --积分类型
	'' AS ZJFMS, --积分类型描述
	'' AS ZJFBGLX, --积分变更类型
	'' AS ZJFBGLXMS, --积分变更类型描述
	substring(T1.posting_date,1,10) AS BUDAT, --过账日期
	substring(T1.voucher_date,1,10) AS ERDAT, --操作日期
	'' AS VBELN, --销售订单
	'' AS POSNR, --销售订单行项目
	'' AS VBELN_VF, --开票抬头
	'' AS POSNR_VF, --开票项目
	'' AS OrderType, --订单类型
	T1.material_code AS MATERIAL, --物料编码
	T1.material_desc AS material_text, --物料描述
	T1.qty AS KWMENG, --订单数量
	'' AS OrderArea, --订单面积
	'' AS SALESUNIT, --销售单位
	'' AS Offset_mark, --冲销标识
	T1.wbs AS WBSID, --WBS号
	T1.wbs_desc AS WBS_Description, --WBS描述
	'' AS ReconciliationDate, --清账日期
	'' AS ReconciliationFlag, --清账标识
	'' AS ProductionRegion, --产区
	T1.local_currency_amt AS PointsValue, --积分金额
	'' AS PolicyName, --政策名称
	'' AS NetPrice2, --静价2
	'' AS EngineeringChange, --工程变更
	'' AS ASPricingDate, --定价日期
	T1.general_ledger_account, --总账科目
	T1.general_ledger_account_desc, --总账科目描述
	CASE
		WHEN T1.general_ledger_account = '0030092000'
		OR general_ledger_account = '0051011000' THEN 'A'
		ELSE NULL
	END AS PointsCategoryCode, --管理口径积分分类编码
	CASE
		WHEN T1.general_ledger_account = '0030092000'
		OR general_ledger_account = '0051011000' THEN '预提积分-手工'
		ELSE NULL
	END AS PointsCategory, --管理口径积分分类
	'' AS PointsBusinessScopeCode, --管理口径使用积分业务范围编码
	'' AS UsePointsBusinessScope, --管理口径使用积分业务范围
	T2.sales_group_code AS sales_group_code, --销售组（主档）
	CASE
		WHEN T1.sales_group_code IS NOT NULL AND T1.sales_group_code <>'' THEN T1.sales_group_code
		WHEN (T1.sales_group_code IS NULL OR T1.sales_group_code = '')
		AND (T1.wbs IS NOT NULL AND T1.wbs <> '') THEN T2.sales_group_code
		WHEN (T2.sales_group_code IS NULL OR T2.sales_group_code = '')
		AND (T1.cost_center_code IS NOT NULL  AND  T1.cost_center_code <> '') THEN T3.sales_grp
		ELSE null
	END AS sales_group_code_WBS, --销售组（主档+WBS）
	T1.sales_group_code AS sales_group_code_actual, -- 销售组（实际）
	CURRENT_TIMESTAMP AS last_update_date,
	'N' AS DEL_FLAG,
	''  AS score_classification,--审计口径积分分类
	T1.item_text AS points_text, -- 文本
	T1.allocation AS points_distribution,--分配
	''  AS cust_refer, --客户参考
	''  AS purchase_order_type,--采购订单类型
	CONCAT(
		SUBSTR(T1.month, 1, 4),
		SUBSTR(T1.month, 6, 2)
	) AS period_id_m--分区字段
FROM           
	DWR_FIN_COST_D_COMPRE_SUBJ_T_T T1
	LEFT JOIN DWRDIM.dwr_dim_cust_sales_area_d T2 ON T2.sales_org_code = T1.sales_org_code
	AND T2.src_cust_num = T1.cust_code
	AND T2.distribution_chnl_code = T1.channel_code
	AND T2.end_date = '4712-12-31'
	AND T2.del_flag = 'N'
	LEFT JOIN COSTCENTER_T T3
    ON T1.cost_center_code = T3.costcenter
    AND T3.RN = 1
