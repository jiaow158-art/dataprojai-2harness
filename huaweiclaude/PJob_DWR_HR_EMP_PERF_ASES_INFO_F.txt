INSERT OVERWRITE TABLE DWRHR.DWR_HR_EMP_PERF_ASES_INFO_F
SELECT A.PERNR AS EMP_NUM --员工号
      ,B.EMP_NAME --员工姓名
      ,A.ZZ_KHZQ AS ASES_PER_CODE --考核周期
      ,A.TEXT1 AS ASES_PER_NAME --考核周期描述	  
      ,SUBSTR(A.BEGDA,1,4)||'-'||SUBSTR(A.BEGDA,5,2)||'-'||SUBSTR(A.BEGDA,7,2) AS START_DATE --开始日期
      ,SUBSTR(A.ENDDA,1,4)||'-'||SUBSTR(A.ENDDA,5,2)||'-'||SUBSTR(A.ENDDA,7,2) AS END_DATE --结束日期
      ,A.ORGEH_T AS DEPT_NAME --部门
      ,A.PLANS_T AS POST_NAME --岗位
      ,A.STAT2 AS EMP_ST_CODE --雇佣状态
      ,A.TEXT2 AS EMP_ST_NAME --雇佣状态描述
      ,A.ZZ_KHF AS ASES_PT --考核分数
      ,A.ZZ_KHXS AS ASES_COFT --考核系数
      ,A.ZZ_KHJG AS ASES_RANK --考核等级
      ,A.ZZ_JXXS AS PERF_COFT --绩效系数
      ,A.ZZ_ZZXS AS ORG_COFT --组织系数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_ZTHRPYJX_1001 A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON A.PERNR = B.EMP_NUM
 WHERE A.MANDT = 800
