DELETE FROM DM.DM_WF_REQUESTS_CONSUMING_OVERTIME_T
 WHERE STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
;

INSERT INTO DM.DM_WF_REQUESTS_CONSUMING_OVERTIME_T
SELECT 
       STAT_MONTH --统计月
      ,REQUEST_CODE --请求ID
      ,REQUEST_EXPLAIN --流水号
      ,DEPT_NAME --所属部门
      ,LASTNAME --节点操作人名称
      ,SAPID --节点操作人SAPID
      ,NODE_NAME --节点名称
      ,WF_TYPE_NAME --L1流程        
      ,WF_NAME --流程名称
      ,CURRENT_NODE_TYPE --当期流程类型
      ,IS_ZX --主线并行
      ,RECEIVE_DATE --接收日期
      ,RECEIVE_TIME --接收时间
      ,OPERATE_DATE --处理日期
      ,OPERATE_TIME --处理时间
      ,OPERATEDATESJ --实际处理日期
      ,OPERATETIMESJ --实际处理时间
      ,OPERATE_TYPE  --操作类型
      ,LAST_OPERATE_TYPE --上一操作类型
      ,HS_HOUR  --耗时24小时制
      ,HS_MINUTE --耗时（工作时间：分钟）
      ,DEPT_JG_NAME --部门架构
      ,NODE_COUNT --节点数
      ,GZ_HOUR --耗时（工作时间：小时）
      ,LEV1_DEPT_NAME --部门层级1
      ,LEV2_DEPT_NAME --部门层级2
      ,LEV3_DEPT_NAME --部门层级3
      ,LEV4_DEPT_NAME --部门层级4
      ,LEV5_DEPT_NAME --部门层级5
      ,LEV6_DEPT_NAME --部门层级6
      ,LEV7_DEPT_NAME --部门层级7
      ,LEV8_DEPT_NAME --部门层级8
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_PRE_T
WHERE 1=1
AND(STAT_MONTH >= 202209 AND STAT_MONTH < 202303 AND GZ_HOUR > 4
OR (STAT_MONTH < 202209 AND GZ_HOUR > 6)
OR (STAT_MONTH >= 202303 AND GZ_HOUR > 3.5))
AND STAT_MONTH = SUBSTR(REPLACE(RECEIVE_DATE,'-',''),1,6)
AND CURRENT_NODE_TYPE = '审批'
and stat_month = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
;
