TRUNCATE TABLE DM.DM_CS_ZDJ_ORDER_STATUS_STAT_T;

INSERT INTO DM.DM_CS_ZDJ_ORDER_STATUS_STAT_T
(
       STAT_MONTH --统计月份
      ,TYPE_NAME --装修类型
	  ,ORDER_STATUS_ID --订单状态ID
	  ,ORDER_STATUS_NAME --订单状态
	  ,SIGN_ORDER_CNT --签单数
	  ,OVER_ORDER_CNT --完工数
	  ,TURN_ORDER_CNT --转单数
	  ,ORDER_CNT --订单数
	  ,DJC_ORDER_CNT --待进场数
	  ,JD_ORDER_CNT --建单数
	  ,ZJD_ORDER_CNT --总部建单数
	  ,QJD_ORDER_CNT --区域建单数
      ,DW_LAST_UPDATE_DATE --DW数据同步时间
)
SELECT STAT_MONTH --统计月份
      ,TYPE_NAME --装修类型
	  ,ORDER_STATUS_ID --订单状态ORDER_NUM
	  ,ORDER_STATUS_NAME --订单状态
	  ,SUM(SIGN_ORDER_CNT) AS SIGN_ORDER_CNT --签单数
	  ,SUM(OVER_ORDER_CNT) AS OVER_ORDER_CNT --完工数
	  ,SUM(TURN_ORDER_CNT) AS TURN_ORDER_CNT --转单数
      ,SUM(ORDER_CNT) AS ORDER_CNT --订单数
	  ,SUM(DJC_ORDER_CNT) AS DJC_ORDER_CNT --待进场数
	  ,SUM(JD_ORDER_CNT) AS JD_ORDER_CNT --建单数
	  ,SUM(ZJD_ORDER_CNT) AS ZJD_ORDER_CNT --总部建单数
	  ,SUM(QJD_ORDER_CNT) AS QJD_ORDER_CNT --区域建单数	  
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM(
SELECT SUBSTR(A.ORDER_SIGN_TIME,1,7) AS STAT_MONTH --统计月份
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END AS TYPE_NAME --装修类型
	  ,A.STAGE_ID AS ORDER_STATUS_ID --订单状态ORDER_NUM
	  ,A.STAGE_NAME AS ORDER_STATUS_NAME --订单状态
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL THEN ORDER_NUM END) AS SIGN_ORDER_CNT --签单数
	  ,0 AS OVER_ORDER_CNT --完工数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE NOT IN(1,2) THEN ORDER_NUM END) AS TURN_ORDER_CNT --转单数
      ,COUNT(DISTINCT ORDER_NUM) AS ORDER_CNT --订单数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_NAME IN('待进场') THEN A.ORDER_NUM END) AS DJC_ORDER_CNT --待进场数
	  ,0 AS JD_ORDER_CNT --建单数
	  ,0 AS ZJD_ORDER_CNT --总部建单数
	  ,0 AS QJD_ORDER_CNT --区域建单数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY
       SUBSTR(A.ORDER_SIGN_TIME,1,7)
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END
	  ,A.STAGE_ID		 
	  ,A.STAGE_NAME
	  
 UNION ALL

SELECT SUBSTR(A.ORDER_OVER_TIME,1,7) AS STAT_MONTH --统计月份
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END AS TYPE_NAME --装修类型
	  ,A.STAGE_ID AS ORDER_STATUS_ID --订单状态ORDER_NUM
	  ,A.STAGE_NAME AS ORDER_STATUS_NAME --订单状态
	  ,0 AS SIGN_ORDER_CNT --签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND A.STAGE_NAME IN('待验收','已完成') THEN ORDER_NUM END) AS OVER_ORDER_CNT --完工数
	  ,0 AS TURN_ORDER_CNT --转单数
      ,0 AS ORDER_CNT --订单数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,0 AS DJC_ORDER_CNT --待进场数
	  ,0 AS JD_ORDER_CNT --建单数
	  ,0 AS ZJD_ORDER_CNT --总部建单数
	  ,0 AS QJD_ORDER_CNT --区域建单数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY
       SUBSTR(A.ORDER_OVER_TIME,1,7)
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END
	  ,A.STAGE_ID		 
	  ,A.STAGE_NAME
	  
 UNION ALL

SELECT SUBSTR(A.CREATE_TIME,1,7) AS STAT_MONTH --统计月份
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END AS TYPE_NAME --装修类型
	  ,A.STAGE_ID AS ORDER_STATUS_ID --订单状态ORDER_NUM
	  ,A.STAGE_NAME AS ORDER_STATUS_NAME --订单状态
	  ,0 AS SIGN_ORDER_CNT --签单数
	  ,0 AS OVER_ORDER_CNT --完工数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE NOT IN(1,2) THEN ORDER_NUM END) AS TURN_ORDER_CNT --转单数
      ,0 AS ORDER_CNT --订单数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,0 AS DJC_ORDER_CNT --待进场数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE IN(2) THEN ORDER_NUM END) AS JD_ORDER_CNT --建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE IN(2) AND A.CREATE_POST_NAME IN('企业管理员','装到家管理员') THEN ORDER_NUM END) AS ZJD_ORDER_CNT --总部建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE IN(2) AND A.CREATE_POST_NAME IN('装到家-运营总监','装到家-销售顾问') THEN ORDER_NUM END) AS QJD_ORDER_CNT --区域建单数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY
       SUBSTR(A.CREATE_TIME,1,7)
      ,CASE WHEN APPT_TYPE_NAME IS NULL OR APPT_TYPE_NAME = '' THEN NULL
	        ELSE CAST(HOME_TYPE_NAME || '-' || APPT_TYPE_NAME AS VARCHAR)
			 END
	  ,A.STAGE_ID		 
	  ,A.STAGE_NAME 
	  ) A
 GROUP BY
       STAT_MONTH --统计月份
      ,TYPE_NAME --装修类型
	  ,ORDER_STATUS_ID --订单状态ORDER_NUM
	  ,ORDER_STATUS_NAME --订单状态
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
