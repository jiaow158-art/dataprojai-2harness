DELETE FROM DM.DM_SJZX_EMP_VISIT_STAT_T
WHERE STAT_MONTH = '${PERIOD_ID_M}'
;
INSERT INTO DM.DM_SJZX_EMP_VISIT_STAT_T(
       STAT_MONTH --统计月份
      ,EMP_NUM --工号
	  ,EMP_NAME --员工姓名
	  ,TEMPL_CODE --模板编码
	  ,VISIT_CNT --访问次数
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,RPT_NAME --报表名称
)
SELECT SUBSTR('${PERIOD_ID_M}',1,7) AS STAT_MONTH --统计月份
      ,A.USERNAME AS EMP_NUM --工号
	  ,A.REALNAME AS EMP_NAME --员工姓名
	  ,A.TEMPL_CODE --模板编码
	  ,COUNT(CASE WHEN B.TYPE IN('0','1','2','3','4','5','6','7','8') THEN 1 END) AS VISIT_CNT --访问次数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.REPORT_NAME AS RPT_NAME --报表名称
  FROM DM.DM_MD_FR_FINE_AUTHORITY_T A
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_RECORD_EXECUTE_T B
    ON B.EMP_NUM = A.USERNAME
   AND A.TEMPL_CODE = B.TEMPL_CODE
   AND SUBSTR(B.BEGIN_TIME,1,7) = '${PERIOD_ID_M}'
   AND B.RPT_TYPE_NAME = '报表'
 INNER JOIN DM.DM_WF_EMP_INFO_T C
    ON A.USERNAME = C.EMP_NUM
   AND C.STATUS IN('0','1','2','3')
 WHERE CASE WHEN COALESCE(C.DEPT_NAME,'SNULL') = '审计法务部审计内控组' THEN 0
 WHEN COALESCE(C.DEPT_NAME,'SNULL') <> '数字赋能平台数据及财经产品部' AND SUBSTR(A.TEMPL_CODE,1,2) <> 'SY' THEN 1
            WHEN COALESCE(C.DEPT_NAME,'SNULL') = '数字赋能平台数据及财经产品部' AND SUBSTR(A.TEMPL_CODE,1,2) = 'SY' THEN 1
			WHEN COALESCE(C.DEPT_NAME,'SNULL') = '数字赋能平台数据及财经产品部' THEN 0
            ELSE 1
             END = 1
   AND A.STAT_MONTH = '${PERIOD_ID_M}'
   AND A.REPORT_FULL_NAME NOT LIKE '%测试%'
 GROUP BY SUBSTR('${PERIOD_ID_M}',1,7) --统计月份
      ,A.USERNAME --工号
	  ,A.REALNAME --员工姓名
	  ,A.TEMPL_CODE --模板编码
	  ,A.REPORT_NAME
	  
 UNION ALL

SELECT DISTINCT
       SUBSTR('${PERIOD_ID_M}',1,7) AS STAT_MONTH --统计月份
      ,B.EMP_NUM --工号
	  ,A.NAME AS EMP_NAME --员工姓名
	  ,CASE WHEN A.TAGNAME = '精益数字看板' THEN 'MPR001'
            WHEN A.TAGNAME = '强商看板' THEN 'MSY005'
            WHEN A.TAGNAME = '数据中心' THEN 'MSY004'
            WHEN A.TAGNAME = '同城报备' THEN 'MSY006'
            WHEN A.TAGNAME = '企微-经营看板' THEN 'MSY001'
            WHEN A.TAGNAME = '瓷砖经营驾驶舱使用人员' THEN 'MSY023'
			 END AS TEMPL_CODE --模板编码
	  ,COALESCE(D.VISIT_CNT,0) AS VISIT_CNT --访问次数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,C.RPT_NAME --报表名称
  FROM DM.DM_WX_DATA_ROLE_T A
 INNER JOIN DM.DM_WF_EMP_INFO_T B
    ON A.USERID = B.EMP_ID
   AND B.STATUS IN('0','1','2','3')
   AND COALESCE(B.DEPT_NAME,'SNULL') <> '数字赋能平台数据及财经产品部'
   AND COALESCE(B.DEPT_NAME,'SNULL') <> '审计法务部审计内控组'
  LEFT JOIN DWRBIT.DWR_OA_RPT_INFO_T C
    ON CASE WHEN A.TAGNAME = '精益数字看板' THEN 'MPR001'
            WHEN A.TAGNAME = '强商看板' THEN 'MSY005'
            WHEN A.TAGNAME = '数据中心' THEN 'MSY004'
            WHEN A.TAGNAME = '同城报备' THEN 'MSY006'
            WHEN A.TAGNAME = '企微-经营看板' THEN 'MSY001'
            WHEN A.TAGNAME = '瓷砖经营驾驶舱使用人员' THEN 'MSY023'
			 END = C.RPT_CODE
   AND C.RQST_TYPE_NAME = '移动端报表'
  LEFT JOIN (SELECT EMP_NUM
                   ,TEMPL_CODE
				   ,SUBSTR(BEGIN_TIME,1,7) AS STAT_MONTH
				   ,COUNT(1) AS VISIT_CNT
               FROM DWIMD.DWI_MD_FR_FINE_RECORD_EXECUTE_T
			  WHERE SUBSTR(BEGIN_TIME,1,7) = SUBSTR('${PERIOD_ID_M}',1,7)
			    AND RPT_TYPE_NAME = '报表'
			  GROUP BY EMP_NUM
                   ,TEMPL_CODE
				   ,SUBSTR(BEGIN_TIME,1,7)
			) D
	ON D.EMP_NUM = B.EMP_NUM
   AND CASE WHEN A.TAGNAME = '精益数字看板' THEN 'MPR001'
            WHEN A.TAGNAME = '强商看板' THEN 'MSY005'
            WHEN A.TAGNAME = '数据中心' THEN 'MSY004'
            WHEN A.TAGNAME = '同城报备' THEN 'MSY006'
            WHEN A.TAGNAME = '企微-经营看板' THEN 'MSY001'
            WHEN A.TAGNAME = '瓷砖经营驾驶舱使用人员' THEN 'MSY023'
			 END = D.TEMPL_CODE
 WHERE CASE WHEN A.TAGNAME = '精益数字看板' THEN 'MPR001'
            WHEN A.TAGNAME = '强商看板' THEN 'MSY005'
            WHEN A.TAGNAME = '数据中心' THEN 'MSY004'
            WHEN A.TAGNAME = '同城报备' THEN 'MSY006'
            WHEN A.TAGNAME = '企微-经营看板' THEN 'MSY001'
            WHEN A.TAGNAME = '瓷砖经营驾驶舱使用人员' THEN 'MSY023'
			 END IS NOT NULL
