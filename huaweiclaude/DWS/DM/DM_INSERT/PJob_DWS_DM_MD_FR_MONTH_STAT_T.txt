DELETE FROM DM.DM_MD_FR_MONTH_STAT_T
 WHERE STAT_MONTH = SUBSTR('${PERIOD_ID_D}',1,7)
;

WITH T1 AS(--存量报表
SELECT '${PERIOD_ID_D}' AS STAT_DATE --统计日期
      ,SUBSTR('${PERIOD_ID_D}',1,7) AS STAT_MONTH --统计月份
      ,COUNT(DISTINCT REPORT_PATH) AS RPT_CNT --报表数
  FROM DM.DM_MD_FR_REPORT_INFO_T
 WHERE SUBSTR(REPORT_PATH,1,3) = 'PC端' 
),T2 AS(--上线报表
SELECT '${PERIOD_ID_D}' AS STAT_DATE --统计日期
      ,SUBSTR('${PERIOD_ID_D}',1,7) AS STAT_MONTH --统计月份
      ,COUNT(DISTINCT A.REPORT_PATH) AS ONLINE_RPT_CNT --上线报表数
  FROM DM.DM_MD_FR_REPORT_INFO_T A
 INNER JOIN (SELECT CONCAT(REPORT_TYPE,REPORT_NUM) AS REPORT_CODE
               FROM UPLOAD.UPLOAD_REPORT_UPDATE_LOG_T
              WHERE UPDATE_TYPE = '上线'
			    AND UPDATE_DATE <= '${PERIOD_ID_D}'
				AND SUBSTR(UPDATE_DATE,1,7) = SUBSTR('${PERIOD_ID_D}',1,7)
			) B
	ON POSITION(B.REPORT_CODE IN A.REPORT_PATH) > 0
 WHERE SUBSTR(A.REPORT_PATH,1,3) = 'PC端'  
),T3 AS(--下线报表
SELECT '${PERIOD_ID_D}' AS STAT_DATE --统计日期
      ,SUBSTR('${PERIOD_ID_D}',1,7) AS STAT_MONTH --统计月份
      ,COUNT(DISTINCT B.REPORT_CODE) AS DOWNLINE_RPT_CNT --下线报表数
  FROM (SELECT CONCAT(REPORT_TYPE,REPORT_NUM) AS REPORT_CODE
          FROM UPLOAD.UPLOAD_REPORT_UPDATE_LOG_T
         WHERE UPDATE_TYPE = '下线'
		   AND UPDATE_DATE <= '${PERIOD_ID_D}'
		   AND SUBSTR(UPDATE_DATE,1,7) = SUBSTR('${PERIOD_ID_D}',1,7)
	    ) B
)
INSERT INTO DM.DM_MD_FR_MONTH_STAT_T
(
       STAT_DATE
      ,STAT_MONTH
	  ,RPT_CNT --报表数
	  ,ONLINE_RPT_CNT --上线报表数
	  ,DOWNLINE_RPT_CNT --下线报表数
	  ,STABLE_RPT_CNT --稳定期报表数
	  ,DW_LAST_UPDATE_DATE
)
SELECT T1.STAT_DATE
      ,T1.STAT_MONTH
	  ,T1.RPT_CNT --报表数
	  ,COALESCE(T2.ONLINE_RPT_CNT,0) AS ONLINE_RPT_CNT --上线报表数
	  ,COALESCE(T3.DOWNLINE_RPT_CNT,0) AS DOWNLINE_RPT_CNT --下线报表数
	  ,COALESCE(T4.RPT_CNT,0) AS STABLE_RPT_CNT --稳定期报表数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T1
  LEFT JOIN T2
    ON T1.STAT_DATE = T2.STAT_DATE
  LEFT JOIN T3
    ON T1.STAT_DATE = T3.STAT_DATE
  LEFT JOIN (SELECT COUNT(DISTINCT REPORT_CODE) AS RPT_CNT
               FROM DM.DM_MD_FRREPORT_LC_STAT_T
              WHERE STAT_MONTH = SUBSTR('${PERIOD_ID_D}',1,7)
			    AND RPT_TYPE = '稳定期'
            ) T4
    ON 1 = 1			
;
