DELETE FROM dm.dm_rpt_demand_forecast_t_2025
WHERE 1=1
and start_month >= 
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
    THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
        ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
         END
         ;
with A as (
select 
cast(t1.calmonth as INTEGER) as start_month,
t2.sales_center_code,
t2.sales_center as sales_center_name,
t2.sales_dep_code,
t2.sales_dep as sales_dep_name,
t2.sales_region_code,
t2.sales_region as sales_region_name,
COALESCE(t3.prod_name1,'其他瓷砖') as prod_name, --品类划分
COALESCE(t3.bigdimension,'其他规格') as dimension,--规格划分
case 
	when channel_desc like '%零售%' then 'GD01'
	when channel_desc like '%工程%' then 'GD03'
	when channel_desc like '%整装%' then 'GD02'
end integrate_channel, --渠道
sum(t1.amount) as zxsyj_dy,--X实际达成业绩单位：万元
sum(t1.area) as zxsmj_dy--X月实际提货面积单位：万方
from DM.DM_RPT_SALE_FORECAST_2023_T t1
join dm.dm_dp_api_sales_group_struct t2 on t1.sales_grp = t2.sales_grp
left join DM.DM_RPT_SALE_FORECAST_T_MID t3 on t1.material = t3.material_num
-- where calmonth = 202506
group by 
t1.calmonth,
t2.sales_center_code,
t2.sales_center,
t2.sales_dep_code,
t2.sales_dep,
t2.sales_region_code,
t2.sales_region,
COALESCE(t3.prod_name1,'其他瓷砖'),
COALESCE(t3.bigdimension,'其他规格'),
integrate_channel
)
,B as 
(
select 
cast(replace(replace(start_month,'01/02','-02'),'-','') as INTEGER) as  start_month,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
dp_cate as prod_name,
dp_dimension as dimension,
integrate_channel,
sum(amount) as yc_amount,
sum(area) as yc_area
From dwrotd.dwr_wm_sales_predict_t
-- where cast(replace(replace(start_month,'01/02','-02'),'-','') as INTEGER) = 202506
group by 
replace(replace(start_month,'01/02','-02'),'-','') ,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
dp_cate,
dp_dimension,
integrate_channel
)
,C as 
(
select 
start_month,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
prod_name,
dimension,
integrate_channel,
zxsyj_dy,
zxsmj_dy,
0 as yc_amount,
0 as yc_area
from A
union all 
select 
start_month,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
prod_name,
dimension,
integrate_channel,
0 as zxsyj_dy,
0 as zxsmj_dy,
yc_amount,
yc_area
from B
)
insert into dm.dm_rpt_demand_forecast_t_2025
select 
start_month,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
prod_name,
dimension,
integrate_channel,
sum(zxsyj_dy) as zxsyj_dy,
sum(zxsmj_dy) as zxsmj_dy,
sum(yc_amount) as yc_amount,
sum(yc_area) as yc_area,
current_timestamp as dw_last_update_date
from C  
where 1=1 
and start_month >= 
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
    THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
        ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
         END
group by 
start_month,
sales_center_code,
sales_center_name,
sales_dep_code,
sales_dep_name,
sales_region_code,
sales_region_name,
prod_name,
dimension,
integrate_channel
