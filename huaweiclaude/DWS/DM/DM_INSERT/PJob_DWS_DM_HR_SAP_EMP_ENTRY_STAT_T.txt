TRUNCATE TABLE DM.DM_HR_SAP_EMP_ENTRY_STAT_T;

WITH T1 AS(
SELECT
       A.STAT_MONTH --统计月份
	  ,SUBSTR(CAST(CONCAT(A.STAT_MONTH,'-01') AS DATE) - INTERVAL '1 MONTH',1,7) AS LAST_MONTH --上个月
	  ,SUBSTR(CAST(CONCAT(A.STAT_MONTH,'-01') AS DATE) - INTERVAL '12 MONTH',1,7) AS LAST_YEAR_MONTH --去年同期月份
	  ,COALESCE(A.LVL1_DEPT_NAME,'东鹏控股') AS LVL1_DEPT_NAME --一级部门名称
      ,A.LVL2_DEPT_NAME --二级部门名称
	  ,A.LVL3_DEPT_NAME --三级部门名称
	  ,A.LVL4_DEPT_NAME --四级部门名称
	  ,A.LVL5_DEPT_NAME --五级部门名称
      ,A.LVL6_DEPT_NAME --六级部门名称
	  ,A.LVL7_DEPT_NAME --七级部门名称
	  ,A.LVL8_DEPT_NAME --八级部门名称
	  ,A.EMP_POST_LVL_NAME AS EMP_LVL_NAME --职级名称
	  ,CASE WHEN A.EMP_POST_LVL_NAME = '员工级' THEN '员工级'
	        WHEN A.EMP_POST_LVL_NAME = '主管级' THEN '主管级'
			WHEN A.EMP_POST_LVL_NAME = '经理级' THEN '经理级'
			WHEN A.EMP_POST_LVL_NAME IN('总监级','总经理级','总裁级','董事长级') THEN '总监及以上'
			 END AS EMP_LVL_TYPE_NAME --职级分类名称
	  ,CASE WHEN A.EMP_POST_LVL_NAME = '员工级' THEN 1
	        WHEN A.EMP_POST_LVL_NAME = '主管级' THEN 2
			WHEN A.EMP_POST_LVL_NAME = '经理级' THEN 3
			WHEN A.EMP_POST_LVL_NAME IN('总监级','总经理级','总裁级','董事长级') THEN 4
			 END AS EMP_LVL_TYPE_RANK --职级分类排序		 
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族') THEN '是'
	        ELSE '否'
			 END AS IS_MGMT_CADRES --是否管理干部
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族','专业职族','营销职族(失效)','行政职族(失效)') THEN 'MP'
            WHEN A.EMP_POST_TYPE_NAME = '操作职族' AND A.EMP_POST_ATTR_NAME = '间接产出' THEN 'O辅'
            WHEN A.EMP_POST_TYPE_NAME = '操作职族' AND A.EMP_POST_ATTR_NAME = '直接产出' THEN 'OO'
            WHEN A.EMP_POST_TYPE_NAME IN('管理职族','专业职族','营销职族(失效)','行政职族(失效)','操作职族')
			 AND (A.EMP_POST_ATTR_NAME = '' OR A.EMP_POST_ATTR_NAME IS NULL) THEN NULL
			ELSE A.EMP_POST_TYPE_NAME
             END AS EMP_POST_TYPE_NAME --职族名称		 
	  ,COUNT(DISTINCT A.EMP_NUM) AS ENTRY_EMP_CNT --入职人数
  FROM DWRHR.DWR_HR_EMP_ENTRY_INFO_F A
 WHERE A.EMP_GROUP_NAME NOT IN('实习生','劳务外包人员','劳务派遣员工','退休人员','离职人员','外部人员','临时工','顾问','残联员工')
 AND (A.LVL5_DEPT_NAME NOT IN ('陈村和园','鹏云','星信厂管理组')  or A.LVL5_DEPT_NAME is null)
 AND (A.LVL3_DEPT_NAME NOT IN ('资本运营部') or A.LVL3_DEPT_NAME is null)
 GROUP BY
       A.STAT_MONTH --统计月份
	  ,SUBSTR(CAST(CONCAT(A.STAT_MONTH,'-01') AS DATE) - INTERVAL '1 MONTH',1,7) --上个月
	  ,SUBSTR(CAST(CONCAT(A.STAT_MONTH,'-01') AS DATE) - INTERVAL '12 MONTH',1,7) --去年同期月份
	  ,COALESCE(A.LVL1_DEPT_NAME,'东鹏控股') --一级部门名称
      ,A.LVL2_DEPT_NAME --二级部门名称
	  ,A.LVL3_DEPT_NAME --三级部门名称
	  ,A.LVL4_DEPT_NAME --四级部门名称
	  ,A.LVL5_DEPT_NAME --五级部门名称
      ,A.LVL6_DEPT_NAME --六级部门名称
	  ,A.LVL7_DEPT_NAME --七级部门名称
	  ,A.LVL8_DEPT_NAME --八级部门名称
	  ,A.EMP_POST_LVL_NAME --职级名称
	  ,CASE WHEN A.EMP_POST_LVL_NAME = '员工级' THEN '员工级'
	        WHEN A.EMP_POST_LVL_NAME = '主管级' THEN '主管级'
			WHEN A.EMP_POST_LVL_NAME = '经理级' THEN '经理级'
			WHEN A.EMP_POST_LVL_NAME IN('总监级','总经理级','总裁级','董事长级') THEN '总监及以上'
			 END --职级分类名称
	  ,CASE WHEN A.EMP_POST_LVL_NAME = '员工级' THEN 1
	        WHEN A.EMP_POST_LVL_NAME = '主管级' THEN 2
			WHEN A.EMP_POST_LVL_NAME = '经理级' THEN 3
			WHEN A.EMP_POST_LVL_NAME IN('总监级','总经理级','总裁级','董事长级') THEN 4
			 END --职级分类排序		 
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族') THEN '是'
	        ELSE '否'
			 END --是否管理干部
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族','专业职族','营销职族(失效)','行政职族(失效)') THEN 'MP'
            WHEN A.EMP_POST_TYPE_NAME = '操作职族' AND A.EMP_POST_ATTR_NAME = '间接产出' THEN 'O辅'
            WHEN A.EMP_POST_TYPE_NAME = '操作职族' AND A.EMP_POST_ATTR_NAME = '直接产出' THEN 'OO'
            WHEN A.EMP_POST_TYPE_NAME IN('管理职族','专业职族','营销职族(失效)','行政职族(失效)','操作职族')
			 AND (A.EMP_POST_ATTR_NAME = '' OR A.EMP_POST_ATTR_NAME IS NULL) THEN NULL
			ELSE A.EMP_POST_TYPE_NAME
             END --职族名称		 
)
INSERT INTO DM.DM_HR_SAP_EMP_ENTRY_STAT_T
(
       STAT_MONTH --统计月份
	  ,LAST_MONTH --上个月
	  ,LAST_YEAR_MONTH --去年同期月份
	  ,LVL1_DEPT_NAME --一级部门名称
      ,LVL2_DEPT_NAME --二级部门名称
	  ,LVL3_DEPT_NAME --三级部门名称
	  ,LVL4_DEPT_NAME --四级部门名称
	  ,LVL5_DEPT_NAME --五级部门名称
      ,LVL6_DEPT_NAME --六级部门名称
	  ,LVL7_DEPT_NAME --七级部门名称
	  ,LVL8_DEPT_NAME --八级部门名称
	  ,EMP_LVL_NAME --职级名称
	  ,EMP_LVL_TYPE_NAME --职级分类名称
	  ,EMP_LVL_TYPE_RANK --职级分类排序		 
	  ,IS_MGMT_CADRES --是否管理干部
	  ,ENTRY_EMP_CNT --本月入职人数
	  ,LM_ENTRY_EMP_CNT --上月入职人数
	  ,LY_ENTRY_EMP_CNT --去年同期入职人数
      ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,EMP_POST_TYPE_NAME --职族名称
	  ,DEPT_JG_NAME --部门结构名称
)
SELECT A.STAT_MONTH --统计月份
	  ,A.LAST_MONTH --上个月
	  ,A.LAST_YEAR_MONTH --去年同期月份
	  ,A.LVL1_DEPT_NAME --一级部门名称
      ,A.LVL2_DEPT_NAME --二级部门名称
	  ,A.LVL3_DEPT_NAME --三级部门名称
	  ,A.LVL4_DEPT_NAME --四级部门名称
	  ,A.LVL5_DEPT_NAME --五级部门名称
      ,A.LVL6_DEPT_NAME --六级部门名称
	  ,A.LVL7_DEPT_NAME --七级部门名称
	  ,A.LVL8_DEPT_NAME --八级部门名称
	  ,A.EMP_LVL_NAME --职级名称
	  ,A.EMP_LVL_TYPE_NAME --职级分类名称
	  ,A.EMP_LVL_TYPE_RANK --职级分类排序		 
	  ,A.IS_MGMT_CADRES --是否管理干部
	  ,A.ENTRY_EMP_CNT --本月入职人数
	  ,COALESCE(B.ENTRY_EMP_CNT,0) AS LM_ENTRY_EMP_CNT --上月入职人数
	  ,COALESCE(C.ENTRY_EMP_CNT,0) AS LY_ENTRY_EMP_CNT --去年同期入职人数
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.EMP_POST_TYPE_NAME --职族名称
	  ,CASE WHEN COALESCE(A.LVL8_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1||A.LVL8_DEPT_NAME1
	        WHEN COALESCE(A.LVL7_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1||A.LVL7_DEPT_NAME1
	        WHEN COALESCE(A.LVL6_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1||A.LVL6_DEPT_NAME1
			WHEN COALESCE(A.LVL5_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1||A.LVL5_DEPT_NAME1
			WHEN COALESCE(A.LVL4_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1||A.LVL4_DEPT_NAME1
			WHEN COALESCE(A.LVL3_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1||A.LVL3_DEPT_NAME1
			WHEN COALESCE(A.LVL2_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME||A.LVL2_DEPT_NAME1
			WHEN COALESCE(A.LVL1_DEPT_NAME,'SNULL') <> 'SNULL'
            THEN A.LVL1_DEPT_NAME
	         END AS DEPT_JG_NAME --部门结构名称
  FROM(SELECT A.*
             ,CASE WHEN COALESCE(A.LVL8_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL8_DEPT_NAME ELSE '' END AS LVL8_DEPT_NAME1
			 ,CASE WHEN COALESCE(A.LVL7_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL7_DEPT_NAME ELSE '' END AS LVL7_DEPT_NAME1
			 ,CASE WHEN COALESCE(A.LVL6_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL6_DEPT_NAME ELSE '' END AS LVL6_DEPT_NAME1
			 ,CASE WHEN COALESCE(A.LVL5_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL5_DEPT_NAME ELSE '' END AS LVL5_DEPT_NAME1
			 ,CASE WHEN COALESCE(A.LVL4_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL4_DEPT_NAME ELSE '' END AS LVL4_DEPT_NAME1
             ,CASE WHEN COALESCE(A.LVL3_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL3_DEPT_NAME ELSE '' END AS LVL3_DEPT_NAME1
			 ,CASE WHEN COALESCE(A.LVL2_DEPT_NAME,'SNULL') <> 'SNULL' THEN '-'||A.LVL2_DEPT_NAME ELSE '' END AS LVL2_DEPT_NAME1  
         FROM T1 A) A
  LEFT JOIN T1 B
    ON A.LAST_MONTH = B.STAT_MONTH
   AND A.EMP_LVL_NAME = B.EMP_LVL_NAME
   AND COALESCE(A.LVL1_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL1_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL2_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL2_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL3_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL3_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL4_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL4_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL5_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL5_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL6_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL6_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL7_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL7_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL8_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(B.LVL8_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.EMP_POST_TYPE_NAME,'CRATE_BY_HWB') = COALESCE(B.EMP_POST_TYPE_NAME,'CRATE_BY_HWB')
  LEFT JOIN T1 C
    ON A.LAST_YEAR_MONTH = C.STAT_MONTH
   AND A.EMP_LVL_NAME = C.EMP_LVL_NAME 
   AND COALESCE(A.LVL1_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL1_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL2_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL2_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL3_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL3_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL4_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL4_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL5_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL5_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL6_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL6_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL7_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL7_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.LVL8_DEPT_NAME,'CRATE_BY_HWB') = COALESCE(C.LVL8_DEPT_NAME,'CRATE_BY_HWB')
   AND COALESCE(A.EMP_POST_TYPE_NAME,'CRATE_BY_HWB') = COALESCE(C.EMP_POST_TYPE_NAME,'CRATE_BY_HWB')
   ;
