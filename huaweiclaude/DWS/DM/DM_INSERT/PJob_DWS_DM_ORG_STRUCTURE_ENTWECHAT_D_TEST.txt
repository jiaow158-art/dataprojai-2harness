DELETE FROM DM.DM_ORG_STRUCTURE_ENTWECHAT_D_TEST;
INSERT INTO DM.DM_ORG_STRUCTURE_ENTWECHAT_D_TEST
with recursive data1 as (
select 
 a.objid ::varchar(250) as objid
,a.SHORT ::varchar(250) as SHORT_OBJID_NAME
,a.stext ::varchar(250) as LONG_OBJID_NAME
,b.sobid ::varchar(250) AS PARENT_OBJID
,'0' as FLAG 
,b.AEDTM
,b.BEGDA 
from dm.dm_sdi_hrp1000_1001_t_test as a 
left join dm.dm_sdi_hrp1001_1001_t_test as b on a.objid = b.objid and b.plvar = '01' AND b.otype = 'O' and b.endda = '99991231'
AND b.istat = '1' AND b.rsign = 'A' AND b.relat = '002' AND b.sclas = 'O' 
where a.plvar = '01' AND a.otype = 'O' AND a.istat = '1' and a.LANGU = '1' and a.objid = '00000201' and a.endda = '99991231'
union 
select 
 c.objid ::varchar(250) as objid
,c.SHORT_OBJID_NAME ::varchar(250) as SHORT_OBJID_NAME
,c.LONG_OBJID_NAME ::varchar(250) as LONG_OBJID_NAME
,c.PARENT_OBJID ::varchar(250) as PARENT_OBJID
,'0' as FLAG 
,c.AEDTM
,c.BEGDA 
from (
select 
 a.objid ::varchar(250) as objid
,a.SHORT ::varchar(250) as SHORT_OBJID_NAME
,a.stext ::varchar(250) as LONG_OBJID_NAME
,b.SOBID ::varchar(250) AS PARENT_OBJID
,b.AEDTM
,b.BEGDA 
,'0' as FLAG 
from dm.dm_sdi_hrp1000_1001_t_test as a 
left join dm.dm_sdi_hrp1001_1001_t_test as b on a.objid = b.objid and b.plvar = '01' AND b.otype = 'O' and b.endda = '99991231'
AND b.istat = '1' AND b.rsign = 'A' AND b.relat = '002' AND b.sclas = 'O' 
where a.plvar = '01' AND a.otype = 'O' AND a.istat = '1' and a.LANGU = '1' and a.endda = '99991231' 
) as c 
join data1 as d on c.PARENT_OBJID = d.objid
)
,
 data2 as ( 
select 
objid
,SHORT_OBJID_NAME
,LONG_OBJID_NAME
,PARENT_OBJID
,FLAG
,AEDTM
,BEGDA 
from data1  
union all
select distinct 
 b.objid 
,b.SHORT as SHORT_OBJID_NAME
,b.stext as LONG_OBJID_NAME
,a.SOBID AS PARENT_OBJID
,'1' as flag -- 删除
,a.AEDTM
,a.BEGDA
from dm.dm_sdi_hrp1000_1001_t_test as b 
inner join dm.dm_sdi_hrp1001_1001_t_test as a 
on a.OBJID = b.objid and a.plvar = '01' AND a.otype = 'O'
AND a.istat = '1'   AND a.sclas = 'O' AND a.relat = '002' AND a.rsign = 'A'
where b.plvar = '01' AND b.otype = 'O' AND b.istat = '1' and b.LANGU = '1' 
and b.objid not in 
(select objid from dm.dm_sdi_hrp1000_1001_t_test where endda = '99991231' and plvar = '01' AND otype = 'O' AND istat = '1' and LANGU = '1') 
)
,
data3 as (
select 
  t.objid  as org_code,             -- 组织代码
  t.SHORT_OBJID_NAME as org_short_name_cn,   -- 组织短名称
  t.LONG_OBJID_NAME  as org_name_cn,   -- 组织长名称
  t.PARENT_OBJID     as up_org_code,   -- 上级组织
  t.flag             as data_flag,   -- 数据标识
  row_number() over(partition by t.objid,t.flag order by t.AEDTM desc) rn
  ,a.SOBID
  ,coalesce (b.sobid,d.sobid) as sobid_a
  ,t.AEDTM
  ,t.BEGDA
from data2 t 
left join dm.dm_sdi_hrp1001_1001_t_test a on a.otype = 'O' AND a.subty = 'B012' AND a.objid = t.objid and a.endda = '99991231' 
left join dm.dm_sdi_hrp1001_1001_t_test b on b.otype = 'S' AND b.subty = 'A008' and a.sobid = b.objid and b.endda = '99991231'
left join dm.dm_sdi_hrp1001_1001_t_test c on c.otype = 'O' AND c.subty = 'B012' AND c.objid = t.PARENT_OBJID and c.endda = '99991231' 
left join dm.dm_sdi_hrp1001_1001_t_test d on d.otype = 'S' AND d.subty = 'A008' and c.sobid = d.objid and d.endda = '99991231' 
)
select t.org_code ,            -- 组织代码
       t.org_short_name_cn ,  -- 组织短名称
	   t.org_name_cn ,  -- 组织长名称
	   t.up_org_code ,  -- 上级组织
	   t.data_flag ,  -- 数据标识
	   current_timestamp() as dw_last_update_date -- 数据同步时间
	   ,t.sobid_a as dept_header
	   ,t1.nachn as dept_header_n
	   ,t.AEDTM as update_date
	   ,t.BEGDA as create_date
from data3 as t
left join dm.dm_rpt_zhrt_oa_pernr_t_test t1 on t1.pernr = t.sobid_a
where t.rn = 1 
order by org_code;
