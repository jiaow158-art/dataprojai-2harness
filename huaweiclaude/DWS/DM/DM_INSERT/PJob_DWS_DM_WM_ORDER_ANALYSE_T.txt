TRUNCATE TABLE DM.DM_WM_ORDER_ANALYSE_T;

WITH T1 AS(
SELECT DISTINCT
       BATCH_START
      ,BELONG_AREA_CODE
	  ,BELONG_AREA_NAME
  FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D
 WHERE BELONG_AREA_CODE IS NOT NULL 
),T2 AS(
SELECT DISTINCT
       BELONG_AREA_CODE
	  ,BELONG_AREA_NAME
  FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D
 WHERE BELONG_AREA_CODE IS NOT NULL
),T3 AS(SELECT 
        A.MATERIAL_NUM
                  --,COALESCE(COALESCE(COALESCE(C.BELONG_AREA_CODE,D.BELONG_AREA_CODE),A.ZCQBM),'GD') AS BELONG_AREA_CODE --产区编码
       ,CASE WHEN C.BELONG_AREA_CODE IS NOT NULL THEN C.BELONG_AREA_CODE
			 WHEN D.BELONG_AREA_CODE IS NOT NULL THEN D.BELONG_AREA_CODE
			 ELSE 'GD'
			  END AS BELONG_AREA_CODE --产区编码
      ,SUBSTR(A.VOUCHER_POST_DATE,1,4) AS STAT_YEAR
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '01' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '01' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_1M --1月份已出库数量数量
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '02' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '02' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_2M --2月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '03' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '03' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_3M --3月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '04' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '04' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_4M --4月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '05' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '05' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_5M --5月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '06' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '06' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_6M --6月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '07' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '07' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_7M --7月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '08' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '08' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_8M --8月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '09' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '09' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_9M --9月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '10' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '10' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_10M --10月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '11' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '11' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_11M --11月份已出库数量数量				 
      ,SUM(CASE WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '12' AND A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN SUBSTR(VOUCHER_POST_DATE,6,2) = '12' AND A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY_12M --12月份已出库数量数量				 				 
      ,SUM(CASE WHEN A.DR_CR_ID = 'H_SHKZG' THEN A.QUANTITY
		        WHEN A.DR_CR_ID = 'S_SHKZG' THEN 0 - A.QUANTITY
		        ELSE 0
			     END) AS OUTWAREHOUSE_QUANTITY --已出库数量数量	
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
 INNER JOIN UPLOAD.UPLOAD_WM_DB_CHANNEL_PROD_T B
	ON A.MATERIAL_NUM = B.MATERIAL_NUM
  			  LEFT JOIN T1 C
			    ON SUBSTR(A.BATCH_NUM,1,3) = C.BATCH_START
			   AND C.BATCH_START IS NOT NULL	
			  LEFT JOIN T1 D
                ON SUBSTR(A.BATCH_NUM,1,2) = D.BATCH_START
               AND D.BATCH_START IS NOT NULL	
 WHERE A.MOVE_TYPE IN('601','602','653','654','927','931','932','933','951','952')
   AND SUBSTR(A.VOUCHER_POST_DATE,1,4) >= '2022'
   AND A.FACTORY_WERKS_CODE NOT LIKE '%80'
   AND A.INV_LOCATION IN('TA','SD02','SA','BI','QA','PA','BZ','BA','DZ','CD','CA','BE','BE01','BE03','BE02','BE05')			 
   GROUP BY A.MATERIAL_NUM
                  --,COALESCE(COALESCE(COALESCE(C.BELONG_AREA_CODE,D.BELONG_AREA_CODE),A.ZCQBM),'GD') AS BELONG_AREA_CODE --产区编码
           ,CASE WHEN C.BELONG_AREA_CODE IS NOT NULL THEN C.BELONG_AREA_CODE
				 WHEN D.BELONG_AREA_CODE IS NOT NULL THEN D.BELONG_AREA_CODE
				 ELSE 'GD'
				  END
           ,SUBSTR(A.VOUCHER_POST_DATE,1,4)
),T4 AS(
            SELECT MATNR AS MATERIAL_NUM
                  --,COALESCE(COALESCE(COALESCE(C.BELONG_AREA_CODE,D.BELONG_AREA_CODE),A.ZCQBM),'GD') AS BELONG_AREA_CODE --产区编码
                  ,CASE WHEN C.BELONG_AREA_CODE IS NOT NULL THEN C.BELONG_AREA_CODE
				        WHEN D.BELONG_AREA_CODE IS NOT NULL THEN D.BELONG_AREA_CODE
						WHEN A.CHARG IS NOT NULL THEN 'GD'
						WHEN A.ZCQBM IS NOT NULL THEN A.ZCQBM
						ELSE 'GD'
						 END AS BELONG_AREA_CODE --产区编码
				  ,SUBSTR(AUDAT,1,4) AS STAT_YEAR
				  ,SUM(KWMENG) AS ORDER_QUANTITY --订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '01' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_1M --1月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '02' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_2M --2月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '03' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_3M --3月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '04' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_4M --4月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '05' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_5M --5月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '06' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_6M --6月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '07' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_7M --7月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '08' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_8M --8月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '09' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_9M --9月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '10' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_10M --10月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '11' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_11M --11月份订单数量
				  ,SUM(CASE WHEN SUBStR(AUDAT,5,2) = '12' THEN KWMENG ELSE 0 END) AS ORDER_QUANTITY_12M --12月份订单数量				  
				  ,SUM(WQRSL) AS UNCONFIRMED_QUANTITY --未交付数量
              FROM DM.DM_OTD_SALES_ORDER_DET_T A
			  INNER JOIN UPLOAD.UPLOAD_WM_DB_CHANNEL_PROD_T B
			    ON A.MATNR = B.MATERIAL_NUM
			  LEFT JOIN T1 C
			    ON SUBSTR(A.CHARG,1,3) = C.BATCH_START
			   AND C.BATCH_START IS NOT NULL	
			  LEFT JOIN T1 D
                ON SUBSTR(A.CHARG,1,2) = D.BATCH_START
               AND D.BATCH_START IS NOT NULL
			 WHERE SUBSTR(AUDAT,1,4) >= '2022'
			   AND (ABGRU = '' OR ABGRU IS NULL)
               AND A.WERKS NOT LIKE '%80'
               AND A.LGORT IN('TA','SD02','SA','BI','QA','PA','BZ','BA','DZ','CD','CA','BE','BE01','BE03','BE02','BE05')
			 GROUP BY 
			       A.MATNR
			      ,CASE WHEN C.BELONG_AREA_CODE IS NOT NULL THEN C.BELONG_AREA_CODE
				        WHEN D.BELONG_AREA_CODE IS NOT NULL THEN D.BELONG_AREA_CODE
						WHEN A.CHARG IS NOT NULL THEN 'GD'
						WHEN A.ZCQBM IS NOT NULL THEN A.ZCQBM
						ELSE 'GD'
						 END
				  ,SUBSTR(AUDAT,1,4) 
           )
,T5 AS(
SELECT DISTINCT YEAR_ID AS STAT_YEAR
  FROM DM.DM_DIM_DATE_D 
 WHERE DAY_ID >= '2022-01-01'
   AND DAY_DT <= SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,10)
)

SELECT  A.CHANNEL_NAME
       ,A.ZCD_NAME
       ,A.MATERIAL_TYPE
       ,A.SERIES_NAME
       ,A.MATERIAL_NUM
       ,A.MATERIAL_NAME
       ,A.FORM_NAME
       ,A.DIMENSION
       ,A.UNIT_AREA
       ,A.MARKET_MONTH
       ,A.PRODUCT_LEVEL_CODE
       ,A.PRODUCT_POST
	   ,D.BELONG_AREA_CODE --产区编码
	   ,D.BELONG_AREA_NAME --产区名称
	   ,G.STAT_YEAR AS STAT_ORDER_YEAR --订单年份
	   ,B.ORDER_QUANTITY --订单数量
	   ,B.ORDER_QUANTITY_1M --1月份订单数量
	   ,B.ORDER_QUANTITY_2M --2月份订单数量
	   ,B.ORDER_QUANTITY_3M --3月份订单数量
	   ,B.ORDER_QUANTITY_4M --4月份订单数量
	   ,B.ORDER_QUANTITY_5M --5月份订单数量
	   ,B.ORDER_QUANTITY_6M --6月份订单数量
	   ,B.ORDER_QUANTITY_7M --7月份订单数量

       ,G.STAT_YEAR AS STAT_ORDER_YEAR --出库年份
	   ,F.OUTWAREHOUSE_QUANTITY --已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_1M --1月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_2M --2月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_3M --3月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_4M --4月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_5M --5月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_6M --6月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_7M --7月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_8M --8月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_9M --9月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_10M --10月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_11M --11月份已出库数量
	   ,F.OUTWAREHOUSE_QUANTITY_12M --12月份已出库数量	   
	   ,C.AVAILABLE_INVENTORY_QUANTITY --可用库存
	   ,C.RESERVE_INVENTORY_QUANTITY --留货数
	   ,C.PREPAR_INVENTORY_QUANTITY --备货数
	   ,C.PROJECT_INVENTORY_QUANTITY --项目库存
	   ,C.TOTAL_INVENTORY_QUANTITY --库存数
	   ,CASE WHEN E.PC_QUANTITY < 0 THEN 0
 	         ELSE E.PC_QUANTITY
			  END AS PRODUCT_QUANTITY --排产数
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
       ,CASE WHEN E.AQ_PC_QUANTITY < 0 THEN 0
 	         ELSE E.AQ_PC_QUANTITY
			  END AS AQ_PC_QUANTITY
	   ,CASE WHEN E.FAQ_PC_QUANTITY < 0 THEN 0
 	         ELSE E.FAQ_PC_QUANTITY
			  END AS FAQ_PC_QUANTITY			  
  FROM UPLOAD.UPLOAD_WM_DB_CHANNEL_PROD_T A
  LEFT JOIN T2 D
    ON 1 = 1
  LEFT JOIN T5 G
    ON 1 = 1  
  LEFT JOIN T4 B
    ON A.MATERIAL_NUM = B.MATERIAL_NUM
   AND D.BELONG_AREA_CODE = B.BELONG_AREA_CODE
   AND G.STAT_YEAR = B.STAT_YEAR
  LEFT JOIN(SELECT A.MATERIAL_NUM
                  ,COALESCE(COALESCE(C.BELONG_AREA_CODE,D.BELONG_AREA_CODE),'GD') AS BELONG_AREA_CODE --产区编码
                  ,SUM(A.AVAILABLE_INVENTORY_QUANTITY) AS AVAILABLE_INVENTORY_QUANTITY --可用库存
				  ,SUM(A.RESERVE_INVENTORY_QUANTITY) AS RESERVE_INVENTORY_QUANTITY --留货数
				  ,SUM(A.PREPAR_INVENTORY_QUANTITY) AS PREPAR_INVENTORY_QUANTITY --备货数
				  ,SUM(A.PROJECT_INVENTORY_QUANTITY) AS PROJECT_INVENTORY_QUANTITY --项目库存
				  ,SUM(COALESCE(A.AVAILABLE_INVENTORY_QUANTITY,0) + COALESCE(A.RESERVE_INVENTORY_QUANTITY,0)
				      + COALESCE(A.PREPAR_INVENTORY_QUANTITY,0) + COALESCE(A.PROJECT_INVENTORY_QUANTITY,0)
					  ) AS TOTAL_INVENTORY_QUANTITY --库存数
              FROM DM.DM_WM_ALL_TYPE_STOCK_T A
			 INNER JOIN UPLOAD.UPLOAD_WM_DB_CHANNEL_PROD_T B
			    ON A.MATERIAL_NUM = B.MATERIAL_NUM
			  LEFT JOIN T1 C
			    ON SUBSTR(A.BATCH_NUM,1,3) = C.BATCH_START
			   AND C.BATCH_START IS NOT NULL	
			  LEFT JOIN T1 D
                ON SUBSTR(A.BATCH_NUM,1,2) = D.BATCH_START
               AND D.BATCH_START IS NOT NULL
			 WHERE A.FACTORY_WERKS_CODE NOT LIKE '%80'
               AND A.INV_LOCATION IN('TA','SD02','SA','BI','QA','PA','BZ','BA','DZ','CD','CA','BE','BE01','BE03','BE02','BE05')	 
			 GROUP BY
                   A.MATERIAL_NUM
                  ,COALESCE(COALESCE(C.BELONG_AREA_CODE,D.BELONG_AREA_CODE),'GD')	 
           ) C
	ON A.MATERIAL_NUM = C.MATERIAL_NUM
   AND D.BELONG_AREA_CODE = C.BELONG_AREA_CODE
  LEFT JOIN(SELECT MATERIAL_NUM
                  ,PRODUCT_AREA_CODE
				  ,SUM(COALESCE(QTY,0) - COALESCE(COMPLETE_QTY,0) - COALESCE(CANCEL_QTY,0) - COALESCE(CLOSE_QTY,0)) AS PC_QUANTITY
              --计划数量-完成数量-取消数量-关闭数量
			      ,SUM(CASE WHEN ORDER_TYPE = '安全库存' THEN COALESCE(QTY,0) - COALESCE(COMPLETE_QTY,0) - COALESCE(CANCEL_QTY,0) - COALESCE(CLOSE_QTY,0)
				            ELSE 0
							 END
					  ) AS AQ_PC_QUANTITY
				  ,SUM(CASE WHEN ORDER_TYPE <> '安全库存' OR ORDER_TYPE IS NULL THEN COALESCE(QTY,0) - COALESCE(COMPLETE_QTY,0) - COALESCE(CANCEL_QTY,0) - COALESCE(CLOSE_QTY,0)
				            ELSE 0
							 END
					  ) AS FAQ_PC_QUANTITY	  
              FROM DWROTD.DWR_PS_APS_PRO_ORDER_F
			 WHERE ORDER_STATE IN(0,10,24,26,28)
			   AND IS_PHANTOM = '0'
             GROUP BY
			       MATERIAL_NUM
                  ,PRODUCT_AREA_CODE
           ) E
	ON A.MATERIAL_NUM = E.MATERIAL_NUM
   AND D.BELONG_AREA_CODE = E.PRODUCT_AREA_CODE	 
  LEFT JOIN T3 F
    ON A.MATERIAL_NUM = F.MATERIAL_NUM
   AND D.BELONG_AREA_CODE = F.BELONG_AREA_CODE
   AND G.STAT_YEAR = F.STAT_YEAR
;
