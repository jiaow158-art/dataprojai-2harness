TRUNCATE TABLE DM.DM_CS_ZDJ_CITY_ORDER_STAT_T;

INSERT INTO DM.DM_CS_ZDJ_CITY_ORDER_STAT_T
SELECT CITY_NAME --城市
      ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS ORDER_CNT --累计引流单数
	  ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) = TO_CHAR(CURRENT_DATE - 1,'yyyy-MM-dd')
						   THEN ORDER_NUM END) AS LD_ORDER_CNT --昨日新增单数
      ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						    AND A.STAGE_NAME NOT IN('待派单','待指派')
						   THEN ORDER_NUM END) AS PD_ORDER_CNT --派单单数
      ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						    AND A.STAGE_NAME IN('合同取消')
						   THEN ORDER_NUM END) AS QX_ORDER_CNT --派单取消单数
      ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						    AND A.STAGE_NAME IN('待签约','待开工','施工中','待验收','已完成')
						   THEN ORDER_NUM END) AS SM_ORDER_CNT --派单上门数
	  ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						    AND A.CNTC_CODE IS NOT NULL
						   THEN ORDER_NUM END) AS SIGN_ORDER_CNT --签约单数
	  ,COUNT(CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
				   AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
				   AND A.CNTC_CODE IS NOT NULL
				  THEN A.SALE_AMT
				  ELSE 0
				   END) AS SIGN_SALE_AMT --签约总额
	  ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) * 150 AS YL_FEE --引流费用			   
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单')
   AND CITY_NAME IN('成都市','西安市','上海市','郑州市','福州市'
                   ,'南昌市','温州市','长沙市','沈阳市','烟台市')
 GROUP BY CITY_NAME
