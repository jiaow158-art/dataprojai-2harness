TRUNCATE TABLE DM.DM_RPT_DEMAND_FORECAST_TARGET_T;
INSERT INTO DM.DM_RPT_DEMAND_FORECAST_TARGET_T
with tmp as 
 (
select  a."CALMONTH" as stat_month ,
        e.lev4_code as sale_grp_name1,
        e.lev5_code as sub_grp_name1, --子销售组名称
        a."ZPLMS" as prod_name3,
        a."ZGG" as dimension,
        a."ZZHQDMS" as  channel_name,
        a."ZXSMJ" as  target_area  -- 目标提货面积
from DM.DM_RPT_BIC_AZSSD_O0282_T a
left join DM.DM_RPT_SALE_GRP_T e
on a."SALES_GRP"  = e.SALE_GRP
union ALL           --目标数据由dw迁移到填报报表
select aa.stat_month,
	ee.lev4_code as sale_grp_name1,
    ee.lev5_code as sub_grp_name1, --子销售组名称
	aa.prod_level3_name as prod_name3,
	aa.dimension,
	aa.prod_channel_name as channel_name,
	aa.sales_area as  target_area
from upload.upload_tax_ceramic_profit_t  aa
left join DM.DM_RPT_SALE_GRP_T ee
on aa.sale_grp_name = ee.SALE_GRP
where aa.stat_month >= 202301 
),
t1 as (
select case when prod_name3 in('原石','玻化砖') then '抛光砖'
            when prod_name3 in('抛釉砖','空气净化砖','通体大理石','新材','ART+','DPI','抛釉砖(不含空气净化砖)') 
			  or (prod_name3 = '世界印象' and dimension = '900X900')
			then '抛釉砖'
            when prod_name3 in('715岩板','918岩板','超大板','大板') 
              or (prod_name3 = '世界印象' and dimension in('1200X1200','1500X750','1800X900','2400X1800','2400X800','2600X800','2800X1200','4800X2400')) then '大板'
            when prod_name3 = '瓷片' then '瓷片'
            when prod_name3 = '精瓷中板' then '中板'
            when prod_name3 = '仿古砖'
			  or (prod_name3 = '世界印象' and dimension not in('1200X1200','1500X750','1800X900','2400X1800','2400X800','2600X800','2800X1200','4800X2400','900X900')) then '仿古砖'
            when prod_name3 = '景观砖' then '景观砖'
            when prod_name3 in('其他瓷砖','其他品类') then '其他品类'
             end as prod_name
       ,case when sale_grp_name1 = '工程市场中心' then '战略工程中心'
	         when sale_grp_name1 in('直营经营管理板块','区域运营中心') then sub_grp_name1
			  end as sub_grp_name --运营中心
	   ,* 
from tmp
where (sale_grp_name1 in('直营经营管理板块','区域运营中心') or ((sale_grp_name1 <> '幕墙工程中心' or sale_grp_name1 is null) and sale_grp_name1 = '工程市场中心' and (sub_grp_name1 is null or sub_grp_name1 <> 'Z031')))
and prod_name3 in('原石','玻化砖','抛釉砖','空气净化砖','通体大理石','新材','ART+','DPI','715岩板','918岩板','超大板'
                 ,'世界印象','瓷片','精瓷中板','仿古砖','景观砖','其他瓷砖','其他品类','抛釉砖(不含空气净化砖)','大板'
				 )
and stat_month >= '202204'	
)
select t1.stat_month,
       t1.prod_name,
       case when t1.prod_name = '抛光砖' and t1.dimension in('600X600','800X800') then t1.dimension
            when t1.prod_name = '抛光砖' then '其他规格'
			when t1.prod_name = '抛釉砖' and t1.dimension in('600X600','800X800','1200X600') then t1.dimension
			when t1.prod_name = '抛釉砖' then '其他规格'
			when t1.prod_name = '大板' and t1.dimension in('1500X750','2600X800','1800X900') then t1.dimension
			when t1.prod_name = '大板' then '其他规格'
			when t1.prod_name = '瓷片' and t1.dimension in('300X300','600X300','800X400') then t1.dimension
			when t1.prod_name = '瓷片' then '其他规格'
			when t1.prod_name = '中板' and t1.dimension in('600X300','800X400') then t1.dimension
			when t1.prod_name = '中板' then '其他规格'
			when t1.prod_name = '仿古砖' and t1.dimension in('600X298','600X600','800X800','1200X600') then t1.dimension
			when t1.prod_name = '仿古砖' then '其他规格'
			when t1.prod_name = '景观砖' and t1.dimension in('600X600','1200X600') then t1.dimension
			when t1.prod_name = '景观砖' then '其他规格'
			when t1.prod_name = '其他品类' then '其他规格'
			 end as dimension,
       case when t1.sub_grp_name = '佛山子公司' or t1.sub_grp_name = '广州子公司' then '广佛子公司' else t1.sub_grp_name end sub_grp_name,
	   t1.CHANNEL_NAME,
	   sum(target_area) as target_area,
	   current_timestamp() as dw_last_update_date
	   from t1
group by case when t1.prod_name = '抛光砖' and t1.dimension in('600X600','800X800') then t1.dimension
            when t1.prod_name = '抛光砖' then '其他规格'
			when t1.prod_name = '抛釉砖' and t1.dimension in('600X600','800X800','1200X600') then t1.dimension
			when t1.prod_name = '抛釉砖' then '其他规格'
			when t1.prod_name = '大板' and t1.dimension in('1500X750','2600X800','1800X900') then t1.dimension
			when t1.prod_name = '大板' then '其他规格'
			when t1.prod_name = '瓷片' and t1.dimension in('300X300','600X300','800X400') then t1.dimension
			when t1.prod_name = '瓷片' then '其他规格'
			when t1.prod_name = '中板' and t1.dimension in('600X300','800X400') then t1.dimension
			when t1.prod_name = '中板' then '其他规格'
			when t1.prod_name = '仿古砖' and t1.dimension in('600X298','600X600','800X800','1200X600') then t1.dimension
			when t1.prod_name = '仿古砖' then '其他规格'
			when t1.prod_name = '景观砖' and t1.dimension in('600X600','1200X600') then t1.dimension
			when t1.prod_name = '景观砖' then '其他规格'
			when t1.prod_name = '其他品类' then '其他规格'
			 end 
       ,t1.prod_name
       ,case when t1.sub_grp_name = '佛山子公司' or t1.sub_grp_name = '广州子公司' then '广佛子公司' else t1.sub_grp_name end 
	   ,t1.stat_month
	   ,t1.CHANNEL_NAME
