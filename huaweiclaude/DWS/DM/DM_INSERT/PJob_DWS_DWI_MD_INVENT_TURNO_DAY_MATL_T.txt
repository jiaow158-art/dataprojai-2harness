DELETE FROM  DWIMD.DWI_MD_INVENT_TURNO_DAY_MATL_T
WHERE STATISTICAL_MONTH >=
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '04'
     THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
         ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
          END;
INSERT INTO  DWIMD.DWI_MD_INVENT_TURNO_DAY_MATL_T
--库存成本
WITH T1 AS(
SELECT 
        A.STAT_MONTH AS STAT_MONTH --月份
       ,A.material_num AS material_num
       ,SUM(A.YM_ZYKCJE) AS YM_ZYKCJE --月末库存金额-自有库存
       ,SUM(A.YM_ZYKCSL) AS YM_ZYKCSL --月末库存数量-自有库存
       ,SUM(A.YM_WXKCJE) AS YM_WXKCJE --月末外协寄售金额
       ,SUM(A.YM_WXKCSL) AS YM_WXKCSL --月末外协寄售数量
FROM upload.upload_wm_prod_stock_dtl_t A
where A.stat_month <= '202112'
GROUP BY A.STAT_MONTH --月份
         ,A.material_num
UNION ALL	   
SELECT 
        CAST(A."CALMONTH" AS INT) AS STAT_MONTH --月份
       ,A."MATERIAL" AS MATERIAL_NUM --物料编码
       ,SUM(A."KCJE_YM") AS YM_ZYKCJE --月末库存金额-自有库存
       ,SUM(A."KCSL_YM") AS YM_ZYKCSL --月末库存数量-自有库存
       ,SUM(A."KCJE_YM_WX") AS YM_WXKCJE --月末外协寄售金额
       ,SUM(A."KCSL_YM_WX") AS YM_WXKCSL --月末外协寄售数量
FROM DM.DM_RPT_BIC_AZMMM_O0101_T A 
LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
ON A."MATERIAL" = B.MATERIAL_NUM
WHERE A."/BIC/ZA_SYAMB" = 'ZAMBID11000001'
AND A."CALMONTH" >= '202201'
GROUP BY CAST(A."CALMONTH" AS INT)
       ,A."MATERIAL" 

)
--销售成本
,T2 AS(
SELECT A.CALMONTH AS STAT_MONTH --月份
      ,A.MATERIAL AS MATERIAL_NUM --物料编码
      ,SUM(A.ZSJCB) AS ZSJCB --实际成本
      ,SUM(A.ZXSYJ) AS ZXSYJ --销售业绩
FROM DM.DM_ZASD_P001_Q066V2_T A
LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
ON A.MATERIAL = B.MATERIAL_NUM
WHERE A.SALES_GRP IN(SELECT DISTINCT SALES_GROUP
                       FROM DWRFIN.DWR_GL_SALES_GROUP_F
                      WHERE UTILITIES_DEPT = '11000001'
		            )  
				
GROUP BY A.CALMONTH
        ,A.MATERIAL
)
--库存周转天数=360/(滚动累计12个月销售成本/(滚动13个月月末库存余额/13))
,T3 AS(
SELECT STAT_MONTH --统计月份
      ,MATERIAL_NUM --物料编码
--      ,CHANNEL_NAME --四大渠道
      ,SUM(YM_ZYKCJE) AS YM_ZYKCJE --月末库存金额-自有库存
      ,SUM(YM_ZYKCSL) AS YM_ZYKCSL --月末库存数量-自有库存
      ,SUM(YM_WXKCJE) AS YM_WXKCJE --月末外协寄售金额
      ,SUM(YM_WXKCSL) AS YM_WXKCSL --月末外协寄售数量
	  ,SUM(ZSJCB) AS ZSJCB --实际成本
	  ,SUM(ZXSYJ) AS ZXSYJ --销售业绩 
FROM(
SELECT STAT_MONTH --统计月份
      ,MATERIAL_NUM --物料编码
--      ,CHANNEL_NAME --四大渠道
      ,YM_ZYKCJE --月末库存金额-自有库存
      ,YM_ZYKCSL --月末库存数量-自有库存
      ,YM_WXKCJE --月末外协寄售金额
      ,YM_WXKCSL --月末外协寄售数量
	  ,0 AS ZSJCB --实际成本
	  ,0 AS ZXSYJ --销售业绩
  FROM T1
  
 UNION ALL

SELECT CAST(STAT_MONTH AS INT) --统计月份
      ,MATERIAL_NUM --物料编码

      ,0 AS YM_ZYKCJE --月末库存金额-自有库存
      ,0 AS YM_ZYKCSL --月末库存数量-自有库存
      ,0 AS YM_WXKCJE --月末外协寄售金额
      ,0 AS YM_WXKCSL --月末外协寄售数量
	  ,ZSJCB --实际成本
	  ,ZXSYJ --销售业绩
  FROM T2) T
GROUP BY STAT_MONTH --统计月份
      ,MATERIAL_NUM --物料编码 

)
,T4 AS(
SELECT STAT_MONTH --统计月份
      ,MATERIAL_NUM --     
	  ,SUM(YM_ZYKCJE) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 12 PRECEDING AND CURRENT ROW) AS YM_ZYKCJE --月末库存金额-自有库存
	  ,SUM(YM_ZYKCSL) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 12 PRECEDING AND CURRENT ROW) AS YM_ZYKCSL --月末库存数量-自有库存
	  ,SUM(YM_WXKCJE) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 12 PRECEDING AND CURRENT ROW) AS YM_WXKCJE --月末外协寄售金额	  
	  ,SUM(YM_WXKCSL) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 12 PRECEDING AND CURRENT ROW) AS YM_WXKCSL --月末外协寄售数量
	  ,SUM(ZSJCB) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 11 PRECEDING AND CURRENT ROW) AS ZSJCB --实际成本
	  ,SUM(ZXSYJ) OVER( PARTITION BY MATERIAL_NUM ORDER BY STAT_MONTH ROWS BETWEEN 11 PRECEDING AND CURRENT ROW) AS ZXSYJ --销售业绩
  FROM T3
)
SELECT 
CAST(STAT_MONTH 		AS INT) AS STATISTICAL_MONTH			--统计月份
,MATERIAL_NUM 			AS MATERIAL_NUM							--物料编码
,SUM(YM_ZYKCJE) 		AS MON_END_STOCK_AMOUNT_O_STOCK  		--月末库存金额-自有库存 
,SUM(YM_ZYKCSL) 		AS MON_END_STOCK_QUANTITY_O_STOCK  		--月末库存数量-自有库存
,SUM(YM_WXKCJE) 		AS MON_END_CONSIGNMENT_AMOUNT  			--月末外协寄售金额
,SUM(YM_WXKCSL) 		AS MON_END_CONSIGNMENT_QUANTITY  		--月末外协寄售数量
,SUM(ZSJCB) 			AS ACTUAL_COST  						--实际成本
,SUM(ZXSYJ) 			AS SALES_PERFORMANCE  					--销售业绩
,'N' 					AS DEL_FLAG								-- 删除标识
,CURRENT_TIMESTAMP() 	AS DW_LAST_UPDATE_DATE					-- 同步数据时间
 FROM T4
WHERE stat_month >=
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '04'
     THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
         ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
          END
 GROUP BY STAT_MONTH --统计月份
      ,MATERIAL_NUM --物料编码
