TRUNCATE TABLE DM.DM_HR_SAP_EMP_SALE_STAT_T
;

WITH T1 AS(
--销售业绩 事业部
SELECT 
        A.CALMONTH AS STAT_MONTH --统计月份
	   ,B.NODE_NAME2 AS DEPT_CODE --部门编码
       ,B.NODE_DESC2 AS DEPT_NAME --部门名称
	   ,SUM(A.AMBPERFORMANCE) AS SALE_AMT --销售业绩
	   ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
  FROM DM.DM_AMBV2_ZZT_XSSR_T A
 INNER JOIN DM.DM_RPT_SALES_GROUP_T B
    ON A.SALES_GRP = B.NODE_NAME9
   AND B.NODE_DESC2 IN('瓷砖事业部','卫浴事业部','生态新材事业部','家居事业部')
   --集团及事业部层面剔除数据来源U
   AND COALESCE(A.DATA_SOURCE,'SNULL') <> 'U'
   AND A.CALMONTH <= '${PERIOD_ID_M}'
   AND A.CALMONTH >= '2023-01'
 GROUP BY 
        A.CALMONTH --统计月份
	   ,B.NODE_NAME2 --部门编码
       ,B.NODE_DESC2 --部门名称 
	
 UNION ALL
 
--销售业绩 运营中心（子公司）
SELECT 
        A.CALMONTH AS STAT_MONTH --统计月份
	   ,COALESCE(COALESCE(C.SALE_DEPT_CODE,D.SALE_DEPT_CODE),E.SALE_DEPT_CODE) AS DEPT_CODE --部门编码
       ,COALESCE(COALESCE(C.SALE_DEPT_NAME,D.SALE_DEPT_NAME),E.SALE_DEPT_NAME) AS DEPT_NAME --部门名称
	   ,SUM(A.AMBPERFORMANCE) AS SALE_AMT --销售业绩
	   ,B.NODE_DESC2 AS UP_DEPT_NAME --上级部门名称
  FROM DM.DM_AMBV2_ZZT_XSSR_T A
 INNER JOIN DM.DM_RPT_SALES_GROUP_T B
    ON A.SALES_GRP = B.NODE_NAME9
   AND B.NODE_DESC2 IN('瓷砖事业部','卫浴事业部','生态新材事业部','家居事业部')	
   AND A.CALMONTH <= '${PERIOD_ID_M}'
   AND A.CALMONTH >= '2023-01'
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T C
    ON B.NODE_NAME4 = C.SALE_DEPT_CODE
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T D
    ON B.NODE_NAME5 = D.SALE_DEPT_CODE
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T E
    ON B.NODE_NAME3 = E.SALE_DEPT_CODE   
 WHERE COALESCE(COALESCE(C.SALE_DEPT_NAME,D.SALE_DEPT_NAME),E.SALE_DEPT_NAME) IS NOT NULL
 GROUP BY 
        A.CALMONTH --统计月份
	   ,COALESCE(COALESCE(C.SALE_DEPT_CODE,D.SALE_DEPT_CODE),E.SALE_DEPT_CODE) --部门编码	
	   ,COALESCE(COALESCE(C.SALE_DEPT_NAME,D.SALE_DEPT_NAME),E.SALE_DEPT_NAME) --部门名称
       ,B.NODE_DESC2 --上级部门名称 
 	
 UNION ALL

--销售业绩 东鹏控股
SELECT 
        A.CALMONTH AS STAT_MONTH --统计月份
	   ,B.SALE_DEPT_CODE AS DEPT_CODE --部门编码
       ,CAST('东鹏控股' AS VARCHAR) AS DEPT_NAME --部门名称
	   ,SUM(A.AMBPERFORMANCE) AS SALE_AMT --销售业绩
	   ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
  FROM DM.DM_AMBV2_ZZT_XSSR_T A
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON 1 = 1
   AND B.EMP_DEPT_NAME = '东鹏控股'
 WHERE COALESCE(A.DATA_SOURCE,'SNULL') <> 'U' --集团及事业部层面剔除数据来源U
   AND A.CALMONTH <= '${PERIOD_ID_M}'
   AND A.CALMONTH >= '2023-01'
 GROUP BY 
        A.CALMONTH --统计月份
	   ,B.SALE_DEPT_CODE --部门编码	
)
,T2 AS(
--员工数 事业部
SELECT A.STAT_MONTH --统计月份
      ,B.SALE_DEPT_CODE AS DEPT_CODE --部门编码
	  ,A.LVL3_DEPT_NAME AS DEPT_NAME --部门名称
      ,COUNT(DISTINCT EMP_NUM) AS EMP_CNT --员工人数
	  ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
	  ,CAST('系统' AS VARCHAR) AS DATA_SRC_NAME --数据源
  FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F A
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON A.LVL3_DEPT_NAME = B.EMP_DEPT_NAME
 WHERE COALESCE(A.FX_PAY_AMT,0) + COALESCE(A.FLT_PAY_AMT,0) + COALESCE(A.BNS_AMT,0)
     + COALESCE(A.HR_WELFARE_AMT,0) + COALESCE(A.PAYROLL_AMT,0) + COALESCE(A.SVRNC_AMT,0) > 0
   AND A.STAT_MONTH <= '${PERIOD_ID_M}'
   AND A.STAT_MONTH >= '2023-06'
   AND (A.EMP_POST_TYPE_NAME <> '操作职族'
    OR A.EMP_POST_TYPE_NAME IS NULL)
 GROUP BY
       A.STAT_MONTH --统计月份
	  ,B.SALE_DEPT_CODE --部门编码	   
      ,A.LVL3_DEPT_NAME --部门名称

 UNION ALL
 
--员工数 运营中心（子公司）
SELECT A.STAT_MONTH --统计月份    
	  ,COALESCE(B.SALE_DEPT_CODE,C.SALE_DEPT_CODE,D.SALE_DEPT_CODE) AS DEPT_CODE --部门编码    
	  ,COALESCE(B.SALE_DEPT_NAME,C.SALE_DEPT_NAME,D.SALE_DEPT_NAME) AS DEPT_NAME --部门名称
      ,COUNT(DISTINCT EMP_NUM) AS EMP_CNT --员工人数
	  ,A.LVL3_DEPT_NAME AS UP_DEPT_NAME --上级部门名称
	  ,CAST('系统' AS VARCHAR) AS DATA_SRC_NAME --数据源
  FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F A
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON A.LVL4_DEPT_CODE = B.EMP_DEPT_CODE
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T C
    ON A.LVL5_DEPT_CODE = C.EMP_DEPT_CODE
 LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T D
    ON A.LVL6_DEPT_CODE = D.EMP_DEPT_CODE
 WHERE COALESCE(A.FX_PAY_AMT,0) + COALESCE(A.FLT_PAY_AMT,0) + COALESCE(A.BNS_AMT,0)
     + COALESCE(A.HR_WELFARE_AMT,0) + COALESCE(A.PAYROLL_AMT,0) + COALESCE(A.SVRNC_AMT,0) > 0
   AND A.STAT_MONTH <= '${PERIOD_ID_M}'
   AND A.STAT_MONTH >= '2023-06'
   AND (A.EMP_POST_TYPE_NAME <> '操作职族'
    OR A.EMP_POST_TYPE_NAME IS NULL)
   AND A.LVL3_DEPT_NAME IN('瓷砖事业部','卫浴事业部','家居事业部')
  AND COALESCE(B.SALE_DEPT_NAME,C.SALE_DEPT_NAME,D.SALE_DEPT_NAME) IS NOT NULL   
 GROUP BY
       A.STAT_MONTH --统计月份
	  ,COALESCE(B.SALE_DEPT_CODE,C.SALE_DEPT_CODE,D.SALE_DEPT_CODE) --部门编码   	   
      ,COALESCE(B.SALE_DEPT_NAME,C.SALE_DEPT_NAME,D.SALE_DEPT_NAME) --部门名称
	  ,A.LVL3_DEPT_NAME --上级部门名称

 UNION ALL

--员工数 集团
SELECT A.STAT_MONTH --统计月份
      ,B.SALE_DEPT_CODE AS DEPT_CODE --部门编码
      ,CAST('东鹏控股' AS VARCHAR) AS DEPT_NAME --部门名称
      ,COUNT(DISTINCT EMP_NUM) AS EMP_CNT --员工人数
	  ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
	  ,CAST('系统' AS VARCHAR) AS DATA_SRC_NAME --数据源
  FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F A
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON 1 = 1
   AND B.EMP_DEPT_NAME = '东鹏控股'
 WHERE COALESCE(A.FX_PAY_AMT,0) + COALESCE(A.FLT_PAY_AMT,0) + COALESCE(A.BNS_AMT,0)
     + COALESCE(A.HR_WELFARE_AMT,0) + COALESCE(A.PAYROLL_AMT,0) + COALESCE(A.SVRNC_AMT,0) > 0
   AND A.STAT_MONTH <= '${PERIOD_ID_M}'
   AND A.STAT_MONTH >= '2023-06'
   AND (A.EMP_POST_TYPE_NAME <> '操作职族'
    OR A.EMP_POST_TYPE_NAME IS NULL)
 GROUP BY
       A.STAT_MONTH --统计月份
	  ,B.SALE_DEPT_CODE --部门编码 
	   
 UNION ALL

--员工数 集团和运营中心（子公司）
SELECT A.STAT_MONTH
      ,B.SALE_DEPT_CODE AS DEPT_CODE --部门编码
      ,B.SALE_DEPT_NAME AS DEPT_NAME --部门名称
      ,SUM(COALESCE(A.EMP_NUM,0)) AS EMP_CNT --员工人数	  
	  ,A.DEPT_NAME AS UP_DEPT_NAME --上级部门名称
	  ,CAST('填报' AS VARCHAR) AS DATA_SRC_NAME --数据源
  FROM UPLOAD.UPLOAD_EHR_MONTHLY_PER_SALES_REVENUE_T A
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON A.SUB_DEPT_CODE = B.EMP_DEPT_CODE
 WHERE A.STAT_MONTH >= '2023-01'
   AND A.STAT_MONTH <= '${PERIOD_ID_M}'
   AND A.EMP_NUM IS NOT NULL
 GROUP BY
       A.STAT_MONTH
      ,B.SALE_DEPT_CODE --部门编码
      ,B.SALE_DEPT_NAME --部门名称
	  ,A.DEPT_NAME --上级部门名称
),T3 AS(
SELECT STAT_MONTH --统计月份
      ,DEPT_CODE --部门编码
      ,DEPT_NAME --部门名称
      ,SUM(EMP_CNT) OVER(PARTITION BY DEPT_NAME,SUBSTR(STAT_MONTH,1,4) ORDER BY STAT_MONTH ASC) / CAST(SUBSTR(STAT_MONTH,6,2) AS NUMERIC) AS EMP_CNT --年累计发薪人数
      ,UP_DEPT_NAME --上级部门名称
FROM (SELECT *
            ,ROW_NUMBER() OVER(PARTITION BY DEPT_CODE,STAT_MONTH ORDER BY CASE WHEN DATA_SRC_NAME = '填报' THEN 1 ELSE 2 END ASC) AS RN
        FROM T2)
WHERE RN = 1
),T4 AS(
--年累计销售业绩 事业部
SELECT 
        C.STAT_MONTH --统计月份
	   ,B.NODE_NAME2 AS DEPT_CODE --部门编码
       ,B.NODE_DESC2 AS DEPT_NAME --部门名称
	   ,SUM(A.AMBPERFORMANCE) AS SALE_AMT --销售业绩
	   ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
  FROM (SELECT DISTINCT SUBSTR(DAY_MBEGIN_DT,1,7) AS STAT_MONTH
                       ,SUBSTR(DAY_MBEGIN_DT,1,4) AS STAT_YEAR
          FROM DM.DM_DIM_DATE_D
	     WHERE DAY_ID >= '20230101'
		   AND SUBSTR(DAY_MBEGIN_DT,1,7) <= '${PERIOD_ID_M}'
	   )  C
 INNER JOIN DM.DM_AMBV2_ZZT_XSSR_T A
    ON C.STAT_YEAR = SUBSTR(A.CALMONTH,1,4)
   AND C.STAT_MONTH >= A.CALMONTH
 INNER JOIN DM.DM_RPT_SALES_GROUP_T B
    ON A.SALES_GRP = B.NODE_NAME9
   AND B.NODE_DESC2 IN('瓷砖事业部','卫浴事业部','生态新材事业部','家居事业部')	
   --集团及事业部层面剔除数据来源U
   AND COALESCE(A.DATA_SOURCE,'SNULL') <> 'U' 
 GROUP BY 
        C.STAT_MONTH --统计月份
	   ,B.NODE_NAME2 --部门编码	
       ,B.NODE_DESC2 --部门名称 
	   
 UNION ALL

SELECT DISTINCT
        C.STAT_MONTH --统计月份
	   ,B.SALE_DEPT_CODE AS DEPT_CODE --部门编码	
       ,CAST('东鹏控股' AS VARCHAR) AS DEPT_NAME --部门名称
	   ,SUM(COALESCE(A.AMBPERFORMANCE,0)) OVER(PARTITION BY C.STAT_YEAR ORDER BY C.STAT_MONTH ASC) AS SALE_AMT --销售业绩
	   ,CAST('东鹏控股' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
  FROM (SELECT DISTINCT SUBSTR(DAY_MBEGIN_DT,1,7) AS STAT_MONTH
                       ,SUBSTR(DAY_MBEGIN_DT,1,4) AS STAT_YEAR
          FROM DM.DM_DIM_DATE_D
	     WHERE DAY_ID >= '20230101'
		   AND SUBSTR(DAY_MBEGIN_DT,1,7) <= '${PERIOD_ID_M}'
	   )  C
 INNER JOIN DM.DM_AMBV2_ZZT_XSSR_T A
    ON C.STAT_YEAR = SUBSTR(A.CALMONTH,1,4)
   AND C.STAT_MONTH >= A.CALMONTH
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T B
    ON 1 = 1
   AND B.EMP_DEPT_NAME = '东鹏控股'	
 WHERE COALESCE(A.DATA_SOURCE,'SNULL') <> 'U' --集团及事业部层面剔除数据来源U
 
 UNION ALL
 
--年累计销售业绩 运营中心（子公司）
SELECT  DISTINCT
        C.STAT_MONTH --统计月份
	   ,COALESCE(COALESCE(D.SALE_DEPT_CODE,E.SALE_DEPT_CODE),F.SALE_DEPT_CODE) AS DEPT_CODE --部门编码
       ,COALESCE(COALESCE(D.SALE_DEPT_NAME,E.SALE_DEPT_NAME),F.SALE_DEPT_NAME) AS DEPT_NAME --部门名称
	   ,SUM(COALESCE(A.AMBPERFORMANCE,0)) OVER(PARTITION BY C.STAT_YEAR ORDER BY C.STAT_MONTH ASC) AS SALE_AMT --销售业绩
	   ,B.NODE_DESC2 AS UP_DEPT_NAME --上级部门名称
  FROM (SELECT DISTINCT SUBSTR(DAY_MBEGIN_DT,1,7) AS STAT_MONTH
                       ,SUBSTR(DAY_MBEGIN_DT,1,4) AS STAT_YEAR
          FROM DM.DM_DIM_DATE_D
	     WHERE DAY_ID >= '20230101'
		   AND SUBSTR(DAY_MBEGIN_DT,1,7) <= '${PERIOD_ID_M}'
	   )  C
 INNER JOIN DM.DM_AMBV2_ZZT_XSSR_T A
    ON C.STAT_YEAR = SUBSTR(A.CALMONTH,1,4)
   AND C.STAT_MONTH >= A.CALMONTH
 INNER JOIN DM.DM_RPT_SALES_GROUP_T B
    ON A.SALES_GRP = B.NODE_NAME9
   AND B.NODE_DESC2 IN('瓷砖事业部','卫浴事业部','生态新材事业部','家居事业部')	
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T D
    ON B.NODE_DESC4 = D.EMP_DEPT_NAME
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T E
    ON B.NODE_DESC5 = E.EMP_DEPT_NAME
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T F
    ON B.NODE_DESC3 = F.EMP_DEPT_NAME  
 WHERE COALESCE(COALESCE(D.SALE_DEPT_NAME,E.SALE_DEPT_NAME),F.SALE_DEPT_NAME) IS NOT NULL
)
INSERT INTO DM.DM_HR_SAP_EMP_SALE_STAT_T
(
       STAT_MONTH --统计月份
	  ,DEPT_CODE --部门编码 
      ,DEPT_NAME --部门名称
      ,EMP_CNT --员工人数
	  ,SALE_AMT --销售业绩
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,STAT_TYPE --统计类型
	  ,UP_DEPT_CODE --上级部门编码
	  ,UP_DEPT_NAME --上级部门名称
)
SELECT A.STAT_MONTH --统计月份
      ,A.DEPT_CODE --部门编码
      ,A.DEPT_NAME --部门名称
      ,A.EMP_CNT --员工人数
	  ,COALESCE(B.SALE_AMT,0) AS SALE_AMT --销售业绩
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CAST('当月' AS VARCHAR) AS STAT_TYPE --统计类型
	  ,C.SALE_DEPT_CODE AS UP_DEPT_CODE --上级部门编码
	  ,A.UP_DEPT_NAME --上级部门名称
  FROM (SELECT *
            ,ROW_NUMBER() OVER(PARTITION BY DEPT_CODE,STAT_MONTH ORDER BY CASE WHEN DATA_SRC_NAME = '填报' THEN 1 ELSE 2 END ASC) AS RN
        FROM T2) A
 INNER JOIN T1 B
    ON A.DEPT_CODE = B.DEPT_CODE
   AND A.STAT_MONTH = B.STAT_MONTH
  LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T C
    ON A.UP_DEPT_NAME = C.EMP_DEPT_NAME
 WHERE A.RN = 1
 
 UNION ALL

SELECT A.STAT_MONTH --统计月份
      ,A.DEPT_CODE --部门编码
      ,A.DEPT_NAME --部门名称
      ,ROUND(A.EMP_CNT) AS EMP_CNT --员工人数
	  ,COALESCE(B.SALE_AMT,0) AS SALE_AMT --销售业绩
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CAST('年累计' AS VARCHAR) AS STAT_TYPE --统计类型
	  ,D.SALE_DEPT_CODE AS UP_DEPT_CODE --上级部门编码
	  ,A.UP_DEPT_NAME --上级部门名称
  FROM T3 A
 INNER JOIN T4 B
    ON A.DEPT_NAME = B.DEPT_NAME
   AND A.STAT_MONTH = B.STAT_MONTH
  LEFT JOIN(SELECT MAX(STAT_MONTH) AS STAT_MONTH
              FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F
           ) C
	ON 1 = 1
    LEFT JOIN UPLOAD.UPLOAD_SALE_EMP_DEPT_REL_T D
    ON A.UP_DEPT_NAME = D.EMP_DEPT_NAME	
 WHERE A.STAT_MONTH <= C.STAT_MONTH
;
