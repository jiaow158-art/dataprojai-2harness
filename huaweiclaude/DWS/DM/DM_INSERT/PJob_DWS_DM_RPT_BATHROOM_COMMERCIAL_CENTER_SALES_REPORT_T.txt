--         如果  "/BIC/ZYL04" <> '00003' 且 STOR_LOC（库存地点）IN ('GT90','GD90','GR90')
--                则：  ZXSCB_DY = DY_QC_XSCB_DD_DY
-- 2、ZXSCB_QNTQ（销售成本-去年同期）调整逻辑：
--         如果  "/BIC/ZYL04" = '00003' 且 STOR_LOC（库存地点） IN ('GT90','GD90','GR90')
--                则：  ZXSCB_QNTQ = DY_QC_XSCB_WL_DD_QNTQ
--         如果  "/BIC/ZYL04" = '00003' 且 STOR_LOC（库存地点） NOT IN ('GT90','GD90','GR90')
--                则：  ZXSCB_QNTQ = DY_QC_XSCB_WL_QNTQ
--         如果  "/BIC/ZYL04" <> '00003' 且 STOR_LOC（库存地点）IN ('GT90','GD90','GR90')
--                则：  ZXSCB_QNTQ = DY_QC_XSCB_DD_QNTQ
-- 3、ZMLE_DY（毛利额-当月）调整逻辑：
--         ZMLE_DY = Z_BHSXSJEGL_DY - ZXSCB_DY
-- 4、ZMLE_QNTQ（毛利额-去年同期）调整逻辑：
--         ZMLE_QNTQ = Z_BHSXSJEGL_QNTQ - ZXSCB_QNTQ
-- ******************************************************************** --

,tt.calyear
,tt.comp_code

,tt.distr_chan

,(tt.z_bhsxsjegl_qntq - tt.zxscb_qntq) as zmle_qntq
,tt.zxssl_qntq
,tt.dw_last_update_date
,tt.zxscb_dy
,tt.zxscb_qntq
,tt.dy_qc_xscb_dd_dy	
,tt.dy_qc_xscb_wl_dy
,tt.dy_qc_xscb_wl_dd_dy
,tt.dy_qc_xscb_dd_qntq
,tt.dy_qc_xscb_wl_qntq
,tt.dy_qc_xscb_wl_dd_qntq
from (
select
 t1.calday
,t1.calmonth
,t1.calyear
,t1.comp_code
,t1.comp_code___t
,t1.distr_chan
,t1.distr_chan___t
,t1.division
,t1.division___t

,t1.plant___t

,t1.sold_to
,t1.sold_to___t
,t1.stor_loc
,t1.stor_loc___t
,t1.wbs_elemt
,t1.wbs_elemt___t
,t1.z_gcbbh
,t1.zxsyj_dy
,nvl(t1.z_bhsxsjegl_dy,0) as z_bhsxsjegl_dy
,nvl(t1.zmle_dy,0) as zmle_dy
,t1.zxssl_dy

,t1.zxssl_qntq
,t1.dw_last_update_date
,case when t2."/BIC/ZYL04" =  '00003' and t1.stor_loc IN ('GT90','GD90','GR90')     then nvl(t1.dy_qc_xscb_wl_dd_dy,0)
      when t2."/BIC/ZYL04" =  '00003' and t1.stor_loc not IN ('GT90','GD90','GR90') then nvl(t1.dy_qc_xscb_wl_dy,0)
      when t2."/BIC/ZYL04" <> '00003' and t1.stor_loc IN ('GT90','GD90','GR90')     then nvl(t1.dy_qc_xscb_dd_dy,0)
      else nvl(to_number(t1.zxscb_dy),0)
    end zxscb_dy

      when t2."/BIC/ZYL04" =  '00003' and t1.stor_loc not IN ('GT90','GD90','GR90') then nvl(t1.dy_qc_xscb_wl_qntq,0)
      when t2."/BIC/ZYL04" <> '00003' and t1.stor_loc IN ('GT90','GD90','GR90')     then nvl(t1.dy_qc_xscb_dd_qntq,0)

where t1.calmonth = '${start_date}'
