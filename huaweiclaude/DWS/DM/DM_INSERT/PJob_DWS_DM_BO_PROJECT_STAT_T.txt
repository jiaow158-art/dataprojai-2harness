WITH T1 AS(
SELECT DAY_DT AS MONTH_END_DATE --月末最后一天
      ,DAY_MEND_MAGO_DT AS LAST_MONTH_END_DATE --上月月末最后一天
      ,DAY_YEND_YAGO_DT --去年年末最后一天
      ,DAY_QAGO_DT --上季度最后一天
      ,CONCAT(YEAR_ID,'-01') AS Q1_START_MONTH --一季度开始月份
      ,CONCAT(YEAR_ID,'-03') AS Q1_END_MONTH --一季度结束月份
      ,CASE WHEN CONCAT(YEAR_ID,'-04') <= '${PERIOD_ID_M}' THEN CONCAT(YEAR_ID,'-04') END AS Q2_START_MONTH --二季度开始月份
      ,CASE WHEN CONCAT(YEAR_ID,'-04') <= '${PERIOD_ID_M}' THEN CONCAT(YEAR_ID,'-06') END AS Q2_END_MONTH --二季度结束月份
      ,CASE WHEN CONCAT(YEAR_ID,'-07') <= '${PERIOD_ID_M}' THEN CONCAT(YEAR_ID,'-07') END AS Q3_START_MONTH --三季度开始月份
      ,CASE WHEN CONCAT(YEAR_ID,'-07') <= '${PERIOD_ID_M}' THEN CONCAT(YEAR_ID,'-09') END AS Q3_END_MONTH --三季度结束月份
      ,CASE WHEN CONCAT(YEAR_ID,'-10') <= '${PERIOD_ID_M}' THEN CONCAT(YEAR_ID,'-10') END AS Q4_START_MONTH --四季度开始月份
      ,CASE WHEN CONCAT(YEAR_ID,'-10') <= '${PERIOD_ID_M}' THEN  CONCAT(YEAR_ID,'-12') END AS Q4_END_MONTH --四季度结束月份
  FROM DM.DM_DIM_DATE_D
 WHERE SUBSTR(DAY_DT,1,7) = '${PERIOD_ID_M}'
   AND MONTH_END_DAY_FLAG = 'Y'
)
INSERT INTO DM.DM_BO_PROJECT_STAT_T(
      STAT_MONTH
     ,DEPT_NAME
     ,AREA_NAME
     ,SALE_GRP_NAME
     ,WORKS_CATEGORY
     ,CUST_CODE
     ,CUST_NAME
     ,APPLY_EMP_NUM
     ,APPLY_EMP_NAME
     ,IS_DIFF_DEPT
     ,FOLW_DEPT_NAME
     ,LM_BO_CNT
     ,NEW_BO_CNT
     ,EXP_BO_CNT
     ,TM_BO_CNT
     ,EXP_BO_AREA
     ,TM_BO_AREA
     ,NEW_BO_AMT
     ,EXP_BO_AMT
     ,TM_BO_AMT
     ,LY_BO_CNT
     ,LY_BO_AMT
     ,Q1_BO_CNT
     ,Q1_BO_AMT
     ,Q2_BO_CNT
     ,Q2_BO_AMT
     ,Q3_BO_CNT
     ,Q3_BO_AMT
     ,Q4_BO_CNT
     ,Q4_BO_AMT
     ,DW_LAST_UPDATE_DATE
	 ,BO_BIG_TYPE --商机大类(商机类型)
     ,BO_SMALL_TYPE --商机小类(项目类型)
	 ,BO_STAGE --商机阶段
	 ,NEW_BO_AREA --新增商机总面积
)
SELECT '${PERIOD_ID_M}' AS STAT_MONTH --统计月份
      ,LEVEL_DEPT_DESC_2 AS DEPT_NAME --事业部
      ,CASE WHEN LEVEL_DEPT_DESC_2 = '卫浴事业部' THEN LEVEL_DEPT_DESC_4
            WHEN LEVEL_DEPT_DESC_2 = '瓷砖事业部' THEN LEVEL_DEPT_DESC_5
             END AS AREA_NAME --区域
      ,LEVEL_DEPT_DESC_10 AS SALE_GRP_NAME --当前线组  
      ,WORKS_CATEGORY  --商机类型
      ,CUST_CODE --客户编码
      ,CUST_NAME --客户名称
      ,APPLY_EMP_NUM --申请人工号
      ,APPLY_EMP_NAME --申请人名称
      ,'否' AS IS_DIFF_DEPT --是否跨事业部
      ,FOLLOW_UP_DIVISION_DEPT AS FOLW_DEPT_NAME --跟进事业部
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,10) <= LAST_MONTH_END_DATE THEN BO_CODE END) AS LM_BO_CNT --商机保有量
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_CODE END) AS NEW_BO_CNT --新增商机数
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_CODE END) AS EXP_BO_CNT --失效商机数
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,10) <= MONTH_END_DATE THEN BO_CODE END) AS TM_BO_CNT --有效商机数
      ,SUM(CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_AREA ELSE 0 END) AS EXP_BO_AREA --失效商机总面积
      ,SUM(CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,10) <= MONTH_END_DATE THEN BO_AREA ELSE 0 END) AS TM_BO_AREA --有效商机总面积
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_AMT ELSE 0 END) NEW_BO_AMT --新增商机金额
      ,SUM(CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_AMT ELSE 0 END) EXP_BO_AMT --失效商机金额
      ,SUM(CASE WHEN SUBSTR(BO_VALIDITY_DATE,1,10) <= MONTH_END_DATE THEN BO_AMT ELSE 0 END) TM_BO_AMT --有效商机金额
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,10) <= DAY_YEND_YAGO_DT
                            AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                           THEN BO_CODE END) AS LY_BO_CNT --上年商机有效数
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,10) <= DAY_YEND_YAGO_DT
                 AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                THEN BO_AMT END) AS LY_BO_AMT --上年商机有效金额                
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q1_START_MONTH
                            AND SUBSTR(BO_APPLY_DATE,1,7) <= Q1_END_MONTH
                            AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                           THEN BO_CODE END) AS Q1_BO_CNT --一季度申请有效数
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q1_START_MONTH
                 AND SUBSTR(BO_APPLY_DATE,1,7) <= Q1_END_MONTH
                 AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                THEN BO_AMT END) AS Q1_BO_AMT --一季度申请有效金额
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q2_START_MONTH
                            AND SUBSTR(BO_APPLY_DATE,1,7) <= Q2_END_MONTH
                            AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                           THEN BO_CODE END) AS Q2_BO_CNT --二季度申请有效数
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q2_START_MONTH
                 AND SUBSTR(BO_APPLY_DATE,1,7) <= Q2_END_MONTH
                 AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                THEN BO_AMT END) AS Q2_BO_AMT --二季度申请有效金额           
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q3_START_MONTH
                            AND SUBSTR(BO_APPLY_DATE,1,7) <= Q3_END_MONTH
                            AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                           THEN BO_CODE END) AS Q3_BO_CNT --三季度申请有效数
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q3_START_MONTH
                 AND SUBSTR(BO_APPLY_DATE,1,7) <= Q3_END_MONTH
                 AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                THEN BO_AMT END) AS Q3_BO_AMT --三季度申请有效金额
      ,COUNT(DISTINCT CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q4_START_MONTH
                            AND SUBSTR(BO_APPLY_DATE,1,7) <= Q4_END_MONTH
                            AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                           THEN BO_CODE END) AS Q4_BO_CNT --四季度申请有效数
      ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) >= Q4_START_MONTH
                 AND SUBSTR(BO_APPLY_DATE,1,7) <= Q4_END_MONTH
                 AND SUBSTR(BO_VALIDITY_DATE,1,10) >= MONTH_END_DATE
                THEN BO_AMT END) AS Q4_BO_AMT --四季度申请有效金额                       
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,BO_BIG_TYPE --商机大类(商机类型)
      ,BO_SMALL_TYPE --商机小类(项目类型)
	  ,BO_STAGE --商机阶段
	  ,SUM(CASE WHEN SUBSTR(BO_APPLY_DATE,1,7) = '${PERIOD_ID_M}' THEN BO_AREA ELSE 0 END) AS NEW_BO_AREA --新增商机总面积
  FROM DWRLTC.DWR_BO_PROJECT_DETAIL_F A
  LEFT JOIN T1
    ON 1 = 1
GROUP BY
       LEVEL_DEPT_DESC_2 --事业部
      ,CASE WHEN LEVEL_DEPT_DESC_2 = '卫浴事业部' THEN LEVEL_DEPT_DESC_4
            WHEN LEVEL_DEPT_DESC_2 = '瓷砖事业部' THEN LEVEL_DEPT_DESC_5
             END --区域
      ,LEVEL_DEPT_DESC_10 --当前线组  
      ,WORKS_CATEGORY  --商机类型
      ,CUST_CODE --客户编码
      ,CUST_NAME --客户名称
      ,APPLY_EMP_NUM --申请人工号
      ,APPLY_EMP_NAME --申请人名称
      ,FOLLOW_UP_DIVISION_DEPT --跟进事业部
	  ,BO_BIG_TYPE --商机大类(商机类型)
      ,BO_SMALL_TYPE --商机小类(项目类型)
	  ,BO_STAGE --商机阶段
