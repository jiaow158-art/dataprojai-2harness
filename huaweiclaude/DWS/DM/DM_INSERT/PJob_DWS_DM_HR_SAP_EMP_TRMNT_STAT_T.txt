WITH T1 AS(
SELECT
       A.STAT_MONTH --统计月份
	  ,SUBSTR(CAST(A.STAT_MONTH || '-01' AS DATE) + INTERVAL '1 YEAR',1,7) AS NY_STAT_MONTH --下一年月份

            WHEN A.EMP_POST_TYPE_NAME = '操作职族' AND A.EMP_POST_ATTR_NAME = '直接产出' THEN 'OO'
            WHEN A.EMP_POST_TYPE_NAME IN('管理职族','专业职族','营销职族(失效)','行政职族(失效)','操作职族')
			 AND (A.EMP_POST_ATTR_NAME = '' OR A.EMP_POST_ATTR_NAME IS NULL) THEN NULL
			ELSE A.EMP_POST_TYPE_NAME
             END AS EMP_POST_TYPE_NAME --职族名称
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族') THEN '是'
	        ELSE '否'
			 END AS IS_MGMT_CADRES --是否管理干部		 
	  ,COUNT(DISTINCT CASE WHEN COALESCE(A.OPER_TYPE_NAME,'$$$') <> '退休' THEN A.EMP_NUM END) AS TRMNT_EMP_CNT --离职人数
	  ,COUNT(DISTINCT CASE WHEN A.TRMNT_TYPE_NAME = '主动离职' OR A.OPER_TYPE_NAME IN('内退') THEN A.EMP_NUM ELSE NULL END) AS VOLRY_TRMNT_EMP_CNT --主动离职人数
	  ,COUNT(DISTINCT CASE WHEN A.TRMNT_TYPE_NAME = '被动离职' AND COALESCE(A.OPER_REASON,'&&&') <> '死亡' THEN A.EMP_NUM ELSE NULL END) AS UNVOLRY_TRMNT_EMP_CNT --被动离职人数
	  ,COUNT(DISTINCT CASE WHEN A.OPER_REASON = '死亡' THEN A.EMP_NUM END) AS OTH_TRMNT_EMP_CNT --其他离职人数
	  ,COUNT(DISTINCT CASE WHEN A.OPER_TYPE_NAME = '退休' THEN A.EMP_NUM END) AS RET_TRMNT_EMP_CNT --退休人数

   AND A.OPER_TYPE_NAME IN('离职','（停用）终止协议用工','退休','内退')

 AND (A.LVL3_DEPT_NAME NOT IN ('资本运营部') or A.LVL3_DEPT_NAME is null)

       A.STAT_MONTH --统计月份

	  ,A.LVL4_DEPT_NAME --四级部门名称
	  ,A.LVL5_DEPT_NAME --五级部门名称
      ,A.LVL6_DEPT_NAME --六级部门名称

	  ,A.LVL8_DEPT_NAME --八级部门名称

			 AND (A.EMP_POST_ATTR_NAME = '' OR A.EMP_POST_ATTR_NAME IS NULL) THEN NULL
			ELSE A.EMP_POST_TYPE_NAME
             END --职族名称
	  ,CASE WHEN A.EMP_POST_TYPE_NAME IN('管理职族') THEN '是'
	        ELSE '否'
			 END --是否管理干部		 
)

SELECT
