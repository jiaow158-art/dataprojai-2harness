TRUNCATE TABLE DM.DM_DMT_USER_STAT_T;

INSERT INTO DM.DM_DMT_USER_STAT_T
(
       STAT_DATE
      ,STORE_ID
      ,STORE_NAME
      ,LEV1_DEPT_NAME
      ,LEV2_DEPT_NAME
      ,LEV3_DEPT_NAME
      ,LEV4_DEPT_NAME
      ,LEV5_DEPT_NAME
      ,LEV6_DEPT_NAME
      ,SOURCE
      ,SOURCE_NAME
      ,GENDER
      ,GENDER_NAME
      ,AGE_GROUP
      ,AGE_GROUP_NAME
      ,DECORATE_PREDILECTION
      ,DECORATE_PREDILECTION_NAME
      ,PROVINCE_NAME
      ,CITY_NAME
      ,DISTRICT_NAME
      ,STAGE_ID
      ,STAGE_NAME
      ,ALREADY_PAY_AMOUNT
      ,ORDER_AMOUNT
      ,CJ_ORDER_AMOUNT
      ,USER_COUNT
      ,DGJ_USER_COUNT
      ,GJZ_USER_COUNT
      ,DD_USER_COUNT
      ,JD_USER_COUNT
      ,KD_USER_COUNT
      ,WC_USER_COUNT
      ,DPJ_USER_COUNT
      ,PJ_USER_COUNT
      ,DW_LAST_UPDATE_DATE
      ,PARENT_SOURCE_ID
      ,PARENT_SOURCE_NAME
	  ,EMP_COUNT
)
SELECT SUBSTR(A.UPDATE_TIME,1,10) AS STAT_DATE --统计日
      ,A.STORE_ID --门店ID
      ,A.STORE_NAME --门店名称
	  ,A.LEV1_DEPT_NAME --一级部门名称
	  ,A.LEV2_DEPT_NAME --二级部门名称
	  ,A.LEV3_DEPT_NAME --三级部门名称
	  ,A.LEV4_DEPT_NAME --四级部门名称
	  ,A.LEV5_DEPT_NAME --五级部门名称
	  ,A.LEV6_DEPT_NAME --六级部门名称
	  ,A.SOURCE --客户来源,参考CLIENTSOURCEENUM类
	  ,A.SOURCE_NAME --客户来源名称
	  ,A.GENDER --性别 1-男性;2-女性;0-未知
	  ,CASE WHEN A.GENDER = 1 THEN '男性'
	        WHEN A.GENDER = 2 THEN '女性'
			WHEN A.GENDER = 0 THEN '未知'
			 END AS GENDER_NAME --性别名称
      ,A.AGE_GROUP --年龄段
	  ,CASE WHEN A.AGE_GROUP = 1 THEN '19岁以下'
	        WHEN A.AGE_GROUP = 2 THEN '20-29岁'
			WHEN A.AGE_GROUP = 3 THEN '30-39岁'
			WHEN A.AGE_GROUP = 4 THEN '40-49岁'
			WHEN A.AGE_GROUP = 5 THEN '50岁及以上'
			 END AS AGE_GROUP_NAME --年龄段名称
	  ,A.DECORATE_PREDILECTION  --	装修喜好，1-中国风，2-欧美风，3-日式
	  ,A.DECORATE_PREDILECTION_NAME --装修喜好名称
	  ,A.PROVINCE_NAME--省区域名
      ,A.CITY_NAME --市区域名
      ,A.DISTRICT_NAME -- 区区域名
	  ,A.STAGE_ID --客户阶段ID
	  ,CASE WHEN A.STAGE_ID = 1 THEN '待跟进'
	        WHEN A.STAGE_ID = 2 THEN '跟进中'
	        WHEN A.STAGE_ID = 8 THEN '已到店'
	        WHEN A.STAGE_ID = 3 THEN '已交定'
	        WHEN A.STAGE_ID = 4 THEN '已开单'
	        WHEN A.STAGE_ID = 5 THEN '已完成'
	        WHEN A.STAGE_ID = 6 THEN '待评价'
	        WHEN A.STAGE_ID = 7 THEN '已评价'
			 END AS STAGE_NAME --客户阶段名称
	  ,SUM(COALESCE(A.ALREADY_PAY_MONEY,0)) AS ALREADY_PAY_AMOUNT --交定金额
	  ,SUM(COALESCE(A.ORDER_MONEY,0)) AS ORDER_AMOUNT --开单金额
	  ,SUM(COALESCE(A.ALREADY_PAY_MONEY,0) + COALESCE(A.ORDER_MONEY,0)) AS CJ_ORDER_AMOUNT --成交金额
	  ,COUNT(DISTINCT A.USER_ID) AS USER_COUNT --用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 1 THEN A.USER_ID END) AS DGJ_USER_COUNT --待跟进用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 2 THEN A.USER_ID END) AS GJZ_USER_COUNT --跟进中用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 8 THEN A.USER_ID END) AS DD_USER_COUNT --已到店用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 3 THEN A.USER_ID END) AS JD_USER_COUNT --已交定用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 4 THEN A.USER_ID END) AS KD_USER_COUNT --已开单用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 5 THEN A.USER_ID END) AS WC_USER_COUNT --已完成用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 6 THEN A.USER_ID END) AS DPJ_USER_COUNT --待评价用户数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_ID = 7 THEN A.USER_ID END) AS PJ_USER_COUNT --已评价用户数
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
      ,A.PARENT_SOURCE_ID --客户来源上级ID
      ,A.PARENT_SOURCE_NAME --客户来源上级名称
	  ,B.EMP_COUNT --门店雇员人数
  FROM DWRDIM.DWR_DIM_DMT_CUST_INFO_D A
  LEFT JOIN(SELECT STORE_ID
                  ,COUNT(DISTINCT EMP_ID) AS EMP_COUNT
              FROM DWIMD.DWI_MD_DMT_EMP_INFO_T
			 WHERE STATUS = 1
			 GROUP BY STORE_ID
           ) B
    ON A.STORE_ID = B.STORE_ID
   AND B.STORE_ID IS NOT NULL
 WHERE A.TEL_NUM IS NOT NULL
   AND A.STAGE_ID IN (1,2,3,4,5,6,7,8)
   AND A.IS_WHITELIST = 0
   AND A.IS_OLD_CLIENT_ADD = 0
 GROUP BY
       SUBSTR(A.UPDATE_TIME,1,10)
	  ,A.STORE_ID --门店ID
      ,A.STORE_NAME --门店名称
	  ,A.LEV1_DEPT_NAME --一级部门名称
	  ,A.LEV2_DEPT_NAME --二级部门名称
	  ,A.LEV3_DEPT_NAME --三级部门名称
	  ,A.LEV4_DEPT_NAME --四级部门名称
	  ,A.LEV5_DEPT_NAME --五级部门名称
	  ,A.LEV6_DEPT_NAME --六级部门名称
	  ,A.SOURCE --客户来源,参考CLIENTSOURCEENUM类
	  ,A.SOURCE_NAME --客户来源名称
	  ,A.GENDER --性别 1-男性;2-女性;0-未知
      ,A.AGE_GROUP --年龄段
	  ,A.DECORATE_PREDILECTION  --	装修喜好，1-中国风，2-欧美风，3-日式
	  ,A.DECORATE_PREDILECTION_NAME --装修喜好名称
	  ,A.PROVINCE_NAME--省区域名
      ,A.CITY_NAME --市区域名
      ,A.DISTRICT_NAME -- 区区域名
	  ,A.STAGE_ID --客户阶段ID
      ,A.PARENT_SOURCE_ID --客户来源上级ID
      ,A.PARENT_SOURCE_NAME --客户来源上级名称	  
      ,B.EMP_COUNT
;
