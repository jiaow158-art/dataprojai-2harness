TRUNCATE TABLE DM.WM_CERAMIC_STOCK_STRUCTURE_T;

INSERT INTO DM.WM_CERAMIC_STOCK_STRUCTURE_T
SELECT 
       A.STAT_MONTH --月份
	  ,A.CATEGORY_NAME --品类
	  ,'自有' AS STOCK_TYPE --库存类型：自有、寄售
	  ,SUM(A.SYB_CHDJ_MJ_1_1_1 + A.SYB_CHDJ_MJ_1_1_2) AS TOTAL_AREA1 --减值损失-半年内-面积
	  ,SUM(A.SYB_CHDJ_MJ_1_2_1 + A.SYB_CHDJ_MJ_1_2_2) AS TOTAL_AREA2 --减值损失-半年到一年-面积
	  ,SUM(A.SYB_CHDJ_MJ_1_3_1 + A.SYB_CHDJ_MJ_1_3_2) AS TOTAL_AREA3 --减值损失-一年到两年-面积
	  ,SUM(A.SYB_CHDJ_MJ_1_4_1 + A.SYB_CHDJ_MJ_1_4_2) AS TOTAL_AREA4 --减值损失-两年以上-面积
	  ,SUM(A.SYB_CHDJ_MJ_1_1_1 + A.SYB_CHDJ_MJ_1_1_2 + A.SYB_CHDJ_MJ_1_2_1 + A.SYB_CHDJ_MJ_1_2_2 + A.SYB_CHDJ_MJ_1_3_1 + A.SYB_CHDJ_MJ_1_3_2 + A.SYB_CHDJ_MJ_1_4_1 + A.SYB_CHDJ_MJ_1_4_2) AS TOTAL_AREA --面积合计
	  ,B.TOTAL_AREA AS LAST_MONTH_TOTAL_AREA --上个月面积
	  ,SUM(CASE WHEN A.FACTORY_WERKS_CODE LIKE '%80' THEN A.SYB_CHDJ_MJ_1_1_1 + A.SYB_CHDJ_MJ_1_1_2 + A.SYB_CHDJ_MJ_1_2_1 + A.SYB_CHDJ_MJ_1_2_2 + A.SYB_CHDJ_MJ_1_3_1 + A.SYB_CHDJ_MJ_1_3_2 + A.SYB_CHDJ_MJ_1_4_1 + A.SYB_CHDJ_MJ_1_4_2
	            ELSE 0
				 END) AS QC_AREA --清仓面积
	  ,SUM(A.KCJE_YM) AS TOTAL_AMOUNT --总库存金额
	  ,B.TOTAL_AMOUNT AS LAST_MONTH_TOTAL_AMOUNT --上个月总库存金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SUM(A.SYB_CHDJ_1_1 + A.SYB_CHDJ_1_2 +
	       A.SYB_CHDJ_1_3 + A.SYB_CHDJ_1_4
		  ) AS INVENTORY_IMPAIR_AMOUNT --减值损失
	  ,SUM(A.SYB_CHDJ_1_1) AS INVENTORY_IMPAIR_AMOUNT1 --减值损失-半年内-金额
	  ,SUM(A.SYB_CHDJ_1_2) AS INVENTORY_IMPAIR_AMOUNT2 --减值损失-半年到一年-金额
	  ,SUM(A.SYB_CHDJ_1_3) AS INVENTORY_IMPAIR_AMOUNT3 --减值损失-一年到两年-金额
	  ,SUM(A.SYB_CHDJ_1_4) AS INVENTORY_IMPAIR_AMOUNT4 --减值损失-两年以上-金额	  
	  ,SUM(A.SYB_CHDJ_1_1 + A.SYB_CHDJ_1_2 +
	       A.SYB_CHDJ_1_3 + A.SYB_CHDJ_1_4
		  ) - B.INVENTORY_IMPAIR_AMOUNT AS INVENTORY_IMPAIR_OCCUR_AMOUNT --减值损失-发生额
	  ,SUM(A.SYB_CHDJ_1_1) - B.INVENTORY_IMPAIR_AMOUNT1 AS INVENTORY_IMPAIR_OCCUR_AMOUNT1 --减值损失-半年内-发生额
	  ,SUM(A.SYB_CHDJ_1_2) - B.INVENTORY_IMPAIR_AMOUNT2 AS INVENTORY_IMPAIR_OCCUR_AMOUNT2 --减值损失-半年到一年-发生额
	  ,SUM(A.SYB_CHDJ_1_3) - B.INVENTORY_IMPAIR_AMOUNT3 AS INVENTORY_IMPAIR_OCCUR_AMOUNT3 --减值损失-一年到两年-发生额
	  ,SUM(A.SYB_CHDJ_1_4) - B.INVENTORY_IMPAIR_AMOUNT4 AS INVENTORY_IMPAIR_OCCUR_AMOUNT4 --减值损失-两年以上-发生额	  
  FROM DM.DM_WM_CERAMIC_STOCK_T A
  LEFT JOIN(SELECT TO_CHAR(TO_DATE(SUBSTR(STAT_MONTH,1,4)||'-'||SUBSTR(STAT_MONTH,5,2) ||'-01')- INTERVAL '-1 MONTH','YYYYMM') AS STAT_MONTH
                  ,CATEGORY_NAME
				  ,SUM(A.KCJE_YM) AS TOTAL_AMOUNT --总库存金额
				  ,SUM(A.SYB_CHDJ_MJ_1_1_1 + A.SYB_CHDJ_MJ_1_1_2 + A.SYB_CHDJ_MJ_1_2_1 + A.SYB_CHDJ_MJ_1_2_2 + A.SYB_CHDJ_MJ_1_3_1 + A.SYB_CHDJ_MJ_1_3_2 + A.SYB_CHDJ_MJ_1_4_1 + A.SYB_CHDJ_MJ_1_4_2) AS TOTAL_AREA --面积合计
	              ,SUM(A.SYB_CHDJ_1_1 + A.SYB_CHDJ_1_2 +
	                   A.SYB_CHDJ_1_3 + A.SYB_CHDJ_1_4
		              ) AS INVENTORY_IMPAIR_AMOUNT --减值损失
	              ,SUM(A.SYB_CHDJ_1_1) AS INVENTORY_IMPAIR_AMOUNT1 --减值损失-半年内-金额
	              ,SUM(A.SYB_CHDJ_1_2) AS INVENTORY_IMPAIR_AMOUNT2 --减值损失-半年到一年-金额
	              ,SUM(A.SYB_CHDJ_1_3) AS INVENTORY_IMPAIR_AMOUNT3 --减值损失-一年到两年-金额
	              ,SUM(A.SYB_CHDJ_1_4) AS INVENTORY_IMPAIR_AMOUNT4 --减值损失-两年以上-金额	
              FROM DM.DM_WM_CERAMIC_STOCK_T A
			 GROUP BY 
                   TO_CHAR(TO_DATE(SUBSTR(STAT_MONTH,1,4)||'-'||SUBSTR(STAT_MONTH,5,2) ||'-01')- INTERVAL '-1 MONTH','YYYYMM')
                  ,CATEGORY_NAME			 
           ) B
	ON A.STAT_MONTH = B.STAT_MONTH
   AND A.CATEGORY_NAME = B.CATEGORY_NAME
 GROUP BY A.STAT_MONTH --月份
	  ,A.CATEGORY_NAME --品类
	  ,B.TOTAL_AREA
	  ,B.TOTAL_AMOUNT
	  ,B.INVENTORY_IMPAIR_AMOUNT
	  ,B.INVENTORY_IMPAIR_AMOUNT1
	  ,B.INVENTORY_IMPAIR_AMOUNT2
	  ,B.INVENTORY_IMPAIR_AMOUNT3
	  ,B.INVENTORY_IMPAIR_AMOUNT4
  
 UNION ALL
