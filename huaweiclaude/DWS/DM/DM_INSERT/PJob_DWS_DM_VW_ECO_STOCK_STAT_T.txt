INSERT INTO DM.DM_VW_ECO_STOCK_STAT_T(
            STAT_MONTH
           ,FACTORY_WERKS_CODE
           ,FACTORY_WERKS_NAME
           ,INV_LOCATION_CODE
           ,INV_LOCATION_NAME
           ,MATERIAL_CODE
           ,MATERIAL_NAME
           ,IS_CLEAR
           ,IS_OEM
           ,MENU_DESC_CODE
           ,MENU_DESC_NAME
           ,TOTAL_QUANTITY
           ,TOTAL_AMOUNT
           ,TOTAL_AMOUNT_6M
           ,TOTAL_AMOUNT_12M
           ,TOTAL_AMOUNT_2Y
           ,TOTAL_AMOUNT_2NY
           ,TOTAL_AREA
           ,TOTAL_AREA_6M
           ,TOTAL_AREA_12M
           ,TOTAL_AREA_2Y
           ,TOTAL_AREA_2NY
           ,TOTAL_IMPAIR_AMOUNT
           ,DW_LAST_UPDATE_DATE

		   ,PRODUCT_AREA_NAME

SELECT 
       SUBSTR(A.CALMONTH,1,4)||'-'||SUBSTR(A.CALMONTH,5,2) AS STAT_MONTH --月份
      ,A.PLANT AS FACTORY_WERKS_CODE --工厂
      ,COALESCE(A.PLANT___T,C.FACTORY_NAME) AS FACTORY_WERKS_NAME --工厂_描述
      ,A.STOR_LOC AS INV_LOCATION_CODE --库存地点
      ,COALESCE(A.STOR_LOC___T,D.LGOBE) AS INV_LOCATION_NAME --库存地点_描述
      ,A.MATERIAL AS MATERIAL_CODE --物料编码
      ,A.MATERIAL___T AS MATERIAL_NAME --物料编码_描述
      ,A.ZISQC AS IS_CLEAR --是否清仓
	  --自产的逻辑要改为：HK和DA产区的都是自产，其他产区属于标准采购
      ,CASE WHEN SUBSTR(A.BATCH,1,2) NOT IN('HK','DA') OR A.BATCH IS NULL THEN 'Y'
	        ELSE 'N'
			 END AS IS_OEM --是否外协
	  ,B.MENU_DESC_CODE --主菜单编码
	  ,B.MENU_DESC_NAME --主菜单名称
      ,SUM(A.QUANTITY) AS TOTAL_QUANTITY --总数量
      ,SUM(A.ZSJKCJE) AS TOTAL_AMOUNT --实际库存金额
      ,SUM(A.BF_2020_WBZQ_6_AMT + A.AF_2020_WBZQ_6_AMT) AS TOTAL_AMOUNT_6M  --库存金额6个月以内
      ,SUM(A.BF_2020_WBZQ_6_12_AMT + A.AF_2020_WBZQ_6_12_AMT) AS TOTAL_AMOUNT_12M --库存金额6-12月
      ,SUM(A.BF_2020_WBZQ_12_24_AMT + A.AF_2020_WBZQ_12_24_AMT) AS TOTAL_AMOUNT_2Y --库存金额1-2年
      ,SUM(A.BF_2020_WBZQ_24_AMT + A.AF_2020_WBZQ_24_AMT) AS TOTAL_AMOUNT_2NY --库存金额2年以上	  
      ,SUM(A.QUANTITY) * B.LENGTH_1 * B.WIDTH / 1000000 AS TOTAL_AREA --总面积
      ,SUM(CASE WHEN A.BF_2020_WBZQ_6_AMT <> 0 OR A.AF_2020_WBZQ_6_AMT <> 0 THEN A.QUANTITY ELSE 0 END) * B.LENGTH_1 * B.WIDTH / 1000000 AS TOTAL_AREA_6M  --库存面积6个月以内
      ,SUM(CASE WHEN A.BF_2020_WBZQ_6_12_AMT <> 0 OR A.AF_2020_WBZQ_6_12_AMT <> 0 THEN A.QUANTITY ELSE 0 END) * B.LENGTH_1 * B.WIDTH / 1000000 AS TOTAL_AREA_12M --库存面积6-12月
      ,SUM(CASE WHEN A.BF_2020_WBZQ_12_24_AMT <> 0 OR A.AF_2020_WBZQ_12_24_AMT <> 0 THEN A.QUANTITY ELSE 0 END) * B.LENGTH_1 * B.WIDTH / 1000000 AS TOTAL_AREA_2Y --库存面积1-2年
      ,SUM(CASE WHEN A.BF_2020_WBZQ_24_AMT <> 0 OR A.AF_2020_WBZQ_24_AMT <> 0 THEN A.QUANTITY ELSE 0 END) * B.LENGTH_1 * B.WIDTH / 1000000 AS TOTAL_AREA_2NY --库存面积2年以上
      ,SUM(A.YBZQ_JC_AMT + A.WBZQ_JC_AMT) AS TOTAL_IMPAIR_AMOUNT --存货跌价准备金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,SUBSTR(A.BATCH,1,2) AS PRODUCT_AREA_CODE
	  ,E.TXTSH AS PRODUCT_AREA_NAME
  FROM DM.DM_FIN_STOCK_DETAIL_ACCAGE_T A
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
    ON A.MATERIAL = B.MATERIAL_NUM
  LEFT JOIN DWRDIM.DWR_DIM_FACTORY_D C
    ON A.PLANT = C.FACTORY_CODE
   AND C.GROUP_NUM = '800'
   AND C.END_DATE = '4712-12-31'
  LEFT JOIN(SELECT DISTINCT
                   LGORT
                  ,LGOBE
				  ,WERKS
              FROM DM.DM_SDI_T001L_1001_T
             WHERE MANDT = '800'          
           ) D
	ON A.STOR_LOC = D.LGORT
   AND A.PLANT = D.WERKS
  LEFT JOIN DM.DM_PRODUCT_AREA_D E
    ON SUBSTR(A.BATCH,1,2) = E."/bic/zww010"
 WHERE CASE
    --SA仓开头物料编码以及T09G197625_A、T09GK190001_A、T09G197625_B、T09GK190001_B算入生态新材物料，其他不算
            WHEN A.STOR_LOC = 'SA'
             AND A.MATERIAL IN('T09G197625_A','T09GK190001_A','T09G197625_B','T09GK190001_B') THEN 1
			WHEN A.STOR_LOC = 'SA'
             AND SUBSTR(A.MATERIAL,1,1) = 'S' THEN 1 
            WHEN A.STOR_LOC = 'SA' THEN 0	
			ELSE 1
			 END = 1
   AND (A.STOCKCAT <> 'K' OR A.STOCKCAT IS NULL)
   AND SUBSTR(A.CALMONTH,1,4) = SUBSTR('${PERIOD_ID_D}',1,4)
   AND ZPRODH2___T = '生态新材'
    --剔除GCSA仓，0030仓（上海仓）、GFZ（广佛子公司仓）、0020仓（深圳仓）、0019（广州仓）、0026（南宁仓）、0028（昆明新仓）、0018（西安仓）
   AND (A.STOR_LOC NOT IN('GCSA','0030','GFZ','0020','0019','0026','0028','0018') OR A.STOR_LOC IS NULL)
   --剔除子公司工厂、BY仓
   AND (SUBSTR(A.PLANT,1,2) NOT IN('80','81','82','83','85','88','8A','BY') OR A.PLANT IS NULL)  
   AND SUBSTR('${PERIOD_ID_D}',1,4) = '2022'
 GROUP BY 
       SUBSTR(A.CALMONTH,1,4)||'-'||SUBSTR(A.CALMONTH,5,2)--月份
      ,A.PLANT --工厂
      ,COALESCE(A.PLANT___T,C.FACTORY_NAME) --工厂_描述
      ,A.STOR_LOC --库存地点
      ,COALESCE(A.STOR_LOC___T,D.LGOBE) --库存地点_描述
      ,A.MATERIAL --物料编码
      ,A.MATERIAL___T --物料编码_描述
      ,A.ZISQC --是否清仓
      ,CASE WHEN SUBSTR(A.BATCH,1,2) NOT IN('HK','DA') OR A.BATCH IS NULL THEN 'Y'
	        ELSE 'N'
			 END --是否外协
	  ,B.MENU_DESC_CODE --主菜单编码
	  ,B.MENU_DESC_NAME --主菜单名称
	  ,B.LENGTH_1
	  ,B.WIDTH
	  ,SUBSTR(A.BATCH,1,2)
	  ,E.TXTSH
	  
 UNION ALL
