        STAT_MONTH

SELECT 
 A.STAT_MONTH --统计月 
,A.LEVEL_ID --部门层级ID
,A.LEV_DEPT_NAME --部门层级名称
,A.LAST_DEPT_NAME --上级部门层级名称
,A.DEPT_NAME--部门名称
,A.WF_TYPE_NAME --流程分类
,A.CURRENT_NODE_TYPE  --节点类型
,SUM(A.WF_COUNT) AS WF_COUNT --流程数
,SUM(A.NODE_COUNT) AS NODE_COUNT --节点数
,SUM(A.GZ_HOUR) AS NODE_COUNT --耗时(工作时间:小时)
,SUM(A.CS_NODE_COUNT) AS CS_NODE_COUNT --超时节点数
,SUM(A.LAST_WF_COUNT) AS LAST_WF_COUNT --上个月流程数
,SUM(A.LAST_NODE_COUNT) AS LAST_NODE_COUNT --上个月节点数
,SUM(A.LAST_GZ_HOUR) AS LAST_GZ_HOUR --上个月耗时(工作时间:小时)
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM(
SELECT 
 A.STAT_MONTH --统计月 
,A.LV AS LEVEL_ID --部门层级ID
,A.LEV_DEPT_NAME --部门层级名称
,A.LEST_DEPT_NAME AS LAST_DEPT_NAME --上级部门层级名称
,A.DEPT_NAME--部门名称
,A.WF_TYPE_NAME --流程分类
,A.CURRENT_NODE_TYPE  --节点类型
,CAST(A.WF_COUNT AS NUMERIC(38,10)) AS WF_COUNT --流程数
,CAST(A.NODE_COUNT AS NUMERIC(38,10)) AS NODE_COUNT --节点数
,CAST(A.GZ_HOUR AS NUMERIC(38,10)) AS GZ_HOUR --耗时(工作时间:小时)
,CAST(A.CS_NODE_COUNT AS NUMERIC(38,10)) AS CS_NODE_COUNT --超时节点数
,CAST(0 AS NUMERIC(38,10)) AS LAST_WF_COUNT --上个月流程数
,CAST(0 AS NUMERIC(38,10)) AS LAST_NODE_COUNT --上个月节点数
,CAST(0 AS NUMERIC(38,10)) AS LAST_GZ_HOUR --上个月耗时(工作时间:小时)
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM(SELECT lv
,lev_dept_name
,lest_dept_name
,DEPT_NAME--部门层级
,stat_month            --统计月
,wf_type_name
,current_node_type  
,WF_COUNT --流程数
,NODE_COUNT --节点数
,GZ_HOUR --耗时(工作时间:小时)
,CS_NODE_COUNT --超时节点数
FROM XXD
where stat_month = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
) A

UNION ALL

SELECT 
 CAST(A.STAT_MONTH AS INT) AS STAT_MONTH --统计月 
,A.LV AS LEVEL_ID --部门层级ID
,A.LEV_DEPT_NAME --部门层级名称
,A.LEST_DEPT_NAME AS LAST_DEPT_NAME --上级部门层级名称
,A.DEPT_NAME--部门名称
,A.WF_TYPE_NAME --流程分类
,A.CURRENT_NODE_TYPE  --节点类型
,CAST(0 AS NUMERIC(38,10)) AS WF_COUNT --流程数
,CAST(0 AS NUMERIC(38,10)) AS NODE_COUNT --节点数
,CAST(0 AS NUMERIC(38,10)) AS GZ_HOUR --耗时(工作时间:小时)
,CAST(0 AS NUMERIC(38,10)) AS CS_NODE_COUNT --超时节点数
,CAST(A.LWF_COUNT AS NUMERIC(38,10)) AS LAST_WF_COUNT --上个月流程数
,CAST(A.LNODE_COUNT AS NUMERIC(38,10)) AS LAST_NODE_COUNT --上个月节点数
,CAST(A.LGZ_HOUR AS NUMERIC(38,10)) AS LAST_GZ_HOUR --上个月耗时(工作时间:小时)
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM (SELECT lv
,lev_dept_name
,lest_dept_name
,DEPT_NAME--部门层级
,to_char(to_date(substr(stat_month,1,4)||'-'||substr(stat_month,5,2) ||'-01')+ interval '1 month','yyyymm') AS stat_month     --统计月
,wf_type_name
,current_node_type  
,WF_COUNT LWF_COUNT --流程数
,NODE_COUNT LNODE_COUNT --节点数
,GZ_HOUR LGZ_HOUR --耗时(工作时间:小时)
,CS_NODE_COUNT --超时节点数
FROM XXD
) A 
where stat_month = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
) A
GROUP BY A.STAT_MONTH --统计月 
,A.LEVEL_ID --部门层级ID
,A.LEV_DEPT_NAME --部门层级名称
,A.LAST_DEPT_NAME --上级部门层级名称
,A.DEPT_NAME--部门名称
,A.WF_TYPE_NAME --流程分类
,A.CURRENT_NODE_TYPE  --节点类型
