DELETE FROM DM.DM_RPT_DEMAND_FORECAST_T
WHERE stat_month >= 202301 
and stat_month >= 
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
    THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
        ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
         END
         ;
with A as (
select 
	COALESCE(a2.bigdimension,'其他规格') as dimension--规格划分
	,COALESCE(a2.prod_name1,'其他瓷砖') as prod_name --品类划分
	,case 
		when a1.level1 in ('战略工程中心','战略地产') then '战略工程'
		when a1.level1 in ('国际营销系统','瓷砖国际营销中心') then '国际营销' 
		when a1.level1 in ('梦之家事业部','梦之家') then '梦之家'
-- 		when a1.level1 = '幕墙工程中心' then '幕墙'
		when a1.level1 in  ('幕墙工程中心','商机资源开发部','战略客户部') then '资源开发运营部' 
		when a1.level1 = 'ART+' then 'ART+'
		else a1.level2
	end as sub_grp_name --运营中心
	,a1.calmonth stat_month
   	,case 
   		when a1.channel_desc in ('整装头部') then '整装头部'
   		when a1.channel_desc in ('工程渠道','工程') then '工程'
   		when a1.channel_desc in ('大零售渠道','大零售','零售渠道') then '零售'
   		else a1.channel_desc
   	end	CHANNEL_NAME --销售渠道	   
   ,a1.amount as zxsyj_dy--X实际达成业绩单位：万元
   ,a1.area as zxsmj_dy--X月实际提货面积单位：万方
from DM.DM_RPT_SALE_FORECAST_2023_T a1 --DM_RPT_SALE_FORECAST_2023_T
left join DM.DM_RPT_SALE_FORECAST_T_MID a2 on a1.material = a2.material_num
)
,B as(
select cast(stat_month as INTEGER) as stat_month
      ,case 
        when sales_department in ('国际营销','国贸') then '国际营销' 
        when sales_department in ('战略地产','战略工程') then '战略工程'
        when sales_department in ('战略客户部','资源开发运营部') then '资源开发运营部'
        else sales_department 
       end as sales_department --外挂表统一名称
      ,category_name as prod_name 
	  ,specifications_code as dimension
	  ,sales__channel as channel_name	  
	  ,cast(estimated_pickup_area as numeric(38,10)) as yc_area --X月预测面积单位：万方
	  ,cast(forecast_amount as numeric(38,10)) as yc_amount --X月预测业绩单位：万元
  from upload.upload_sd_forecast_aroam_t
)
,C as(
select stat_month
      ,sub_grp_name
      ,prod_name
	  ,dimension
	  ,channel_name
	  ,sum(zxsyj_dy)/10000 as zxsyj_dy --X实际达成业绩单位：万元
	  ,sum(zxsmj_dy)/10000 as zxsmj_dy--X月实际提货面积单位：万方
	  ,sum(cast(0 as numeric(38,10)))  as yc_area --X月预测面积单位：万方
	  ,sum(cast(0 as numeric(38,10))) as yc_amount --X月预测业绩单位：万元
from A
group by stat_month,sub_grp_name,prod_name,dimension,channel_name

union all

select stat_month
      ,sales_department
      ,prod_name
	  ,dimension
	  ,channel_name
	  ,sum(cast(0 as numeric(38,10))) as zxsyj_dy --X实际达成业绩单位：万元
	  ,sum(cast(0 as numeric(38,10))) as zxsmj_dy --X月实际提货面积单位：万方
	  ,sum(yc_area) as yc_area--X月预测面积单位：万方
	  ,sum(yc_amount) as yc_amount --X月预测业绩单位：万元
from B
group by stat_month,sales_department,prod_name,dimension,channel_name
)
,D AS (
select stat_month
	  --可以忽略这个字段，前端统一一下
      ,case when sub_grp_name like '%运营中%' then '运营中心'
            when sub_grp_name like '%子公司'   then '子公司'
            when sub_grp_name like '资源开发运营部'   then '工程市场中心'
            when sub_grp_name = '战略工程'   then '工程市场中心'
            when sub_grp_name = '国际营销' then '国际营销'
            when sub_grp_name = '梦之家' then '梦之家'
            when sub_grp_name = '幕墙' then '工程市场中心'
             when sub_grp_name = 'ART+' then 'ART+'
            end  sub_name
      ,sub_grp_name
      ,prod_name
	  ,dimension
	  ,channel_name
	  ,sum(zxsyj_dy) as zxsyj_dy --X实际达成业绩单位：万元
	  ,sum(zxsmj_dy) as zxsmj_dy --X月实际提货面积单位：万方
	  ,sum(yc_area) as yc_area --X月预测面积单位：万方
	  ,sum(yc_amount) as yc_amount --X月预测业绩单位：万元
	  ,abs(sum(zxsmj_dy) - sum(yc_area)) as ycpc_area--预测偏差面积
	  ,abs(sum(zxsyj_dy) - sum(yc_amount)) as ycpc_amount --预测偏差金额
	  ,'X' as FLAG
from C
WHERE 1=1
 group by 
       stat_month
      ,sub_grp_name
      ,prod_name
	  ,dimension
	  ,channel_name
)
INSERT INTO DM.DM_RPT_DEMAND_FORECAST_T
SELECT 
	cast(stat_month as INTEGER) as stat_month
	,prod_name
	,dimension 
	,zxsyj_dy  --X实际达成业绩单位：万元
	,zxsmj_dy  --X月实际提货面积单位：万方
	,yc_area  --X月预测面积单位：万方
	,yc_amount  --X月预测业绩单位：万元
	,ycpc_area --预测偏差面积
	,ycpc_amount 
	,current_timestamp as dw_last_update_date -- 数据同步时间
	,sub_name
	,sub_grp_name
	,FLAG
	,channel_name  
FROM D
WHERE FLAG = 'X' 
-- and sub_grp_name <> '国际营销'
and 
stat_month >=202301
and stat_month >= 
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
    THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
        ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
         END
         ;
