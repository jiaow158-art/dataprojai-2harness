DELETE FROM DM.DM_PAM_MAKE_PREPAREDELIVERY_T WHERE zj_month <= SUBSTRING('${PERIOD_ID_D}',1,7);
INSERT INTO DM.DM_PAM_MAKE_PREPAREDELIVERY_T
with tmp_a as
(
SELECT 
	  b.base as base_name --"基地"
	 ,a.RESOURCE_NAME --"资源名称"
	 ,ITEM_CODE as MATERIAL_NUM --"物料编码"
	 ,ITEM_PRODUCT_PROPERTY --"物料组1"、
	 ,substr(ZJHS_TIME,1,7) as zj_month
	 ,sum(PLAN_AREA) PLAN_AREA--"计划数量(平方米)"
	 ,sum(ontime_AREA1_1) ontime_AREA1_1 --"按时按质报工面积1"
	 --,sum(Submitted_AREA1) Submitted_AREA1--"准交面积1"
	 ,sum(ontime_AREA2_2)  ontime_AREA2_2--"按时按质报工面积2"
	 --,sum(Submitted_AREA2) Submitted_AREA2--"准交面积2"
	 --按时按质报工面积汇总（方）“/”计划数量（平方米）
	 ,round(sum(ontime_AREA1_1)/sum(PLAN_AREA),3) as zj_rate1 
from dm.DM_PS_BASE_QUASI_DELIVERY_T a
left join upload.upload_md_resource_base_t b
on a.RESOURCE_NAME = b.RESOURCE_NAME
where A.base not like '%OEM%'
and A.base not like '%加工%'
--AND concat(to_number(substr(zj_month,6,2),'999999999'),'月') = '{ym}'
and A.ZJHS_TIME <= '${PERIOD_ID_D}' --substr(current_date+ INTERVAL '-1 day',1,10)   -- 传昨天
group by
  b.base --"基地"
 ,a.RESOURCE_NAME --"资源名称"
 ,MATERIAL_NUM --"物料编码"
 ,ITEM_PRODUCT_PROPERTY --"物料组1"
 ,zj_month
)
select   a.base_name --'基地'
		,a.RESOURCE_NAME --'资源名称'
		,a.MATERIAL_NUM --'物料编码'
		,a.ITEM_PRODUCT_PROPERTY --'物料组1'        
		,a.zj_month --准交月份                        
		,b.dimension --规格
		,b.product_level2_name --品类
		,case when a.zj_rate1 >= c.zj_rate1 then '是'
			  else '否'
		 end as SKU_delivered_status  -- SKU_交付状态
		,a.PLAN_AREA --'计划数量(平方米)'
		,a.ontime_AREA1_1 as ontime_AREA --'按时按质报工面积'
		,case when a.ontime_AREA1_1 >= a.PLAN_AREA then PLAN_AREA
			  else a.ontime_AREA1_1 
		 end as Submitted_AREA--'准交面积'
		,current_timestamp() as dw_last_update_date -- 数据同步时间
	 from tmp_a a
  left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T b
  on a.MATERIAL_NUM = b.material_num 
  left join upload.upload_wm_item_zj_rate1_M c
  on a.zj_month = c.stat_month
  and a.ITEM_PRODUCT_PROPERTY = c.ITEM_PRODUCT_PROPERTY;
