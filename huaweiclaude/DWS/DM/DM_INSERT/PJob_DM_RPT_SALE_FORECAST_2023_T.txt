DELETE FROM DM.DM_RPT_SALE_FORECAST_2023_T
WHERE 1=1
and calmonth >= 202301
and calmonth >=
CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
    THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
        ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
        END
        ;
INSERT INTO DM.DM_RPT_SALE_FORECAST_2023_T

with data_datail as 
(
select 
    case when  substr(REPLACE(a.calmonth,'-',''),5,2)='01' then  substr(REPLACE(a.calmonth,'-',''),1,4)||'02' else REPLACE(a.calmonth,'-','') end  as calmonth,
    case
        when b.node_name3 in ('REQ') then b.node_desc3 --瓷砖国际营销中心
	    when b.node_name4 in ('H04230600', 'H04230655') then b.node_desc4
        when b.node_name3 in ('H03230617', 'H03230628') then b.node_desc3
        when b.node_name5 in ('H05240601', 'Z031', 'H05230609') then b.node_desc5
        end level1,
    case
        when b.node_name4 in ('H04230600', 'H04230655')  then b.node_desc5
        else ''
        end   level2,
    a.integrate_channel  as channel_code,
    case
       when a.integrate_channel = 'GD01' then '零售渠道'
       when a.integrate_channel = 'GD02' then '整装头部'
       when a.integrate_channel = 'GD03' then '工程渠道'
       end   channel_desc,
    a.salesorg,--销售组织代码
    a.salesorg___t,--销售组织名称
    a.sales_grp,--销售组编码
    a.sales_grp___t,--销售组名称
    a.distr_chan,--分销渠道
    a.customer,--客户编码
    a.customer___t,--客户名称
    coalesce(t4.stat_cate, '其他品类')  as prod_name,
    a.material,--产品编码
    coalesce(a.size_dim, '其他规格')  as dimension,
    a.ambperformance           amount,
    zxsmj        area,
    case
        when b.node_name4 in ('H04230600', 'H04230655') then b.node_desc4
        when b.node_name3 in ('H03230617', 'H03230628') then b.node_desc3
        when b.node_name5 in ('H05240601', 'Z031', 'H05230609') then b.node_desc5
        end ||   coalesce(t4.stat_cate, '其他品类')  || dimension                 mainid1,
    case
        when b.node_name4 in ('H04230600', 'H04230655')  then b.node_desc5
        else ''
        end ||   coalesce(t4.stat_cate, '其他品类') || dimension                 mainid2,
    current_timestamp as dw_last_update_date
from dm.dm_ambv2_zzt_xssr_t a
         left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
         --left join DM.DM_RPT_PROD_LEVEL_T c on a.ZPRODH3 = c.prod_code3 and c.PROD_CODE3 IS NOT NULL
         left join DM.DM_RPT_SALE_FORECAST_T_MID t4 on a.material = t4.material_num
where 1=1 
    and  NODE_NAME2 = 'H11000001'
  	and ((node_name4 in ('H04230600', 'H04230655') and
        (strpos(node_desc5, '子公司') > 0 or strpos(node_desc5, '运营中心') > 0))
    	or (node_name3 in ('H03230617', 'H03230628' ,'REQ'))
    	or node_name5 in ('H05240601','Z031', 'H05230609')
    )
--        其他杂类过滤
    --and (lve2_code not in ('ART+', 'DPI', '1+N产品', '其他品类') or lve3_code = '其他杂类')
    and t4.prod_property IN('A0001', 'A0002', 'C0002')
    and a.data_source <> 'U'

union all 

-- ART+
select 
    case when  substr(REPLACE(a.calmonth,'-',''),5,2)='01' then  substr(REPLACE(a.calmonth,'-',''),1,4)||'02' else REPLACE(a.calmonth,'-','') end  as calmonth,
    case
        when b.node_name5 in ('H03230627') then b.node_desc5
        when b.node_name6 in ('H03230627') then b.node_desc6
        when b.node_name7 in ('H03230627') then b.node_desc7
        end level1,
    '' as level2,
    a.integrate_channel  as channel_code,
    case
       when a.integrate_channel = 'GD01' then '大零售渠道'
       when a.integrate_channel = 'GD02' then '整装头部'
       when a.integrate_channel = 'GD03' then '工程渠道'
       end   channel_desc,
    a.salesorg,--销售组织代码
    a.salesorg___t,--销售组织名称
    a.sales_grp,--销售组编码
    a.sales_grp___t,--销售组名称
    a.distr_chan,--分销渠道
    a.customer,--客户编码
    a.customer___t,--客户名称
    coalesce(t4.stat_cate, '其他品类')  as prod_name,
    a.material,--产品编码
    coalesce(a.size_dim, '其他规格')  as dimension,
    a.ambperformance           amount,
    zxsmj  area,
    case
        when b.node_name5 in ('H03230627') then b.node_desc5
        end ||   coalesce(t4.stat_cate, '其他品类')  || dimension     mainid1,
    '' ||   coalesce(t4.stat_cate, '其他品类')  || dimension                 mainid2,
    current_timestamp as dw_last_update_date
from dm.dm_ambv2_zzt_xssr_t a
         left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
         --left join DM.DM_RPT_PROD_LEVEL_T c on a.ZPRODH3 = c.prod_code3 and c.PROD_CODE3 IS NOT NULL
         left join DM.DM_RPT_SALE_FORECAST_T_MID t4 on a.material = t4.material_num
where 1=1 
and  NODE_NAME2 = 'H11000001'
and (node_name5 in ('H03230627') or node_name6 in ('H03230627') or node_name7 in ('H03230627'))
and a.data_source <> 'U'
)


select 
	calmonth,
	level1,
	level2,
	channel_code,
	channel_desc,
	salesorg,
	salesorg___t,
	sales_grp,
	sales_grp___t,
	distr_chan,
	customer,
	customer___t,
	prod_name,
	material,
	dimension,
	sum(amount) as amount,
	sum(area) as area,
	mainid1,
	mainid2,
	dw_last_update_date
from data_datail
where 1=1
    and REPLACE(calmonth,'-','') >= '202301'
    and REPLACE(calmonth,'-','') >= 
        CASE WHEN SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',9,2) <= '06'
            THEN REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 MONTH',1,7),'-','')
            ELSE REPLACE(SUBSTR(CURRENT_DATE - INTERVAL '1 DAY',1,7),'-','')
        END
group by 
	calmonth,
	level1,
	level2,
	channel_code,
	channel_desc,
	salesorg,
	salesorg___t,
	sales_grp,
	sales_grp___t,
	distr_chan,
	customer,
	customer___t,
	prod_name,
	material,
	dimension,
	mainid1,
	mainid2,
	dw_last_update_date
