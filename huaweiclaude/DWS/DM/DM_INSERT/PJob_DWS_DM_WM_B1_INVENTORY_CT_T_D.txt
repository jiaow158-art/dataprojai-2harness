DELETE FROM DM.DM_WM_B1_INVENTORY_CT_T_D
where "日期" IN
(
  select day_dt from dm.dm_dim_date_d where  day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) and day_dt >= substr(to_date('${PERIOD_ID_D}') + interval '-3 day',1,10)

INSERT INTO DM.DM_WM_B1_INVENTORY_CT_T_D
with dm_b1_customer_data AS
(
  select distinct customer_code,customer_type,customer_flag from dm.dm_b1_customer_data_t
),
oivl as (
	select 
		 "docdate"
		,"transseq"
		,"loccode"
		,"transtype"
		,"itemcode"
		,"messageid"
		,sum("inqty") "inqty"
		,sum("outqty") "outqty" 
		,sum("inqty")-sum("outqty")
		,t1."customer_code" as customer_code
		,createdate 
	from dwiotd.dwi_wm_b1_oivl_t t1 
	join dm_b1_customer_data t3 
	on t1.customer_code =t3.customer_code 
	and t3.customer_type ='CT' 
	and t3.customer_flag ='N'   
	and t1.customer_type=t3.customer_type
	where 1=1 
	group by "docdate",
			 "transseq",
			 "loccode",
			 "transtype",
			 "itemcode",
			 "messageid",
			 t1."customer_code",
			 createdate 
),
ivl1 as (
	select transseq,
		  t1.customer_code ,
		  sum(transvalue) transvalue  
	from dwiotd.dwi_wm_b1_ivl1_t  t1 
	join dm_b1_customer_data t2 
	on t1.customer_code =t2.customer_code 
	and t2.customer_type ='CT' 
	and t2.customer_flag ='N'   
	and t1.customer_type=t2.customer_type
	group by transseq,
			t1.customer_code 
),
oitm as (
	select itemcode,
		   itemname,
		   codebars,
		   frgnname,
		   suppcatnum,
		   invntryuom,
		   cardcode,
		   itmsgrpcod,
		   firmcode,
		   slength1,
		   swidth1,
		   t1.customer_code 
	  from dwiotd.dwi_wm_b1_oitm_t t1 
	  join dm_b1_customer_data t2 
	  on t1.customer_code =t2.customer_code 
	  and t1.customer_type = t2.customer_type
	  and t2.customer_type ='CT' 
	  and t2.customer_flag ='N'
	  group by itemcode,
			   itemname,
			   codebars,
			   frgnname,
			   suppcatnum,
			   invntryuom,
			   cardcode,
			   itmsgrpcod,
			   firmcode,
			   slength1,
			   swidth1,
			   t1.customer_code 
),
oitb as (
	select itmsgrpcod,
		  itmsgrpnam,
		  t1.customer_code  
	from dwiotd.dwi_wm_b1_oitb_t t1 
	join dm_b1_customer_data t2 
	on t1.customer_code =t2.customer_code 
	and t1.customer_type=t2.customer_type 
	and t2.customer_type ='CT' 
	and t2.customer_flag ='N'
	group by itmsgrpcod,
			 itmsgrpnam,
			 t1.customer_code
),
omrc as (
	select firmcode,
		   firmname,
		   t1.customer_code  
	from dwiotd.dwi_wm_b1_omrc_t  t1 
	join dm_b1_customer_data t2 
	on t1.customer_code =t2.customer_code 
	and t1.customer_type=t2.customer_type 
	and t2.customer_type ='CT' 
	and t2.customer_flag ='N'
	group by  firmcode,
			  firmname,
			  t1.customer_code 
),
oilm as (
	select messageid,
		   effectqty,
		   bpcardcode,
		   t1.customer_code  
	from dwiotd.dwi_wm_b1_oilm_t t1 
	join dm_b1_customer_data t2 
	on t1.customer_code =t2.customer_code 
	and t2.customer_type ='CT' 
	and t2.customer_flag ='N'  
	and t1.customer_type=t2.customer_type
	 group by messageid,
			  effectqty,
			  bpcardcode,
			  t1.customer_code 
),
ocrd as (
	select cardcode,
		   cardname,
		   t1.customer_code  
	from dwiotd.dwi_wm_b1_ocrd_t t1 
	join dm_b1_customer_data t2 
	on t1.customer_code =t2.customer_code 
	and t2.customer_type ='CT' 
	and t2.customer_flag ='N'   
	and t1.customer_type=t2.customer_type
	 group by cardcode,
			  cardname,
			  t1.customer_code 
), 
dt as (
	select day_dt 
	from dm.dm_dim_date_d 
	where  day_dt <= SUBSTR('${PERIOD_ID_D}',1,10) 
	and day_dt >= substr(to_date(SUBSTR('${PERIOD_ID_D}',1,10)) + interval '-3 day',1,10)
	order by day_dt 
) ,
pdn1 AS (
	 select 
		 t1.customer_code
		,t3.itemcode
		,t3.dscription
		,substr(dt.day_dt,1,10) day_dt
		,t1.docdate
		,row_number() over(partition by  t1.customer_code,t3.itemcode,dt.day_dt order by substr(t1.docdate,1,10)  desc ) rk 
	 from DWIOTD.DWI_WM_b1_OPDN_T t1  
	 join dm_b1_customer_data t2 on t1.customer_code =t2.customer_code and t2.customer_type ='CT' and t2.customer_flag ='N' and t1.customer_type=t2.customer_type
	 join DWIOTD.DWI_WM_b1_PDN1_T t3 on t1.customer_code =t3.customer_code and t1.docentry =t3.docentry
	 join  dt on 1=1 
	 where 1=1  
	 and t1.docdate<=dt.day_dt
),
tmp1 as (
		  select 
		       dt.day_dt,
			   case when t0.docdate< dt.day_dt then 'qc' 
					when t0.docdate  >=  dt.day_dt and  t0.docdate< substr(to_date(dt.day_dt ) + interval '+1 day',1,10)then 'qj' 
					else 'qm'  end as dttype,       
				t0.itemcode,
				t2.itemname,
				t2.codebars,
				t2.frgnname,
				t2.suppcatnum,
				t2.invntryuom,
				t3.itmsgrpnam,
				t4.firmname,
				t7.cardcode,
				t7.cardname,
				t0.inqty,
				t0.outqty,       
				case when nvl(t1.transvalue,0.0)>0.0 then t1.transvalue 
					 else 0.0 
				end as inlinebalance,       
				case when nvl(t1.transvalue,0.0)<0.0 then -t1.transvalue 
					 else 0.0 
				end as outlinebalance,	       
				case when (t0.transtype=18 or t0.transtype=20) then t0.inqty 
					 else 0.0 
				end as puch_inqty,        
				case when (t0.transtype=14 or t0.transtype=16) then t0.inqty 
					else 0.0
				end as sale_inqty,       
				case when nvl(t1.transvalue,0.0)>=0.0 and t0.transtype<>18 and t0.transtype<>20 and t0.transtype<>14 and t0.transtype<>16 then t0.inqty 
					else 0.0 
				end as other_inqty,        
				case when (t0.transtype=19 or t0.transtype=21) then t0.outqty 
					else 0.0 
				end as puch_outqty,        
				case when (t0.transtype=13 or t0.transtype=15) then t0.outqty 
					else 0.0 
				end as sale_outqty,       
				case when nvl(t1.transvalue,0.0)<=0.0 and t0.transtype<>19 and t0.transtype<>21 and t0.transtype<>13 and t0.transtype<>15 then t0.outqty 
					else 0.0 
				end as other_outqty,     
				t1.transvalue as linebalance,
				t0.loccode as loccode ,
				t0.customer_code      
		from oivl t0  
		inner join ivl1 t1 on t0.transseq = t1.transseq  and t0.customer_code = t1.customer_code         
		inner join oitm t2  on t0.itemcode = t2.itemcode  and t0.customer_code = t2.customer_code      
		inner join oitb t3  on t2.itmsgrpcod=t3.itmsgrpcod  and t2.customer_code = t3.customer_code
		inner join omrc t4  on t2.firmcode=t4.firmcode    and t2.customer_code = t4.customer_code
		left join oilm t5 on nvl(t0.messageid,'') = nvl(t5.messageid,'') and nvl(t5.effectqty,0.0)<>0.0      and t0.customer_code = t5.customer_code
		left join ocrd t6 on nvl(t5.bpcardcode,'') = nvl(t6.cardcode,'')    and t5.customer_code = t6.customer_code   
		left join ocrd t7 on nvl(t2.cardcode,'')=nvl(t7.cardcode,'')        and t2.customer_code = t7.customer_code 
	    join dt on 1=1 
),
tmp as (
    select 
     t.day_dt
    ,t.dttype
    ,t.itemcode
    ,t.itemname
    ,t.codebars
    ,t.frgnname
    ,t.suppcatnum
    ,t.invntryuom
    ,t.itmsgrpnam
    ,t.firmname
    ,t.loccode
    ,t.cardcode
    ,t.cardname
    , t.customer_code
    ,sum(case when t.dttype='qc' then nvl(t.inqty,0.0) else 0.0 end) as qcinqty
    ,sum(case when t.dttype='qj' then nvl(t.inqty,0.0) else 0.0 end) as qjinqty
    ,sum(case when t.dttype='qm' then nvl(t.inqty,0.0) else 0.0 end) as qminqty
    ,sum(case when t.dttype='qc' then nvl(t.outqty,0.0) else 0.0 end) as qcoutqty
    ,sum(case when t.dttype='qj' then nvl(t.outqty,0.0) else 0.0 end) as qjoutqty
    ,sum(case when t.dttype='qm' then nvl(t.outqty,0.0) else 0.0 end) as qmoutqty
    ,sum(case when t.dttype='qj' then nvl(t.inlinebalance,0.0) else 0.0 end )as qjinlinebalance
    ,sum(case when t.dttype='qj' then nvl(t.outlinebalance,0.0) else 0.0 end) as qjoutlinebalance
    ,sum(case when t.dttype='qc' then nvl(t.linebalance,0.0) else 0.0 end) as qclinebalance
    ,sum(case when t.dttype='qj' then nvl(t.linebalance,0.0) else 0.0 end) as qjlinebalance
    ,sum(case when t.dttype='qm' then nvl(t.linebalance,0.0) else 0.0 end) as qmlinebalance
    ,sum(case when t.dttype='qj' then nvl(t.puch_inqty,0.0) else 0.0 end) as qjpuch_inqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_inqty,0.0) else 0.0 end) as qjsale_inqty
    ,sum(case when t.dttype='qj' then nvl(t.other_inqty,0.0) else 0.0 end) as qjother_inqty
    ,sum(case when t.dttype='qj' then nvl(t.puch_outqty,0.0) else 0.0 end)as qjpuch_outqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_outqty,0.0) else 0.0 end) as qjsale_outqty
    ,sum(case when t.dttype='qj' then nvl(t.other_outqty,0.0) else 0.0 end) as qjother_outqty
      from tmp1 t  
	   group by t.itemcode,
				t.itemname,
				t.codebars,
				t.frgnname,
				t.suppcatnum,
				t.invntryuom,
				t.itmsgrpnam,
				t.firmname,
				t.loccode,
				t.cardcode,
				t.cardname,
				t.customer_code,
				t.dttype,
				t.day_dt
)
select  
	 tt.day_dt as  "日期"
	,tt.customer_code "客户别名"
	,tt.itemcode     "物料编码"
	,tt.itemname     "物料描述"
	,t5.material_type "分类"
	,t5.product_class "品类"
	,t5.analyze_product_class "分析品类"
	,sum(nvl(tt.qcinqty,0.0))-sum(nvl(tt.qcoutqty,0.0)) as "期初数量"
	,sum(nvl(tt.qjinqty,0.0)) as "本期入库"
	,sum(nvl(tt.qjpuch_inqty,0.0)) as "期间采购收货数量"
	,sum(nvl(tt.qjsale_inqty,0.0)) as "期间销售退货数量"
	,sum(nvl(tt.qjother_inqty,0.0)) as "期间其他入库数量"
	,sum(nvl(tt.qjpuch_outqty,0.0)) as "期间采购退货数量"
	,sum(nvl(tt.qjoutqty,0.0)) as "本期出库"
	,sum(nvl(tt.qjsale_outqty,0.0)) as "期间销售出库数量"
	,sum(nvl(tt.qjother_outqty,0.0)) as "期间其他出库数量"
	,(sum(nvl(tt.qcinqty,0.0))-sum(nvl(tt.qcoutqty,0.0)))+(sum(nvl(tt.qjinqty,0.0))-sum(nvl(tt.qjoutqty,0.0))) as "期末数量"
	,sum(nvl(tt.qjinqty,0.0))-sum(nvl(tt.qjoutqty,0.0)) as "期间发生"
	,sum(nvl(tt.qclinebalance,0)) as "期初金额"
	,sum(nvl(tt.qjinlinebalance,0.0)) as "本期入库金额"
	,sum(nvl(tt.qjoutlinebalance,0.0)) as "本期出库金额"
	,sum(nvl(tt.qclinebalance,0.0))+sum(nvl(tt.qjlinebalance,0.0)) as "期末金额"
	,sum(nvl(tt.qjlinebalance,0.0)) as "本期发生金额"
	,t8.docdate "最末入库日期"
	,t9.material_id "物料ID"
    ,t9.material_name "SAP物料名称"
    ,t9.matl_type_code "SAP物料类型"
    ,t9.matl_type_name "SAP物料类型名称"
    ,t9.product_group_code "SAP产品组"
    ,t9.product_group_name "SAP产品组名称"
    ,t9.product_classify_code    "SAP产品分级"
    ,t9.product_classify_name    "SAP产品分级名称"
    ,t9.product_level_code       "SAP层次代码"
    ,t9.product_level_name       "SAP层次名称"
    ,t9.product_level1_code      "SAP层次1"
    ,t9.product_level1_name      "SAP层次1名称"
    ,t9.product_level2_code      "SAP层次2"
    ,t9.product_level2_name      "SAP层次2名称"
    ,t9.product_level3_code      "SAP层次3"
    ,t9.product_level3_name      "SAP层次3名称"
    ,t9.product_level4_code      "SAP层次4"
    ,t9.product_level4_name      "SAP层次4名称"
	,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	,'' as "体积"
from  tmp  tt 
    join oitm t4  on tt.itemcode=t4.itemcode AND tt.customer_code  =t4.customer_code 
    left join  upload.upload_vm_b1_inv_cate_match_t  t5 on tt.itemcode = t5.material_num
    left join  dm.dm_rpt_bi0_pmaterial_t t6 on tt.itemcode = t6."MATERIAL" 
    left join  DM.DM_RPT_PROD_CATEGORY_T t7 on tt.itemcode = t7."prod_level3_code" 
    left join pdn1 t8 on t4.itemcode=t8.itemcode AND t4.customer_code  =t8.customer_code and tt.day_dt = t8.day_dt and rk = 1 
    left join dwimd.dwi_md_data_material_general_t_his t9 on tt.itemcode = t9.material_num 
    and  t9.end_date > '${PERIOD_ID_D}'   and t9.start_date <= '${PERIOD_ID_D}'
   --and  t9.end_date > tt.day_dt   and t9.start_date <= tt.day_dt
group by  tt.customer_code ,
		  tt.itemcode,
		  tt.itemname,
		  t5.material_type,
		  t5.product_class ,
		  t5.analyze_product_class ,
		  t8.docdate,
		  t9.material_id ,
          t9.material_name ,
          t9.matl_type_code ,
          t9.matl_type_name ,
          t9.product_group_code ,
          t9.product_group_name ,
          t9.product_classify_code  ,
          t9.product_classify_name ,
          t9.product_level_code ,
          t9.product_level_name ,
          t9.product_level1_code ,
          t9.product_level1_name ,
          t9.product_level2_code ,
          t9.product_level2_name ,
          t9.product_level3_code ,
          t9.product_level3_name ,
          t9.product_level4_code ,
          t9.product_level4_name ,
		  tt.day_dt;
