-- 逻辑修改：十城建单数、十城线下建单数、十城400建单数 要取建单日期在2023-11-16及以后 2023-12-07 黄文彬
-- 逻辑修改：十城的完工数和签单数也保持与建单数一样的逻辑 2023-12-07 黄文彬

TRUNCATE TABLE DM.DM_CS_ZDJ_ORDER_STAT_T;

INSERT INTO DM.DM_CS_ZDJ_ORDER_STAT_T(
       STAT_MONTH --统计月份
	  ,ORDER_CNT --签单数
	  ,XX_ORDER_CNT --线下签单数
	  ,F400_ORDER_CNT --400签单数
	  ,XS_ORDER_CNT --线上签单数
	  ,DATA_TYPE --统计类型
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SC_ORDER_CNT --10城建单数
	  ,SC_XX_ORDER_CNT --10城线下建单数
	  ,SC_F400_ORDER_CNT --10城400建单数
)
SELECT SUBSTR(A.ORDER_SIGN_TIME,1,7) AS STAT_MONTH --统计月份
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL THEN ORDER_NUM END) AS ORDER_CNT --签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND A.SOURCE_CODE = 2 THEN ORDER_NUM END) AS XX_ORDER_CNT --线下签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND A.SOURCE_CODE = 4 THEN ORDER_NUM END) AS F400_ORDER_CNT --400签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL) THEN ORDER_NUM END) AS XS_ORDER_CNT --线上签单数
	  ,CAST('签单' AS VARCHAR) AS DATA_TYPE --统计类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  --新零售订单总数中，建单日期在2023-11-16及以后，成都市、福州市、南昌市、上海市、沈阳市、温州市、西安市、烟台市、长沙市、郑州市的订单数之和
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL 
	                        AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_ORDER_CNT --10城签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL 
	                        AND A.SOURCE_CODE = 2 
						    AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_XX_ORDER_CNT --十城线下签单数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL
	                        AND A.SOURCE_CODE = 4 
						    AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_F400_ORDER_CNT --十城400签单数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY
       SUBSTR(A.ORDER_SIGN_TIME,1,7)
	   
 UNION ALL

SELECT SUBSTR(A.ORDER_OVER_TIME,1,7) AS STAT_MONTH --统计月份 
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND A.STAGE_NAME IN('待验收','已完成') THEN ORDER_NUM END) AS OVER_ORDER_CNT --完工数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL
	                        AND A.SOURCE_CODE = 2 
						    AND A.STAGE_NAME IN('待验收','已完成')	
						   THEN ORDER_NUM END) AS XX_ORDER_CNT --线下完工数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL 
	                        AND A.SOURCE_CODE = 4 
							AND A.STAGE_NAME IN('待验收','已完成')
						   THEN ORDER_NUM END) AS F400_ORDER_CNT --400完工数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL
	                        AND (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL)
							AND A.STAGE_NAME IN('待验收','已完成')
						   THEN ORDER_NUM END) AS XS_ORDER_CNT --线上完工数
	  ,CAST('完工' AS VARCHAR) AS DATA_TYPE --统计类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  --新零售订单总数中，建单日期在2023-11-16及以后，成都市、福州市、南昌市、上海市、沈阳市、温州市、西安市、烟台市、长沙市、郑州市的订单数之和
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL AND A.STAGE_NAME IN('待验收','已完成')
						    AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_ORDER_CNT --10城完工数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL
	                        AND A.SOURCE_CODE = 2 
						    AND A.STAGE_NAME IN('待验收','已完成')	
						    AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_XX_ORDER_CNT --十城线下完工数
	  ,COUNT(DISTINCT CASE WHEN CNTC_CODE IS NOT NULL 
	                        AND A.SOURCE_CODE = 4 
							AND A.STAGE_NAME IN('待验收','已完成')
						    AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_F400_ORDER_CNT --十城400完工数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY
       SUBSTR(A.ORDER_OVER_TIME,1,7)
	   
 UNION ALL

SELECT SUBSTR(A.CREATE_TIME,1,7) AS STAT_MONTH --统计月份 
      ,COUNT(DISTINCT ORDER_NUM) AS ORDER_CNT --建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE = 2 THEN ORDER_NUM END) AS XX_ORDER_CNT --线下建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE = 4 THEN ORDER_NUM END) AS F400_ORDER_CNT --400建单数
	  ,COUNT(DISTINCT CASE WHEN (A.SOURCE_CODE <> 2 OR A.SOURCE_CODE IS NULL) THEN ORDER_NUM END) AS XS_ORDER_CNT --线上建单数
      ,CAST('建单' AS VARCHAR) AS DATA_TYPE --统计类型
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  --新零售订单总数中，建单日期在2023-11-16及以后，成都市、福州市、南昌市、上海市、沈阳市、温州市、西安市、烟台市、长沙市、郑州市的订单数之和
	  ,COUNT(DISTINCT CASE WHEN CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_ORDER_CNT --十城建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE = 2
	                        AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
						    AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_XX_ORDER_CNT --十城线下建单数
	  ,COUNT(DISTINCT CASE WHEN A.SOURCE_CODE = 4
	                        AND CITY_NAME IN('成都市','福州市','南昌市','上海市','沈阳市','温州市','西安市','烟台市','长沙市','郑州市')
	                        AND SUBSTR(A.CREATE_TIME,1,10) >= '2023-11-16'
						   THEN ORDER_NUM END) AS SC_F400_ORDER_CNT --十城400建单数						   
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE STAGE_NAME NOT IN('废弃订单')
 GROUP BY
       SUBSTR(A.CREATE_TIME,1,7)
