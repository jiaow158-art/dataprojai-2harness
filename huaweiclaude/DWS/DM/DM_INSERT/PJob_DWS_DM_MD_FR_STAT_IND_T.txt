DELETE FROM DM.DM_MD_FR_STAT_IND_T
 WHERE STAT_MONTH = '${PERIOD_ID_M}'
;

WITH T1 AS(
SELECT SUBSTR(BEGIN_TIME,1,7) STAT_MONTH --统计月份
      ,COUNT(1) VISIT_CNT --访问次数
      ,COUNT(DISTINCT USERNAME) AS VISIT_USER_CNT --访问人数
  FROM DWIMD.DWI_MD_FR_FINE_RECORD_EXECUTE_T A
 WHERE SUBSTR(BEGIN_TIME,1,7) = '${PERIOD_ID_M}'
 GROUP BY SUBSTR(BEGIN_TIME,1,7))
,T2 AS(
SELECT '${PERIOD_ID_M}' AS STAT_MONTH --统计月份
      ,COUNT(1) AS IND_CNT --指标数
  FROM DM.DM_OA_UF_INDEXLIBRARY_t
  WHERE BGLX IN ('0','1')
)
,T3 AS(
SELECT '${PERIOD_ID_M}' AS STAT_MONTH --统计月份
      ,COUNT(DISTINCT REPORT_PATH) AS RPT_CNT --报表数
  FROM DM.DM_MD_FR_REPORT_INFO_T
 WHERE SUBSTR(REPORT_PATH,1,3) = 'PC端' 
)
INSERT INTO DM.DM_MD_FR_STAT_IND_T(
       STAT_MONTH
	  ,VISIT_CNT
	  ,VISIT_USER_CNT
	  ,IND_CNT
	  ,RPT_CNT
	  ,DW_LAST_UPDATE_DATE
)
SELECT A.STAT_MONTH
	  ,A.VISIT_CNT
	  ,A.VISIT_USER_CNT
	  ,B.IND_CNT
	  ,C.RPT_CNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T1 A
  LEFT JOIN T2 B
    ON A.STAT_MONTH = B.STAT_MONTH
  LEFT JOIN T3 C
    ON A.STAT_MONTH = C.STAT_MONTH
