TRUNCATE TABLE DM.DM_MD_FR_REPORT_INFO_T;

INSERT INTO DM.DM_MD_FR_REPORT_INFO_T
WITH T0 AS(
SELECT SPLIT_PART(FULLPATH,'-_-',1) C0
      ,SPLIT_PART(FULLPATH,'-_-',2) C1
      ,SPLIT_PART(FULLPATH,'-_-',3) C2
      ,SPLIT_PART(FULLPATH,'-_-',4) C3
	  ,SPLIT_PART(FULLPATH,'-_-',5) C4
      ,SPLIT_PART(FULLPATH,'-_-',6) C5
	  ,SPLIT_PART(FULLPATH,'-_-',7) C6
      ,SPLIT_PART(FULLPATH,'-_-',8) C7
	  ,SPLIT_PART(FULLPATH,'-_-',9) C8
      ,*
FROM DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T
WHERE DEVICETYPE <> 0
),T1 AS(
SELECT B.DISPLAYNAME AS DISPLAYNAME1
      ,C.DISPLAYNAME AS DISPLAYNAME2
	  ,D.DISPLAYNAME AS DISPLAYNAME3
	  ,E.DISPLAYNAME AS DISPLAYNAME4
	  ,F.DISPLAYNAME AS DISPLAYNAME5
	  ,G.DISPLAYNAME AS DISPLAYNAME6
	  ,H.DISPLAYNAME AS DISPLAYNAME7
	  ,I.DISPLAYNAME AS DISPLAYNAME8
	  ,A.*
  FROM T0 A
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T B
    ON A.C1 = B.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T C
    ON A.C2 = C.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T D
    ON A.C3 = D.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T E
    ON A.C4 = E.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T F
    ON A.C5 = F.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T G
    ON A.C6 = G.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T H
    ON A.C7 = H.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_AUTHORITY_OBJECT_T I
    ON A.C8 = I.ID
),T2 AS(
SELECT CASE WHEN DISPLAYNAME8 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3||'/'||DISPLAYNAME4||'/'||DISPLAYNAME5||'/'||DISPLAYNAME6||'/'||DISPLAYNAME7||'/'||DISPLAYNAME8
			WHEN DISPLAYNAME7 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3||'/'||DISPLAYNAME4||'/'||DISPLAYNAME5||'/'||DISPLAYNAME6||'/'||DISPLAYNAME7
			WHEN DISPLAYNAME6 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3||'/'||DISPLAYNAME4||'/'||DISPLAYNAME5||'/'||DISPLAYNAME6
			WHEN DISPLAYNAME5 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3||'/'||DISPLAYNAME4||'/'||DISPLAYNAME5
			WHEN DISPLAYNAME4 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3||'/'||DISPLAYNAME4
			WHEN DISPLAYNAME3 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2||'/'||DISPLAYNAME3
			WHEN DISPLAYNAME2 IS NOT NULL
            THEN DISPLAYNAME1||'/'||DISPLAYNAME2
			WHEN DISPLAYNAME1 IS NOT NULL
            THEN DISPLAYNAME1
			 END AS REPORT_FULL_NAME
	   ,ID
	   ,C0
	   ,C1	   
	   ,C2	   
	   ,C3	   
	   ,C4	   
	   ,C5	   
	   ,C6	   
	   ,C7	   
	   ,C8	   	   
       ,DISPLAYNAME AS REPORT_NAME
       ,PATH AS REPORT_PATH
	   ,DISPLAYNAME1 AS DEPT_NAME
	   ,EXPANDID
  FROM T1
 WHERE PATH IS NOT NULL
)
SELECT A.ID AS REPORT_ID
	  ,ROW_NUMBER() OVER(ORDER BY A.REPORT_FULL_NAME ASC) AS REPORT_NUM
      ,SUBSTR(A.DEPT_NAME,3,90) AS DEPT_NAME
	  ,A.REPORT_FULL_NAME
	  ,A.REPORT_NAME
	  ,A.REPORT_PATH
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,A.C0 AS ROOT_ID
	  ,A.C1 AS ROOT_ID1
	  ,A.C2 AS ROOT_ID2
	  ,A.C3 AS ROOT_ID3
	  ,A.C4 AS ROOT_ID4
	  ,A.C5 AS ROOT_ID5
	  ,A.C6 AS ROOT_ID6
	  ,A.C7 AS ROOT_ID7
	  ,A.C8 AS ROOT_ID8
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',1) AS ROOT_NAME1
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',2) AS ROOT_NAME2
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',3) AS ROOT_NAME3
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',4) AS ROOT_NAME4
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',5) AS ROOT_NAME5
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',6) AS ROOT_NAME6
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',7) AS ROOT_NAME7
	  ,SPLIT_PART(A.REPORT_FULL_NAME,'/',8) AS ROOT_NAME8
      ,CASE WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('FI','SA','PR','SY','HR')
	        THEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,5)
            ELSE NULL
			 END AS REPORT_CODE --报表编码
	  ,SUBSTR(B.UPDATE_DATE,1,10) AS ONLINE_DATE --上线日期
	  ,CASE WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('FI','SA','PR','SY','HR')
	        THEN SUBSTR(SPLIT_PART(REPLACE(REPLACE(REPLACE(UPPER(A.REPORT_PATH),'.CPTX',''),'.FRM',''),'.CPT',''),'/',LENGTH(REPLACE(REPLACE(REPLACE(UPPER(A.REPORT_PATH),'.CPTX',''),'.FRM',''),'.CPT','')) - LENGTH(REPLACE(REPLACE(REPLACE(REPLACE(UPPER(A.REPORT_PATH),'.CPTX',''),'.FRM',''),'.CPT',''),'/','')) + 1),6,500)
	         END AS REPORT_TEMPL_NAME --模板名称
	  ,CASE WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('FI') THEN '财经'
	        WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('SA') THEN '研销'
			WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('PR') THEN '供应链'
			WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('SY') THEN '系统'
			WHEN SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,2) IN('HR') THEN '人资'
			 END REPORT_TYPE --报表类型
      ,B.REQUIRER --需求确认人
	  ,B.DEVELOPER --开发人员
	  ,A.EXPANDID --扩展ID
	  ,C.SHOWTYPE AS SHOW_TYPE_CODE --预览类型编码
	  ,CASE WHEN C.SHOWTYPE = '0' THEN '填报'
	        WHEN C.SHOWTYPE = '1' THEN '分页'
			WHEN C.SHOWTYPE = '2' THEN '分析'
			WHEN C.SHOWTYPE = '3' THEN '表单预览'
			WHEN C.SHOWTYPE = '4' THEN '新填报'
			WHEN C.SHOWTYPE = '5' THEN '预览'
             END AS SHOW_TYPE_NAME --预览类型描述
  FROM T2 A
  LEFT JOIN (SELECT CONCAT(REPORT_TYPE,REPORT_NUM) AS REPORT_CODE
                   ,ROW_NUMBER() OVER(PARTITION BY CONCAT(REPORT_TYPE,REPORT_NUM)
				                          ORDER BY UPDATE_DATE ASC
				                     ) AS RN
				   ,*					 
			   --,MIN(CASE WHEN UPDATE_TYPE = '上线' THEN SUBSTR(UPDATE_DATE,1,10) ELSE NULL END) AS ONLINE_DATE --上线日期
               FROM UPLOAD.UPLOAD_REPORT_UPDATE_LOG_T
              WHERE UPDATE_TYPE IN('上线')
			) B
	ON B.REPORT_CODE = SUBSTR(SPLIT_PART(A.REPORT_PATH,'/',LENGTH(A.REPORT_PATH) - LENGTH(REPLACE(A.REPORT_PATH,'/','')) + 1),1,5)
   AND B.RN = 1
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_REPORT_EXPAND_T C
    ON A.EXPANDID = C.ID
 WHERE A.REPORT_FULL_NAME NOT LIKE '测试%' 
;
