DELETE FROM DM.DM_RPT_SALE_GRP_T;
with t1 as(
SELECT H."NODENAME",
CASE 
WHEN H."IOBJNM" = '0SALES_GRP' THEN ST.TXTSH
WHEN H."IOBJNM" = '0SALES_OFF' THEN ST1."TXTSH" ELSE R."TXTSH"
END AS NODEDESC,
H."NODEID",
H."PARENTID" , 
H."TLEVEL",
H."IOBJNM"
FROM DM.dm_rpt_bi0_hsales_grp_t AS H
LEFT JOIN dm.DM_RPT_BI0_TSALES_GRP_T AS ST ON H."NODENAME" = ST.SALES_GRP AND ST.LANGU = '1' AND H."IOBJNM" = '0SALES_GRP'
LEFT JOIN dm.dm_rpt_bi0_tsales_off_t AS ST1 ON H."NODENAME" = ST1."SALES_OFF" AND ST1."LANGU" = '1' AND H."IOBJNM" = '0SALES_OFF'
LEFT JOIN dm.dm_rpt_rsthiernode_t R ON R."NODENAME" = H."NODENAME" AND R."OBJVERS" = 'A' AND R."HIEID" = (select HIEID from DM.DM_RPT_RSHIEDIR_T where hienm = 'DP_JTJG_2021')/*替换 DP_JTJG_2021 */ AND R."LANGU" = '1'
WHERE H."HIEID" = (select HIEID from DM.DM_RPT_RSHIEDIR_T where hienm = 'DP_JTJG_2021')) /*替换 DP_JTJG_2021 */
INSERT INTO DM.DM_RPT_SALE_GRP_T
SELECT 
DISTINCT
CASE 
WHEN H15."IOBJNM" = '0SALES_GRP' THEN H15."NODENAME"
WHEN H14."IOBJNM" = '0SALES_GRP' THEN H14."NODENAME"
WHEN H13."IOBJNM" = '0SALES_GRP' THEN H13."NODENAME"
WHEN H12."IOBJNM" = '0SALES_GRP' THEN H12."NODENAME"
WHEN H11."IOBJNM" = '0SALES_GRP' THEN H11."NODENAME"
WHEN H10."IOBJNM" = '0SALES_GRP' THEN H10."NODENAME"
WHEN H9."IOBJNM" = '0SALES_GRP' THEN H9."NODENAME"
WHEN H8."IOBJNM" = '0SALES_GRP' THEN H8."NODENAME"
WHEN H7."IOBJNM" = '0SALES_GRP' THEN H7."NODENAME"
WHEN H6."IOBJNM" = '0SALES_GRP' THEN H6."NODENAME"
WHEN H5."IOBJNM" = '0SALES_GRP' THEN H5."NODENAME"
WHEN H4."IOBJNM" = '0SALES_GRP' THEN H4."NODENAME"
WHEN H3."IOBJNM" = '0SALES_GRP' THEN H3."NODENAME"
WHEN H2."IOBJNM" = '0SALES_GRP' THEN H2."NODENAME"
WHEN H1."IOBJNM" = '0SALES_GRP' THEN H1."NODENAME"
END SALE_GRP, --销售组
H1."NODENAME" AS LEV1_CODE, --层级1
H1.NODEDESC AS LEV1_NAME, --层级1描述
H2."NODENAME" AS LEV2_CODE, --层级2
H2.NODEDESC AS LEV2_NAME, --层级2描述
H3."NODENAME" AS LEV3_CODE, --层级3
H3.NODEDESC AS LEV3_NAME, --层级3描述
H4."NODENAME" AS LEV4_CODE, --层级4
H4.NODEDESC AS LEV4_NAME, --层级4描述
H5."NODENAME" AS LEV5_CODE, --层级5 
H5.NODEDESC AS LEV5_NAME, --层级5描述
H6."NODENAME" AS LEV6_CODE, --层级6
H6.NODEDESC AS LEV6_NAME, --层级6描述
H7."NODENAME" AS LEV7_CODE, --层级7
H7.NODEDESC AS LEV7_NAME, --层级7描述
H8."NODENAME" AS LEV8_CODE, --层级8
H8.NODEDESC AS LEV8_NAME, --层级8描述
H9."NODENAME" AS LEV9_CODE, --层级9
H9.NODEDESC AS LEV9_NAME, --层级9描述
H10."NODENAME" AS LEVEL10C, 
H10.NODEDESC AS LEVEL10N,
H11."NODENAME" AS LEVEL11C, 
H11.NODEDESC AS LEVEL11N,
H12."NODENAME" AS LEVEL12C, 
H12.NODEDESC AS LEVEL12N,
H13."NODENAME" AS LEVEL13C, 
H13.NODEDESC AS LEVEL13N,
H14."NODENAME" AS LEVEL14C, 
H14.NODEDESC AS LEVEL14N,
H15."NODENAME" AS LEVEL15C, 
H15.NODEDESC AS LEVEL15N,
CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM t1 H1
LEFT JOIN t1 H2 ON H1."NODEID" = H2."PARENTID"
LEFT JOIN t1 H3 ON H2."NODEID" = H3."PARENTID"
LEFT JOIN t1 H4 ON H3."NODEID" = H4."PARENTID"
LEFT JOIN t1 H5 ON H4."NODEID" = H5."PARENTID"
LEFT JOIN t1 H6 ON H5."NODEID" = H6."PARENTID"

WHERE H1."TLEVEL" = '01'
--AND H2."NODENAME" = '瓷砖事业部'
AND H1."NODENAME" = 'DP集团'

WHEN H15."IOBJNM" = '0SALES_GRP' THEN H15."NODENAME"
WHEN H14."IOBJNM" = '0SALES_GRP' THEN H14."NODENAME"
WHEN H13."IOBJNM" = '0SALES_GRP' THEN H13."NODENAME"
WHEN H12."IOBJNM" = '0SALES_GRP' THEN H12."NODENAME"
WHEN H11."IOBJNM" = '0SALES_GRP' THEN H11."NODENAME"
WHEN H10."IOBJNM" = '0SALES_GRP' THEN H10."NODENAME"
WHEN H9."IOBJNM" = '0SALES_GRP' THEN H9."NODENAME"
WHEN H8."IOBJNM" = '0SALES_GRP' THEN H8."NODENAME"
WHEN H7."IOBJNM" = '0SALES_GRP' THEN H7."NODENAME"
WHEN H6."IOBJNM" = '0SALES_GRP' THEN H6."NODENAME"
WHEN H5."IOBJNM" = '0SALES_GRP' THEN H5."NODENAME"
WHEN H4."IOBJNM" = '0SALES_GRP' THEN H4."NODENAME"
WHEN H3."IOBJNM" = '0SALES_GRP' THEN H3."NODENAME"
WHEN H2."IOBJNM" = '0SALES_GRP' THEN H2."NODENAME"
WHEN H1."IOBJNM" = '0SALES_GRP' THEN H1."NODENAME"
END IS NOT NULL
