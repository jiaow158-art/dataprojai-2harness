TRUNCATE TABLE DM.DM_RPT_PROD_LEVEL_T;

SELECT H."NODENAME" as NODENAME,
CASE 
WHEN H."IOBJNM" = 'ZPRODH3' THEN ST."TXTSH"
END AS NODEDESC,
H."NODEID" as NODEID,
H."PARENTID" as PARENTID,

LEFT JOIN dm.dm_rpt_rsthiernode_t R ON R."NODENAME" = H."NODENAME" AND R."OBJVERS" = 'A' AND R."HIEID" = (select HIEID from DM.DM_RPT_RSHIEDIR_T where hienm = 'ZPRODH3_2020')/*替换 DP_JTJG_2021 */ AND R."LANGU" = '1'
WHERE H."HIEID" = (select HIEID from DM.DM_RPT_RSHIEDIR_T where hienm = 'ZPRODH3_2020')) /*替换 DP_JTJG_2021 */
