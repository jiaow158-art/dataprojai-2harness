--应收整体状态表（客编维度）

--涉及底表：
--1.度量表：
--dm.dm_rpt_receivable_quota_t --BW_应收余额表
--dm.dm_rpt_tradeticket_quota_t drtqt --商票余额表
--dm.dm_rpt_tradeticket_accage_t drtat --商票账龄报表
--dm.dm_rpt_receivable_devalue_t drrdt --应收减值报表
--dm.dm_rpt_pjobect_adpay_t --工程代垫款报表
--dm.dm_ar_return_payment_t --应收回款报表
--dm.dm_zzt_xssr_t --销售收入成本表
--2.维度表：
--DM.DM_RPT_SALE_GRP_T --利用销售组匹配组织
--dwrdim.dwr_dim_cust_sales_area_d  --客户主数据（销售区域）


TRUNCATE TABLE DM.DM_AR_OVERDUE_RECEIVABLES_T;
INSERT INTO DM.DM_AR_OVERDUE_RECEIVABLES_T
with TMP_TK as ( -- 商票金额及逾期商票
select 
d.month_id as edition_mon,
n.comp_code
,n.cust_num
,n.sale_grp
,n.wbs_num
,0 zysye
,0 zyq
,coalesce(t.spje,0) as spje
,coalesce(t.yqsp,0) as yqsp
,0 zysjz
,0 bf_zysjz_1
,0 zyspjjz
,0 bf_zyspjjz_1
,0 yqe_yc
,nvl(lead(coalesce(t.yqsp,0),1) over(partition by n.comp_code,n.cust_num order by d.month_id asc),0) as yqsp_yc --下月逾期商票(不回款情况下)
,0 zddk
,0 bhszyywsr
,0 yqhk
,0 wyqhk
from
(select month_id from dm.dm_dim_date_d  
where month_id >= '202101'   --取大于2021年1月的数据
and month_id <= substr(now() + interval '1 MONTH',1,4) || substr(now() + interval '1 MONTH' ,6,2)  --取小于当月数据
 group by month_id) d 
 left join (select 
            comp_code--公司代码
			,customer as cust_num--客编
			,sales_grp as sale_grp
			,wbs_elemt as wbs_num
		from dm.dm_rpt_tradeticket_quota_close_t drtqt --商票余额表
			group by comp_code,
			         cust_num
					 ,sales_grp
					 ,wbs_elemt
					 ) n
		on 1 = 1    --通过left join来计算笛卡尔积，再通过开窗函数提取下月逾期的商票数据（保证某月数据缺失时，不会跳过月份来取数）
left join (select 
	       substring(edition_date,1,6) as edition_mon--定版月份
	       ,comp_code  --公司代码
			,customer as cust_num --客编
			,sales_grp as sale_grp
			,wbs_elemt as wbs_num
			,sum(zpjje) as spje --商票金额
			,sum(case when substring(zafi_p007_netduedate,1,6)<=substring(edition_date,1,6)  then zpjje else 0 end) as yqsp --逾期商票  --当前日期小于等于当前日期
		from dm.dm_rpt_tradeticket_quota_close_t drtqt --商票余额表
			group by comp_code
			         ,cust_num
					 ,edition_mon
					 ,sales_grp
					 ,wbs_elemt
					 ) t
on n.cust_num = t.cust_num
and n.comp_code= t.comp_code
and d.month_id = t.edition_mon
and n.sale_grp = t.sale_grp
and COALESCE(n.wbs_num,'SNULL') = COALESCE(t.wbs_num,'SNULL')
)
,tmp_ar_total_data as( --按客编、公司汇总应收数据
select edition_date
       ,comp_code 
       ,debitor
	   ,sales_grp as sale_grp
	   ,wbs_elemt as wbs_num
       ,sum(zysye) as zysye
       ,sum(zyq) as zyq
       ,sum(zdqyj) as zdqyj
from dm.dm_rpt_receivable_quota_close_t --应收余额表
group by edition_date,comp_code,debitor,sales_grp,wbs_elemt
), 
tmp_ar_total_data_fore as (--把当月的到期预警数拼接到上月的数据里
select replace(substring(cast(last_day(add_months(t.edition_date,1)) as varchar),1,10),'-','') as FORECAST_DATE, --输出下个月的最后一天
		   t.*  from tmp_ar_total_data t
),
TMP_RE as(  -- 应收+预测
       select 
	    t.edition_mon--定版月份
	   ,t.comp_code
	   ,t.cust_num--客编
       ,t.sale_grp --销售组编码
       ,t.wbs_num --WBS号	   
	   ,sum(t.zysye) as zysye  --应收余额
	   ,sum(t.zyq) as zyq  --逾期额
	   ,0 spje
       ,0 yqsp
       ,0 zysjz
       ,0 bf_zysjz_1
       ,0 zyspjjz
       ,0 bf_zyspjjz_1
--	   ,t1.zdqyj --到期预警（不回款情况下)
       ,sum(t.yqe_yc) as yqe_yc --下月逾期额预测（不回款情况下)
       ,0 yqsp_yc 
       ,0 zddk
       ,0 bhszyywsr
       ,0 yqhk
       ,0 wyqhk
       from (select substr(t1.edition_date,1,6) as edition_mon--定版月份
	               ,t1.comp_code
	               ,t1.debitor as cust_num--客编
				   ,t1.sale_grp --销售组编码
				   ,t1.wbs_num --WBS号
	               ,t1.zysye  --应收余额
	               ,t1.zyq  --逾期额
				   ,0 as yqe_yc --下月逾期额预测（不回款情况下）
	           from tmp_ar_total_data_fore t1
			   
			  union all

             select substr(t2.edition_date,1,6) as edition_mon--定版月份
                   ,t2.comp_code	 
				   ,t2.debitor as cust_num--客编
				   ,t2.sale_grp --销售组编码
				   ,t2.wbs_num --WBS号				   
	               ,0 as zysye  --应收余额
	               ,0 as zyq  --逾期额
				   ,t2.zdqyj as yqe_yc --下月逾期额预测（不回款情况下）
			  from tmp_ar_total_data  t2
			 ) t
	 group by t.edition_mon--定版月份
	   ,t.comp_code
	   ,t.cust_num--客编
       ,t.sale_grp --销售组编码
       ,t.wbs_num --WBS号			 
       )
 --  用于回款计算
, tmp_ar_total_data2 as( --按客编、公司、wbs汇总应收数据
select edition_date
       ,comp_code 
       ,debitor
	   ,sales_grp as sale_grp
       ,wbs_elemt as wbs_num
       ,sum(zysye) as zysye
       ,sum(zyq) as zyq
       ,sum(zdqyj) as zdqyj
from dm.dm_rpt_receivable_quota_close_t --应收余额表
group by edition_date,comp_code,debitor,wbs_elemt,sales_grp
)
-- 应收（打上下月标记和本月标记）
,temp_q_pre as (select replace(substring(cast(add_months(t.edition_date,1) as varchar),1,7),'-','') as next_mon,  --下月（用于回款计算）
substr(t.edition_date,1,6) as edition_mon, t.*  from tmp_ar_total_data2 t)

,TMP_TK_RD as ( -- 商票减值
select d.month_id as edition_mon
       ,n.comp_code
       ,n.cust_num
	   ,n.sale_grp
	   ,n.wbs_num
	   ,0 zysye
       ,0 zyq
       ,0 spje
       ,0 yqsp
       ,0 zysjz
       ,0 bf_zysjz_1
       ,coalesce(t.zyspjjz,0) as zyspjjz
       ,nvl(lag(coalesce(t.zyspjjz,0),1) over(partition by n.comp_code,n.cust_num order by d.month_id asc),0) as bf_zyspjjz_1 --总应收票据减值(上月)
       ,0 yqe_yc
       ,0 yqsp_yc 
       ,0 zddk
       ,0 bhszyywsr
       ,0 yqhk
       ,0 wyqhk
from
(select month_id from dm.dm_dim_date_d  where month_id >= '202101' and month_id <= substr(sysdate,1,4) || substr(sysdate,6,2)
 group by month_id ) d 
 left join (select 
             comp_code
			,customer as cust_num --客编
			,sales_grp as sale_grp
			,wbs_elemt as wbs_num
		from dm.dm_rpt_tradeticket_accage_close_t drtqt --商票账龄、减值报表（定版）
			group by comp_code
			         ,cust_num
					 ,sales_grp
					 ,wbs_elemt
					 ) n
		on 1 = 1
left join (select 
	             substring(edition_date,1,6) as edition_mon--定版月份
	            ,comp_code
			    ,customer as cust_num --客编
			    ,sales_grp as sale_grp
			    ,wbs_elemt as wbs_num						
			    ,sum(zyspj_jc_amt) as zyspjjz --总应收票据减值
		from dm.dm_rpt_tradeticket_accage_close_t drtqt --商票账龄、减值报表（定版）
			group by comp_code
			         ,cust_num
					 ,edition_mon
			         ,sales_grp
			         ,wbs_elemt				 
					 ) t
on n.cust_num = t.cust_num
and n.comp_code=t.comp_code
and d.month_id = t.edition_mon
and n.sale_grp = t.sale_grp
and COALESCE(n.wbs_num,'SNULL') = COALESCE(t.wbs_num,'SNULL')
)
, TMP_RE_RD as ( -- 应收减值
select d.month_id as edition_mon
       ,n.comp_code as comp_code 
       ,n.cust_num
	   ,n.sale_grp
	   ,null as wbs_num
	   ,0 zysye
       ,0 zyq
       ,0 spje
       ,0 yqsp
       ,coalesce(t.zysjz,0) as zysjz
       ,nvl(lag(coalesce(t.zysjz,0),1) over(partition by n.comp_code,n.cust_num,n.sale_grp order by d.month_id asc),0) as bf_zysjz_1 --总应收减值(上月)
       ,0 zyspjjz
       ,0 bf_zyspjjz_1
       ,0 yqe_yc
       ,0 yqsp_yc 
       ,0 zddk
       ,0 bhszyywsr
       ,0 yqhk
       ,0 wyqhk
from
(select month_id from dm.dm_dim_date_d  where month_id >= '202101' and month_id <= substr(sysdate,1,4) || substr(sysdate,6,2)
 group by month_id ) d 
 left join (select 
             omp_code as comp_code
			,debitor as cust_num --客编
			,sales_grp as sale_grp
		from dm.dm_rpt_receivable_devalue_close_t drtqt --应收减值报表（定版）
			group by omp_code
			         ,cust_num
					 ,sales_grp
					 ) n
		on 1 = 1
left join (select 
	         substring(edition_date,1,6) as edition_mon--定版月份
	        ,omp_code as comp_code
			,debitor as cust_num --客编
			,sales_grp as sale_grp
--			,sum(zyszkwyq) as zyszkwyq--总应收账款未逾期
--			,sum(zyszkyq)  as zyszkyq--总应收账款逾期
			,nvl(sum(zyszkwyq),0)+ nvl(sum(zyszkyq),0) as zysjz
		from dm.dm_rpt_receivable_devalue_close_t drtqt --应收减值报表（定版）
			group by  omp_code
			         ,cust_num
					 ,edition_mon
					 ,sales_grp
					 ) t
on n.cust_num = t.cust_num
and n.comp_code = t.comp_code
and d.month_id = t.edition_mon
and n.sale_grp = t.sale_grp
)
, TMP_IC as (
	select  -- 不含税主营业务收入（管理）
		 calmonth as edition_mon--月份
		,comp_code
		,customer as cust_num       --客编
		,dzxt.sales_grp as sale_grp
		,dzxt.wbs_elemt as wbs_num
		,0 zysye
        ,0 zyq
        ,0 spje
        ,0 yqsp
        ,0 zysjz
        ,0 bf_zysjz_1
        ,0 zyspjjz
        ,0 bf_zyspjjz_1
        ,0 yqe_yc
        ,0 yqsp_yc 
        ,0 zddk
		,sum( nvl(zbhsje,0)  -nvl(zsyjf,0) + nvl(zfdje,0) -nvl(zytjf_xt,0) -nvl(zytjf_sg,0)  )  as bhszyywsr  -- 不含税主营业务收入（管理）
     	,0 yqhk
        ,0 wyqhk
	from dm.dm_zzt_xssr_close_t dzxt  --销售收入成本表（定版）
	left join dm.dm_rpt_sale_grp_t drsgt 
	on drsgt.sale_grp = dzxt.sales_grp 
	where distr_chan ='06'  --06：工程渠道
	and lev4_name in ('直营经营管理板块','工程市场中心')  --自营工程（含子公司工程渠道、工程市场中心）
	group by edition_mon,comp_code,cust_num,zysye,zyq,spje,yqsp, zysjz, bf_zysjz_1, zyspjjz,bf_zyspjjz_1, yqe_yc, yqsp_yc , zddk,yqhk,wyqhk,dzxt.sales_grp,dzxt.wbs_elemt	
)
, TMP_RP as ( -- 代垫款
	select  
		 edition_date as edition_mon --月份
		,comp_code
		,debitor as cust_num      --客编
		,sales_grp as sale_grp
		,wbs_elemt as wbs_num
		,0 zysye
        ,0 zyq
        ,0 spje
        ,0 yqsp
        ,0 zysjz
        ,0 bf_zysjz_1
        ,0 zyspjjz
        ,0 bf_zyspjjz_1
        ,0 yqe_yc
        ,0 yqsp_yc 
        ,sum(NVL(zddk_dq,0))as zddk --工程代垫款   --取的当前工程代垫款
        ,0 bhszyywsr
        ,0 yqhk
        ,0 wyqhk
	from dm.dm_rpt_pjobect_adpay_t drpat --工程代垫款报表
	group by edition_mon,comp_code,cust_num,zysye,zyq,spje,yqsp, zysjz, bf_zysjz_1, zyspjjz,bf_zyspjjz_1, yqe_yc, yqsp_yc , bhszyywsr,yqhk,wyqhk,sales_grp,wbs_elemt
)

-- 此处开始计算回款：匹配上月末应收余额大于0的数据；
, temp_r as (select   -- 回款（获取组织架构）
calmonth,debitor,comp_code ,wbs_elemt as wbs_num,sales_grp as sale_grp,
-sum(nvl(zhkje_wqrx,0)+nvl(zhkje_dadz,0)+nvl(zhkje_wqyc,0)+nvl(zhkje_yqyc,0)) as zhk  --总回款=未确认项金额+DADZ客户直接回款+未清银承+已清银承
from dm.dm_ar_return_payment_close_t darpt -- 应收回款报表（定版）
group by calmonth,debitor,comp_code ,wbs_elemt,sales_grp )

-- 有WBS项目号的，按WBS项目匹配
,temp_p as  (select A.calmonth as calmonth,  
A.debitor as debitor,
A.comp_code  as comp_code,
A.wbs_num  as wbs_num,
a.sale_grp,
B.zysye as pre_ysye,  --上月应收余额
A.zhk 
from (select *  from temp_r where temp_r.wbs_num is not null ) as A   
inner join (select * from temp_q_pre where temp_q_pre.wbs_num is not null) as B
on A.wbs_num =B.wbs_num 
and A.debitor  =B.debitor  
and A.comp_code =B.comp_code 
and a.sale_grp = b.sale_grp
and A.calmonth =B.next_mon 
where pre_ysye>0 ) --应收余额大于0
,temp_p2 as (
select C.*,nvl(D.zdqyj,0) as zdqyj from temp_p C   --本月到期预警
left join (select * from temp_q_pre where temp_q_pre.wbs_num is not null) as D
on C.wbs_num =D.wbs_num 
and C.debitor  =D.debitor  
and C.comp_code =D.comp_code 
and C.calmonth =D.edition_mon 
and c.sale_grp = d.sale_grp
)
 -- 无WBS项目号的，按客编+公司编码项目匹配
,temp_px as (
select E.calmonth as calmonth,
E.debitor as debitor,
E.comp_code  as comp_code,
E.wbs_num  as wbs_num,
e.sale_grp,
F.zysye  as pre_ysye,  --上月应收余额
E.zhk 
from (select *  from temp_r where temp_r.wbs_num is null ) as E 
inner join (select * from temp_q_pre where temp_q_pre.wbs_num is null) as F
on E.debitor  =F.debitor  
and E.comp_code =F.comp_code 
and E.calmonth =F.next_mon 
and e.sale_grp = f.sale_grp
where pre_ysye>0 )--应收余额大于0
,temp_px2 as (
select G.*,nvl(H.zdqyj,0) as zdqyj from temp_px G
left join (select * from temp_q_pre where temp_q_pre.wbs_num is  null) as H
on G.debitor  =H.debitor  
and G.comp_code =H.comp_code 
and G.calmonth =H.edition_mon 
and g.sale_grp = h.sale_grp
)

,TMP_RT as(  -- 回款数据
	select calmonth,debitor,comp_code,wbs_num,pre_ysye,zdqyj,zhk,sale_grp,
case when zdqyj>zhk then zhk else zdqyj end as yqhk,  --按先进先出原则，到期预警大于总回款时，逾期回款=总回款；否则，逾期回款=到期预警
case when zdqyj>zhk then  0 else zhk-zdqyj end as wyqhk  --按先进先出原则，到期预警大于总回款时，未逾期回款=0；否则，未逾期回款=总回款-到期预警
from temp_p2
where zhk>0

union all 

select calmonth,debitor,comp_code,wbs_num,pre_ysye,zdqyj,zhk,sale_grp,
case when zdqyj>zhk then zhk else zdqyj end as yqhk,
case when zdqyj>zhk then  0 else zhk-zdqyj end as wyqhk
from temp_px2
where zhk>0
)
,TMP_RT1 as (SELECT
calmonth as edition_mon
,comp_code
,debitor as cust_num
,sale_grp
,wbs_num
,0 zysye
,0 zyq
,0 spje
,0 yqsp
,0 zysjz
,0 bf_zysjz_1
,0 zyspjjz
,0 bf_zyspjjz_1
,0 yqe_yc
,0 yqsp_yc 
,0 zddk
,0 bhszyywsr
,sum(yqhk) as yqhk
,sum(wyqhk) as wyqhk
from TMP_RT
group by edition_mon,comp_code,cust_num,zysye,zyq,spje,yqsp, zysjz, bf_zysjz_1, zyspjjz,bf_zyspjjz_1, yqe_yc, yqsp_yc , zddk, bhszyywsr,wbs_num,sale_grp
)

,TMP_ALL as(
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk 
FROM TMP_RE
UNION ALL
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk 
FROM TMP_TK
UNION ALL
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk 
FROM TMP_RE_RD
UNION ALL 
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk
FROM TMP_TK_RD
UNION ALL
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk
FROM TMP_IC
UNION ALL
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk
FROM TMP_RP
UNION ALL
SELECT edition_mon,comp_code,cust_num,sale_grp,wbs_num,zysye,zyq,spje,yqsp,zysjz,bf_zysjz_1,zyspjjz,bf_zyspjjz_1,yqe_yc,yqsp_yc,zddk,bhszyywsr,yqhk,wyqhk
FROM TMP_RT1
)
,TMP_SUM as(
SELECT  
    edition_mon
	,comp_code
	,cust_num
	,sale_grp
	,wbs_num
	,sum(zysye) as zysye
	,sum(zyq) as zyq
	,sum(spje) as spje
	,sum(yqsp) as yqsp
	,sum(zysjz) as zysjz
	,sum(bf_zysjz_1) as bf_zysjz_1
	,sum(zyspjjz) as zyspjjz

 SELECT 
 TMP_SUM.*
 ,g.cust_name
 ,CASE WHEN d.sales_group_name IN ('中国恒大','华夏幸福','泰禾集团') THEN 1 ELSE 0 END is_ignore --是否剔除
 FROM TMP_SUM
left join 

(select *,ROW_NUMBER() over(PARTITION by cust_num,sales_org_code

			  else 2 end asc
) as rn

from dwrdim.dwr_dim_cust_general_d ) g
 on g.cust_num=TMP_SUM.cust_num 
 and rn2=1
