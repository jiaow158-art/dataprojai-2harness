truncate table DWRFIN.DWR_AR_RISK_LEVEL_F_T;
insert into  DWRFIN.DWR_AR_RISK_LEVEL_F_T 
with tmp as (
	select 
	WBS,comp_code,cust_code,min(case when  cur_cu_clos_balance_amt >0 then 
 due_date else null end ) as min_due_date,
	sum(cur_cu_clos_balance_amt) as cur_cu_clos_balance_amt_1,
	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(voucher_date,1,10), 'YYYY-MM-DD')) 
between  365 and 545 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as cur_cu_clos_balance_amt_545,
	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(voucher_date,1,10), 'YYYY-MM-DD')) 
between  545 and 730 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as cur_cu_clos_balance_amt_730,	
	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(voucher_date,1,10), 'YYYY-MM-DD')) 
>730 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as cur_cu_clos_balance_amt_730up ,

	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(due_date,1,10), 'YYYY-MM-DD')) 
between  0 and 90 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as ar_cur_cu_clos_balance_amt_1,--逾期0-3个月应收金额（前不含后含）


	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(due_date,1,10), 'YYYY-MM-DD')) 
between  90 and 180 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as ar_cur_cu_clos_balance_amt_2,	--逾期3-6个月应收金额（前不含后含）

	
	sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(due_date,1,10), 'YYYY-MM-DD')) 
between  180 and 365 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as ar_cur_cu_clos_balance_amt_3,	--逾期6-12个月应收金额（前不含后含）
	
		sum(case when extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(due_date,1,10), 'YYYY-MM-DD')) 
 >365 then coalesce(cur_cu_clos_balance_amt,0) else 0  end 
	) as ar_cur_cu_clos_balance_amt_4	--逾期12个月以上应收金额（前不含后含）
from 
DWRFIN.DWR_AR_CUST_DETAIL_F T 
LEFT JOIN DWRDIM.DWR_DIM_CUST_GENERAL_D  T3 
            ON T.CUST_CODE = T3.cust_num
            and T3.end_date = '4712-12-31' 
where
substr(clear_date,1,4)  = '4712' 
and coalesce(special_general_ledger,'#') in ('U','Q','M','&','W','V','#')
group by WBS,comp_code,cust_code	
	)
	,tmp2 as(
	select 
	wbs, comp_code, cust_code, min_due_date,extract('day' from to_date('${PERIOD_ID_D}', 'YYYY-MM-DD') - to_date(substr(min_due_date,1,10), 'YYYY-MM-DD')) as max_overdue_days,
	cur_cu_clos_balance_amt_1, cur_cu_clos_balance_amt_545, cur_cu_clos_balance_amt_730, cur_cu_clos_balance_amt_730up,
	ar_cur_cu_clos_balance_amt_1,ar_cur_cu_clos_balance_amt_2,ar_cur_cu_clos_balance_amt_3,ar_cur_cu_clos_balance_amt_4
	from  
	tmp 
)
,CREDIT_DEVALUE as (
select  
comp_code
,cust_code
,wbs
,sum(overdue_receivables) as overdue_receivables
from 
DWRFIN.DWR_AR_CREDIT_DEVALUE_F
 where query_date = '${PERIOD_ID_D}'
group by 
comp_code
,cust_code
,wbs
)
select  
t1.comp_code
,t1.cust_code
,t1.wbs
,t3.cust_risk_level
,t1.overdue_receivables
,t2.min_due_date
,t2.max_overdue_days
,t2.cur_cu_clos_balance_amt_1
,t2.cur_cu_clos_balance_amt_545
,t2.cur_cu_clos_balance_amt_730
,t2.cur_cu_clos_balance_amt_730up
,CASE 
    -- 情况一：暴雷
    WHEN t3.cust_risk_level = '暴雷' 
         AND (t2.ar_cur_cu_clos_balance_amt_3 + t2.ar_cur_cu_clos_balance_amt_4) > 10000 
    THEN '暴雷'
    
    -- 情况二：高风险
    WHEN t2.ar_cur_cu_clos_balance_amt_4 > 10000 
    THEN '高风险'
    
    -- 情况三：高风险
    WHEN t1.overdue_receivables <= 0 
         AND t2.cur_cu_clos_balance_amt_730up > 50000 
    THEN '高风险'
    
    -- 情况四：不良
    WHEN (t2.ar_cur_cu_clos_balance_amt_2 + t2.ar_cur_cu_clos_balance_amt_3 + t2.ar_cur_cu_clos_balance_amt_4) > 10000 
    THEN '不良'
    
    -- 情况五：不良
    WHEN t1.overdue_receivables <= 0 
         AND t2.cur_cu_clos_balance_amt_730up <= 50000 
         AND (t2.cur_cu_clos_balance_amt_730 + t2.cur_cu_clos_balance_amt_730up)  > 50000 
    THEN '不良'
    
    -- 情况六：问题
    WHEN t1.overdue_receivables > 10000 
    THEN '问题'
    
    -- 情况七：问题
    WHEN t1.overdue_receivables <= 0 
         AND (t2.cur_cu_clos_balance_amt_545 + t2.cur_cu_clos_balance_amt_730 + t2.cur_cu_clos_balance_amt_730up) > 50000 
         AND (t2.cur_cu_clos_balance_amt_730 + t2.cur_cu_clos_balance_amt_730up) <= 50000 
    THEN '问题'
    
    -- 情况八：关注
    WHEN t1.overdue_receivables <= 10000 and t1.overdue_receivables  > 0
    THEN '关注'
    
    -- 情况九：关注
    WHEN t2.max_overdue_days >= -45 
         AND t2.max_overdue_days < 0 
    THEN '关注'
    
    -- 其他：正常
    ELSE '正常'
END AS risk_level
,current_timestamp  
,ar_cur_cu_clos_balance_amt_1
,ar_cur_cu_clos_balance_amt_2
,ar_cur_cu_clos_balance_amt_3
,ar_cur_cu_clos_balance_amt_4
from 
 CREDIT_DEVALUE  t1 
 left join tmp2 t2 
 on t1.WBS = t2.wbs  
 and t1.comp_code = t2.comp_code
 and t1.cust_code =t2.cust_code
 left join upload.uplaod_ar_eng_det_t t3
 on t1.WBS = t3.wbs  
 and t1.cust_code =t3.cust_code
