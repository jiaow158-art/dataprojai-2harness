DELETE FROM DWRFIN.DWR_AR_OTHER_RECEIVABLE_AGING_F 
WHERE query_date in (SELECT CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) END                            -- 当月月结2-5日传上月月底
                            UNION ALL
                            SELECT '${PERIOD_ID_D}'     -- 昨天
                            UNION ALL
                            SELECT substr(to_date('${PERIOD_ID_D}','yyyy-MM-dd') - 1,1,10)  -- 前天
    );
INSERT INTO DWRFIN.DWR_AR_OTHER_RECEIVABLE_AGING_F
SELECT
    T.QUERY_DATE
    ,T.DRAFT_SHORT_NUM
    ,T.VOUCHER_CODE
    ,T.VOUCHER_TYPE
    ,T.VOUCHER_DATE
    ,T.POSTING_CODE
    ,T.VOUCHER_CURRENCY
    ,T.TXT
    ,T.CHARGE_ACCOUNT
    ,T.COMP_CODE
    ,T.POSTING_DATE
    ,T.YEAR_MONTH
    ,T.ACCOUNT
    ,T.AMT
    ,T.DUE_DAYS
    ,T.WITHIN_1YEAR_AMT
    ,T."1_2YEAR_AMT"
    ,T."2_3YEAR_AMT"
    ,T."3_4YEAR_AMT"
    ,T.OVER_4YEAR_AMT
    ,T.WITHIN_1YEAR_AGING
    ,T."1_2YEAR_AGING"
    ,T."2_3YEAR_AGING"
    ,T."3_4YEAR_AGING"
    ,T.OVER_4YEAR_AGING
    ,T.BUSINESS_DEPARTMENT
    ,T.DW_LAST_UPDATE_DATE
    ,T.SUPPLIER_CODE                                        -- 供应商编码
    ,T.CUST_CODE                                            -- 客户编码
    ,T.REF_DOC                                              -- 参照
    ,T.CUST_GRP_CODE                                        -- 客户组
    ,T.WBS_NUM                                              -- WBS编码
    ,T.SALES_GRP_CODE                                       -- 销售组编码
    ,case when coalesce (t1.del_flag,t2.del_flag) is not null then coalesce (t1.del_flag,t2.del_flag)
		else 'N' end as is_overdue_cust                   -- 是否暴雷客户
	,T.INPUT_DATE  --输入日期（新增于6月13日）
FROM DWRFIN.DWR_AR_OTHER_RECEIVABLE_AGING_F_01 T  
	LEFT JOIN UPLOAD.UPLOAD_AMB_CREDIT_CUSTCODE_T T1
	ON T.CUST_CODE = T1.CUST_CODE 
	AND T.COMP_CODE = T1.COMP_CODE 
	AND TO_DATE(T.QUERY_DATE,'yyyy-MM-dd') BETWEEN T1.STAT_DATE AND T1.END_DATE
	LEFT JOIN UPLOAD.UPLOAD_AMB_CREDIT_CUSTCODE_T T2
	ON T.CUST_CODE = T2.CUST_CODE 
	AND T2.COMP_CODE IS NULL
	AND TO_DATE(T.QUERY_DATE,'yyyy-MM-dd') BETWEEN T2.STAT_DATE AND T2.END_DATE 
WHERE query_date = CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END
