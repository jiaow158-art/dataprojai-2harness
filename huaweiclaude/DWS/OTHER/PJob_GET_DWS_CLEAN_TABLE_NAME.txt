(
select schemaname ||'.'||relname TABLE_NAME
from PGXC_GET_STAT_ALL_TABLES
where schemaName not in('pg_toast', 'pg_catalog', 'information_schema', 'cstore', 'pmk')
        and dirty_page_rate >20
order by
        n_dead_tup desc,dirty_page_rate desc
)
union all
(
select schemaname ||'.'||relname TABLE_NAME
from PGXC_GET_STAT_ALL_TABLES
where schemaName in ('api_log')
)
;
