truncate table upload.upload_ct_authority_t;

insert into upload.upload_ct_authority_t (工号, 业务员, 零售线组权限, 工程线组权限, 大包线组权限, 设计师线组权限, 渠道权限, 三级权限, 四级权限, 五级权限)

select 工号,
       业务员,
       case when strpos(a.渠道, '零售') > 0 or strpos(a.渠道, '市场中心') > 0 then a.线组 else '' end  零售线组权限,
       case when strpos(a.渠道, '工程') > 0 or strpos(a.渠道, '市场中心') > 0 then a.线组 else '' end  工程线组权限,
       case when strpos(a.渠道, '大包') > 0 or strpos(a.渠道, '市场中心') > 0 then a.线组 else '' end  大包线组权限,
       case when strpos(a.渠道, '设计师') > 0 or strpos(a.渠道, '市场中心') > 0 then a.线组 else '' end 设计师线组权限,
       (case
            when strpos(a.渠道, '运营中心') > 0 or strpos(a.渠道, '子公司') > 0 then '无'
            when strpos(a.渠道, '零售') > 0 then '零售'
            when strpos(a.渠道, '工程') > 0 then '工程'
            when strpos(a.渠道, '大包') > 0 then '大包'
            when strpos(a.渠道, '设计师') > 0 then '设计师'
            else a.渠道 end
           )                                                                               渠道权限,
       case 运营中心 when '大包市场部' then '大包市场中心' else 运营中心 end                                  三级权限,
       case a.渠道 when '大包市场部' then '大包市场中心' else a.渠道 end                                  四级权限,
       case a.渠道 when '大包市场部' then '大包市场中心' else a.线组 end                                  五级权限
from upload.upload_salesgroup_person_relation a
where 1 = 1
