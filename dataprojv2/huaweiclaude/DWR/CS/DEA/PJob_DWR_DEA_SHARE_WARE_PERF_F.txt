INSERT OVERWRITE TABLE DWRCS.DWR_DEA_SHARE_WARE_PERF_F PARTITION (STAT_MONTH)

       --t1.out_return_goods,  --出退货 
       t1.dealer_code as customer_code, -- 客户编码,也存了经销商编码
       t1.dealer_name as customer_name, -- 客户名称，也存了经销商名称
       '' as cust_address,		--客户地址
       t1.channel_name_cn, 		--渠道
       SUBSTRING(t1.doc_date,1,10) AS doc_date, -- 单据日期
       t1.doc_num, 			--单号

       t1.material_name,  -- 物料描述
       t1.matl_group_code, --物料组代码
	   t1.matl_group_name, 		--物料组名称
	   t1.external_matl_group_code, -- 外部物料组
	   t1.external_matl_group_name, -- 外部物料组名称
       t1.matl_type_code   , -- 物料类型
  	   t1.matl_type_name   , --物料类型名称
       t1.product_group_code  ,-- 产品组
  	   t1.product_group_name  , --产品组名称 
       t1.product_classify_code  ,-- 产品分级
       t1.product_classify_name  ,-- 产品分级名称
       t1.product_level_code ,--  '产品层次' 
       t1.product_level_name ,--  '产品层次名称' 
  	   t1.product_level1_code,--   '产品层次1' 
       t1.product_level1_name,--   '产品层次1名称' 
       t1.product_level2_code,--   '产品层次2' 
       t1.product_level2_name,--   '产品层次2名称' 
       t1.product_level3_code,--   '产品层次3' 
       t1.product_level3_name,--   '产品层次3名称'
       t1.product_level4_code,--   '产品层次4'
       t1.product_level4_name,--   '产品层次4名称' 
       case when t1.out_return_goods = '出货' then 
               t1.material_qty * -1
            when t1.out_return_goods = '退货' then 
               t1.material_qty * -1 
            else material_qty end as material_qty,				--数量
       t1.batch_num,			--批次
       case when t1.out_return_goods = '出货' then 
               t1.Price * -1
            when t1.out_return_goods = '退货' then 
               t1.Price * -1 
            else Price end as Price,				--单价
       case when t1.out_return_goods = '出货' then 
               t1.line_total_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.line_total_amt * -1
            else line_total_amt end as line_total_amt, -- 行合计
       t1.unit,				--单位
       t1.bar_code,				--条形码
       case when t1.out_return_goods = '出货' then 
               t1.discount_price_ago_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.discount_price_ago_amt * -1
       else discount_price_ago_amt end as discount_price_ago_amt, -- 折扣前价格
       case when t1.out_return_goods = '出货' then 
               t1.discount_price_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.discount_price_amt * -1
       else discount_price_amt end as discount_price_amt, -- 折扣
       case when t1.out_return_goods = '出货' then 
               t1.cost_price_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.cost_price_amt * -1
       else cost_price_amt end as cost_price_amt, -- 成本价
       case when t1.out_return_goods = '出货' then 
               t1.cost_total_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.cost_total_amt * -1
       else cost_total_amt end as cost_total_amt, -- 成本合计
       case when t1.out_return_goods = '出货' then 
               t1.gross_profit_amt * -1
            when t1.out_return_goods = '退货' then 
               t1.gross_profit_amt * -1
       else gross_profit_amt end as gross_profit_amt, -- 毛利
       case when t1.out_return_goods = '出货' then 
               t1.no_clear_qty * -1
            when t1.out_return_goods = '退货' then 
               t1.no_clear_qty * -1
       else no_clear_qty end as no_clear_qty, -- 未清数量
       t1.line_num,					--行号
       t1.wm_name,				-- 仓库
       t1.batch_num_stock,		-- 批次库存
       t1.group_name,					--客户组
       t1.b1_material_group_name,					--物料组
	   t1.cust_from,				-- 客户来源
