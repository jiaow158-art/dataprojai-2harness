INSERT OVERWRITE TABLE DWRCS.DWR_CS_MOS_PRTCT_INFO_F
SELECT  A.ID --ID
       ,A.CREATE_TIME --创建时间
       ,A.CREATER_ID AS CREATE_EMP_ID --创建人ID
       ,A.CREATER_NAME AS CREATE_EMP_NAME --创建人名称
       ,A.MODIFY_TIME AS UPDATE_TIME --修改时间	   
       ,A.MODIFIER_ID AS UPDATE_EMP_ID --修改人ID
       ,A.MODIFIER_NAME AS UPDATE_EMP_NAME --修改人名称
       ,A.VERSION AS VERS_NUM --数据版本号
       ,A.INTENTION_ORDER_ID AS ORDER_ID --意向订单ID
       ,A.PROTECT_TYPE AS PRTCT_TYPE_CODE --报备类型编码 1 门店报备 2 区域报备
	   ,CASE WHEN A.PROTECT_TYPE = 1 THEN '门店报备'
	         WHEN A.PROTECT_TYPE = 2 THEN '区域报备'
			  END AS PRTCT_TYPE_NAME --报备类型描述
       ,A.PROTECT_STATUS AS PRTCT_ST_CODE --报备状态编码 0 无保护 1 保护期内 2 保护过期
	   ,CASE WHEN A.PROTECT_STATUS = 0 THEN '无保护'
	         WHEN A.PROTECT_STATUS = 1 THEN '保护期内'
			 WHEN A.PROTECT_STATUS = 2 THEN '保护过期'
			  END AS PRTCT_ST_NAME --报备状态描述
       ,A.PROTECT_STORE_ID AS PRTCT_STR_ID --报备门店ID
       ,A.PROTECT_STORE_NAME AS PRTCT_STR_NAME --报备门店
       ,A.PROTECT_EMPLOYEE_ID AS PRTCT_EMP_ID --报备员工ID
       ,A.PROTECT_EMPLOYEE_NAME AS PRTCT_EMP_NAME --报备员工
       ,A.PROTECT_DEALER_ID AS PRTCT_DEALER_ID --报备经销商ID
       ,A.PROTECT_DEALER_NAME AS PRTCT_DEALER_NAME --报备经销商
       ,A.PROTECT_CITY AS PRTCT_CITY_NAME --报备城市
       ,A.PROTECT_STAGE_CODE AS PRTCT_STAGE_CODE --报备阶段编码
       ,A.PROTECT_STAGE_NAME AS PRTCT_STAGE_NAME --报备阶段描述
       ,A.PROTECT_SOURCE_ID AS PRTCT_SRC_ID --报备来源ID
       ,B.NAME AS PRTCT_SRC_NAME --客户来源
	   ,COALESCE(B.PARENT_ID,A.PROTECT_SOURCE_ID) AS PARNT_PRTCT_SRC_ID --客户上级来源ID
	   ,COALESCE(B.PARENT_NAME,B.NAME) AS PARNT_PRTCT_SRC_NAME --客户上级来源
       ,A.PROTECT_START_TIME AS PRTCT_START_TIME --报备开始时间
       ,A.PROTECT_END_TIME AS PRTCT_END_TIME --报备结束时间
       ,COALESCE(A.TRANSFER_POOL_NUM,0) AS TFR_POOL_CNT --流入公共池次数
       ,COALESCE(A.POOL_PROTECT_NUM,0) AS POOL_PRTCT_CNT --公共池保护次数
       ,COALESCE(A.PROTECT_NUM,0) AS PRTCT_CNT --报备次数
       ,A.FIRST_PROTECT_EMPLOYEE_ID AS FST_PRTCT_EMP_ID --首次报备人ID
       ,A.FIRST_PROTECT_EMPLOYEE_NAME AS FST_PRTCT_EMP_NAME --首次报备人
       ,A.FIRST_PROTECT_STORE_ID AS FST_PRTCT_STR_ID --首次报备门店ID
       ,A.FIRST_PROTECT_STORE_NAME AS FST_PRTCT_STR_NAME --首次报备门店
       ,A.FIRST_PROTECT_DEALER_ID AS FST_PRTCT_DEALER_ID --首次报备经销商ID
       ,A.FIRST_PROTECT_DEALER_NAME AS FST_PRTCT_DEALER_NAME --首次报备经销商
       ,A.FIRST_PROTECT_TIME AS FST_PRTCT_TIME --首次报备时间
       ,COALESCE(A.DISPATCH_NUM,0) AS DSPCH_CNT --派单次数
       ,A.FOLLOW_RULE_ID AS FOLOW_RULE_ID --报备规则ID
       ,A.FOLLOW_RULE_DETAIL_ID AS FOLOW_RULE_DTL_ID --报备规则阶段设置ID
       ,A.FOLLOW_RULE_DETAIL_DAY_ID AS FOLOW_RULE_DTL_DAY_ID --报备规则阶段时间ID
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SAL_CLIENT_INTENTION_PROTECT_1012 A
  LEFT JOIN SDI.SDI_SAL_CLIENT_SOURCE_INFO_1012 B
    ON A.PROTECT_SOURCE_ID = B.ID
