WITH T1 AS(
SELECT DISTINCT
       A.DICT_TYPE
      ,A.DICT_VALUE
	  ,A.DICT_NAME
  FROM SDI.SDI_BS_DICT_1030 A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.DICT_TYPE IN('zdj_order_stage','contract_status','zdj_order_customer_source')
   --订单阶段 zdj_order_stage
   --合同状态 contract_status
   --订单自定义来源 zdj_order_customer_source
),T2 AS(
SELECT 
       QUOTATIONCATEGORYPRICELIST
      ,REPLACE(REPLACE(SPLIT(REPLACE(REPLACE(QUOTATIONCATEGORYPRICELIST,'}, {',';'),'"',''),';')[0],'}',''),'{','') AS PRICE1
      ,REPLACE(REPLACE(SPLIT(REPLACE(REPLACE(QUOTATIONCATEGORYPRICELIST,'}, {',';'),'"',''),';')[1],'}',''),'{','') AS PRICE2
      ,REPLACE(REPLACE(SPLIT(REPLACE(REPLACE(QUOTATIONCATEGORYPRICELIST,'}, {',';'),'"',''),';')[2],'}',''),'{','') AS PRICE3
      ,ID
	  ,ORDER_CODE
	  ,ENGINEERING_AREA
	  ,SPLIT(SPLIT(QUOTATION_PRICE,'totalMoney": ')[1],',')[0] AS SALE_AMT
FROM SDI.SDI_HD_ORDER_QUOTATION_1030
WHERE EP_ID = 3044
AND APP_ID = 1831
AND IS_DELETED = 0
),T3 AS(
SELECT
       QUOTATIONCATEGORYPRICELIST
      ,CASE WHEN PRICE1 LIKE '%AUXILIARY_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE1,', bizType: AUXILIARY_MATERIAL',''),'price: ','')
            WHEN PRICE2 LIKE '%AUXILIARY_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE2,', bizType: AUXILIARY_MATERIAL',''),'price: ','')
			WHEN PRICE3 LIKE '%AUXILIARY_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE3,', bizType: AUXILIARY_MATERIAL',''),'price: ','')
			 END AS FC_SALE_AMT --辅材销售额
	  ,CASE WHEN PRICE1 LIKE '%PRINCIPAL_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE1,', bizType: PRINCIPAL_MATERIAL',''),'price: ','')
            WHEN PRICE2 LIKE '%PRINCIPAL_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE2,', bizType: PRINCIPAL_MATERIAL',''),'price: ','')
			WHEN PRICE3 LIKE '%PRINCIPAL_MATERIAL%'
	        THEN REPLACE(REPLACE(PRICE3,', bizType: PRINCIPAL_MATERIAL',''),'price: ','')
			 END AS ZC_SALE_AMT --主材销售额
	  ,CASE WHEN PRICE1 LIKE '%MANUAL_LABOR%'
	        THEN REPLACE(REPLACE(PRICE1,', bizType: MANUAL_LABOR',''),'price: ','')
            WHEN PRICE2 LIKE '%MANUAL_LABOR%'
	        THEN REPLACE(REPLACE(PRICE2,', bizType: MANUAL_LABOR',''),'price: ','')
			WHEN PRICE3 LIKE '%MANUAL_LABOR%'
	        THEN REPLACE(REPLACE(PRICE3,', bizType: MANUAL_LABOR',''),'price: ','')
			 END AS RG_SALE_AMT --人工销售额
      ,ID
	  ,ORDER_CODE
	  ,ENGINEERING_AREA AS CNSTR_AREA --施工面积
	  ,SALE_AMT --合同金额
  FROM T2	  
),T4 AS(
SELECT ORDER_CODE
      ,SUM(SCORE) AS EVALU_SCORE --评价分数
	  ,COUNT(DISTINCT ID) AS EVALU_CNT --评价条数
  FROM SDI.SDI_HD_ORDER_EVALUATION_1030 A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND IS_DELETED = 0
   AND TYPE IN(1,0)
 GROUP BY ORDER_CODE  
),T5 AS(
SELECT ORDER_CODE
      ,CONTRACT_SIGN_TIME --签单时间
	  ,PROJECT_END_TIME --完工时间
	  ,ASSIGN_TIME	--指派师傅时间
	  ,FOLLOW_TIME	--完成沟通时间
	  ,PROJECT_START_TIME	--师傅开工时间
  FROM SDI.SDI_HD_ORDER_STAGE_INFO_1030 A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.IS_DELETED = 0   
),T6 AS(
SELECT ORDER_CODE
      ,CONCAT_WS(',',COLLECT_SET(DECORATOR_NAME)) AS DECOR_NAME --师傅姓名
  FROM SDI.SDI_HD_ORDER_DECORATOR_REL_1030 A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.IS_DELETED = 0
   AND A.TYPE = 2
 GROUP BY ORDER_CODE 
),T7 AS(
SELECT ORDER_CODE
      ,CREATE_TIME
	  ,ROW_NUMBER() OVER(PARTITION BY ORDER_CODE ORDER BY CREATE_TIME ASC) AS RN
  FROM SDI.SDI_HD_ORDER_STAGE_LOG_1030 A
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.IS_DELETED = 0 
   AND A.stage_name = '已签约'
)

	  ,S.SHORT_NAME AS CITY_SHORT_NAME --城市简称
	  ,T.SHORT_NAME AS DIST_SHORT_NAME --区区域简称
	  ,X.LVL1_DEPT_NAME	--一级部门名称
      ,X.LVL2_DEPT_NAME	--二级部门名称
	  ,X.LVL3_DEPT_NAME	--三级部门名称
	  ,X.LVL4_DEPT_NAME	--四级部门名称
	  ,P.ASSIGN_TIME	--指派师傅时间
	  ,P.FOLLOW_TIME	--完成沟通时间
	  ,P.PROJECT_START_TIME AS ORDER_START_TIME	--师傅开工时间
	  ,Y.DECOR_NAME --师傅姓名
	  ,Z.CREATE_TIME AS SYS_ORDER_SIGN_TIME --系统签单时间
  FROM SDI.SDI_HD_ORDER_INFO_1030 A
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_1030 B
    ON A.EMPLOYEE_ID = B.ID
   AND A.EP_ID = B.EP_ID
   AND A.APP_ID = B.APP_ID
  LEFT JOIN T1 C
    ON A.ORDER_TYPE = C.DICT_VALUE
   AND A.ORDER_TYPE IS NOT NULL
   AND C.DICT_TYPE = 'zdj_order_stage'
  LEFT JOIN T1 D
    ON A.ORDER_STATUS = D.DICT_VALUE
   AND D.DICT_TYPE = 'zdj_order_stage'	
  LEFT JOIN T1 E
    ON A.STAGE_ID = E.DICT_VALUE 
   AND E.DICT_TYPE = 'zdj_order_stage'
  LEFT JOIN T1 F
    ON A.CONTRACT_STATUS = F.DICT_VALUE
   AND F.DICT_TYPE = 'contract_status'
  LEFT JOIN T1 G
    ON A.SOURCE = G.DICT_VALUE
   AND G.DICT_TYPE = 'zdj_order_customer_source'
  LEFT JOIN T1 H
    ON A.CUSTOMER_SOURCE = H.DICT_VALUE
   AND H.DICT_TYPE = 'zdj_order_customer_source'
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 I
    ON B.STORE_ID = I.ID
   AND A.EP_ID = I.EP_ID
   AND A.APP_ID = I.APP_ID
  LEFT JOIN SDI.SDI_HD_PROVIDER_1030 J
    ON A.PROVIDER_CODE = J.PROVIDER_CODE
   AND A.EP_ID = J.EP_ID
   AND A.APP_ID = J.APP_ID
  LEFT JOIN T3 K
    ON A.QUOTATION_ID = K.ID
  LEFT JOIN T4 O
    ON A.ORDER_CODE = O.ORDER_CODE 
  LEFT JOIN T5 P
    ON A.ORDER_CODE = P.ORDER_CODE
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_1030 Q
    ON A.CREATE_BY = Q.ID
   AND A.EP_ID = Q.EP_ID
   AND A.APP_ID = Q.APP_ID
  LEFT JOIN SDI.SDI_BUZ_REGION_1030 R
    ON A.PROVINCE_ID = R.ID
   AND R.STATUS = 1
  LEFT JOIN SDI.SDI_BUZ_REGION_1030 S
    ON A.CITY_ID = S.ID
   AND S.STATUS = 1
  LEFT JOIN SDI.SDI_BUZ_REGION_1030 T
    ON A.DISTRICT_ID = T.ID
   AND T.STATUS = 1  
  LEFT JOIN(SELECT DISTINCT LVL1_DEPT_NAME	--一级部门名称
                           ,LVL2_DEPT_NAME	--二级部门名称
						   ,LVL3_DEPT_NAME	--三级部门名称
						   ,LVL4_DEPT_NAME	--四级部门名称
						   ,DEPT_NUM --部门编码
              FROM DWRDIM.DWR_DIM_DMT_DEPT_INFO_D
			 WHERE BUSI_TYPE LIKE '%5%' 
           ) X
	ON LPAD(A.PROVIDER_CODE,10,'0') = X.DEPT_NUM
  LEFT JOIN T6 Y
    ON A.ORDER_CODE = Y.ORDER_CODE
  LEFT JOIN T7 Z
    ON A.ORDER_CODE = Z.ORDER_CODE
   AND Z.RN = 1
  LEFT JOIN(SELECT ORDER_QUOTATION_ID
                  ,SUM(COALESCE(AREA,0)) AS AREA
              FROM SDI.SDI_HD_ORDER_QUOTATION_SCHEME_ITEMS_1030
             WHERE EP_ID = 3044
               AND APP_ID = 1831
               AND IS_DELETED = 0
			 GROUP BY ORDER_QUOTATION_ID
           ) AA
	ON AA.ORDER_QUOTATION_ID = A.QUOTATION_ID
 WHERE A.EP_ID = 3044
   AND A.APP_ID = 1831
   AND A.IS_DELETED = 0
   AND COALESCE(A.PROVINCE,'') NOT LIKE '%台湾%'
   AND COALESCE(A.PROVINCE,'') NOT LIKE '%澳门%'
