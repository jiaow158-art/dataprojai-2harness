INSERT OVERWRITE TABLE DWRCS.DWR_CS_MOS_CONFLCT_INFO_F
SELECT  A.ID --ID
       ,A.CREATER_ID AS CREATE_EMP_ID --创建人ID
       ,A.CREATER_NAME AS CREATE_EMP_NAME --创建人名称
       ,A.CREATE_TIME --创建时间
       ,A.MODIFIER_ID AS UPDATE_EMP_ID --修改人ID
       ,A.MODIFIER_NAME AS UPDATE_EMP_NAME --修改人名称
       ,A.MODIFY_TIME AS UPDATE_TIME --修改时间
       ,A.VERSION AS VERS_NUM --数据版本号
       ,A.FOLLOW_EXAMINATION_ID AS FOLOW_EXAM_ID --审核记录ID
       ,A.CONFLICT_CLIENT_ID AS CONFLCT_CUST_ID --冲突客户ID
       ,A.CONFLICT_CLIENT_NAME AS CONFLCT_CUST_NAME --冲突客户名
       ,A.CONFLICT_INTENTION_ORDER_ID AS CONFLCT_ORDER_ID --冲突客户的意向订单ID
       ,A.CONFLICT_STORE_ID AS CONFLCT_STR_ID --冲突客户的门店ID
       ,A.CONFLICT_STORE_NAME AS CONFLCT_STR_NAME --冲突客户的门店
       ,A.CONFLICT_EMPLOYEE_ID AS CONFLCT_EMP_ID --冲突客户的跟进员工ID
       ,A.CONFLICT_EMPLOYEE_NAME AS CONFLCT_EMP_NAME --冲突客户的跟进员工
       ,A.CONFLICT_PHONE AS CONFLCT_PHONE_NUM --冲突客户的手机
       ,A.CONFLICT_ADDRESS AS CONFLCT_ADDR --冲突客户的地址
       ,A.CONFLICT_STAGE_CODE AS CONFLCT_STAGE_CODE --冲突客户的阶段编码
       ,A.CONFLICT_STAGE_NAME AS CONFLCT_STAGE_NAME --冲突客户的阶段
       ,A.CONFLICT_SOURCE_ID AS CONFLCT_SRC_ID --冲突客户来源ID
       ,B.NAME AS CONFLCT_SRC_NAME --冲突客户来源
	   ,COALESCE(B.PARENT_ID,A.CONFLICT_SOURCE_ID) AS PARNT_CUST_SRC_ID --客户上级来源ID
	   ,COALESCE(B.PARENT_NAME,B.NAME) AS PARNT_CUST_SRC_NAME --客户上级来源
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SAL_CLIENT_FOLLOW_EXAMINATION_CONFLICT_1012 A
  LEFT JOIN SDI.SDI_SAL_CLIENT_SOURCE_INFO_1012 B
    ON A.CONFLICT_SOURCE_ID = B.ID
 WHERE CAST(A.DELETE_FLAG AS STRING) = 'FALSE'
