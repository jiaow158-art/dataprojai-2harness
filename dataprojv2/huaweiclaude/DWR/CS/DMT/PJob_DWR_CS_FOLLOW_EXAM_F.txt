INSERT OVERWRITE TABLE DWRCS.DWR_CS_FOLLOW_EXAM_F
SELECT
       CASE WHEN A.EP_ID = 3044 AND A.APP_ID = 1831 THEN '瓷砖'
            WHEN A.EP_ID = 3045 AND A.APP_ID = 1832 THEN '卫浴'
			 END AS DATA_TYPE --数据所属类型
      ,A.APP_ID	--APP ID
      ,A.EP_ID	--企业ID
      ,A.CREATE_BY AS CREATE_EMP_ID	--创建人
	  ,B.EMP_NAME AS CREATE_EMP_NAME --创建人名称
	  ,B.POST_NAME AS CREATE_POST_NAME --创建人职位名称
      ,A.CREATE_TIME --创建时间
      ,A.MODIFY_BY AS MODIFY_EMP_ID --修改人
	  ,C.EMP_NAME AS MODIFY_EMP_NAME --创建人名称
	  ,C.POST_NAME AS MODIFY_POST_NAME --创建人职位名称	  
      ,A.MODIFY_TIME	--修改时间
      ,A.STORE_ID --门店ID
	  ,L.NAME AS STORE_NAME --门店名称
	  ,L.ABBREVIATION AS STORE_BY_NAME --门店简称
      ,A.CLIENT_ID AS USER_ID --客户ID
	  ,H.REAL_NAME AS USER_NAME --客户名称
	  ,H.SOURCE AS CUST_SRC_CODE --客户来源
	  ,J.NAME AS CUST_SRC_NAME --客户来源名称
	  ,COALESCE(K.NAME,J.NAME) AS CUST_SRC_CAT_NAME --客户来源类别名称
      ,A.FOLLOW_EMPLOYEE_ID AS FOLOW_EP_ID --跟进员工ID（触发人）
	  ,D.EMP_NAME AS FOLOW_EMP_NAME --跟进员工人名称
	  ,D.POST_NAME AS FOLOW_POST_NAME --跟进员工人职位名称		  
      ,A.BIND_ID	--客户绑定表ID
      ,A.ORDER_ID	--客户订单表ID
      ,A.STAGE_ID	--客户阶段
	  ,E.STAGE_NAME --客户阶段名称
      ,A.FOLLOW_RULE_ID AS FOLOW_RULE_ID	--报备设置ID
      ,A.FOLLOW_RULE_DETAIL_ID AS FOLOW_RULE_DTL_ID --报备设置详情ID
      ,A.FOLLOW_RULE_DETAIL_DAY_ID AS FOLOW_RULE_DTL_DAY_ID --报备天数设置ID
      ,A.FOLLOW_RULE_DETAIL_AREA_DAY_ID AS FOLOW_RULE_DTL_AREA_DAY_ID --区域报备天数设置ID
      ,A.TYPE AS FOLOW_TYPE --类型 1.报备审核 2.奖励审核
	  ,CASE WHEN A.TYPE = 1 THEN ''
	        WHEN A.TYPE = 2 THEN ''
			 END AS FOLOW_TYPE_NAME --类型名称
      ,A.TRIGGER_TIME	--触发时间
      ,A.STATUS	--状态 -1.已拒绝 0.待审核 1.审核通过
	  ,CASE WHEN A.STATUS = -1 THEN '已拒绝'
	        WHEN A.STATUS = 0 THEN '待审核'
			WHEN A.STATUS = 1 THEN '审核通过'
			 END AS STATUS_NAME --状态名称
      ,A.EXAMINATION_EMPLOYEE_ID AS EXAM_EMP_ID --审核员工ID
	  ,D.EMP_NAME AS EXAM_EMP_NAME --审核员工人名称
	  ,D.POST_NAME AS EXAM_POST_NAME --审核员工人职位名称	  
      ,A.EXAMINATION_TIME AS EXAM_TIME --审核时间
      ,A.REWARD_AMOUNT AS REWARD_AMT --奖励金额
      ,A.REJECT_REASON --审核不通过原因
      ,A.LAST_PROTECT_STORE_ID AS LAST_PRTCT_STORE_ID --上次保护门店
	  ,M.NAME AS LAST_PRTCT_STORE_NAME --门店名称
	  ,M.ABBREVIATION AS LAST_PRTCT_STORE_BY_NAME --门店简称
      ,A.LAST_PROTECT_EMPLOYEE_ID AS LAST_PRTCT_EMP_ID --上次保护员工
	  ,G.EMP_NAME AS LAST_PRTCT_EMP_NAME --上次保护员工人名称
	  ,G.POST_NAME AS LAST_PRTCT_POST_NAME --上次保护员工人职位名称
      ,A.LAST_PROTECT_CLIENT_ID AS LAST_PRTCT_USER_ID --上次保护客户ID 
	  ,I.REAL_NAME AS LAST_PRTCT_USER_NAME --上次保护客户名称
      ,A.CONFLICT_PARAM	--冲突参数
      ,A.REMARK	--备注
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_CLIENT_FOLLOW_EXAMINATION_1030 A
  LEFT JOIN DWRDIM.DWR_DIM_DMT_EMP_INFO_D B
    ON A.EP_ID = B.EP_ID
   AND A.APP_ID = B.APP_ID
   AND A.CREATE_BY = B.EMP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_EMP_INFO_D C
    ON A.EP_ID = C.EP_ID
   AND A.APP_ID = C.APP_ID
   AND A.MODIFY_BY = C.EMP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_EMP_INFO_D D
    ON A.EP_ID = D.EP_ID
   AND A.APP_ID = D.APP_ID
   AND A.FOLLOW_EMPLOYEE_ID = D.EMP_ID
  LEFT JOIN SDI.SDI_CLIENT_STAGE_1030 E
    ON A.STAGE_ID = E.STAGE_ID
   AND A.EP_ID = E.EP_ID	
   AND E.STATUS = 1
   AND E.IS_DELETED = 0
  LEFT JOIN DWRDIM.DWR_DIM_DMT_EMP_INFO_D F
    ON A.EP_ID = F.EP_ID
   AND A.APP_ID = F.APP_ID
   AND A.EXAMINATION_EMPLOYEE_ID = F.EMP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_EMP_INFO_D G
    ON A.EP_ID = G.EP_ID
   AND A.APP_ID = G.APP_ID
   AND A.LAST_PROTECT_EMPLOYEE_ID = G.EMP_ID 
  LEFT JOIN SDI.SDI_CLIENT_INFO_1030 H
    ON A.CLIENT_ID = H.ID
   AND A.EP_ID = H.EP_ID
   AND A.APP_ID = H.APP_ID
  LEFT JOIN SDI.SDI_CLIENT_INFO_1030 I
    ON A.LAST_PROTECT_CLIENT_ID = I.ID
   AND A.EP_ID = I.EP_ID
   AND A.APP_ID = I.APP_ID
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 J
    ON H.SOURCE = J.ID
   AND A.EP_ID = J.EP_ID
   AND J.DEL_FLAG = 'N'
   AND J.DELETED = 0   
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 K
    ON J.PARENT_ID = K.ID
   AND A.EP_ID = K.EP_ID	
   AND K.DEL_FLAG = 'N'
   AND K.DELETED = 0
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 L
    ON A.EP_ID = L.EP_ID
   AND A.STORE_ID = L.ID
   AND L.STATUS = 1
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 M
    ON A.EP_ID = M.EP_ID
   AND A.LAST_PROTECT_STORE_ID = M.ID
   AND M.STATUS = 1 
 WHERE A.IS_DELETED = 0
--瓷砖epId:3044  appId:1831
--卫浴epId:3045  appId:1832  
   AND CASE WHEN A.EP_ID = 3044 AND A.APP_ID = 1831 THEN 1
            WHEN A.EP_ID = 3045 AND A.APP_ID = 1832 THEN 1
			ELSE 0
			 END = 1
