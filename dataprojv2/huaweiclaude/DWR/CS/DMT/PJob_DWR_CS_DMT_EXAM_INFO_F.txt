INSERT OVERWRITE TABLE DWRCS.DWR_CS_DMT_EXAM_INFO_F
SELECT
       A.ID --ID
      ,A.CREATE_BY --创建人
      ,A.CREATE_TIME --创建时间
      ,A.MODIFY_BY  --修改人
      ,A.MODIFY_TIME --修改时间
      ,A.STORE_ID	--门店ID
	  ,B.NAME AS STORE_NAME --门店名称
	  ,B.PROVINCE AS PROV_NAME --门店所在省份
	  ,B.CITY AS CITY_NAME --门店所在城市
	  ,B.OUT_STORE_ID --OA门店编码
	  ,B.PARTICIPATION_IN_BUSINESS AS BUSI_TYPE --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）
      ,F.LVL1_DEPT_NAME --一级部门名称
	  ,F.LVL2_DEPT_NAME --二级部门名称
	  ,F.LVL3_DEPT_NAME --三级部门名称
	  ,F.LVL4_DEPT_NAME --四级部门名称
	  ,F.LVL5_DEPT_NAME --五级部门名称
	  ,F.LVL6_DEPT_NAME --六级部门名称
	  ,F.LVL7_DEPT_NAME --七级部门名称
	  ,F.LVL8_DEPT_NAME --八级部门名称
	  ,A.CLIENT_ID AS CUST_ID --客户ID
	  ,D.REAL_NAME AS CUST_REAL_NAME --客户真实姓名
	  ,D.NICK_NAME AS CUST_NICK_NAME --客户昵称
      ,A.FOLLOW_EMPLOYEE_ID AS FOLW_EMP_ID --跟进员工ID（触发人）
	  ,C.NAME AS FOLW_EMP_NAME --跟进员工名称（触发人）
      ,A.BIND_ID	--客户绑定表ID
      ,A.ORDER_ID	--客户订单表ID
      ,A.STAGE_ID	--客户阶段ID
	  ,G.STAGE_NAME --客户阶段名称
      ,A.FOLLOW_RULE_ID AS FOLW_RULE_ID --报备设置ID
      ,A.FOLLOW_RULE_DETAIL_ID AS FOLW_RULE_DET_ID --报备设置详情ID
      ,A.FOLLOW_RULE_DETAIL_DAY_ID AS FOLW_RULE_DET_DAY_ID --报备天数设置ID
      ,A.FOLLOW_RULE_DETAIL_AREA_DAY_ID AS FOLW_RULE_DET_AREA_DAY_ID --区域报备天数设置ID
      ,A.TYPE AS EXAM_TYPE_CODE --类型编码 1.报备审核 2.奖励审核
	  ,CASE WHEN A.TYPE = 1 THEN '报备审核'
	        WHEN A.TYPE = 2 THEN '奖励审核'
			 END AS EXAM_TYPE_NAME --类型名称
      ,A.TRIGGER_TIME AS TRIG_TIME --触发时间
      ,A.STATUS	AS EXAM_STATUS_CODE --状态 -1.已拒绝 0.待审核 1.审核通过
	  ,CASE WHEN A.STATUS = -1 THEN '已拒绝'
	        WHEN A.STATUS = 0 THEN '待审核'
			WHEN A.STATUS = 1 THEN '审核通过'
			 END AS EXAM_STATUS_NAME --状态名称
      ,A.EXAMINATION_EMPLOYEE_ID AS EXAM_EMP_ID --审核员工ID
	  ,E.NAME AS EXAM_EMP_NAME --审核员工名称
      ,A.EXAMINATION_TIME AS EXAM_TIME --审核时间
      ,A.REWARD_AMOUNT AS RWARD_AMT --奖励金额
      ,A.REJECT_REASON AS REJ_RSN --审核不通过原因
      ,A.LAST_PROTECT_STORE_ID AS LAST_PRTCT_STORE_ID --上次保护门店
      ,A.LAST_PROTECT_EMPLOYEE_ID AS LAST_PRTCT_EMP_ID --上次保护员工
      ,A.LAST_PROTECT_CLIENT_ID AS LAST_PRTCT_CUST_ID --上次保护客户
      ,A.REMARK	--备注
	  ,EXAMINE_NO AS EXAM_NUM --检查编码
      ,EXAMIMATION_DURATION AS EXAM_DUR --审核时长
      ,APPLY_PHONE AS APPLY_PHONE_NUM --申请人手机号码
      ,APPLY_ADDRESS AS APPLY_ADDR --申请人地址
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,F.DEALER_NAME AS DEALER_NAME --经销商名称
	  ,F.DEALER_NO AS DEALER_NUM --经销商编号
	  ,D.SOURCE AS CUST_SRC_CODE --客户来源
	  ,H.NAME AS CUST_SRC_NAME --客户来源名称
	  ,COALESCE(I.NAME,H.NAME) AS CUST_SRC_CAT_NAME --客户来源类别名称	  
  FROM SDI.SDI_CLIENT_FOLLOW_EXAMINATION_1030 A
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 B
    ON A.STORE_ID = B.ID
   AND A.APP_ID = B.APP_ID
   AND A.EP_ID = B.EP_ID
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_1030 C
    ON A.FOLLOW_EMPLOYEE_ID = C.ID
   AND A.APP_ID = C.APP_ID
   AND A.EP_ID = C.EP_ID
  LEFT JOIN SDI.SDI_CLIENT_INFO_1030 D
    ON A.CLIENT_ID = D.ID
   AND A.APP_ID = D.APP_ID
   AND A.EP_ID = D.EP_ID
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_1030 E
    ON A.EXAMINATION_EMPLOYEE_ID = E.ID
   AND A.APP_ID = E.APP_ID
   AND A.EP_ID = E.EP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_DEPT_INFO_D F
    ON A.STORE_ID = F.STORE_ID
   AND A.APP_ID = F.APP_ID
   AND A.EP_ID = F.EP_ID
  LEFT JOIN SDI.SDI_CLIENT_STAGE_1030 G
    ON A.STAGE_ID = G.STAGE_ID
   AND A.EP_ID = G.EP_ID
   AND G.IS_DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 H
    ON D.SOURCE = H.ID
   AND A.EP_ID = H.EP_ID
   AND H.DEL_FLAG = 'N'
   AND H.DELETED = 0   
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 I
    ON H.PARENT_ID = I.ID
   AND A.EP_ID = I.EP_ID	
   AND I.DEL_FLAG = 'N'
   AND I.DELETED = 0   
 WHERE A.IS_DELETED = 0
   AND A.APP_ID = 1831
   AND A.EP_ID = 3044
