WITH T1 AS(
SELECT DISTINCT
       AAA.CLIENT_ID
      ,AAA.STORE_ID
      ,CONCAT(CONCAT(CONCAT(CONCAT(BBB.MODEL,'/'),BBB.`NAME`),'/'),AAA.ITEM_NUM) AS ITEM_LIST_NAME --产品名称
  FROM SDI.SDI_CLIENT_ITEM_BIND_1030 AS AAA
 INNER JOIN SDI.SDI_ITEM_1030 BBB
   ON AAA.ITEM_NO = BBB.ITEM_NO
 WHERE AAA.DELETED = 0
),T2 AS(
SELECT CLIENT_ID
      ,STORE_ID
	  ,CONCAT_WS(';',COLLECT_LIST(ITEM_LIST_NAME)) AS ITEM_LIST_NAME --产品名称
  FROM T1
 GROUP BY 
       CLIENT_ID
      ,STORE_ID
)
INSERT OVERWRITE TABLE DWRCS.DWR_CS_DMT_LIFETIME_F
SELECT
       CLIENT.ID AS ID 
      ,CLIENT.REAL_NAME AS REAL_NAME --姓名
      ,CLIENT.NICK_NAME AS NICK_NAME --昵称
      ,CLIENT.PHONE AS PHONE_NUM --手机号码
      ,BIND.SOURCE AS SOURCE_CODE --客户来源
      ,SOUR.NAME AS SOURCE_NAME --客户来源名称
      ,SOUR1.NAME AS SOURCE_CAT_NAME --客户来源类别名称
      ,SOUR.LINK_NAME AS SOURCE_LINK_NAME --客户来源链式名称
      ,CLIENT.PROTECT_STATUS AS PROTECT_STATUS --保护状态

	        WHEN CLIENT.PROTECT_STATUS = 1 THEN '保护期内'
			WHEN CLIENT.PROTECT_STATUS = 2 THEN '保护过期'
			 END AS PROTECT_STATUS_NAME --保护状态名称
      ,CLIENT.PROTECT_AREA_STATUS AS PROTECT_AREA_STATUS --区域保护状态
      ,CASE WHEN CLIENT.PROTECT_AREA_STATUS = 0 THEN '无保护'
	        WHEN CLIENT.PROTECT_AREA_STATUS = 1 THEN '保护期内'
			WHEN CLIENT.PROTECT_AREA_STATUS = 2 THEN '保护过期'
			 END AS PROTECT_AREA_STATUS_NAME --区域保护状态名称
      ,CLIENT.PROTECT_EMPLOYEE_ID AS PROTECT_EMP_ID --保护员工
      ,CLIENT.PROTECT_STORE_ID AS PROTECT_STORE_ID --保护门店
      ,CLIENT.PROTECT_AREA_STORE_ID AS PROTECT_AREA_STORE_ID --区域保护门店
      ,CLIENT.PROTECT_START_TIME AS PROTECT_START_TIME --保护开始时间
      ,CLIENT.PROTECT_AREA_START_TIME AS PROTECT_AREA_START_TIME --区域保护开始时间
      ,CASE WHEN TICKET.ID IS NOT NULL THEN '是'
	        ELSE '否'
			 END AS IS_RECEIPT_COUPON --是否接受优惠券
      ,TICKET.RECEIVE_WAY AS RECEIVE_WAY --领取方式
	  ,CASE WHEN TICKET.RECEIVE_WAY = 0 THEN '页面转发'
	        WHEN TICKET.RECEIVE_WAY = 1 THEN '领券中心'
			WHEN TICKET.RECEIVE_WAY = 2 THEN '首礼'
			WHEN TICKET.RECEIVE_WAY = 3 THEN '群发'
			 END AS RECEIVE_WAY_NAME --领取方式名称
      ,TICKET.EMPLOYEE_ID AS TICKET_EMP_ID --发放优惠券导购员工ID
      ,TICKET.RECEIVE_TIME AS TICKET_RECEIVE_TIME --领取时间
      ,COUPON.NAME AS TICKET_NAME --优惠券名称
      ,COUPON.COUPON_CATEGORY AS COUPON_CAT_CODE --优惠券类别
	  ,CASE WHEN COUPON.COUPON_CATEGORY = 0 THEN '普通优惠券'
	        WHEN COUPON.COUPON_CATEGORY = 1 THEN '运费券'
			 END AS COUPON_CAT_NAME --优惠券类别名称
      ,TICKET.STATUS AS TICKET_STATUS --优惠券状态
	  ,CASE WHEN TICKET.STATUS = 0 THEN '无效'
	        WHEN TICKET.STATUS = 1 THEN '有效'
			WHEN TICKET.STATUS = 2 THEN '已使用'
			WHEN TICKET.STATUS = 3 THEN '已过期'
			WHEN TICKET.STATUS = 4 THEN '已转送'
			 END AS TICKET_STATUS_NAME --优惠券状态名称
      ,TICKET.RECEIVE_STORE_ID AS TICKET_USE_STORE_ID --领取的门店ID
      ,MEMBER.ID AS MEMBER_ID --用户ID
	  ,MEMBER.CREATE_TIME AS MEMBER_CREATE_TIME --创建时间
      ,BIND.BIND_TIME AS BIND_TIME --绑定时间
      ,CLIENT.CREATE_TIME AS CREATE_TIME --用户创建时间
	  ,BIND.STAGE_ID --客户阶段
      ,STAGE.STAGE_NAME AS STAGE_NAME --客户阶段名称
      ,CASE WHEN MEMBER.GENDER = 1 THEN '男性' 
	        WHEN MEMBER.GENDER = 2 THEN '女性'
			ELSE ''
			 END AS GENDER --性别
      ,CLIENT.AGE_GROUP AS AGE_GROUP --年龄层
	  ,CASE WHEN CLIENT.AGE_GROUP = 1 THEN '18-25岁'
	        WHEN CLIENT.AGE_GROUP = 2 THEN '26-35岁'
			WHEN CLIENT.AGE_GROUP = 3 THEN '36岁及以上'
			ELSE ''
			 END AS AGE_GROUP_NAME --年龄层
      ,BIND.DECORATE_PREDILECTION AS DEC_PRCT_CODE --装修风格
      ,DECORATE.NAME AS DEC_PRCT_NAME --装修风格名称
	  ,BIND.ID AS BIND_ID --绑定ID
      ,BIND.PROVINCE AS BIND_PROVINCE_NAME --绑定省份
      ,BIND.CITY AS BIND_CITY_NAME --绑定城市
      ,BIND.DISTRICT AS BIND_DISTRICT_NAME --绑定区域
      ,BIND.COMMUNITY_ADDRESS AS CMY_ADDR --小区名称
      ,BIND.DETAILED_ADDRESS AS DTL_ADDR --详细地址
      ,BIND.FOLLOW_TIME AS FOLOW_TIME --跟进时间
      ,ORD.DEPOSIT_TIME AS DEPOSIT_TIME --交定时间
      ,ORD.CREATE_ORDER_TIME AS ORDER_TIME --开单时间
      ,ORD.BARGAIN_MONEY AS BRGN_AMT --定金
      ,COALESCE(ORD.ORDER_MONEY,0) - COALESCE(ORD.BARGAIN_MONEY,0) AS BAL_AMT --余额
      ,ORD.ID AS ORDER_ID --订单ID
	  ,T2.ITEM_LIST_NAME --产品名称
	  ,ORD.ORDER_MONEY AS ORDER_AMT --订单金额
      ,BIND.LITTLE_DEPOSIT AS LITTLE_DEPOSIT_AMT --小定金额
      ,BIND.BIG_DEPOSIT AS BIG_DEPOSIT_AMT --大定金额
      ,CASE WHEN SH.IS_315_REWARD = 1 THEN '是' 
	        ELSE '否' 
			 END AS IS_315_REWARD --是否315奖励
      ,CASE WHEN SH.IS_PRODUCT_REWARD = 1 THEN '是'
	        ELSE '否'
			 END AS IS_PRODUCT_REWARD --是否产品激励奖励
      ,SH.CREATE_TIME AS SHARE_TIME --分享时间
      ,CASE WHEN SH.CREATE_TIME IS NOT NULL THEN '是'
	        ELSE '否'
			 END AS IS_SHARE_REWARD --是否分享奖励           
      ,STORE.`ID` AS STORE_ID --门店ID
      ,STORE.`NAME` AS STORE_NAME --门店名称
      ,DEPT.`DEPT_ID` AS DEPT_ID --部门ID	  
      ,DEPT.NAME AS DEPT_NAME --部门名称
      ,STORE.`ABBREVIATION` AS STORE_BY_NAME --门店简称
      ,STORE.CONTACT_PHONE AS STORE_CONTACT_PHONE --门店联系人电话
      ,CASE WHEN STORE.IS_OPEN = 1 THEN '营业'
	        ELSE '打烊'
			 END  AS STORE_STATUS_NAME --门店状态
      ,STORE.STORE_TAG AS STORE_TAG --门店标签
      ,STORE.OUT_STORE_ID AS OUT_STORE_ID --外部门店编码
      ,EMP.`NAME` AS EMP_NAME --员工名称
      ,EMP.`PHONE` AS EMP_PHONE_NUM --员工电话
      ,STORE.PROVINCE AS STORE_PROVINCE_NAME --门店所在省份
      ,STORE.CITY AS STORE_CITY_NAME --门店所在城市
      ,STORE.DISTRICT AS STORE_DISTRICT_NAME --门店所在区域
      ,STORE.ADDRESS AS STORE_ADDR --门店地址
      ,DEPT.DMT_SALESMAN_NAME AS DMT_SALE_NAME --DMT业务员姓名
      ,DEPT.O2O_SALESMAN_NAME AS O2O_SALE_NAME --O2O业务员姓名
      ,CASE WHEN REGULAR.ID IS NOT NULL THEN '是'
	        ELSE '否'
			 END AS IS_REG --是否定时客户
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间		 
  FROM SDI.SDI_CLIENT_INFO_1030 AS CLIENT
 INNER JOIN SDI.SDI_CLIENT_EMPLOYEE_BIND_1030 AS BIND
    ON CLIENT.ID = BIND.CLIENT_ID
   AND BIND.IS_DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_BIND_CLIENTNUM_1030 AS NUM
    ON BIND.CLIENT_ID = NUM.CLIENT_ID
   AND BIND.STORE_ID = NUM.STORE_ID
   AND NUM.DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_SHARE_RECORD_1030 AS SH
    ON BIND.CLIENT_ID = SH.CLIENT_ID
   AND BIND.STORE_ID = SH.STORE_ID
   AND BIND.EMPLOYEE_ID = SH.EMPLOYEE_ID
  LEFT JOIN SDI.SDI_CLIENT_ORDER_INFO_1030 AS ORD
    ON ORD.CLIENT_ID = BIND.CLIENT_ID 
   AND ORD.STORE_ID = BIND.STORE_ID
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 AS SOUR
    ON BIND.SOURCE = SOUR.ID 
   AND SOUR.DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 AS SOUR1
    ON SOUR.PARENT_ID  = SOUR1.ID 
   AND SOUR1.DELETED = 0 
  LEFT JOIN SDI.SDI_CLIENT_STAGE_1030 AS STAGE
    ON STAGE.STAGE_ID = BIND.STAGE_ID
   AND STAGE.EP_ID = BIND.EP_ID
   AND STAGE.IS_DELETED = 0
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_1030 AS EMP
    ON EMP.ID = BIND.EMPLOYEE_ID
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 AS STORE
    ON STORE.ID = BIND.STORE_ID
  LEFT JOIN SDI.SDI_BUZ_DEPT_INFO_1030 AS DEPT
    ON DEPT.EP_ID = STORE.EP_ID
   AND DEPT.STORE_ID = STORE.ID
  LEFT JOIN SDI.SDI_MEM_USER_1030 AS MEMBER
    ON MEMBER.ID = CLIENT.MEMBER_ID
  LEFT JOIN SDI.SDI_CLIENT_REGULAR_CUSTOMER_1030 AS REGULAR
    ON REGULAR.CLIENT_ID = CLIENT.ID
  LEFT JOIN (SELECT *
                   ,ROW_NUMBER() OVER(PARTITION BY RECEIPTOR_ID,RECEIVE_STORE_ID ORDER BY ID DESC) AS RN
               FROM SDI.SDI_MT_COUPON_TICKET_1030
			) AS TICKET 
    ON TICKET.RECEIPTOR_ID = CLIENT.MEMBER_ID
   AND TICKET.RECEIVE_STORE_ID = BIND.STORE_ID
   AND TICKET.RN = 1
  LEFT JOIN SDI.SDI_MT_COUPON_INFO_1030 AS COUPON
    ON COUPON.ID = TICKET.COUPON_INFO_ID
  LEFT JOIN SDI.SDI_CLIENT_DECORATE_PREDILECTION_1030 AS DECORATE
    ON DECORATE.ID = BIND.DECORATE_PREDILECTION
  LEFT JOIN T2
    ON T2.CLIENT_ID = CLIENT.ID
   AND T2.STORE_ID = BIND.STORE_ID
