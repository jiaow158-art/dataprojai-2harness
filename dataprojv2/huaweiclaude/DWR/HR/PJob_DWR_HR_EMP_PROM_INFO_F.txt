INSERT OVERWRITE TABLE DWRHR.DWR_HR_EMP_PROM_INFO_F
SELECT A.PERNR AS EMP_NUM --工号
      ,G.EMP_NAME --姓名
      ,A.MASSN AS OP_TYPE_CODE --操作类型编码
	  ,C.MNTXT AS OP_TYPE_NAME --操作类型描述
	  ,A.MASSG AS OP_RSN_CODE --操作原因编码
	  ,D.MGTXT AS OP_RSN_NAME --操作原因描述
	  ,A.STAT1 AS CUST_SPEC_ST_CODE --客户特定状态编码
	  ,E.TEXT1 AS CUST_SPEC_ST_NAME --客户特定状态描述
	  ,A.STAT2 AS EMP_ST_CODE --雇佣状态编码
	  ,F.TEXT1 AS EMP_ST_NAME --雇佣状态描述
      ,SUBSTR(A.BEGDA,1,4) || '-' || SUBSTR(A.BEGDA,5,2) || '-' || SUBSTR(A.BEGDA,7,2) AS START_DATE --开始日期
	  ,SUBSTR(A.ENDDA,1,4) || '-' || SUBSTR(A.ENDDA,5,2) || '-' || SUBSTR(A.ENDDA,7,2) AS END_DATE --结束日期
	  ,SUBSTR(A.AEDTM,1,4) || '-' || SUBSTR(A.AEDTM,5,2) || '-' || SUBSTR(A.AEDTM,7,2) AS UPDATE_DATE --更新日期
	  ,A.UNAME AS UPDATE_EMP_NUM --更新人工号
	  ,B.EMP_NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_PA0000_1001 A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON A.UNAME = B.EMP_NUM
  LEFT JOIN SDI.SDI_T529T_1001 C
    ON A.MASSN = C.MASSN
   AND C.MANDT = 800
   AND C.SPRSL = '1'
  LEFT JOIN SDI.SDI_T530T_1001 D
    ON A.MASSN = D.MASSN
   AND A.MASSG = D.MASSG
   AND D.MANDT = 800
   AND D.SPRSL = '1'
  LEFT JOIN SDI.SDI_T529U_1001 E
    ON A.STAT1 = E.STATV
   AND E.MANDT = 800
   AND E.SPRSL = '1'
   AND E.STATN = '1'
  LEFT JOIN SDI.SDI_T529U_1001 F
    ON A.STAT2 = F.STATV
   AND F.MANDT = 800
   AND F.SPRSL = '1'
   AND F.STATN = '2'
  LEFT JOIN DM.DM_WF_EMP_INFO_T G
    ON A.PERNR = G.EMP_NUM
 WHERE A.MANDT = 800
