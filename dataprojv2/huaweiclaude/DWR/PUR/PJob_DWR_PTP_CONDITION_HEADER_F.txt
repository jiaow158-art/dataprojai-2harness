---SAP条件抬头表
set hive.cbo.enable=true;				--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWRPUR.DWR_PTP_CONDITION_HEADER_F		--DWR.SAP条件抬头表
SELECT
	 DPCH.GROUP_NUM						AS	GROUP_NUM
	,DPCH.CONDITION_RECORD_NO            AS	CONDITION_RECORD_NO
	,DPCH.RECORD_CREATE_DATE             AS	RECORD_CREATE_DATE
	,DPCH.APPLICATION_PROGRAM_CODE       AS	APPLICATION_PROGRAM_CODE
	,DPCH.CREATE_BY                      AS	CREATE_BY
	,DPCH.START_DATE                     AS	START_DATE
	,DPCH.CONDITION_TABLE                AS	CONDITION_TABLE
	,DPCH.VALID_EXPIRY_DATE              AS	VALID_EXPIRY_DATE
	,DPCH.CONDITION_TYPE                 AS	CONDITION_TYPE
	,NULL                    AS	CONDITION_TYPE_NAME
	,DPCH.REFER                          AS	REFER
	,DPCH.CONDITION_PURPOSE              AS	CONDITION_PURPOSE
	,DPCH.ALLOW_PERMIT_DATE              AS	ALLOW_PERMIT_DATE
	,DPCH.DEL_FLAG                       AS	DEL_FLAG
	,CURRENT_TIMESTAMP()            AS	DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_CONDITION_HEADER_T DPCH		--DWI.SAP条件抬头表
WHERE DPCH.DEL_FLAG = 'N'
