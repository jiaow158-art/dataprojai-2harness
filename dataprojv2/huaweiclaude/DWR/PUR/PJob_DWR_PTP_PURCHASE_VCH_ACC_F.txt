INSERT OVERWRITE TABLE DWRPUR.DWR_PTP_PURCHASE_VCH_ACC_F	--DWR-SAP采购凭证中的帐户设置表
SELECT
	 DPPVA.GROUP_NUM							AS	GROUP_NUM
	,DPPVA.PURCHASE_VOUCHER_NUM               AS	PURCHASE_VOUCHER_NUM
	,DPPVA.PURCHASE_VCH_ITEM_NUM              AS	PURCHASE_VCH_ITEM_NUM
	,DPPVA.PURCHASE_VCH_ACC_SERIALNO          AS	PURCHASE_VCH_ACC_SERIALNO
	,DPPVA.QUANTITY                           AS	QUANTITY
	,DPPVA.SALES_BILL_ITEM                    AS	SALES_BILL_ITEM
	,DPPVA.MORE_ACCOUNT_PERCENT               AS	MORE_ACCOUNT_PERCENT
	,DPPVA.COST_CENTER_CODE                   AS	COST_CENTER_CODE
	,T.CONFIG_NAME                        AS	COST_CENTER_NAME
	,DPPVA.ORDER_NUM                          AS	ORDER_NUM
	,DPPVA.PLAN_ITEM                          AS	PLAN_ITEM
	,DPPVA.BUSINESS_SCOPE                     AS	BUSINESS_SCOPE
	,T1.CONFIG_NAME                        AS	BUSINESS_SCOPE_NAME
	,DPPVA.GL_ACCOUNT_NUM                     AS	GL_ACCOUNT_NUM
	,DPPVA.CONTROL_SCOPE                      AS	CONTROL_SCOPE
	,T2.CONFIG_NAME                        AS	CONTROL_SCOPE_NAME
	,DPPVA.SALES_VCH_NUM                      AS	SALES_VCH_NUM
	,DPPVA.MAIN_ASSET_NUM                     AS	MAIN_ASSET_NUM
	,DPPVA.PROFIT_CENTER                      AS	PROFIT_CENTER
	,T3.CONFIG_NAME                        AS	PROFIT_CENTER_NAME
	,DPPVA.FUNCTION_SCOPE                     AS	FUNCTION_SCOPE
	,DPPVA.DISCHARGE                          AS	DISCHARGE
	,DPPVA.FUND_CENTER_CODE                   AS	FUND_CENTER_CODE
	,DPPVA.ASSETS_SON_CODE                    AS	ASSETS_SON_CODE
	,DPPVA.PROFITABILITY_SEGMENT_CODE         AS	PROFITABILITY_SEGMENT_CODE
	,DPPVA.WBS_ELEMENT                        AS	WBS_ELEMENT
	,T4.CONFIG_NAME                        AS	WBS_ELEMENT_NAME
	,DPPVA.COMMITMENT_ITEM_ID                 AS	COMMITMENT_ITEM_ID
	,T5.CONFIG_NAME                        AS	COMMITMENT_ITEM_NAME
	,DPPVA.RECORD_CREATE_DATE                 AS	RECORD_CREATE_DATE
	,DPPVA.TR_RESERVE_VCH_ITEM                AS	TR_RESERVE_VCH_ITEM
	,DPPVA.ORDER_PROCESS_ROUTE_NO1            AS	ORDER_PROCESS_ROUTE_NO1
	,DPPVA.ORDER_PROCESS_ROUTE_NO2            AS	ORDER_PROCESS_ROUTE_NO2
	,DPPVA.SERIVICE_DATE                      AS	SERIVICE_DATE
	,DPPVA.DEL_FLAG                           AS	DEL_FLAG
	,CURRENT_TIMESTAMP()                AS	DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_PURCHASE_VCH_ACC_T DPPVA		--DWI-SAP采购凭证中的帐户设置表
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T		--关联通用配置表
	 ON T.CONFIG_CODE = DPPVA.COST_CENTER_CODE		--取成本中心中文名称
	 AND T.GROUP_NUM = DPPVA.GROUP_NUM				--集团
	 AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1		--关联通用配置表
	 ON T1.CONFIG_CODE = DPPVA.BUSINESS_SCOPE		--取业务范围中文名称
	 AND T1.GROUP_NUM = DPPVA.GROUP_NUM 			--集团
	 AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表
	 ON T2.CONFIG_CODE = DPPVA.CONTROL_SCOPE		--取控制范围中文名称
	 AND T2.GROUP_NUM = DPPVA.GROUP_NUM				--集团
	 AND T2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T3		--关联通用配置表
	 ON T3.CONFIG_CODE = DPPVA.PROFIT_CENTER		--取利润中心中文名称
	 AND T3.GROUP_NUM = DPPVA.GROUP_NUM				--集团
	 AND T3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4		--关联通用配置表
	 ON T4.CONFIG_CODE = DPPVA.WBS_ELEMENT			--取工作分解结构元素 (WBS 元素)中文名称
	 AND T4.GROUP_NUM = DPPVA.GROUP_NUM				--集团
	 AND T4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T5		--关联通用配置表
	 ON T5.CONFIG_CODE = DPPVA.COMMITMENT_ITEM_ID	--取承诺项目中文名称
	 AND T5.GROUP_NUM = DPPVA.GROUP_NUM				--集团
	 AND T5.LANG_CODE = '1'
WHERE DPPVA.DEL_FLAG = 'N'
