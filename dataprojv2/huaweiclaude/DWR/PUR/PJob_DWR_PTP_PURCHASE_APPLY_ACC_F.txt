set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWRPUR.DWR_PTP_PURCHASE_APPLY_ACC_F		--DWR.SAP采购申请账户设置表
SELECT
	 DPPAA.GROUP_NUM						  AS	GROUP_NUM
	,DPPAA.PURCHASE_APPLY_NUM                 AS	PURCHASE_APPLY_NUM
	,DPPAA.PURCHASE_APPLY_ITEM_NUM            AS	PURCHASE_APPLY_ITEM_NUM
	,DPPAA.PURCHASE_APPLY_ACC_SERIALNO        AS	PURCHASE_APPLY_ACC_SERIALNO
	,DPPAA.SALE_VOUCHER_ITEM                  AS	SALE_VOUCHER_ITEM
	,DPPAA.PURCHASE_APPLY_QTY                 AS	PURCHASE_APPLY_QTY
	,DPPAA.COST_CENTER_CODE                   AS	COST_CENTER_CODE
	,T.CONFIG_NAME                        	  AS	COST_CENTER_NAME
	,DPPAA.ORDER_NUM                          AS	ORDER_NUM
	,DPPAA.PLAN_ITEM                          AS	PLAN_ITEM
	,DPPAA.BUSINESS_SCOPE                     AS	BUSINESS_SCOPE
	,T1.CONFIG_NAME                        	  AS	BUSINESS_SCOPE_NAME
	,DPPAA.GL_ACCOUNT_NUM                     AS	GL_ACCOUNT_NUM
	,DPPAA.CONTROL_SCOPE                      AS	CONTROL_SCOPE
	,T2.CONFIG_NAME                        	  AS	CONTROL_SCOPE_NAME
	,DPPAA.SALES_VCH_NUM                      AS	SALES_VCH_NUM
	,DPPAA.MAIN_ASSET_NUM                     AS	MAIN_ASSET_NUM
	,DPPAA.PROFIT_CENTER                      AS	PROFIT_CENTER
	,T3.CONFIG_NAME                        	  AS	PROFIT_CENTER_NAME
	,DPPAA.FUNCTION_SCOPE                     AS	FUNCTION_SCOPE
	,DPPAA.DISCHARGE                          AS	DISCHARGE
	,DPPAA.FUND_CENTER_CODE                   AS	FUND_CENTER_CODE
	,DPPAA.ASSETS_SON_CODE                    AS	ASSETS_SON_CODE
	,DPPAA.PROFITABILITY_SEGMENT_CODE         AS	PROFITABILITY_SEGMENT_CODE
	,DPPAA.COMMITMENT_ITEM_ID                 AS	COMMITMENT_ITEM_ID
	,T4.CONFIG_NAME                        	  AS	COMMITMENT_ITEM_NAME
	,DPPAA.RECORD_CREATE_DATE                 AS	RECORD_CREATE_DATE
	,DPPAA.TR_RESERVE_VCH_ITEM                AS	TR_RESERVE_VCH_ITEM
	,DPPAA.WBS_ELEMENT                        AS	WBS_ELEMENT
	,T5.CONFIG_NAME                        	  AS	WBS_ELEMENT_NAME
	,DPPAA.ORDER_PROCESS_ROUTE_NO1            AS	ORDER_PROCESS_ROUTE_NO1
	,DPPAA.ORDER_PROCESS_ROUTE_NO2            AS	ORDER_PROCESS_ROUTE_NO2
	,DPPAA.SERIVICE_DATE                      AS	SERIVICE_DATE
    ,DPPAA.DEL_FLAG                           AS  DEL_FLAG
    ,CURRENT_TIMESTAMP()                	  AS  DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_PURCHASE_APPLY_ACC_T DPPAA		--DWI.SAP采购申请账户设置表
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T		--关联通用配置表
	 ON T.CONFIG_CODE = DPPAA.COST_CENTER_CODE		--取成本中心中文名称
	 AND T.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T1		--关联通用配置表
	 ON T1.CONFIG_CODE = DPPAA.BUSINESS_SCOPE		--取业务范围中文名称
	 AND T1.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表
	 ON T2.CONFIG_CODE = DPPAA.CONTROL_SCOPE		--取控制范围中文名称
	 AND T2.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T3		--关联通用配置表
	 ON T3.CONFIG_CODE = DPPAA.PROFIT_CENTER		--取利润中心中文名称
	 AND T3.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4		--关联通用配置表
	 ON T4.CONFIG_CODE = DPPAA.COMMITMENT_ITEM_ID	--取承诺项目中文名称
	 AND T4.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T5		--关联通用配置表
	 ON T5.CONFIG_CODE = DPPAA.WBS_ELEMENT			--取工作分解结构元素 (WBS 元素)中文名称
	 AND T5.GROUP_NUM = DPPAA.GROUP_NUM				--集团
	 AND T5.LANG_CODE = '1'
WHERE DPPAA.DEL_FLAG = 'N'
