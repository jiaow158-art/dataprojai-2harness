---SAP采购一般数据表
set hive.cbo.enable=true;													--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWRPUR.DWR_PTP_PURCHASE_GENERAL_F		--DWR.SAP采购一般数据表
SELECT
	 DPPG.GROUP_NUM						AS	GROUP_NUM
	,DPPG.PURCHASE_INFO_RECORD_NO        AS	PURCHASE_INFO_RECORD_NO
	,DPPG.RECORD_CREATE_DATE             AS	RECORD_CREATE_DATE
	,DPPG.CREATE_BY                      AS	CREATE_BY
	,DPPG.LONG_UTC_TIMESTAMP             AS	LONG_UTC_TIMESTAMP
	,DPPG.MATERIAL_ID                    AS	MATERIAL_ID
	,DPPG.MATERIAL_NUM                   AS	MATERIAL_NUM
	,T.MATERIAL_NAME                  AS	MATERIAL_NAME
	,DPPG.SUPPLIER_OR_CREDITOR_ID        AS	SUPPLIER_OR_CREDITOR_ID
	,DPPG.SUPPLIER_OR_CREDITOR_NUM       AS	SUPPLIER_OR_CREDITOR_NUM
	,T1.SUPPLIER_OR_CREDITOR_NAME      AS	SUPPLIER_OR_CREDITOR_NAME
	,DPPG.REGION_CODE                    AS	REGION_CODE
	,DPPG.BASE_UNIT                      AS	BASE_UNIT
	,DPPG.PURCHASE_INFO                  AS	PURCHASE_INFO
	,DPPG.PURCHASE_INFO_FLAG             AS	PURCHASE_INFO_FLAG
	,DPPG.BASE_UNTI_DENOMINATOR          AS	BASE_UNTI_DENOMINATOR
	,DPPG.VENDOR_PHONE_NUMBER            AS	VENDOR_PHONE_NUMBER
	,DPPG.MATERIAL_GROUP                 AS	MATERIAL_GROUP
	,T2.CONFIG_NAME                    AS	MATERIAL_GROUP_NAME
	,DPPG.PURCHASE_ORDER_UNIT            AS	PURCHASE_ORDER_UNIT
	,DPPG.VENDOR_USE_MATERIAL_NUM        AS	VENDOR_USE_MATERIAL_NUM
	,DPPG.START_DATE                     AS	START_DATE
	,DPPG.NO_STOCK_INFO                  AS	NO_STOCK_INFO
	,DPPG.SALES_USER                     AS	SALES_USER
	,DPPG.BASE_UNTI_NUMERATOR            AS	BASE_UNTI_NUMERATOR
	,DPPG.REMINDER_DAY1                  AS	REMINDER_DAY1
	,DPPG.END_DATE                       AS	END_DATE
	,DPPG.DELIVERY_COUNTRY_CODE          AS	DELIVERY_COUNTRY_CODE
	,DPPG.SORT_SEQ_NUM                   AS	SORT_SEQ_NUM
    ,DPPG.DEL_FLAG                       AS  DEL_FLAG
    ,CURRENT_TIMESTAMP()            AS  DW_LAST_UPDATE_DATE
FROM DWIPUR.DWI_PTP_PURCHASE_GENERAL_T DPPG		--DWI事实表						
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T	--关联物料主数据
	 ON T.MATERIAL_ID = DPPG.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DPPG.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DPPG.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T T1	--关联供应商主数据
	 ON T1.SUPPLIER_OR_CREDITOR_ID = DPPG.SUPPLIER_OR_CREDITOR_ID	--供应商ID
	 AND T1.GROUP_NUM = DPPG.GROUP_NUM		--集团
	 AND T1.END_DATE ='4712-12-31'			--结束时间为'4712-12-31'
	 AND DPPG.DEL_FLAG = 'N'	 			--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表
	 ON T2.CONFIG_CODE = DPPG.MATERIAL_GROUP	--取物料组中文名称
	 AND T2.GROUP_NUM = DPPG.GROUP_NUM			--集团
	 AND T2.LANG_CODE = '1'
WHERE DPPG.DEL_FLAG = 'N'
