insert OVERWRITE table dwrpi.dwr_wf_node_basic_into_f
select 
node_id
,node_name
,is_create_node
,is_return_node
,is_reopen_node
,is_end_node
,node_display_abscissa
,node_display_ordinate
,node_author_group_total
,node_attribute
,pass_branch_num
,is_custom_node
,flow_order
,sign_type
,operators_1
,request_code
,start_node_code
,creator
,ecology_pinyin_search
,flowchart_style
,uuid
,del_flag
,dw_last_update_date
,case when node_name like '%【停用】%' then '停用'
      when node_name like '%【申请】%' then '申请'
	  when node_name like '%【归档】%' then '归档'
	  when node_name like '%【审批】%' then '审批'
	  when node_name like '%【停用流程】%' then '停用'
	  when node_name like '%【操作】%' then '操作'
	  else ''
	   end as node_type --节点类型
,case when substr(node_name,3,1) = 'A' then '主线'
      when substr(node_name,3,1) = 'B' then '并行'
	  else '停用流程'
	   end as is_zx --是否主线
from dwipi.dwi_wf_node_basic_into_t
where del_flag = 'N';
