INSERT OVERWRITE TABLE DWRQMS.DWR_QMS_ERR_ORDER_INFO_F
SELECT  A.HANDLE AS ID --主键
       ,A.SITE AS FACTORY_CODE --工厂编码
	   ,D.SITE_NAME AS FACTORY_NAME --工厂名称
	   ,D.SITE_SHORT_NAME AS FACTORY_SHORT_NAME --工厂简称
       ,A.ERROR_ORDER_NO AS ERR_NUM --异常单号
       ,REPLACE(SPLIT(A.WORK_SHOP_HANDLE,':')[1],A.SITE||',','') AS WORK_SHOP_CODE --车间编码
       ,REPLACE(SPLIT(A.PRODUCT_LINE_HANDLE,':')[2],SPLIT(A.WORK_SHOP_HANDLE,':')[1]||',','') AS PROD_LINE_CODE --产线编码
       ,SPLIT(A.PROCESS_HANDLE,',')[LENGTH(PROCESS_HANDLE) - LENGTH(REPLACE(PROCESS_HANDLE,',',''))] AS PROC_CODE --工序编码
       ,SPLIT(A.TEAM_HANDLE,',')[LENGTH(A.TEAM_HANDLE)-LENGTH(REPLACE(A.TEAM_HANDLE,',',''))] AS TEAM_CODE --班组编码
       ,SPLIT(A.ITEM_HANDLE,':')[1] AS MATERIAL_NUM --物料编码
       ,A.CONVEY_TIME AS CON_TIME --下发时间
       ,A.MONITOR_POINT_HANDLE AS MONIT_PT_ID --监控点主键
       ,A.MONITOR_PUSH_MECHANISM_HANDLE AS MONIT_PUSH_ID --监控点推送机制主键
       ,A.CORRECT_UPPER_LIMIT AS CORR_UP_LMT --向上纠正线
       ,A.CORRECT_LOWER_LIMIT AS CORR_LOW_LMT --向下纠正线
       ,A.WARNING_UPPER_LIMIT AS WARN_UP_LMT --向上预警线
       ,A.WARNING_LOWER_LIMIT AS WARN_LOW_LMT --向下预警线
       ,A.WRITE_QTY  --填报数量
       ,A.MONITOR_CONCLUSION AS MONIT_CNCLD_CODE --监控结论编码
       ,T1.NAME AS MONIT_CNCLD_NAME --监控结论描述
       ,A.ERROR_HAND_REMARK AS ERR_REMARK --异常处理备注
       ,A.ERROR_HAND_STATUS AS ERR_ST_CODE --异常处理状态编码
       ,T2.NAME AS ERR_ST_NAME --异常处理状态描述
       ,A.CORRECT_ORDER_NO AS CORR_NUM --纠正预防单号
       ,A.TACKLE_STATUS AS TACK_ST_CODE --预警确认情况编码：0待确认 1已确认
       ,CASE WHEN A.TACKLE_STATUS = 0 THEN '待确认'
             WHEN A.TACKLE_STATUS = 1 THEN '已确认'
              END AS TACK_ST_NAME --预警确认情况描述
       ,A.CORRECT_STATUS AS CORR_ST_CODE --纠正预防情况编码：0无 1待发起 2已发起
       ,CASE WHEN A.CORRECT_STATUS = 0 THEN '无'
             WHEN A.CORRECT_STATUS = 1 THEN '待发起'
             WHEN A.CORRECT_STATUS = 2 THEN '已发起'
              END AS CORR_ST_NAME --纠正预防情况描述
       ,A.PUSH_LEVEL AS PUSH_LVL_CODE --当前推送等级编码
       ,T3.NAME AS PUSH_LVL_NAME --当前推送等级描述
       ,A.PUSH_STATUS AS PUTH_ST_CODE --推送状态编码：0继续推送 1完成推送
       ,CASE WHEN A.PUSH_STATUS = 0 THEN '继续推送'
             WHEN A.PUSH_STATUS = 1 THEN '完成推送'
              END AS PUTH_ST_NAME --推送状态描述
       ,A.PUSH_TIME --推送时间
	   ,A.STATE AS ST_CODE --是否启用编码：0是 1否
       ,CASE WHEN A.STATE = 0 THEN '是'
             WHEN A.STATE = 1 THEN '否'
              END AS ST_NAME --是否启用描述
       ,A.CREATE_BY AS CREATE_EMP_NUM --创建人工号
       ,COALESCE(B.EMP_NAME,A.CREATE_BY) AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATE_DATE AS CREATE_TIME --创建时间
       ,A.UPDATE_BY AS UPDATE_EMP_NUM --修改人工号
       ,COALESCE(C.EMP_NAME,A.UPDATE_BY) AS UPDATE_EMP_NAME --修改人姓名
       ,A.UPDATE_DATE AS UPDATE_TIME --修改时间
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_M_QMS_MONITOR_ERROR_ORDER_1053 A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON A.CREATE_BY = B.EMP_NUM
  LEFT JOIN DM.DM_WF_EMP_INFO_T C
    ON A.UPDATE_BY = C.EMP_NUM
  LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D D
    ON A.SITE = D.SITE_CODE
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T1
    ON A.MONITOR_CONCLUSION = T1.CODE
   AND T1.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,monitorConclusion'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T2
    ON A.ERROR_HAND_STATUS = T2.CODE
   AND T2.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,monitorErrorDataStatus'
  LEFT JOIN SDI.SDI_S_REFERENCE_DATA_DETAILED_1053 T3
    ON A.PUSH_LEVEL = T3.CODE
   AND T3.REFERENCE_DATA_HANDLE = 'ReferenceDataBO:QMS,monitorPushLevel'
