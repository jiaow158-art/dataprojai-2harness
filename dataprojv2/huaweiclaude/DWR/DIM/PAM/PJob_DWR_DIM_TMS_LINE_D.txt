WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_TMS_LINE_D
SELECT A.ID AS LINE_ID --线路ID
      ,A.LINE_CODE --线路编码
	  ,A.LINE_NAME --线路名称
	  ,A.SHIPMETHOD AS SHIP_METH_CODE --运输方式编码
	  ,T1.DESCR AS SHIP_METH_NAME --运输方式描述
	  ,A.SKU_TYPE AS MATERIAL_TYPE_CODE --货品类型编码
	  ,T2.DESCR AS MATERIAL_TYPE_NAME --货品类型描述
	  ,A.ORDER_TYPE AS ORDER_TYPE_CODE --业务类型编码
	  ,CASE WHEN A.ORDER_TYPE = '1' THEN '工程'
            WHEN A.ORDER_TYPE = '100' THEN '其它'
			WHEN A.ORDER_TYPE = '11' THEN '洁具售后单'
			WHEN A.ORDER_TYPE = '21' THEN '共享仓干线'
			WHEN A.ORDER_TYPE = '31' THEN '城配'
	         END AS ORDER_TYPE_NAME --业务类型描述
	  ,A.ENTRYDT AS CREATE_TIME --创建时间
	  ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	  ,T3.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T3.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATEDT AS UPDATE_TIME --更新时间
	  ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	  ,T4.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T4.NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_LINE_1005 A 
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON A.SHIPMETHOD = T1.CODE
   AND T1.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T2
    ON A.SKU_TYPE = T2.CODE
   AND T2.TYPENAME = 'BMS_SKU_TYPE'
  LEFT JOIN T AS T3
    ON A.ENTRYID = T3.ID
  LEFT JOIN T AS T4
    ON A.UPDATEID = T4.ID
 WHERE A.TENANT_ID = 1
