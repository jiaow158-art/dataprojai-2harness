WITH T AS(
SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_TMS_SHIP_METH_D
SELECT A.ID AS ID --运输方式ID
	  ,A.LOADING_PROVINCE AS DELV_PROV_CODE --提货省编码
	  ,T1.STREET_NAME AS DELV_PROV_NAME --提货省名称
	  ,A.LOADING_CITY AS DELV_CITY_CODE --提货市编码
	  ,T2.STREET_NAME AS DELV_CITY_NAME --提货市名称
	  ,A.UNLOADING_PROVINCE AS RECV_PROV_CODE --收货省编码
	  ,T3.STREET_NAME AS RECV_PROV_NAME --收货省名称
	  ,A.UNLOADING_CITY AS RECV_CITY_CODE --收货市编码
	  ,T4.STREET_NAME AS RECV_CITY_NAME --收货市名称
	  ,A.SHIP_METHOD AS SHIP_METH_CODE --运输方式编码
	  ,T6.DESCR AS SHIP_METH_NAME --运输方式名称
	  ,A.BMS_SKU_TYPE AS DIV_CODE --事业部编码
	  ,T7.DESCR AS DIV_NAME --事业部描述
	  ,A.SHIP_DAYS AS SHIP_DAY --运输天数
	  ,A.PRIORITY AS PRIR_CODE --优先级编码
	  ,T5.NAME AS PRIR_NAME --优先级描述
	  ,A.ENTRYDT AS CREATE_TIME --创建时间
	  ,A.ENTRYID AS CREATE_EMP_ID --创建人ID
	  ,T8.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T8.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATEDT AS UPDATE_TIME --更新时间
	  ,A.UPDATEID AS UPDATE_EMP_ID --更新人ID
	  ,T9.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T9.NAME AS UPDATE_EMP_NAME --更新人姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_TMS_SITE_SHIP_1005 A
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T1
    ON A.LOADING_PROVINCE = T1.STREET_CODE
   AND T1.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T2
    ON A.LOADING_CITY = T2.STREET_CODE
   AND T2.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T3
    ON A.UNLOADING_PROVINCE = T3.STREET_CODE
   AND T3.STREET_LEVEL = '1'
  LEFT JOIN SDI.SDI_TMS_SITE_STREET_1005 T4
    ON A.UNLOADING_CITY = T4.STREET_CODE
   AND T4.STREET_LEVEL = '2'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T6
    ON A.SHIP_METHOD = T6.CODE
   AND T6.TYPENAME = 'BMS_CONTRACT_SHIP_METHOD'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T7
    ON A.BMS_SKU_TYPE = T7.CODE
   AND T7.TYPENAME = 'BMS_SKU_TYPE' 
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T5
    ON A.PRIORITY = T5.CODE
   AND T5.TYPENAME = 'TMS_PRIORITY'
  LEFT JOIN T AS T8
    ON A.ENTRYID = T8.ID
  LEFT JOIN T AS T9
    ON A.UPDATEID = T9.ID
 WHERE A.TENANT_ID = 1
