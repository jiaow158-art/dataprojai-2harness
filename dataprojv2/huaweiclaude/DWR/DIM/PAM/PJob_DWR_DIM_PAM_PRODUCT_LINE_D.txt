INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_PAM_PRODUCT_LINE_D
SELECT  
    T1.site AS SITE_CODE, -- 基地编码
    T2.site_desc AS SITE_NAME, -- 基地名称
    regexp_extract(T1.work_shop_bo, '[^,]+:?\\d+?,(.*)', 1) AS work_shop_code,      -- 车间编码    
    T3.work_shop_desc AS work_shop_NAME, -- 车间名称
    T1.product_line AS PRODUCT_LINE_CODE, -- 产线编码
    T1.product_line_desc AS PRODUCT_LINE_NAME, -- 产线名称
    'N' AS del_flag, -- 删除数据标识
	 current_timestamp() AS dw_last_update_date, -- 最后更新日期
	 CASE WHEN t1.site = '1300' THEN '清远'
	   	  WHEN t1.site = '1500' THEN '丰城'
	   	  WHEN t1.site = '1600' THEN '澧县'
	   	  WHEN t1.site = '1800' THEN '重庆'
	   	  WHEN t1.site = '1900' THEN '山西'
	  ELSE '' END SITE_SHORT_NAME, -- 基地简称
	  T3.work_shop_type, -- 车间类型
	  case when T1.site = '1300' then T1.product_line else regexp_extract(T1.work_shop_bo, '[^,]+:?\\d+?,(.*)', 1) end as PRODUCT_LINE_A_CODE, --新产线编码
	  case when T1.site = '1300' then T1.product_line_desc else T3.work_shop_desc end PRODUCT_LINE_A_NAME --新产线编码
FROM SDI.SDI_M_PRODUCT_LINE_1006 T1  -- 产线
LEFT JOIN SDI.SDI_M_SITE_1006 T2  -- 基地
ON T1.site = T2.site 
AND T2.enable_flag = '1'
LEFT JOIN SDI.SDI_M_WORK_SHOP_1006 T3 -- 车间
ON T1.site = T3.site 
AND regexp_extract(T1.work_shop_bo, '[^,]+:?\\d+?,(.*)', 1) = T3.work_shop 
AND T3.state = '1'
WHERE T1.state = '1';
