WITH WORK_CENTER AS (
SELECT 
T.SITE_CODE,
T.SITE_NAME,
T.WORK_SHOP_CODE,
T.PRODUCT_LINE_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
T.S_PRODUCT_LINE_CODE,
T.S_PRODUCT_LINE_NAME
FROM ( 
SELECT   
T.SITE_CODE,
T.SITE_NAME,
T.WORK_SHOP_CODE,
T.PRODUCT_LINE_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_CODE 
     ELSE '' END AS S_PRODUCT_LINE_CODE,  --服务产线编码
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_NAME 
     ELSE '' END AS S_PRODUCT_LINE_NAME  --服务产线名称
FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 T --基地产线成本表
WHERE M_PRODUCT_LINE_NAME NOT IN ('原料车间','辅助车间','分级车间','釉料车间')
-- LEFT JOIN SDI.SDI_RADIATION_WORKSHOP_T_1037 T2  --辅助车间辐射车间
-- ON T.SITE_CODE = T2.SITE_CODE 
-- AND T.COST_CENTER_CODE = T2.COST_CENTER_CODE
-- AND T.M_PRODUCT_LINE_CODE = T2.M_PRODUCT_LINE_CODE
-- AND T2.EFFECT_START_DATE <= current_date() AND T2.EFFECT_END_DATE >= current_date()
) T
GROUP BY  
T.SITE_CODE,
T.SITE_NAME,
T.WORK_SHOP_CODE,
T.PRODUCT_LINE_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.M_PRODUCT_LINE_CODE,
T.M_PRODUCT_LINE_NAME,
T.S_PRODUCT_LINE_CODE,
T.S_PRODUCT_LINE_NAME
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_PAM_PROD_VERSION_D
SELECT 
 A.site AS site_code 
,A.site_describe as site_name
,A.work_shop as work_shop_code
,A.work_shop_desc as work_shop_name
,A.product_line_a as product_line_code
,A.product_line_a_desc as product_line_name
,A.verid as verid_code
,A.ver_desc as verid_name
,A.creater
,A.updater
,A.create_time
,A.update_time
,current_timestamp() as dw_last_update_date
,B.M_PRODUCT_LINE_CODE
,B.M_PRODUCT_LINE_NAME
,B.S_PRODUCT_LINE_CODE
,B.S_PRODUCT_LINE_NAME
FROM SDI.SDI_DP_PRODUCTION_VERSION_MAINT_1006 A
LEFT JOIN WORK_CENTER B
ON A.site = B.SITE_CODE
AND A.work_shop = B.WORK_SHOP_CODE 
AND A.product_line_a = B.PRODUCT_LINE_CODE
where delete_flag = 1
