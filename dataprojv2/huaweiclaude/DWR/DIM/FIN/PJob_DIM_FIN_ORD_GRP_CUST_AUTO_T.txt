-- DWRDIM.DIM_FIN_ORD_GRP_CUST_AUTO_T
with tmp as (
select a.cust_num,a.sales_org_code,case when c.show_cust_code is not null then '02' else '' end as dis_chnl_code
,row_number() over(partition by a.cust_num,b.so_group order by a.sales_org_code ) as rn ,b.so_group,b.kunnr as gb_cust_num
from DWRDIM.DWR_DIM_CUST_SALES_AREA_D a
join (SELECT case when substr(sg_describe,1,1) = '/' then regexp_extract(sg_describe, '/(\\d{10})', 1) 
else regexp_extract(sg_describe, '^\\d{10}', 0) end as cust_num ,so_group,kunnr
FROM SDI.SDI_ZGATHERING_SG_1001
WHERE KUNNR IN (SELECT GB_CUST_CODE FROM SDI.SDI_XSSR_GB_CUST_T_1037 WHERE FLAG = 'SW') ) b
on a.cust_num = b.cust_num
LEFT JOIN SDI.SDI_GB_SHOWROOM_CUST_T_1037 c
ON b.cust_num = c.show_cust_code 
where a.end_date = '4712-12-31' and a.del_flag = 'N' and (a.sales_org_code like '3%' or a.sales_org_code = '1030')
and a.cust_order_frozen_code = ''  and a.distribution_chnl_code <> '06'
union all 
select a.cust_num,a.sales_org_code,case when c.show_cust_code is not null then '02' else '' end as dis_chnl_code
,row_number() over(partition by a.cust_num,b.so_group order by a.sales_org_code ) as rn ,b.so_group,b.kunnr as gb_cust_num
from DWRDIM.DWR_DIM_CUST_SALES_AREA_D a
join (SELECT case when substr(sg_describe,1,1) = '/' then regexp_extract(sg_describe, '/(\\d{10})', 1) 
else regexp_extract(sg_describe, '^\\d{10}', 0) end as cust_num ,so_group,kunnr
FROM SDI.SDI_ZGATHERING_SG_1001
WHERE KUNNR IN (SELECT GB_CUST_CODE FROM SDI.SDI_XSSR_GB_CUST_T_1037 WHERE FLAG = 'CT') ) b
on a.cust_num = b.cust_num
LEFT JOIN SDI.SDI_GB_SHOWROOM_CUST_T_1037 c
ON b.cust_num = c.show_cust_code 
LEFT JOIN (
    SELECT cust_num, COUNT(DISTINCT sales_org_code) AS org_count
    FROM DWRDIM.DWR_DIM_CUST_SALES_AREA_D
    where end_date = '4712-12-31' and del_flag = 'N'
    and (sales_org_code not like '2%' and sales_org_code not like '3%' and sales_org_code not in ('7100','7300','7810','7910','7820','1310','7M20','7930','7940'))
    and cust_order_frozen_code = ''
    GROUP BY cust_num
) e ON a.cust_num = e.cust_num  --存在有多个销售组织时排除7620组织
where a.end_date = '4712-12-31' and a.del_flag = 'N' and (a.sales_org_code not like '2%' and a.sales_org_code not like '3%' and a.sales_org_code not in ('7100','7300','7810','7910','7820','1310','7M20','7930','7940'))
and a.cust_order_frozen_code = ''  and case when e.org_count > 1 then a.sales_org_code <> '7620' else 1=1 end 
and a.distribution_chnl_code <> '06'
)
insert overwrite table DWRDIM.DIM_FIN_ORD_GRP_CUST_AUTO_T 
select 
 gb_cust_num
,cust_num
,sales_org_code
,dis_chnl_code
,so_group
,current_timestamp as dw_last_update_date
from tmp 
where rn = 1
