--层次结构表
INSERT
  OVERWRITE TABLE DWRDIM.DWR_DIM_LEVEL_STRUCTURE_D
SELECT
  T1.NODE_ID AS LV1_NODE_ID,
  --第一层级组织ID
  T1.NODE_NAME AS LV1_NODE_NAME,
  --第一层级组织名称
  T2.NODE_ID AS LV2_NODE_ID,
  --第二层级组织ID
  T2.NODE_NAME AS LV2_NODE_NAME,
  --第二层级组织名称
  T3.NODE_ID AS LV3_NODE_ID,
  --第三层级组织ID
  T3.NODE_NAME AS LV3_NODE_NAME,
  --第三层级组织名称
  T4.NODE_ID AS LV4_NODE_ID,
  --第四层级组织ID
  T4.NODE_NAME AS LV4_NODE_NAME,
  --第四层级组织名称
  T5.NODE_ID AS LV5_NODE_ID,
  --第五层级组织ID
  T5.NODE_NAME AS LV5_NODE_NAME,
  --第五层级组织名称
  T6.NODE_ID AS LV6_NODE_ID,
  --第六层级组织ID
  T6.NODE_NAME AS LV6_NODE_NAME,
  --第六层级组织名称
  T7.NODE_ID AS LV7_NODE_ID,
  --第七层级组织ID
  T7.NODE_NAME AS LV7_NODE_NAME,
  --第七层级组织名称
  T8.NODE_ID AS LV8_NODE_ID,
  --第八层级组织ID
  T8.NODE_NAME AS LV8_NODE_NAME,
  --第八层级组织名称
  T9.NODE_ID AS LV9_NODE_ID,
  --第九层级组织ID
  T9.NODE_NAME AS LV9_NODE_NAME,
  --第九层级组织名称
  T10.NODE_ID AS LV10_NODE_ID,
  --第十层级组织ID
  T10.NODE_NAME AS LV10_NODE_NAME,
  --第十层级组织名称
  COALESCE(
    T10.NODE_ID,
    T9.NODE_ID,
    T8.NODE_ID,
    T7.NODE_ID,
    T6.NODE_ID,
    T5.NODE_ID,
    T4.NODE_ID,
    T3.NODE_ID,
    T2.NODE_ID,
    T1.NODE_ID
  ) AS NODE_ID,
  --组织ID
  COALESCE(
    T10.NODE_NAME,
    T9.NODE_NAME,
    T8.NODE_NAME,
    T7.NODE_NAME,
    T6.NODE_NAME,
    T5.NODE_NAME,
    T4.NODE_NAME,
    T3.NODE_NAME,
    T2.NODE_NAME,
    T1.NODE_NAME
  ) AS NODE_NAME,
  --组织名称
  'N' AS DEL_FLAG,
  --删除符号
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE --DW最后更新日期
FROM
  DWIMD.DWI_MD_LEVEL_STRUCTURE_T T1
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T2 ON T1.INNER_LEVEL_ID = T2.INNER_LEVEL_ID
  AND T2.PARENT_ID = T1.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T3 ON T2.INNER_LEVEL_ID = T3.INNER_LEVEL_ID
  AND T3.PARENT_ID = T2.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T4 ON T3.INNER_LEVEL_ID = T4.INNER_LEVEL_ID
  AND T4.PARENT_ID = T3.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T5 ON T4.INNER_LEVEL_ID = T5.INNER_LEVEL_ID
  AND T5.PARENT_ID = T4.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T6 ON T5.INNER_LEVEL_ID = T6.INNER_LEVEL_ID
  AND T6.PARENT_ID = T5.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T7 ON T6.INNER_LEVEL_ID = T7.INNER_LEVEL_ID
  AND T7.PARENT_ID = T6.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T8 ON T7.INNER_LEVEL_ID = T8.INNER_LEVEL_ID
  AND T8.PARENT_ID = T7.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T9 ON T8.INNER_LEVEL_ID = T9.INNER_LEVEL_ID
  AND T9.PARENT_ID = T8.NODE_ID
  LEFT JOIN DWIMD.DWI_MD_LEVEL_STRUCTURE_T T10 ON T9.INNER_LEVEL_ID = T10.INNER_LEVEL_ID
  AND T10.PARENT_ID = T9.NODE_ID
WHERE
  1 = 1
  AND UPPER(T1.INNER_LEVEL_ID) = 'N6X45W6B2GPOKGEL94IFLA949'
  AND T1.NODE_LEVEL = '01'
  AND UPPER(T1.SOURCE_TYPE) = 'HSALES_GRP'
GROUP BY
  T1.NODE_ID,
  T1.NODE_NAME,
  T2.NODE_ID,
  T2.NODE_NAME,
  T3.NODE_ID,
  T3.NODE_NAME,
  T4.NODE_ID,
  T4.NODE_NAME,
  T5.NODE_ID,
  T5.NODE_NAME,
  T6.NODE_ID,
  T6.NODE_NAME,
  T7.NODE_ID,
  T7.NODE_NAME,
  T8.NODE_ID,
  T8.NODE_NAME,
  T9.NODE_ID,
  T9.NODE_NAME,
  T10.NODE_ID,
  T10.NODE_NAME,
  COALESCE(
    T10.NODE_ID,
    T9.NODE_ID,
    T8.NODE_ID,
    T7.NODE_ID,
    T6.NODE_ID,
    T5.NODE_ID,
    T4.NODE_ID,
    T3.NODE_ID,
    T2.NODE_ID,
    T1.NODE_ID
  ),
  COALESCE(
    T10.NODE_NAME,
    T9.NODE_NAME,
    T8.NODE_NAME,
    T7.NODE_NAME,
    T6.NODE_NAME,
    T5.NODE_NAME,
    T4.NODE_NAME,
    T3.NODE_NAME,
    T2.NODE_NAME,
    T1.NODE_NAME
  );
