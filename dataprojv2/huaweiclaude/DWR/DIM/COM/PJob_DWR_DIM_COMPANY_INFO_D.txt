--DWI_MD_COMPANY_INFO_T 到 DWR_DIM_COMPANY_INFO_D
--OSS公司信息表
INSERT
  OVERWRITE TABLE DWRDIM.DWR_DIM_COMPANY_INFO_D
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.DB_LAST_UPD_SRC AS DB_LAST_UPD_SRC,
  DWI.COMPANY_CODE_NAME AS COMPANY_CODE_NAME,
  DWI.COMPANY_CODE AS COMPANY_CODE,
  DWI.DB_LAST_UPD_DATE AS DB_LAST_UPD_DATE,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.MODIFICATION_NUM AS MODIFICATION_NUM,
  DWI.LAST_UPD_BY AS LAST_UPD_BY,
  DWI.LAST_UPD_DATE AS LAST_UPD_DATE,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.CREATED_DATE AS CREATED_DATE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIMD.DWI_MD_COMPANY_INFO_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
