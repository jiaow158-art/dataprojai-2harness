--物料品牌主数据
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MATERIAL_BRAND_D
SELECT A.FACTORY_WERKS_CODE
,D.FACTORY_NAME AS FACTORY_WERKS_NAME
,A.MATL_GROUP_CODE
,C.WGBEZ AS MATL_GROUP_NAME
,A.EXTERNAL_MATL_GROUP_CODE
,B.EWBEZ AS EXTERNAL_MATL_GROUP_NAME
,A.BRAND_NAME
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIMD.DWI_MD_MATERIAL_BRAND_T A
LEFT JOIN SDI.SDI_TWEWT_1001 B
ON A.EXTERNAL_MATL_GROUP_CODE = B.EXTWG
AND B.SPRAS = '1'
AND B.MANDT = '800'
LEFT JOIN SDI.SDI_T023T_1001 C
ON A.MATL_GROUP_CODE = C.MATKL
AND C.SPRAS = '1'
AND C.MANDT = '800'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T D
ON A.FACTORY_WERKS_CODE = D.FACTORY_CODE
AND D.END_DATE = '4712-12-31'     
;
