WITH FAC AS (
SELECT material_num,factory_code,row_number() over(partition by material_num order by factory_code) as rn 
FROM DWIMD.DWI_MD_DATA_MATERIAL_FACTORY_T A
where SUBSTR(end_date,1,10) = '4712-12-31' and del_flag = 'N' 
)

insert overwrite table DWRDIM.DWR_DIM_MAT_QUALITY_DETAIL_T_01
SELECT 
    g.material_num, -- 物料编号
    g.material_name, -- 物料名称
    g.matl_type_code, --物料类型编码
    g.matl_type_name, --物料类型编码
    g.product_listed_date, -- 产品上线日期
    g.prod_property, -- 产品所有权
    g.prod_property_name, --产品所有权名称
    g.dimension, --大小量纲 
    g.length_1,--长
    g.width, --宽
    g.gross_weight, -- 毛重
    g.net_weight, -- 净重 
    g.busi_volume, --业务量
    g.product_level_code, -- 产品层次代码(基础)
    g.matl_group_code, --物料组代码
    g.matl_group_name,  --物料组名称
    g.product_group_code, --产品组
    g.product_group_name, --产品组名称
    g.x_plant_matl_status,--跨工厂物料状态
    g.x_dstrb_c_mm_status_code,--跨分销物料状态
    g.batch_mngnt_require_flag, --批次管理标识
    g.general_item_ctgry_group_code, --普通项目类别组
    g.general_item_ctgry_group_name,--普通项目类别组名称
    g.category,-- 品类 
    g.category_name, -- 品类名称
    g.reserved_field_1,--产品渠道
    g.reserved_field1_name,--产品渠道名称
    g.product_place_code,--产品工艺编码
    g.product_place_name,--产品工艺名称
    g.product_brand_code,--系列编码
    g.product_brand_name,--系列名称
    g.product_position_code, --高值说明编码
    g.product_position_name, --高值说明名称
    g.reserved_field5_name, --新品说明
    g.reserved_field_4,  --清库存标识
    g.reserved_field4_name, --清库存标识名称
    g.matl_texture,  --物料材质
    g.matl_texture_name,   --物料材质名称
    g.product_series_name, --产品规格
    -- 归类标识：根据产品属性和物料类型对产品进行分类
    CASE 
        WHEN g.prod_property IN ('A0001', 'A0002', 'C0001', 'C0002', 'C0003', 'C0004', 'C0006') THEN '瓷砖(产成品)'
        WHEN g.prod_property = 'B0001' THEN '卫浴(产成品)'
        WHEN g.prod_property = 'D0001' THEN '生态新材(产成品)'
        WHEN g.prod_property = 'E0001' THEN '国贸(产成品)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND FAC.factory_code IN ('1200','1300','1350','1500','1550','1600','1650','1700','1750','1800','1850','1900','1950','7220','7320','7920','7A20','7D00','7D50','8120','8820') THEN '瓷砖(原材料)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND FAC.factory_code IN ('3200','3320','3500','3602','3603','3606','3620','3700','3820','3920','3A20') THEN '卫浴(原材料)'
        ELSE '其他' 
    END AS flag,    
    -- 产品上线时间总行数：物料类型为 FERT 且上线日期不为空的记录数
    CASE WHEN g.matl_type_code = 'FERT' AND g.product_listed_date IS NOT NULL and g.product_listed_date <> ''  AND g.x_dstrb_c_mm_status_code <> 'GG'  THEN 1 ELSE 0 END AS pld_sum_count,
    -- 产品上线时间错误数：物料类型为 FERT、上线日期不为空且格式错误的记录数
    CASE WHEN g.matl_type_code = 'FERT' AND g.product_listed_date IS NOT NULL and g.product_listed_date <> ''  AND g.x_dstrb_c_mm_status_code <> 'GG'
    AND (g.product_listed_date not rlike '^\\d{4}-(0[1-9]|1[0-2])$') THEN 1 ELSE 0 END AS pld_err_count,
    -- 重量总行数：毛重或净重不为 0 的记录数
    CASE WHEN g.matl_type_code = 'FERT' and (g.gross_weight <> '0' and g.net_weight <> '0') AND g.x_dstrb_c_mm_status_code <> 'GG' THEN 1 ELSE 0 END AS weight_sum_count,
    -- 重量错误总行数：毛重和净重均不为 0 且毛重小于净重的记录数
    CASE WHEN g.matl_type_code = 'FERT' and (g.gross_weight <> '0' and g.net_weight <> '0') AND g.x_dstrb_c_mm_status_code <> 'GG' 
    AND CAST(g.gross_weight AS FLOAT) < CAST(g.net_weight AS FLOAT) THEN 1 ELSE 0 END AS weight_error_count,
    -- 产品所有权总行数：物料类型为 FERT 或 HALB 或者产品层次代码前四位为 9006 的记录数
    CASE WHEN (g.matl_type_code IN ('FERT', 'HALB') or (g.matl_type_code='ROH' and SUBSTR(g.product_level_code, 1, 4) = '9006')) AND g.x_dstrb_c_mm_status_code <> 'GG' THEN 1 ELSE 0 END AS pp_sum_count,
    -- 产品所有权错误行数：物料类型为 FERT 或 HALB 或者产品层次代码前四位为 9006 且产品属性为空的记录数
    CASE WHEN (g.matl_type_code IN ('FERT', 'HALB') OR (g.matl_type_code='ROH' and SUBSTR(g.product_level_code, 1, 4) = '9006')) AND g.x_dstrb_c_mm_status_code <> 'GG'
        AND g.prod_property = '' THEN 1 ELSE 0 END AS pp_error_count,
    --产品组与物料组的关系校验
    CASE WHEN g.x_dstrb_c_mm_status_code <> 'GG'  AND g.product_group_code <> '00' THEN 1 ELSE 0 END AS group_sum_count,
    case when g.product_group_code <> f.product_group_code AND g.x_dstrb_c_mm_status_code <> 'GG' AND g.product_group_code <> '00'
    	 then 1 else 0 end as group_error_count,  
    1 AS status_sum_count,
    -- 状态检查错误行数：工厂物料状态和分销渠道物料状态不一致的记录数
    CASE WHEN NVL(CASE WHEN g.x_plant_matl_status = '99' THEN 'GG' ELSE g.x_plant_matl_status END, '#') <> NVL(g.x_dstrb_c_mm_status_code, '#')
        THEN 1 ELSE 0 END AS status_error_count,
    -- 批次管理总行数：符合特定产品属性和物料类型要求的记录数
    CASE WHEN (g.prod_property IN ('A0001', 'A0002', 'C0001', 'C0002', 'C0003', 'D0001', 'C0006', 'E0001') 
              OR (g.prod_property = 'B0001' AND g.general_item_ctgry_group_code <> 'LUMF')) 
             AND g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' THEN 1 ELSE 0 END AS batch_sum_count,
    -- 批次管理错误行数：符合特定产品属性和物料类型要求且批次管理标志为空的记录数
    CASE WHEN (g.prod_property IN ('A0001', 'A0002', 'C0001', 'C0002', 'C0003', 'D0001', 'C0006', 'E0001')  
              OR (g.prod_property = 'B0001' AND g.general_item_ctgry_group_code <> 'LUMF')) AND g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG'
             AND (g.batch_mngnt_require_flag = '' OR g.batch_mngnt_require_flag IS NULL) THEN 1 ELSE 0 END AS batch_error_count,
    --品类不能为空校验
    case when g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end as category_sum_count,
    case when g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' and (g.category is null or g.category = '') then 1 else 0 end as category_error_count, 
   --产品渠道不能为空
    case when g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' and g.prod_property in ('A0001','A0002','C0001','C0002','B0001','E0001') then 1 else 0 end as chanel_sum_count,
    case when g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' and g.prod_property in ('A0001','A0002','C0001','C0002','B0001','E0001') 
    and (g.reserved_field_1 is null or g.reserved_field_1 = '') then 1 else 0 end as chanel_sum_count,
    --产品工艺不能为空
    case when g.prod_property in ('A0001','C0001','C0002','A0002') and g.category like 'PL%' and g.matl_type_code = 'FERT'
    	 and g.category <> 'PL015' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end as tech_sum_count,
    case when g.prod_property in ('A0001','C0001','C0002','A0002') and g.category like 'PL%' and g.matl_type_code = 'FERT'
    	 and g.category <> 'PL015' AND g.x_dstrb_c_mm_status_code <> 'GG' and g.product_place_code = '' then 1 else 0 end as tech_error_count,
   --高值说明异常
    case when g.matl_type_code = 'FERT' and g.material_num not like '%_B' and g.prod_property in ('A0001','A0002','C0001','C0002','E0001') 
    	 and category <> 'PL015' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end as pos_sum_count,	 
    case when g.matl_type_code = 'FERT' and (g.product_series_name in ('918','超大板') or g.product_brand_code in ('A1','A13','A14','A16','A18') )
         and g.material_num not like '%_B' and g.prod_property in ('A0001','A0002','C0001','C0002','E0001') and g.category <> 'PL015' 	
    	 and (g.product_position_code <> 'A100' or g.product_position_code is null or g.product_position_code = '') AND g.x_dstrb_c_mm_status_code <> 'GG'
    	 then 1 else 0 end as pos_error_count,
   --大小量纲校验
    case when g.matl_type_code = 'FERT' and g.dimension is not null and g.dimension <> '' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end as dim_sum_count,
    case when g.matl_type_code = 'FERT' and g.dimension is not null and g.dimension <> '' AND g.x_dstrb_c_mm_status_code <> 'GG'
    and (instr(g.dimension,'*') > 0 or instr(g.dimension,'x') > 0 or (instr(g.dimension,'X') > 0 
    and cast(split(g.dimension,'X')[0] as float ) < cast(split(g.dimension,'X')[1] as float ))) then 1 else 0 end as dim_error_count,
   --品类异常
    case when g.prod_property in ('A0001','C0001','C0002','A0002') and g.category like 'PL%' and g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end cate_sum_count,
    case when g.prod_property in ('A0001','C0001','C0002','A0002') and g.category like 'PL%' and g.matl_type_code = 'FERT' AND g.x_dstrb_c_mm_status_code <> 'GG'
    	 and g.category not in ('PL001','PL002','PL006','PL008','PL010','PL011','PL015') then 1 else 0 end cate_error_count,
   --大小量纲等于长宽
    case when g.matl_type_code = 'FERT' and g.dimension is not null and g.dimension <> '' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end size_sum_count,
    case when g.matl_type_code = 'FERT' and g.dimension is not null and g.dimension <> '' AND g.x_dstrb_c_mm_status_code <> 'GG' and instr(g.dimension,'X') > 0 
	     and (split(g.dimension,'X')[0] <> g.length_1 or split(g.dimension,'X')[1] <> g.width) then 1 else 0 end size_error_count,
   --新品说明与产品上市时间的关系校验
	case when g.matl_type_code = 'FERT' and g.prod_property in ('A0001','C0001','C0002','D0001','E0001') AND g.x_dstrb_c_mm_status_code <> 'GG'
	     and g.product_listed_date is not null and g.product_listed_date <> '' then 1 else 0 end as new_sum_count,
	case when g.matl_type_code = 'FERT' and g.prod_property in ('A0001','C0001','C0002','D0001','E0001') AND g.x_dstrb_c_mm_status_code <> 'GG'
	     and g.product_listed_date is not null and g.product_listed_date <> ''
     	 and ((g.product_listed_date <= substring(cast(add_months(date_add(current_date,-1),-12) as string),1,7) and g.reserved_field5_name = '新品')
	     or (g.product_listed_date > substring(cast(add_months(date_add(current_date,-1),-12) as string),1,7) and g.reserved_field5_name = '非新品' )) 
	     then 1 else 0 end as new_error_count ,
   --清库存标识与库存消耗类型的关系校验
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' AND g.x_dstrb_c_mm_status_code <> 'GG' then 1 else 0 end as clear_sum_count,
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' AND g.x_dstrb_c_mm_status_code <> 'GG'
    	 and case when g.reserved_field_4 = '03' then 'A001' end <> g.matl_texture then 1 else 0 end as clear_sum_count,
    CURRENT_TIMESTAMP()	AS DW_LAST_UPDATE_DATE			-- 数据同步时间
FROM dwimd.dwi_md_data_material_general_t g
LEFT JOIN FAC 
ON g.material_num = fac.material_num and fac.rn = 1
-- 关联评估类与产品层次参考表，根据参考表 PRDHA 长度匹配产品层次代码
LEFT JOIN sdi.sdi_zmmt016_1001 e ON 
    CASE 
        WHEN LENGTH(e.PRDHA) = 4 THEN g.product_level2_code
        WHEN LENGTH(e.PRDHA) = 7 THEN g.product_level3_code end = e.PRDHA
left join sdi.sdi_pro_mat_relat_t_1037 f 
on g.matl_group_code = f.material_group_code
