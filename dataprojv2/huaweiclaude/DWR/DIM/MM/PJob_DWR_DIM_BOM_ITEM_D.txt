INSERT
  OVERWRITE TABLE DWRDIM.DWR_DIM_BOM_ITEM_D
SELECT
  BOM.MATL_TYPE AS MATL_TYPE,
  BOM.MATL_LIST AS MATL_LIST,
  BOM.INNER_COUNTER AS INNER_COUNTER,
  BOM.GROUP_NUM AS GROUP_NUM,
  BOM.BOM_PROJ_NODE_NUM AS BOM_PROJ_NODE_NUM,
  BOM.LAST_CHG_DATE AS LAST_CHG_DATE,
  BOM.COMPONENT_NUM AS COMPONENT_NUM,
  BOM.COMPONENT_MEASURE_UNIT_CODE AS COMPONENT_MEASURE_UNIT_CODE,
  BOM.COMPONENT_SCRAP_PCT AS COMPONENT_SCRAP_PCT,
  BOM.VALID_DATE_BEGIN AS VALID_DATE_BEGIN,
 T.CONFIG_NAME AS PROJ_TYPE_CODE,
  BOM.PROJ_CHG_STATUS_G_FLAG AS PROJ_CHG_STATUS_G_FLAG,
  BOM.PROD_ORDER_SHIP_LOC_CODE AS PROD_ORDER_SHIP_LOC_CODE,
  BOM.REC_CRT_DATE AS REC_CRT_DATE,
  BOM.PRE_ITEM_COUNTER AS PRE_ITEM_COUNTER,
  BOM.NET_WASTE_FLAG AS NET_WASTE_FLAG,
  BOM.FIX_NUM AS FIX_NUM,
  BOM.CHG_OBJECT_NAME AS CHG_OBJECT_NAME,
  BOM.PARENT_NODE AS PARENT_NODE,
  BOM.BOM_PROJ_OBJECT_TYPE AS BOM_PROJ_OBJECT_TYPE,
  BOM.CRT_REC_USER AS CRT_REC_USER,
  BOM.COST_RELATED_FLAG AS COST_RELATED_FLAG,
  BOM.SALES_RELATED_FLAG AS SALES_RELATED_FLAG,
  BOM.PRODUCT_RELATED_FLAG AS PRODUCT_RELATED_FLAG,
  BOM.PROJECT_RELATED_FLAG AS PROJECT_RELATED_FLAG,
  BOM.BOM_COMPONENT AS BOM_COMPONENT,
  BOM.BOM_PROJ_TEXT AS BOM_PROJ_TEXT,
  BOM.BOM_PROJ_NUM AS BOM_PROJ_NUM,
  BOM.BOM_PROJ_INHERIT_NODE_NUM AS BOM_PROJ_INHERIT_NODE_NUM,
  BOM.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  BOM.START_DATE AS START_DATE,
  BOM.END_DATE AS END_DATE
FROM
  DWIMD.DWI_MD_BOM_ITEM_T BOM
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T 
  ON BOM.PROJ_TYPE_CODE = T.CONFIG_CODE
  AND BOM.GROUP_NUM = T.GROUP_NUM
  AND T.LANG_CODE = '1'
