WITH FAC AS (
SELECT material_num,factory_code,row_number() over(partition by material_num order by factory_code) as rn 
FROM DWIMD.DWI_MD_DATA_MATERIAL_FACTORY_T A
where SUBSTR(end_date,1,10) = '4712-12-31' and del_flag = 'N' 
)
insert overwrite table DWRDIM.DWR_DIM_MAT_QUALITY_DETAIL_T_02
SELECT 
    g.material_num, -- 物料编号
    g.material_name, -- 物料名称
    s.sales_org_code, -- 销售组织代码
    s.distribution_channel_code, -- 分销渠道代码
    g.matl_type_code, --物料类型编码
    g.matl_type_name, --物料类型编码
    g.gross_weight, --毛重,
    g.net_weight, --净重,
    g.busi_volume, --业务量,
    g.prod_property, -- 产品所有权  
    g.prod_property_name, --产品所有权名称
    s.matl_subj_set_group_code, -- 物料科目设置组代码
    g.product_level_code, -- 产品层次代码(基础)
    s.product_level_code,-- 产品层次代码(销售)
    g.matl_group_code, --物料组
    g.matl_group_name, --物料组名称
    g.general_item_ctgry_group_code, --普通项目类别组
    g.general_item_ctgry_group_name,--普通项目类别组名称
    s.matl_m_file_item_c_group_code,--项目类别编码(销售)
    s.matl_m_file_item_c_group_name,--项目类别名称(销售)
    s.matl_price_group_code,--物料价格组
    s.matl_price_group_name,--物料价格组名称
    s.matl_group1_code,--产品主次(物料组1)
    s.matl_group1_name,--产品主次(物料组1)名称
    s.matl_group2_code,--在市退市状态(物料组2)
    s.matl_group2_name,--在市退市状态(物料组2)名称
    s.matl_group3_code,--配件标识(物料组3)
    s.matl_group3_name,--配件标识(物料组3)名称
    s.matl_group4_code,--排产计划分类(物料组4)
    s.matl_group4_name,--排产计划分类(物料组4)名称
    -- 归类标识：根据产品属性和物料类型对产品进行分类
    CASE 
        WHEN g.prod_property IN ('A0001', 'A0002', 'C0001', 'C0002', 'C0003', 'C0004', 'C0006') THEN '瓷砖(产成品)'
        WHEN g.prod_property = 'B0001' THEN '卫浴(产成品)'
        WHEN g.prod_property = 'D0001' THEN '生态新材(产成品)'
        WHEN g.prod_property = 'E0001' THEN '国贸(产成品)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND FAC.factory_code IN ('1200','1300','1350','1500','1550','1600','1650','1700','1750','1800','1850','1900','1950','7220','7320','7920','7A20','7D00','7D50','8120','8820') THEN '瓷砖(原材料)'
        WHEN g.prod_property = '' AND g.matl_type_code IN ('ROH','VERP') AND FAC.factory_code IN ('3200','3320','3500','3602','3603','3606','3620','3700','3820','3920','3A20') THEN '卫浴(原材料)'
        ELSE '其他' 
    END AS flag,    
    --基础视图与销售视图的产品层次一致性校验
    1 as prodh_sum_count,
    case when g.product_level_code <> s.product_level_code then 1 else 0 end as prodh_error_count,    
    --基础视图与销售视图的项目类别组一致性校验
    1 as item_sum_count,
    case when g.general_item_ctgry_group_code <> s.matl_m_file_item_c_group_code then 1 else 0 end as item_error_count,
    --科目设置组在各个销售视图的一致性
    1 as set_sum_count,
    case when count(distinct s.matl_subj_set_group_code) over(partition by s.material_num) > 1 THEN 1 ELSE 0 END AS set_error_count,
    --物料价格组在各个销售视图的一致性
    1 as konda_sum_count,
    case when count(distinct s.matl_price_group_code) over(partition by s.material_num) > 1 THEN 1 ELSE 0 END AS konda_error_count,
   --产品主次（物料组1）在各个销售视图的一致性
    1 as mat1_sum_count,
    case when count(distinct s.matl_group1_code) over(partition by s.material_num) > 1 THEN 1 ELSE 0 END AS mat1_error_count,
   --卫浴重量体积校验
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group2_code= 'J9'  --在市
    	 and s.matl_group4_code in ('DZ','DJ','ZS','TX') then 1 else 0 end wv_sum_count,
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group2_code= 'J9' 
    	 and s.matl_group4_code in ('DZ','DJ','ZS','TX') and (g.gross_weight = '0' or g.busi_volume = '0' ) then 1 else 0 end wv_error_count,
   --在市退市状态（物料组2）与排产计划分类（物料组4）的关系校验	 
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group2_code = 'J10' then 1 else 0 end ts_sum_count, --退市
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group2_code = 'J10' 
    	 and s.matl_group4_code <> 'TS' then 1 else 0 end ts_error_count, --退市
   --配件标识（物料组3）与市退市状态（物料组2）的关系校验
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group3_code = 'PJ' then 1 else 0 end pj_sum_count,
    case when g.matl_type_code = 'FERT' and g.prod_property = 'B0001' and s.matl_group3_code = 'PJ' 
         and (s.matl_group2_code is not null and s.matl_group2_code <> '') then 1 else 0 end pj_error_count,
   --配件标识（物料组3）在各个销售视图的一致性
    1 as mat3_sum_count,
    case when count(distinct s.matl_group3_code) over(partition by s.material_num) > 1 THEN 1 ELSE 0 END AS mat3_error_count,
    1 as mat4_sum_count,
    case when count(distinct s.matl_group4_code) over(partition by s.material_num) > 1 THEN 1 ELSE 0 END AS mat4_error_count,
    CURRENT_TIMESTAMP()	AS DW_LAST_UPDATE_DATE			-- 数据同步时间
FROM (
    SELECT 
        material_num,
        sales_org_code,
        distribution_channel_code,
        matl_subj_set_group_code,
        matl_m_file_item_c_group_code,
        matl_m_file_item_c_group_name,
        matl_price_group_code,
        matl_price_group_name,
        product_level_code,
        matl_group2_code,
        matl_group1_code,
        matl_group3_code,
        matl_group4_code,
        matl_group2_name,
        matl_group1_name,
        matl_group3_name,
        matl_group4_name
    FROM 
        dwimd.dwi_md_data_material_sales_t
    WHERE 
        end_date = '4712-12-31' AND del_flag = 'N'
) s -- 关联销售数据，条件为物料编号相同且销售数据有效
LEFT JOIN  dwimd.dwi_md_data_material_general_t g  
ON g.material_num = s.material_num  
LEFT JOIN FAC 
ON g.material_num = fac.material_num and fac.rn = 1
where g.x_dstrb_c_mm_status_code <> 'GG'
