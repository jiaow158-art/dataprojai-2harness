INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_DEALER_TGT_D PARTITION(STAT_YEAR)
SELECT A.ID
      ,A.STAT_MONTH --月份
	  ,A.DIV_CODE --所属事业部编码
	  ,A.DIV_NAME --所属事业部描述
	  ,A.CUST_NUM --客编  
	  ,A.CHNL_NAME --渠道名称
	  ,A.TGT_TYPE_NAME --任务目标类型
	  ,A.TGT_AMT --任务目标
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间 
	  ,B.IS_USE_MTH_TGT_CODE --是否与母客编共同使用销售任务编码
	  ,B.IS_USE_MTH_TGT_NAME --是否与母客编共同使用销售任务描述
	  ,A.STAT_YEAR --年份
  FROM (SELECT *
             ,ROW_NUMBER() OVER(PARTITION BY CUST_NUM,SIGN_CHNL_CODE,STAT_MONTH ORDER BY VAILD_DATE DESC) AS RN
         FROM T1) AS A
  LEFT JOIN DWRDIM.DWR_DIM_DEALER_CNTC_D B
    ON A.ID = B.ID
   AND B.DATA_SRC_NAME = '台账'
 WHERE A.RN = 1
   AND A.CUST_NUM IS NOT NULL
