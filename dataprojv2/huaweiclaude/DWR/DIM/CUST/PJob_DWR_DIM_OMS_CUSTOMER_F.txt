INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_OMS_CUSTOMER_F
SELECT 
       DI.ORGANIZATION_TYPE AS ORG_TYPE --经营主体类型
	  ,A.MEANING AS ORG_TYPE_NAME --经营主体类型名称
      ,DI.ORG_CODE --经营主体编码
      ,DI.ORG_NAME --经营主体名称
      ,DI.DISTRIBUTOR_CODE AS CUSTOMER_CODE --经销商编码
      ,DI.DISTRIBUTOR_NAME AS CUSTOMER_NAME --经销商名称
      ,DI.CONTACTS AS CONTACT_NAME --联系人
      ,DI.CONTACTS_PHONE AS CONTACT_PHONE --联系人电话
      ,DI.SERVICES_PHONE AS SERVICE_PHONE --客服电话
      ,CASE WHEN DI.ENABLED_FLAG = 'Y' THEN '是'
	        WHEN DI.ENABLED_FLAG = 'N' THEN '否'
			 END AS IS_ENABLE --是否生效
      ,DI.PROVINCE AS PROVINCE_NAME --省
      ,DI.CITY AS CITY_NAME --市
      ,DI.COUNTY AS COUNTY_NAME --县
      ,DI.ADDRESS --详细地址
      ,DI.DESCRIBED AS DESCRIB --描述
      ,CASE WHEN DI.NEGATIVE_INVENTORY = 'Y' THEN '是'
	        WHEN DI.NEGATIVE_INVENTORY = 'N' THEN '否'
			 END AS IS_NEG_INV --是否允许负库存
      ,DI.CREATION_DATE AS CREATE_DATE --制单日期
	  ,CO.SAP_CHANNEL_CODE --SAP渠道编码
	  ,CO.CHANNEL_CODE --渠道编码
	  ,CO.CHANNEL_NAME --渠道名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_DT_DISTRIBUTOR_INFO_1029 DI 
 INNER JOIN SDI.SDI_DTCUST_CHANNEL_ORG_1031 CO 
    ON CO.ORG_ID = DI.ORG_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 A
    ON DI.ORGANIZATION_TYPE = A.LOOKUP_CODE
   AND A.LOOKUP_TYPE = 'ORGANIZATION_TYPE'
   AND A.ENABLED_FLAG = 'Y'
