INSERT OVERWRITE TABLE dwrdim.DWR_CUST_LATEST_DIMENSION_T
                        SELECT
                                G.KUNNR AS cust_num,
                                G.NAME2 AS cust_name,
                                G.VKORG AS sales_org,
                                G.VKGRP AS sales_grp,
                                G.KDGRP AS cust_grp,
                                G.PERNR AS sales_man,
                                current_timestamp AS dw_last_update_date
                        FROM
                                (
                                        SELECT
                                                G.KUNNR,
                                                G1.NAME2,
                                                SUBSTR(G.VKORG, 1, 2) AS VKORG,
                                                G.VKGRP,
                                                G.KDGRP,
                                                G.VTWEG,
                                                G2.PERNR,
                                                ROW_NUMBER() OVER(
                                                        PARTITION BY G.KUNNR,
                                                        SUBSTR(G.VKORG, 1, 2)
                                                        ORDER BY
                                                                G.VKORG,G.KUNNR,G.VTWEG ASC
                                                ) RN
                                        FROM
                                                SDI.SDI_KNVV_1001 G
                                        LEFT JOIN SDI.SDI_KNA1_1001 G1 ON G.KUNNR = G1.KUNNR
                                        LEFT JOIN sdi.sdi_knvp_1001 G2 ON G2.KUNNR=G.KUNNR
                                   AND SUBSTR(G.VKORG, 1, 2)=SUBSTR(G2.VKORG, 1, 2)
                                           AND G.VTWEG=G2.VTWEG
                                           and G2.parvw='VE'
                                        WHERE G.VKGRP <> '' AND G.FAKSD = ''
                                ) G
                        WHERE
                                G.RN = 1;
