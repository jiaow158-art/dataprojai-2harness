SELECT
       A.USER_ID --自增ID
      ,A.FACE_ID	 --	人脸ID
      ,A.WX_EXT_USER_ID	 --	企业微信外部联系人ID
      ,A.WX_UNION_ID	 --	微信UNIONID
      ,A.WX_OPEN_ID	 --	微信OPENID
      ,A.MEMBER_ID	 --	关联的会员ID
      ,A.STORE_ID	 --	所属门店
	  ,B.STORE_NAME --门店名称
      ,A.SOURCE	 --	客户来源,参考CLIENTSOURCEENUM类
	  ,D.NAME AS SOURCE_NAME --客户来源名称
      ,A.REAL_NAME	 --	真实姓名
      ,A.NICK_NAME	 --	昵称
      ,A.TEL_NUM --手机号
      ,A.BIRTHDAY	 --	生日YYYY-MM-DD
      ,A.GENDER	 --	性别 1-男性;2-女性;0-未知
      ,A.REMARK	 --	备注名
      ,A.PROVINCE_ID --	省区域ID
      ,A.PROVINCE_NAME --省区域名
      ,A.CITY_ID	 --	市区域ID
      ,A.CITY_NAME --市区域名
      ,A.DISTRICT_ID	 --	区区域ID
      ,A.DISTRICT_NAME -- 区区域名
      ,A.COMMUNITY_ADDRESS	 --	小区地址
      ,A.DETAILED_ADDRESS	 --	详细地址
      ,A.DETAILED_ADDRESS_LONGITUDE	 --	详细地址经度
      ,A.DETAILED_ADDRESS_LATITUDE	 --	详细地址纬度
      ,A.BIND_EMPLOYEE_ID	 --	归属员工ID
      ,A.BIND_RULE_ID	 --	客户归属关系绑定规则的ID
      ,A.LAST_BIND_TIME --最后绑定时间
      ,A.CREATE_TIME	 --	创建时间
      ,A.UPDATE_TIME	 --	更新时间
      ,A.PROMOTER	 --	推广人
      ,A.GRADE	 --	客户等级
      ,A.PROMOTION_TYPE	 --	推广方式， 1-导购推广 2-员工推广 3-客户推广
      ,A.CORP_NAME	 --	企业名称(只有添加的企业外部联系人才有)
      ,A.INVALID_CAUSE	 --	无效原因:1-客户不想要 2-客户选择其他品牌 3-客户收房延期 4-内部人员 5-其他 参考CLIENTINVALIDENUM枚举类
      ,A.TIKTOK_ID	 --	抖音数据ID
      ,A.FOLLOW_INFO	 --	跟进情况
      ,A.INVALID_TIME	 --	无效时间
      ,A.TELESALES_TIME	 --	电销时间
      ,A.DESCRIPTION	 --	备注/描述
      ,A.APP_FOLLOW_INFO	 --	APP跟进情况
      ,A.PROMOTER_ID	 --	推广人客户ID
      ,A.AMBASSADOR_STORE_ID	 --	推广大使门店
      ,A.AMBASSADOR_CREATE_TIME	 --	成为大使时间
      ,A.AGE_GROUP	 --	年龄层，1-18~25岁，2-26~35岁，3-36岁及以上
      ,A.DECORATE_PREDILECTION	 --	装修喜好，1-中国风，2-欧美风，3-日式
	  ,C.NAME AS DECORATE_PREDILECTION_NAME --装修喜好名称
      ,A.DEALER_ID	 --	经销商ID
      ,A.DEALER_NAME	 --	归属经销商
      ,A.LONGITUDE	 --	经度
      ,A.LATITUDE	 --	纬度
      ,A.IS_ASSET_CLIENT	 --	是否多资产客户:0-否/1-是
      ,A.ASSET_TYPE	 --	多资产类型:1-主客户/2-多资产客户
      ,A.IS_OLD_CLIENT_ADD	 --	是否老客户新增
      ,A.ACTIVATION_TIME	 --	激活时间
      ,A.ACTIVATION_EMPLOYEE_ID	 --	激活导购ID
      ,A.FOLLOW_TIME	 --	跟进时间
      ,A.FOLLOW_EMPLOYEE_ID	 --	跟进导购ID
      ,A.REACH_STORE_TIME	 --	到店时间
      ,A.REACH_STORE_EMPLOYEE_ID	 --	到店导购ID
      ,A.ALREADY_PAY_TIME	 --	交定时间
      ,A.ALREADY_PAY_TYPE	 --	交定方式
      ,A.ALREADY_PAY_MONEY	 --	交定金额
      ,A.ALREADY_PAY_EMPLOYEE_ID	 --	交定导购ID
      ,A.ORDER_TIME	 --	开单时间
      ,A.ORDER_MONEY	 --	开单金额
      ,A.ORDER_EMPLOYEE_ID	 --	开单导购ID
      ,A.ACCOMPLISH_TIME	 --	完成时间
      ,A.ACCOMPLISH_EMPLOYEE_ID	 --	完成导购ID
      ,A.SEND_GOODS_TIME	 --	发货时间
      ,A.TAKE_GOODS_TIME	 --	收货时间
      ,A.EVALUATION_TIME	 --	评价时间
      ,A.CLOSE_TIME	 --	关闭时间
      ,A.CLOSE_TYPE	 --	关闭原因
      ,A.CLOSE_EMPLOYEE_ID	 --	关闭导购ID
      ,A.STATUS	 --	客户状态:0待分配线索,1待认领客户,2线索,3客户
      ,A.STAGE_ID	 --	客户阶段ID
      ,A.UPDATE_STAGE_REMARK --	客户阶段变更备注
      ,A.BIND_TIME --客户绑定时间
      ,A.CUSTOMER_SOURCE --企业自定义来源
      ,A.IS_WXWORK_EXT_USER	 --客户是否是企微外部联系人
      ,A.IS_WHITELIST	 --是否白名单，1：是；0：否；
	  ,B.LEV1_DEPT_NAME --一级部门名称
	  ,B.LEV2_DEPT_NAME --二级部门名称
	  ,B.LEV3_DEPT_NAME --三级部门名称
	  ,B.LEV4_DEPT_NAME --四级部门名称
	  ,B.LEV5_DEPT_NAME --五级部门名称
	  ,B.LEV6_DEPT_NAME --六级部门名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,COALESCE(D.PARENT_ID,A.SOURCE) AS PARENT_SOURCE_ID --客户来源上级ID
	  ,COALESCE(E.NAME,D.NAME) AS PARENT_SOURCE_NAME --客户来源上级名称
  FROM DWIMD.DWI_MD_DMT_CUST_INFO_T A
  LEFT JOIN DWIMD.DWI_MD_DMT_DEPT_INFO_T B
    ON A.STORE_ID = B.STORE_ID
  LEFT JOIN SDI.SDI_CLIENT_DECORATE_PREDILECTION_1030 C
    ON A.DECORATE_PREDILECTION = C.ID
   AND C.DEL_FLAG = 'N'
   AND C.EP_ID = 3044
   AND C.IS_DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 D
    ON A.SOURCE = D.ID
   AND D.DEL_FLAG = 'N'
   AND D.EP_ID = 3044
   AND D.DELETED = 0
  LEFT JOIN SDI.SDI_CLIENT_SOURCE_INFO_1030 E
    ON D.PARENT_ID = E.ID
   AND E.DEL_FLAG = 'N'
   AND E.EP_ID = 3044
   AND E.DELETED = 0   
;
