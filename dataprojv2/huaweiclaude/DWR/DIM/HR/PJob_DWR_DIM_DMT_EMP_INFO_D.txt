INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_DMT_EMP_INFO_D
SELECT DISTINCT
       A.ID	AS EMP_ID --自增主键
      ,A.NAME AS EMP_NAME --员工姓名	   
      ,A.STORE_ID AS DEPT_ID --部门ID
	  ,COALESCE(B.DEPT_NAME,D.DEPT_NAME) AS DEPT_NAME --部门名称
      ,A.EP_ID --企业ID
      ,A.APP_ID	--APPID	  
      ,A.PHONE --员工手机号=账号
      ,A.SEX --性别
      ,A.POSITION_ID AS POST_ID --职位ID
      ,A.POSITION_NAME AS POST_NAME --职位名
      ,A.EMPLOYEE_GROUP_ID AS GROUP_ID --员工所在组ID
      ,A.EMPLOYEE_GROUP_NAME AS GROUP_NAME --员工所在组名
      ,A.STATUS	--员工状态：1在职2离职
	  ,CASE WHEN A.STATUS = '1' THEN '在职'
	        WHEN A.STATUS = '2' THEN '离职'
			 END AS STATUS_NAME --员工状态名称
      ,A.CREATE_TIME --创建时间
      ,A.UPDATE_TIME --更新时间
      ,A.EMAIL --邮箱
      ,A.IS_LINE_GROUP --是否线组长，1：是；0：否；
	  ,CASE WHEN A.IS_LINE_GROUP = '1' THEN '是'
	        WHEN A.IS_LINE_GROUP = '0' THEN '否'
			 END AS IS_LINE_GROUP_NAME --是否线组长名称
      ,A.IS_MARKET_MANAGER AS IS_MRKT_MAG	--是否市场经理，1：是；0：否；
	  ,CASE WHEN A.IS_MARKET_MANAGER = '0' THEN '否'
	        WHEN A.IS_MARKET_MANAGER = '1' THEN '是'
			 END AS IS_MRKT_MAG_NAME --是否市场经理名称
      ,A.IS_PRODUCT_MANAGER AS IS_PROD_MAG --是否产品经理，1：是；0：否；
	  ,CASE WHEN A.IS_PRODUCT_MANAGER = '0' THEN '否'
	        WHEN A.IS_PRODUCT_MANAGER = '1' THEN '是'
			 END AS IS_PROD_MAG_NAME --是否产品经理名称
      ,A.RETAIL_FLAG AS IS_RETAIL --承接新零售标识，1 是 0 否
	  ,CASE WHEN A.RETAIL_FLAG = '0' THEN '否'
	        WHEN A.RETAIL_FLAG = '1' THEN '是'
			 END AS IS_RETAIL_NAME --是否承接新零售名称
      ,A.ENABLE_WJDH AS IS_WJDH --是否是万家灯火，1 是 0 否
	  ,CASE WHEN A.ENABLE_WJDH = '0' THEN '否'
	        WHEN A.ENABLE_WJDH = '1' THEN '是'
			 END AS IS_WJDH_NAME --是否是万家灯火名称
	  ,COALESCE(B.LVL1_DEPT_ID,D.LVL1_DEPT_ID) AS LVL1_DEPT_ID --一级部门ID
	  ,COALESCE(B.LVL1_DEPT_SYC_ID,D.LVL1_DEPT_SYC_ID) AS LVL1_DEPT_SYC_ID --一级部门原ID
	  ,COALESCE(B.LVL1_DEPT_NAME,D.LVL1_DEPT_NAME) AS LVL1_DEPT_NAME --一级部门名称
	  ,COALESCE(B.LVL2_DEPT_ID,D.LVL2_DEPT_ID) AS LVL2_DEPT_ID --二级部门ID
	  ,COALESCE(B.LVL2_DEPT_SYC_ID,D.LVL2_DEPT_SYC_ID) AS LVL2_DEPT_SYC_ID --二级部门原ID
	  ,COALESCE(B.LVL2_DEPT_NAME,D.LVL2_DEPT_NAME) AS LVL2_DEPT_NAME --二级部门名称
	  ,COALESCE(B.LVL3_DEPT_ID,D.LVL3_DEPT_ID) AS LVL3_DEPT_ID --三级部门ID
	  ,COALESCE(B.LVL3_DEPT_SYC_ID,D.LVL3_DEPT_SYC_ID) AS LVL3_DEPT_SYC_ID --三级部门原ID
	  ,COALESCE(B.LVL3_DEPT_NAME,D.LVL3_DEPT_NAME) AS LVL3_DEPT_NAME --三级部门名称
	  ,COALESCE(B.LVL4_DEPT_ID,D.LVL4_DEPT_ID) AS LVL4_DEPT_ID --四级部门ID
	  ,COALESCE(B.LVL4_DEPT_SYC_ID,D.LVL4_DEPT_SYC_ID) AS LVL4_DEPT_SYC_ID --四级部门原ID
	  ,COALESCE(B.LVL4_DEPT_NAME,D.LVL4_DEPT_NAME) AS LVL4_DEPT_NAME --四级部门名称
	  ,COALESCE(B.LVL5_DEPT_ID,D.LVL5_DEPT_ID) AS LVL5_DEPT_ID --五级部门ID
	  ,COALESCE(B.LVL5_DEPT_SYC_ID,D.LVL5_DEPT_SYC_ID) AS LVL5_DEPT_SYC_ID --五级部门原ID
	  ,COALESCE(B.LVL5_DEPT_NAME,D.LVL5_DEPT_NAME) AS LVL5_DEPT_NAME --五级部门名称
	  ,COALESCE(B.LVL6_DEPT_ID,D.LVL6_DEPT_ID) AS LVL6_DEPT_ID --六级部门ID
	  ,COALESCE(B.LVL6_DEPT_SYC_ID,D.LVL6_DEPT_SYC_ID) AS LVL6_DEPT_SYC_ID --六级部门原ID
	  ,COALESCE(B.LVL6_DEPT_NAME,D.LVL6_DEPT_NAME) AS LVL6_DEPT_NAME --六级部门名称
	  ,COALESCE(B.LVL7_DEPT_ID,D.LVL7_DEPT_ID) AS LVL7_DEPT_ID --七级部门ID
	  ,COALESCE(B.LVL7_DEPT_SYC_ID,D.LVL7_DEPT_SYC_ID) AS LVL7_DEPT_SYC_ID --七级部门原ID
	  ,COALESCE(B.LVL7_DEPT_NAME,D.LVL7_DEPT_NAME) AS LVL7_DEPT_NAME --七级部门名称
	  ,COALESCE(B.LVL8_DEPT_ID,D.LVL8_DEPT_ID) AS LVL8_DEPT_ID --八级部门ID
	  ,COALESCE(B.LVL8_DEPT_SYC_ID,D.LVL8_DEPT_SYC_ID) AS LVL8_DEPT_SYC_ID --八级部门原ID
	  ,COALESCE(B.LVL8_DEPT_NAME,D.LVL8_DEPT_NAME) AS LVL8_DEPT_NAME --八级部门名称
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,A.PER_USER_ID AS USER_ID --用户ID
	  ,COALESCE(E.PROVINCE,F.PROVINCE) AS PROV_NAME --省份
	  ,COALESCE(E.CITY,F.CITY) AS CITY_NAME --城市
	  ,COALESCE(E.OUT_STORE_ID,F.OUT_STORE_ID) AS OUT_STORE_ID --OA门店编码
	  ,COALESCE(E.PARTICIPATION_IN_BUSINESS,F.PARTICIPATION_IN_BUSINESS) AS BUSI_TYPE --参与业务（1.瓷砖派单、2.洁具派单、3.线下客户、4.万家灯火派单、5.装到家 6.同城项目）
  FROM SDI.SDI_BUZ_EMPLOYEE_1030 A
  LEFT JOIN SDI.SDI_BUZ_EMPLOYEE_STORE_ROLE_1030 C
    ON A.ID = C.EMPLOYEE_ID
   AND A.EP_ID = C.EP_ID
   AND A.APP_ID = C.APP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_DEPT_INFO_D B
    ON A.STORE_ID = B.STORE_ID
   AND A.EP_ID = B.EP_ID
   AND A.APP_ID = B.APP_ID
  LEFT JOIN DWRDIM.DWR_DIM_DMT_DEPT_INFO_D D
    ON C.DEPT_ID = D.DEPT_SYC_ID
   AND A.EP_ID = D.EP_ID
   AND A.APP_ID = D.APP_ID   
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 E
    ON A.STORE_ID = E.ID
   AND A.EP_ID = E.EP_ID
   AND A.APP_ID = E.APP_ID
  LEFT JOIN SDI.SDI_BUZ_STORE_1030 F
    ON D.STORE_ID = F.ID
   AND A.EP_ID = F.EP_ID
   AND A.APP_ID = F.APP_ID 
 WHERE CASE WHEN A.EP_ID = 3044 AND A.APP_ID = 1831 THEN 1
            WHEN A.EP_ID = 3045 AND A.APP_ID = 1832 THEN 1
			ELSE 0
			 END = 1
