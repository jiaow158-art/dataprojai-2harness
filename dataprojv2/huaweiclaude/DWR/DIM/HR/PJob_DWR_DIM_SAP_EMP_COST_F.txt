set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

set hive.exec.dynamic.partition=true;

WITH T AS(--调整部门
SELECT A.MATCH_DEPT_CODE
      ,A.AFTER_ADJUST_DEPT_CODE AS DEPT_CODE
	  ,REPLACE(A.STAT_MONTH,'-','') AS STAT_MONTH
      ,B.DEPT_SHORT_NAME AS DEPT_NAME 
  FROM SDI.SDI_EHR_LABOR_COST_ADJUSTDEPARTMENT_1037 A
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B
    ON A.AFTER_ADJUST_DEPT_CODE = B.DEPT_CODE
   AND A.STAT_MONTH = B.STAT_MONTH
 WHERE A.AFTER_ADJUST_DEPT_CODE IS NOT NULL
   AND A.STAT_MONTH IS NOT NULL
),T1 AS(
SELECT 
       A.PERNR AS EMP_NUM --员工编号
      ,A.NACHN AS EMP_NAME --姓名
	  ,COALESCE(H.LVL1_DEPT_CODE,B.LVL1_DEPT_CODE) AS LVL1_DEPT_CODE --一级架构编号
	  ,COALESCE(H.LVL1_DEPT_SHORT_NAME,B.LVL1_DEPT_SHORT_NAME) AS LVL1_DEPT_NAME --一级架构名称
	  ,COALESCE(D.DEPT_CODE,COALESCE(B.LVL2_DEPT_CODE,A.ZLOO)) AS LVL2_DEPT_CODE --二级架构编号
	  ,COALESCE(D.DEPT_NAME,COALESCE(B.LVL2_DEPT_SHORT_NAME,A.ZLOO_TXT)) AS LVL2_DEPT_NAME --二级架构名称
      ,COALESCE(E.DEPT_CODE,A.ZL1O) AS LVL3_DEPT_CODE --三级架构编号
      ,COALESCE(E.DEPT_NAME,A.ZL1O_TXT) AS LVL3_DEPT_NAME --三级架构名称
      ,COALESCE(F.DEPT_CODE,A.ZL2O) AS LVL4_DEPT_CODE --四级架构编号
      ,COALESCE(F.DEPT_NAME,A.ZL2O_TXT) AS LVL4_DEPT_NAME --四级架构名称
      ,COALESCE(G.DEPT_CODE,A.ZL3O) AS LVL5_DEPT_CODE --五级架构编号
      ,COALESCE(G.DEPT_NAME,A.ZL3O_TXT) AS LVL5_DEPT_NAME --五级架构名称
      ,A.ZL4O AS LVL6_DEPT_CODE --六级架构编号
      ,A.ZL4O_TXT AS LVL6_DEPT_NAME --六级架构名称
      ,A.KOSTL AS COST_CENTER_CODE --成本中心
      ,A.PLANS AS EMP_POST_NAME --岗位
      ,A.ZZ_ZWLB AS EMP_POST_TYPE_NAME --职族
      ,A.TXART AS PAY_PLACE --发薪地
      ,A.ZGDGZ AS FX_PAY_AMT --固定工资（税前）
      ,COALESCE(A.ZFDGZ,0) - COALESCE(C.ADJUST_WAGES,0) AS FLT_PAY_AMT --浮动工资（税前）
      ,COALESCE(A.ZJL,0) - COALESCE(C.ADJUST_BONUS,0) AS BNS_AMT --奖金（税前）  专项激励
      ,A.ZSB AS SOC_INSR_AMT --社保（公司部分）
      ,A.ZGJJ AS ACCM_FUND_AMT --公积金（公司部分）
      ,A.ZGHJF AS EXP_AMT --工会经费
      ,A.ZFL AS WELFARE_AMT --福利
      ,A.ZZPF AS RECR_AMT --招聘费
      ,A.ZPX AS ED_AMT --培训及教育经费
      ,A.ZYWX AS ACDNT_INSR_AMT --意外险
      ,A.ZDFJJ AS PAYROLL_AMT --代发奖金
      ,A.ZFLHR AS HR_WELFARE_AMT --福利（HR系统取值）
      ,A.ZLZBC AS SVRNC_AMT --离职补偿金
      ,A.ZXZ AS SALE_QTY --销值/产量
      ,A.ZDW AS UNIT_ARTF_QTY --单位人工产出
      ,A.ZRG AS COST_COMENT --人工成本增加说明
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SUBSTR(A.ZDATE,1,4) || '-' || SUBSTR(A.ZDATE,5,2) AS STAT_MONTH --统计月份
      ,CASE WHEN D.MATCH_DEPT_CODE IS NOT NULL THEN 2
	        WHEN E.MATCH_DEPT_CODE IS NOT NULL THEN 3
			WHEN F.MATCH_DEPT_CODE IS NOT NULL THEN 4
			WHEN G.MATCH_DEPT_CODE IS NOT NULL THEN 5
			 END CHANGE_ID --调整部门的上级层级ID
  FROM (SELECT A.*
              ,ROW_NUMBER() OVER(PARTITION BY PERNR,SUBSTR(ZDATE,1,4) || '-' || SUBSTR(ZDATE,5,2)) AS RN
          FROM SDI.SDI_ZHRTPY045_1001 A
	     WHERE A.MANDT = 800 
	   ) A
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B
    ON A.ZL1O = B.DEPT_CODE
   AND A.ZDATE = REPLACE(B.STAT_MONTH,'-','')
  LEFT JOIN SDI.SDI_EHR_LABOR_COST_ADJUSTAMOUNT_1037 C
    ON SUBSTR(A.ZDATE,1,4) || '-' || SUBSTR(A.ZDATE,5,2) = C.STAT_MONTH
   AND A.PERNR = C.EMP_NUM
   AND C.EMP_NUM IS NOT NULL
   AND A.RN = 1
  LEFT JOIN T AS D
    ON A.ZDATE = D.STAT_MONTH
   AND A.ZL1O = D.MATCH_DEPT_CODE
  LEFT JOIN T AS E
    ON A.ZDATE = E.STAT_MONTH
   AND A.ZL2O = E.MATCH_DEPT_CODE
  LEFT JOIN T AS F
    ON A.ZDATE = F.STAT_MONTH
   AND A.ZL3O = F.MATCH_DEPT_CODE
  LEFT JOIN T AS G
    ON A.ZDATE = G.STAT_MONTH
   AND A.ZL4O = G.MATCH_DEPT_CODE
  --二级部门关联部门表取一级部门
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F H
    ON A.ZLOO = H.DEPT_CODE
   AND A.ZDATE = REPLACE(H.STAT_MONTH,'-','') 
 )
 
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_SAP_EMP_COST_F PARTITION(STAT_MONTH)
SELECT 
       A.EMP_NUM --员工编号
      ,A.EMP_NAME --姓名
	  ,CASE WHEN B.LVL1_DEPT_CODE IS NOT NULL THEN B.LVL1_DEPT_CODE
	        WHEN C.LVL1_DEPT_CODE IS NOT NULL THEN C.LVL1_DEPT_CODE
			WHEN D.LVL1_DEPT_CODE IS NOT NULL THEN D.LVL1_DEPT_CODE
			WHEN E.LVL1_DEPT_CODE IS NOT NULL THEN E.LVL1_DEPT_CODE
	        ELSE A.LVL1_DEPT_CODE
			 END AS LVL1_DEPT_CODE--一级架构编号
	  ,CASE WHEN B.LVL1_DEPT_NAME IS NOT NULL THEN B.LVL1_DEPT_NAME
	        WHEN C.LVL1_DEPT_NAME IS NOT NULL THEN C.LVL1_DEPT_NAME
			WHEN D.LVL1_DEPT_NAME IS NOT NULL THEN D.LVL1_DEPT_NAME
			WHEN E.LVL1_DEPT_NAME IS NOT NULL THEN E.LVL1_DEPT_NAME
	        ELSE A.LVL1_DEPT_NAME
			 END AS LVL1_DEPT_NAME --一级架构名称
	  ,CASE WHEN C.LVL2_DEPT_CODE IS NOT NULL THEN C.LVL2_DEPT_CODE
            WHEN D.LVL2_DEPT_CODE IS NOT NULL THEN D.LVL2_DEPT_CODE
	        WHEN E.LVL2_DEPT_CODE IS NOT NULL THEN E.LVL2_DEPT_CODE	        
	        ELSE A.LVL2_DEPT_CODE
			 END AS LVL2_DEPT_CODE --二级架构编号
	  ,CASE WHEN C.LVL2_DEPT_NAME IS NOT NULL THEN C.LVL2_DEPT_NAME
	        WHEN D.LVL2_DEPT_NAME IS NOT NULL THEN D.LVL2_DEPT_NAME
			WHEN E.LVL2_DEPT_NAME IS NOT NULL THEN E.LVL2_DEPT_NAME
	        ELSE A.LVL2_DEPT_NAME
			 END AS LVL2_DEPT_NAME --二级架构名称
      ,CASE WHEN D.LVL3_DEPT_CODE IS NOT NULL THEN D.LVL3_DEPT_CODE
	        WHEN E.LVL3_DEPT_CODE IS NOT NULL THEN E.LVL3_DEPT_CODE
	        ELSE A.LVL3_DEPT_CODE
			 END AS LVL3_DEPT_CODE --三级架构编号
      ,CASE WHEN D.LVL3_DEPT_NAME IS NOT NULL THEN D.LVL3_DEPT_NAME
	        WHEN E.LVL3_DEPT_NAME IS NOT NULL THEN E.LVL3_DEPT_NAME
	        ELSE A.LVL3_DEPT_NAME
			 END AS LVL3_DEPT_NAME --三级架构名称
      ,CASE WHEN E.LVL4_DEPT_CODE IS NOT NULL THEN E.LVL4_DEPT_CODE
	        ELSE A.LVL4_DEPT_CODE
			 END AS LVL4_DEPT_CODE --四级架构编号
      ,CASE WHEN E.LVL4_DEPT_NAME IS NOT NULL THEN E.LVL4_DEPT_NAME
	        ELSE A.LVL4_DEPT_NAME
			 END AS LVL4_DEPT_NAME --四级架构名称
      ,A.LVL5_DEPT_CODE --五级架构编号
      ,A.LVL5_DEPT_NAME --五级架构名称
      ,A.LVL6_DEPT_CODE --六级架构编号
      ,A.LVL6_DEPT_NAME --六级架构名称
      ,A.COST_CENTER_CODE --成本中心
      ,A.EMP_POST_NAME --岗位
      ,A.EMP_POST_TYPE_NAME --职族
      ,A.PAY_PLACE --发薪地
      ,A.FX_PAY_AMT --固定工资（税前）
      ,A.FLT_PAY_AMT --浮动工资（税前）
      ,A.BNS_AMT --奖金（税前）  专项激励
      ,A.SOC_INSR_AMT --社保（公司部分）
      ,A.ACCM_FUND_AMT --公积金（公司部分）
      ,A.EXP_AMT --工会经费
      ,A.WELFARE_AMT --福利
      ,A.RECR_AMT --招聘费
      ,A.ED_AMT --培训及教育经费
      ,A.ACDNT_INSR_AMT --意外险
      ,A.PAYROLL_AMT --代发奖金
      ,A.HR_WELFARE_AMT --福利（HR系统取值）
      ,A.SVRNC_AMT --离职补偿金
      ,A.SALE_QTY --销值/产量
      ,A.UNIT_ARTF_QTY --单位人工产出
      ,A.COST_COMENT --人工成本增加说明
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.STAT_MONTH --统计月份
  FROM T1 AS A
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B --重新关联调整部门后的架构
    ON A.LVL2_DEPT_CODE = B.DEPT_CODE
   AND A.STAT_MONTH = B.STAT_MONTH
   AND A.CHANGE_ID = 2
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F C --重新关联调整部门后的架构
    ON A.LVL3_DEPT_CODE = C.DEPT_CODE
   AND A.STAT_MONTH = C.STAT_MONTH
   AND A.CHANGE_ID = 3 
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F D --重新关联调整部门后的架构
    ON A.LVL4_DEPT_CODE = D.DEPT_CODE
   AND A.STAT_MONTH = D.STAT_MONTH
   AND A.CHANGE_ID = 4
  LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F E --重新关联调整部门后的架构
    ON A.LVL5_DEPT_CODE = E.DEPT_CODE
   AND A.STAT_MONTH = E.STAT_MONTH
   AND A.CHANGE_ID = 5
 
  UNION ALL
  
SELECT 
        A.EMP_NUM
       ,A.EMP_NAME
	   ,B.LVL1_DEPT_CODE
	   ,B.LVL1_DEPT_NAME
       ,A.L0_CODE AS LVL2_DEPT_CODE
       ,A.L0_NAME AS LVL2_DEPT_NAME
       ,A.L1_CODE AS LVL3_DEPT_CODE
       ,A.L1_NAME AS LVL3_DEPT_NAME
       ,A.L2_CODE AS LVL4_DEPT_CODE
       ,A.L2_NAME AS LVL4_DEPT_NAME
       ,A.L3_CODE AS LVL5_DEPT_CODE
       ,A.L3_NAME AS LVL5_DEPT_NAME
       ,A.L4_CODE AS LVL6_DEPT_CODE
       ,A.L4_NAME AS LVL6_DEPT_NAME   
       ,A.COST_CENTER AS COST_CENTER_CODE
       ,A.POST AS EMP_POST_NAME	   
       ,A.VOCATIONAL_RACE AS EMP_POST_TYPE_NAME
       ,A.PAYPLACE AS PAY_PLACE
       ,A.FIXED_SALARY AS FX_PAY_AMT
       ,A.FLOATING_WAGE AS FLT_PAY_AMT
       ,A.BONUS AS BNS_AMT
       ,A.SOCIAL_SECURITY AS SOC_INSR_AMT
       ,A.ACCUMULATION_FUND AS ACCM_FUND_AMT
       ,A.TRADE_UNION_OUTLAYS AS EXP_AMT
       ,A.WELFARE AS WELFARE_AMT
       ,A.RECRUITMENT_FEES AS RECR_AMT
       ,A.TRAINING_EDUCATION_EXPENSES AS ED_AMT
       ,A.ACCIDENT_INSURANCE AS ACDNT_INSR_AMT
       ,A.AGENT_BONUS AS PAYROLL_AMT
       ,A.WELFARE_HR AS HR_WELFARE_AMT
       ,A.SEVERANCE_PACKAGES AS SVRNC_AMT
       ,A.SALES_VALUE_OUTPUT AS SALE_QTY 
       ,A.UNIT_LABOR_OUTPUT AS UNIT_ARTF_QTY
       ,CAST(A.INCREASED_LABOR_COSTS AS STRING) AS COST_COMENT
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	   ,A.STAT_MONTH
   FROM SDI.SDI_EHR_LABOR_COST_DETAIL_1037 A
   LEFT JOIN DWRDIM.DWR_DIM_SAP_DEPT_INFO_F B
     ON A.L0_CODE = B.DEPT_CODE
    AND A.STAT_MONTH = B.STAT_MONTH
