with org_pos as (
select  t1.id , t1.code , t2.id as position_id  ,t4.code as bp_employee_code ,t1.parent_id  ,t1.name_path
        ,t1.ref_code   -- ehr部门代码
        ,t1.name                                                     --    共享组织架构描述
        ,t1.short_name                                                --    共享组织架构缩写
        ,t1.cost_center_code                               --    共享成本中心
        ,t1.cost_center_name                               --    共享成本中心描述
from  sdi.sdi_ORGANIZATION_1039 t1  
left join sdi.sdi_position_1039 t2 ON t1.id = t2.organization_id and  t2.category_id = 'BE63B549-58DF-11ED-972D-0242AC120023'  -- 岗位类别为财经BP
LEFT JOIN sdi.sdi_employee_position_1039 t3 
 ON t2.id = t3.position_id
 LEFT JOIN sdi.sdi_employee_1039 t4
 ON t3.employee_id = t4.ID  
 where not t1.frozen 
)
,
org_bp as (
select t1.id   ,t1.code ,t1.bp_employee_code ,t1.parent_id ,t1.name_path,t1.ref_code 
       ,t1.name                                                     --    共享组织架构描述
        ,t1.short_name                                                --    共享组织架构缩写
        ,t1.cost_center_code                               --    共享成本中心
        ,t1.cost_center_name                               --    共享成本中心描述
        ,COALESCE(T1.bp_employee_code,T2.bp_employee_code, T3.bp_employee_code, T4.bp_employee_code, T5.bp_employee_code, T6.bp_employee_code, T7.bp_employee_code, T8.bp_employee_code, T9.bp_employee_code, 
 T10.bp_employee_code, T11.bp_employee_code) AS  final_bp_employee_code
from org_pos  T1 
LEFT JOIN org_pos T2   -- 往上找有设置BP的部门
ON T1.parent_id = T2.ID 
LEFT JOIN org_pos T3 
ON T2.parent_id = T3.ID
LEFT JOIN org_pos T4
ON T3.parent_id = T4.ID
LEFT JOIN org_pos T5 
ON T4.parent_id = T5.ID
LEFT JOIN org_pos T6 
ON T5.parent_id = T6.ID
LEFT JOIN org_pos T7
ON T6.parent_id = T7.ID
LEFT JOIN org_pos T8
ON T7.parent_id = T8.ID
LEFT JOIN org_pos T9
ON T8.parent_id = T9.ID
LEFT JOIN org_pos T10
ON T9.parent_id = T10.ID
LEFT JOIN org_pos T11
ON T10.parent_id = T11.ID
)
INSERT OVERWRITE TABLE dwrdim.DWR_DIM_ORG_COST_EMP_D
select
     d.emp_num                                     emp_num                                    --   人员编码
     ,d.emp_name                 emp_name                    --   人员名称
    ,d.department_code                      EHR_ORG_CODE               --    EHR组织架构代码
    ,d.department_name               EHR_ORG_DESC                    --    EHR组织架构描述
    ,s.code                                          FSSC_ORG_CODE                --          共享组织架构代码
    ,s.name                                          FSSC_ORG_DESC                    --    共享组织架构描述
    ,s.short_name                             FSSC_ORG_SHORT                    --    共享组织架构缩写
    ,s.cost_center_code                   FSSC_COST_CENTER            --    共享成本中心
    ,c.costcenter                                    SAP_COST_CENTER                --    BW成本中心
    ,s.cost_center_name                   COST_CENTER_DESC            --    共享成本中心描述
    ,c.comp_code                                   comp_code                                    --   公司
    ,c.sales_grp                                    sales_grp                                    --   销售组
    ,t.txtsh  sales_desc                                --   销售组描述
    ,`current_timestamp`()  dw_last_update_date
    ,s.final_bp_employee_code as bp_employee_code
    ,bpn.emp_name as bp_employee_name
    from  DWRDIM.DWR_DIM_EMPLOYEE_BASIC_INFO_D d 
    left join  org_bp s on  s.ref_code  = d.department_code
    left join sdi.SDI_BI0_QCOSTCENTER_1009 c
     on substr(S.cost_center_code,2,10) = substr(c.costcenter,2,10) and c.dateto = '99991231'  and c.co_area='DP00'
    LEFT JOIN SDI.SDI_TSALES_GRP_1009 t ON c.sales_grp= t.sales_grp and t.langu='1'
     LEFT JOIN DWRDIM.DWR_DIM_EMPLOYEE_BASIC_INFO_D bpn ON  s.final_bp_employee_code = bpn.emp_num and substr(bpn.end_date,1,10) ='4712-12-31'
     where 1=1 
       and substr(d.end_date,1,10) ='4712-12-31';
