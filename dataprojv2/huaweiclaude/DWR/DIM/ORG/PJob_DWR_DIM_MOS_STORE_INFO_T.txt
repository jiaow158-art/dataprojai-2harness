WITH T1 AS(
SELECT ID
      ,PARTICIPATION_IN_BUSINESS
      ,TABLE_VIEW.BUSI_TYPE
  FROM SDI.SDI_SYS_CLIENT_1012 A
 LATERAL VIEW EXPLODE(SPLIT(PARTICIPATION_IN_BUSINESS,',')) TABLE_VIEW AS BUSI_TYPE
)
INSERT OVERWRITE TABLE DWRDIM.DWR_DIM_MOS_STORE_INFO_T
SELECT A.ID --ID
      ,A.CODE AS STORE_CODE --门店编码
	  ,A.OA_STORE_CODE --OA门店编码
	  ,A.NAME AS STORE_NAME --门店名称
	  ,A.ABBREVIATION AS STORE_ABBR_NAME --门店简称
	  ,A.ROLE_ID --角色ID
	  ,A.ROLE_NAME --角色名称
	  ,A.TYPE_ID AS CUST_TYPE_ID --客户类型ID
	  ,A.TYPE_NAME AS CUST_TYPE_NAME --客户类型名称
	  ,A.ATTRIBUTE_ID AS CUST_ATTR_ID --客户属性ID
	  ,A.ATTRIBUTE_NAME AS CUST_ATTR_NAME --客户属性名称
	  ,A.CATEGORY_ID AS CUST_CAT_ID --客户分类ID
	  ,A.CATEGORY_NAME AS CUST_CAT_NAME --客户分类名称
	  ,A.DEPARTMENT_ID AS DEPT_ID --部门ID
	  ,A.DEPARTMENT_NAME AS DEPT_NAME --部门名称
	  ,A.SALESMAN_ID AS SALE_EMP_ID --业务员ID
	  ,A.SALESMAN_NAME AS SALE_EMP_NAME --业务员名称
	  ,A.CHANNEL_ID AS CHNL_ID --渠道ID
	  ,A.CHANNEL_NAME AS CHNL_NAME --渠道名称
	  ,A.CONTACTS AS CNTCT_NAME --联系人
	  ,A.CONTACTS_PHONE AS CNTCT_PHONE_NUM --联系人电话
	  ,A.PROVINCE AS PROV_NAME --省份
	  ,A.CITY AS CITY_NAME --城市
	  ,A.DISTRICT AS DIST_NAME --区县
	  ,A.ADDRESS AS ADDR --详细地址
	  ,A.ACCEPT_TYPE AS ACC_TYPE_ID --接单类型ID
	  ,CASE WHEN A.ACCEPT_TYPE = 0 THEN '经销商接单'
	        WHEN A.ACCEPT_TYPE = 1 THEN '门店接单'
			WHEN A.ACCEPT_TYPE = 2 THEN '导购接单'
			 END AS ACC_TYPE_NAME --接单类型描述
	  ,A.LAT --纬度
	  ,A.LNG --经度
	  ,A.SCOPE_SERVICE AS SRVC_SCP --服务范围（米）
	  ,CAST(A.AREA AS DECIMAL(38,10)) AS STORE_AREA --门店面积（平方米）
	  ,A.CREATE_TIME --创建时间
	  ,A.CREATER_NAME AS CREATE_EMP_NAME --创建人名称
	  ,A.MODIFY_TIME AS UPDATE_TIME --更新时间
	  ,A.MODIFIER_NAME AS UPDATE_EMP_NAME --更新人名称
	  ,A.PARENT_ID AS PARNT_ID --上级ID
	  ,A.PARENT_CODE AS PARNT_CODE --上级编码
	  ,A.PARENT_NAME AS PARNT_NAME --上级名称
	  ,A.LEVEL --层级
	  ,A.PARENT_DEP_ID AS PARNT_DEPT_ID --上级部门ID
	  ,A.PARENT_DEPT_NAME AS PARNT_DEPT_NAME --上级部门名称
	  ,A.DISTRIBUTOR_ID AS DEALER_ID --经销商ID
	  ,A.DISTRIBUTOR_NAME AS DEALER_NAME --经销商名称
	  ,A.MARKETING_ID AS MKT_ID --营销部ID
	  ,A.MARKETING_NAME AS MKT_NAME --营销部名称
	  ,A.LARGE_AREA_ID --大区ID
	  ,A.LARGE_AREA_NAME --大区名称
	  ,A.CLIENT_LEVEL_ID AS CUST_LVL_ID --客户级别ID
	  ,A.CLIENT_LEVEL_CODE AS CUST_LVL_CODE --客户级别编码	  
	  ,A.CLIENT_LEVEL_NAME AS CUST_LVL_NAME --客户级别名称
	  ,LPAD(A.SAP_CODE,10,'0') AS CUST_NUM --SAP客户编码
	  ,A.SAP_CHANNEL_CODE AS CHNL_CODE --SAP渠道编码
	  ,A.SAP_SALES_ORG AS SALE_ORG_CODE --SAP销售组织编码
	  ,A.LINE_GROUP_ID AS SALE_GRP_ID --线组ID
	  ,A.LINE_GROUP_NAME AS SALE_GRP_NAME --线组名称
	  ,A.STORE_TAG --店态
	  ,A.PARTICIPATION_IN_BUSINESS AS BUSI_ID --参与业务ID
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN B.ID IS NOT NULL THEN '是' ELSE '否' END IS_TC --是否参与同城业务
	  ,A.JOIN_CITY_PROTECT_TIME AS TC_TIME --参与同城业务时间
	  ,A.SAP_SALES_GROUP_CODE AS SALE_GRP_CODE --线组编码
  FROM SDI.SDI_SYS_CLIENT_1012 A
  LEFT JOIN (SELECT DISTINCT ID,BUSI_TYPE FROM T1 WHERE BUSI_TYPE = 6) B
    ON A.ID = B.ID
 WHERE A.USABLE_FLAG = 1
