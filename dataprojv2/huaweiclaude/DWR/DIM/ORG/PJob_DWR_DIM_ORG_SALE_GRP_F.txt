DROP TABLE IF EXISTS DWRDIM.DWR_DIM_ORG_SALE_GRP_TMP01;

CREATE TABLE DWRDIM.DWR_DIM_ORG_SALE_GRP_TMP01
AS
SELECT H.NODENAME,
		  CASE WHEN H.IOBJNM = '0SALES_GRP' THEN ST.TXTSH
			   WHEN H.IOBJNM = '0SALES_OFF' THEN ST1.TXTSH 
		  ELSE R.TXTSH	
		  END AS NODEDESC,
		  H.NODEID,
		  H.PARENTID , 
		  H.TLEVEL,
		  H.IOBJNM
	FROM SDI.SDI_BI0_HSALES_GRP_1001 H
	LEFT JOIN SDI.SDI_BI0_TSALES_GRP_1009  ST 
	ON H.NODENAME = ST.SALES_GRP 
	AND ST.LANGU = '1' 
	AND H.IOBJNM = '0SALES_GRP'
	LEFT JOIN SDI.SDI_BI0_TSALES_OFF_1009 AS ST1 
	ON H.NODENAME = ST1.SALES_OFF 
	AND ST1.LANGU = '1' 
	AND H.IOBJNM = '0SALES_OFF'
	LEFT JOIN SDI.SDI_RSTHIERNODE_1009 R 
	ON R.NODENAME = H.NODENAME 
	AND R.OBJVERS = 'A' 
	AND R.HIEID = H.HIEID
	AND R.LANGU = '1'
	WHERE H.HIEID IN ('JH3NA9Q0QI18EFL6T3J3CCNRK')
	AND H.DEL_FLAG = 'N'
	AND H.OBJVERS = 'A'
	;
