INSERT OVERWRITE TABLE DWRLTC.DWR_BO_SIGN_ORDER_F
SELECT A.DELIVER_SIGN_ID AS SIGN_ID --签收ID
      ,A.SIGN_CODE AS SIGN_NUM --签收单号
      ,A.BUSINESS_ID AS BUSI_ID --商机ID
	  ,A.BUSINESS_CODE AS BUSI_CODE	--商机编号
	  ,A.DESPATCH_ID AS DSPT_ID --发运单ID
	  ,A.DESPATCH_CODE AS DSPT_CODE --发运单号
	  ,A.DELIVERY_NO AS DELV_NUM --提货单号
	  ,A.WBS AS WBS_NUM --WBS号
	  ,A.SHIPMENT_ID AS DELV_SHIP_ID --出货计划ID
      ,A.SHIPMENT_CODE AS DELV_SHIP_CODE --出货计划编码
      ,A.DELIVER_DEMAND_ID AS DMAND_ID --要货单ID
      ,A.DEMAND_CODE AS DMAND_NUM --要货单编号
	  ,A.CREATION_DATE AS SIGN_DATE --签收日期
      ,A.SIGN_STATUS AS SIGN_ST_CODE --签收状态编码
      ,CASE WHEN A.SIGN_STATUS = 'DRAFT' THEN '草稿'
	        WHEN A.SIGN_STATUS = 'ALREADY_SIGNED' THEN '已签收'
			WHEN A.SIGN_STATUS = 'REFUSAL' THEN '审批拒绝'
			WHEN A.SIGN_STATUS = 'ALREADY_ACCEPT' THEN '已验收'
			WHEN A.SIGN_STATUS = 'APPROVAL' THEN '审批中'
			 END AS SIGN_ST_NAME --签收状态描述
	  ,A.LOGISTICS_COST_CODE AS LGSTCS_COST_CENT_CODE --物流费成本中心
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_DELIVER_SIGN_HEADER_1033 A
 WHERE A.DELETE_FLAG = 0
