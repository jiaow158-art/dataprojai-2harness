--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with user_pd_zy as 
(
select employee_number from SDI.SDI_report_user_trajectory_1050
where trajectory_url in (
'/order/plan/page',--自营用户
'/order/statistics',--自营用户
'/order/dmand/page'--自营用户
) 
group by employee_number
),
user_pd_fzy as 
(
select employee_number from SDI.SDI_report_user_trajectory_1050
where trajectory_url in (
'/order/user/page'--非自营用户
) 
group by employee_number
),
cust_relation as 
(
SELECT DISTINCT 
CASE
	WHEN ZZKUNNR IS NULL OR ZZKUNNR=''THEN KUNNR
	ELSE ZZKUNNR
END AS ZZKUNNR,--母客编
KNVV.KUNNR,--子客编
T171T.BZTXT
FROM sdi.sdi_knvv_1001 KNVV
INNER JOIN sdi.sdi_t171t_1001 T171T
ON KNVV.BZIRK=T171T.BZIRK
AND T171T.SPRAS='1'
AND T171T.BZTXT IN('华南运营中心','西北运营中心','湘鄂运营中心','东北运营中心','赣皖运营中心','浙江运营中心','华北运营中心','鲁豫晋运营中心','西南运营中心','江苏运营中心','江浙运营中心')
)
,dqgc_kb as (   -- 仅仅代签工程的客编
select cust_num,count(distinct sale_org) c1,count(distinct case when sale_org='1020' then sale_org end) c2
from dm.dm_otd_so_order_user_t 
group by cust_num  having count(distinct sale_org) =1 and count(distinct case when sale_org='1020' then sale_org end)=1
) 
,base_data as 
(
--自营用户基数：工程市场中心 + 子公司
select 
t1.item_submit_date,--创建日期
t1.create_emp_num,--业务员账号
t1.create_emp_name,--业务员姓名
t1.cust_num,--客户编码
t1.sale_org_code,--销售组织
t1.sale_chcl_code,--分销渠道
case 
	when short_objid_name_t5 like '%子公司' then short_objid_name_t5
	when short_objid_name_t5 = '工程市场中心' and 
		short_objid_name_t7 IN('战略地产一部','战略地产二部','战略地产三部','战略地产四部','战略地产五部','企商组') then short_objid_name_t7
end as sub_grp,
case when t2.employee_number is not null then '自营用户-是'  else '自营用户-否' end as user_type,
null as ZZKUNNR
From dm.dm_bo_plan_order_t t1
join DWIMD.DWI_ORG_PERNR_STRUCTURE_INFO_T org On t1.create_emp_num = org.employee_code
left join user_pd_zy t2 on t1.create_emp_num = t2.employee_number
where 1=1
and from_unixtime(unix_timestamp(t1.item_submit_date, 'yyyy-MM')) >= from_unixtime(unix_timestamp('${PERIOD_ID_D}', 'yyyy-MM'))
union all
--非自营用户：客户编码-区域运营中心
select 
t1.create_date,--创建日期
null as create_emp_num,--业务员账号
null as create_emp_name,--业务员姓名
t1.cust_num,--客户编码
null as sale_org_code,--销售组织
sale_channel_code as sale_chcl_code,--分销渠道
t3.BZTXT as sub_grp,
case when t2.employee_number is not null then '非自营用户-是'  else '非自营用户-否' end as user_type,
t3.ZZKUNNR
From dm.dm_otd_so_order_user_t t1
left join user_pd_fzy t2 on t1.cust_num = case when length(t2.employee_number)>=12 then substring(t2.employee_number,1,10) else t2.employee_number end
join cust_relation t3 on t1.cust_num = t3.KUNNR 
left join dqgc_kb dqgc ON t1.cust_num = dqgc.cust_num
where 1=1
and dqgc.cust_num is null -- 仅代签工程的客编排除掉
and from_unixtime(unix_timestamp(t1.create_date, 'yyyy-MM')) >= from_unixtime(unix_timestamp('${PERIOD_ID_D}', 'yyyy-MM'))
)
INSERT OVERWRITE TABLE dwrltc.dwr_yddd_active_d PARTITION(PERIOD_ID_M)
select 
item_submit_date,--创建日期
create_emp_num,--业务员账号
create_emp_name,--业务员姓名
cust_num,--客户编码
sale_org_code,--销售组织
sale_chcl_code,--分销渠道
sub_grp,--组织
user_type,--用户标识
ZZKUNNR,--母客编
CURRENT_TIMESTAMP() as dw_last_update_date,
substring(item_submit_date,1,7) as PERIOD_ID_M
from base_data
where sub_grp is not null
