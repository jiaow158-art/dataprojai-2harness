WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_RQST_PAYOT_INFO_F
SELECT
       A.REQUEST_PAYOUT_ID AS RQST_PAYOT_ID --请款单ID
      ,A.REQUEST_PAYOUT_CODE AS RQST_PAYOT_NUM --请款单单号
      ,A.CONTRACT_CODE AS CNTC_NUM --合同号
      ,A.CONTRACT_NAME AS CNTC_NAME --合同名称
      ,A.BUSINESS_ID AS BUSI_ID --商机ID
      ,A.BUSINESS_CODE AS BUSI_NUM --商机编号
      ,A.BUSINESS_NAME AS BUSI_NAME --商机名称
      ,A.CUSTOMER_CODE AS CUST_NUM --客户编号
      ,A.CUSTOMER_NAME AS CUST_NAME --客户名称
      ,A.SETTLEMENT_MODE AS SETLM_TYPE_CODE --结算方式编码
	  ,T1.MEANING AS SETLM_TYPE_NAME --结算方式描述
      ,A.SETTLEMENT_DATE_START AS SETLM_START_DATE --结算时段开始
      ,A.SETTLEMENT_DATE_END AS SETLM_END_DATE --结算时间段结束
      ,A.WBS AS WBS_NUM --WBS号
      ,A.REQUEST_PAYOUT_PAY AS RQST_PAYOT_TYPE_CODE --请款类型编码
	  ,T2.MEANING AS RQST_PAYOT_TYPE_NAME --请款类型描述
      ,A.CUSTOMER_TICKET_MODE AS OPEN_TKT_TYPE_CODE --客户开票方式编码
	  ,T3.MEANING AS OPEN_TKT_TYPE_NAME --客户开票方式描述
      ,COALESCE(A.ACCRUED_REQUEST_AMOUNT,0) AS ACR_RQST_AMT --应计请款金额
      ,COALESCE(A.PRACTICAL_REQUEST_AMOUNT,0) AS TACT_RQST_AMT --实际请款金额
      ,COALESCE(A.ASK_TICKET_AMOUNT,0) AS ASK_TKT_AMT --要求开票金额
      ,A.PREDICT_TICKET_DATE AS PRDCT_TKT_TIME --预计开票日期
      ,A.TICKET_STATUS AS TKT_ST_CODE --开票状态编码
	  ,T4.MEANING AS TKT_ST_NAME --开票状态描述
      ,A.CREATION_DATE AS CREATE_TIME --创建时间	  
      ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	  ,T5.USER_NAME AS CREATE_EMP_NUM --创建人工号
      ,A.CREATED_NAME AS CREATE_EMP_NAME --创建人名字[业务员]
      ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
      ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	  ,T6.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	  ,T6.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名
      ,A.VERSION_NUM AS VERS_NUM --版本号
      ,A.OVERDUE_FLAG AS IS_OVRDU_CODE --是否逾期编码
	  ,T7.MEANING AS IS_OVRDU_NAME --是否逾期描述
      ,COALESCE(A.FEE_REQUEST_AMOUNT,0) AS RQST_FEE_AMT --本次费用请款金额
      ,COALESCE(A.TOTAL_REQUEST_AMOUNT,0) AS TOT_RQST_AMT --请款金额合计
      ,A.SOURCE_SALES_MAN_ID AS SRC_SALE_EMP_ID --源业务员ID,商机继承后赋值
      ,A.SOURCE_SALES_MAN_CODE AS SRC_SALE_EMP_NUM --源业务员编码，商机继承后赋值
      ,A.SOURCE_SALES_MAN_NAME AS SRC_SALE_EMP_NAME --源业务员名称,商机继承后赋值
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_1033 A
  LEFT JOIN T AS T1
    ON A.SETTLEMENT_MODE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'SETTLEMENT'
  LEFT JOIN T AS T2
    ON A.REQUEST_PAYOUT_PAY = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'SETTLE_ASK_PAYMENT'
  LEFT JOIN T AS T3
    ON A.CUSTOMER_TICKET_MODE = T3.LOOKUP_CODE
   AND T3.LOOKUP_TYPE = 'CUSTOMER_BILLING_METHOD'
  LEFT JOIN T AS T4
    ON A.TICKET_STATUS = T4.LOOKUP_CODE
   AND T4.LOOKUP_TYPE = 'SETTLE_OPEN_TICKET_STATUS'
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T5
    ON A.CREATED_BY = T5.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T6
    ON A.LAST_UPDATED_BY = T6.USER_ID
  LEFT JOIN T AS T7
    ON A.OVERDUE_FLAG = T7.LOOKUP_CODE
   AND T7.LOOKUP_TYPE = 'YES_OR_NO'
 WHERE A.DELETE_FLAG = 0
