-- LTC商机清单明细表
INSERT OVERWRITE TABLE DM.DM_LTC_BUSINESS_OP_LIST_DETAIL_T
select  
 a.business_code                as business_code                -- 商机编号
,a.business_name                as business_name                -- 商机名称
,a.strategic_business_code      as strategic_business_code      -- 战略商机编号
,a.strategic_business_name      as strategic_business_name      -- 战略商机名称
,a.customer_name                as customer_name                -- 客户名称
,a.apply_user                   as apply_user                   -- 申请人
,a.business_type                as business_type                -- 商机类别
,blv1.meaning                   as business_type_name           -- 商机类别名称
,a.effective_date               as effective_date               -- 有效日期
,a.business_stage               as business_stage               -- 商机阶段
,a.s0approve_state              as s0approve_state              -- S0审批状态
,blv2.meaning                   as s0approve_state_name         -- S0审批状态名称
,a.s1approve_state              as s1approve_state              -- S1审批状态
,blv3.meaning                   as s1approve_state_name         -- S1审批状态
,a.detail_address               as detail_address               -- 详细地址
,a.product_category             as product_category             -- 产品品类
,blv4.meaning                   as product_category_name        -- 产品品类名称
,case a.division 
        when 'BATHROOM_DIVISION' then '卫浴事业部'
    when 'CERAMIC_TILE_DIVISION' then '瓷砖事业部' 
    when 'HOUSEHOLD_DIVISION' then '家居事业部' 
    when 'NEW_ECOLOGICAL_TIMBER_DIVISION' then '生态新材事业部' 
    else '其他'
    end                         as business_dept                -- 事业部
,case a.transfer_flag 
        when 'N' then '否'
        else '是'
--    when 'Y' then '是' 
    end                         as business_op_transfer         -- 是否来自转让商机
,CURRENT_TIMESTAMP()			as DW_LAST_UPDATE_DATE			-- 数据同步时间
,a.pid 							as pid							-- 父级ID			-----新增
,p.business_code 				as p_business_code 				-- 父级商机编号		-----新增
,p.business_name 				as p_business_name 				-- 父级商机名称		-----新增
,ur.user_name  					as apply_user_code 				-- 申请人工号		-----新增
FROM
        SDI.SDI_BUSINESS_OPPORTUNITY_1033 a
        LEFT JOIN SDI.SDI_BASE_USERS_1029 bu ON a.created_by = bu.user_id
        LEFT JOIN SDI.SDI_BASE_SAP_ORGANIZATIONAL_STRUCTURE_1029 bsos ON bu.user_name = bsos.employee
        LEFT JOIN SDI.SDI_BASE_SAP_ORGANIZATIONAL_STRUCTURE_1029 bsos2 ON a.factory_salesman_num = bsos2.employee 
		left join SDI.SDI_BASE_USERS_1029 ur on ur.user_id = a.apply_user_id 	-----新增
        left join SDI.SDI_BUSINESS_OPPORTUNITY_1033 P on P.business_id = a.pid 		-----新增
        left join SDI.SDI_BASE_LOOKUP_VALUES_1029 blv1 on blv1.lookup_code = a.business_type
              and blv1.system_code = 'LTC'
                    and blv1.enabled_flag = 'Y'
                    and blv1.lookup_type = 'BUSINESS_CATEGORY'
        left join SDI.SDI_BASE_LOOKUP_VALUES_1029 blv2 on blv2.lookup_code = a.s0approve_state
              and blv2.system_code = 'LTC'
                    and blv2.enabled_flag = 'Y'
                    and blv2.lookup_type = 'BUSINESS_APPROVE_STATE'
        left join SDI.SDI_BASE_LOOKUP_VALUES_1029 blv3 on blv3.lookup_code = a.s1approve_state
              and blv3.system_code = 'LTC'
                    and blv3.enabled_flag = 'Y'
                    and blv3.lookup_type = 'BUSINESS_APPROVE_STATE'
        left join SDI.SDI_BASE_LOOKUP_VALUES_1029 blv4 on blv4.lookup_code = a.product_category
              and blv4.system_code = 'LTC'
                    and blv4.enabled_flag = 'Y'
                    and blv4.lookup_type = 'PRODUCT_CATEGORY'
WHERE
        1 = 1 
        AND a.delete_flag = 0 
--         AND ( a.created_by = 997 OR a.factory_salesman_num = 10004006 ) 
        AND a.business_code NOT IN ( SELECT business_code FROM SDI.SDI_BUSINESS_DARE_1033 WHERE dare_status NOT IN ( 'ALLOW' ) )
