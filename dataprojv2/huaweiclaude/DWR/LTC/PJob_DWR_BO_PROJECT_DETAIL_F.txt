WITH BUSINESS_BUSINESS_SALE_FORECAST_DATA AS( -- 商机产品品类数据
	SELECT T1.business_id,
		   CONCAT_WS(',',collect_set(T2.meaning)) as PROD_CATEGORY,  -- 产品品类
		   sum(nvl(t1.estimated_amount,0)) as estimated_amt, -- 预计标的金额
		   sum(nvl(t1.estimated_year_deliver_amount,0)) as estimated_year_deliver_amt -- 预计本年度可发货金额
	FROM SDI.sdi_business_sale_forecast_1033 T1  -- 商机销售预测
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T2 
	ON T1.big_category = T2.lookup_code 
	AND T2.lookup_type = 'PRODUCT_CATEGORY' -- 产品品类
	group by T1.business_id
),
PROJECT_APPLY_AREA_DATA AS(  -- 项目评审面积/件数数据
	SELECT T1.business_id ,
		   T1.business_code,
		   SUM(T2.quantity) AS PROJECT_APPLY_AREA -- 项目评审面积/件数
	FROM SDI.SDI_PROJECT_REVIEW_1033 T1    -- LTC 项目评审
	LEFT JOIN SDI.SDI_PROJECT_REVIEW_PROFIT_1033 T2  
	ON T1.project_id = T2.project_id 
	group by T1.business_id ,
			 T1.business_code
),
CONTRACT_LOGIN_AREA_DATA AS( -- 合同登记面积/件数 数据
	select T3.business_id ,
	 	   T3.business_code ,
	 	   SUM(T4.quantity) AS contract_login_area, -- 合同登记面积/件数
	 	   SUM(T4.contract_amount) AS contract_login_amount -- 合同登记金额
	FROM SDI.SDI_CONTRACT_1033 T3   -- LTC 合同
	LEFT JOIN SDI.SDI_CONTRACT_PROFIT_1033 T4
	ON T3.contract_id = T4.contract_id 
	WHERE T3.business_id IS NOT NULL   -- 有商机为空的数据
	GROUP BY T3.business_id ,
			 T3.business_code 
),
HIS_OA_SJ_DATA AS( -- OA 历史商机数据
	select T.dh AS BO_CODE,
		   NVL(T.yjnchl,0) AS oa_expected_year_out_amt -- 预计年出货量(万元)
	from(
	SELECT T.lsh ,T.dh ,T.bbsj,T.yjnchl,
		   row_number() over(PARTITION by dh order by bbsj desc) rn
	FROM SDI.SDI_FORMTABLE_MAIN_60_1004 T   -- OA 历史商机
	WHERE NVL(T.dh,'ISNULL') <> 'ISNULL'
	) T
	where t.rn = 1
),
estimated_signed_date_data as(  -- 预计合同签订时间数据
    SELECT business_id,
    	   estimated_signed_date  -- 预计合同签订时间
    FROM(
    	select business_id,
    		   estimated_signed_date,
    		   ROW_NUMBER() OVER(PARTITION BY business_id ORDER BY estimated_signed_date asc) RN
    	from SDI.sdi_business_sale_forecast_1033 t
    ) T
    WHERE RN = 1
),
estimated_delivery_date_data as(  -- 预计出货时间数据
    SELECT business_id,
    	   estimated_delivery_date  -- 预计出货时间
    FROM(
    	select business_id,
    		   estimated_delivery_date,
    		   ROW_NUMBER() OVER(PARTITION BY business_id ORDER BY estimated_delivery_date asc) RN
    	from SDI.sdi_business_sale_forecast_1033 t
    ) T
    WHERE RN = 1
),
DELIVER_SHIPMENT_DATA AS(  -- 大货要货单最早时间数据
	SELECT business_code,
		   min(SUBSTRING(CAST(creation_date AS STRING),1,10)) as  MIN_DELIVER_DATE -- 大货要货单最早时间
	FROM SDI.SDI_DELIVER_SHIPMENT_HEADER_1033 
	where plan_type in ('DT_PLAN','ZL_PLAN') 
	group by business_code
),
GL_INVOICE_CONDITIONAL_DATA AS(  -- 发票开票时间最早时间数据
	select engineer_report_code AS business_code,
		   min(SUBSTRING(CAST(bill_date AS STRING),1,10)) as  MIN_BILL_DATE -- 发票开票时间最早时间
	from dwrfin.dwr_gl_invoice_conditional_f 
	WHERE 1=1  
	--and period_id_m >= REPLACE('${PERIOD_ID_M}','-','')
	--WHERE period_id_m >= '201806' -- 初始化数据放开此条件
	GROUP BY engineer_report_code
),

	   T8.node_name3 AS LEVEL_DEPT_NAME_3, -- 3层部门名称

	   T8.node_desc4 AS LEVEL_DEPT_DESC_4, -- 4层部门描述
	   T8.node5 AS LEVEL_DEPT_CODE_5, -- 5层部门编码

	   T15.estimated_signed_date, -- 预计合同签订时间

	   NVL(T1.JJYJYL,0) AS JJYJYL_QTY, -- 洁具预计用量

	   CURRENT_TIMESTAMP() AS dw_last_update_date,  -- 数据同步时间

       T17.meaning as division_name, -- 事业部名称
	   T3.engineering_big_category , -- 工程大类
       T3.engineering_small_category , -- 工程小类
       T3.sale_org as sale_org_code, -- 销售组织
       T3.sale_org_name, -- 销售组织名称
       CASE WHEN T1.GCLB = '战略子项目' THEN T3.customer_code ELSE '' END AS strategy_customer_code,-- 百强战略客户编码
       CASE WHEN T1.GCLB = '战略子项目' THEN T3.customer_name ELSE '' END AS strategy_cust_name,-- 百强战略客户编码
       SUBSTRING(T3.effective_date,1,10) AS  effective_date, -- 商机有效期

       NVL(T18.meaning,T1.sjlx) AS engineering_big_category_name, -- 工程大类名称
