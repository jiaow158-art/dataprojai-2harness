WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_VRTUL_CHK_INFO_F
SELECT  A.VIRTUAL_CHECK_ID AS VRTUL_CHK_ID --虚拟验收单ID
       ,A.VIRTUAL_CHECK_CODE AS VRTUL_CHK_NUM --虚拟验收单号
       ,A.CONTRACT_CODE AS CNTC_NUM --合同号
       ,A.CONTRACT_NAME AS CNTC_NAME --合同名称
       ,A.BUSINESS_ID AS BUSI_ID --商机ID
       ,A.BUSINESS_CODE AS BUSI_NUM --商机编号
       ,A.BUSINESS_NAME AS BUSI_NAME --商机名称
       ,A.CUSTOMER_CODE AS CUST_NUM --客户编号
       ,A.CUSTOMER_NAME AS CUST_NAME --客户名称
       ,A.SETTLEMENT_MODE AS SETLM_TYPE_CODE --结算方式编码
	   ,T1.MEANING AS SETLM_TYPE_NAME --结算方式描述
       ,A.SETTLEMENT_DATE_START AS SETLM_START_DATE --结算时段开始
       ,A.SETTLEMENT_DATE_END AS SETLM_END_DATE --结算时间段结束
       ,A.WBS AS WBS_NUM --WBS号
       ,A.VIRTUAL_CHECK_STATUS AS VRTUL_ST_CODE --虚拟验收单状态编码
	   ,T2.MEANING AS VRTUL_ST_NAME --虚拟验收单状态描述
       ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	   ,T3.USER_NAME AS CREATE_EMP_NUM --创建人工号
	   ,T3.USER_FULL_NAME AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATED_NAME AS SALE_EMP_NAME --创建人名字[业务员]
       ,A.CREATION_DATE AS CREATE_TIME --创建时间
       ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	   ,T4.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	   ,T4.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名
       ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
       ,A.VERSION_NUM AS VERS_NUM --版本号
       ,A.OVERDUE_FLAG AS IS_OVRDU_CODE --是否逾期编码
	   ,T5.MEANING AS IS_OVRDU_NAME --是否逾期描述
       ,A.ACCEPTANCE_RECOVERY_TIME AS ACC_RCVY_DAY --验收单回收时间（天）
       ,A.ACCEPTANCE_REQUEST_TIME AS ACC_RQST_DAY --验收到请款时间（天）
       ,A.REQUEST_INVOICE_TIME AS RQST_INVC_DAY --请款到开票时间（天）
       ,A.INVOICE_RETURNED_TIME AS INVC_RTRN_DAY --开票到回款时间（天）
       ,CAST(A.PROGRESS_SCALE / 100 AS DECIMAL(38,10)) AS PROG_RATE --进度款比例（%）
       ,CAST(A.SETTLEMENT_SCALE / 100 AS DECIMAL(38,10)) AS SETLM_RATE --结算款比例（%）
       ,CAST(A.RETENTION_SCALE / 100 AS DECIMAL(38,10)) AS RETN_RATE --质保金比例（%）
       ,A.EXPIRATION_DATE AS EXPN_DAY --质保期（天）
       ,A.CUSTOMER_BILLING_METHOD AS OPEN_TKT_TYPE_CODE --客户开票方式编码
	   ,T6.MEANING AS OPEN_TKT_TYPE_NAME --客户开票方式描述
       ,A.CONTRACT_DELIVER_SETTLEMENT AS DELV_SETLM_TYPE_CODE --超合同发货结算方式编码
	   ,T7.MEANING AS DELV_SETLM_TYPE_NAME --超合同发货结算方式描述
       ,A.CHECK_WAY AS CHK_TYPE_CODE --验收单方式编码
	   ,T8.MEANING AS CHK_TYPE_NAME --验收单方式描述
       ,A.CONTRACT_AMOUNT AS CNTC_AMT --合同金额（元）
       ,A.TOTAL_AMOUNT AS TOT_AMT --累计虚拟验收总金额
       ,A.COMPANY_CODE AS CO_CODE --公司代码
       ,A.CHECK_ACCOUNTS_DATE AS CHK_ACCT_DATE --对账起始日
       ,CAST(A.SALES_TAX_RATE / 100 AS DECIMAL(38,10)) AS SALE_TAX_RATE --开票税率
       ,A.CONTACT_NAME AS RECV_NAME --收件人
       ,A.CONTACT_WAY AS RECV_PHONE_NUM --联系电话
       ,A.PROVINCE AS PROV_NAME --省份
       ,A.CITY AS CITY_NAME --城市
       ,A.AREA AS DIST_NAME --区
       ,A.ADDRESS AS ADDR --详细地址
       ,A.SOURCE_SALES_MAN_ID AS SRC_SALE_EMP_ID --源业务员ID,商机继承后赋值
       ,A.SOURCE_SALES_MAN_CODE AS SRC_SALE_EMP_NUM --源业务员编码，商机继承后赋值
       ,A.SOURCE_SALES_MAN_NAME AS SRC_SALE_EMP_NAME --源业务员名称,商机继承后赋值
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_1033 A
  LEFT JOIN T AS T1
    ON A.SETTLEMENT_MODE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'SETTLEMENT'
  LEFT JOIN T AS T2
    ON A.VIRTUAL_CHECK_STATUS = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'SETTLE_VIRTUAL_CHECK_STATUS'
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T3
    ON A.CREATED_BY = T3.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T4
    ON A.LAST_UPDATED_BY = T4.USER_ID
  LEFT JOIN T AS T5
    ON A.OVERDUE_FLAG = T5.LOOKUP_CODE
   AND T5.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T6
    ON A.CUSTOMER_BILLING_METHOD = T6.LOOKUP_CODE
   AND T6.LOOKUP_TYPE = 'CUSTOMER_BILLING_METHOD'
  LEFT JOIN T AS T7
    ON A.CONTRACT_DELIVER_SETTLEMENT = T7.LOOKUP_CODE
   AND T7.LOOKUP_TYPE = 'CONTRACT_DELIVER_SETTLEMENT'
  LEFT JOIN T AS T8
    ON A.CHECK_WAY = T8.LOOKUP_CODE
   AND T8.LOOKUP_TYPE = 'SETTLE_CHECK_WAY'
 WHERE A.DELETE_FLAG = 0
