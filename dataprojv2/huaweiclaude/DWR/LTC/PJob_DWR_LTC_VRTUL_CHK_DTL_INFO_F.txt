WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_VRTUL_CHK_DTL_INFO_F
SELECT   A.VIRTUAL_CHECK_DETAIL_ID AS VRTUL_CHK_DTL_ID --虚拟细表ID
        ,A.VIRTUAL_CHECK_ID AS VRTUL_CHK_ID --虚拟主表ID
        ,A.LINE_ID AS DSPT_LINE_ID --发运单行ID		
        ,A.DEMAND_CODE AS DMAND_NUM --要货单号
        ,A.MATERIAL_CODE AS MATERIAL_NUM --物料编码
        ,A.MATERIAL_NAME --物料名称
        ,A.MATERIAL_SPEC AS DIMENSION --规格型号		
        ,A.UNIT --单位
        ,COALESCE(A.DEMAND_QTY,0) AS DMAND_QTY --要货单需求数
        ,A.PRICE AS PRC --单价
	    ,B.WBS AS PLAN_ITEM_NUM --计划单行号		
        ,A.PICK_CODE AS DELV_NUM --提货单号
	    ,B.DELIVERY_LINE_NO AS DELV_LINE_NUM --提货单行号
        ,A.TRANSPORT_ENTRUST_CODE AS DSPT_NUM --运输委托号
	    ,B.PROJECT_PLAN_NO AS PROJ_PLAN_NUM --项目计划号
	    ,B.LOADING_PLAN_NO AS LOAD_PLAN_NUM --装载计划号		
        ,A.PICK_DATE AS DELV_DATE --提货日期
        ,COALESCE(A.PICK_QTY,0) AS DELV_QTY --提货数量
        ,A.PREDICT_ORIGINAL_DATE AS PRDCT_ORIG_TIME --预计原件收回日期
        ,A.OVERDUE_FLAG  AS IS_OVRDU_CODE --是否逾期编码
	    ,T1.MEANING AS IS_OVRDU_NAME --是否逾期描述
        ,COALESCE(A.TOTAL_VIRTUAL_QTY,0) AS TOT_VRTUL_CHK_QTY --累计虚拟验收数
        ,COALESCE(A.TOTAL_VIRTUAL_AMOUNT,0) AS TOT_VRTUL_CHK_AMT --累计虚拟验收金额
        ,COALESCE(A.TOTAL_FORMAL_QTY,0) AS TOT_FRM_QTY --累计验收数
        ,COALESCE(A.TOTAL_FORMAL_AMOUNT,0) AS TOT_FRM_AMT --累计验收金额
        ,COALESCE(A.TOTAL_OFFICIAL_DIFFERENCE,0) AS DIFF_TOT_QTY --虚拟-正式差值
        ,COALESCE(A.TOTAL_REQUEST_QTY,0) AS TOT_RQST_QTY --累计请款数
        ,COALESCE(A.TOTAL_REQUEST_AMOUNT,0) AS TOT_RQST_AMT --累计请款金额
        ,COALESCE(A.TOTAL_TICKET_QTY,0) AS TOT_TKT_QTY --累计开票数
        ,COALESCE(A.TOTAL_TICKET_AMOUNT,0) AS TOT_TKT_AMT --累计开票金额
        ,COALESCE(A.TOTAL_RECIPIENT_QTY,0) AS TOT_RCPNT_QTY --累计收款数
        ,COALESCE(A.TOTAL_RECIPIENT_AMOUNT,0) AS TOT_RCPNT_AMT --累计收款金额
        ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	    ,T2.USER_NAME AS CREATE_EMP_NUM --创建人工号
	    ,T2.USER_FULL_NAME AS CREATE_EMP_NAME --创建人姓名
        ,A.CREATION_DATE AS CREATE_TIME --创建时间
        ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	    ,T3.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	    ,T3.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名
        ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
        ,A.VERSION_NUM AS VERS_NUM --版本号	  
        ,A.BOX_NUM --包装信息
        ,A.PRODUCT_CHECK_STATUS AS PROD_CHK_ST_CODE --货物验收状态编码
		,CASE WHEN A.PRODUCT_CHECK_STATUS = 'WAIT_OVERDUE' THEN '待办逾期'
		      WHEN A.PRODUCT_CHECK_STATUS = 'WAIT_NOT_OVERDUE' THEN '待办未逾期'
		      WHEN A.PRODUCT_CHECK_STATUS = 'COMPLETE_OVERDUE' THEN '已办逾期'
			  WHEN A.PRODUCT_CHECK_STATUS = 'COMPLETE_NOT_OVERDUE' THEN '已办未逾期'
			   END AS PROD_CHK_ST_NAME --货物验收状态描述
	    ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_DETAIL_1033 A
  LEFT JOIN SDI.SDI_DELIVER_DESPATCH_LINE_1033 B
    ON A.LINE_ID = B.LINE_ID
   AND B.DELETE_FLAG = 0
  LEFT JOIN T AS T1
    ON A.OVERDUE_FLAG = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T2
    ON A.CREATED_BY = T2.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T3
    ON A.LAST_UPDATED_BY = T3.USER_ID
 WHERE A.DELETE_FLAG = 0
