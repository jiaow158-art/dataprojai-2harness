				   ,POSNR

      ,NULL as fwf --服务费（%）
	  ,0.00 as FD --服务费（%）
      ,NULL as JGF --加工费（元/片）
      ,NULL as YF --运费（元/片）
      ,'LTC010-经销商工程价格审批流程(OSS)' AS wf_name --流程名称
	  ,B.YSACB_MX as ycdj --预测成本(元/片)	  
	  ----预测含税业绩=销售额*（100-服务费）	  
	  ,b.SPJG_MX * b.SQSL_MX as ycyj --预测含税业绩额
	  ,B.YSACB_MX * b.SQSL_MX as yccb --预测成本	
	  ,B.YSACBMLL_MX as ycml --预测毛利

 UNION ALL

  LEFT JOIN (SELECT DISTINCT * FROM DM.DM_GL_GROUP_STRUCTURE_D) E
    ON D.SALES_GROUP_CODE = E.SALES_GROUP_CODE

INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_OFR_ORDER_F
SELECT DISTINCT
      a.SQRQ AS APLY_DATE --申请日期
     ,a.GCBBH AS ENGR_CODE --工程报备号
     ,a.KHBM AS CUST_NUM --售达方客编
     ,a.CGFMC AS CUST_NAME --客户简称
     ,a.LSH AS SER_NUM --流水号
     ,a.BJDYXQ AS EXP_DATE --终止日期
     ,a.GCMC AS ENGR_NAME --工程名称
     ,COALESCE(g.EMP_NAME,a.SQR) as APLY_EMP_NUM --业务员
     ,a.ZZCLR AS FNL_EMP_NAME --最终处理人
     ,a.sales_area AS SALE_AREA_NAME --销售区域 
     ,a.BJDH AS OFR_NUM --oss单号
     ,a.WLBM AS MATERIAL_NUM --物料号
     ,a.WLMS AS MATERIAL_NAME --物料描述
	 ,d.PRODUCT_LEVEL_CODE --产品层次
	 ,e.VTEXT AS PRODUCT_STAGE --品类
	 ,d.DIMENSION --规格
     ,d.LENGTH_1 * d.WIDTH /1000000 as UNIT_AREA --单片面积
     ,a.SQSL AS APLY_QTY --申请数量
	 ,a.SQSL * d.LENGTH_1 * d.WIDTH /1000000 as SALE_AREA --销售面积
     ,a.XSE AS ORDER_AMT --销售额
     ,a.ZHSPJG AS APPR_PRC --审批价格
     ,a.fwf AS SRVC_RATE --服务费（%）
     ,a.JGF AS PROC_AMT --加工费（元/片）
     ,a.YF AS SHIP_AMT --运费（元/片）
     ,a.WF_NAME --流程名称
	 ,a.ycdj AS PRDCT_UNIT_COST_AMT --预测成本(元/片)
	 ,a.ycyj AS PRDCT_INCM_AMT --预测含税业绩额
	 ,a.yccb AS PRDCT_COST_AMT --预测成本	
	 ,a.ycml AS PRDCT_GRS_AMT --预测毛利
	 ,case when CURRENT_DATE - interval 1 day <= a.BJDYXQ then '有效期内'
           else '有效期外'
            end as IS_VAILD_DATE --是否有效期内 当前查询日期>终止日期比较，有效期外，当前查询日期<=终止日期比较，有效期内
	 ,a.REQUESTID as REQUEST_ID --流程id
	 ,o.LASTOPERATEDATE as GD_DATE --归档日期
	 ,a.BJDHH AS OFR_ITEM_NUM --OSS报价单行号
	 ,a.DW_MX AS UNIT --计量单位
	 ,a.GCFDJ_MX AS GCF_PRC --工程方价格
	 ,a.SQJG_MX AS APLY_PRC --申请价格
	 ,a.SAP_ORDER_NUM --SAP报价单号
	 ,a.SAP_ORDER_ITEM_NUM --SAP报价单行号
	 ,a.OFR_TYPE_NAME AS OFR_TYPE_NAME --报价单类型
	 ,a.DEALER_CODE --经销商客编
	 ,a.DEALER_NAME --经销商名称
	 ,CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T1 a
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D d -- 物料
    ON a.WLBM = d.MATERIAL_NUM
   AND d.END_DATE = '4712-12-31' 
   AND D.DEL_flag = 'N'
  LEFT JOIN SDI.SDI_T179T_1001 e --产品品类
    ON e.MANDT = d.GROUP_NUM
   AND e.SPRAS = '1'
   AND SUBSTR(d.PRODUCT_LEVEL_CODE,1,4) = e.PRODH  
  LEFT JOIN DM.DM_WF_EMP_INFO_T g
    ON a.SQR = g.EMP_NUM
  LEFT JOIN sdi.sdi_Workflow_Requestbase_1004 o
    ON a.REQUESTID = o.REQUESTID
   AND o.CURRENTNODETYPE = 3	
   AND o.del_flag = 'N'
 WHERE o.LASTOPERATEDATE IS NOT NULL
   AND coalesce(coalesce(a.BJDH,a.SAP_ORDER_NUM),'$$$') <> '$$$'
   AND SUBSTR(a.BJDYXQ,1,4) > '2021'
