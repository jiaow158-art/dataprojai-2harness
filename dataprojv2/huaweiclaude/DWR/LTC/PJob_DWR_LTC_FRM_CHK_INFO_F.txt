WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_FRM_CHK_INFO_F
SELECT  A.FORMAL_CHECK_ID AS FRM_CHK_ID --正式验收单ID
       ,A.FORMAL_CHECK_CODE AS FRM_CHK_NUM --正式验收单号
       ,A.CONTRACT_CODE AS CNTC_NUM --合同号
       ,A.CONTRACT_NAME AS CNTC_NAME --合同名称
       ,A.BUSINESS_ID AS BUSI_ID --商机ID
       ,A.BUSINESS_CODE AS BUSI_NUM --商机编号
       ,A.BUSINESS_NAME AS BUSI_NAME --商机名称
       ,A.CUSTOMER_CODE AS CUST_NUM --客户编号
       ,A.CUSTOMER_NAME AS CUST_NAME --客户名称
       ,A.SETTLEMENT_MODE AS SETLM_MODE_CODE --结算方式编码
	   ,T1.MEANING AS SETLM_MODE_NAME --结算方式描述
       ,A.SETTLEMENT_DATE_START AS SETLM_START_DATE --结算时段开始
       ,A.SETTLEMENT_DATE_END AS SETLM_END_DATE --结算时间段结束
       ,A.WBS AS WBS_NUM --WBS
       ,A.CONTACT_NAME AS CNTCT_NAME --收货人
       ,A.CONTACT_WAY AS CNTCT_PHONE_NUM --联系方式
       ,A.PROVINCE AS PROV_NAME --省
       ,A.CITY AS CITY_NAME --市
       ,A.AREA AS DIST_NAME --区
       ,A.ADDRESS AS ADDR --详细地址
       ,A.CHECK_WAY AS CHK_TYPE_CODE --验收单方式编码
	   ,T2.MEANING AS CHK_TYPE_NAME --验收单方式描述
       ,A.PREDICT_REQUEST_DATE AS PRDCT_RQST_TIME --预计请款日期
       ,A.REQUEST_STATUS AS RQST_ST_CODE --请款状态编码
	   ,T3.MEANING AS RQST_ST_NAME --请款状态描述
       ,A.AFFIRM_RESULT AS AFFR_RSULT_CODE --确认结果编码
	   ,T4.MEANING AS AFFR_RSULT_NAME --确认结果描述
       ,A.AFFIRM_NAME AS AFFR_EMP_NAME --确认人
       ,A.AFFIRM_DATE AS AFFR_TIME --确认日期
       ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	   ,T5.USER_NAME AS CREATE_EMP_NUM --创建人工号
	   ,T5.USER_FULL_NAME AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATION_DATE AS CREATE_TIME --创建时间
       ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	   ,T6.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	   ,T6.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名	   
       ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
       ,A.VERSION_NUM AS VERS_NUM --版本号
       ,A.CREATED_NAME AS SALE_EMP_NAME --创建人名字[业务员]
       ,A.OVERDUE_FLAG AS IS_OVRDU_CODE --是否逾期编码
	   ,T7.MEANING AS IS_OVRDU_NAME --是否逾期描述
       ,A.FORMAL_CHECK_STATUS AS FRM_CHK_ST_CODE --正式验收单状态编码
	   ,T8.MEANING AS FRM_CHK_ST_NAME --正式验收单状态描述
       ,A.FULL_INVOICE_FLAG AS IS_FULL_INVC_CODE --全额开票标识编码
	   ,T9.MEANING AS IS_FULL_INVC_NAME --全额开票标识描述
       ,A.REQUEST_PLAN_FLAG AS IS_RQST_PLAN_CODE --进度款标识编码
	   ,T10.MEANING AS IS_RQST_PLAN_NAME --进度款标识描述
       ,A.REQUEST_PLAN_AMOUNT AS RQST_PLAN_AMT --已请款金额
       ,A.REQUEST_SETTLE_FLAG AS IS_RQST_SET_CODE --结算款标识编码
	   ,T11.MEANING AS IS_RQST_SET_NAME --结算款标识描述
       ,A.REQUEST_QUALITY_FLAG AS IS_RQST_QTY_CODE --质保款标识编码
	   ,T12.MEANING AS IS_RQST_QTY_NAME --质保款标识描述
       ,A.ORIGINAL_FLAG AS IS_ORIG_CODE --是否原件编码
	   ,T13.MEANING AS IS_ORIG_NAME --是否原件描述
       ,A.IS_COLLECT_SOURCE_FILE AS IS_COLL_SRC_FILE_CODE --是否回收原件编码
       ,A.COLLECT_DATE AS COLL_DATE --原件回收时间
       ,A.SOURCE_SALES_MAN_ID AS SRC_SALE_EMP_ID --源业务员ID,商机继承后赋值
       ,A.SOURCE_SALES_MAN_CODE AS SRC_SALE_EMP_NUM --源业务员编码，商机继承后赋值
       ,A.SOURCE_SALES_MAN_NAME AS SRC_SALE_EMP_NAME --源业务员名称,商机继承后赋值
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_FORMAL_CHECK_1033 A
  LEFT JOIN T AS T1
    ON A.SETTLEMENT_MODE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'SETTLEMENT'
  LEFT JOIN T AS T2
    ON A.CHECK_WAY = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'SETTLE_CHECK_WAY'
  LEFT JOIN T AS T3
    ON A.REQUEST_STATUS = T3.LOOKUP_CODE
   AND T3.LOOKUP_TYPE = 'SETTLE_REQUEST_PAYOUT_STATUS'
  LEFT JOIN T AS T4
    ON A.AFFIRM_RESULT = T4.LOOKUP_CODE
   AND T4.LOOKUP_TYPE = 'SETTLE_FORMAL_CHECK_AFFIRM'
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T5
    ON A.CREATED_BY = T5.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T6
    ON A.LAST_UPDATED_BY = T6.USER_ID
  LEFT JOIN T AS T7
    ON A.OVERDUE_FLAG = T7.LOOKUP_CODE
   AND T7.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T8
    ON A.FORMAL_CHECK_STATUS = T8.LOOKUP_CODE
   AND T8.LOOKUP_TYPE = 'SETTLE_FORMAL_CHECK_STATUS'
  LEFT JOIN T AS T9
    ON A.FULL_INVOICE_FLAG = T9.LOOKUP_CODE
   AND T9.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T10
    ON A.REQUEST_PLAN_FLAG = T10.LOOKUP_CODE
   AND T10.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T11
    ON A.REQUEST_SETTLE_FLAG = T11.LOOKUP_CODE
   AND T11.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T12
    ON A.REQUEST_QUALITY_FLAG = T12.LOOKUP_CODE
   AND T12.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T13
    ON A.ORIGINAL_FLAG = T13.LOOKUP_CODE
   AND T13.LOOKUP_TYPE = 'YES_OR_NO'
 WHERE A.DELETE_FLAG = 0
