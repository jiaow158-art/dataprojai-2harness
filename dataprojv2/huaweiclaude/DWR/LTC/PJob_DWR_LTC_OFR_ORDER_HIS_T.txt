WITH T1 AS( --报价单
SELECT CASE WHEN T9.X_OD_SUBTYPE = 'Third Party Project Quote' THEN PAY.OU_NUM
            WHEN T9.X_OD_SUBTYPE = 'Dealers Project Quote' THEN AC.OU_NUM
			 END AS CUST_NUM --客编
	  ,REP.PROJ_REP_CODE AS BUSI_NUM --商机号
      ,T1.PROD_ID AS MATERIAL_ID --物料ID
      ,PROD.NAME AS MATERIAL_NUM --物料编码
	  ,PROD.DESC_TEXT AS MATERIAL_NAME --物料名称
      ,T1.X_APP_PRI AS APPR_AMT --审批价格
	  ,T1.X_CONFIRM_AMOUNT AS ORDER_QTY --下单数量
	  ,COALESCE(T1.X_CONFIRM_AMOUNT,0) * COALESCE(T1.X_UNIT_AREA,0) AS ORDER_AREA --报价单面积
	  ,T1.CREATED AS CREATE_TIME --价审创建日期
	  ,ROW_NUMBER() OVER(PARTITION BY CASE WHEN T9.X_OD_SUBTYPE = 'Third Party Project Quote' THEN PAY.OU_NUM
                                           WHEN T9.X_OD_SUBTYPE = 'Dealers Project Quote' THEN AC.OU_NUM
			                                END,REP.PROJ_REP_CODE,T1.PROD_ID ORDER BY T1.CREATED DESC) AS RN
  FROM SDI.SDI_S_ORDER_1002 T9
 INNER JOIN SDI.SDI_S_ORDER_ITEM_1002 T1
    ON T1.ORDER_ID = T9.ROW_ID
 INNER JOIN SDI.SDI_CX_PROJ_REP_1002 REP
    ON T9.X_PROJ_REP_ID = REP.ROW_ID
  LEFT JOIN SDI.SDI_S_ORG_EXT_1002 AC
    ON T9.ACCNT_ID = AC.PAR_ROW_ID
  LEFT JOIN SDI.SDI_S_ORG_EXT_1002 PAY
    ON T9.X_PAYER_ID = PAY.ROW_ID
  LEFT JOIN SDI.SDI_S_PROD_INT_1002 PROD
    ON T1.PROD_ID = PROD.ROW_ID
 WHERE T9.STATUS_CD IN('已完成','部分转单','全部转单')
   AND T1.STATUS_CD IN('新建')
   AND T9.X_OD_SUBTYPE IN('Third Party Project Quote','Dealers Project Quote'))
,T2 AS( --订单
SELECT CASE WHEN T9.X_OD_SUBTYPE = 'Third Party Project Order' THEN PA.OU_NUM
            WHEN T9.X_OD_SUBTYPE = 'Dealers Project Order' THEN AC.OU_NUM
			 END AS CUST_NUM --客编
	  ,REP.PROJ_REP_CODE AS BUSI_NUM --商机号
      ,T1.PROD_ID AS MATERIAL_ID --物料ID
      ,SUM(COALESCE(T1.X_QTY_REQ,0)) AS TURN_ORDER_QTY --累计已转单数量
	  ,SUM(COALESCE(T1.X_QTY_REQ,0) * COALESCE(T1.X_UNIT_AREA,0)) AS TURN_ORDER_AREA --累计已转单面积
  FROM SDI.SDI_S_ORDER_1002 T9
 INNER JOIN SDI.SDI_S_ORDER_ITEM_1002 T1
    ON T1.ORDER_ID = T9.ROW_ID
 INNER JOIN SDI.SDI_CX_PROJ_REP_1002 REP
    ON T9.X_PROJ_REP_ID = REP.ROW_ID
  LEFT JOIN SDI.SDI_S_ORG_EXT_1002 AC
    ON T9.ACCNT_ID = AC.PAR_ROW_ID
  LEFT JOIN SDI.SDI_S_ORG_EXT_1002 PA
    ON T9.X_PRJ_ACCT_ID = PA.PAR_ROW_ID
 WHERE T9.X_OD_TYPE = 'Purchase Order'
   AND T9.STATUS_CD NOT IN('无效','新建','提交失败','取消')
   AND T1.X_ITEM_STATUS_CD NOT IN('新建','无效','评审退回','取消')
   AND T9.X_OD_SUBTYPE IN('Third Party Project Order','Dealers Project Order')
 GROUP BY CASE WHEN T9.X_OD_SUBTYPE = 'Third Party Project Order' THEN PA.OU_NUM
            WHEN T9.X_OD_SUBTYPE = 'Dealers Project Order' THEN AC.OU_NUM
			 END --客编
	  ,REP.PROJ_REP_CODE --商机号
      ,T1.PROD_ID --物料ID
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_OFR_ORDER_HIS_T
SELECT T1.CUST_NUM --客编
	  ,T1.BUSI_NUM --商机号
      ,T1.MATERIAL_ID --物料ID
      ,T1.MATERIAL_NUM --物料编码
	  ,T1.MATERIAL_NAME --物料名称
	  ,MAX(CASE WHEN T1.RN = 1 THEN COALESCE(T1.APPR_AMT,0) ELSE 0 END) AS LST_APPR_AMT --最后一次审批价格
	  ,MAX(CASE WHEN T1.RN = 1 THEN T1.CREATE_TIME END) AS LST_CREATE_TIME --最后一次价审创建日期
	  ,SUM(COALESCE(T1.ORDER_QTY,0)) AS TOT_ORDER_QTY --报价单总下单数量
	  ,SUM(COALESCE(T1.ORDER_AREA,0)) AS TOT_ORDER_AREA --报价单总下单面积
	  ,T2.TURN_ORDER_QTY --累计已转单数量
	  ,T2.TURN_ORDER_AREA --累计已转单面积
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T1
 INNER JOIN T2
    ON T1.CUST_NUM = T2.CUST_NUM
   AND T1.BUSI_NUM = T2.BUSI_NUM
   AND T1.MATERIAL_ID = T2.MATERIAL_ID
 GROUP BY T1.CUST_NUM --客编
	  ,T1.BUSI_NUM --商机号
      ,T1.MATERIAL_ID --物料ID
      ,T1.MATERIAL_NUM --物料编码
	  ,T1.MATERIAL_NAME --物料名称
	  ,T2.TURN_ORDER_QTY --累计已转单数量
	  ,T2.TURN_ORDER_AREA --累计已转单面积
