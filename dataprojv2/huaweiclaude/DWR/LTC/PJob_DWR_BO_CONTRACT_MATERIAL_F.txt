--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with DM_OTD_SALES_ORDER_DET_SW_DATA AS(
SELECT  t.dh as business_code, -- 商机
	    t.material_num , -- 物料编码
	    t.maid as initial_material_id, -- 原始物料ID
	    SUM(t.order_qty) AS order_qty,-- 已转计划单数量
	    SUM(t.res_stock_qty) AS res_stock_qty,-- 留货数量/已确认数量
	    SUM(t.prepare_stock_qty) AS prepare_stock_qty,-- 已提货单数量
	    SUM(t.stockout_qty) AS stockout_qty-- 已出库数量	
FROM DM_SW.DM_OTD_SALES_ORDER_DET_SW t
where t.maid <> ''
and t.cancal_code = ''
group by t.dh ,
		 t.material_num ,
		 t.maid 
),
TMP_DATA AS(
	SELECT SUBSTRING(CAST(T0.apply_date AS STRING),1,10) AS business_apply_date, -- 申请日期
	       T0.business_id ,  -- 商机ID
		   T0.business_code , -- 商机编码
		   T0.business_name , -- 商机名称
		   T01.project_id ,  -- 项目ID
		   T01.project_code ,  -- 项目编码
		   T1.contract_id ,  -- 合同ID
		   T1.contract_code,  -- 合同编码
		   T1.contract_name , -- 合同名称
		   T1.contract_status , -- 合同状态
		   T2.meaning as contract_status_name , -- 合同状态名称
		   T1.receipts_status , -- 单据状态
		   T3.meaning as receipts_status_name , -- 单据状态名称
		   T4.initial_material_id,  -- 原始物料ID
		   T4.material as material_num, -- 物料编码
		   T4.material_description as material_name, -- 物料名称
		   T01.project_type , -- 工程类型
	       T01.is_allograph,  -- 是否代签
	       T01.engineering_sale_org, -- 工程方销售组织
	       T01.engineering_customer_code, -- 工程方客编
	       T01.engineering_customer_name, -- 工程方客编名称
	       SUBSTRING(CAST(T01.apply_date AS STRING),1,10) AS PROJECT_apply_date,  -- 项目申请日期
	       T01.delivery_period,  -- 预计供货周期
	       T01.quotation_results, -- 报价反馈
	       T01.receipts_status AS PROJECT_receipts_status, -- 单据状态
	       T03.meaning AS PROJECT_receipts_status_name, -- 单据状态名称
	       '06' as distribution_channel,  -- 分销渠道
	       T01.is_palletize, -- 是否打托
		   T011.delivery,-- T01.delivery,  -- 配送方式
		   T02.meaning AS delivery_NAME, -- 配送方式
		   T7.meaning AS freight_payers, -- T01.freight_payers,  -- 物流支付方
		   T011.shipping_address,--T01.shipping_address,  -- 配送地址
		   T011.detailed_address, -- T01.detailed_address,  -- 详细地址
		   T4.unit , -- 单位
		   T4.weight_volume, -- 重量/体积
		   T4.delivery_factory, -- 交付工厂
		   T4.technical_standard, -- 技术标准
		   NVL(T4.review_quotation,0) AS review_quotation, -- 评审报价
		   NVL(T4.product_naked_price,0) as product_naked_price, -- 产品单价
		   NVL(T4.review_dosage,0) AS review_dosage, -- 评审用量
		   NVL(T4.service_rate,0) AS service_rate, -- 综合服务费率
		   T1.new_flag, -- 合同版本标识
		   T1.sale_org , -- 销售渠道
		   T4.material_group2_code , -- 物料组2编码
		   T4.material_group2 as material_group2_name, -- 物料组2名称
		   T4.material_group4_code , -- 物料组4编码
		   T4.material_group4 , -- 物料组4名称
		   T4.contract_quantity, -- 合同物料数量
		   nvl(t6.order_qty,0) as order_qty,-- 已转计划单数量
		   nvl(t6.res_stock_qty,0) as res_stock_qty,-- 留货数量/已确认数量
		   nvl(t6.prepare_stock_qty,0) as prepare_stock_qty,-- 已提货单数量
		   nvl(t6.stockout_qty,0) as stockout_qty,-- 已出库数量	
		   CURRENT_TIMESTAMP() AS dw_last_update_date , -- 数据同步时间
		   SUBSTRING(CAST(T0.apply_date AS STRING),1,7) AS PERIOD_ID_M -- 分区月
	FROM SDI.SDI_BUSINESS_OPPORTUNITY_1033 T0
	LEFT JOIN SDI.SDI_PROJECT_REVIEW_JJJXS_1033 T01  -- 项目评审
	ON T0.business_code = T01.business_code 
	AND T01.delete_flag = 0
	LEFT JOIN SDI.SDI_PROJECT_REVIEW_JJJXS_LOGISTICS_1033 T011 -- 物流信息
	ON T01.project_id = T011.project_id 
	AND T011.delete_flag = 0
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T02  -- RTM LOOKUP 表
	ON T011.delivery = T02.lookup_code 
	AND T02.lookup_type = 'DELIVERY'   --  配送方式
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T03
	ON T01.receipts_status = T03.lookup_code 
	AND T03.lookup_type = 'SAMPLE_GENERAL_STATUS' -- 单据状态
	LEFT JOIN SDI.SDI_CONTRACT_1033 T1
	ON T0.business_code = T1.business_code 
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T2  -- RTM LOOKUP 表
	ON T1.contract_status = T2.lookup_code 
	AND T2.lookup_type = 'CONTRACT_SEAL_STATUS'   --  合同状态
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T3  -- RTM LOOKUP 表
	ON T1.receipts_status = T3.lookup_code 
	AND T3.lookup_type = 'RECIPS_CODE'   --  单据状态
	LEFT JOIN SDI.SDI_CONTRACT_MATERIAL_1033 T4 -- 合同物料
	ON T1.contract_id = T4.contract_id 
	AND T4.delete_flag = 0
	LEFT JOIN DWIMD.dwi_md_data_material_sales_t T5 -- 物料销售主数据
	ON T1.sale_org = T5.sales_org_code 
	AND T4.material = T5.material_num 
	AND T5.distribution_channel_code = '06'
	AND T5.end_date = '4712-12-31'
	LEFT JOIN DM_OTD_SALES_ORDER_DET_SW_DATA T6
	ON T0.business_code = T6.business_code
	AND T4.material = T6.material_num
	AND T4.initial_material_id = T6.initial_material_id
	LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T7  -- RTM LOOKUP 表
	ON T011.freight_payers = T7.lookup_code 
	AND T7.lookup_type = 'SHIPPING_METHODS'   --  物流支付方
	where T0.division = 'BATHROOM_DIVISION'  -- 卫浴事业部
	AND T1.new_flag = 'NEW'
	AND T1.delete_flag = 0
	and T1.receipts_status = 'PASS' -- RECIPS_CODE
	and T0.delete_flag = 0
	and T4.initial_material_id is not null
	AND NVL(T4.material_status,'ISNULL') <> 'STOP' -- 排除 停用
	--AND SUBSTRING(CAST(T0.apply_date AS STRING),1,7) >= '${PERIOD_ID_M}' -- 传昨天所在月份的上月
)
insert overwrite table DWRLTC.DWR_BO_CONTRACT_MATERIAL_F PARTITION(PERIOD_ID_M)
SELECT t.*
FROM TMP_DATA t;
