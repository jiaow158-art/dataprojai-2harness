WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
),TT1 AS(
SELECT PROJECT_ID
      ,NEW_PROJECT_CONTRACT_CATEGORY
  FROM SDI.SDI_PROJECT_REVIEW_1033
LATERAL VIEW EXPLODE(SPLIT(PROJECT_CONTRACT_CATEGORY,',')) TBL AS NEW_PROJECT_CONTRACT_CATEGORY
),TTT1 AS(
SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS CNTC_CAT_NAME 
  FROM TT1 AS A
  LEFT JOIN T
    ON A.NEW_PROJECT_CONTRACT_CATEGORY = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'PRODUCT_CATEGORY'
 GROUP BY A.PROJECT_ID
),TT2 AS(
SELECT PROJECT_ID
      ,NEW_PROJECT_COOPERATION
  FROM SDI.SDI_PROJECT_REVIEW_1033

SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS CNTC_COOP_NAME
  FROM TT2 AS A
  LEFT JOIN T
    ON A.NEW_PROJECT_COOPERATION = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'EXPECTED_COOPERATION_TYPE'
 GROUP BY A.PROJECT_ID
),TT3 AS(
SELECT PROJECT_ID
      ,NEW_PRODUCT_CATEGORY
  FROM SDI.SDI_PROJECT_REVIEW_1033
LATERAL VIEW EXPLODE(SPLIT(PRODUCT_CATEGORY,',')) TBL AS NEW_PRODUCT_CATEGORY
),TTT3 AS(
SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS PROD_CAT_NAME
  FROM TT3 AS A
  LEFT JOIN T
    ON A.NEW_PRODUCT_CATEGORY = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'PRODUCT_CATEGORY'
 GROUP BY A.PROJECT_ID
),TT4 AS(
SELECT PROJECT_ID
      ,NEW_PROJECT_COOPERATION
  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_1033
LATERAL VIEW EXPLODE(SPLIT(PROJECT_COOPERATION,',')) TBL AS NEW_PROJECT_COOPERATION
),TTT4 AS(
SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS CNTC_COOP_NAME
  FROM TT4 AS A
  LEFT JOIN T
    ON A.NEW_PROJECT_COOPERATION = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'EXPECTED_COOPERATION_TYPE'

),TT5 AS(
SELECT PROJECT_ID
      ,NEW_ORDERS_PUSH_METHOD
  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_1033

),TTT5 AS(
SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS ORDER_PUSH_METH_NAME
  FROM TT5 AS A
  LEFT JOIN T
    ON A.NEW_ORDERS_PUSH_METHOD = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'ORDER_PUSH_METHOD'

),TT6 AS(
SELECT PROJECT_ID
      ,NEW_CONTRACT_SIGNING_METHOD
  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_1033
LATERAL VIEW EXPLODE(SPLIT(CONTRACT_SIGNING_METHOD,',')) TBL AS NEW_CONTRACT_SIGNING_METHOD
),TTT6 AS(
SELECT A.PROJECT_ID
      ,CONCAT_WS(',',COLLECT_SET(T.MEANING)) AS CNTR_SIGN_METH_NAME
  FROM TT6 AS A
  LEFT JOIN T
    ON A.NEW_CONTRACT_SIGNING_METHOD = T.LOOKUP_CODE
   AND T.LOOKUP_TYPE = 'CONTRACT_SIGNING_METHOD'
 GROUP BY A.PROJECT_ID

  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_1033

SELECT PROJECT_ID
      ,NEW_PRODUCT_CATEGORY
  FROM SDI.SDI_PROJECT_REVIEW_JJJXS_1033

SELECT A.PROJECT_ID
