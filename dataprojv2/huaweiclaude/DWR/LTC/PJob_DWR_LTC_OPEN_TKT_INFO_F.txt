WITH T AS(
SELECT LOOKUP_CODE
      ,LOOKUP_TYPE
	  ,MEANING
  FROM SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
 WHERE T1.DELETE_FLAG = 0
)
INSERT OVERWRITE TABLE DWRLTC.DWR_LTC_OPEN_TKT_INFO_F
SELECT  A.OPEN_TICKET_ID AS OPEN_TKT_ID --开票单ID
       ,A.OPEN_TICKET_CODE AS OPEN_TKT_NUM --开票单号
       ,A.CONTRACT_CODE AS CNTC_NUM --合同号
       ,A.CONTRACT_NAME AS TNTC_NAME --合同名称
       ,A.BUSINESS_ID AS BUSI_ID --商机ID
       ,A.BUSINESS_CODE AS BUSI_NUM --商机编号
       ,A.BUSINESS_NAME AS BUSI_NAME --商机名称
       ,A.CUSTOMER_CODE AS CUST_NUM --客户编号
       ,A.CUSTOMER_NAME AS CUST_NAME --客户名称
       ,CASE WHEN COALESCE(A.SETTLEMENT_MODE,'') <> '' THEN A.SETTLEMENT_MODE ELSE NULL END AS SETLM_TYPE_CODE --结算方式编码
	   ,T1.MEANING AS SETLM_TYPE_NAME --结算方式描述
       ,A.SETTLEMENT_DATE_START AS SETLM_START_DATE --结算时段开始
       ,A.SETTLEMENT_DATE_END AS SETLM_END_DATE --结算时间段结束
       ,A.WBS AS WBS_NUM --WBS
       ,A.REQUEST_PAYOUT_PAY AS RQST_PAYOT_TYPE_CODE --请款类型编码
	   ,T2.MEANING AS RQST_PAYOT_TYPE_NAME --请款类型描述
       ,A.CUSTOMER_TICKET_MODE AS OPEN_TKT_TYPE_CODE --客户开票方式编码
	   ,T3.MEANING AS OPEN_TKT_TYPE_NAME --客户开票方式描述
       ,A.INVOICE_TYPE AS INVC_TYPE_CODE --发票类型编码
	   ,T4.MEANING AS INVC_TYPE_NAME --发票类型描述
       ,COALESCE(A.ACCRUED_TICKET_AMOUNT,0) AS ACR_TKT_AMT --应计开票金额
       ,COALESCE(A.CURRENT_TICKET_AMOUNT,0) AS CUR_TKT_AMT --本次开票金额
       ,COALESCE(A.URPLUS_TICKET_AMOUNT,0) AS RPLS_TKT_AMT --剩余开票金额
       ,COALESCE(A.TOTAL_RECIPIENT_AMOUNT,0) AS TOT_RCPNT_AMT --累计收款金额
       ,COALESCE(A.UN_CHECK_AMOUNT,0) AS UN_CHK_AMT --未钩稽金额
       ,A.PREDICT_RECIPIENT_DATE AS PRDCT_RCPNT_DATE --预计收款日期
       ,A.INVOICE_TITLE AS INVC_TIL --开票抬头
       ,A.TAXPAYER_IDENTIFICATION AS TAX_NUM --纳税识别号
       ,A.BANK_NAME --银行
       ,A.BANK_ACCOUNT AS BANK_ACCT_NUM --银行账号
       ,A.INVOICE_ADDRESS AS BANK_ADDR --开户行地址
       ,A.VERIFICATION_STATUS AS VERIF_ST_CODE --核销状态编码
	   ,CASE WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_NO_COMMIT' THEN '未核销'
             WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_SOME_COMMIT' THEN '部分认领'
			 WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_COMMIT' THEN '已认领'
			 WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_F' THEN '未核销'
			 WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_B' THEN '部分核销'
			 WHEN A.VERIFICATION_STATUS = 'WRITE_OFF_STATUS_T' THEN '已核销'
			 WHEN A.VERIFICATION_STATUS = 'BACK_VERIFICATION' THEN '反核销'
			 WHEN A.VERIFICATION_STATUS = 'REPORT_COST' THEN '报备费'
			 WHEN A.VERIFICATION_STATUS = 'DELAY_COST' THEN '延期费'
			 WHEN A.VERIFICATION_STATUS = 'ARBITRATION_EARNEST_MONEY' THEN '仲裁诚意金'
			 WHEN A.VERIFICATION_STATUS = 'ARBITRATION_DEPOSIT' THEN '仲裁保证金'
			 WHEN A.VERIFICATION_STATUS = 'TRANSFER_FEE' THEN '转让费'
			 WHEN A.VERIFICATION_STATUS = 'MARGIN_BID' THEN '投标保证金'
			 WHEN A.VERIFICATION_STATUS = 'MARGIN_QUALITY' THEN '质量保证金'
			 WHEN A.VERIFICATION_STATUS = 'PERFORMANCE_BOND' THEN '履约保证金'
			 WHEN A.VERIFICATION_STATUS = 'TOMARGIN_QUALITY' THEN '转质量保证金'
			 WHEN A.VERIFICATION_STATUS = 'TOPERFORMANCE_BOND' THEN '转履约保证金'
		      END AS VERIF_ST_NAME --核销状态描述
       ,A.COMMIT_STATUS AS CMMT_ST_CODE --回款认领状态编码
	   ,T5.MEANING AS CMMT_ST_NAME --回款认领状态描述
       ,COALESCE(A.COMMIT_TOTAL_AMOUNT,0) AS TOT_CMMT_AMT --回款认领金额累计
       ,A.RECEIVED_BUSINESS_TICKET AS IS_RECV_TKT_CODE --已收商票编码
	   ,T6.MEANING AS IS_RECV_TKT_NAME --已收商票描述
       ,COALESCE(A.BUSINESS_TICKET_AMOUNT,0) AS BUSI_TKT_AMT --商票金额
       ,A.SALESMAN_PICK_FLAG AS IS_SLF_PICK_CODE --是否业务员自提编码
	   ,T7.MEANING AS IS_SLF_PICK_NAME --是否业务员自提描述
       ,A.SEND_DATE --发出日期
       ,A.EXPRESS_NUMBER AS EXP_NUM --快递单号
       ,A.EXPRESS_NAME AS RECV_NAME --收件人
       ,A.EXPRESS_PHONE AS RECV_PHONE_NUM --联系电话
       ,A.PROVINCE AS PROV_NAME --省
       ,A.CITY AS CITY_NAME --市
       ,A.AREA AS DIST_NAME --区
       ,A.ADDRESS AS ADDR --详细地址
       ,A.COURIER_NUM AS COUR_NUM --回执快递单号
       ,A.RECEIVED_DATE AS RECV_DATE --收到时间
       ,A.RECEIPT_SIGN_FLAG AS IS_RCPT_SIGN_CODE --回执已签收编码
	   ,T8.MEANING AS IS_RCPT_SIGN_NAME --回执已签收描述
       ,A.RECEIVE_REASON AS NO_RECV_RSN --未收到原因
       ,CASE WHEN A.PRO_TICKET_FLAG = 'X' THEN 'N' ELSE A.PRO_TICKET_FLAG END AS IS_PRO_TKT_CODE --预开票标识编码
	   ,T9.MEANING AS IS_PRO_TKT_NAME --预开票标识描述
       ,A.PROCESS_STATUS AS PRCS_ST_CODE --开票单据状态编码
	   ,CASE WHEN A.PROCESS_STATUS = 'DRAFT' THEN '草稿'
             WHEN A.PROCESS_STATUS = 'APPROVAL' THEN '审批中'
		     WHEN A.PROCESS_STATUS = 'ALLOW' THEN '审批通过'
		     WHEN A.PROCESS_STATUS = 'FEEDBACK' THEN '结果反馈'
		     WHEN A.PROCESS_STATUS = 'COMPLETE' THEN '完成'
		     WHEN A.PROCESS_STATUS = 'CLOSE' THEN '关闭'
		     WHEN A.PROCESS_STATUS = 'PASS' THEN '通过'
		     WHEN A.PROCESS_STATUS = 'REJECT' THEN '驳回'
		     WHEN A.PROCESS_STATUS = 'FAIL' THEN '失败'
		     WHEN A.PROCESS_STATUS = 'REFUSE' THEN '拒绝'
		     WHEN A.PROCESS_STATUS = 'DISCARD_WAIT' THEN '待作废'
		     WHEN A.PROCESS_STATUS = 'DISCARD' THEN '作废'
			  END AS PROCESS_STATUS_NAME --开票状态描述
       ,A.CREATED_BY AS CREATE_EMP_ID --创建人ID
	   ,T10.USER_NAME AS CREATE_EMP_NUM --创建人工号
	   ,T10.USER_FULL_NAME AS CREATE_EMP_NAME --创建人姓名
       ,A.CREATION_DATE AS CREATE_TIME --创建时间
       ,A.LAST_UPDATED_BY AS UPDATE_EMP_ID --更新人ID
	   ,T11.USER_NAME AS UPDATE_EMP_NUM --更新人工号
	   ,T11.USER_FULL_NAME AS UPDATE_EMP_NAME --更新人姓名
       ,A.LAST_UPDATE_DATE AS UPDATE_TIME --更新时间
       ,A.VERSION_NUM AS VERS_NUM --版本号
       ,A.OVERDUE_FLAG AS IS_OVRDU_CODE --是否逾期编码
	   ,T12.MEANING AS IS_OVRDU_NAME --是否逾期描述
       ,A.RECEIPT_MAILBOX AS RCPT_EMAIL --电子收票邮箱
       ,A.RECEIPT_PHONE AS RCPT_PHONE_NUM --电子收票电话
       ,A.INVOICE_REMARKS AS INVC_REMARK --发票备注
       ,A.INVOICE_PHONE AS INVC_PHONE_NUM --开票电话
       ,A.COMPANY_CODE AS CO_CODE --公司代码
       ,A.INVOICE_AX_TYPE AS INVC_AX_TYPE_CODE --开票汇总方式编码
	   ,T13.MEANING AS INVC_AX_TYPE_NAME --开票汇总方式描述
       ,A.INVOICE_UNIT AS INVC_UNIT_CODE --产品开票单位编码
	   ,T14.MEANING AS INVC_UNIT_NAME --产品开票单位描述
       ,A.SALES_TAX_RATE AS TKT_TAX_RATE --开票税率
       ,COALESCE(A.FEE_TICKET_AMOUNT,0) AS TKT_AMT --本次费用开票金额
       ,COALESCE(A.TOTAL_TICKET_AMOUNT,0) AS TOT_TKT_AMT --开票金额合计
       ,A.REQUEST_PAYOUT_IDS AS RQST_PAYOT_ID --请款主表ID
       ,A.COST_CENTER_CODE AS COST_CTR_CODE --成本中心编码
       ,A.COST_CENTER_NAME AS COST_CTR_NAME --成本中心名称
       ,A.QTPJY_CODE AS OTH_DOC_EMP_NUM --其他单据员编码
       ,A.QTPJY_NAME AS OTH_DOC_EMP_NAME --其他单据员名称
       ,A.ZCBDJY_CODE AS OFF_DOC_EMP_NUM --总裁办单据员编码
       ,A.ZCBDJY_NAME AS OFF_DOC_EMP_NAME --总裁办单据员名称
       ,A.JJF_CODE AS SHIP_EMP_NUM --寄件方联系人编码
       ,A.JJF_NAME AS SHIP_EMP_NAME --寄件方联系人名称
       ,A.JJF_PHONE AS SHIP_PHONE_NUM --寄件方联系人电话
       ,A.KD_OA_ID AS OA_RQST_ID --快递OA流程ID
       ,A.CAN_DUE AS IS_DUE_CODE --能进行开票处理标识编码
	   ,T15.MEANING AS IS_DUE_NAME --能进行开票处理标识秒速
       ,A.SUPPLEMENTARY_NOTE AS SPLMN_NOTE --补充说明
       ,A.SOURCE_SALES_MAN_ID AS SRC_SALE_EMP_ID --源业务员ID,商机继承后赋值
       ,A.SOURCE_SALES_MAN_CODE AS SRC_SALE_EMP_NUM --源业务员编码，商机继承后赋值
       ,A.SOURCE_SALES_MAN_NAME AS SRC_SALE_EMP_NAME --源业务员名称,商机继承后赋值
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_OPEN_TICKET_1033 A
  LEFT JOIN T AS T1 
    ON A.SETTLEMENT_MODE = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'SETTLEMENT'
  LEFT JOIN T AS T2
    ON A.REQUEST_PAYOUT_PAY = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'SETTLE_ASK_PAYMENT'
  LEFT JOIN T AS T3
    ON A.REQUEST_PAYOUT_PAY = T3.LOOKUP_CODE
   AND T3.LOOKUP_TYPE = 'CUSTOMER_BILLING_METHOD'
  LEFT JOIN T AS T4
    ON A.INVOICE_TYPE = T4.LOOKUP_CODE
   AND T4.LOOKUP_TYPE = 'INVOICE_TYPE'
  LEFT JOIN T AS T5
    ON A.COMMIT_STATUS = T5.LOOKUP_CODE
   AND T5.LOOKUP_TYPE = 'SETTLEMENT_VERIFICATION_COMMIT_STATUS'
  LEFT JOIN T AS T6
    ON A.RECEIVED_BUSINESS_TICKET = T6.LOOKUP_CODE
   AND T6.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T7
    ON A.SALESMAN_PICK_FLAG = T7.LOOKUP_CODE
   AND T7.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T8
    ON A.RECEIPT_SIGN_FLAG = T8.LOOKUP_CODE
   AND T8.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T9
    ON REPLACE(A.PRO_TICKET_FLAG,'X','N') = T9.LOOKUP_CODE
   AND T9.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T10
    ON A.CREATED_BY = T10.USER_ID
  LEFT JOIN SDI.SDI_BASE_USERS_1029 T11
    ON A.LAST_UPDATED_BY = T11.USER_ID
  LEFT JOIN T AS T12
    ON A.OVERDUE_FLAG = T12.LOOKUP_CODE
   AND T12.LOOKUP_TYPE = 'YES_OR_NO'
  LEFT JOIN T AS T13
    ON A.INVOICE_AX_TYPE = T13.LOOKUP_CODE
   AND T13.LOOKUP_TYPE = 'INVOICE_AX_TYPE'
  LEFT JOIN T AS T14
    ON A.INVOICE_UNIT = T14.LOOKUP_CODE
   AND T14.LOOKUP_TYPE = 'INVOICE_UNIT'
  LEFT JOIN T AS T15
    ON A.CAN_DUE = T15.LOOKUP_CODE
   AND T15.LOOKUP_TYPE = 'YES_OR_NO'
 WHERE A.DELETE_FLAG = 0
