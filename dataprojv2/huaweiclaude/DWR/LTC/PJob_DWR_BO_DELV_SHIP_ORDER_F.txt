      ,COALESCE(COALESCE(G.DP_ACTUAL_QUANTITY,G.TRAN_NUM),0) AS ACTL_SHIP_QTY --实际开单
      ,COALESCE(G.TRAN_NUM,0) - COALESCE(COALESCE(G.DP_ACTUAL_QUANTITY,G.TRAN_NUM),0) RSDU_SHIP_QTY --出货数剩余未开单数
	  ,COALESCE(G.PROD_NUM,0) - COALESCE(G.TRAN_NUM,0) AS RSDU_DMAND_QTY --要货单剩余未开单数
	  ,0 AS INV_QTY --当前可用库存数
      ,0 AS CMPRN_PRC --综合价
      ,0 AS BARE_PRC	--裸价
      ,NULL AS RTN_ST_CODE --退货状态编码
	  ,NULL AS RTN_ST_NAME --退货状态描述
	  ,G.DP_ORDER_ID AS OFR_ORDER_ID --报价单ID
	  ,G.DP_ORDER_ITEM_ID AS OFR_ORDER_ITEM_ID --报价单行ID
	  ,'OSS' AS DATA_SRC_CODE --数据来源编码	  
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,NULL AS PROJ_ID --项目评审行ID
	  ,NULL AS CNTC_ITEM_ID --合同行ID
	  ,SO.ORDER_NUM AS OFR_ORDER_NUM --报价单号
	  ,1 AS IS_HIDE_CODE --是否前端展示
  FROM SDI.SDI_CX_PURCH_1002 A
  LEFT JOIN SDI.SDI_CX_PROJ_REP_1002 B
    ON A.REP_PRO_ID = B.ROW_ID
  LEFT JOIN T1
    ON A.STATUS = T1.NAME
   AND T1.TYPE = 'DP_PURCHASE_STATUS'
  LEFT JOIN SDI.SDI_S_ORG_EXT_1002 C
    ON B.ACCOUNT_NUM = C.OU_NUM
  LEFT JOIN T2
    ON A.DP_ACCOUNT_ORDER_NUMBER = T2.PARTY_UID
  LEFT JOIN SDI.SDI_CX_SALES_ORG_1002 D
    ON B.DP_SALES_GROUP_CODE = D.SALES_ORG_CD
  LEFT JOIN SDI.SDI_CX_SALES_CHAN_1002 E
    ON B.DP_SALES_CHANNEL_CODE = E.SALES_CHAN_CD
  LEFT JOIN SDI.SDI_CX_SALES_AREA_1002 F
    ON C.ROW_ID = F.ACCOUNT_ID
   AND D.ROW_ID = F.SALES_ORG_ID
   AND E.ROW_ID = F.SALES_CHANNEL_ID
  LEFT JOIN SDI.SDI_CX_PURCH_ITM_1002 G
    ON G.HEAD_ID = A.ROW_ID
  LEFT JOIN SDI.SDI_S_PROD_INT_1002 H
    ON G.PROD_ID = H.ROW_ID
  LEFT JOIN (SELECT OD.X_SAP_NUM
                   ,OD.X_DP_SHIPMENT_PLAN_ORDER_ID
				   ,ROW_NUMBER() OVER(PARTITION BY OD.X_DP_SHIPMENT_PLAN_ORDER_ID ORDER BY CASE WHEN OD.X_SAP_NUM IS NOT NULL THEN 1 ELSE 2 END) AS RN
               FROM SDI.SDI_S_ORDER_1002 OD 
			  WHERE OD.STATUS_CD <> '无效' 
                AND OD.X_SAP_NUM IS NOT NULL
				AND OD.X_OD_TYPE = 'Purchase Order'
				AND OD.X_OD_SUBTYPE = 'Project Order'  
            ) O
	ON A.ROW_ID = O.X_DP_SHIPMENT_PLAN_ORDER_ID
   AND O.RN = 1
  LEFT JOIN (SELECT DISTINCT
                    ROW_ID
                   ,PURCH_NUM
               FROM SDI.SDI_CX_PURCH_1002
			  WHERE DP_REQUITE_ORDER_TYPE in('Purchase Single','Purchase Single UpOne')  
			 ) Q
	ON SUBSTR(A.PURCH_NUM,1,LENGTH(A.PURCH_NUM) - 2) = Q.PURCH_NUM
  LEFT JOIN SDI.SDI_S_ORDER_1002 SO
    ON G.DP_ORDER_ID = SO.ROW_ID
  LEFT JOIN T1 AS P
    ON A.DP_REQUITE_ORDER_TYPE = P.NAME
   AND P.TYPE = 'DP_REQUIRE_GOODS_TYPE'   
  LEFT JOIN T1 AS T3
    ON A.DP_IF_PLAY = T3.NAME
   AND T3.TYPE = 'WF_PROC_PERSIS_FREQ_CD'
  LEFT JOIN T1 AS T4
    ON A.DP_DISBURDEN = T4.NAME
   AND T4.TYPE = 'DISCHARGER_TYPE'
  LEFT JOIN T1 AS T5
    ON A.DP_FORWARDER = T5.NAME
   AND T5.TYPE = 'FORWARDER_TYPE'
  LEFT JOIN T1 AS T6
    ON A.DP_FREIGHT_PAYMENT_METHOD = T6.NAME
   AND T6.TYPE = 'DP_FREIGHT_PAY_MODE'
  LEFT JOIN T1 AS T8
    ON A.DP_DISTRIBUTION_MODE = T8.NAME
   AND T8.TYPE = 'DP_DISTRI_MODE'
  LEFT JOIN T1 AS T9
    ON A.DP_CONDITIONS_OF_CARRIAGE = T9.NAME
   AND T9.TYPE = 'DP_CARRIAGE_TERMS'
  LEFT JOIN SDI.SDI_TVGRT_1001 T10
    ON F.SALES_GROUP_CODE = T10.VKGRP
   AND T10.SPRAS = '1'
   AND T10.MANDT = '800'
  LEFT JOIN T1 AS T14
    ON A.DP_IF_MACHINING = T14.NAME
   AND T14.TYPE = 'DP_MATCH_PROO_FLAG'
  LEFT JOIN T1 AS T15
    ON A.DP_TRANSPORTATION_MODE = T15.NAME
   AND T15.TYPE = 'DP_TRANS_TYPE'
  LEFT JOIN SDI.SDI_TVKBT_1001 T16
    ON F.SALES_DEPART_CODE = T16.VKBUR
   AND T16.SPRAS = '1'
   AND T16.MANDT = '800'
  LEFT JOIN T1 AS T17
    ON H.X_MAT_GRP = T17.NAME
   AND T17.TYPE = 'DP_MAT_GRP_TYPE'
 WHERE A.DP_REQUITE_ORDER_TYPE in('Pro Deliver Plan')
