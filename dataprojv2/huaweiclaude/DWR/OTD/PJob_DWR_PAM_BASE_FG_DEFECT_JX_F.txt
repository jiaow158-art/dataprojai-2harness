INSERT OVERWRITE TABLE  dwrotd.DWR_PAM_BASE_FG_DEFECT_JX_F
SELECT 
    Y.CATEGORY,--类别
    Y.PROMODEL AS PROMODEL,--型号
    Y.CALDAY,
    SUM(Y.FJNUM) AS FJNUM,--分级总数
    SUM(Y.YDNUM) AS YDNUM,--优等品
    SUM(Y.YDPBNUM) AS YDPBNUM,--优等喷补
    SUM(Y.YDDMNUM) AS YDDMNUM,--优等打磨
    SUM(Y.HGNUM) AS HGNUM,--合格品
    SUM(Y.CPPBNUM) AS CPPBNUM,--次品喷补
    SUM(Y.CPDMNUM) AS CPDMNUM,--次品打磨
    SUM(Y.FSNUM) AS FSNUM,--返烧
    SUM(Y.LQNUM) AS LQNUM,--漏气
    SUM(Y.FPNUM) AS FPNUM,--废品
    SUM(Y.QXA1) AS QXA1,--浆料缺陷数
    SUM(Y.QXA2) AS QXA2,--釉料缺陷数
    SUM(Y.QXB) AS QXB,--注浆缺陷数
    SUM(Y.QXD) AS QXD,--施釉缺陷数
    SUM(Y.QXEF) AS QXEF,--装卸车缺陷数
    SUM(Y.QXG) AS QXG,--吹尘缺陷数
    SUM(Y.QXH) AS QXH,--司炉缺陷数
    SUM(Y.SSBHGNUM) AS SSNUM,--一烧试水不合格数
    SUM(Y.FQFPNUM) AS FXNUM,   --一烧试水不合格数
    substr(Y.CALDAY,1,7) as period_id_m
FROM (
    SELECT 
        P.CATEGORY,
        CF.PROMODEL,
        CF.FJDATE  AS CALDAY,
        COUNT(DISTINCT cf.code) + COALESCE(SUM(CASE WHEN cf.code = '' THEN 1 ELSE 0 END), 0) AS FJNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '优等' THEN cf.code ELSE NULL END) AS YDNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '优等喷补' THEN cf.code ELSE NULL END) AS YDPBNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '优等打磨' THEN cf.code ELSE NULL END) AS YDDMNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '合格' THEN cf.code ELSE NULL END) AS HGNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '次品打磨' THEN cf.code ELSE NULL END) AS CPDMNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '次品喷补' THEN cf.code ELSE NULL END) AS CPPBNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '返烧' THEN cf.code ELSE NULL END) AS FSNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '漏气' THEN cf.code ELSE NULL END) AS LQNUM,
        COUNT(DISTINCT CASE WHEN CF.PROGRADE = '废品' THEN cf.code ELSE NULL END) AS FPNUM,
        SUM(QX_ALLA1.QXNUM) AS QXA1,
        SUM(QX_ALLA2.QXNUM) AS QXA2,
        SUM(QX_ALLB.QXNUM) AS QXB,
        SUM(QX_ALLD.QXNUM) AS QXD,
        COALESCE(SUM(QX_ALLE.QXNUM), 0) + COALESCE(SUM(QX_ALLF.QXNUM), 0) AS QXEF,
        SUM(QX_ALLG.QXNUM) AS QXG,
        SUM(QX_ALLH.QXNUM) AS QXH,
        0 AS FQFPNUM,
        0 AS SSBHGNUM
    FROM 
        SDI.SDI_QIS_CLASSIFICATION_1011 CF
    LEFT JOIN 
        SDI.SDI_PRODUCTMODEL_1011 P 
        ON P.MODELID = CF.PROMODEL AND P.status <> '9'
    LEFT JOIN ( SELECT  COUNT(DISTINCT QT.TESTID) AS QXNUM,QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE  QT.STATUS <> '9' AND D.DICTCODE = '125' AND QT.ERRTABLE = 'QIS_CLASSIFICATION'GROUP BY  QT.TESTID) QX_ALLA1 ON QX_ALLA1.TESTID = CF.UUID
    LEFT JOIN (SELECT  COUNT(DISTINCT QT.TESTID) AS QXNUM,QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE  QT.STATUS <> '9'  AND (D.DICTCODE = '124' OR D.DICTCODE = '267') AND QT.ERRTABLE = 'QIS_CLASSIFICATION'GROUP BY  QT.TESTID) QX_ALLA2  ON QX_ALLA2.TESTID = CF.UUID
    LEFT JOIN (SELECT  COUNT(DISTINCT QT.TESTID) AS QXNUM,QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT JOIN SDI.SDI_DP_DICTIONARY_1011 D  ON QT.QXCODE = D.DICTCODE WHERE QT.STATUS <> '9'  AND D.PARENTCODE = 'QXB' AND QT.ERRTABLE = 'QIS_CLASSIFICATION'
        GROUP BY QT.TESTID) QX_ALLB ON QX_ALLB.TESTID = CF.UUID
    LEFT JOIN (SELECT COUNT(DISTINCT QT.TESTID) AS QXNUM,QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE  QT.STATUS <> '9' AND D.PARENTCODE = 'QXD' AND QT.ERRTABLE = 'QIS_CLASSIFICATION'
    GROUP BY QT.TESTID) QX_ALLD ON QX_ALLD.TESTID = CF.UUID
    LEFT JOIN (SELECT COUNT(DISTINCT QT.TESTID) AS QXNUM,  QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT
    JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE QT.STATUS <> '9' AND D.PARENTCODE = 'QXE' AND QT.ERRTABLE = 'QIS_CLASSIFICATION' GROUP BY QT.TESTID) QX_ALLE ON QX_ALLE.TESTID = CF.UUID
    LEFT JOIN (SELECT  COUNT(DISTINCT QT.TESTID) AS QXNUM, QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT
    JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE QT.STATUS <> '9' AND D.PARENTCODE = 'QXF' AND QT.ERRTABLE = 'QIS_CLASSIFICATION' GROUP BY QT.TESTID) QX_ALLF ON QX_ALLF.TESTID = CF.UUID
    LEFT JOIN (SELECT COUNT(DISTINCT QT.TESTID) AS QXNUM, QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT
    JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE QT.STATUS <> '9' AND D.PARENTCODE = 'QXG' AND QT.ERRTABLE = 'QIS_CLASSIFICATION'GROUP BY  QT.TESTID) QX_ALLG ON QX_ALLG.TESTID = CF.UUID
    LEFT JOIN (SELECT  COUNT(DISTINCT QT.TESTID) AS QXNUM,  QT.TESTID FROM SDI.SDI_QIS_TESTERR_1011 QT
    JOIN SDI.SDI_DP_DICTIONARY_1011 D ON QT.QXCODE = D.DICTCODE WHERE  QT.STATUS <> '9' AND D.PARENTCODE = 'QXH' AND QT.ERRTABLE = 'QIS_CLASSIFICATION' GROUP BY QT.TESTID) QX_ALLH ON QX_ALLH.TESTID = CF.UUID
    WHERE 
        CF.STATUS <> '9'
        -- AND CF.FACTORY IN ('S_FACTORY')
        -- AND CF.FJDATE >= 'S_DATE_B'
        -- AND CF.FJDATE <= 'S_DATE_E'
        GROUP BY 
        P.CATEGORY,
        CF.PROMODEL,
        CF.FJDATE 
        
    UNION ALL 
SELECT 
    P.CATEGORY,
    X.PROMODEL,
    X.FJDATE ,
    0 AS FJNUM,
    0 AS YDNUM,
    0 AS YDPBNUM,
    0 AS YDDMNUM,
    0 AS HGNUM,
    0 AS CPPBNUM,
    0 AS CPDMNUM,
    0 AS FSNUM,
    0 AS LQNUM,
    0 AS FPNUM,
    SUM(CASE WHEN dd.DICTCODE = '125' THEN e.errnum ELSE 0 END) AS QXA1,
    SUM(CASE WHEN dd.DICTCODE = '124' THEN e.errnum ELSE 0 END) + 
    SUM(CASE WHEN dd.DICTCODE = '267' THEN e.errnum ELSE 0 END) AS QXA2,
    SUM(CASE WHEN dd.parentcode = 'QXB' THEN e.errnum ELSE 0 END) AS QXB,
    SUM(CASE WHEN dd.parentcode = 'QXD' THEN e.errnum ELSE 0 END) AS QXD,
    SUM(CASE WHEN dd.parentcode IN ('QXE', 'QXF') THEN e.errnum ELSE 0 END) AS QXEF,
    SUM(CASE WHEN dd.parentcode = 'QXG' THEN e.errnum ELSE 0 END) AS QXG,
    SUM(CASE WHEN dd.parentcode = 'QXH' THEN e.errnum ELSE 0 END) AS QXH,
    0 AS FQFPNUM,
    0 AS SSBHGNUM
FROM 
    SDI.SDI_QIS_XJYSINFO_1011 X
LEFT JOIN 
    SDI.SDI_PRODUCTMODEL_1011 P 
    ON P.MODELID = X.PROMODEL
LEFT JOIN 
    SDI.SDI_QIS_TESTERR_1011 E 
    ON E.TESTID = X.UUID AND E.STATUS <> '9'
LEFT JOIN 
    SDI.SDI_DP_DICTIONARY_1011 dd 
    ON dd.dictcode = e.QXCODE
WHERE 
    X.STATUS <> '9'
GROUP BY 
    P.CATEGORY,
    X.PROMODEL,
    X.FJDATE 
    UNION ALL 
SELECT 
    p.CATEGORY,
    W.PRODUCTGC,
    W.WINDDATE,
    0 AS FJNUM,
    0 AS YDNUM,
    0 AS YDPBNUM,
    0 AS YDDMNUM,
    0 AS HGNUM,
    0 AS CPPBNUM,
    0 AS CPDMNUM,
    0 AS FSNUM,
    0 AS LQNUM,
    0 AS FPNUM,
    0 AS QXA1,
    0 AS QXA2,
    0 AS QXB,
    0 AS QXD,
    0 AS QXEF,
    0 AS QXG,
    0 AS QXH,
    SUM(CASE WHEN W.QUALIFIED = '0' THEN 1 ELSE 0 END) AS FQFPNUM,
    0 AS SSBHGNUM
FROM 
    SDI.SDI_QIS_WIND_1011 W
LEFT JOIN 
    SDI.SDI_PRODUCTMODEL_1011 P 
    ON P.MODELID = W.PRODUCTGC
LEFT JOIN 
    SDI.SDI_QIS_CLASSIFICATION_1011 C 
    ON C.CODE = W.BARCODE 
    AND C.STATUS <> '9'
    AND C.PROGRADE IN ('优等', '优等喷补', '优等打磨', '合格')
LEFT JOIN 
    SDI.SDI_QIS_FSFJINFO_1011 FS 
    ON FS.CODE = W.BARCODE 
    AND FS.STATUS <> '9'
WHERE 
    FS.CODE IS NULL
GROUP BY 
    p.CATEGORY,
    W.PRODUCTGC,
    W.WINDDATE
UNION ALL 
SELECT 
    P.CATEGORY,
    ls.product AS promodel,
    ls.testDATE,
    0 AS FJNUM,
    0 AS YDNUM,
    0 AS YDPBNUM,
    0 AS YDDMNUM,
    0 AS HGNUM,
    0 AS CPPBNUM,
    0 AS CPDMNUM,
    0 AS FSNUM,
    0 AS LQNUM,
    0 AS FPNUM,
    0 AS QXA1,
    0 AS QXA2,
    0 AS QXB,
    0 AS QXD,
    0 AS QXEF,
    0 AS QXG,
    0 AS QXH,
    0 AS FQFPNUM,
    COUNT(DISTINCT CASE 
        WHEN ls.Testresult IN ('CCC', 'DDD', 'HHH', 'EEE', 'FFF', 'GGG') 
        THEN ls.BARCODE 
        ELSE NULL 
    END) AS SSBHGNUM
FROM 
    SDI.SDI_qis_testwater_1011 ls
LEFT JOIN 
    SDI.SDI_PRODUCTMODEL_1011 P 
    ON P.MODELID = ls.product
LEFT JOIN 
    SDI.SDI_QIS_CLASSIFICATION_1011 c 
    ON c.code = ls.barcode AND c.status <> '9'
LEFT JOIN 
    SDI.SDI_qis_fsfjinfo_1011 fs 
    ON fs.code = ls.barcode
WHERE 
    fs.code IS NULL
    AND c.prograde IN ('优等', '优等打磨', '优等喷补', '合格')
GROUP BY 
    P.CATEGORY,
    ls.product,
    ls.testDATE
) Y
GROUP BY 
    Y.CATEGORY,
    Y.PROMODEL  ,
    Y.CALDAY,
    substr(Y.CALDAY,1,7)
