WITH COST_CENTER AS (
SELECT   
T.SHIFT_DATE,
T.SITE_CODE,   
T.WORK_SHOP_CODE,   
T.PRODUCT_LINE_CODE,   
T.COST_CENTER_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
T.S_PRODUCT_LINE_CODE,  --服务产线编码
T.S_PRODUCT_LINE_NAME,  --服务产线名称
ROW_NUMBER() OVER(PARTITION BY T.SHIFT_DATE,T.SITE_CODE,T.WORK_SHOP_CODE,T.PRODUCT_LINE_CODE,T.S_PRODUCT_LINE_CODE,T.PCS_SEG_CODE ORDER BY T.COST_CENTER_CODE) AS RN
FROM (SELECT 
T1.DAY_DT AS SHIFT_DATE,
T.SITE_CODE,   
T.WORK_SHOP_CODE,   
T.PRODUCT_LINE_CODE,   
T.COST_CENTER_CODE,
T.PCS_SEG_CODE,
T.PCS_SEG_NAME,
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_CODE 
     ELSE '' END AS S_PRODUCT_LINE_CODE,  --服务产线编码
CASE WHEN T.PCS_SEG_CODE = 'C' THEN T.M_PRODUCT_LINE_NAME
     ELSE '' END AS S_PRODUCT_LINE_NAME  --服务产线名称
FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 T --基地产线成本表
JOIN (SELECT DAY_DT FROM DM.DM_DIM_DATE_D 
	  WHERE SUBSTR(DATE_ADD(DAY_DT,1),1,7) BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}') T1 
ON T1.DAY_DT BETWEEN SUBSTR(T.START_DATE,1,10) AND SUBSTR(T.END_DATE,1,10)
LEFT JOIN SDI.SDI_RADIATION_WORKSHOP_T_1037 T2  --辅助车间辐射车间
ON T.SITE_CODE = T2.SITE_CODE 
AND T.COST_CENTER_CODE = T2.COST_CENTER_CODE
AND T.M_PRODUCT_LINE_CODE = T2.M_PRODUCT_LINE_CODE
AND T1.DAY_DT BETWEEN T2.EFFECT_START_DATE AND T2.EFFECT_END_DATE
WHERE T.M_PRODUCT_LINE_NAME NOT IN ('分级车间','辅助车间') ) T
 )

,PRESS_PROD AS ( --压机
SELECT 
 A.SITE_CODE	--	基地
,A.WORK_SHOP_CODE	--	车间
,A.PRODUCT_LINE_CODE_A AS PRODUCT_LINE_CODE
,A.SHIFT_DATE  --班次日期
,SUM(A.PRODUCTION_NUM * A.UNIT_AREA) AS PRODUCTION_AREA -- 压机数/M2
FROM DWROTD.DWR_PAM_BASE_PRESS_WORK_F A 
WHERE SUBSTR(DATE_ADD(A.SHIFT_DATE,1),1,7) BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}' 
GROUP BY 
 A.SITE_CODE	--	基地
,A.WORK_SHOP_CODE	--	车间
,A.PRODUCT_LINE_CODE_A
,A.SHIFT_DATE
)
,TMP AS (
SELECT 
 DATE_ADD(T1.SHIFT_DATE,1) AS SHIFT_DATE 
,T1.SITE_CODE   
,T1.WORK_SHOP_CODE
,T1.PRODUCT_LINE_CODE   
,T1.COST_CENTER_CODE 
,T1.S_PRODUCT_LINE_CODE  --服务产线编码
,T1.S_PRODUCT_LINE_NAME  --服务产线名称
,T1.PCS_SEG_CODE
,T1.PCS_SEG_NAME
,CASE WHEN NVL(T2.PRODUCTION_AREA,0) <> 0 THEN 1 ELSE 0 END AS COUNT_NUM
,CASE WHEN NVL(T2.PRODUCTION_AREA,0) = 0 THEN T1.SHIFT_DATE END AS START_DATE
,CASE WHEN NVL(T2.PRODUCTION_AREA,0) = 0 THEN T1.SHIFT_DATE END AS END_DATE
---停产置0，否置1，再每天按月汇总截止所在日期的COUNT_NUM
,SUM(CASE WHEN NVL(T2.PRODUCTION_AREA,0) <> 0 THEN 1 ELSE 0 END) OVER(PARTITION BY SUBSTR(DATE_ADD(T1.SHIFT_DATE,1),1,7),T1.SITE_CODE,T1.WORK_SHOP_CODE,T1.PRODUCT_LINE_CODE,T1.S_PRODUCT_LINE_CODE ORDER BY T1.SHIFT_DATE) AS COUNT_SUM 
FROM COST_CENTER T1
LEFT JOIN PRESS_PROD T2 
ON T1.SHIFT_DATE =T2.SHIFT_DATE 
AND T1.SITE_CODE = T2.SITE_CODE
AND T1.WORK_SHOP_CODE = T2.WORK_SHOP_CODE
AND T1.PRODUCT_LINE_CODE = T2.PRODUCT_LINE_CODE 
WHERE T1.RN = 1  
)
,TMP1 AS (
SELECT 
 SUBSTR(T.SHIFT_DATE,1,7) AS STAT_MONTH
,T.SITE_CODE   
,T.WORK_SHOP_CODE
,T.PRODUCT_LINE_CODE  
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.COST_CENTER_CODE 
,MIN(T.START_DATE) AS START_DATE
,MAX(T.END_DATE) AS END_DATE
FROM TMP T
WHERE T.COUNT_NUM = '0' AND T.PCS_SEG_CODE = 'C' --成型
GROUP BY  
 SUBSTR(T.SHIFT_DATE,1,7)
,T.SITE_CODE   
,T.WORK_SHOP_CODE
,T.PRODUCT_LINE_CODE   
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.COST_CENTER_CODE 
,T.COUNT_SUM

)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_LOCKOUT_DATE_T PARTITION(PERIOD_ID_M)
SELECT 
 T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T1.SITE_NAME --'基地名称'
,T.WORK_SHOP_CODE --'车间编码'
,T2.WORK_SHOP_NAME --'车间名称'
,T.PRODUCT_LINE_CODE --'产线编码'
,T3.PRODUCT_LINE_A_NAME AS PRODUCT_LINE_NAME  --'产线名称'
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
,T.COST_CENTER_CODE  --'成本中心'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.START_DATE --'开始日期'
,T.END_DATE --'结束日期'
,NVL(DATEDIFF(T.END_DATE,T.START_DATE) + 1 ,0) AS LOCKOUT_DAYS --'停产天数'
,CURRENT_TIMESTAMP () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,STAT_MONTH AS PERIOD_ID_M
FROM TMP1 T
LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D T1 -- 基地维表
ON T.SITE_CODE = T1.SITE_CODE
LEFT JOIN DWRDIM.DWR_DIM_PAM_WORK_SHOP_D T2 --车间维表
ON T.SITE_CODE = T2.SITE_CODE 
AND T.WORK_SHOP_CODE = T2.WORK_SHOP_CODE
LEFT JOIN (SELECT DISTINCT SITE_CODE,SITE_NAME,WORK_SHOP_CODE,WORK_SHOP_NAME,PRODUCT_LINE_A_CODE,PRODUCT_LINE_A_NAME 
FROM DWRDIM.DWR_DIM_PAM_PRODUCT_LINE_D) T3 --产线维表
ON T.SITE_CODE = T3.SITE_CODE
AND T.WORK_SHOP_CODE = T3.WORK_SHOP_CODE
AND T.PRODUCT_LINE_CODE = T3.PRODUCT_LINE_A_CODE
