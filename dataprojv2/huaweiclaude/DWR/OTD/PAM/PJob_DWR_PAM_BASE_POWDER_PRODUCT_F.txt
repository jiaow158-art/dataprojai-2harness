--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;

insert overwrite table DWROTD.DWR_PAM_BASE_POWDER_PRODUCT_F partition(period_id_m)
select 
 T.SITE AS SITE_CODE  --基地代码
,T.WORKSHOP AS WORK_SHOP_CODE --车间代码
,T.PRODUCT_LINE AS PRODUCT_LINE_CODE  --产线代码
,T.TEAM AS GROUP_CODE  -- 班组代码
,T.WORK_NO_CODE   --工序代码
,T.SHIFT_DATE    --班次日期
,T.SHIFT AS SHIFT_CODE   --班次代码
,T.SHIFT_START_TIME --班次开始时间
,T.SHIFT_END_TIME  --班次结束时间
,T.EQUIPENT_SERIAL_NO  --设备编号
,T.EQUIPENT_SERIAL_NO_NAME  --设备名称
,T.POWDER_NAME  --粉料名称
,T.ODO_WATER  --当班产量T（含水）
,T.CONTAIN_WATER  --含水%
,T.ODO_DRIER  --当班产量T（干料）
,T.ODO_WEIGHT  --当班产量T（干重）
,T1.key_value AS tran_rate --'粉料-气耗转换比例'
,T.REMARKS  --备注
,T.CREATE_BY  --创建人
,T.CREATE_DATE  --创建时间
,T.LAST_UPDATE_BY  --修改人
,T.LAST_UPDATE_DATE  --修改时间
,'N' AS DEL_FLAG 
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
,substr(SHIFT_DATE,1,7) as period_id_m
from SDI.SDI_D_POWDER_OUTPUT_REPORT_1006 T
left join (SELECT key_value FROM sdi.sdi_iap_dict_item_t_1006 WHERE iap_dict_id in (SELECT id FROM sdi.sdi_iap_dict_t_1006 WHERE code = 'NX202405131807')) T1  on 1=1 --粉料产量换算天然气
where site = '1800' and delete_flag = 1
and EQUIPENT_SERIAL_NO = 'TA2'
-- and shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
