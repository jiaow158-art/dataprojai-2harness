-- 瓷砖各基地入库产量表（月）
WITH MATDOC_DETAIL_DATA AS( -- SAP物料凭证数据
	SELECT period_id_m, -- 月份
		   budat , -- 过账日期
		   werks , -- 基地
		   AUFNR,  -- 订单
		   ERFME, -- 单位
		   charg, -- 批次
		   MATNR, --物料编码
		   lgort, -- 库存地点 
		   mblnr, -- 物料凭证
		   bwart,-- 移动类型
		   sgtxt, -- 文本
		   sakto , -- 会计科目
	   	   SUM(NVL(stock_qty,0)) AS stock_qty  -- 数量
	FROM SDI.sdi_matdoc_1001 
	WHERE  del_flag = 'N'
	AND 1 = 1
	AND period_id_m >= REPLACE(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-','')
	AND SUBSTRING(matnr,-2,1) = '_' -- 成品
	AND SUBSTRING(matnr,-1) IN ('A','B') -- A 优等 B 一级
	AND bwart IN ('131','132','531','532') -- 移动类型
	AND werks IN('1300','1500','1600','1800','1900') -- 限制 清远、江西、湖南、重庆、山西
	GROUP BY  period_id_m, -- 月份
			  budat , -- 过账日期
			   werks , -- 基地
			   AUFNR,  -- 订单
			   ERFME, -- 单位
			   charg, -- 批次
			   MATNR, --物料编码
			   lgort, -- 库存地点 
			   mblnr, -- 物料凭证
			   bwart,-- 移动类型
			   sgtxt, -- 文本
			   sakto  -- 会计科目
),
MATDOC_DETAIL_DATA01 AS( -- 获取面积
	SELECT T1.period_id_m, -- 月份
		   T1.budat , -- 过账日期
		   T1.werks , -- 基地
		   T1.AUFNR,  -- 订单
		   T1.ERFME, -- 单位
		   T1.charg, -- 批次
		   T1.MATNR, --物料编码
		   T1.lgort, -- 库存地点 
		   T1.mblnr, -- 物料凭证
		   T1.bwart,-- 移动类型
		   T1.sgtxt, -- 文本
		   T1.sakto , -- 会计科目
	   	   T1.stock_qty,  -- 数量
		   t2.dimension, -- 量纲
		   t2.length_1,--  长
		   t2.width -- 宽
	FROM MATDOC_DETAIL_DATA T1
	LEFT JOIN DWIMD.dwi_md_data_material_general_t t2
	ON t1.MATNR = t2.material_num 
),
MATDOC_DETAIL_DATA02 AS( -- 换算面积
	SELECT T.period_id_m, -- 月份
		   T.werks , -- 基地
		   T.AUFNR,  -- 订单
		   T.ERFME, -- 单位
		   T.charg, -- 批次
		   T.MATNR, --物料编码
		   T.lgort, -- 库存地点 
		   T.mblnr, -- 物料凭证
		   T.bwart,-- 移动类型
		   T.sgtxt, -- 文本
		   T.sakto,  -- 会计科目
		   SUM(T.stock_qty) AS stock_qty -- 入库产量
	FROM(SELECT T1.period_id_m, -- 月份
	            T1.budat, -- 过账日期
			    T1.werks , -- 基地
			    T1.AUFNR,  -- 订单
			    T1.ERFME, -- 单位
			    T1.charg, -- 批次
			    T1.MATNR, --物料编码
			    T1.lgort, -- 库存地点 
			    T1.mblnr, -- 物料凭证
			    T1.bwart,-- 移动类型
			    T1.sgtxt, -- 文本
			    T1.sakto,  -- 会计科目
			    NVL(SUM(stock_qty * ((length_1 * width)/1000000)),0) AS stock_qty
	     FROM MATDOC_DETAIL_DATA01 T1
	     GROUP BY T1.period_id_m, -- 月份
	   			  T1.budat, -- 过账日期
			      T1.werks , -- 基地
			      T1.AUFNR,  -- 订单
			      T1.ERFME, -- 单位
			      T1.charg, -- 批次
			      T1.MATNR, --物料编码
			      T1.lgort, -- 库存地点 
			      T1.mblnr, -- 物料凭证
			      T1.bwart,-- 移动类型
			      T1.sgtxt, -- 文本
			      T1.sakto  -- 会计科目
		)T
		GROUP BY T.period_id_m, -- 月份
	   			 T.budat, -- 过账日期
			     T.werks , -- 基地
			     T.AUFNR,  -- 订单
			     T.ERFME, -- 单位
			     T.charg, -- 批次
			     T.MATNR, --物料编码
			     T.lgort, -- 库存地点 
			     T.mblnr, -- 物料凭证
			     T.bwart,-- 移动类型
			     T.sgtxt, -- 文本
			     T.sakto  -- 会计科目
)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_INSTOCKYIELD_F PARTITION(period_id_m)
SELECT CONCAT(SUBSTRING(T1.period_id_m,1,4),'-',SUBSTRING(T1.period_id_m,5,2)) AS start_month, -- 月份
	   T1.werks AS site_code, -- 基地
	   T3.work_shop AS work_shop_code, --  车间编码
	   T3.product_line AS product_line_code, -- 产线编码
	   T1.AUFNR AS order_header_code,  -- 生产订单头
	   T1.ERFME AS unit, -- 单位
	   T1.charg AS batch_code, -- 批次
	   T1.MATNR AS material_num, --物料编码
	   T1.lgort AS stock_place, -- 库存地点 
	   T1.mblnr AS material_voucher, -- 物料凭证
	   T1.bwart AS move_type,-- 移动类型
	   T1.sgtxt AS APS_ORDER_NO, -- 文本(APS订单号)为空情况：一级品 、手工操作
	   T1.sakto AS SUBJECT, -- 会计科目
	   T2.verid AS PROD_VERSION_CODE,-- 生产版本
   	   T1.stock_qty , -- 入库数量
   	   'N' AS del_flag, -- 删除数据标识
   	   CURRENT_TIMESTAMP() AS dw_last_update_date , -- 数据同步时间
   	   CONCAT(SUBSTRING(T1.period_id_m,1,4),'-',SUBSTRING(T1.period_id_m,5,2)) AS period_id_m -- 分区月
FROM MATDOC_DETAIL_DATA02 T1   -- 换算面积
LEFT JOIN SDI.SDI_CKMLMV013_1001 T2 -- SAP产品订单分配表
ON T1.AUFNR = T2.AUFNR
AND T2.del_flag = 'N'
LEFT JOIN SDI.SDI_DP_PRODUCTION_VERSION_MAINT_1006 T3 -- 产线生产版本维护表
ON T1.werks = T3.site 
AND T2.verid = T3.verid 
AND T3.delete_flag = 1;
