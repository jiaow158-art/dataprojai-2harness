WITH one_grade_data_01 as(
	SELECT P.CATEGORY, 
	       P.MUDID, 
	       P.WEIGHT,  
	       CF.PROMODEL,
	       CF.FJDATE,
COUNT(DISTINCT CASE
WHEN 
			CF.ZJPROPERTY = '正常'
			AND CF.SYPROPERTY = '正常'
			AND CF.SCPROPERTY = '正常'
	 THEN CF.CODE
		ELSE NULL 
	END
) + SUM(
	CASE
		WHEN CF.CODE = ''
		AND (
			CF.ZJPROPERTY = '正常'
			AND CF.SYPROPERTY = '正常'
			AND CF.SCPROPERTY = '正常'
		) THEN 1
		ELSE 0
	END
)  AS FJNUM,--
	       COUNT(DISTINCT CASE WHEN CF.PROGRADE = '优等' THEN CF.CODE END)  YSYDNUM, --
	       COUNT(DISTINCT CASE WHEN CF.PROGRADE = '合格' THEN  CF.CODE END) YSHGNUM, --
	       0 DMYDNUM, --
	       0 DMHGNUM, --
	       0 PBYDNUM, --
	       0 PBHGNUM, --
	       0 QTYDNUM, --
	       0 QTHGNUM, --
	       0 CFYDNUM, --
	       0 CFHGNUM, --
	       0 LQYDNUM, --
	       0 LQHGNUM, --
	       0 LSYDNUM, --
	       0 LSHGNUM, --
	       0 SSBHGNUM, --
	       0 FQFPNUM , --
	       0 JJNUM --
	FROM SDI.sdi_qis_classification_1011 CF -- 一烧分级信息
	LEFT JOIN SDI.SDI_PRODUCTMODEL_1011 P  -- 光瓷产品信息表
	ON P.MODELID = CF.PROMODEL 
	and p.status <> '9'
	WHERE CF.STATUS <> '9'
	AND CF.FACTORY IN ('3200','3100')
	--AND CF.FJDATE BETWEEN '2021-09-12' AND '2021-09-12'  
	AND ((CF.ZJPROPERTY = '正常'  
	AND CF.SYPROPERTY = '正常'  
	AND CF.SCPROPERTY = '正常') OR 
	(CF.ZJPROPERTY = '试制'  
	AND CF.SYPROPERTY = '试制'  
	AND CF.SCPROPERTY = '试制'))
	AND P.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')  
	GROUP BY P.CATEGORY, 
			 P.MUDID, 
			 P.WEIGHT,  
			 CF.PROMODEL,
			 CF.FJDATE
	union ALL
	select PM.CATEGORY, 
	       PM.MUDID ,  
	       PM.WEIGHT, 
	       pm.modelid  PROMODEL ,  
	       dm.FJDATE,
	       0 FJNUM,
	       0 YSYDNUM, 
	       0 YSHGNUM, 
	       COUNT(distinct case when  DM.PROGRADE = '优等' then dm.code  end ) DMYDNUM,
	       COUNT(DISTINCT case when DM.PROGRADE = '合格' then DM.CODE end) DMHGNUM ,
	       0 PBYDNUM,
	       0 PBHGNUM,
	       0 QTYDNUM,
	       0 QTHGNUM, 
	       0 CFYDNUM, 
	       0 CFHGNUM, 
	       0 LQYDNUM, 
	       0 LQHGNUM, 
	       0 LSYDNUM, 
	       0 LSHGNUM,
	       0 SSBHGNUM,  
	       0 FQFPNUM , 
	       0 JJNUM
	from SDI.SDI_QIS_DMINFO_1011 dm   -- 打磨分级
	LEFT JOIN SDI.SDI_PRODUCTMODEL_1011 PM 
	ON PM.MODELID = dm.promodel 
	AND PM.STATUS <> '9'
	left join SDI.SDI_qis_classification_1011 cf_dm
	on dm.code = cf_dm.code   
	and cf_dm.status <> '9'
	where 1=1
	and DM.STATUS <> '9'

	and not exists(select 1 from SDI.sdi_qis_fsfjinfo_1011 fs where fs.code = dm.code ) 
	AND ((cf_dm.ZJPROPERTY = '正常'  
	AND cf_dm.SYPROPERTY = '正常'

	(cf_dm.ZJPROPERTY = '试制'  
	AND cf_dm.SYPROPERTY = '试制'

			  PM.MUDID,  
	          PM.WEIGHT, 			  
			  pm.MODELID,
			  dm.FJDATE
	union  ALL   
	select  NVL(PM.CATEGORY, 'NULL') CATEGORY, 
			PM.MUDID ,  
	        PM.WEIGHT, 			
			pm.modelid  PROMODEL , 
			pb.FJDATE,
			0 FJNUM,
	        0 YSYDNUM, 
	        0 YSHGNUM,

	        0 QTHGNUM,
	        0 CFYDNUM,
	        0 CFHGNUM,
	        0 LQYDNUM,
	        0 LQHGNUM, 
	        0 LSYDNUM,
	        0 LSHGNUM,
	        0 SSBHGNUM,  
	        0 FQFPNUM , 
	        0 JJNUM
	from  SDI.SDI_QIS_PBFJINFO_1011 PB    -- 喷补信息
	LEFT JOIN SDI.SDI_PRODUCTMODEL_1011 PM
	ON PM.MODELID = pb.promodel 
	AND PM.STATUS <> '9'
	left join SDI.SDI_qis_classification_1011 cf_pb 
	on pb.code = cf_pb.code    
	and cf_pb.status <> '9'
	where 1=1 
	AND PB.STATUS <> '9' 
	--AND pb.FJDATE BETWEEN '2021-09-12' AND '2021-09-12'
	and not exists(select 1 from SDI.sdi_qis_fsfjinfo_1011 fs where fs.code = pb.code )
	AND pb.FACTORY IN ('3200','3100')   
	AND ((cf_pb.ZJPROPERTY = '正常'  
	AND cf_pb.SYPROPERTY = '正常'  
	AND cf_pb.SCPROPERTY = '正常') OR 
	(cf_pb.ZJPROPERTY = '试制'  
	AND cf_pb.SYPROPERTY = '试制'  
	AND cf_pb.SCPROPERTY = '试制'))
	AND PM.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15') 
	group by  PM.CATEGORY,
			  PM.MUDID,  
	          PM.WEIGHT, 			  
			  pm.MODELID ,
			  pb.FJDATE
	union ALL
	SELECT NVL(PM.CATEGORY, 'NULL') CATEGORY,

  	          PM.WEIGHT,

             PM.WEIGHT ,

SELECT Y.FJDATE AS GRADE_DATE, -- 分级日期
	   Y.CATEGORY AS PROD_CATEGORY, -- 产品类别
       Y.MUDID AS MUDID_MODEL, -- 泥坯型号
       Y.PROMODEL AS PROD_MODEL, -- 产品型号
       SUM(Y.FJNUM) AS ONE_GRADE_QTY,  -- 一烧分级数
       SUM(NVL(Y.YSYDNUM, '0') + NVL(Y.YSHGNUM, '0') + NVL(DMYDNUM, '0') + NVL(DMHGNUM, '0') + NVL(PBYDNUM, '0') + NVL(Y.PBHGNUM, '0') +
        NVL(QTYDNUM, '0') + NVL(QTHGNUM, '0') + NVL(CFYDNUM, '0') + NVL(CFHGNUM, '0') + NVL(LQYDNUM, '0') + NVL(LQHGNUM, '0') +
        NVL(LSYDNUM, '0') + NVL(LSHGNUM, '0') - NVL(SSBHGNUM, '0') - nvl(JJNUM,0)  - NVL(y.FQFPNUM, '0')) ONE_FINE_QTY, -- 一烧优等数
       Y.WEIGHT
 FROM one_grade_data_01 Y
where 1=1 
and y.category <> '烧托垫'    
GROUP BY Y.CATEGORY, 
         Y.MUDID, 
         Y.PROMODEL,
         Y.FJDATE,
         Y.WEIGHT
)

 grade_date, prod_category, mudid_model, prod_model, one_grade_qty, one_fine_qty, CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, weight
