--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

with tmp0 as (
select SHIFT_DATE,site_code,work_shop_code,product_line_code,sum(NVL(HAND_NUM * unit_area,0)) as HAND_AREA
 from DWROTD.DWR_PAM_BASE_KEEP_BRICK_f 
 where site_code in ('1600' ,'1300','1900') and flag <> '3'
--  and SHIFT_DATE >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
 group by SHIFT_DATE,site_code,work_shop_code,product_line_code
)

, tmp1 as (
select * 
from (
select 
 site_code
,work_shop_code
,day_dt
,HAND_NUM
,unit_area
,id
,product_line_code
,ROW_NUMBER ( ) OVER ( PARTITION BY day_dt,work_shop_code, product_code ORDER BY create_time desc) dayrow --重庆昨日存砖数
from DWROTD.DWR_PAM_BASE_KEEP_BRICK_F 
join (select day_dt from dm.dm_dim_date_d where --day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')and 
day_dt >= '2023-01-01' AND day_dt <= substr(date_add(current_date,-1),1,10)) b on 1=1
 where  SHIFT_DATE < day_dt and site_code = '1800'
) as T
where HAND_NUM <> 0 and dayrow = 1
)
,tmp2 as (
select * 
from (
select 
 site_code
,work_shop_code
,day_dt
,HAND_NUM
,unit_area
,id
,product_line_code
,ROW_NUMBER ( ) OVER ( PARTITION BY day_dt,work_shop_code, product_code ORDER BY create_time desc) dayrow  --重庆当日存砖数
from DWROTD.DWR_PAM_BASE_KEEP_BRICK_F 
join (select day_dt from dm.dm_dim_date_d where --day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')and 
day_dt >= '2023-01-01' AND day_dt <= substr(date_add(current_date,-1),1,10)) b on 1=1
 where  SHIFT_DATE <= day_dt and site_code = '1800'
) as T
where  HAND_NUM <> 0 and dayrow = 1
)
,tmp3 as (
select * 
from (
select 
 site_code
,work_shop_code
,product_line_code
,day_dt
,HAND_AREA
,SHIFT_DATE
,ROW_NUMBER ( ) OVER ( PARTITION BY day_dt,work_shop_code,product_line_code order by SHIFT_DATE desc) dayrow  
from tmp0
join (select day_dt from dm.dm_dim_date_d where --day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')and 
day_dt >= '2023-01-01' AND day_dt <= substr(date_add(current_date,-1),1,10)) b on 1=1
 where  SHIFT_DATE < day_dt 
) as T
where  HAND_AREA <> 0 and dayrow = 1 
)
,tmp4 as (
select * 
from (
select 
 site_code
,work_shop_code
,product_line_code
,day_dt
,HAND_AREA
,SHIFT_DATE
,ROW_NUMBER ( ) OVER ( PARTITION BY day_dt,work_shop_code,product_line_code order by SHIFT_DATE desc) dayrow  
from tmp0
join (select day_dt from dm.dm_dim_date_d where --day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')and 
day_dt >= '2023-01-01' AND day_dt <= substr(date_add(current_date,-1),1,10)) b on 1=1
 where  SHIFT_DATE <= day_dt --and site_code = '1600'
) as T
where  HAND_AREA <> 0 and dayrow = 1 
)

-- )
insert overwrite table DWROTD.DWR_PAM_BASE_KEEP_BRICK_S partition(period_id_m)
select --重庆
 b.day_dt as start_date
,a.site_code
,a.work_shop_code
,a.product_line_code
,sum(nvl(b.HAND_NUM * b.unit_area,0)) as K_AREA --留砖接班 
,'1' as flag --留砖接班 
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,substr(b.day_dt,1,7) AS period_id_m
from DWROTD.DWR_PAM_BASE_KEEP_BRICK_F a
left join tmp1 b
on a.id = b.id
where b.day_dt <> ''
-- and b.day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
group by 
 a.site_code
,a.work_shop_code
,b.day_dt
,a.product_line_code
union all 
select  --重庆
 b.day_dt as start_date
,a.site_code
,a.work_shop_code
,a.product_line_code
,sum(nvl(b.HAND_NUM * b.unit_area,0)) as K_AREA --留砖交班 
,'2' as flag --留砖交班 
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,substr(b.day_dt,1,7) AS period_id_m
from DWROTD.DWR_PAM_BASE_KEEP_BRICK_F a
left join tmp2 b
on a.id = b.id
where  b.day_dt <> '' 
-- and b.day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
group by 
 a.site_code
,a.work_shop_code
,b.day_dt
,a.product_line_code
union all 
select --湖南、清远、山西
 a.day_dt as start_date
,a.site_code
,a.work_shop_code
,a.product_line_code
,a.HAND_AREA as K_AREA --留砖接班 
,'1' as flag --留砖接班 
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,substr(a.day_dt,1,7) AS period_id_m
from tmp3 a
where a.day_dt <> '' 
-- and a.day_dt >= CONCAT(SUBSTR(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select --湖南、清远、山西
 a.day_dt as start_date
,a.site_code
,a.work_shop_code
,a.product_line_code
,a.HAND_AREA as K_AREA--留砖交班 
,'2' as flag --留砖交班 
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,substr(a.day_dt,1,7) AS period_id_m
from tmp4 a
where a.day_dt <> ''
union all 
select --山西当日交班= 上日接班+当日交班
 a.day_dt as start_date
,a.site_code
,a.work_shop_code
,a.product_line_code
,a.HAND_AREA as K_AREA--留砖接班 
,'2' as flag --留砖交班 
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,substr(a.day_dt,1,7) AS period_id_m
from tmp3 a
where a.day_dt <> '' and a.site_code = '1900'
