INSERT OVERWRITE TABLE DWROTD.DWR_VM_INPUT_OUTPUT_T PARTITION(STAT_MONTH)
SELECT
       A.AUFNR AS ORDER_NUM --订单号
      ,A.MATNR AS MATERIAL_CODE --物料编号
      ,A.MATNRI AS INTO_MATERIAL_CODE --投入产品编码
      ,A.FLAG AS INOUT_FLAG --投入/产出标识
      ,A.KOSTL AS COST_CENTER --成本中心
      ,A.MENGE AS QTY --数量
      ,A.DMBTR AS AMT --金额
      ,A.MENGEI AS INTO_QTY --投入数量
      ,A.DMBTRI AS INTO_AMT --投入金额
      ,A.BUKRS AS COM_CODE --公司代码
      ,A.WERKS AS FACTORY_WERKS_CODE --工厂
      ,A.MAKTX AS MATERIAL_NAME --物料描述
      ,A.KTEXT AS ORDER_DESC --描述
      ,A.MAKTXI AS INTO_MATERIAL_NAME --投入产品描述
      ,A.MEINS AS BASE_MEASURE_UNIT --基本计量单位
      ,A.MEINSI AS INTO_MEASURE_UNIT --投入计量单位
      ,SUBSTR(A.UPDATE_DATE,1,4)||'-'||''||SUBSTR(A.UPDATE_DATE,5,2)||'-'||SUBSTR(A.UPDATE_DATE,7,2) AS UPDATE_DATE --更新日期
	  ,B.LENGTH_1 --长度
	  ,B.WIDTH --宽度
	  ,B.LENGTH_1 * B.WIDTH /1000000 AS UNIT_AREA --单位面积(平方米)
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,A.YEAR_MONTH AS STAT_MONTH --年月
  FROM SDI.SDI_ZMM_ZPIOS_1001 A
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
    ON A.MATNR = B.MATERIAL_NUM
 INNER JOIN (SELECT YEAR_MONTH
                   ,MAX(UPDATE_DATE) AS UPDATE_DATE
               FROM SDI.SDI_ZMM_ZPIOS_1001
			  GROUP BY YEAR_MONTH
			) C
	ON A.YEAR_MONTH = C.YEAR_MONTH
   AND A.UPDATE_DATE = C.UPDATE_DATE
 WHERE A.YEAR_MONTH = '${PERIOD_ID_M}'
;
