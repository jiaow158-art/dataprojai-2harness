,one_grade_data_02 as(
	SELECT SUBSTRING(Y.FJDATE,1,7) AS GRADE_DATE, -- 分级月份
		   Y.FACTORY AS FACTORY_CODE, -- 工厂编码
		   --Y.CATEGORY AS PROD_CATEGORY, -- 产品类别
	       --Y.MUDID AS MUDID_MODEL, -- 泥坯型号
	       --Y.PROMODEL AS MATERIAL_NUM, -- 产品型号(物料编码)
	       t1.material_num AS MATERIAL_NUM,
	       SUM(Y.FJNUM) AS YSFJ_QTY,  -- 一烧分级数
	       SUM(NVL(Y.YSYDNUM, '0') + NVL(Y.YSHGNUM, '0') + NVL(DMYDNUM, '0') + NVL(DMHGNUM, '0') + NVL(PBYDNUM, '0') + NVL(Y.PBHGNUM, '0') +
	        NVL(QTYDNUM, '0') + NVL(QTHGNUM, '0') + NVL(CFYDNUM, '0') + NVL(CFHGNUM, '0') + NVL(LQYDNUM, '0') + NVL(LQHGNUM, '0') +
	         NVL(LSYDNUM, '0') + NVL(LSHGNUM, '0') - NVL(SSBHGNUM, '0') - nvl(JJNUM,0)  - NVL(y.FQFPNUM, '0')) YSYD_QTY -- 一烧优等数
	 FROM one_grade_data_01 Y
	 left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t1 -- 生产BOM关系数据
	 on y.FACTORY = t1.factory_code
	 and y.PROMODEL = t1.small_material_num
	where 1=1 
	and y.category <> '烧托垫'    
	and Y.FJDATE between '2022-06-01' and '${PERIOD_ID_D}'  -- 传当天
	GROUP BY --Y.CATEGORY, 
	         --Y.MUDID, 
	         --Y.PROMODEL,
	         t1.material_num,
	         SUBSTRING(Y.FJDATE,1,7),
	         Y.FACTORY
),
JX_ACTUAL_DATA_01 as(  -- 江西实收数
select t.in_stock_date , -- 入库日期
	   t.factory_code, -- 工厂编码
	   --t.material_num, --物料编码
	   t1.material_num,
       sum(t.in_stock_qty)  AS actual_receipt_qty -- 收坯数   
FROM(
		select SUBSTRING(a.indate,1,7) as in_stock_date , -- 入库日期
		       --case when c.monitorname is not null then c.monitorname else gl.MONITORNAME end  as MONITOR_NAME, -- 带班班长名称
		       --a.produceline as GROUTING_lineno,  -- 注浆线号
		       --a.producttype as mud_model_code,  -- 泥坯型号
		       a.FACTORY as factory_code, -- 工厂编码
		       t3.MODELID as material_num, --物料编码
		       sum(a.innum) as in_stock_qty   -- 入库数量
		from SDI.SDI_qis_modifyoffline_1011 a   -- 改洗下线记录表（即入库条码）
		left join SDI.SDI_QIS_GROUTINGPLAN_BARCODE_1011 c   -- 注浆条码列表
		on a.barcode=c.barcode 
		and c.status<>'9'
		left join sdi.sdi_qis_mud_1011 b  -- 泥坯
		on a.producttype =b.mudid 
		and b.status<>'9'
		left join SDI.SDI_QIS_GROUTINGLINE_1011 gl   -- 注浆线
		on a.produceline=gl.LINENO 
		and gl.status <> '9' 
		and gl.factoryid = a.factory
		LEFT JOIN  SDI.SDI_PRODUCTMODEL_1011  T3
		ON a.producttype = T3.mudid
		where a.status<>'9' 
		and (a.isabnormal='00000' or (a.gbgrade='1' and a.is_gb='0') )
		and not exists (
		  select 1 from SDI.SDI_QIS_REPEATCODE_1011 re  -- 重贴条码
		  where re.code= a.barcode and re.status <>'9' 
		  )
		AND a.FACTORY IN ('3200')   
		AND a.productproperty = 'AAAAA'  
		AND b.CATEGORY NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15') 
		and a.indate between '2022-06-01' and '${PERIOD_ID_D}'
		group by SUBSTRING(a.indate,1,7),
				 a.FACTORY,
				 t3.MODELID
		         --case when c.monitorname is not null then c.monitorname else gl.MONITORNAME end,
		         --a.produceline,
		         --a.producttype 
		UNION ALL
		select substring(d.cjdate,1,7) AS IN_STOCK_DATE, -- 入库日期
		       a.FACTORY as factory_code,
		       t3.MODELID as material_num, --物料编码
		       --c.monitorname as MONITOR_NAME, -- 带班班长名称
		       --a.produceline as GROUTING_lineno,  -- 注浆线号
		       --a.producttype as mud_model_code,  -- 泥坯型号
		       -sum(case when d.uuid is not null then 1 else 0 end) as in_stock_qty   -- 入库数量
		from SDI.SDI_QIS_REBACKLINK_1011 d    -- 跨部门报废
		left join SDI.SDI_qis_modifyoffline_1011 a   -- 改洗下线记录表（即入库条码）
		on d.code=a.barcode 
		and a.status<>'9'
		left join SDI.SDI_qis_mud_1011 b       -- 泥坯
		on a.producttype =b.mudid 
		and b.status<>'9'
		left join SDI.SDI_qis_groutingplan_barcode_1011 c   -- 注浆条码列表
		on a.barcode=c.barcode 
		and c.status<>'9'
		LEFT JOIN SDI.SDI_PRODUCTMODEL_1011  T3
		ON a.producttype = T3.mudid
		where   d.status<>'9' 
		and d.acceptid like 'ZJ%'
		and d.STATUS ='2'
		and d.link in ('010203','010204','釉坯装车','装车')
		AND a.FACTORY IN ('3200')   
		AND d.cjdate between '2022-06-01' and '${PERIOD_ID_D}'
		AND a.productproperty = 'AAAAA'
		 AND b.CATEGORY NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15') 
		 group by substring(d.cjdate,1,7),
		          a.FACTORY,
				  t3.MODELID
		          --c.monitorname,
		          --a.produceline,
		          --a.producttype  
  )T
  left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t1 -- 生产BOM关系数据
  on t.factory_code = t1.factory_code
  and t.material_num = t1.small_material_num
  GROUP BY t.in_stock_date , -- 入库日期
	       t.factory_code, -- 工厂编码
	       --t.material_num --物料编码
	       t1.material_num
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_BASE_YSFJYD_JX_T
SELECT nvl(t1.GRADE_DATE,t2.in_stock_date) AS GRADE_DATE,  -- 时间
       NVL(T1.factory_code,T2.factory_code) AS factory_code, -- 工厂编码
       NVL(T1.material_num,T2.material_num) AS material_num, -- 物料编码
       NVL(T1.YSFJ_QTY,0) AS YSFJ_QTY,  -- 一烧分级数
       NVL(T1.YSYD_QTY,0) AS YSYD_QTY, -- 一烧优等数
       NVL(T2.actual_receipt_qty,0) AS actual_receipt_qty,  -- 收坯数
       CURRENT_TIMESTAMP() AS dw_last_update_date 
FROM one_grade_data_02 t1
FULL outer join JX_ACTUAL_DATA_01 t2
on nvl(t1.GRADE_DATE,'NULL') = nvl(t2.in_stock_date,'NULL')
and NVL(T1.factory_code,'NULL') = NVL(T2.factory_code,'NULL')
and NVL(T1.material_num,'NULL') = NVL(T2.material_num,'NULL');
