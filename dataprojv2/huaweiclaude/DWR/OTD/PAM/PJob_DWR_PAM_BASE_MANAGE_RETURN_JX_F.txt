WITH repeat_grade_data_01 as(  -- 重烧明细数据1
 SELECT fs.fjdate
        ,pm.category
          ,pm.MUDID
        ,fs.promodel
        ,pm.weight
        ,COUNT(DISTINCT fs.uuid) cnt   -- 分级数
        ,COUNT(CASE WHEN fs.prograde in('优等','合格') THEN fs.code END )  yds -- 优等数
   FROM sdi.sdi_productmodel_1011 pm    -- 光瓷产品信息表
    join SDI.SDI_qis_fsfjinfo_1011 fs  ON  fs.promodel = pm.modelid -- 返烧信息
    LEFT  JOIN sdi.sdi_QIS_CLASSIFICATION_1011 C ON C.CODE = FS.CODE AND C.STATUS <> '9'
  WHERE 1 = 1
    AND fs.status <> '9'
    AND ((C.zjproperty in ('正常') -- = '正常'
    AND C.syproperty in ('正常')
    AND C.scproperty in ('正常')) or 
    (C.zjproperty in ('试制') -- = '正常'
    AND C.syproperty in ('试制')
    AND C.scproperty in ('试制')))
    --AND fs.qxclass = '普通缺陷'
    AND fs.factory IN ('3200','3202')
  --  AND fs.fjdate >= '2022-11-10'   AND fs.fjdate <= '2022-11-10'
    AND PM.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')
  GROUP BY fs.fjdate, 
           pm.category, 
           pm.MUDID,
           fs.promodel,
           pm.weight
 UNION ALL 
 SELECT fs.fjdate
        ,pm.category
        ,pm.MUDID
        ,fs.promodel
        ,pm.weight
        ,SUM(fs.fjnum) cnt
        ,SUM(fs.ydnum) yds
   FROM sdi.sdi_productmodel_1011 pm  -- 光瓷产品信息表
       ,SDI.SDI_qis_xjfsinfo_1011 fs  -- 小件产品返烧分级
  WHERE fs.promodel = pm.modelid
    --AND fs.qxclass = '普通缺陷'
    AND fs.factory IN ('3200','3202')
  --  AND fs.fjdate >= '2022-11-10'  AND fs.fjdate <= '2022-11-10'
    AND PM.CATEGORYID NOT IN ('lb05','lb08','lb09','lb11','lb13l','lb15')
  GROUP BY fs.fjdate,
      pm.category, 
      pm.MUDID,
      fs.promodel,
      pm.weight
)
,repeat_grade_data_02 as(  -- 重烧明细数据2
 SELECT t.fjdate AS grand_date, -- 分级日期
     t.category,  -- 产品类别
     t.MUDID AS MUDID_MODEL,   -- 泥坯型号
     t.promodel AS PROD_MODEL, -- 产品型号
     sum(t.cnt) AS grade_qty,  -- 分级数
     sum(t.yds) AS goods_qty,  -- 优等数
     t.weight as weight --重量     
 FROM repeat_grade_data_01 T
 GROUP BY t.fjdate,
       t.category, 
     t.MUDID,
       t.promodel,
       t.weight 
)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_MANAGE_RETURN_JX_F
SELECT grand_date, category, mudid_model, prod_model, grade_qty, goods_qty,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
       weight
FROM repeat_grade_data_02 T;
