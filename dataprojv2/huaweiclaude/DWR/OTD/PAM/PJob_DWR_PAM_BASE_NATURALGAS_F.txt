--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式

WITH QY_TMP_DATA AS( 	-- 单线分表的明细数据，实际为要分摊的数据
	SELECT  T1.shift_date AS START_DATE, -- 日期
			 T1.site AS SITE_CODE, -- 基地编码
			 T1.work_shop AS WORK_SHOP_CODE, -- 车间编码
			 T1.product_line AS product_line_CODE, -- 产线编码
			 T1.work_no_code , -- 工序编码
		     sum(natural_gas_general_qty) AS natural_gas_general_qty -- 天然气总表-用量
	FROM SDI.SDI_dp_natural_gas_consumption_record_info_1006 T1 -- 天然气耗用记录
	WHERE T1.delete_flag = 1
	AND T1.site = '1300'
	AND T1.shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
	GROUP BY T1.shift_date,
			 T1.site ,
			 T1.work_shop ,
			 T1.product_line ,
			 T1.work_no_code 
),
QY_SUM_DATA AS(  -- 清远总表数据
   SELECT meter_reading_date AS START_DATE, -- 日期
	   site AS SITE_CODE, -- 基地编码
	   SUM(NVL(this_day_use_number,0)) AS this_day_use_number -- 当天用量
	FROM SDI.SDI_dp_regulating_natural_gas_1006 -- 调压站天然气记录表
	WHERE delete_flag = 1
	AND site = '1300'
	AND meter_reading_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
	GROUP BY site,
			 meter_reading_date
),
QY_DETAIL_DATA AS( -- 清远分表明细表数据
   SELECT T1.START_DATE, -- 日期
		  T1.SITE_CODE, -- 基地
	      SUM(NVL(T1.natural_gas_general_qty,0)) AS natural_gas_general_qty  -- 天然气总表-用量
	FROM QY_TMP_DATA T1 --  单线分表的明细数据，实际为要分摊的数据
	GROUP BY T1.START_DATE,
		     T1.SITE_CODE
),
QY_YL_DATA AS(  --清远原料数据
  SELECT shift_date AS START_DATE, -- 日期
	   site AS SITE_CODE, -- 基地
	   sum(nvl(natural_gas_consumption,0)) AS natural_gas_consumption -- 天然气用量
	FROM SDI.SDI_dp_fuel_consumption_record_material_workshop_info_1006 -- 原料车间燃料耗用记录表
	WHERE delete_flag = 1
	AND site = '1300'
	AND shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
	GROUP BY shift_date,
			 site
),
SHARE_RATIO_DATA AS( -- 分摊比例数据
	SELECT T1.START_DATE, -- 日期
		   T1.SITE_CODE,  -- 基地编码
		   NVL(T1.this_day_use_number,0) AS this_day_use_number, -- 当天用量
		   NVL(T2.natural_gas_general_qty,0) AS natural_gas_general_qty,  -- 天然气总表-用量
		   NVL(T3.natural_gas_consumption,0) AS natural_gas_consumption, -- 天然气用量(原料)
		   NVL(T1.this_day_use_number,0) - NVL(T2.natural_gas_general_qty,0) - NVL(T3.natural_gas_consumption,0) AS DIFFERENCE_QTY, --差异数据
		   NVL(T2.natural_gas_general_qty,0) + NVL(T3.natural_gas_consumption,0) AS DENOMINATOR_QTY, -- 分母数据（分表总计）
		   CASE WHEN NVL(T2.natural_gas_general_qty,0) + NVL(T3.natural_gas_consumption,0) <> 0 THEN
		     (NVL(T1.this_day_use_number,0) - NVL(T2.natural_gas_general_qty,0) - NVL(T3.natural_gas_consumption,0)) /
		      (NVL(T2.natural_gas_general_qty,0) + NVL(T3.natural_gas_consumption,0)) 
		     ELSE 0 END AS SHARE_RATIO -- 分摊比例
	FROM QY_SUM_DATA T1
	LEFT JOIN QY_DETAIL_DATA T2
	ON T1.SITE_CODE = T2.SITE_CODE
	AND T1.START_DATE = T2.START_DATE
	LEFT JOIN QY_YL_DATA T3
	ON T1.SITE_CODE = T3.SITE_CODE
	AND T1.START_DATE = T3.START_DATE
),
QY_GAS_RESULT_DATA AS( -- 清远气的结果数据
	SELECT T1.START_DATE, -- 日期
		   T1.SITE_CODE, -- 基地
		   T1.WORK_SHOP_CODE, -- 车间
		   T1.PRODUCT_LINE_CODE,  -- 产线
		   T1.WORK_NO_CODE, -- 工序
		   T1.cost_center_CODE, -- 成本中心编码
		   T1.USE_QTY AS GAS_QTY-- 用气量
	FROM(
		SELECT T1.START_DATE, -- 日期
			   T1.SITE_CODE, -- 基地
			   T1.WORK_SHOP_CODE, -- 车间
			   T1.PRODUCT_LINE_CODE,  -- 产线
			   T1.WORK_NO_CODE, -- 工序
			   T3.cost_center AS cost_center_CODE, -- 成本中心编码
			   T1.natural_gas_general_qty,  -- 天然气总表用量
			   T2.SHARE_RATIO * T1.natural_gas_general_qty AS Y_SHARE_QTY, -- 应分摊数
			   (T2.SHARE_RATIO * T1.natural_gas_general_qty) + T1.natural_gas_general_qty AS USE_QTY-- 用气量
		FROM QY_TMP_DATA T1
		LEFT JOIN SHARE_RATIO_DATA T2
		ON T1.START_DATE = T2.START_DATE
		AND T1.SITE_CODE = T2.SITE_CODE
		LEFT JOIN SDI.SDI_dj_gas_costcenter_maintenance_1006 T3 -- 清远-用气与成本中心维护表
		ON T1.SITE_CODE = T3.SITE
		AND T1.WORK_SHOP_CODE = T3.work_shop
		AND T1.PRODUCT_LINE_CODE = T3.product_line
		AND T3.delete_flag = 1
		AND T3.SITE = '1300'
	  ) T1
),
jx_cx_gas_yl_data as ( -- 江西成型用气量数据
	select  t.site,
			t.shift_date ,
			t.work_shop ,
			t.product_line ,
			t.work_no_code ,
			t.name_of_gas ,
			t.gas_name ,
			t.num,
		  SUM (t.num) OVER ( PARTITION BY t.shift_date,t.name_of_gas ) AS total_usenum
	FROM(
		SELECT
			site,
			shift_date ,
			work_shop ,
			product_line ,
			work_no_code ,
			name_of_gas ,
			gas_name ,
			sum(nvl( use_gas_number, 0) + nvl( compensation, 0)) num
		FROM
			sdi.sdi_dj_use_gas_1006
		WHERE
			delete_flag = 1
			AND site = '1500'
			AND shift_date <> ''
			and shift_date is not null
			AND name_of_gas IS NOT NULL
			AND work_shop <> 'CJ_YL1'
		group by
			site,
			shift_date ,
			work_shop ,
			product_line ,
			work_no_code ,
			name_of_gas,
			gas_name
		) T
),
jx_yl_gas_yl_data as ( --江西气原料车间用量数据
	SELECT
		site,
		shift_date ,
		work_shop ,
		product_line ,
		work_no_code ,
		name_of_gas ,
		gas_name,
		sum(nvl( use_gas_number, 0) + nvl( compensation, 0)) ylnum
	FROM
		sdi.sdi_dj_use_gas_1006
	WHERE
		delete_flag = 1
		AND site = '1500'
		AND shift_date <> ''
		and shift_date is not null
		AND name_of_gas IS NOT NULL
		AND work_shop = 'CJ_YL1'
	group by
		site,
		shift_date ,
		work_shop ,
		product_line ,
		work_no_code ,
		name_of_gas,
		gas_name
),
jx_gh_gas_data as ( --港华用量总表
	select
		site,
		shift_date ,
		gas_name ,
		sum(nvl( consumption, 0) + nvl(  compensation, 0)) ghnum
	from
		sdi.sdi_dj_onsumption_of_ganghua_gas_1006
	WHERE
		delete_flag = 1
		and site = '1500'
	group by
		site,
		shift_date ,
		gas_name 
),
jx_gas_cost_center_data as (	--用气成本中心表
	select 
		site, 
		work_shop, 
		work_no_code, 
		name_of_gas_consumption, 
		product_line, 
		cost_center 
	from sdi.sdi_dj_name_of_gas_consumption_1006 
	where delete_flag  = 1 
	and site = '1500' 
),
JX_GAS_RESULT_DATA as( -- 江西用气结果数据
	select t.shift_date AS START_DATE, -- 日期
		   t.site AS SITE_CODE, -- 基地编码
		   t.work_shop AS WORK_SHOP_CODE, -- 车间编码
		   t.product_line AS PRODUCT_LINE_CODE, -- 产线编码
		   t.work_no_code AS work_no_code, -- 工序编码
		   t.cost_center AS COST_CENTER_CODE, -- 成本中心编码
		   sum(t.qty) AS GAS_QTY, -- 用气量
		   sum(case when t.name_of_gas = '焦化气' then t.qty/(3900 / 7900) else 0 end ) AS GAS_QTY_jlq,
		   sum(case when t.name_of_gas = '天然气' then t.qty else 0 end) AS GAS_QTY_trq
	from (
		select
		yl.site,
		yl.shift_date,
		yl.work_shop,
		yl.product_line,
		yl.work_no_code,
		yl.name_of_gas,
		yl.gas_name,
		cc.cost_center,
		case when nvl(yl.total_usenum,0) <> 0 then
		case 
			when yl.name_of_gas = '焦化气' then ((gh.ghnum-nvl( yluse.ylnum,0)) / yl.total_usenum) * yl.num * (3900 / 7900) 	--(（港华-原料） / 成型合计) *成型数量
			else ((gh.ghnum-nvl( yluse.ylnum,0)) / 	yl.total_usenum) * yl.num  
		end 
		else 0 end as qty
	from
		jx_cx_gas_yl_data yl
	left join jx_yl_gas_yl_data yluse 
	on	yl.site = yluse.site
	and yl.name_of_gas = yluse.name_of_gas
	and yl.shift_date = yluse.shift_date
	left join jx_gh_gas_data gh 
	on gh.site = yl.site
	and gh.shift_date=yl.shift_date
	and gh.gas_name = yl.name_of_gas
	left join jx_gas_cost_center_data cc 
	on cc.site= yl.site
	and cc.work_shop= yl.work_shop
	and cc.name_of_gas_consumption = yl.gas_name
	union all 
	select 
		yluse.site,
		yluse.shift_date ,
		yluse.work_shop ,
		yluse.product_line ,
		yluse.work_no_code ,
		yluse.name_of_gas ,
		yluse.gas_name,
		cc.cost_center,
		sum(CASE 
			when yluse.name_of_gas = '焦化气' then yluse.ylnum * 3900 /7900
			else yluse.ylnum
		END) qty
		from jx_yl_gas_yl_data yluse
		left join jx_gas_cost_center_data cc 
		on cc.site= yluse.site
		and cc.work_shop= yluse.work_shop
		and cc.name_of_gas_consumption = yluse.gas_name
	group by 
		yluse.site,
		yluse.shift_date ,
		yluse.work_shop ,
		yluse.product_line ,
		yluse.work_no_code ,
		yluse.name_of_gas ,
		yluse.gas_name,
		cc.cost_center
	) t 
	group by t.site,
			 t.shift_date,
			 t.work_shop,
			 t.product_line,
			 t.work_no_code,
			 t.cost_center
),
HN_GAS_DETAIL_DATA AS(  -- 湖南用气明细数据
	SELECT START_DATE, -- 日期
		   SITE_CODE, -- 基地编码
		   WORK_SHOP_CODE, -- 车间编码
		   WORK_SHOP_QTY, -- 车间用量
		   SUM(WORK_SHOP_QTY) OVER(PARTITION BY START_DATE,SITE_CODE) AS WORK_SHOP_TOTAL_QTY -- 车间总用量
	FROM (
		SELECT  shift_date AS START_DATE, -- 日期
				site AS SITE_CODE, -- 基地编码
			    work_shop AS WORK_SHOP_CODE, -- 车间编码
				SUM(natural_gas_general_qty) AS WORK_SHOP_QTY -- 天然气总表-用量(车间用量)
		FROM SDI.SDI_dp_natural_gas_consumption_record_info_1006 --天然气耗用记录表
		WHERE delete_flag = 1
		AND site = '1600'
		AND shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
		GROUP BY shift_date,
				 site,
			     work_shop
	) T
),
HN_GAS_SUM_DATA AS(  -- 湖南用气汇总数据
	SELECT meter_reading_date AS START_DATE, -- 抄表日期
		   site AS SITE_CODE, -- 基地编码
	       SUM(NVL(this_day_use_number,0)) AS  GAS_DAY_QTY-- 当日用量
	FROM SDI.sdi_dp_regulating_natural_gas_1006  -- 调压站天然气记录表
	WHERE delete_flag = 1
	AND site = '1600'
	AND meter_reading_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
	GROUP BY meter_reading_date,
			 site
),
HN_GAS_DATA AS( -- 湖南气数据
	SELECT T1.START_DATE, -- 日期
	       T1.SITE_CODE, -- 基地编码
		   T1.WORK_SHOP_CODE, -- 车间编码
		   T1.WORK_SHOP_QTY,  -- 车间用量
		   T1.WORK_SHOP_TOTAL_QTY, -- 车间总用量
		   NVL(T2.GAS_DAY_QTY,0) AS GAS_DAY_QTY, -- 当天用量
		   (T1.WORK_SHOP_QTY / T1.WORK_SHOP_TOTAL_QTY) * NVL(T2.GAS_DAY_QTY,0) AS HN_GAS_QTY -- 用气量
	FROM HN_GAS_DETAIL_DATA T1
	LEFT JOIN HN_GAS_SUM_DATA T2
	ON T1.START_DATE = T2.START_DATE
	AND T1.SITE_CODE = T2.SITE_CODE
),
HN_WORK_SHOP_REL_DATA AS( -- 湖南车间关系数据
	SELECT T1.site ,
		   T1.work_shop ,
		   T1.product_line ,
		   T2.work_no_code ,
		   T1.cost_center 
	FROM SDI.SDI_dj_gas_costcenter_maintenance_1006 T1 -- 用气与成本中心维护表
	LEFT JOIN SDI.sdi_dj_powerconsumption_costcenter_maintenance_1006 T2 --  各基地成本中心和车间产线对应关系表
	ON T1.site = T2.site 
	AND T1.work_shop = T2.work_shop 
	AND T1.product_line = T2.product_line 
	AND T1.cost_center = T2.cost_center 
	AND T2.delete_flag = 1
	WHERE T1.delete_flag = 1
	AND T1.site = '1600'
),
HN_GAS_RESULT_DATA AS(  -- 湖南气结果数据  
	SELECT T1.START_DATE, -- 日期
	       T1.SITE_CODE, -- 基地编码
		   T1.WORK_SHOP_CODE, -- 车间编码
		   T2.product_line AS product_line_CODE,-- 产线编码
		   T2.work_no_code, -- 工序编码
		   T2.cost_center AS cost_center_CODE, -- 成本中心编码
		   T1.HN_GAS_QTY AS GAS_QTY -- 用气量
	FROM HN_GAS_DATA T1 -- 湖南气数据
	LEFT JOIN HN_WORK_SHOP_REL_DATA T2 -- 湖南车间关系数据
	ON T1.SITE_CODE = T2.site
	AND T1.WORK_SHOP_CODE = T2.work_shop
),
CQ_GAS_RESULT_DATA AS( -- 重庆气的结果数据
	SELECT t1.START_DATE, -- 使用日期
		   t1.SITE_CODE, -- 基地编码
		   t1.work_shop_code, -- 车间编码
		   t1.product_line_code, -- 产线编码
		   t1.work_no_code, -- 工序编码
		   t1.COST_CENTER_CODE, -- 成本中心编码
		   SUM(t1.gas_qty) AS gas_qty -- 用气量
	FROM DWROTD.DWR_PAM_BASE_GAS_CQ_F T1 -- 瓷砖天然气中间表(重庆)
    WHERE UPPER(t1.gas_code) NOT LIKE '%ZB%' -- 排除天然气为ZB的
    AND t1.START_DATE >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
    GROUP BY start_date,
             site_code,
             work_shop_code,
             product_line_code,
             work_no_code,
             cost_center_code
),
SX_GAS_DETAIL_DATA AS(  -- 山西气明细数据

	GROUP BY T1.shift_date,
			 T2.site,
			 T2.work_shop
),

SX_GAS_RESULT_DATA AS( -- 山西气的结果数据

		   T2.product_line AS product_line_CODE, -- 产线编码
		   T2.work_no_code , -- 工序编码

	UNION ALL
	SELECT start_date,

	FROM JX_GAS_RESULT_DATA -- 江西气的结果数据
	UNION ALL
	SELECT start_date,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code,
		   gas_qty,
		   null as GAS_QTY_jlq,
		   null as GAS_QTY_trq
	FROM HN_GAS_RESULT_DATA -- 湖南气的结果数据
	UNION ALL
	SELECT start_date,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code,
		   gas_qty,
		   null as GAS_QTY_jlq,
		   null as GAS_QTY_trq
	FROM CQ_GAS_RESULT_DATA -- 重庆气的结果数据
	UNION ALL
	SELECT start_date,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code,
		   gas_qty,
		   null as GAS_QTY_jlq,
		   null as GAS_QTY_trq
	FROM SX_GAS_RESULT_DATA -- 山西气的结果数据
),
dim_date as (    -- 日期维度,发散维度用
	select day_dt 
	     from dm.dm_dim_date_d 
	  where day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	    and day_dt >=  CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01') -- 前3月，调度传昨天,格式：YYYY-MM-DD 
),
ACTICAL_DETAIL_DATA AS( -- 实际数据
	SELECT SUBSTRING(START_DATE,1,7) as period_id_m,
		   START_date,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code,
		   SUM(gas_qty) AS gas_qty,
		    SUM(gas_qty_jlq) AS gas_qty_jlq,
		     SUM(gas_qty_trq) AS gas_qty_trq
	FROM GAS_RESULT_DATA T 
	WHERE START_DATE >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
	GROUP BY SUBSTRING(START_DATE,1,7),
			 START_date,
			 site_code,
			 work_shop_code,
			 product_line_code,
			 work_no_code,
			 cost_center_code
),
ACTICAL_DETAIL_DIM_DATA AS( -- 维度数据
	SELECT period_id_m,
		   site_code,
		   work_shop_code,
		   product_line_code,
		   work_no_code,
		   cost_center_code
	FROM ACTICAL_DETAIL_DATA
	GROUP BY period_id_m,
			 site_code,
			 work_shop_code,
			 product_line_code,
			 work_no_code,
			 cost_center_code
),
result_data as(
	SELECT t1.day_dt AS start_date,
		   t2.period_id_m ,
		   t2.site_code ,
		   t2.work_shop_code,
		   t2.product_line_code ,
		   t2.work_no_code ,
		   t2.cost_center_code ,
		   nvl(t3.gas_qty,0) AS gas_qty,
		    nvl(t3.gas_qty_jlq,0) AS gas_qty_jlq,
		     nvl(t3.gas_qty_trq,0) AS gas_qty_trq
	FROM dim_date T1
	LEFT JOIN ACTICAL_DETAIL_DIM_DATA t2
	ON SUBSTRING(T1.day_dt,1,7) = T2.period_id_m  -- 按月发散
	LEFT JOIN ACTICAL_DETAIL_DATA t3
	ON t1.day_dt = t3.START_date
	AND t2.period_id_m = t3.period_id_m
	AND t2.site_code = t3.site_code
	AND t2.work_shop_code = t3.work_shop_code
	AND t2.product_line_code = t3.product_line_code
	AND t2.work_no_code = t3.work_no_code
	AND t2.cost_center_code = t3.cost_center_code
)

	   product_line_code, -- 产线编码
	   work_no_code, -- 工序编码

	   gas_qty_trq,
	    NVL(PERIOD_ID_M,SUBSTRING(START_DATE,1,7)) AS PERIOD_ID_M -- 分区月
