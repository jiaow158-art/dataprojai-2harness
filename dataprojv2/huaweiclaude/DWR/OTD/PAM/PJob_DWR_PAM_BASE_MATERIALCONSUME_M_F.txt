set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
--瓷砖各基地物耗表(月)
WITH SDI_ACDOCA_1001_DATA AS( -- 明细数据
	SELECT PERIOD_ID_M, -- 分区字段
		   T.RBUKRS, -- 基地
		   T.BUDAT , -- 过帐日期
		   T.RACCT, -- 科目号
		   T.RCNTR, -- 成本中心
		   T.MATNR, -- 物料
		   T.SGTXT, -- 文本
		   SUM(NVL(T.TSL,0)) AS MATERIAL_AMT -- 金额
	FROM SDI.SDI_ACDOCA_1001 T -- 会计凭证
	WHERE 1 = 1
	AND T.PERIOD_ID_M >= REPLACE(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-','') -- 前3月，调度传昨天,格式：YYYY-MM-DD 
	AND T.RBUKRS IN ('1300','1500','1600','1800','1900') -- 基地
	AND T.RACCT IN ('0061401000','0061408000','0061406000','0061499000','0060530100') -- 科目
	AND T.RCNTR NOT IN ('1300MC0001','1500MA0055','1600MA0037','1800MA0029') -- 成本中心
	AND T.RFAREA = '4105' -- 功能范围 (生产制造)
	GROUP BY T.PERIOD_ID_M, -- 分区字段
			 T.RBUKRS, -- 基地
			 T.BUDAT , -- 过帐日期
			 T.RACCT, -- 科目号
			 T.RCNTR, -- 成本中心
			 T.MATNR, -- 物料
			 T.SGTXT -- 文本
),
SITE_MATERIAL_SUM_DATA AS(-- 基地汇总数据
	SELECT T.PERIOD_ID_M AS START_MONTH,
		   T.RBUKRS AS SITE_CODE,
		   SUM(NVL(T.MATERIAL_AMT,0)) AS SITE_MATERIAL_AMT -- 基地金额(月)
	FROM SDI_ACDOCA_1001_DATA T  -- 明细数据
	GROUP BY T.PERIOD_ID_M,
		 	 T.RBUKRS
),
SDI_ZTCODG007_INTYF_1001_DATA AS( -- 分段成本还原表
	SELECT T.LFGJA, -- 年（2024）
		   T.LFMON,	-- 本期（04）
		   T.ZCPDJ, -- 产品等级
		   T.AUFNR, -- 订单
		   T.WERKS, -- 基地
		   T.VERID, -- 生产版本
		   T.ZGXD, -- 工序段
		   T.ZCLASS, -- 品类
		   T.ZSERIAL, -- 系列
		   T.ZHOUDU, -- 规格厚度
		   T.GROES, -- 规格
		   T.ZDWMJ, -- 单位面积
		   T.PLNBEZ, -- 物料
		   T.MEINS, -- 基本计量单位
		   T.ZCB_SUM, -- 成本合计
		   T.ZAREA, -- 面积(M2单位)
		   T.BSTMG, -- 订单产量(PC单位)
		   NVL(T.ZWLXH1,0)  AS ZWLXH1,-- 原料段-物料消耗
		   NVL(T.ZWXF1,0) AS ZWXF1,-- 原料段-维修费
		   NVL(T.ZWLXH2,0) AS ZWLXH2,-- 釉料段-物料消耗
		   NVL(T.ZWXF2,0) AS ZWXF2,-- 釉料段-维修费
		   NVL(T.ZWLXH3,0) AS ZWLXH3, -- 成型段-物料消耗
		   NVL(T.ZWXF3,0) AS ZWXF3-- 成型段-维修费
	FROM SDI.SDI_ZTCODG007_INTYF_1001 T -- 分段成本还原本期结存-研发费用
	WHERE T.del_flag = 'N'
	AND T.WERKS IN ('1300','1500','1600','1800','1900') -- 基地
	AND SUBSTRING(PLNBEZ,-2) IN ('_A','_B')  -- 限制物料为_A、_B
	AND CONCAT(lfgja,CASE WHEN LENGTH(lfmon) < 2 THEN CONCAT('0',lfmon) ELSE lfmon END) >= REPLACE(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-','') -- 前3月，调度传昨天,格式：YYYY-MM-DD
),
SITE_COST_RETURN_MATERIAL_DATA AS( -- 基地还原成本物耗金额
	SELECT CONCAT(lfgja,CASE WHEN LENGTH(lfmon) < 2 THEN CONCAT('0',lfmon) ELSE lfmon END) AS START_MONTH, -- 月份
		   werks AS SITE_CODE, -- 基地
		   SUM(NVL(T.ZAREA,0)) AS ZAREA, -- 面积(M2单位)
		   CASE WHEN SUM(NVL(T.ZAREA,0)) <> 0 THEN SUM(NVL(T.ZWLXH1,0) + NVL(T.ZWXF1,0) + NVL(T.ZWLXH2,0) + NVL(T.ZWXF2,0) + NVL(T.ZWLXH3,0) + NVL(T.ZWXF3,0))
		   ELSE 0 END  AS COST_RETRUN_MATERIAL_AMT  -- 成本还原物耗金额
	FROM SDI_ZTCODG007_INTYF_1001_DATA T
	WHERE SUBSTRING(VERID,1,1) = 'R'  -- 只取生产版本为R
	GROUP BY lfgja,
			 lfmon,
			 werks
),
DP_DEDUCT_BALLDAN_AMOUNT_MAINT_1006_DATA AS( --  扣减球旦金额维护表
	SELECT REPLACE(SUBSTRING(budat,1,7),'-','') AS START_MONTH, -- 月份
		   site AS SITE_CODE, -- 基地编码
		   cost_center AS cost_center_code, -- 成本中心
		   sum(amount) AS KJ_QD_AMT -- 扣减球旦金额
	FROM SDI.SDI_DP_DEDUCT_BALLDAN_AMOUNT_MAINT_1006 --  扣减球旦金额维护表
	WHERE delete_flag = 1
	AND budat >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM-DD
	AND site IN ( '1300','1500','1600','1800','1900' )  -- 基地
	GROUP BY SUBSTRING(budat,1,7),
			 site,
			 cost_center
),
qd_detail_data as( -- 球旦的明细数据，含数采部分的
	SELECT T.PERIOD_ID_M AS START_MONTH, -- 分区字段
		   T.RBUKRS AS SITE_CODE, -- 基地
		   T.RCNTR  AS cost_center_code, -- 成本中心
		   SUM(MATERIAL_AMT) AS QD_MATERIAL_AMT -- 物耗金额
	FROM SDI_ACDOCA_1001_DATA t  -- 会计凭证的球旦数据
	WHERE 1 = 1
	AND ((t.RBUKRS IN ('1300','1900') AND t.PERIOD_ID_M >= '202405') -- 1300/1900，只取202405以后的，非1300/1900，只取202401以后的  
	  OR (t.RBUKRS NOT IN ('1300','1900') AND t.PERIOD_ID_M >= '202401'))
	AND  (T.sgtxt LIKE '%球旦%' OR T.sgtxt LIKE '%球石%' OR T.sgtxt LIKE '%自动打包%' OR T.sgtxt LIKE '%系统数不考核%')  --  只取 球旦、球石、自动打包、系统数不考核
	GROUP BY T.PERIOD_ID_M, -- 分区字段
			 T.RBUKRS, -- 基地
			 T.RCNTR -- 成本中心
	UNION ALL
	SELECT START_MONTH, -- 月份
		   SITE_CODE, -- 基地编码
		   cost_center_code, -- 成本中心
		   KJ_QD_AMT AS QD_MATERIAL_AMT -- 扣减球旦金额
	FROM DP_DEDUCT_BALLDAN_AMOUNT_MAINT_1006_DATA  --  扣减球旦金额维护表
),
CY_DETAIL_DATA01 AS( -- 差异值计算（这是第一次剔除球旦，只是为了计算差异值）
	SELECT T1.START_MONTH,  -- 月份
		   T1.SITE_CODE,  -- 基地
		   NVL(T1.COST_RETRUN_MATERIAL_AMT,0) AS COST_RETRUN_MATERIAL_AMT, -- 成本还原金额
		   NVL(T2.SITE_MATERIAL_AMT,0) AS SITE_MATERIAL_AMT,  -- 含球旦金额
		   NVL(T1.COST_RETRUN_MATERIAL_AMT,0) - NVL(T2.SITE_MATERIAL_AMT,0) AS MATERIAL_AMT  -- 物耗金额
	FROM SITE_COST_RETURN_MATERIAL_DATA T1 -- 基地还原成本物耗金额
	LEFT JOIN SITE_MATERIAL_SUM_DATA T2-- 基地汇总数据
	ON T1.START_MONTH = T2.START_MONTH
	AND T1.SITE_CODE = T2.SITE_CODE
),
SDI_ACDOCA_1001_COST_DATA AS( -- acdoca的成本中心数据
	SELECT PERIOD_ID_M AS START_MONTH, -- 月份
		   RBUKRS AS SITE_CODE, -- 基地
		   RCNTR AS COST_CENTER_CODE, -- 成本中心
		   SUM(NVL(MATERIAL_AMT,0)) AS MATERIAL_AMT -- 金额
	FROM SDI_ACDOCA_1001_DATA
	GROUP BY PERIOD_ID_M,
			 RBUKRS,
			 RCNTR
),
DP_RM_WORKSHOP_COSTCENTER_MAINT_DATA AS( --  原料车间成本中心数据
	SELECT DISTINCT site AS SITE_CODE,  -- 基地编码
			    cost_center AS COST_CENTER_CODE -- 成本中心编码
	FROM SDI.SDI_DP_RM_WORKSHOP_COSTCENTER_MAINT_1006 --  原料车间成本中心维护表
	WHERE delete_flag = 1
),
NO_YL_WORKSHOP_DATA AS(  -- 只取原料车间后的数据
	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.MATERIAL_AMT -- 金额
	FROM SDI_ACDOCA_1001_COST_DATA T1 -- acdoca的成本中心数据
	JOIN DP_RM_WORKSHOP_COSTCENTER_MAINT_DATA T2 --  原料车间成本中心数据
	ON T1.SITE_CODE = T2.SITE_CODE
	AND T1.COST_CENTER_CODE = T2.COST_CENTER_CODE
),
TMP_01 AS( -- 计算剔除球旦金额明细，在1)步取出的金额字段减去第2）步取出的球旦金额字段得到”剔除球旦金额明细“
	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.MATERIAL_AMT, -- 金额
		   NVL(T2.QD_MATERIAL_AMT,0) AS QD_MATERIAL_AMT, -- 球旦物耗金额
		   T1.MATERIAL_AMT - NVL(T2.QD_MATERIAL_AMT,0) AS NO_QD_MATERIAL_AMT -- 剔除球旦金额明细
	FROM NO_YL_WORKSHOP_DATA t1 -- 只取原料车间后的数据
	LEFT JOIN qd_detail_data t2 -- 球旦的明细数据，含数采部分的
	ON t1.START_MONTH = t2.START_MONTH
	AND t1.SITE_CODE = t2.SITE_CODE
	AND t1.COST_CENTER_CODE = t2.COST_CENTER_CODE
),
TMP_02 AS( -- 计算剔除球旦金额汇总，按工厂，月份，汇总第3）步明细的”剔除球旦金额“字段
	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.MATERIAL_AMT, -- 金额
		   T1.QD_MATERIAL_AMT, -- 球旦物耗金额
		   T1.NO_QD_MATERIAL_AMT, -- 剔除球旦金额明细
		   SUM(T1.NO_QD_MATERIAL_AMT) OVER(PARTITION BY T1.START_MONTH,T1.SITE_CODE) AS NO_QD_SUM_MATERIAL_AMT -- 剔除球旦金额
	FROM TMP_01 T1
),
TMP_03 AS( -- 计算原料成本中心金额占比，用第3）明细除以4）汇总得到占比
   	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.MATERIAL_AMT, -- 金额
		   T1.QD_MATERIAL_AMT, -- 球旦物耗金额
		   T1.NO_QD_MATERIAL_AMT, -- 剔除球旦金额明细
		   T1.NO_QD_SUM_MATERIAL_AMT, -- 剔除球旦金额
		   --CASE WHEN T1.NO_QD_SUM_MATERIAL_AMT <> 0 THEN CAST(T1.NO_QD_MATERIAL_AMT  / T1.NO_QD_SUM_MATERIAL_AMT AS DECIMAL(38,12)) ELSE 0 END RATIO_VALUE -- 比例
		   CASE WHEN T1.NO_QD_SUM_MATERIAL_AMT <> 0 THEN CAST(CAST(T1.NO_QD_MATERIAL_AMT AS STRING) / CAST(T1.NO_QD_SUM_MATERIAL_AMT AS STRING) AS DECIMAL(38,12)) ELSE 0 END RATIO_VALUE -- 比例
	FROM TMP_02 T1
),
TMP_04 AS( -- 关联差异值，通过年，月，工厂和E表关联，取差异值字段,算出应调整值
	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地
		   T1.COST_CENTER_CODE, -- 成本中心
		   T1.MATERIAL_AMT, -- 金额
		   T1.QD_MATERIAL_AMT, -- 球旦物耗金额
		   T1.NO_QD_MATERIAL_AMT, -- 剔除球旦金额明细
		   T1.NO_QD_SUM_MATERIAL_AMT, -- 剔除球旦金额
		   T1.RATIO_VALUE, -- 比例
		   NVL(t2.MATERIAL_AMT,0) AS CY_MATERIAL_AMT, -- 物耗金额
		   T1.RATIO_VALUE * NVL(t2.MATERIAL_AMT,0) AS tz_material_amt -- 应调整物耗金额
	FROM TMP_03 t1
	LEFT JOIN CY_DETAIL_DATA01 t2 -- 差异值计算（这是第一次剔除球旦，只是为了计算差异值）
	ON t1.START_MONTH = t2.START_MONTH
	AND t1.SITE_CODE = t2.SITE_CODE
),
TMP_05 AS ( -- 写进过账日期字段，按表中月份取月末最后一天作为过账日期,及拼接凭证原始数据
	SELECT T1.START_MONTH, -- 月份
		   T1.SITE_CODE, -- 基地编码
		   T2.day_mend_dt AS ACCOUNT_DATE, -- 过账日期
		   '0061499000' AS SUBJECT_CODE_CODE, -- 科目
		   T1.COST_CENTER_CODE, -- 成本中心编码
		   '' AS MATERIAL_CODE, -- 物料编码
		   '调整值' AS TEXTS, -- 文本
		   T1.TZ_MATERIAL_AMT AS MATERIAL_AMT -- 应调整数
	FROM TMP_04 t1
	LEFT JOIN (SELECT DISTINCT month_id,day_mend_dt FROM DM.dm_dim_date_d WHERE day_dt <= current_date()) T2 -- 日期维度
	ON T1.START_MONTH = T2.month_id
	UNION ALL
	SELECT T.PERIOD_ID_M AS START_MONTH, -- 分区字段
		   T.RBUKRS AS SITE_CODE, -- 基地
		   T.BUDAT AS ACCOUNT_DATE, -- 过帐日期
		   T.RACCT AS SUBJECT_CODE_CODE, -- 科目号
		   T.RCNTR AS COST_CENTER_CODE, -- 成本中心
		   T.MATNR AS MATERIAL_CODE, -- 物料
		   T.SGTXT AS TEXTS, -- 文本
		   T.MATERIAL_AMT AS MATERIAL_AMT -- 金额
	FROM SDI_ACDOCA_1001_DATA T -- 会计凭证
),
NO_PACKAGE_data as( -- 剔除的数据，不含自动打包
	SELECT START_MONTH, -- 月份
		   SITE_CODE, -- 基地编码
		   cost_center_code, -- 成本中心
	 	   sum(QD_MATERIAL_AMT) AS QD_MATERIAL_AMT -- 物耗金额
	FROM(
		SELECT T.PERIOD_ID_M AS START_MONTH, -- 分区字段
			   T.RBUKRS AS SITE_CODE, -- 基地
			   T.RCNTR  AS cost_center_code, -- 成本中心
			   SUM(MATERIAL_AMT) AS QD_MATERIAL_AMT -- 物耗金额
		FROM SDI_ACDOCA_1001_DATA t  -- 会计凭证的球旦数据
		WHERE 1 = 1
		AND ((t.RBUKRS IN ('1300','1900') AND t.PERIOD_ID_M >= '202405') -- 1300/1900，只取202405以后的，非1300/1900，只取202401以后的  
		  OR (t.RBUKRS NOT IN ('1300','1900') AND t.PERIOD_ID_M >= '202401'))
		AND  (T.sgtxt LIKE '%球旦%' OR T.sgtxt LIKE '%球石%' OR T.sgtxt LIKE '%系统数不考核%')  --  只取 球旦、球石、系统数不考核
		GROUP BY T.PERIOD_ID_M, -- 分区字段
				 T.RBUKRS, -- 基地
				 T.RCNTR -- 成本中心
		UNION ALL
		SELECT START_MONTH, -- 月份
			   SITE_CODE, -- 基地编码
			   cost_center_code, -- 成本中心
			   KJ_QD_AMT AS QD_MATERIAL_AMT -- 扣减球旦金额
		FROM DP_DEDUCT_BALLDAN_AMOUNT_MAINT_1006_DATA  --  扣减球旦金额维护表
		) t
		GROUP BY START_MONTH, -- 月份
			     SITE_CODE, -- 基地编码
			     cost_center_code -- 成本中心
),
TMP_06 AS(   -- 按月份基地成本中心汇总
	SELECT start_month,
		   site_code,
		   cost_center_code,
		   SUM(material_amt) AS material_amt
	FROM TMP_05
	GROUP BY start_month,
		     site_code,
		     cost_center_code
),
tmp_07 AS ( -- 剔除非自动打包数据
	SELECT t1.start_month,
		   t1.site_code,
		   t1.cost_center_code,
		   t1.material_amt - NVL(t2.QD_MATERIAL_AMT,0) AS material_amt
	FROM TMP_06 T1
	LEFT JOIN NO_PACKAGE_data t2
	ON t1.START_MONTH = t2.START_MONTH
	AND t1.SITE_CODE = t2.SITE_CODE
	AND t1.cost_center_code = t2.cost_center_code
),
tmp_08 AS( -- 物耗结果数据，没有发散前
	SELECT CONCAT(SUBSTRING(t1.start_month,1,4),'-',SUBSTRING(t1.start_month,5,2)) AS start_month,  -- 月份
		   t1.site_code,   -- 基地编码
		   t2.work_shop AS work_shop_code, -- 车间编码
		   t2.product_line AS product_line_code, -- 产线编码
		   t2.work_no_code, -- 工序编码
		   t1.cost_center_code, -- 成本中心编码
		   t1.material_amt,  -- 物耗金额
		   --'N' AS del_flag, -- 删除数据标识
		   --CURRENT_TIMESTAMP() AS dw_last_update_date , -- 数据同步时间
		   CONCAT(SUBSTRING(t1.start_month,1,4),'-',SUBSTRING(t1.start_month,5,2)) AS PERIOD_ID_M -- 分区月
	FROM tmp_07 T1  -- 按月份、基地、成本中心汇总金额
	LEFT JOIN SDI.SDI_dj_powerconsumption_costcenter_maintenance_1006 T2 -- 清远-用电与成本中心维护表
	ON T1.SITE_CODE = T2.SITE
	AND T1.COST_CENTER_CODE = T2.COST_CENTER
	AND T2.delete_flag = 1
),
dim_date_data as (    -- 日期维度,发散维度用
	select  DISTINCT concat(substring(month_id,1,4),'-',substring(month_id,5,2)) AS start_month
	     from dm.dm_dim_date_d 
	  where day_dt <=SUBSTR('${PERIOD_ID_D}',1,10) 
	    and day_dt >=  CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01') -- 前3月，调度传昨天,格式：YYYY-MM-DD 
),
dim_data as( -- 维度
	SELECT
		DISTINCT site AS site_code,
		work_shop AS work_shop_code,
		product_line AS product_line_code,
		work_no_code,
		cost_center AS cost_center_code
	FROM SDI.SDI_dj_powerconsumption_costcenter_maintenance_1006 -- 各基地成本关系表
	WHERE delete_flag = 1
),
dim_date_result_data as( 
	SELECT t1.start_month,
		   t2.site_code,
		   t2.work_shop_code,
		   t2.product_line_code,
		   t2.work_no_code,
		   t2.cost_center_code
	FROM dim_date_data t1
	LEFT JOIN dim_data t2
	ON 1 = 1
)
INSERT OVERWRITE TABLE DWROTD.DWR_PAM_BASE_MATERIALCONSUME_M_F PARTITION(PERIOD_ID_M)
SELECT t1.start_month,  -- 月份
	   t1.site_code,  -- 基地编码
	   t1.work_shop_code,  -- 车间编码
	   t1.product_line_code,  -- 产线编码
	   t1.work_no_code,  -- 工序编码
	   t1.cost_center_code, -- 成本中心编码
	   nvl(t2.material_amt,0) AS material_amt,
	   'N' AS del_flag, -- 删除数据标识
	   CURRENT_TIMESTAMP() AS dw_last_update_date , -- 数据同步时间
	   t1.start_month AS period_id_m -- 分区月
FROM dim_date_result_data t1
LEFT JOIN tmp_08 t2
ON t1.start_month = t2.start_month
AND t1.site_code = t2.site_code
AND t1.work_shop_code = t2.work_shop_code
AND t1.product_line_code =  t2.product_line_code
AND t1.work_no_code = t2.work_no_code
AND t1.cost_center_code = t2.cost_center_code
;
