with tmp as (
select  
 T.SITE as site_code
,T.WORKSHOP as work_shop_code
,T.PRODUCT_LINE as PRODUCT_LINE_CODE
,T.TEAM as group_code
,T.WORK_NO_CODE 
,T.DATES as SHIFT_DATE
,T.MATERIAL_CODE as PRODUCT_CODE
,T.SLICE_NUM 
,T.CREATE_BY as creater
,T.CREATE_DATE as create_time
,T.LAST_UPDATE_BY as updater
,T.LAST_UPDATE_DATE as update_time
from sdi.sdi_d_technical_1006 T  --技术部中试（研发）用坯记录表
where delete_flag = 1 and site = '1800' --重庆
and DATES >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select  --釉线交接班记录表
 T.site as site_code
,T.work_shop as work_shop_code
,T.product_line as product_line_code
,T.group_code as group_code
,T.work_no_code as work_no_code
,T.shift_date as shift_date
,T.PRODUCT_CODE -- '产品编号'
,T.pilot_brick as SLICE_NUM --中试砖
,T.creater -- '创建人姓名'
,T.create_time -- '创建时间'
,T.updater -- '修改人姓名'
,T.update_time -- '修改时间'
from SDI.SDI_dh_glaze_line_handover_1006 T
WHERE T.delete_flag = 1 and T.site = '1600' --湖南
and T.shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select 
 T.site as site_code
,T.code_of_the_collecting_workshop as work_shop_code
,T.product_line as product_line_code
,T.group_code as group_code
,T.work_no_code as work_no_code
,T.shift_date as shift_date
,T.product_number as PRODUCT_CODE -- '产品编号'
,T.number as SLICE_NUM 
,T.creater -- '创建人姓名'
,T.create_time -- '创建时间'
,T.updater -- '修改人姓名'
,T.update_time -- '修改时间'
from sdi.sdi_dj_medium_trial_embryo_1006 T --技术部中试(研发)用坯
WHERE T.delete_flag = 1 and T.site in ('1500','1900') --江西、山西
and T.shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
union all 
select 
 T.site as site_code
,T.work_shop as work_shop_code
,T.product_line as product_line_code
,T.group_code as group_code
,T.work_no_code as work_no_code
,T.shift_date as shift_date
,T.product_code as PRODUCT_CODE -- '产品编号'
,T.production as SLICE_NUM 
,T.creater -- '创建人姓名'
,T.create_time -- '创建时间'
,T.updater -- '修改人姓名'
,T.update_time -- '修改时间'
from sdi.sdi_dp_press_work_record_info_1006 T --清远中试跟压机放一起
WHERE T.delete_flag = 1 and T.site = '1300' and T.trial_production = '0'--是否试制0是1否
and T.shift_date >= CONCAT(SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7),'-01')
)
insert overwrite table DWROTD.DWR_PAM_BASE_BRICK_USE_LOSS_F partition(period_id_m)
select 
 T.site_code
,T.work_shop_code
,T.PRODUCT_LINE_CODE
,T.group_code
,T.WORK_NO_CODE 
,substr(T.SHIFT_DATE,1,10) as SHIFT_DATE
,T.PRODUCT_CODE
,T.SLICE_NUM 
,T1.UNIT_AREA --单位面积
,T.creater
,T.create_time
,T.updater
,T.update_time
,'N' AS DEL_FLAG  --  '删除标识'
,current_timestamp () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,SUBSTR(T.SHIFT_DATE,1,7) AS period_id_m
FROM TMP T
left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T1
on T.PRODUCT_CODE = T1.material_num
