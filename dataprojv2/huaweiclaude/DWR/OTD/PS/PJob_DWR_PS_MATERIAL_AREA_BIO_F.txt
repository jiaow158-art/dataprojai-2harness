---SAP物料产区归属表
set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_PS_MATERIAL_AREA_BIO_F	--DWR.SAP物料产区归属表
SELECT		--查询
  DPMAB.GROUP_NUM AS GROUP_NUM,	--集团
  DPMAB.FACTORY_AREA_CODE AS FACTORY_AREA_CODE,	--产区编码
  DPMAB.FACTORY_WERKS_ID AS FACTORY_WERKS_ID,	--工厂ID
  DPMAB.FACTORY_WERKS_CODE AS FACTORY_WERKS_CODE,	--工厂编码
  T.FACTORY_NAME AS FACTORY_NAME,	--工厂名称
  DPMAB.FACTORY_WERKS2_ID AS FACTORY_WERKS2_ID,	--工厂ID
  DPMAB.FACTORY_WERKS2_CODE AS FACTORY_WERKS2_CODE, --工厂编码
  T1.FACTORY_NAME AS FACTORY_NAME,	--工厂名称
  DPMAB.MATERIAL_ID AS MATERIAL_ID,	--物料ID
  DPMAB.MATERIAL_NUM AS MATERIAL_NUM, --物料编号
  T2.MATERIAL_NAME AS MATERIAL_NAME,	--物料名称
  DPMAB.SOURCE_MATERIAL_NUM AS SOURCE_MATERIAL_NUM,	--源物料编号
  DPMAB.DEL_FLAG AS DEL_FLAG,	--删除标识
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE	--DW最后更新时间
FROM DWIOTD.DWI_PS_MATERIAL_AREA_BIO_T DPMAB	--DWI.SAP物料产区归属表
  INNER JOIN DWIMD.DWI_MD_FACTORY_T T 	--关联工厂主数据取工厂名称
  ON T.FACTORY_ID = DPMAB.FACTORY_WERKS_ID	--工厂ID
  AND T.GROUP_NUM = DPMAB.GROUP_NUM		--集团
  AND T.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
  AND DPMAB.DEL_FLAG = 'N'		--删除标识为'N'
  INNER JOIN DWIMD.DWI_MD_FACTORY_T T1 	--关联工厂主数据取工厂名称
  ON T1.FACTORY_ID = DPMAB.FACTORY_WERKS2_ID	--工厂ID
  AND T1.GROUP_NUM = DPMAB.GROUP_NUM	--集团
  AND T1.END_DATE = '4712-12-31'	--结束时间为'4712-12-31'
  AND DPMAB.DEL_FLAG = 'N'	--删除标识为'N'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T2 	--关联物料主数据取物料ID
  ON T2.MATERIAL_ID = DPMAB.MATERIAL_ID		--物料ID
  AND T2.GROUP_NUM = DPMAB.GROUP_NUM		--集团
  AND T2.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
  AND DPMAB.DEL_FLAG = 'N'		--删除标识为'N'
WHERE DPMAB.DEL_FLAG = 'N'
