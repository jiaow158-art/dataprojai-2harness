INSERT OVERWRITE TABLE DWROTD.DWR_PS_PRCH_ORDER_F
SELECT A.ID
      ,A.PURCHASE_REQUISITION_NO AS APLY_NUM --申请单号
	  ,A.PURCHASE_ORDER_NO AS PRCH_NUM --采购订单号
	  ,A.PURCHASE_ORDER_PROJ_NO AS PRCH_LINE_NUM --采购订单行号
	  ,A.APS_PLAN_TASK_ID AS PLAN_TASK_ID --排产计划ID
	  ,A.APS_PLAN_TASK_DTL_ID AS PLAN_TASK_DTL_ID --排产计划明细ID
	  ,A.APS_PROTECH_ORDER_ID AS PRCS_TASK_ID --制程工艺ID
	  ,A.SUPPLIER_ID AS SPLR_ID --供应商ID
	  ,B.CODE AS SPLR_CODE --供应商编码
	  ,B.NAME AS SPLR_NAME --供应商名称
	  ,A.REQUIRE_DELIVERY AS RQST_DELV_DATE --要求交期
	  ,A.REPLY_DELIVERY AS REPLY_DELV_DATE --回复交期
	  ,A.REMARK --备注
	  ,A.BIBULOUS_RATE AS BIBU_RATE_CODE --吸水率编码
	  ,CASE WHEN A.BIBULOUS_RATE = 10 THEN 'E 〉10'
	        WHEN A.BIBULOUS_RATE = 20 THEN '3%〈 E≤6%'
			WHEN A.BIBULOUS_RATE = 30 THEN '6%〈 E≤10%'
			WHEN A.BIBULOUS_RATE = 40 THEN '0.5%〈 E≤3%'
			WHEN A.BIBULOUS_RATE = 50 THEN 'E≤0.5%'
			 END AS BIBU_RATE_NAME --吸水率描述
	  ,A.ACCEPTANCE_CRITERIA AS ACC_CRITA_CODE --验收标准编码
	  ,CASE WHEN A.ACCEPTANCE_CRITERIA = 10 THEN '东鹏A类标准'
	        WHEN A.ACCEPTANCE_CRITERIA = 20 THEN '东鹏C类标准'
			WHEN A.ACCEPTANCE_CRITERIA = 30 THEN '国标'
			WHEN A.ACCEPTANCE_CRITERIA = 40 THEN '供应商标准'
			WHEN A.ACCEPTANCE_CRITERIA = 50 THEN '双方签板标准'
			 END AS ACC_CRITA_NAME --验收标准描述
      ,A.ADOBE_BODY AS ADOBE_BODY_CODE --坯体编码
	  ,CASE WHEN A.ADOBE_BODY = 10 THEN '普通坯'
	        WHEN A.ADOBE_BODY = 20 THEN '色坯'
			WHEN A.ADOBE_BODY = 30 THEN '通体'
			WHEN A.ADOBE_BODY = 40 THEN '通体布料'
			WHEN A.ADOBE_BODY = 50 THEN '斑点坯'
			 END AS ADOBE_BODY_NAME --坯体描述
	  ,A.PACKAGE_WAY AS PACK_TYPE_CODE --包装方式编码
	  ,CASE WHEN A.PACKAGE_WAY = 10 THEN '东鹏常规纸箱'
	        WHEN A.PACKAGE_WAY = 20 THEN '东鹏旧版纸箱'
			WHEN A.PACKAGE_WAY = 30 THEN '世界印象纸箱'
			WHEN A.PACKAGE_WAY = 40 THEN '梦之家纸箱'
			WHEN A.PACKAGE_WAY = 50 THEN 'ART+纸箱'
			WHEN A.PACKAGE_WAY = 60 THEN '中性纸箱'
	        WHEN A.PACKAGE_WAY = 70 THEN '供应商纸箱'
			WHEN A.PACKAGE_WAY = 80 THEN '东鹏零售纸箱'
			WHEN A.PACKAGE_WAY = 90 THEN '东鹏工程纸箱'
			WHEN A.PACKAGE_WAY = 100 THEN '东鹏致美家纸箱'
			 END AS PACK_TYPE_NAME --包装方式描述
      ,COALESCE(A.STATUS,10) AS PRCH_ST_CODE --采购计划状态
	  ,CASE WHEN A.STATUS = 10 THEN '未处理'
	        WHEN A.STATUS = 20 THEN '审批中'
			WHEN A.STATUS = 30 THEN '审批退回'
			WHEN A.STATUS = 40 THEN '待回复交期'
			WHEN A.STATUS = 50 THEN '已回复交期'
			WHEN A.STATUS = 60 THEN '已确认并转单'
			WHEN A.STATUS = 70 THEN '部分交付'
			WHEN A.STATUS = 80 THEN '已完成交付'
			WHEN A.STATUS = 90 THEN '超期审批中'
			WHEN A.STATUS = 95 THEN '特批退回'
			WHEN A.STATUS = 110 THEN '已关闭'
	        ELSE '未处理'
			 END AS PRCH_ST_NAME --采购计划状态描述
	  ,A.OA_ID AS OA_RQST_ID --OA审批ID
	  ,A.OA_NO AS OA_RQST_NUM --OA审批流水号
	  ,A.OA_NODE AS OA_RQST_NODE --OA审批节点
	  ,A.OA_DATE AS OA_RQST_DATE --OA申请日期
	  ,A.ALREADY_LEVITE_QTY AS ALRD_WHSE_QTY --已入仓数
      ,A.PURCHASE_VOUCHER_TYPE AS PRCH_VCHR_TYPE_CODE --采购凭证类型
      ,A.PURCHASE_VOUCHER AS PRCH_VCHR_NUM --采购凭证号
	  ,A.TMS_REMARK --TMS确认信息
	  ,A.IS_TMS AS IS_TMS_CODE --是否可以TMS确认
	  ,CASE WHEN A.IS_TMS = 0 THEN '否'
	        WHEN A.IS_TMS = 1 THEN '是'
			 END AS IS_TMS_NAME --是否可以TMS确认描述
	  ,A.NOT_LEVITE_QTY AS UN_WHSE_QTY --未入仓数
	  ,A.BOTTOM_LABEL AS BOT_LBL --底标
	  ,A.CLOSE_BILL_REASON AS CLSE_RSN_CODE --关单原因编码
	  ,A.CLOSE_BILL_REASON_TXT AS CLSE_RSN_DESC --关单原因描述
	  ,A.BASE_CODE AS BASE_NAME --基地
	  ,A.ASSIGN_REASON AS ASGN_RSN_DESC --分配原因描述
	  ,A.LAST_OA_ID AS LAST_OA_RQST_ID --上一回OA审批ID
      ,A.LAST_OA_NO AS LAST_OA_RQST_NUM --上一回OA审批流水号
	  ,A.LAST_OA_NODE AS LAST_OA_RQST_NODE --上一回OA审批节点
	  ,A.LAST_OA_DATE AS LAST_OA_RQST_DATE --上一回OA申请日期	  
      ,A.PURCHASE_ORG_ID AS PRCH_ORG_ID --采购组织ID
	  ,C.EKORG AS PRCH_ORG_CODE --采购组织编码
	  ,C.EKOTX AS PRCH_ORG_NAME --采购组织名称
	  ,A.PURCHASE_GROUP_ID AS PRCH_GRP_ID --采购组ID
	  ,E.EKGRP AS PRCH_GRP_CODE --采购组编码
	  ,E.EKNAM AS PRCH_GRP_NAME --采购组描述
	  ,A.APS_INVENTORY_LOCATION_ID AS WH_ID --仓库ID
	  ,D.LGORT AS WH_CODE --仓库编码
	  ,D.LGOBE AS WH_NAME --仓库名称
	  ,A.SAP_COMPANY_ID AS CO_ID --公司ID
	  ,F.BUKRS AS CO_CODE --公司编码
	  ,F.BUTXT AS CO_NAME --公司名称
	  ,A.SAP_STOCK_AREA_ID AS FACTORY_ID --工厂ID
	  ,G.WERKS AS FACTORY_CODE --工厂编码
	  ,G.NAME AS FACTORY_NAME --工厂名称
	  ,A.SECOND_REPLY_DELIVERY_STATE AS SEC_REPLY_DELV_ST_CODE --二次回复交期有效编码
	  ,CASE WHEN A.SECOND_REPLY_DELIVERY_STATE = 0 THEN '禁用'
	        WHEN A.SECOND_REPLY_DELIVERY_STATE = 1 THEN '可用'
			 END AS SEC_REPLY_DELV_ST_NAME --二次回复交期有效描述
	  ,A.SECOND_REPLY_DELIVERY AS SEC_REPLY_DELV_DATE --二次回复交期
	  ,A.RES_WIP_SCHE_ID AS PROD_LINE_ID --生产资源ID
	  ,H.CODE AS PROD_LINE_CODE --生产资源编码
	  ,H.NAME AS PROD_LINE_NAME --生产资源名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_APS_OEM_PO_SUM_1007 A
  LEFT JOIN SDI.SDI_SUPPLIER_1007 B
    ON A.SUPPLIER_ID = B.ID
  LEFT JOIN SDI.SDI_APS_PURCHASE_ORG_1007 C
    ON A.PURCHASE_ORG_ID = C.ID
  LEFT JOIN SDI.SDI_APS_INVENTORY_LOCATION_1007 D
    ON A.APS_INVENTORY_LOCATION_ID = D.ID
  LEFT JOIN SDI.SDI_APS_PURCHASE_GROUP_1007 E
	ON A.PURCHASE_GROUP_ID = E.ID
  LEFT JOIN SDI.SDI_APS_COMPANY_1007 F
    ON A.SAP_COMPANY_ID = F.ID
  LEFT JOIN SDI.SDI_APS_SAP_STOCK_AREA_1007 G
    ON A.SAP_STOCK_AREA_ID = G.ID
  LEFT JOIN SDI.SDI_RES_WIP_SCHE_1007 H
    ON A.RES_WIP_SCHE_ID = H.ID
 WHERE A.IS_PHANTOM = 0
;
