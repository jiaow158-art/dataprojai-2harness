--DWI_PS_RES_ENTERPRISE_T 到 DWR_PS_RES_ENTERPRISE_F
--UPDATE FROM CHENZHIYONG 20221123 新增PRODUCT_AREA_NEW
--企业模型
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_RES_ENTERPRISE_F
SELECT
  DWI.ID AS ID,
  DWI.CODE AS CODE,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.ERP_ORG_ID AS ERP_ORG_ID,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_BY_HAND AS IS_BY_HAND,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.IS_RESOURCE AS IS_RESOURCE,
  DWI.LEVEL_ID AS LEVEL_ID,
  DWI.NAME AS NAME,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.TREE_PID AS TREE_PID,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  DWI.PRODUCT_AREA_NEW
FROM
  DWIOTD.DWI_PS_RES_ENTERPRISE_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
