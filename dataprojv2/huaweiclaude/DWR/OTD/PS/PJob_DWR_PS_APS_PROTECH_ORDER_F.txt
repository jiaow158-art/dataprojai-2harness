--DWI_PS_APS_PROTECH_ORDER_T 到 DWR_PS_APS_PROTECH_ORDER_F
--制程工艺单
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_APS_PROTECH_ORDER_F
SELECT
  DWI.ID AS ID,
  DWI.BATCH_RATIO AS BATCH_RATIO,
  DWI.PROTECH_NUM AS PROTECH_NUM,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_END_NODE AS IS_END_NODE,
  DWI.IS_LOCKED AS IS_LOCKED,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.IS_PROGRESS_COMPLETE AS IS_PROGRESS_COMPLETE,
  DWI.IS_START_NODE AS IS_START_NODE,
  DWI.NODE_NUMBER AS NODE_NUMBER,
  DWI.ORDER_STATE AS ORDER_STATE,
  DWI.PRO_TECH_NAME AS PRO_TECH_NAME,
  DWI.PROCESS_TECH_ID AS PROCESS_TECH_ID,
  DWI.SRC_MATERIAL_ID AS SRC_MATERIAL_ID,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.PROCESS_TECH_ROUTE_ID AS PROCESS_TECH_ROUTE_ID,
  DWI.PRODUCT_ORDER_ID AS PRODUCT_ORDER_ID,
  DWI.QTY AS QTY,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.TRANSFER_BATCH AS TRANSFER_BATCH,
  DWI.TRANSPORT_TIME AS TRANSPORT_TIME,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_APS_PROTECH_ORDER_T DWI
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
