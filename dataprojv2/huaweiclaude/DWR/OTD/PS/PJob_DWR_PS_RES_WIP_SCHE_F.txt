--DWI_PS_RES_WIP_SCHE_T 到 DWR_PS_RES_WIP_SCHE_F
--生产资源
INSERT
  OVERWRITE TABLE DWROTD.DWR_PS_RES_WIP_SCHE_F
SELECT
  DWI.ID AS ID,
  DWI.ALGORITHM_MARKING AS ALGORITHM_MARKING,
  DWI.AUTOMATION_TYPE AS AUTOMATION_TYPE,
  DWI.CODE AS CODE,
  DWI.CREATE_BY AS CREATE_BY,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.SRC_BELONG_FACT_ID AS SRC_BELONG_FACT_ID,
  DWI.SRC_FACTORY_ID AS SRC_FACTORY_ID,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.INV_ORG_ID AS INV_ORG_ID,
  DWI.IS_OUT_MADE AS IS_OUT_MADE,
  DWI.IS_PHANTOM AS IS_PHANTOM,
  DWI.IS_SMALL_SINGLE AS IS_SMALL_SINGLE,
  DWI.NAME AS NAME,
  DWI.PROCESS_TECH_TYPE_ID AS PROCESS_TECH_TYPE_ID,
  DWI.QTY AS QTY,
  DWI.RESOURCE_STATE AS RESOURCE_STATE,
  DWI.SCHEME_ID AS SCHEME_ID,
  DWI.SEQUENCE AS SEQUENCE,
  DWI.SOURCE_ID AS SOURCE_ID,
  DWI.SOURCE_TYPE AS SOURCE_TYPE,
  DWI.SYNC_ID AS SYNC_ID,
  DWI.TAKT_TIME AS TAKT_TIME,
  DWI.UPDATE_BY AS UPDATE_BY,
  DWI.UPDATE_DATE AS UPDATE_DATE,
  DWI.WORK_SHOP_ID AS WORK_SHOP_ID,
  DWI.IS_PROCESSING_BRICK AS IS_PROCESSING_BRICK,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_RES_WIP_SCHE_T DWI
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
