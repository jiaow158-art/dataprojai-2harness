--SDI_SALE_ORDER_1007 到 DWR_PS_SALE_ORDER_F
--APS销售计划单头表
INSERT OVERWRITE TABLE DWROTD.DWR_PS_SALE_ORDER_F
SELECT
   SRC.ID AS ID
  ,SRC.BILL_TYPE AS BILL_TYPE
  ,SRC.CODE AS OSS_SO_ID
  ,SRC.CREATE_BY AS CREATE_BY
  ,SRC.CREATE_DATE AS CREATE_DATE
  ,SRC.CUSTOMER_ID AS SRC_CUST_ID
  --源系统客户ID
  --取主数据取ID,源表关联字段为NULL,赋值-999999,源表关联字段有数据关联不上主数据表,赋值-999998
  ,CASE WHEN SRC.CUSTOMER_ID IS NULL OR SRC.CUSTOMER_ID = '' THEN -999999
        ELSE NVL(TMP.CUST_ID, -999998)
         END AS CUST_ID --客户ID
  ,CASE WHEN SRC.CUSTOMER_ID IS NULL OR SRC.CUSTOMER_ID = '' THEN 'SNULL'
        ELSE NVL(TMP.CUST_NUM, 'NOT EXISTS')
         END AS CUST_NUM --客户编码
  ,TMP.CUST_NAME --客户名称		 
  ,SRC.DISTRIBUTION_STATE AS DISTRIBUTION_STATE
  ,SRC.FULL_ORDER_FLAG AS FULL_ORDER_FLAG
  ,SRC.INV_ORG_ID AS INV_ORG_ID
  ,SRC.IS_PHANTOM AS IS_PHANTOM
  ,SRC.LINE_COUNT AS LINE_COUNT
  ,SRC.MAKE_TIME AS MAKE_TIME
  ,SRC.ORDER_TYPE AS ORDER_TYPE
  ,SRC.OSS_PLAN_NUM AS OSS_PLAN_NUM
  ,SRC.POST_OSS_STATE AS POST_OSS_STATE
  ,SRC.REVIEW_STATUS AS REVIEW_STATUS
  ,SRC.SALE_DEPT AS SALE_DEPT
  ,SRC.SALE_GROUP AS SALE_GROUP
  ,SRC.SHARE_QTY AS SHARE_CUST_QTY
  ,SRC.SOURCE_TYPE AS SOURCE_TYPE
  ,SRC.SYNC_ID AS SYNC_ID
  ,SRC.TRACE_STATE AS TRACE_STATE
  ,SRC.UN_HAND_QTY AS UN_HAND_QTY
  ,SRC.UPDATE_BY AS UPDATE_BY
  ,SRC.UPDATE_DATE AS UPDATE_DATE
  ,SRC.SALE_ORG AS SALE_ORG
  ,SRC.SALE_DEPT_NAME AS SALE_DEPT_NAME
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
  ,CASE WHEN SRC.DISTRIBUTION_STATE = 0 THEN '未分配'
	    WHEN SRC.DISTRIBUTION_STATE = 1 THEN '已分配'
	    WHEN SRC.DISTRIBUTION_STATE = 2 THEN '分配完毕'
         END AS DSTN_ST_NAME --分配状态描述
  ,CASE WHEN SRC.FULL_ORDER_FLAG = 0 THEN '否'
        WHEN SRC.FULL_ORDER_FLAG = 1 THEN '是'
		 END IS_ALL_DELV_NAME --整单标识描述
  ,CASE WHEN SRC.REVIEW_STATUS = 0 THEN '未评审'
        WHEN SRC.REVIEW_STATUS = 1 THEN '已评审'
		WHEN SRC.REVIEW_STATUS = 2 THEN '全部'
		 END AS APPR_ST_NAME --评审状态描述
  ,CASE WHEN SRC.SOURCE_TYPE = 0 THEN '自建'
        WHEN SRC.SOURCE_TYPE = 1 THEN '外部'
		 END SRC_NAME --来源描述
  ,SRC.APPLY_BY AS APPL_EMP_NAME --申请人姓名
  ,SRC.BUSINESS_BY AS SALE_EMP_NAME --业务员姓名
  ,SRC.SALE_GROUP_NAME AS SALE_GRP_NAME --销售组名称
  ,SRC.GROSS_MARGIN_FLAG AS IS_ALL_GPM --整单成本毛利率标识
  ,SRC.GROSS_MARGIN AS ALL_GPM_RATE --整单成本毛利率
  ,SRC.CHANEL_CODE AS SALE_CHNL_CODE --分销渠道
  FROM SDI.SDI_SALE_ORDER_1007 SRC
  LEFT JOIN (
    SELECT
       M.ID
      ,CUST.CUST_NUM --客户编码
      ,CUST.CUST_ID --客户ID
	  ,CUST.CUST_NAME --客户名称
    FROM
      SDI.SDI_CSM_CUSTOMER_1007 M --客户编码表
      INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
      ON M.CODE = CUST.CUST_NUM
      AND CUST.END_DATE = '4712-12-31'
      AND CUST.DEL_FLAG = 'N'
      AND M.IS_PHANTOM = 0 --取有效属性数据
      AND CUST.GROUP_NUM = '800'
  ) TMP ON SRC.CUSTOMER_ID = TMP.ID
 WHERE SRC.DEL_FLAG = 'N'
