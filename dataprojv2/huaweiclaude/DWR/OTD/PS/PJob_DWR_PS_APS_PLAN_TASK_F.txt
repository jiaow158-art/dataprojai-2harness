--UPDATE FROM CHENZHIYONG 20220308 新增了 CLOSE_FLAG、CLOSE_QTY、OPEN_FLAG、MUST_OUT_MADE、APPOINT_RESOURCE_ID
--UPDATE FROM HUANGWENBIN 20220414 新增了 CLOSE_TIMEPREPARE_CLOSE_FLAG
--UPDATE FROM HUANGWENBIN 20220826 新增了资源编码等18个字段，并切换数据源SDI-->DWR
--UPDATE FROM HUANGWENBIN 20220829 新增字段：排产计划明细ID
--update from huangwenbin 2025-08-21 新增字段：减数原因、减数状态编码
--计划任务表
INSERT OVERWRITE TABLE DWROTD.DWR_PS_APS_PLAN_TASK_F
SELECT
       T.ID AS ID
      ,T.ACTUAL_START_TIME AS ACTUAL_START_TIME --实际开始时间
      ,T.CREATE_BY AS CREATE_BY
      ,T.CREATE_DATE AS CREATE_DATE
      ,T.DURATION AS DURATION
      ,T.END_TIME AS END_TIME --结束时间
      ,T.FRONT_TRANSFER_VALUE AS FRONT_TRANSFER_VALUE
      ,T.INV_ORG_ID AS INV_ORG_ID
      ,T.IS_ALERT AS IS_ALERT
      ,T.IS_CANCEL AS IS_CANCEL --是否关闭
      ,T.IS_COMBINATION AS IS_COMBINATION
      ,T.IS_LOCKED AS IS_LOCKED
      ,T.IS_PROGRESS_COMPLETE AS IS_PROGRESS_COMPLETE
      ,T.IS_SAME_MODE AS IS_SAME_MODE
      ,T.PLAN_NO AS PLAN_NO --报工号
      ,T.RESOURCE_ID AS RESOURCE_ID
      ,T.START_TIME AS START_TIME --开始时间
      ,T.STATUS AS STATUS
      ,T.SUB_RES_STRATEGYS_ID AS SUB_RES_STRATEGYS_ID
      ,T.UPDATE_BY AS UPDATE_BY
      ,T.UPDATE_DATE AS UPDATE_DATE
      ,T.UPLOAD_REVIEW AS UPLOAD_REVIEW
      ,T.IS_RE_UPLOAD AS IS_RE_UPLOAD
      ,T.RE_DATE AS RE_DATE
      ,T.RE_STATE AS RE_STATE
      ,T.IS_MANUAL_MODIFY AS IS_MANUAL_MODIFY
      ,T.CLOSE_FLAG AS CLOSE_FLAG
      ,T.CLOSE_QTY AS CLOSE_QTY
      ,T.OPEN_FLAG AS OPEN_FLAG
      ,T.MUST_OUT_MADE AS MUST_OUT_MADE
      ,T.APPOINT_RESOURCE_ID AS APPOINT_RESOURCE_ID
      ,T.DEL_FLAG AS DEL_FLAG
      ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
      ,T.CLOSE_TIME --关闭时间
      ,T.PREPARE_CLOSE_FLAG
	  ,A.CODE AS RSRS_CODE --资源编码
	  ,A.NAME AS RSRS_NAME --资源名称
	  ,CASE WHEN T.STATUS = 10 THEN '未完成'
	        WHEN T.STATUS = 20 THEN '已下达'
			WHEN T.STATUS = 30 THEN '已完成'
			 END AS PLAN_ST_NAME --计划单状态
      ,B.IS_MAIN_ITEM 
	  ,B.ORDER_QTY --发放数量
	  ,B.PLAN_AMOUNT AS PLAN_QTY --计划数量
	  ,B.PLAN_RAW_MATERIAL_DATE
	  ,B.PROCESS_TECH_TASK_ID AS PRCS_TASK_ID --制程工艺ID
	  ,B.RELEASE_START_TIME AS RLSE_START_TIME --下达时间
	  ,B.SAM --工艺定额（秒）
	  ,B.WORK_ORDER AS PRCH_NUM --OEM单号/SAP单号
	  ,B.FINISHED_AMOUNT AS CMPLT_QTY --完成数量
	  ,B.OEM_ORDER_DATE AS PRCH_DATE --OEM采购日期
	  ,T.RE_STATE AS IS_RE_CODE --是否复版编码
	  ,CASE WHEN T.RE_STATE = 0 THEN '否'
	        WHEN T.RE_STATE = 1 THEN '是'
			 END AS IS_RE_NAME --是否复版描述
      ,T.OEM_DELIVERY_DATE AS PRCH_DELV_DATE --外协交期
	  ,T.BASE_DELIVERY_DATE AS BASE_DELV_DATE --基地交期
	  ,T.CLOSE_FLAG AS IS_CLSE_CODE --是否关闭编码
	  ,CASE WHEN T.CLOSE_FLAG = 0 THEN '否'
	        WHEN T.CLOSE_FLAG = 1 THEN '是'
			 END AS IS_CLSE_NAME --是否关闭描述
	  ,T.CLOSE_QTY AS CLSE_QTY --关闭数量
	  ,CASE WHEN T.PREPARE_CLOSE_FLAG = 0 THEN '否'
	        WHEN T.PREPARE_CLOSE_FLAG = 1 THEN '是'
			 END AS IS_PREP_CLSE_NAME --是否预关闭描述
	  ,T.SPLIT_STATE AS SPLT_ST_CODE --减数状态编码
      ,CASE WHEN T.SPLIT_STATE = 1 THEN '待确认'
            WHEN T.SPLIT_STATE = 2 THEN '已确认'
             END AS SPLT_ST_NAME --减数状态描述			 
	  ,T.SPLIT_REASON AS SPLT_RSN --减数原因
      ,B.WORK_ORDER_TYPE AS ORDER_TYPE_CODE --工单类型编码
      ,CASE WHEN B.WORK_ORDER_TYPE = 0 THEN '量产'
            WHEN B.WORK_ORDER_TYPE = 1 THEN '试产'
            WHEN B.WORK_ORDER_TYPE = 2 THEN '返工'
            WHEN B.WORK_ORDER_TYPE = 3 THEN '校验'
             END AS ORDER_TYPE_NAME --工单类型描述
      ,B.ID AS PLAN_TASK_DTL_ID --排产计划明细ID    
      ,T.SPLIT_REASON --减数原因
      ,T.SPLIT_STATE --减数状态编码
  FROM SDI.SDI_APS_PLAN_TASK_1007 T
  LEFT JOIN SDI.SDI_RES_WIP_SCHE_1007 A
    ON A.ID = T.RESOURCE_ID
  LEFT JOIN SDI.SDI_APS_PLAN_TASK_DTL_1007 B
    ON T.ID = B.PLAN_TASK_ID  
WHERE T.DEL_FLAG = 'N';
