SELECT T2.ID
      ,T3.EMPLOYEE_NO
	  ,T3.NAME
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 T2
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 T3
    ON T2.EMPLOYEE_ID = T3.ID

INSERT OVERWRITE TABLE DWROTD.DWR_TRA_BILL_RSULT_T
SELECT A.ID --计费结果ID
      ,A.BILL_ORDER_NO AS DO_NUM --TMS运单号
	  ,A.SOURCE_ORDER_LINENO AS DO_LINE_NUM --运单行号
	  ,B.EXTERNNO AS DSPT_NUM --外部订单号
	  ,B.ORDER_NO AS TMS_ORDER_NUM --TMS订单号
	  ,B.PLAN_NO AS TMS_PLAN_NUM --TMS计划号
	  ,A.PROJECT_PLAN AS DELV_PLAN_LINE_NUM --SAP项目计划号
	  ,A.ERP_NO AS DELV_NUM --SAP提货单号
	  ,A.DUMP_NO AS SAP_SALE_NUM --SAP转储单号（SAP销售订单号）
	  ,A.WITHHOLDING_NO AS WTHL_NUM --预提单号
	  ,A.STATEMENT_NO AS SUBL_NUM --对账单号
	  ,B.STATUS AS DO_ST_CODE --运单状态编码
	  ,T1.DESCR AS DO_ST_NAME --运单状态描述
	  ,A.CONTRACT_NO AS CNTC_NUM --合同编号
	  ,A.QUOTE_TYPE AS CNTC_TYPE_CODE --合同种类编码
	  ,CASE WHEN A.QUOTE_TYPE = '01' THEN '应收' ELSE '应付' END AS CNTC_TYPE_NAME --合同种类描述
	  ,A.QUOTE_ID AS OFR_NUM --报价编码
	  ,A.TASK_RUN_DT AS CHRG_TIME --计费时间
	  ,B.ACTUAL_SHIP_DATE AS SHIP_TIME --承运时间
	  ,CASE WHEN B.ORDERTYPE IN ('32','33','34','35','36','37') THEN C.DELIVERY_DATE ELSE NULL END AS DELV_TIME --配送时间
	  ,A.WAREHOUSEKEY AS WH_CODE --提货仓库编码
	  ,T2.DESCR AS WH_NAME --提货仓库名称
	  ,A.LOADING_SITE_CODE AS LOAD_SITE_CODE --提货站点编码
	  ,T3.NAME AS LOAD_SITE_NAME --提货站点名称
	  ,A.SHIP_METHOD AS SHIP_METH_CODE --运输方式编码
	  ,T4.DESCR AS SHIP_METH_NAME --运输方式描述
	  ,A.LOADING_PROVINCE_CODE AS DELV_PROV_CODE --提货省编码
	  ,A.LOADING_PROVINCE_NAME AS DELV_PROV_NAME --提货省名称
	  ,A.LOADING_CITY_CODE AS DELV_CITY_CODE --提货市编码
      ,A.LOADING_CITY_NAME AS DELV_CITY_NAME --提货市名称
	  ,A.LOADING_COUNTY_CODE AS DELV_DIST_CODE --提货区编码
	  ,A.LOADING_COUNTY_NAME AS DELV_DIST_NAME --提货区名称
	  ,B.LOADING_ADDR AS DELV_ADDR --提货详细地址
	  ,A.UNLOADING_SITE_CODE AS RECV_SITE_CODE --收货站点编码
	  ,T5.NAME AS RECV_SITE_NAME --收货站点名称
 	  ,A.UNLOADING_PROVINCE_CODE AS RECV_PROV_CODE --收货省编码
	  ,A.UNLOADING_PROVINCE_NAME AS RECV_PROV_NAME --收货省名称
	  ,A.UNLOADING_CITY_CODE AS RECV_CITY_CODE --收货市编码
      ,A.UNLOADING_CITY_NAME AS RECV_CITY_NAME --收货市名称
	  ,A.UNLOADING_COUNTY_CODE AS RECV_DIST_CODE --收货区编码
	  ,A.UNLOADING_COUNTY_NAME AS RECV_DIST_NAME --收货区名称
	  ,B.UNLOADING_ADDR AS RECV_ADDR --收货地址
	  ,A.FEE_ITEM_NAME AS FEE_PROJ_NAME --费用项目
	  ,A.CHARGE_TYPE_NAME AS CHRG_TYPE_NAME --计费类型
	  ,A.STATUS AS BILL_ST_CODE --计费状态编码
	  ,T22.DESCR AS BILL_ST_NAME --计费状态描述
	  ,B.CUSTOMER_CODE AS CUST_NUM --客户编码
	  ,T6.DESCR AS CUST_NAME --客户名称
	  ,A.CHARGE_METHOD_NAME AS CHRG_METH_NAME --计费方式
	  ,A.SKU_CODE AS MATERIAL_NUM --物料编号
	  ,A.SKU_NAME AS MATERIAL_NAME --物料名称
	  ,T7.UDF1 AS DIMENSION --规格
	  ,B.MOTHER_CUSTOMER_NO AS MTH_CUST_NUM --母客编
	  ,T8.DESCR AS MTH_CUST_NAME --母客编名称
	  ,C.OA_APPROVAL AS OA_RQST_ID --OA审批流水号
	  ,C.OA_NOTES AS OA_REMARK --备注
	  ,COALESCE(A.BILL_UNIT_PRICE,0) AS TAX_PRC --含税单价
	  ,COALESCE(A.EXTAX_BILL_UNIT_PRICE,0) AS NO_TAX_PRC --不含税单价
	  ,A.CAL_QTY AS BILL_COEFT --计费系数
	  ,A.CAL_UOM AS BILL_UNIT_CODE --计费单位编码
	  ,T20.DESCR AS BILL_UNIT_NAME --计费单位描述
	  ,A.CHARGE_UNIT_NAME AS CHRG_UNIT_NAME --单位
	  ,COALESCE(A.BILL_TOTAL,0) AS TOT_TAX_AMT --含税金额
	  ,COALESCE(A.EXTAX_BILL_TOTAL,0) AS TOT_NO_TAX_AMT --不含税金额
	  ,A.TAX_POINT AS TAX_PT --税率
	  ,A.BILL_CYCLE AS BILL_CYC_CODE --计费周期编码
	  ,T21.DESCR AS BILL_CYC_NAME --计费周期描述
      ,COALESCE(B.ACTUAL_ROUGH_WEIGHT,0) AS DELV_WT --发货重量(吨)
	  ,A.BILL_ORDER_TABLE AS FEE_TYPE_CODE --费用说明编码
	  ,T9.DESCR AS FEE_TYPE_NAME --费用说明描述
	  ,COALESCE(B.IS_BACKINGUP,0) AS IS_BACK_CODE --是否倒车编码
      ,CASE WHEN B.IS_BACKINGUP = '1' THEN '是' ELSE '否' END AS IS_BACK_NAME --是否倒车描述
	  ,COALESCE(B.IS_MACHINING,0) AS IS_PROC_CODE --是否加工编码
      ,CASE WHEN B.IS_MACHINING = '1' THEN '是' ELSE '否' END AS IS_PROC_NAME --是否加工描述
	  ,COALESCE(B.IS_UNLOADING,0) AS IS_DSCHG_CODE --是否卸货编码
      ,CASE WHEN B.IS_UNLOADING = '1' THEN '是' ELSE '否' END AS IS_DSCHG_NAME --是否卸货描述
	  ,B.DEPARTMENT_CODE AS DEPT_CODE --部门编码
	  ,T10.DESCR AS DEPT_NAME --部门名称
	  ,B.SALE_LINEGROUP AS SALE_GRP_CODE --线组编码
	  ,T11.DESCR AS SALE_GRP_NAME --线组名称
	  ,A.COSTCENTER_NO AS COST_CNTR_CODE --成本中心编码
	  ,T12.COSTCENTER_NAME AS COST_CNTR_NAME --成本中心名称
	  ,COALESCE(B.TMS_PAY_METHOD,A.TMS_PAY_METHOD) AS PAY_METH_NAME --运费支付方式
	  ,B.ORDERTYPE AS ORDER_TYPE_CODE --订单类型编码
	  ,T13.DESCR AS ORDER_TYPE_NAME --订单类型描述
	  ,CASE WHEN D.DO_NO IS NOT NULL THEN '已回单' ELSE '未回单' END AS IS_RCPT_NAME --是否有回单
	  ,T12.COMPANY AS CO_CODE --公司编码
	  ,CASE WHEN B.ORDERTYPE = '38' THEN T15.DESCR ELSE T14.DESCR END AS CO_NAME --公司名称
	  ,A.OWNER_KEY AS BILL_OWN_CODE --计费对象编码
	  ,T16.DESCR AS BILL_OWN_NAME --计费对象名称
	  ,A.CREATE_DT AS CREATE_TIME --创建时间
	  ,B.PROCESSOR AS PROC_NAME --单证处理人
	  ,T19.DESCR AS EMP_TYPE_NAME --员工类别
	  ,A.CREATE_ID AS CREATE_EMP_ID --创建人ID
	  ,T17.EMPLOYEE_NO AS CREATE_EMP_NUM --创建人工号
	  ,T17.NAME AS CREATE_EMP_NAME --创建人姓名
	  ,A.UPDATE_DT AS UPDATE_TIME --更新时间
	  ,A.UPDATE_ID AS UPDATE_EMP_ID --更新人ID
	  ,T18.EMPLOYEE_NO AS UPDATE_EMP_NUM --更新人工号
	  ,T18.NAME AS UPDATE_EMP_NAME --更新人姓名
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间	  
  FROM SDI.SDI_BMS_BILL_RESULT_TMS_1005 A
  LEFT JOIN SDI.SDI_TMS_DO_1005 B
    ON A.BILL_ORDER_NO = B.DO_NO
  LEFT JOIN SDI.SDI_TMS_TRANSPORT_PLAN_1005 C
    ON B.PLAN_NO = C.PLAN_NO
  LEFT JOIN (SELECT DISTINCT DO_NO FROM SDI.SDI_TMS_RECEIPT_1005 WHERE STATUS IN ('30','40')) D
    ON A.BILL_ORDER_NO = D.DO_NO
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T1
    ON B.STATUS = T1.CODE 
   AND T1.TYPENAME = 'TRANSORDERSTATUS'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T2
    ON T2.PARTNER_KEY = A.WAREHOUSEKEY
   AND T2.TRADE_TYPE = 'WAR'
  LEFT JOIN SDI.SDI_TMS_SITE_1005 T3
    ON A.LOADING_SITE_CODE = T3.CODE
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T4
    ON A.SHIP_METHOD = T4.CODE 
   AND T4.TYPENAME = 'TMS_SHIPMETHOD'
  LEFT JOIN SDI.SDI_TMS_SITE_1005 T5
    ON A.UNLOADING_SITE_CODE = T5.CODE
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T6
    ON T6.PARTNER_KEY = B.CUSTOMER_CODE
   AND T6.TRADE_TYPE = 'CUS'
  LEFT JOIN SDI.SDI_TMS_SKU_1005 T7
    ON A.SKU_CODE = T7.SKU_CODE
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T8
    ON T8.PARTNER_KEY = B.MOTHER_CUSTOMER_NO
   AND T8.TRADE_TYPE = 'CUS'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T9
    ON A.BILL_ORDER_TABLE = T9.CODE 
   AND T9.TYPENAME = 'BILL_ORDER_TABLE'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T10
    ON T10.PARTNER_KEY = B.DEPARTMENT_CODE
   AND T10.TRADE_TYPE = 'DEP'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T11
    ON T11.PARTNER_KEY = B.SALE_LINEGROUP
   AND T11.TRADE_TYPE = 'LIN'
  LEFT JOIN SDI.SDI_BMS_COSTCENTER_1005 T12
    ON T12.COSTCENTER_CODE = A.COSTCENTER_NO
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T13
    ON B.ORDERTYPE = T13.CODE 
   AND T13.TYPENAME = 'TMS_ORDER_TYPE'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T14
    ON T14.PARTNER_KEY = T12.COMPANY
   AND T14.TRADE_TYPE = 'COM'
  LEFT JOIN SDI.SDI_PLAT_PARTNER_1005 T15
    ON T15.PARTNER_KEY = B.INVOICEKEY
   AND T15.TRADE_TYPE = 'COM'
  LEFT JOIN (SELECT DISTINCT PARTNER_KEY,DESCR
               FROM SDI.SDI_PLAT_PARTNER_1005
              WHERE TRADE_TYPE IN('STOR','CAR')) T16
    ON T16.PARTNER_KEY = A.OWNER_KEY
  LEFT JOIN T AS T17
    ON A.CREATE_ID = T17.ID
  LEFT JOIN T AS T18
    ON A.UPDATE_ID = T18.ID
  LEFT JOIN (SELECT E.NAME
                   ,MAX(C.DESCR) AS DESCR
               FROM SDI.SDI_PLAT_EMPLOYEE_1005 E 
               LEFT JOIN SDI.SDI_PLAT_CODE_1005 C 
                 ON TYPENAME = 'LOGIN_USER_CATEGORY'
                AND C.CODE = E.CATEGORY
              GROUP BY E.NAME
            ) T19
	ON B.PROCESSOR = T19.NAME
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T20
    ON A.CAL_UOM = T20.CODE 
   AND T20.TYPENAME = 'UOM'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T21
    ON A.BILL_CYCLE = T21.CODE 
   AND T21.TYPENAME = 'BILLCYCLECODE'
  LEFT JOIN SDI.SDI_PLAT_CODE_1005 T22
    ON A.STATUS = T22.CODE 
   AND T22.TYPENAME = 'BMS_BILL_RESULT_STATUS'
 WHERE A.TENANT_ID = 1
