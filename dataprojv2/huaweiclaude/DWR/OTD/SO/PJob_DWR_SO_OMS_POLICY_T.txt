insert overwrite into DWROTD.DWR_SO_OMS_POLICY_T
with tmp as (
select 
 pfp.policy_fixed_price_code as policy_code
,pfp.policy_title as policy_name
,pfp.deposit_return
,pfp.policy_start_date
,pfp.policy_end_date
,'FIXED_PRICE' as policy_type
,pfp.performance_completion_rate as per_comp_rate
from dwrotd.dwr_dtma_policy_fixed_price_t pfp
where pfp.delete_flag = 0 
union all 
select 
 pfp.policy_discount_code as policy_code
,pfp.policy_title as policy_name
,pfp.deposit_return
,pfp.policy_start_date
,pfp.policy_end_date
,'DISCOUNT' as policy_type
,pfp.performance_completion_rate as per_comp_rate
from  dwrotd.dwr_dtma_policy_discount_t pfp
where pfp.delete_flag = 0 
)
,tmp1 as (
select lpad(客户编码,10,0) as CUST_NUM ,sum(一月+二月+三月+四月+五月+六月+七月+八月+九月+十月+十一月+十二月) * 10000 as sum_goat
from upload.upload_achievementcustomer
where 渠道 = '零售渠道' and 年份 = substr(current_date - 1,1,4)
group by lpad(客户编码,10,0)
)
,tmp2 as (
select distinct "ACCOUNT_NUMBER" as CUST_NUM  --强商
from dm.dm_rpt_zmaster_cus_t 
where accnt_group8code is not null and "PARVW" = 'AG'
)
,tmp3 as (
SELECT DISTINCT 
       IS_SUB_CUST_NAME --是否为子客编描述
          ,CNTC_STATUS_NAME --合同状态名称
          ,IS_WF_FIL_NAME --流程是否归档描述（瓷砖）
          ,CUST_TYPE_NAME --客户类型描述
          ,CUST_NUM --客户编码
          ,APLY_DATE --申请日期
          ,CUST_ATTR_NAME --客户属性描述
          ,LCGD_DATE AS GD_DATE --归档日期
  FROM DWRDIM.DWR_DIM_DEALER_CNTC_D
 WHERE 1=1
   AND DATA_SRC_NAME = '台账' 
   AND CNTC_STATUS_NAME = '履约中'
   AND LEFT(APLY_DATE,4) >= substr(current_date()-1,1,4)
   AND LEFT(CNTC_END_DATE,4) >= substr(current_date()-1,1,4)
   AND IS_WF_FIL_NAME = '已归档'
   AND CUST_TYPE_NAME IN('零售商A','零售商B')
)
,perform as (
select case when T.ZCMC in ('25年IW总裁班新品促销政策_一档','国补_25年IW总裁班新品促销政策_一档','25年IW总裁班新品促销政策_二档','国补_25年IW总裁班新品促销政策_二档') 
                 then '25年IW总裁班新品促销政策'  
                 when T.ZCMC in ('25年秋季新品订货会_五档','国补_25年秋季新品订货会_五档','25年秋季新品订货会_二档','国补_25年秋季新品订货会_二档','25年秋季新品订货会_一档','国补_25年秋季新品订货会_一档','25年秋季新品订货会_二挡','25年秋季新品订货会_四档') 
                 then '25年秋季新品订货会' else T.ZCMC end as ZCMC,
T.BUS_AREA,T.MATERIAL,T.doc_number,T.s_ord_item,t.salesorg,t.customer,T.distr_chan,T.CALMONTH,T.ambperformance 
from dm.dm_ambv2_zzt_xssr_t T
where T.ZCMC is not null 
)
,perform1 as (
select 
 sum(CASE WHEN T.ZCMC IN (SELECT DISTINCT policy_name FROM TMP) THEN T.ambperformance ELSE 0 END) as ambperformance1
,SUM(CASE WHEN T.BUS_AREA = 'G000' AND T.MATERIAL IN (SELECT DISTINCT product_code from dwrotd.dwr_dtma_policy_fixed_price_item_t
union all SELECT DISTINCT product_code from dwrotd.dwr_DTMA_POLICY_DISCOUNT_ITEM_t) THEN T.ambperformance ELSE 0 END) as ambperformance2
,sum(T1.mwsbp + T1.netwr) as ord_amt 
,coalesce(T2.parent_cust_num,T.customer) as ZZKUNNR
,T.CALMONTH
,T.ZCMC
from perform T
left join (SELECT * FROM DM.DM_OTD_SALES_ORDER_DET_T WHERE ABGRU IS NULL)t1 
on T.doc_number = T1.VBELN  AND T.s_ord_item = T1.POSNR
left join DWRDIM.DWR_DIM_CUST_SALES_AREA_D t2
on t.salesorg = t2.sales_org_code
and t.customer = t2.cust_num
and t2.distribution_chnl_code = T.distr_chan
and t2.end_date = '4712-12-31'
and t2.del_flag = 'N'
where (T.ZCMC IN (SELECT DISTINCT policy_name FROM TMP) or T.BUS_AREA = 'G000') 
group by 
 coalesce(T2.parent_cust_num,T.customer)
,T.CALMONTH
,T.ZCMC
)
,perform2 as (
select 
 sum(T.ambperformance) as ambperformance
,T.customer as cust_num
,T.CALMONTH
,T.ZCMC
from perform T
-- left join (SELECT * FROM DM.DM_OTD_SALES_ORDER_DET_T WHERE ABGRU IS NULL)t1 
-- on T.doc_number = T1.VBELN  AND T.s_ord_item = T1.POSNR
-- left join DWRDIM.DWR_DIM_CUST_SALES_AREA_D t2
-- on t.salesorg = t2.sales_org_code
-- and t.customer = t2.cust_num
-- and t2.distribution_chnl_code = T.distr_chan
-- and t2.end_date = '4712-12-31'
-- and t2.del_flag = 'N'
where T.ZCMC IN (SELECT DISTINCT policy_name FROM TMP) 
group by 
 T.customer
,T.CALMONTH
,T.ZCMC
)
,TMPP AS (
select 
 t.policy_activity_code
,t.policy_code
,t1.policy_name
,t1.policy_start_date
,t1.policy_end_date
,t.policy_position ---报名档位
,t1.deposit_return --退订规则
,t.customer_code as cust_code-- 客户编码
,t3.cust_name as cust_name-- 客户名称
,t.sale_org_code 
,t.channel_code as dis_chnl_code 
,t2.sales_group_code
,t5.node_name5
,t5.node_desc5
,case when t2.cust_group_code = 'B1' then t2.cust_num else coalesce(t2.parent_cust_num,t2.cust_num) end as par_cust_num
,t4.cust_name as par_cust_name
,t.enroll_time as policy_apply_date --报名时间
,t6.cust_type_name as cust_lev_name  --客户等级
,case when t7.cust_num is not null and t3.ATTRI_CODE6 = '06A' then '强商&共享仓运营商'
      when t7.cust_num is not null then '强商'
      when t3.ATTRI_CODE6 = '06A' then '共享仓运营商' else '' end as CUST_ATTR_NAME --客户属性
,t3.cust_ache_class
,case when t3.cust_ache_class = '一类' then 'T≥1000万或共享仓或强商'
      when t3.cust_ache_class = '二类' then '500≤T<1000'
      when t3.cust_ache_class = '三类' then '250≤T<500'
      when t3.cust_ache_class = '四类' then 'T<250' else '' end as cust_ache_class_name  --客户分类名称 
,t8.sum_goat as task_sum_amt -- 任务合计      
,t.task_target as dh_target_amt--订货目标金额
,t.deposit_amount as deposit_amt --定金金额
,t1.per_comp_rate --保证金返回门槛（业绩完成率）
from dwrotd.dwr_dtma_policy_activity_t t
left join tmp t1 
on t.policy_code = t1.policy_code and t.policy_type = t1.policy_type
left join DWRDIM.DWR_DIM_CUST_SALES_AREA_D t2
on t.sale_org_code = t2.sales_org_code
and t.customer_code = t2.cust_num
and t2.distribution_chnl_code = t.channel_code
and t2.end_date = '4712-12-31'
and t2.del_flag = 'N'
left join DWRDIM.DWR_DIM_CUST_GENERAL_D t3
on t.customer_code = t3.cust_num
and t3.end_date = '4712-12-31'
and t3.del_flag = 'N'
left join DWRDIM.DWR_DIM_CUST_GENERAL_D t4
on case when t2.cust_group_code = 'B1' then t2.cust_num else t2.parent_cust_num end = t4.cust_num
and t4.end_date = '4712-12-31'
and t4.del_flag = 'N'
left join dm.dm_rpt_sales_group_t t5
on t2.sales_group_code = t5.node_name9
left join tmp3 t6
on t.customer_code = t6.cust_num
left join tmp2 t7
on t.customer_code = t7.cust_num
left join tmp1 t8
on case when t2.cust_group_code = 'B1' then t2.cust_num else t2.parent_cust_num end = t8.cust_num

)
SELECT 
 t.policy_activity_code 
,t.policy_code
,t.policy_name
,t.policy_start_date
,t.policy_end_date
,t.policy_position ---报名档位
,t.deposit_return
,t.cust_code-- 客户编码
,t.cust_name-- 客户名称
,t.sale_org_code  as sales_org_code
,t.dis_chnl_code 
,t.sales_group_code
,t.node_name5
,t.node_desc5
,t.par_cust_num
,t1.calmonth
,t.par_cust_name
,t.policy_apply_date --报名时间
,t.cust_lev_name  --客户等级
,t.CUST_ATTR_NAME --客户属性
,t.cust_ache_class
,t.cust_ache_class_name  --客户分类名称 
,t.task_sum_amt -- 任务合计      
,t.dh_target_amt--订货目标金额
,t1.ord_amt  --订单金额
,t1.ambperformance1 as fac_yj_amt  --出库金额（对厂）
,t1.ambperformance2 as ware_yj_amt --出库金额（对仓）
,t.deposit_amt --定金金额
,0 as policy_cust_amt  --保证金返回门槛（业绩完成率）
,current_date() AS DW_LAST_UPDATE_DATE
,per_comp_rate 
FROM TMPP t
left join perform1 t1 
on t1.calmonth between substr(t.policy_apply_date,1,7) and substr(t.policy_end_date,1,7)
and t1.zzkunnr = t.par_cust_num
and t.policy_name = t1.ZCMC
union all
SELECT 
 t.policy_activity_code 
,t.policy_code
,t.policy_name
,t.policy_start_date
,t.policy_end_date
,t.policy_position ---报名档位
,t.deposit_return
,t.cust_code-- 客户编码
,t.cust_name-- 客户名称
,t.sale_org_code  as sales_org_code
,t.dis_chnl_code 
,t.sales_group_code
,t.node_name5
,t.node_desc5
,t.par_cust_num
,t2.calmonth
,t.par_cust_name
,t.policy_apply_date --报名时间
,t.cust_lev_name  --客户等级
,t.CUST_ATTR_NAME --客户属性
,t.cust_ache_class
,t.cust_ache_class_name  --客户分类名称 
,t.task_sum_amt -- 任务合计      
,t.dh_target_amt--订货目标金额
,0 as ord_amt  --订单金额
,0 as fac_yj_amt  --出库金额（对厂）
,0 as ware_yj_amt --出库金额（对仓）
,t.deposit_amt --定金金额
,t2.ambperformance as policy_cust_amt  --保证金返回门槛（业绩完成率）
,current_timestamp() AS DW_LAST_UPDATE_DATE
,per_comp_rate
FROM TMPP t
left join perform2 t2 
on t2.calmonth between substr(t.policy_apply_date,1,7) and substr(t.policy_end_date,1,7)
and t2.cust_num = t.cust_code
and t.policy_name = t2.ZCMC
