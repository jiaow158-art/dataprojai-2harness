--SAP销售订单行临时表TMP03(关联DD07配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_ITEM_SAP_TMP03
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  MCD1.DOMAIN_VALUE_NAME AS GIVEN_INV_REVALUED_NAME,
  MCD2.DOMAIN_VALUE_NAME AS INCOMPLETE_STATUS_NAME,
  MCD3.DOMAIN_VALUE_NAME AS ITEM_CONFIRM_STATUS_NAME,
  MCD4.DOMAIN_VALUE_NAME AS PROJ_TYPE_NAME,
  MCD5.DOMAIN_VALUE_NAME AS REFERENCE_DOC_NUM_NAME,
  MCD6.DOMAIN_VALUE_NAME AS PROJ_DELIVERY_STATUS_NAME,
  MCD7.DOMAIN_VALUE_NAME AS DELIVERY_STATUS_NAME,
  MCD8.DOMAIN_VALUE_NAME AS INCOMP_DELIVERY_STATUS_NAME,
  MCD9.DOMAIN_VALUE_NAME AS REFERENCE_TOTAL_STATUS_NAME,
  MCD10.DOMAIN_VALUE_NAME AS REFERENCE_STATUS_NAME
FROM
  DWIOTD.DWI_SO_ITEM_SAP_T DWI --SAP销售订单行表
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD1 --特定库存的评估中文名称
  ON DWI.GIVEN_INV_REVALUED = MCD1.DOMAIN_VALUE_CODE
  AND MCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD2 --一般项目的不完整状态中文名称
  ON DWI.INCOMPLETE_STATUS = MCD2.DOMAIN_VALUE_CODE
  AND MCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD3 --凭证项目的确认状态中文名称
  ON DWI.ITEM_CONFIRM_STATUS = MCD3.DOMAIN_VALUE_CODE
  AND MCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD4 --项目类型中文名称
  ON DWI.PROJ_TYPE = MCD4.DOMAIN_VALUE_CODE
  AND MCD4.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD5 --参考单据的单据编号中文名称
  ON DWI.REFERENCE_DOC_NUM = MCD5.DOMAIN_VALUE_CODE
  AND MCD5.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD6 --项目的总体交货状态中文名称
  ON DWI.PROJ_OVERALL_DELIVERY_STATUS = MCD6.DOMAIN_VALUE_CODE
  AND MCD6.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD7 --交货状态中文名称
  ON DWI.DELIVERY_STATUS = MCD7.DOMAIN_VALUE_CODE
  AND MCD7.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD8 --关于交货的项目不完整状态中文名称
  ON DWI.INCOMPLETE_DELIVERY_STATUS = MCD8.DOMAIN_VALUE_CODE
  AND MCD8.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD9 --引用的总状态中文名称
  ON DWI.REFERENCE_TOTAL_STATUS = MCD9.DOMAIN_VALUE_CODE
  AND MCD9.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD10 --参考状态中文名称
  ON DWI.REFERENCE_STATUS = MCD10.DOMAIN_VALUE_CODE
  AND MCD10.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
