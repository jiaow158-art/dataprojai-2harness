insert overwrite table dwrotd.dwr_decoration_cust_price_maintenance_f  
select ph.price_code, -- 申请单号
       ph.price_name, -- 价目表名称
       c1.customer_code, -- 客户编码
       c1.customer_name, -- 客户
       c2.customer_code, -- 所属头部编码
       c2.customer_name, -- 所属头部名称
       d.distributor_code, -- 经销商编码
       d.distributor_name, -- 经销商名称
       ph.creation_date, -- 制单日期
       case ph.price_status 
           when 'EFFECTIVE' then '已生效' 
           when 'WASTE' then '已作废' 
           when 'INVALID' then '失效' 
           when 'NEW' then '草稿' 
       end as price_status, -- 状态
       ph.channel_code, -- 渠道编码
       ph.channel_name, -- 渠道名称
       ph.bid_price_ratio, -- 默认标价比例
       ph.described, -- 头备注
       plt.product_code, -- 产品编码
       plt.product_name, -- 产品名称
       plt.auxiliary_name, -- 辅助名称
       plt.cost_price, -- 成本价
       plt.sale_price, -- 销售价
       plt.retail_price, -- 零售价
       plt.directly_price, -- 直营价
       plt.freight, -- 运费
       plt.sale_unit_name, -- 销售单位
       plt.stock_price, -- 基本单位价格
       plt.stock_unit_name, -- 基本单位
       plt.trans_scale, -- 转换率
       case plt.operation_status 
           when 'DELETE' then '删除' 
           when 'SUBMIT' then '已提交' 
           when 'UPDATE' then '更新中' 
       end as operation_status, -- 操作状态
       plt.described, -- 行备注
       CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
from sdi.sdi_dtcust_price_header_1031  ph
left join sdi.sdi_dtcust_price_line_temp_1031  plt on ph.header_id = plt.header_id
left join sdi.sdi_dtcust_customer_info_1029  c1 on ph.customer_id = c1.customer_id
left join sdi.sdi_dtcust_customer_info_1029 c2 on c1.head_enterprise_id = c2.customer_id
left join sdi.sdi_dt_distributor_info_1029  d on ph.distributor_id = d.distributor_id
where ph.channel_code = '10008'
