INSERT OVERWRITE TABLE DWROTD.DWR_VM_OMS_SALE_ORDER_F
SELECT 
       SH.ORDER_CODE --订单号
      ,SH.ORDER_STATUS  --订单状态
	  ,A.MEANING AS ORDER_STATUS_NAME --订单状态名称
      ,SH.SETTLEMENT_TYPE --结算方式
	  ,B.MEANING AS SETTLEMENT_TYPE_NAME --结算方式名称
      ,SH.FINACE_PAY_TYPE AS PAY_TYPE --付款状态
	  ,C.MEANING AS PAY_TYPE_NAME --付款状态名称
      ,SH.CUSTOM_CODE AS CUST_CODE --客户编码
      ,SH.CUSTOM_NAME AS CUST_NAME --客户名称
      ,SH.AMOUNT AS ORDER_AMT --本单金额
      ,SH.OTHER_PRICE AS ORDER_OTHER_AMT --其他费用
      ,SH.DESCRIBED AS ORDER_REMARK --备注
      ,SH.PROVINCE AS PROVINCE_NAME --省
      ,SH.CITY AS CITY_NAME --市
      ,SH.COUNTY AS COUNTY_NAME --区
      ,SH.ADDRESS --详细地址
      ,SH.CONTACTS AS CONTACT_NAME -- 收货人
      ,SH.CONTACTS_PHONE AS CONTACT_PHONE -- 收货人电话
      ,SL.PRODUCT_CODE AS MATERIAL_CODE --产品编码
      ,SL.PRODUCT_NAME AS MATERIAL_NAME --产品名称
	  ,E.LENGTH_1 --长度
	  ,E.WIDTH --宽度
	  ,E.PRODUCT_LEVEL_CODE --产品层次编码
	  ,E.PRODUCT_LEVEL_NAME --产品层次名称
	  ,E.PRODUCT_LEVEL1_CODE --产品层次1编码
	  ,E.PRODUCT_LEVEL1_NAME --产品层次1名称
	  ,E.PRODUCT_LEVEL2_CODE --产品层次2编码
	  ,E.PRODUCT_LEVEL2_NAME --产品层次2名称
	  ,E.PRODUCT_LEVEL3_CODE --产品层次3编码
	  ,E.PRODUCT_LEVEL3_NAME --产品层次3名称
	  ,E.PRODUCT_LEVEL4_CODE --产品层次4编码
	  ,E.PRODUCT_LEVEL4_NAME --产品层次4名称
	  ,E.PRODUCT_LEVEL5_CODE --产品层次5编码
	  ,E.PRODUCT_LEVEL5_NAME --产品层次5名称
	  ,E.PRODUCT_LEVEL6_CODE --产品层次6编码
	  ,E.PRODUCT_LEVEL6_NAME --产品层次6名称
	  ,E.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道
	  ,E.RESERVED_FIELD1_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称
      ,SL.SALE_QTY  --销售数量
	  ,CASE WHEN E.MATL_GROUP_CODE = '1000010' 
	        THEN SL.SALE_QTY * E.LENGTH_1 * E.WIDTH /1000000
	        ELSE NULL
			 END AS SALE_AREA --销售面积
      ,SL.DELIVERED_QTY AS DELIVER_QTY --已发货数量
      ,SL.UNIT_CODENAME AS SALE_UNIT --销售单位
      ,SL.SALE_PRICE  --销售单价
      ,SL.SALE_QTY * SL.SALE_PRICE AS SALE_AMT --销售金额
      ,SH.CREATION_DATE AS CREATE_DATE --制单时间
	  ,SH.CHANNEL_CODE --渠道编码  
	  ,SH.CHANNEL_NAME --渠道名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SH.DISTRIBUTOR_CODE AS CUSTOMER_CODE --经销商编码
	  ,SH.DISTRIBUTOR_NAME AS CUSTOMER_NAME --经销商名称
      ,F.EMPLOYEE_NUMBER AS SALE_EMP_NUM --销售员编码
	  ,SH.SALES_MAN AS SALE_EMP_NAME --销售员名称
	  ,SH.WAREHOUSE_CODE --仓库编码
	  ,SH.WAREHOUSE_NAME --仓库名称
      ,SH.LOCATION_CODE AS INV_LOCATION_CODE --库存地点编码
	  ,SH.LOCATION_NAME AS INV_LOCATION_NAME --库存地点名称
	  ,SH.SOURCE_CODE --来源单据号
	  ,SH.SERVICE_DESCRIBED AS SERVICE_REMARK --服务商备注
  FROM SDI.SDI_DTWMS_SO_HEADERS_1031 SH 
 INNER JOIN SDI.SDI_DTWMS_SO_LINE_1031 SL 
    ON SH.HEADER_ID = SL.HEADER_ID
   AND SL.DEL_FLAG = 'N'	
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 A
    ON SH.ORDER_STATUS = A.LOOKUP_CODE
   AND A.LOOKUP_TYPE = 'SALE_ORDER_STATUS'
   AND A.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B
    ON SH.SETTLEMENT_TYPE = B.LOOKUP_CODE
   AND B.LOOKUP_TYPE = 'FIT_SETTLEMENT_TYPE'
   AND B.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 C
    ON SH.FINACE_PAY_TYPE = C.LOOKUP_CODE
   AND C.LOOKUP_TYPE = 'FINACE_PAYTYPE'
   AND C.ENABLED_FLAG = 'Y'
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T E
    ON SL.PRODUCT_CODE = E.MATERIAL_NUM
  LEFT JOIN SDI.SDI_BASE_PERSON_1029 F
    ON F.PERSON_ID = SH.SALESMAN_ID
   AND F.ENABLED = 'Y'
 WHERE 1 = 1
