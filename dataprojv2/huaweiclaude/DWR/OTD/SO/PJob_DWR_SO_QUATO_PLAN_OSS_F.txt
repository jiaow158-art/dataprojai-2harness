--DWI_SO_QUATO_PLAN_OSS_T 到 DWR_SO_QUATO_PLAN_OSS_F
--报价单与计划单中间表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_QUATO_PLAN_OSS_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.DB_LAST_UPDATE_DATE AS DB_LAST_UPDATE_DATE,
  DWI.LAST_UPDATE_DATE AS LAST_UPDATE_DATE,
  DWI.PLAN_ITEM_ID AS PLAN_ITEM_ID,
  DWI.QUATO_ITEM_ID AS QUATO_ITEM_ID,
  DWI.APPLY_QUANTITY AS APPLY_QUANTITY,
  DWI.COMMENTS AS COMMENTS,
  DWI.STATUS AS STATUS,
  DWI.MODIFY_QTY AS MODIFY_QTY,
  DWI.DB_LAST_UPDATE_DATE_SRC AS DB_LAST_UPDATE_DATE_SRC,
  DWI.LAST_UPDATE_BY AS LAST_UPDATE_BY,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_QUATO_PLAN_OSS_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
