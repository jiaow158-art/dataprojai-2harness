--DWI_SO_PURCH_ITEM_OSS_T 到 DWR_SO_PURCH_ITEM_OSS_F
--要货单行
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_PURCH_ITEM_OSS_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.LAST_UPDATE_DATE AS LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE AS DB_LAST_UPDATE_DATE,
  DWI.HEAD_ID AS HEAD_ID,
  DWI.DP_REQUITE_ORDE_ITEMR_ID AS DP_REQUITE_ORDE_ITEMR_ID,
  DWI.PLAN_NUM AS PLAN_NUM,
  DWI.DP_ORDER_ITEM_ID AS DP_ORDER_ITEM_ID,
  DWI.YI_PICK_AMT AS YI_PICK_AMT,
  DWI.PROD_NUM AS PROD_NUM,
  DWI.TRAN_NUM AS TRAN_NUM,
  DWI.DP_ORDER_ID AS DP_ORDER_ID,
  DWI.DP_ACTUAL_QUANTITY AS DP_ACTUAL_QUANTITY,
  DWI.YI_PICK AS YI_PICK,
  DWI.SRC_FACTORY_ID AS SRC_FACTORY_ID,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.PROD_ID AS PROD_ID,
  DWI.DP_REMARK AS DP_REMARK,
  DWI.CANCEL_NUM AS CANCEL_NUM,
  DWI.ARIVAL_TIME AS ARIVAL_TIME,
  DWI.UPDATE_ID AS UPDATE_ID,
  DWI.CREAT_ID AS CREAT_ID,
  DWI.YI_RETURN_PICK AS YI_RETURN_PICK,
  DWI.YI_CHECK AS YI_CHECK,
  DWI.MODIFY_QTY AS MODIFY_QTY,
  DWI.DP_PLAN_WAREHOUSE AS DP_PLAN_WAREHOUSE,
  DWI.UNIT_PRI AS UNIT_PRI,
  DWI.YI_RETURN_CHECK AS YI_RETURN_CHECK,
  DWI.LAST_UPDATE_BY AS LAST_UPDATE_BY,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC AS DB_LAST_UPDATE_DATE_SRC,
  DWI.STLE_STATUS AS STLE_STATUS,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PURCH_ITEM_OSS_T DWI
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
