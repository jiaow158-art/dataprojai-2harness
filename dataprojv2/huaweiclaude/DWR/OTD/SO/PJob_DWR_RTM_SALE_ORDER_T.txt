INSERT OVERWRITE TABLE DWROTD.DWR_RTM_SALE_ORDER_T
SELECT A.CREATION_DATE AS ORDER_DATE --订单创建时间
      ,A.ID AS ORDER_ID --订单ID
      ,C.ORDER_NUMBER AS ORDER_NUM --订单编码
      ,A.QUOTATION_NUMBER AS OFR_ORDER_NUM --报价单编码
      ,A.CREATED_BY AS CREATOR_ID --创建人ID
      ,A.CREATED_BY_NAME AS CREATOR_NAME --创建人名称
      ,A.TOTAL_ORDER_PRICE AS ORDER_AMT --订单总价
      ,B.ID AS ORDER_LINE_ID --订单行ID
      ,B.MATERIAL_NUMBER AS MATERIAL_CODE --物料编码
      ,B.MATERIAL_NAME --物料名称
      ,B.SUPPLY_AREA AS SUP_AREA --产区名称
      ,B.FACTORY_CODE --工厂编码
      ,B.STOCK_LOCATION_CODE AS WAREHOUSE_CODE --库存地点
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.QUANTITY
			ELSE B.QUANTITY * -1
			 END AS ORDER_QTY --订单数量
      ,B.BATCH AS BATCH_NUM --批次号
      ,B.DEFAULT_DELIVERY_DATE AS DEL_DELV_DATE --默认交货日期
      ,B.DEMAND_DELIVERY_DATE AS DEM_DELV_DATE --需求交货日期
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.ITEM_AMOUNT
			ELSE B.ITEM_AMOUNT * -1
			 END AS ORDER_LINE_AMT --订单行金额
      ,B.SUBMIT_DATE AS SUBMIT_TIME --提交时间
      ,B.FACTORY_PROCESSING_DAYS AS FACTORY_PRCSG_DAY --工厂加工天数
      ,B.FACTORY_DELIVERY_DATE AS FACTORY_DELV_DATE --工厂交货日期
      ,B.SHIPPING_METHOD AS SHIP_METH_CODE --运输方式编码
      ,D.MEANING AS SHIP_METH_NAME --运输方式名称
      ,B.STATUS AS LINE_STATUS_CODE --订单行状态编码
      ,E.MEANING AS LINE_STATUS_NAME --订单行状态名称
      ,B.UNIT AS MATERIAL_UNIT --计量单位
      ,B.UNIT_WEIGHT --单位重量
      ,B.UNIT_AREA --单位面积(平方米)
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.UNIT_AREA * B.QUANTITY
			ELSE B.UNIT_AREA * B.QUANTITY * -1
			 END AS SAEL_AREA --销售面积(平方米)
      ,B.MATERIAL_SPECIFICATIONS AS DIMENSION --物料规格
      ,B.MATERIAL_GROUP1 AS MATL_GROUP1_CODE --物料组1编码
      ,B.MATERIAL_GROUP2 AS MATL_GROUP2_CODE --物料组2编码
      ,B.SAP_ORDER_NUMBER AS SAP_ORDER_NUM --SAP订单号
      ,REPLACE(B.SAP_ORDER_ITEM_NUMBER,B.SAP_ORDER_NUMBER,'') AS SAP_ORDER_LINE_NUM --SAP订单行号
      ,B.DENIAL_REASON AS DENL_RSN --拒绝原因
      ,B.STOCK_TYPE --留货类型
      ,B.AGENT_PRICE --基准价
      ,B.UNIT_PRICE --单价
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.UNKNOWN_QUANTITY
			ELSE B.UNKNOWN_QUANTITY * -1
			 END AS UN_CNFRM_QTY --未确认数量
	  ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.UNIT_AREA * B.UNKNOWN_QUANTITY
			ELSE B.UNIT_AREA * B.UNKNOWN_QUANTITY * -1
			 END AS UN_CNFRM_AREA --未确认面积
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.CONFIRM_QUANTITY
			ELSE B.CONFIRM_QUANTITY * -1
			 END AS CNFRM_QTY --确认数量
	  ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.UNIT_AREA * B.CONFIRM_QUANTITY
			ELSE B.UNIT_AREA * B.CONFIRM_QUANTITY * -1
			 END AS CNFRM_AREA --确认面积
      ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.SEND_QUANTITY
			ELSE B.SEND_QUANTITY * -1
			 END AS YCK_QTY --已出库数量
	  ,CASE WHEN COALESCE(B.STATUS,'') <> 'CANCEL' THEN B.UNIT_AREA * B.SEND_QUANTITY
			ELSE B.UNIT_AREA * B.SEND_QUANTITY * -1
			 END AS YCK_AREA --已出库面积
      ,B.WEIGHT_UNIT --重量单位
      ,B.MATERIAL_COST --物料成本
	  ,F.CREATE_DATE --报价单创建时间
      ,F.OFR_TYPE_CODE --报价单类型编码
      ,F.OFR_TYPE_NAME --报价单类型名称
	  ,F.SALE_ORG_CODE --销售组织编码
      ,F.SALE_ORG_NAME --销售组织名称
      ,F.SALE_CHNL_CODE --分销渠道编码
      ,F.SALE_CHNL_NAME --分销渠道名称
      ,F.CUST_NUM --客户编码
      ,F.CUST_NAME --客户名称
	  ,F.STATUS_CODE --报价单状态编码
      ,F.STATUS_NAME --报价单状态名称
	  ,F.DEPOSIT_AMT --报价单定金金额（初始定金）
      ,F.RETURN_AMT --报价单退回定金
      ,F.SALE_EMP_NUM --销售员工编码
      ,F.SALE_EMP_NAME --销售员工名称
	  ,F.PLCY_NAME --活动政策名称
	  ,F.OFR_QTY --报价单数量
	  ,F.OFR_ORDER_AMT --报价单单据金额
      ,F.SALE_AREA --报价单销售面积
	  ,F.YZ_DEPT_NAME --运营中心
      ,F.XQ_DEPT_CODE --销区编码
      ,F.XQ_DEPT_NAME --销区名称
      ,F.SALE_GRP_CODE --销售组编码
      ,F.SALE_GRP_NAME --销售组名称
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM DWRCS.DWR_CS_ENGIN_P_ORDER_HEADER_F A
  LEFT JOIN DWRCS.DWR_CS_ENGIN_P_ORDER_LINE_F B
    ON A.ID = B.HEAD_ID
  LEFT JOIN DWRCS.DWR_CS_ORDER_ALL_BUSINESS_F C
    ON A.ID = C.ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 D
    ON B.SHIPPING_METHOD = D.LOOKUP_CODE
   AND D.LOOKUP_TYPE = 'DEALER_TRANSPORT'
   AND D.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 E
    ON B.STATUS = E.LOOKUP_CODE
   AND E.LOOKUP_TYPE = 'ORDER_LINE_NEW_STATUS'
   AND E.ENABLED_FLAG = 'Y'
  LEFT JOIN DWROTD.DWR_RTM_OFR_ORDER_T F
    ON B.QUOTATION_ITEM_ID = F.OFR_ORDER_LINE_ID  
;
