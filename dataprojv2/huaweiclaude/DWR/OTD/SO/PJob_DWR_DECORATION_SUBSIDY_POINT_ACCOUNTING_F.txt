insert overwrite table DWROTD.dwr_decoration_subsidy_point_accounting_f
SELECT 
    s.distributor_code 	    	-- 服务商编码, 
    ,s.distributor_name 		-- 服务商名称, 
    ,s.customer_code			-- 装企编码, 
    ,s.customer_name		    -- 装企名称, 
    ,s.product_code             -- 物料编码, 
    ,s.product_name             -- 物料名称, 
    ,s.size						 -- 尺寸, 
    ,s.channel_code 			-- 渠道编码, 
    ,s.channel_name 			-- 渠道名称, 
    ,s.supply_reference_price 	-- 供货基准价(片), 
    ,subsidy_ratio				 -- 倒推补贴比例, 
    ,subsidy_after_price 		-- 补贴后价格, 
    ,CASE s.subsidy_points_status 
        WHEN 'PASS' THEN '已审核' 
        WHEN 'APPROVED' THEN '待审核' 
    END subsidy_points_status -- 状态, 
    ,crt_user.user_full_name 	-- 创建人, 
    ,s.creation_date 	-- 创建时间
    ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM 
    SDI.SDI_dtac_subsidy_points_info_1031 s
LEFT JOIN 
    SDI.SDI_base_users_1029 crt_user ON s.created_by = crt_user.user_id
