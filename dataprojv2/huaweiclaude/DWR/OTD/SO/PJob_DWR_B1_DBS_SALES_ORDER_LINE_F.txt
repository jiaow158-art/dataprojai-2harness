INSERT OVERWRITE TABLE DWROTD.DWR_B1_DBS_SALES_ORDER_LINE_F 
SELECT 
 t3.DocEntry    
,t3.ItemCode
,t3.Quantity
,t3.Price
,t3.FreeTxt
,'销售出库'                 in_out_flag                      -- 出入库标识
,current_timestamp() DW_LAST_UPDATE_DATE
,lpad(t1.customer_code,10,0)        distributor_code 
FROM DWIOTD.DWI_WM_B1_oinv_T t1 
JOIN DWIOTD.DWI_WM_B1_OCRD_T t2 on t1.customer_code =t2.customer_code and t1.customer_type = t2.customer_type  and t1.cardcode = t2.cardcode
JOIN DWIOTD.DWI_WM_B1_inv1_T t3 on t1.DocEntry = t3.DocEntry AND T1.customer_code = T3.customer_code
-- where t2.customer_code ='1000006131'
where  t2.u_sfdbs ='Y'
union all 
SELECT 
 t3.DocEntry    
,t3.ItemCode
,t3.Quantity
,t3.Price
,t3.FreeTxt
,'销售退货'             in_out_flag                         -- 出入库标识
,current_timestamp() DW_LAST_UPDATE_DATE
,lpad(t1.customer_code,10,0)        distributor_code 
FROM DWIOTD.DWI_WM_B1_orin_T t1 
JOIN DWIOTD.DWI_WM_B1_OCRD_T t2 on t1.customer_code =t2.customer_code and t1.customer_type = t2.customer_type  and t1.cardcode = t2.cardcode
JOIN DWIOTD.DWI_WM_B1_rin1_T t3 on t1.DocEntry = t3.DocEntry AND T1.customer_code = T3.customer_code
-- where t2.customer_code ='1000006131'
 where  t2.u_sfdbs ='Y'
