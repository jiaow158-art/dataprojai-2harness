--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP交货到期表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_DELIVERY_DUE_SAP_F
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SHIPPING_DATE AS SHIPPING_DATE,
  DWI.RECEIVE_POINTS AS RECEIVE_POINTS,
  RCD1.CONFIG_NAME AS RECEIVE_POINTS_NAME,
  DWI.ROUTE AS ROUTE,
  RCD2.CONFIG_NAME AS ROUTE_NAME,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  RCD3.CONFIG_NAME AS SAL_VCH_TYPE_NAME,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.CUST_SRV_NUM AS CUST_SRV_NUM,
  DWI.CUST_SAL_ADDR_ID AS CUST_SAL_ADDR_ID,
  DWI.CUST_SRV_ADDR_ID AS CUST_SRV_ADDR_ID,
  DWI.SAL_ORG_CODE AS SAL_ORG_CODE,
  RCD4.CONFIG_NAME AS SAL_ORG_CODE_NAME,
  DWI.DIS_CHANNEL AS DIS_CHANNEL,
  RCD5.CONFIG_NAME AS DIS_CHANNEL_NAME,
  DWI.PROD_GROUP AS PROD_GROUP,
  RCD6.CONFIG_NAME AS PROD_GROUP_NAME,
  DWI.SAL_OFFICE_CODE AS SAL_OFFICE_CODE,
  RCD7.CONFIG_NAME AS SAL_OFFICE_CODE_NAME,
  DWI.SAL_ORG_GROUP AS SAL_ORG_GROUP,
  RCD8.CONFIG_NAME AS SAL_ORG_GROUP_NAME,
  DWI.PROJ_GROSS_WEIGHT AS PROJ_GROSS_WEIGHT,
  DWI.WEIGHT_UNIT AS WEIGHT_UNIT,
  DWI.PROJ_VOL AS PROJ_VOL,
  DWI.DELIVERY_CREATE_DATE AS DELIVERY_CREATE_DATE,
  DWI.VOL_UNIT AS VOL_UNIT,
  DWI.DEAL_DAYS AS DEAL_DAYS,
  DWI.BACK_SUM_FLAG AS BACK_SUM_FLAG,
  DWI.DELIVERY_PRIORITY AS DELIVERY_PRIORITY,
  RCD9.CONFIG_NAME AS DELIVERY_PRIORITY_NAME,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DELIVERY_DUE_SAP_T DWI --SAP交货到期表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD1 --装运点/接收点中文名称
  ON DWI.GROUP_NUM = RCD1.GROUP_NUM
  AND DWI.RECEIVE_POINTS = RCD1.CONFIG_CODE
  AND RCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD2 --路线中文名称
  ON DWI.GROUP_NUM = RCD2.GROUP_NUM
  AND DWI.ROUTE = RCD2.CONFIG_CODE
  AND RCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD3 --销售凭证类型中文名称
  ON DWI.GROUP_NUM = RCD3.GROUP_NUM
  AND DWI.SAL_VCH_TYPE = RCD3.CONFIG_CODE
  AND RCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD4 --销售组织中文名称
  ON DWI.GROUP_NUM = RCD4.GROUP_NUM
  AND DWI.SAL_ORG_CODE = RCD4.CONFIG_CODE
  AND RCD4.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD5 --分销渠道中文名称
  ON DWI.GROUP_NUM = RCD5.GROUP_NUM
  AND DWI.DIS_CHANNEL = RCD5.CONFIG_CODE
  AND RCD5.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD6 --产品组中文名称
  ON DWI.GROUP_NUM = RCD6.GROUP_NUM
  AND DWI.PROD_GROUP = RCD6.CONFIG_CODE
  AND RCD6.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD7 --销售办事处中文名称
  ON DWI.GROUP_NUM = RCD7.GROUP_NUM
  AND DWI.SAL_OFFICE_CODE = RCD7.CONFIG_CODE
  AND RCD7.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD8 --销售组中文名称
  ON DWI.GROUP_NUM = RCD8.GROUP_NUM
  AND DWI.SAL_ORG_GROUP = RCD8.CONFIG_CODE
  AND RCD8.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD9 --交货优先权中文名称
  ON DWI.GROUP_NUM = RCD9.GROUP_NUM
  AND DWI.DELIVERY_PRIORITY = RCD9.CONFIG_CODE
  AND RCD9.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
