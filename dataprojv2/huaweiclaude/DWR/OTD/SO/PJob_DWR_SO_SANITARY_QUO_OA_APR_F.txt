--DEA-A1-005-洁具询价报价OA审批流程主表(工程及家装)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_SANITARY_QUO_OA_APR_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.REQUEST_ID AS REQUEST_ID,
  DWI.OA_CODE AS OA_CODE,
  DWI.WHOLE_A_GAIN_LOSS_AMT AS WHOLE_A_GAIN_LOSS_AMT,
  DWI.WHOLE_B_GAIN_LOSS_AMT AS WHOLE_B_GAIN_LOSS_AMT,
  DWI.TOTAL_AMT AS TOTAL_AMT,
  DWI.CONTRACT_ANNEX AS CONTRACT_ANNEX,
  DWI.RETURN_INFO AS RETURN_INFO,
  DWI.SAP_QUOTE_NUM AS SAP_QUOTE_NUM,
  DWI.QUOTE_VALID_DATE AS QUOTE_VALID_DATE,
  DWI.APPLICATION_DATE AS APPLICATION_DATE,
  DWI.INSURANCE_DATE AS INSURANCE_DATE,
  DWI.PROJ_NAME AS PROJ_NAME,
  DWI.PROJ_REP_NUM1 AS PROJ_REP_NUM1,
  DWI.PROJ_REP_NUM2 AS PROJ_REP_NUM2,
  DWI.WBS AS WBS,
  DWI.CUST_NAME AS CUST_NAME,
  DWI.CUST_ID2 AS CUST_ID2,
  DWI.CUST_NUM2 AS CUST_NUM2,
  CUST.CUST_NAME AS CUST_NAME2,
  --客户名称2
  DWI.OLD_PROJ_REP_NUM AS OLD_PROJ_REP_NUM,
  DWI.CONTACT_PHONE AS CONTACT_PHONE,
  DWI.APPLY_USER AS APPLY_USER,
  DWI.SALES_DEPT AS SALES_DEPT,
  DWI.CUST_ID1 AS CUST_ID1,
  DWI.CUST_NUM1 AS CUST_NUM1,
  CUST1.CUST_NAME AS CUST_NAME1,
  --客户名称1
  DWI.APPLY_DEPT AS APPLY_DEPT,
  DWI.OSS_RETURN_INFO AS OSS_RETURN_INFO,
  DWI.PROJ_CLASS2 AS PROJ_CLASS2,
  DWI.PROPERTY AS PROPERTY,
  DWI.COMPANY_ID AS COMPANY_ID,
  DWI.COMPANY_CODE AS COMPANY_CODE,
  COMP.COMPANY_NAME AS COMPANY_NAME,
  --公司名称
  DWI.APP_NODE_ASSESS_EMP_ID AS APP_NODE_ASSESS_EMP_ID,
  DWI.CHANNEL AS CHANNEL,
  DWI.SALE_ORG_CODE AS SALE_ORG_CODE,
  DWI.GROSS_PROFIT_LESS0_FLAG AS GROSS_PROFIT_LESS0_FLAG,
  DWI.CONFIRM_NODE_FLAG AS CONFIRM_NODE_FLAG,
  DWI.APP_DISCOUNT AS APP_DISCOUNT,
  DWI.FLAG AS FLAG,
  DWI.SIGN_CONTRACT_FLAG AS SIGN_CONTRACT_FLAG,
  DWI.HAVED_CUST_NUM_FLAG AS HAVED_CUST_NUM_FLAG,
  DWI.QUOTE_BUSINESS_SCOPE AS QUOTE_BUSINESS_SCOPE,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SANITARY_QUO_OA_APR_T DWI
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON DWI.CUST_ID2 = CUST.CUST_ID
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST1 --客户主数据表
  ON DWI.CUST_ID1 = CUST1.CUST_ID
  AND CUST1.END_DATE = '4712-12-31'
  AND CUST1.DEL_FLAG = 'N'
  AND CUST1.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_COMPANY_T COMP --公司主数据表
  ON DWI.COMPANY_ID = COMP.COMPANY_ID
  AND COMP.END_DATE = '4712-12-31'
  AND COMP.DEL_FLAG = 'N'
  AND COMP.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
