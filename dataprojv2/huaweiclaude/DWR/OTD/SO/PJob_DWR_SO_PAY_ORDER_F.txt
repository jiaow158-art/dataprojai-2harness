INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_PAY_ORDER_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.ACT_UPDATED_BY AS ACT_UPDATED_BY,
  DWI.ACT_CREATED_BY AS ACT_CREATED_BY,
  DWI.ACCOUNT_ID AS ACCOUNT_ID,
  DWI.EXCHANGE_DATE AS EXCHANGE_DATE,
  DWI.AMOUNT AS AMOUNT,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.MODIFICATION_NUM AS MODIFICATION_NUM,
  DWI.LAST_UPD_BY AS LAST_UPD_BY,
  DWI.LAST_UPD_DATE AS LAST_UPD_DATE,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.CREATED_DATE AS CREATED_DATE,
  DWI.STLMEN_TEMS AS STLMEN_TEMS,
  DWI.STATUS AS STATUS,
  DWI.RECEVING_NAME AS RECEVING_NAME,
  DWI.RECEVING_ACCOUNT AS RECEVING_ACCOUNT,
  DWI.PAY_FOLLOW_ID AS PAY_FOLLOW_ID,
  DWI.PAY_FOLLOW AS PAY_FOLLOW,
  DWI.PAYMENT_TYPE AS PAYMENT_TYPE,
  DWI.PAYER_NAME AS PAYER_NAME,
  DWI.PAYER_ACCOUNT AS PAYER_ACCOUNT,
  DWI.COMPANY_NAME AS COMPANY_NAME,
  DWI.COMPANY_ID AS COMPANY_ID,
  DWI.BANK_ORDER_NUMBER AS BANK_ORDER_NUMBER,
  DWI.BANK_ORDER_ID AS BANK_ORDER_ID,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PAY_ORDER_T DWI
WHERE
  DWI.DEL_FLAG = 'N';
