set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售订单头临时表(关联DD07T配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_HEAD_SAP_TMP03
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  MCD15.DOMAIN_VALUE_NAME AS WHOLE_DELIVERY_STATUS_NAME,
  MCD16.DOMAIN_VALUE_NAME AS DELIVERY_STATUS_NAME,
  MCD17.DOMAIN_VALUE_NAME AS DELIVERY_FROZEN_STATUS_NAME,
  MCD18.DOMAIN_VALUE_NAME AS WHOLE_PROJ_REF_STATUS_NAME,
  MCD19.DOMAIN_VALUE_NAME AS REFER_VCH_TITLE_STATUS_NAME,
  MCD20.DOMAIN_VALUE_NAME AS WHOLE_FORZEN_STATUS_NAME,
  MCD21.DOMAIN_VALUE_NAME AS SHIPMENT_PLAN_STATUS_NAME,
  MCD22.DOMAIN_VALUE_NAME AS TITLE_ROUTINE_STATUS_NAME,
  MCD23.DOMAIN_VALUE_NAME AS TOTAL_NCOMP_STATUS_NAME,
  MCD24.DOMAIN_VALUE_NAME AS BILL_NOWHOLE_STATUS_NAME,
  MCD25.DOMAIN_VALUE_NAME AS PROJ_BILL_INCOMP_STATUS_NAME,
  MCD26.DOMAIN_VALUE_NAME AS VCH_NOWHOLE_STATUS_NAME,
  MCD27.DOMAIN_VALUE_NAME AS DELIVERY_NOWHOLE_STATUS_NAME,
  MCD28.DOMAIN_VALUE_NAME AS PROJ_DELV_NCOMP_STATUS_NAME
FROM
  DWIOTD.DWI_SO_HEAD_SAP_T DWI --SAP销售订单头表
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD15 --所有项目的总体交货状态中文名称
  ON DWI.WHOLE_DELIVERY_STATUS = MCD15.DOMAIN_VALUE_CODE
  AND MCD15.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD16 --交货状态中文名称
  ON DWI.DELIVERY_STATUS = MCD16.DOMAIN_VALUE_CODE
  AND MCD16.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD17 --总体交货冻结状态中文名称
  ON DWI.DELIVERY_FROZEN_STATUS = MCD17.DOMAIN_VALUE_CODE
  AND MCD17.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD18 --所有项目的总参考状态中文名称
  ON DWI.WHOLE_PROJ_REF_STATUS = MCD18.DOMAIN_VALUE_CODE
  AND MCD18.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD19 --参考凭证抬头状态中文名称
  ON DWI.REFER_VCH_TITLE_STATUS = MCD19.DOMAIN_VALUE_CODE
  AND MCD19.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD20 --总体冻结的状态中文名称
  ON DWI.WHOLE_FORZEN_STATUS = MCD20.DOMAIN_VALUE_CODE
  AND MCD20.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD21 --运输计划状态中文名称
  ON DWI.SHIPMENT_PLAN_STATUS = MCD21.DOMAIN_VALUE_CODE
  AND MCD21.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD22 --抬头的常规不完整状态中文名称
  ON DWI.TITLE_ROUTINE_STATUS = MCD22.DOMAIN_VALUE_CODE
  AND MCD22.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD23 --大体上所有项目的总计不完整状态中文名称
  ON DWI.TOTAL_NCOMP_STATUS = MCD23.DOMAIN_VALUE_CODE
  AND MCD23.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD24 --相对于开票的抬头不完整状态中文名称
  ON DWI.BILL_NOWHOLE_STATUS = MCD24.DOMAIN_VALUE_CODE
  AND MCD24.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD25 --所有项目的总计不完整状态开票中文名称
  ON DWI.PROJ_BILL_INCOMP_STATUS = MCD25.DOMAIN_VALUE_CODE
  AND MCD25.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD26 --对于定价,凭证不完整中文名称
  ON DWI.VCH_NOWHOLE_STATUS = MCD26.DOMAIN_VALUE_CODE
  AND MCD26.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD27 --关于交货的抬头不完整状态中文名称
  ON DWI.DELIVERY_NOWHOLE_STATUS = MCD27.DOMAIN_VALUE_CODE
  AND MCD27.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD28 --所有项目的总计不完整状态交货中文名称
  ON DWI.PROJ_DELV_NCOMP_STATUS = MCD28.DOMAIN_VALUE_CODE
  AND MCD28.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
