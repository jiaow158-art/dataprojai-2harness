set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售订单行临时表TMP02(关联通用配置表取中文名称)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_ITEM_SAP_TMP02
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  RCD12.CONFIG_NAME AS AVAIL_CHECK_GROUP_NAME,
  RCD13.CONFIG_NAME AS EXTER_MATERIAL_GROUP_NAME,
  RCD14.CONFIG_NAME AS MATERIAL_PRICE_GROUP_NAME,
  RCD15.CONFIG_NAME AS MATERIAL_GROUP1_NAME,
  RCD16.CONFIG_NAME AS MATERIAL_GROUP2_NAME,
  RCD17.CONFIG_NAME AS MATERIAL_GROUP3_NAME,
  RCD18.CONFIG_NAME AS MATERIAL_GROUP4_NAME,
  RCD19.CONFIG_NAME AS MATERIAL_GROUP5_NAME,
  RCD20.CONFIG_NAME AS TERM_SETTING_GROUP_NAME,
  RCD21.CONFIG_NAME AS SO_REJECT_REASON_NAME,
  RCD22.CONFIG_NAME AS MATERIAL_PRICE_GROUP_NAME
FROM
  DWIOTD.DWI_SO_ITEM_SAP_T DWI --SAP销售订单行表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD12 --可用性检查的检查组中文名称
  ON DWI.GROUP_NUM = RCD12.GROUP_NUM
  AND DWI.AVAIL_CHECK_GROUP = RCD12.CONFIG_CODE
  AND RCD12.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD13 --外部物料组中文名称
  ON DWI.GROUP_NUM = RCD13.GROUP_NUM
  AND DWI.EXTERNAL_MATERIAL_GROUP = RCD13.CONFIG_CODE
  AND RCD13.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD14 --物料价格组中文名称
  ON DWI.GROUP_NUM = RCD14.GROUP_NUM
  AND DWI.MATERIAL_PRICE_GROUP = RCD14.CONFIG_CODE
  AND RCD14.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD15 --物料组1中文名称
  ON DWI.GROUP_NUM = RCD15.GROUP_NUM
  AND DWI.MATERIAL_GROUP1 = RCD15.CONFIG_CODE
  AND RCD15.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD16 --物料组2中文名称
  ON DWI.GROUP_NUM = RCD16.GROUP_NUM
  AND DWI.MATERIAL_GROUP2 = RCD16.CONFIG_CODE
  AND RCD16.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD17 --物料组3中文名称
  ON DWI.GROUP_NUM = RCD17.GROUP_NUM
  AND DWI.MATERIAL_GROUP3 = RCD17.CONFIG_CODE
  AND RCD17.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD18 --物料组4中文名称
  ON DWI.GROUP_NUM = RCD18.GROUP_NUM
  AND DWI.MATERIAL_GROUP4 = RCD18.CONFIG_CODE
  AND RCD18.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD19 --物料组5中文名称
  ON DWI.GROUP_NUM = RCD19.GROUP_NUM
  AND DWI.MATERIAL_GROUP5 = RCD19.CONFIG_CODE
  AND RCD19.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD20 --该物料的科目设置组中文名称
  ON DWI.GROUP_NUM = RCD20.GROUP_NUM
  AND DWI.TERM_SETTING_GROUP = RCD20.CONFIG_CODE
  AND RCD20.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD21 --报价和销售订单的拒绝原因中文名称
  ON DWI.GROUP_NUM = RCD21.GROUP_NUM
  AND DWI.SO_REJECT_REASON = RCD21.CONFIG_CODE
  AND RCD21.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD22 --主要项目的物料定价组中文名称
  ON DWI.GROUP_NUM = RCD22.GROUP_NUM
  AND DWI.MATERIAL_PRICE_GROUP = RCD22.CONFIG_CODE
  AND RCD22.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
