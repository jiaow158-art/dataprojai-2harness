INSERT OVERWRITE table DWROTD.DWR_B1_DBS_SALES_ORDER_HEAD_F 
SELECT 
 t1.DocEntry              doc_num                              -- 订单头
,lpad(t1.customer_code,10,0)        distributor_code           -- 服务商编码
,t1.docstatus           order_status                        -- 订单状态
,t2.cardcode             erp_customer_code                -- erp客户编码
,t2.cardname            erp_customer_name                   -- erp客户名称
,t1.comments            described                           -- 头备注
,t2.cardname           consignee                           -- 收货人
,t1.u_docphone          consignee_phone                     -- 收货人电话      
,t1.address             address                             -- 收货地址
,null                distributor_singer                  -- 经销商签收人
,t1.createdate          distributor_singer_time             -- 经销商签收时间
,'销售出库'                 in_out_flag                      -- 出入库标识
,substr(t1.createdate,1,10) order_ym_attribution               -- 订单年月归属
,current_timestamp() DW_LAST_UPDATE_DATE
,t1.DocNum
FROM DWIOTD.DWI_WM_B1_OINV_T t1 
JOIN DWIOTD.DWI_WM_B1_OCRD_T t2 on t1.customer_code =t2.customer_code and t1.customer_type = t2.customer_type  and t1.cardcode = t2.cardcode
-- where t1.customer_code in ('1000006131','7500310575','310518')
where  t2.u_sfdbs ='Y'
union all 
SELECT 
 t1.DocEntry              doc_num                              -- 订单头
,lpad(t1.customer_code,10,0)    distributor_code            -- 服务商编码
,t1.docstatus           order_status                        -- 订单状态
,t2.cardcode             erp_customer_code                -- erp客户编码
,t2.cardname            erp_customer_name                   -- erp客户名称
,t1.comments            described                           -- 头备注
,t1.cntctcode           consignee                           -- 收货人
,t1.u_docphone          consignee_phone                     -- 收货人电话      
,t1.address             address                             -- 收货地址
,null           distributor_singer                  -- 经销商签收人
,t1.createdate          distributor_singer_time             -- 经销商签收时间
,'销售退货'             in_out_flag                         -- 出入库标识
,substr(t1.createdate,1,10)  order_ym_attribution                 -- 订单年月归属
,current_timestamp() DW_LAST_UPDATE_DATE
,t1.DocNum
FROM DWIOTD.DWI_WM_B1_orin_T t1 
JOIN DWIOTD.DWI_WM_B1_OCRD_T t2 on t1.customer_code =t2.customer_code and t1.customer_type = t2.customer_type  and t1.cardcode = t2.cardcode
-- where t1.customer_code in ('1000006131','7500310575','310518')
where  t2.u_sfdbs ='Y'
-- where t1.customer_code ='1000006131'
