insert overwrite table dwrotd.dwr_otd_so_remain_goods_t partition(period_id_m)
select 
 t1.vbeln as sales_num  -- 销售凭证
,t1.POSNR as sales_item -- 销售凭证项目
,FROM_UNIXTIME(UNIX_TIMESTAMP(t2.ERDAT,'yyyyMMdd'),'yyyy-MM-dd') as create_date --订单创建时间 
,t2.SUBMI as pre_type -- 备货类型
,t2.kunnr as cust_num -- 客户
,FROM_UNIXTIME(UNIX_TIMESTAMP(t2.BSTDK,'yyyyMMdd'),'yyyy-MM-dd') as cust_ref_date -- 客户参考日期
,t3.matnr as material_num  --
,t3.werks as plant  --工厂
,t3.lgort as loc_code  --库存地点
,t3.charg as batch_num  --批次
,t4.posid as wbs_level3  --第三层wbs
,t5.wbs_level2 --第二层wbs
,t5.wbs_level1 --第一层wbs
,t3.kwmeng as sales_qty -- 销售数量
,t1.BMENG as rmn_qty  -- 留货数量
,FROM_UNIXTIME(UNIX_TIMESTAMP(t1.EDATU,'yyyyMMdd'),'yyyy-MM-dd') as rmn_date  -- 留货日期
,current_timestamp() as dw_last_update_date --数据同步时间
,substr(t1.EDATU,1,4)||'-'||substr(t1.EDATU,5,2) as period_id_m
from sdi.sdi_vbep_1001 t1
join sdi.sdi_vbak_1001 t2
on t1.vbeln = t2.vbeln
join sdi.sdi_vbap_1001 t3
on t1.vbeln = t3.vbeln
and t1.posnr = t3.posnr
left join sdi.sdi_prps_1001 t4
on t3.ps_psp_pnr = t4.pspnr
left join DWRDIM.DWR_DIM_WBS_BASIS_INFO_F t5
on t4.posid = t5.wbs_level3
where t1.del_flag = 'N'
and t2.del_flag = 'N'
and t3.del_flag = 'N'
and substr(t1.EDATU,1,6) in (select distinct substr(CPD_UPDAT,1,6) from sdi.sdi_vbap_1001 where dw_last_update_date >= '${PERIOD_ID_D}')
