--DWI_SO_SANIT_QUO_OA_APR_DTL_T 到 DWR_SO_SANIT_QUO_OA_APR_DTL_F
--DEA-A1-005-洁具询价报价OA审批流程明细表(工程及家装)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_SANIT_QUO_OA_APR_DTL_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.MAIN_ID AS MAIN_ID,
  DWI.A_GROSS_PROFIT_AMT AS A_GROSS_PROFIT_AMT,
  DWI.SUBTOTAL_AMT AS SUBTOTAL_AMT,
  DWI.A_GROSS_PROFIT_RATE AS A_GROSS_PROFIT_RATE,
  DWI.B_GROSS_PROFIT_AMT AS B_GROSS_PROFIT_AMT,
  DWI.B_GROSS_PROFIT_RATE AS B_GROSS_PROFIT_RATE,
  DWI.APPLY_DISCOUNT AS APPLY_DISCOUNT,
  DWI.APPROVE_DISCOUNT AS APPROVE_DISCOUNT,
  DWI.APPROVE_BARE_PRICE AS APPROVE_BARE_PRICE,
  DWI.APPROVE_PRICE AS APPROVE_PRICE,
  DWI.APPLY_PRICE AS APPLY_PRICE,
  DWI.ITEM_ABBREVIATION AS ITEM_ABBREVIATION,
  DWI.A_COST AS A_COST,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.B_COST AS B_COST,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.SAP_SETTLEMENT_PRICE AS SAP_SETTLEMENT_PRICE,
  DWI.SALE_QTY AS SALE_QTY,
  DWI.COMPR_SERVICE_RATE AS COMPR_SERVICE_RATE,
  DWI.SERVICE_EVALUATE AS SERVICE_EVALUATE,
  DWI.FREIGHT AS FREIGHT,
  DWI.MIDDLEMEN AS MIDDLEMEN,
  DWI.MATERIAL_GROUP AS MATERIAL_GROUP,
  DWI.MEASUREMENT_UNIT AS MEASUREMENT_UNIT,
  DWI.AGEN AS AGEN,
  DWI.PROD_CHANNELG101_RETAIL AS PROD_CHANNELG101_RETAIL,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SANIT_QUO_OA_APR_DTL_T DWI
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
