--DWI_SO_PRE_ORDER_ITEM_OSS_T 到 DWR_SO_PRE_ORDER_ITEM_OSS_F
--OSS准订单行
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_PRE_ORDER_ITEM_OSS_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.PLAN_ORDER_ITEM_ID AS PLAN_ORDER_ITEM_ID,
  DWI.QUOTA_ITEM_ID AS QUOTA_ITEM_ID,
  DWI.DB_LAST_UPDATE_DATE AS DB_LAST_UPDATE_DATE,
  DWI.CREATE_DATE AS CREATE_DATE,
  DWI.LAST_UPDATE_DATE AS LAST_UPDATE_DATE,
  DWI.ORDER_ID AS ORDER_ID,
  DWI.QTY AS QTY,
  DWI.SYH_PRICE AS SYH_PRICE,
  DWI.CURRENT_PRICE AS CURRENT_PRICE,
  DWI.SRC_FACTORY_ID AS SRC_FACTORY_ID,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.PRODUCT_ID AS PRODUCT_ID,
  DWI.PRICE AS PRICE,
  DWI.SHIP_EXPEN AS SHIP_EXPEN,
  DWI.SHIP_CITY AS SHIP_CITY,
  DWI.LAST_UPDATE_BY AS LAST_UPDATE_BY,
  DWI.SERVER_EXPEN AS SERVER_EXPEN,
  DWI.SHIP_PROVINCE AS SHIP_PROVINCE,
  DWI.SHIP_TYPE AS SHIP_TYPE,
  DWI.STATUS AS STATUS,
  DWI.PROJECT_TYPE AS PROJECT_TYPE,
  DWI.CREATED_BY AS CREATED_BY,
  DWI.MODIFY_QTY AS MODIFY_QTY,
  DWI.CONFLICT_ID AS CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC AS DB_LAST_UPDATE_DATE_SRC,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PRE_ORDER_ITEM_OSS_T DWI
  INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
