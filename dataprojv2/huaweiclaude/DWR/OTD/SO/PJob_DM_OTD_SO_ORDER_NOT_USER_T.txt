WITH DWR_WF_CSTM_PROD_NODE_F AS(
SELECT MATERIAL_NUM
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '01A' THEN OP_TIME END) AS OP_DATE_01A --O1A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '06A' THEN OP_TIME END) AS OP_DATE_06A --06A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '07A' THEN OP_TIME END) AS OP_DATE_07A --07A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '15A' THEN OP_TIME END) AS OP_DATE_15A --15A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '05A' THEN OP_TIME END) AS OP_DATE_05A --05A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '04B' THEN OP_TIME END) AS OP_DATE_04B --04B节点操作日期
  FROM DWRPI.DWR_WF_CSTM_PROD_NODE_F
 WHERE MATERIAL_NUM IS NOT NULL
 GROUP BY MATERIAL_NUM
),TMP_DELIVERY AS(
select delivery as delv_num
      ,delivery_project as delv_item_num
      ,reference_bill_num as sales_num
      ,reference_proj_num as sales_item_num
      ,sum(actual_delivery_qty) as sales_qty
from dwrotd.dwr_wm_ladingbill_item_f
group by delivery
      ,delivery_project
      ,reference_bill_num
      ,reference_proj_num
)
insert overwrite  table dm.dm_otd_so_order_not_user_t

 t1.order_num           -- 订单编号

,t1.submit_time         submission_time -- 提交时间

          || TRUNC((MOD((UNIX_TIMESTAMP(t1.submit_time) - UNIX_TIMESTAMP(t1.create_date)),86400)) / 3600,0) || '小时','.0','')      billing_time              -- 开单耗时

          || TRUNC((MOD((UNIX_TIMESTAMP(t1.review_pass_date) - UNIX_TIMESTAMP(t1.submit_time)),86400)) / 3600,0) || '小时','.0','')      evaluation_time   -- 评审耗时

,t1.batch             batch   -- 批次
,t3.rmn_qty           confirmed_quantity   -- 已确定数量
,case when t1.qty - t3.rmn_qty  > 0 then '待留货'
	 when t1.qty  - t3.rmn_qty  = 0 then '已留货' end  holding_status  -- 留货状态
,case when t3.rmn_qty <> 0 then t3.rmn_date   end          holding_time -- 留货时间
,REPLACE(TRUNC((UNIX_TIMESTAMP(t3.rmn_date) - UNIX_TIMESTAMP(t1.review_pass_date)) / 86400,0) || '天'
          || TRUNC((MOD((UNIX_TIMESTAMP(t3.rmn_date) - UNIX_TIMESTAMP(t1.review_pass_date)),86400)) / 3600,0) || '小时','.0','')  holding_duration                     -- 留货耗时

       when t3.rmn_qty - sum(nvl(t4.sales_qty,0)) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num  )   > 0 then round(datediff(current_date,t3.rmn_date),2)
       when t3.rmn_qty - sum(nvl(t4.sales_qty,0))  over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num  )   =  0 then null
       end			holding_days -- 留货天数
,null                holding_warning -- 留货预警

      when t1.qty - sum(t11.OUT_STOCK_QTY) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num) = 0 then '已出库' end  outbound_status -- 出库状态

	 when t1.qty  - case when  t8.do_num is not null  then sum(t6.actl_delv_qty ) over(partition by t3.sales_num,t3.batch_num,t1.sap_order_line_num ) else 0 end   = 0 then '已签收' end   receiving_status    -- 签收状态
