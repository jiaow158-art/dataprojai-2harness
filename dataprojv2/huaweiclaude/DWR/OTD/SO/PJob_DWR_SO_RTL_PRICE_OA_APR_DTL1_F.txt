--SBC-DCM-A5-001-直营价格审批申请流程明细表1(零售)
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_RTL_PRICE_OA_APR_DTL1_F
SELECT
  DWI.ROW_ID AS ROW_ID,
  DWI.MAIN_ID AS MAIN_ID,
  DWI.SUBTOTAL_AMT AS SUBTOTAL_AMT,
  DWI.GAIN_LOSS_AMT AS GAIN_LOSS_AMT,
  DWI.APP_EXAMINE_DISCOUNT AS APP_EXAMINE_DISCOUNT,
  DWI.APP_EXAMINE_DISCOUNT1 AS APP_EXAMINE_DISCOUNT1,
  DWI.SUBTOTAL_AMT AS SUBTOTAL_AMT,
  DWI.APP_PRICE AS APP_PRICE,
  DWI.APPLY_PRICE AS APPLY_PRICE,
  DWI.SAP_SETTLEMENT_PRICE AS SAP_SETTLEMENT_PRICE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  FAC.FACTORY_NAME AS FACTORY_NAME,
  --工厂名称
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  MAT.MATERIAL_NAME AS MATERIAL_NAME,
  --物料名称
  DWI.REFER_COST_GROSS_RATE AS REFER_COST_GROSS_RATE,
  DWI.MATERIAL_ABBREVIATION AS MATERIAL_ABBREVIATION,
  DWI.B_GROSS_MARGIN_RATE AS B_GROSS_MARGIN_RATE,
  DWI.REFER_COST AS REFER_COST,
  DWI.SALE_QTY AS SALE_QTY,
  DWI.B_COST AS B_COST,
  DWI.SALE_AREA AS SALE_AREA,
  DWI.GROSS_MARGIN_RATE AS GROSS_MARGIN_RATE,
  DWI.A_COST AS A_COST,
  DWI.PROD_LEVEL AS PROD_LEVEL,
  DWI.AMOEBA_COST AS AMOEBA_COST,
  DWI.BATCH_NUM AS BATCH_NUM,
  DWI.SERVICE_FEE AS SERVICE_FEE,
  DWI.ITEM_ID AS ITEM_ID,
  DWI.UNIT_AREA AS UNIT_AREA,
  DWI.MEASURING_UNIT AS MEASURING_UNIT,
  DWI.MATERIAL_GROUP1 AS MATERIAL_GROUP1,
  DWI.ADVERT_POINTS_FLAG AS ADVERT_POINTS_FLAG,
  DWI.YEARS_POINTS_FLAG AS YEARS_POINTS_FLAG,
  DWI.MONTHLY_INTEGRAL AS MONTHLY_INTEGRAL,
  DWI.RENOVATION_POINTS_FLAG AS RENOVATION_POINTS_FLAG,
  DWI.CERAMIC_TILE_DIVISION_FLAG AS CERAMIC_TILE_DIVISION_FLAG,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL1_T DWI
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON DWI.FACTORY_ID = FAC.FACTORY_ID
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON DWI.MATERIAL_ID = MAT.MATERIAL_ID
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
WHERE
  DWI.DEL_FLAG = 'N';
