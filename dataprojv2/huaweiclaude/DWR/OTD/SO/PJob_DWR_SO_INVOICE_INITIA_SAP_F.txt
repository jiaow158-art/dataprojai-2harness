--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP出具发票初始表
INSERT
  OVERWRITE TABLE DWROTD.DWR_SO_INVOICE_INITIA_SAP_F
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.PROD_GROUP AS PROD_GROUP,
  RCD1.CONFIG_NAME AS PROD_GROUP_NAME,
  DWI.AIM_COUNTRY AS AIM_COUNTRY,
  DWI.SAL_CURRENCY_CODE AS SAL_CURRENCY_CODE,
  DWI.ISSUE_INVOICE_TYPE AS ISSUE_INVOICE_TYPE,
  DWI.SAL_DIS_VCH_TYPE AS SAL_DIS_VCH_TYPE,
  MCD1.DOMAIN_VALUE_NAME AS SAL_DIS_VCH_TYPE_CN_NAME,
  DWI.BILL_TYPE AS BILL_TYPE,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  DWI.DIS_CHANNEL AS DIS_CHANNEL,
  RCD2.CONFIG_NAME AS DIS_CHANNEL_NAME,
  DWI.RECEIVE_POINTS AS RECEIVE_POINTS,
  RCD3.CONFIG_NAME AS RECEIVE_POINTS_NAME,
  DWI.SAL_ORG_CODE AS SAL_ORG_CODE,
  RCD4.CONFIG_NAME AS SAL_ORG_CODE_NAME,
  DWI.BILL_DATE AS BILL_DATE,
  DWI.CUST_ADDR_NUM AS CUST_ADDR_NUM,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.VC_NET_WORTH_AMT AS VC_NET_WORTH_AMT,
  DWI.DEL_FLAG AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_INVOICE_INITIA_SAP_T DWI --SAP出具发票初始表
  --关联通用配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD1 --产品组中文名称
  ON DWI.GROUP_NUM = RCD1.GROUP_NUM
  AND DWI.PROD_GROUP = RCD1.CONFIG_CODE
  AND RCD1.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD2 --分销渠道中文名称
  ON DWI.GROUP_NUM = RCD2.GROUP_NUM
  AND DWI.DIS_CHANNEL = RCD2.CONFIG_CODE
  AND RCD2.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD3 --装运点/接收点中文名称
  ON DWI.GROUP_NUM = RCD3.GROUP_NUM
  AND DWI.RECEIVE_POINTS = RCD3.CONFIG_CODE
  AND RCD3.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R RCD4 --销售组织中文名称
  ON DWI.GROUP_NUM = RCD4.GROUP_NUM
  AND DWI.SAL_ORG_CODE = RCD4.CONFIG_CODE
  AND RCD4.LANG_CODE = '1' 
  --关联DD07T配置表，取中文名称
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R MCD1 --销售和分销凭证类别中文名称
  ON DWI.SAL_DIS_VCH_TYPE = MCD1.DOMAIN_VALUE_CODE
  AND MCD1.LANG_CODE = '1'
WHERE
  DWI.DEL_FLAG = 'N';
