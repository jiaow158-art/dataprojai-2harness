--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
insert overwrite table dwrotd.dwr_otd_loading_plan_t partition(period_id_m)
select 
 t.plan_no as plan_num --装载计划号
,t1.spec as plan_item_num--计划项目号
,t1.vbeln_vl as delv_num --提货单号
,t1.posnr_vl as delv_item_num --提货单行号
,t1.vbeln as sales_num --销售单号
,t1.posnr as sales_item_num --销售单行号
,t1.auart as sales_type -- 销售凭证类型
,t1.arktx as sales_item_text --销售订单项目的短文本
,t.cust_no as cust_num --客户编码
,t.cust_name  --客户名称
,t1.matnr as material_num -- 物料编号
,t1.werks as plant --工厂
,t1.lgort as inv_loc --库存地点
,t1.charg as batch_num --批号
,t2.lfimg as sales_qty --销售数量
,t1.lgmng as confirm_qty --确认数量
,t1.vrkme as sales_unit --销售单位
,t1.brgew as gross_wt--毛重
,t1.gewei as wt_unit --重量单位
,t1.volum as volume --体积
,t1.voleh as vol_unit --体积单位
,t1.class_type -- 等级
,t1.groes as dimension --大小/量纲
,t.source_location as source_loc --起始站点
,t.dest_location as dest_loc -- 目的站点
,case when t.pay_method = '' then t.zpay_method else t.pay_method end as pay_method -- 运费支付方式
,t.trans_tools  --运输工具
,t.driver_name -- 司机姓名
,t.driver_id -- 证件号
,substr(t.plan_loading_dt,1,4)||'-'|| substr(t.plan_loading_dt,5,2)||'-'|| substr(t.plan_loading_dt,7,2)||' '||
 substr(t.plan_loading_tm,1,2)||':'|| substr(t.plan_loading_tm,3,2)||':'||substr(t.plan_loading_tm,5,2) as plan_load_date -- 计划装载日期
-- ,substr(t.plan_loading_tm,1,2)||':'|| substr(t.plan_loading_tm,3,2)||':'|| substr(t.plan_loading_tm,5,2) as plan_load_time -- 计划装载时间
,substr(t.create_dt,1,4)||'-'|| substr(t.create_dt,5,2)||'-'|| substr(t.create_dt,7,2)||' '||
 substr(t.create_tm,1,2)||':'|| substr(t.create_tm,3,2)||':'|| substr(t.create_tm,5,2) as create_date -- 创建日期
-- ,substr(t.create_tm,1,2)||':'|| substr(t.create_tm,3,2)||':'|| substr(t.create_tm,5,2) as create_time -- 创建时间
,t.create_user -- 创建用户
,t.proxy_id as cnsgn_num --委托书流水号
,t.car_no as car_num --车牌号
,t.log_information as cour_info --快递信息文本
,t.driver_phone -- 司机电话
,case when t.zis_batch_modify = 0 then '否' 
      when t.zis_batch_modify = 1 then '是' 
      else t.zis_batch_modify end as is_batch_mod --是否修改批次
,case when t.pallet = 0 then '否' 
      when t.pallet = 1 then '是'
      else t.pallet end as is_pallet --是否打托
,t.costcenter  -- 成本中心
,case when t.pyfs = '' then t.zdelivery else t.pyfs end as ship_meth --配运方式
,t.shdz as cnsgc_addr --收货地址
,case when t.shr = '' then t.salesman else t.shr end as cnsgn --收货人
,case when t.shrdh = '' then t.salesman_phone else t.shrdh end as cnsgn_pnone --收货人电话
,t.ywry as emp_name --业务人员
,t.ywrydh as emp_phone --业务电话
,t.zbz as remark -- 备注
,current_timestamp () AS DW_LAST_UPDATE_DATE
,substr(t.create_dt,1,4)||'-'|| substr(t.create_dt,5,2) as period_id_m
from sdi.sdi_zdel_plan_h_1001 t
left join sdi.sdi_zdel_plan_d_1001 t1
on t.plan_no = t1.plan_no
left join sdi.sdi_lips_1001 t2 
on t1.VBELN_VL = t2.vbeln and t1.POSNR_VL = t2.POSNR
where substr(t.create_dt,1,6) in (
 select substr(a.create_dt,1,6) as y_month
 from sdi.sdi_zdel_plan_h_1001 a 
 where substr(a.dw_last_update_date,1,10) >='${start_date}' GROUP BY substr(a.create_dt,1,6)) ;
