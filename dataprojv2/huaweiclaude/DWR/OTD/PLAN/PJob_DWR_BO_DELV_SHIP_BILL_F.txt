INSERT OVERWRITE TABLE DWROTD.DWR_BO_DELV_SHIP_BILL_F
SELECT A.DELIVER_SHIPMENT_BILL_ID AS DELV_SHIP_BILL_ID --出货单票据行ID
      ,A.DELIVER_SHIPMENT_LINE_ID AS DELV_SHIP_LINE_ID --出货单行ID
      ,A.DELIVER_SHIPMENT_ID AS DELV_SHIP_ID --出货单ID
      ,A.PRODUCT_CODE AS MATERIAL_CODE --物料编码
	  ,A.PRODUCT_NAME AS MATERIAL_NAME --物料描述	  
      ,A.CUSTOMER_MATERIAL AS CUST_MATERIAL_CODE --客户产品编码
      ,A.PRODUCT_SPECIFICATION AS DIMENSION --产品规格
      ,COALESCE(A.ORDER_QTY,0) AS DMAND_QTY --要货单量
      ,A.DELIVER_DEMAND_LINE_ID AS DMAND_LINE_ID --要货单行ID
      ,A.MATERIAL_PRICE_ID AS PROJ_LINE_ID --评审单行ID
      ,A.ROLL_PLAN_LINE_ID --滚动行ID
      ,A.INV_QTY --当前可用库存数
      ,A.BATCH_CODE AS BATCH_NUM --批次
      ,COALESCE(A.BATCH_WEIGHT,0) AS BATCH_WT --批次重量
      ,COALESCE(A.TOTAL_WEIGHT,0) AS TOT_WT --总重量
      ,A.UNIT --单位
      ,COALESCE(A.SURPLUS_BILL_QTY,0) AS RSDU_UN_ORDER_QTY --剩余未开单数量
      ,COALESCE(A.SHIPMENT_PLAN_QTY,0) AS SHIP_PLAN_QTY --本次计划出货数
      ,COALESCE(A.SHIPMENT_QTY,0) AS ACTL_SHIP_QTY --实际开单
      ,A.PLAN_FACTORY_CODE --计划出货工厂编码
      ,A.PLAN_WAREHOUSE_CODE AS PLAN_WH_CODE --计划出货仓库编码
      ,A.COMPREHENSIVE_PRICE AS CMPRN_PRC --综合价
      ,A.BARE_PRICE AS BARE_PRC	--裸价
      ,A.FULL_FLAG AS IS_FULL_DELV_CODE --是否整箱编码
	  ,T1.MEANING AS IS_FULL_DELV_NAME --是否整箱描述
	  ,A.PRODUCT_CATEGORY MATERIAL_CAT_NAME --产品品类描述
	  ,A.BILL_STATUS AS BILL_ST_CODE --票据状态编码
	  ,T2.MEANING AS BILL_ST_NAME --票据状态描述
	  ,COALESCE(A.ACTUAL_SHIPMENT_QTY,0) AS RTN_ACTL_SHIP_QTY --回写: 实际发货数
	  ,A.SAP_LINE_NO AS SAP_ORDER_LINE_NUM --SAP订单接口返回的行号
	  ,A.WBS AS PLAN_ITEM_NUM --计划单行号
	  ,A.IS_INSTALL AS IS_INSTL_CODE --是否安装编码
	  ,COALESCE(T3.MEANING,A.IS_INSTALL) AS IS_INSTL_NAME --是否安装描述
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,B.SAP_ORDER_CODE AS SAP_ORDER_NUM --SAP订单号
  FROM SDI.SDI_DELIVER_SHIPMENT_BILL_LINE_1033 A
  LEFT JOIN SDI.SDI_DELIVER_SHIPMENT_HEADER_1033 B
    ON A.DELIVER_SHIPMENT_ID = B.DELIVER_SHIPMENT_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T1
    ON A.FULL_FLAG = T1.LOOKUP_CODE
   AND T1.LOOKUP_TYPE = 'YES_OR_NO'
   AND T1.DELETE_FLAG = 0
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T2
    ON A.BILL_STATUS = T2.LOOKUP_CODE
   AND T2.LOOKUP_TYPE = 'DELIVER_STATUS'
   AND T2.DELETE_FLAG = 0
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 T3
    ON A.IS_INSTALL = T3.LOOKUP_CODE
   AND T3.LOOKUP_TYPE = 'YES_OR_NO'
   AND T3.DELETE_FLAG = 0 
 WHERE A.DELETE_FLAG = 0
