set	hive.exec.max.dynamic.partitions = 100000;

set hive.cbo.enable=true;

with tmp as (
 select  
    c.jm_category_name as product_category,
    a.jm_gc_code as gc_model,
    substr(a.jm_fj_date,1,10) as calday,
    0 as total_grading,
    sum(case when  b.jm_fj_defect_code BETWEEN '020004014' and '020004016' 
    then 1 else 0 end ) as  blowing_dust_defect,   --吹尘
  
    sum(case when  b.jm_fj_defect_code ='020005001'
    then 1 else 0 end ) as slip_defect,         --浆料

    sum(case when   b.jm_fj_defect_code BETWEEN '020003001' and '020003014' 
    then 1 else 0 end ) as glaze_app_defect,   --施釉
    
    sum(case when   b.jm_fj_defect_code  BETWEEN '020004001' AND '020004008'  
    then 1 else 0 end )as  furnace_op_defect,  --司炉 
    
    sum(case when   b.jm_fj_defect_code ='020005002'
    then 1 else 0 end )as  glaze_defect,  --釉料缺陷 
    
    sum(case when  b.jm_fj_defect_code  BETWEEN '020001001' and '020001018'
    then 1 else 0 end ) as casting_defect,  --注浆
    
    sum(case when  b.jm_fj_defect_code BETWEEN '020004015' and '020004016'
    then 1 else 0 end ) as handling_defect  --装卸车
    from  
    SDI.SDI_JM_T_FJ_FIRST_1038 a 
    join SDI.SDI_JM_T_FJ_DEFECT_1038  b on b.jm_fj_id=a.jm_uuid     
    left join SDI.SDI_jm_t_wy_base_gc_1038 c on a.jm_gc_id=c.jm_uuid    
    where    a.jm_is_del=0 and b.jm_is_del=0  and a.JM_IS_BM=0
    group by 
 substr(a.jm_fj_date,1,10),
    a.jm_gc_code,
    c.jm_category_name
   union all 
   select 
    b.jm_category_name as product_category,
    a.jm_gc_code as model,
    substr(a.jm_fj_date,1,10) as calday,       
    sum(case when a.jm_uuid is not null then 1 else 0 end ) as total_grading,
    0,
    0,
    0,
    0,
    0,
    0,
    0
    from   SDI.SDI_JM_T_FJ_FIRST_1038 a 
    left join SDI.SDI_jm_t_wy_base_gc_1038 b on a.jm_gc_id=b.jm_uuid
    where a.jm_is_del=0 and a.JM_IS_BL_COMPLEMENT=0 and a.JM_IS_BM=0    
    group by 
    b.jm_category_name ,
    a.jm_gc_code,
    substr(a.jm_fj_date,1,10)  
   )
INSERT OVERWRITE TABLE  dwrotd.DWR_PAM_BASE_FG_DEFECT_JM_F  PARTITION(PERIOD_ID_M)
 select  
product_category,
gc_model,
calday,
sum(total_grading),
sum(blowing_dust_defect),
sum(slip_defect),
sum(glaze_app_defect),
sum(furnace_op_defect),
sum(glaze_defect),
sum(casting_defect),
sum(handling_defect),
current_timestamp  as last_update_date ,
substr(calday,1,7)  as period_id_m
from 
tmp 
group by  
product_category,
gc_model,
calday,
substr(calday,1,7)
