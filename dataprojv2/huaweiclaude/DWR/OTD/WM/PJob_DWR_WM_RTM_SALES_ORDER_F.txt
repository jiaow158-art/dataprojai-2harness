,customChannel as (
             select t.customer_id,
	                t.channel_code,
	                t.channel_name,
	                t.group_name 
	           from(
	                select t.customer_id,
	                       t.channel_code,
	                       t.channel_name,
	                       t.group_name,
		                   ROW_NUMBER () over(PARTITION by t.customer_id,t.channel_code order by t.creation_date desc) rn
	                  from SDI.SDI_DTCUST_CUSTOMER_CHANNEL_GROUP_1031 t
	               ) t
	         where t.rn = 1
            
			
)
,productInfo as (
 SELECT
 pi.product_line
,pi.product_id
,pc.category_code_key
,pb.brand_name AS brand_name
,pc.category_name
FROM SDI.SDI_prod_info_1031 pi
LEFT JOIN SDI.SDI_prod_brand_1031 pb ON pb.brand_id = pi.brand_id
LEFT JOIN SDI.SDI_prod_category_1031 pc ON pc.category_id = pi.category_id
group by  pi.product_line,pi.product_id,pc.category_code_key,pb.brand_name,pc.category_name
)

,distributorInfo as (
       select t.customer_code,
		t.rtm_customer_code,
		t.customer_type
		from(SELECT cust_code as customer_code,rtm_cust_code as rtm_customer_code,
					cust_type as customer_type,
					ROW_NUMBER() OVER(PARTITION BY rtm_cust_code ORDER BY CASE WHEN is_main = '是' THEN 1 ELSE 0 END DESC) AS RN
					FROM sdi.SDI_B1_CUSTOMER_DATA_1037 
					where cust_type = 'SW'   -- 卫浴and is_rtm = 'Y'
					) t
					where t.rn = 1
					UNION ALL
					SELECT customer_code,
						   customer_code as rtm_customer_code,
						   'CT' AS customer_type   -- 瓷砖
					FROM SDI.SDI_CT_CUSTOMER_DATA_1037 T
					WHERE T.data_source_falg = 'RTM_B1'

)
