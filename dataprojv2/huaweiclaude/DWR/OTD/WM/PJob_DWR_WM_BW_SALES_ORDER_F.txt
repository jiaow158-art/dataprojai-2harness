--性能优化SQL
set hive.cbo.enable=true;				
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_BW_SALES_ORDER_F
SELECT 'CT' AS customer_type, -- 客户类型 CT:瓷砖  SW:洁具
	   CASE WHEN LENGTH(T1.distid) < 10 THEN CONCAT('0000',T1.distid) ELSE T1.distid END AS dealer_code ,    -- 经销商编码
	   CASE WHEN objtype = '13' THEN '出货'
	        ELSE '退货' 
	   end AS out_return_goods, -- 出退货	 
	   T1.DOCENTRY  AS doc_num,   -- 单号
	   CASE WHEN LENGTH(T1.distid) < 10 THEN CONCAT('0000',T1.distid) ELSE T1.distid END AS cust_code,  -- 客户编码
	   T3.customer_name AS dealer_name ,-- 客户名称
	   T1.SALESCHANN as channel_name_cn, -- 渠道
	   '' as cust_address, -- 客户地址
	   CONCAT(SUBSTRING(T1.DOCDATE,1,4),'-',SUBSTRING(T1.DOCDATE,5,2),'-',SUBSTRING(T1.DOCDATE,7,2)) as  doc_date, -- 单据日期
	   T1.itemcode AS material_num, -- 物料编码
	   T2.material_name ,  -- 物料描述
	   T2.matl_group_code, --物料组代码
	   T2.matl_group_name, 		--物料组名称
	   T2.external_matl_group_code, -- 外部物料组
	   T2.external_matl_group_name, -- 外部物料组名称
       T2.matl_type_code   , -- 物料类型
  	   T2.matl_type_name   , --物料类型名称
       T2.product_group_code  ,-- 产品组
  	   T2.product_group_name  , --产品组名称 
       T2.product_classify_code  ,-- 产品分级
       T2.product_classify_name  ,-- 产品分级名称
       T2.product_level_code ,--  '产品层次' 
       T2.product_level_name ,--  '产品层次名称' 
  	   T2.product_level1_code,--   '产品层次1' 
       T2.product_level1_name,--   '产品层次1名称' 
       T2.product_level2_code,--   '产品层次2' 
       T2.product_level2_name,--   '产品层次2名称' 
       T2.product_level3_code,--   '产品层次3' 
       T2.product_level3_name,--   '产品层次3名称'
       T2.product_level4_code,--   '产品层次4'
       T2.product_level4_name,--   '产品层次4名称' 
       CASE WHEN T1.objtype = '13' THEN T1.QUANTITY  
	        ELSE T1.QUANTITY  * -1
	   end AS  material_qty, -- 数量
	   T1.DISTNUMBER as batch_num, -- 批次
	   CASE WHEN T1.distid = '352516' THEN -- 杭州东箭的单价字段存的是行合计
		     CASE WHEN T1.objtype = '13' THEN T1.price / T1.QUANTITY  
		        ELSE (T1.price / T1.QUANTITY)   * -1
		     end
	   ELSE
		      CASE WHEN T1.objtype = '13' THEN T1.price 
		        ELSE T1.price    * -1
		     end
	   END AS Price	,				--单价
	   CASE WHEN T1.distid = '352516' THEN -- 杭州东箭完整家居有限公司
		   CASE WHEN objtype = '13' THEN T1.price  
		        ELSE T1.price  * -1
		   end
	   ELSE
	       CASE WHEN objtype = '13' THEN T1.price  * T1.QUANTITY
		        ELSE (T1.price * T1.QUANTITY) * -1
		   end
	   END AS  line_total_amt,				--行合计
	   T1.UNITMSR AS unit,				--单位
	   '' as bar_code,				--条形码
	   NVL(T1.PRICEBEFDI,0) AS  discount_price_ago_amt,			--折扣前价格
	   0 as discount_price_amt,			--折扣
	   nvl(T1.STOCKPRICE,0) as cost_price_amt,			--成本价
	   0 as cost_total_amt,	--成本合计
	   nvl(T1.GROSSPROFIT,0) as gross_profit_amt,				--毛利
	   0 as no_clear_qty,					--未清数量
	   0 as line_num,					--行号
	   '' as wm_name,				-- 仓库
	   '' as batch_num_stock,		-- 批次库存
	   '' as group_name,					--客户组
	   '' as b1_material_group_name,					--b1物料组
	   '' as cust_from,				-- 客户来源
	   '' as bill_Address,				-- 单据地址
	   '' as cloud_order_flag,-- 云订货下单
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	   T2.MATERIAL_ID, -- 物料代理键
	   T1.SALESCHANN as sales_channel ,  -- 销售渠道
	   NVL(T2.length_1,0) AS length_1,  -- 长度
	   NVL(T2.width,0) AS width,    -- 宽度
	   NVL(T2.height,0) AS height,     --  高度
	   T2.DIMENSION   ,      -- 规格(大小量纲)
	   CASE WHEN T1.distid = '352516' THEN -- 杭州东箭完整家居有限公司
		   CASE WHEN objtype = '13' THEN T1.price  
		        ELSE T1.price  * -1
		   end
	   ELSE
	       CASE WHEN objtype = '13' THEN T1.price  * T1.QUANTITY
		        ELSE (T1.price * T1.QUANTITY) * -1
		   end
	   END AS total_sales_amt,			--销售总额
	   '' as U_jtOrderId --云订货单号
FROM SDI.SDI_ZIM_CUS_DATA_1009 T1 
JOIN DWIMD.dwi_md_data_material_general_t T2
ON T1.itemcode = T2.material_num 
AND T2.prod_property = 'A0001' -- 瓷砖客户
join sdi.sdi_ct_customer_data_1037 T3
ON CASE WHEN LENGTH(T1.distid) < 10 THEN CONCAT('0000',T1.distid) ELSE T1.distid END = T3.customer_code
AND T3.data_source_falg = 'BW'
WHERE T1.distid in ('0000316827','352516') -- 0000316827 福建石狮市三跃建材贸易有限公司 , 352516 杭州东箭完整家居有限公司
AND T1.objtype IN ('13','14') -- 13 销售出库  14 销售退货
;
