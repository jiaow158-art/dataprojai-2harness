WITH DM_B1_CUSTOMER_DATA AS 
(SELECT distinct cust_code as CUSTOMER_CODE,
			    abbreviation as CUSTOMER_SHORT_NAME 
	FROM SDI.sdi_b1_customer_data_1037 t     -- 客户关系外挂表
	where  cust_type = 'SW'
 )
INSERT OVERWRITE TABLE DWROTD.DWR_WM_B1_OTHER_OUTINSTOCK_F
SELECT
    T0.CUSTOMER_TYPE ,-- AS "经销商标识",
    T0.CUSTOMER_CODE ,-- AS "经销商编码",
    T3.CUSTOMER_SHORT_NAME ,--  AS "运营中心",
    '其它入库' AS BILL_TYPE,--   AS "单据类型",
    T2.SERIESNAME,--   "单据序列",
    T0.DOCNUM,--   "单据编号",
    T0.DOCDATE,--   "单据日期",
    T0.CARDCODE,--   "客编",
    T0.CARDNAME ,--  "客户名称",
    T0.DOCTOTAL,--   "单据总金额",
    T0.ADDRESS ,--  "地址",
    T0.COMMENTS ,--  "单据备注",
    T1.LINENUM ,--  "行号",
    T1.ITEMCODE ,--  "物料编码",
    T1.DSCRIPTION ,--  "物料描述",
    T1.QUANTITY ,--  "数量",
    T1.PRICE ,--  "单价",
    T1.LINETOTAL ,--  "行合计",
    T4.WHSNAME ,--  "仓库",
    T1.U_DISTNUMBER ,--  "批次",
    T1.FREETXT ,--  "行备注"
    CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM DWIOTD.DWI_WM_B1_OIGN_T T0
JOIN DM_B1_CUSTOMER_DATA  T3 ON T0.CUSTOMER_CODE =T3.CUSTOMER_CODE   
JOIN DWIOTD.DWI_WM_B1_IGN1_T T1 ON T0.DOCENTRY = T1.DOCENTRY AND T0.CUSTOMER_CODE = T1.CUSTOMER_CODE
INNER JOIN DWIOTD.DWI_WM_B1_NNM1_T T2 ON T0.SERIES=T2.SERIES AND T1.CUSTOMER_CODE = T2.CUSTOMER_CODE
LEFT JOIN DWIOTD.DWI_WM_B1_OWHS_T T4 ON T1.WHSCODE = T4.WHSCODE AND T1.CUSTOMER_CODE = T4.CUSTOMER_CODE
WHERE T0.CUSTOMER_TYPE = 'SW'
AND T1.CUSTOMER_TYPE = 'SW'
AND T2.CUSTOMER_TYPE = 'SW'
AND T4.CUSTOMER_TYPE = 'SW'
UNION ALL
SELECT
    T0.CUSTOMER_TYPE,--    AS "经销商标识",
    T0.CUSTOMER_CODE,--    AS "经销商编码",
    T3.CUSTOMER_SHORT_NAME,--    AS "运营中心",
	'其它出库' AS BILL_TYPE,-- AS  "单据类型",
	T2.SERIESNAME,--  "单据序列",
	T0.DOCNUM,--  "单据编号",
	T0.DOCDATE,--  "单据日期",
	T0.CARDCODE,--  "客编",
	T0.CARDNAME,--  "客户名称",
	T0.DOCTOTAL,--  "单据总金额",
	T0.ADDRESS,--  "地址",
	T0.COMMENTS,--  "单据备注",
	T1.LINENUM,--  "行号",
	T1.ITEMCODE,--  "物料编码",
	T1.DSCRIPTION,--  "物料描述",
	T1.QUANTITY * -1,--  "数量",
	T1.PRICE * -1,-- "单价",
	T1.LINETOTAL * -1,-- "行合计",
	T4.WHSNAME,-- "仓库",
	T1.U_DISTNUMBER,-- "批次",
	T1.FREETXT,-- "行备注"
	CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM DWIOTD.DWI_WM_B1_OIGE_T T0
JOIN DM_B1_CUSTOMER_DATA  T3 
ON T0.CUSTOMER_CODE =T3.CUSTOMER_CODE   
JOIN DWIOTD.DWI_WM_B1_IGE1_T T1 
ON  T0.DOCENTRY = T1.DOCENTRY
AND T0.CUSTOMER_CODE = T1.CUSTOMER_CODE
INNER JOIN DWIOTD.DWI_WM_B1_NNM1_T T2 
ON T0.SERIES=T2.SERIES 
AND T1.CUSTOMER_CODE = T2.CUSTOMER_CODE
LEFT JOIN DWIOTD.DWI_WM_B1_OWHS_T T4 
ON T1.WHSCODE = T4.WHSCODE
AND T1.CUSTOMER_CODE = T4.CUSTOMER_CODE
WHERE T0.CUSTOMER_TYPE = 'SW'
AND T1.CUSTOMER_TYPE = 'SW'
AND T2.CUSTOMER_TYPE = 'SW'
AND T4.CUSTOMER_TYPE = 'SW'
UNION ALL -- RTM其它出入库数据
SELECT 'SW' AS CUSTOMER_TYPE,  -- 客户类型 CT:瓷砖 SW：洁具
	   t1.distributor_code as CUSTOMER_CODE,     -- 经销商编码
       t5.cust_short_name,      -- 经销商名称
       case when t2.order_type = 'otherIn' then '其它入库'
       		when t2.order_type = 'otherOut' then '其它出库'
       end AS BILL_TYPE, -- 单据类型
       --t2.header_id as SERIESNAME, -- 单据序列
	   case when upper(t2.inv_other_out_type) = 'INITIALIZE' then '期初中转'
			when upper(t2.inv_other_out_type) = 'BATCH_ADJUST' then '库存批次调整'
			when upper(t2.inv_other_out_type) = 'BROKEN_OUT' then '破损出库'
			when upper(t2.inv_other_out_type) = 'LOSSES' then '盘亏出库'
			when upper(t2.inv_other_in_type) = 'INITIALIZE' then '期初中转'
			when upper(t2.inv_other_in_type) = 'BATCH_ADJUST' then '库存批次调整'
			when upper(t2.inv_other_in_type) = 'BROKEN_IN' then '破损入库'
			when upper(t2.inv_other_in_type) = 'SURPLUS' then '盘盈入库'
	   end as SERIESNAME, -- 单据序列
       t2.order_code as DOCNUM, -- 单据编号
       t2.creation_date as DOCDATE, -- 单据日期
       t2.supplier_code as CARDCODE , -- 客编
       t2.supplier_name as CARDNAME , -- 客户名称
       t2.sum_amount as DOCTOTAL, -- 单据总金额
       '' as ADDRESS, -- 地址
       t2.described as COMMENTS,  -- 单据备注
       t3.line_id as LINENUM, -- 行号
       t3.product_code as ITEMCODE, -- 物料编码
       t3.product_name as DSCRIPTION, -- 物料描述
	   case when t2.order_type = 'otherIn' and t2.yes_or_no = 'N' then t3.qty
       		when t2.order_type = 'otherOut' and t2.yes_or_no = 'N' then t3.qty * -1
       		when t2.yes_or_no = 'Y' and (t2.other_in_type = 'otherRed' or t2.other_out_type = 'otherRed')then t3.qty * -1
       end as QUANTITY, -- 数量
	   case when t2.order_type = 'otherIn'  and t2.yes_or_no = 'N'  then t3.price
       		when t2.order_type = 'otherOut'  and t2.yes_or_no = 'N' then t3.price * -1
       		when t2.yes_or_no = 'Y' and (t2.other_in_type = 'otherRed' or t2.other_out_type = 'otherRed')then t3.price * -1
       end as PRICE, -- 单价
	   case when t2.order_type = 'otherIn' and t2.yes_or_no = 'N' then t3.amount
       		when t2.order_type = 'otherOut' and t2.yes_or_no = 'N' then t3.amount * -1
       		when t2.yes_or_no = 'Y' and (t2.other_in_type = 'otherRed' or t2.other_out_type = 'otherRed')then t3.amount * -1
       end as LINETOTAL, -- 行合计
       t4.warehouse_name as WHSNAME, -- 仓库
       t3.batch_code as U_DISTNUMBER, -- 批次
       t3.described as FREETXT,   -- 行备注
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM SDI.SDI_DT_DISTRIBUTOR_INFO_1029 t1    -- 分销信息
LEFT JOIN SDI.SDI_DTWMS_INV_SUNDRY_HEADERS_1031 t2   -- 其他出入库头表
on t1.distributor_id = t2.distributor_id 
left join SDI.SDI_DTWMS_INV_SUNDRY_LINE_1031 t3  -- 其他出入库行表
on t2.header_id = t3.header_id 
left join SDI.SDI_DTWMS_WAREHOUSE_INFO_1031 t4   -- 仓库信息表
on t2.warehouse_id = t4.warehouse_id 
join (SELECT rtm_cust_code ,abbreviation as cust_short_name 
	FROM SDI.sdi_b1_customer_data_1037     -- 客户关系外挂表
	where is_main = '是' 
	and cust_type = 'SW') t5
on t1.distributor_code = t5.rtm_cust_code
where t1.business_type IN ('WY','WY_CZ')   -- 卫浴类型
and t2.order_type in ('otherIn','otherOut') -- 订单类型
and t2.order_status NOT IN ('draft','toVoid') -- 订单状态 草稿/作废
;
