INSERT OVERWRITE TABLE DWROTD.dwr_wm_decoration_sales_order_f
select 
  doc_num order_code   -- 单据编号
,sale_order_code       -- 订单编号
,order_status 		   -- 单据状态
,order_status_desc 	   -- 单据状态描述 
,sih.distributor_name  -- 经销商名称
, sih.custom_code 	   -- 客户编码 
, sih.cust_name_cn     -- 客户名称 
, case sih.order_type when 'sale_inventory' then '销售出库' when 'sale_return_inventory' then '退货入库' end  order_type_desc -- 单据类型 
, if(sih.subsidy_status = 'Y', '是', '否') as subsidy_status -- 是否已补贴 
, sih.doc_date   -- 制单日期 
, sih.user_full_name  -- 制单人 
, sih.head_described -- 头备注 
, if(sih.import_from_excel = 'Y', '是', '否') -- 是否Excel导入 
, case sih.aog_way when 'TO_WAREHOUSE' then '到仓' when 'TO_HOME' then '到户' end aog_way  -- 到货方式
, sih.consignee -- 收货人 
, sih.consignee_phone -- 收货人电话 
--, concat(sih.province) as `省市区`
,sih.province -- 省份
, sih.city    -- 城市
, sih.county  -- 县/区
, sih.cust_address -- 详细地址 
, case sih.recipient when 'both' then '双方' when 'dealers' then '经销商' when 'customer' then '装企' end recipient -- 签收方 
, sih.distributor_signer -- 经销商签收人 
, sih.distributor_signer_time -- 经销商签收时间 
, sih.customer_signer -- 装企签收人 
, sih.customer_signer_time -- 装企签收时间 
, sih.order_ym_attribution -- 订单年月归属 
,sih.material_num  -- 产品编码 
, sih.material_name -- 产品名称 
, sih.decoration_product_code -- 装企产品编码 
, sih.decoration_product_name -- 装企产品名称 
, sih.decoration_product_size -- 装企产品尺寸 
, sih.batch  -- 批次 
, sih.unit   -- 销售单位 
, sih.material_qty  -- 出库数量 
, sih.discount_tax_price  tax_price -- 含税单价 
, sih.discount_tax_amount tax_amount -- 含税金额 
, sih.line_described   -- 行备注 
, sih.weight  -- 重量 
, sih.area  -- 面积 
, sih.volume -- 体积 
, sih.square_unit_price -- 平方单价 
,'RTM-B1' SOURCE_DATA 
,CURRENT_TIMESTAMP()
,sih.distributor_code  -- 经销商编码
from DWROTD.DWR_WM_RTM_SALES_ORDER_F sih
where channel_code = '10008' and order_type = 'sale_inventory'
