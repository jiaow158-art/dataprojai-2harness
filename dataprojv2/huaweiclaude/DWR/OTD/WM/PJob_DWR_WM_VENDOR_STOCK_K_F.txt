---SAP供应商特殊库存(K)
set hive.cbo.enable=true;				--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_VENDOR_STOCK_K_F	--DWR-SAP供应商特殊库存(K)
SELECT
	DWVSK.GROUP_NUM					  AS	GROUP_NUM		--集团
	,DWVSK.MATERIAL_ID                    AS	MATERIAL_ID	--物料ID
	,DWVSK.MATERIAL_NUM                   AS	MATERIAL_NUM	--物料编号
	,T.MATERIAL_NAME                  	  AS	MATERIAL_NAME	--物料名称
	,DWVSK.FACTORY_ID                     AS	FACTORY_ID		--工厂ID
	,DWVSK.FACTORY_CODE                   AS	FACTORY_CODE	--工厂编码
	,T1.FACTORY_NAME                   	  AS	FACTORY_NAME	--工厂名称
	,DWVSK.STOCK_PLACE                    AS	STOCK_PLACE		--库存地点
	,DWVSK.BATCH_NUM                      AS	BATCH_NUM		--批号
	,DWVSK.SPECIAL_STOCK_FLAG             AS	SPECIAL_STOCK_FLAG	--特殊库存标识
	,T2.CONFIG_NAME                    	  AS	SPECIAL_STOCK_FLAG_NAME		--特殊库存标识名称
	,DWVSK.SUPPLIER_OR_CREDITOR_ID        AS	SUPPLIER_OR_CREDITOR_ID		--供应商帐户ID
	,DWVSK.SUPPLIER_OR_CREDITOR_NUM       AS	SUPPLIER_OR_CREDITOR_ACCT	--供应商或债权人的帐号
	,T3.SUPPLIER_OR_CREDITOR_NAME      	  AS	SUPPLIER_OR_CREDITOR_NAME	--供应商帐户名称
	,DWVSK.LAST_MXXX_PERIOD               AS	LAST_MXXX_PERIOD	--在最后的 Mxxx 期间之前，期间的 MxxxH 已经存在
	,DWVSK.CURRENT_STOCK_FIN_YEAR         AS	CURRENT_STOCK_FIN_YEAR	--当前实际库存指示的财政年度
	,DWVSK.CREATE_DATE                    AS	CREATE_DATE		--创建日期
	,DWVSK.CREATE_OBJECT_USER_NAME        AS	CREATE_OBJECT_USER_NAME		--创建对象的人员名称
	,DWVSK.CURRENT_PERIOD                 AS	CURRENT_PERIOD	--当前期间（过账期间）
	,DWVSK.CURRENT_PERIOD_FISCAL_YEAR     AS	CURRENT_PERIOD_FISCAL_YEAR	--当前期间的会计年度
	,DWVSK.MIN_ORDER_QTY                  AS	MIN_ORDER_QTY	--最小订购量
	,DWVSK.LAST_UPDATE_DATE               AS	LAST_UPDATE_DATE	--上次更改的日期
	,DWVSK.SUPPLEMENT_QTY                 AS	SUPPLEMENT_QTY	--补充数量
	,DWVSK.DSTRT_STOCK_QTY                AS	DSTRT_STOCK_QTY	--分配的库存数量
	,DWVSK.FROZEN_RETAIL_STOCK            AS	FROZEN_RETAIL_STOCK	--冻结的寄售库存
	,DWVSK.LAST_FROZEN_RETAIL_STOCK       AS	LAST_FROZEN_RETAIL_STOCK	--冻结的寄售库存,前期
	,DWVSK.LIMIT_USE_RETAIL_STOCK         AS	LIMIT_USE_RETAIL_STOCK	--限制使用的寄售库存
	,DWVSK.LAST_PASS_CHECK_DATE           AS	LAST_PASS_CHECK_DATE	--上次过账的盘点日期
	,DWVSK.DEL_FLAG                       AS	DEL_FLAG	--删除标识
	,CURRENT_TIMESTAMP()            	  AS	DW_LAST_UPDATE_DATE	--DW最后更新时间
FROM DWIOTD.DWI_WM_VENDOR_STOCK_K_T DWVSK	--DWI-SAP供应商特殊库存(K)
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T		--关联物料主数据,取MATERIAL_NAME
	 ON T.MATERIAL_ID = DWVSK.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DWVSK.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWVSK.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T1				--关联工厂主数据,取FACTORY_NAME
	 ON  T1.FACTORY_ID = DWVSK.FACTORY_ID		--工厂ID
	 AND T1.GROUP_NUM = DWVSK.GROUP_NUM			--集团
	 AND T1.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWVSK.DEL_FLAG = 'N'					--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表,取特殊库存标识名称中文名称
	 ON T2.CONFIG_CODE = DWVSK.SPECIAL_STOCK_FLAG	--特殊库存标识名称
	 AND T2.GROUP_NUM = DWVSK.GROUP_NUM				--集团
	 AND T2.LANG_CODE = '1'
INNER JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T T3		--关联供应商主数据,取SUPPLIER_OR_CREDITOR_NAME
	 ON T3.SUPPLIER_OR_CREDITOR_ID = DWVSK.SUPPLIER_OR_CREDITOR_ID	--供应商ID
	 AND T3.GROUP_NUM = DWVSK.GROUP_NUM				--集团
	 AND T3.END_DATE ='4712-12-31'					--结束时间为'4712-12-31'
	 AND DWVSK.DEL_FLAG = 'N'						--删除标识为'N'
WHERE DWVSK.DEL_FLAG = 'N'
