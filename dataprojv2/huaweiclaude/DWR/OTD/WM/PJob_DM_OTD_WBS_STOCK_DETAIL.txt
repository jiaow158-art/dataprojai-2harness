INSERT OVERWRITE TABLE dm.dm_otd_wbs_stock_detail PARTITION (partition_ym)
select  t1.sjh,
        t1.posid,          --wbs元素（三层wbs号）
        t1.post1,  -- WBS元素描述
        case when t2.jjyy='转换订单' then '否' else '是'  end is_plan,  -- 是否计划单
        t1.jhddlx, -- 计划订单类型
        t1.ZCJRQ, -- 创建日期
        t1.cpudt_mkpf, -- 留货日期
        t1.zqypcjf,   -- 确认排产交付时间
        t1.zzzjh,   --到达项目地日期
        t1.yqrq,  -- 开始计算逾期日期
        t1.kzts, -- 控制天数
        t1.yqts, -- 逾期天数
        t1.SFYQ,  -- 是否逾期
        t1.werks , --工厂        
        t1.lgort, -- 库存地点        
        t1.charg,--批次        
        t1.matnr,--物料编码        
        t1.maktx,--物料名称        
        t1.mvgr1, --物料组1        
        t1.menge, -- 
        t1.left_menge , -- 留货数量
        t1.mianji, --留货面积        
        t2.pernr, -- 创建员ID
        t2.pernrname,  --计划订单创建人名称
        t3.sales_emp,   -- 销售雇员ID
        t3.sales_emp_name, -- 销售雇员名称
        t3.sales_grp,      -- 销售组
        current_timestamp() as dw_last_update_date,
        date_format(date_sub(current_date(),1),'YYYY-MM') partition_ym
from sdi.sdi_ztmmdg002_detail_1001 t1 
left join sdi.sdi_zpst005_1001 t2 On t1.posid = t2.zjhdid 
left join dwrdim.dwr_dim_wbs_basis_info_f t3 ON t1.posid = t3.wbs_level3;
