-- 按物料批次公司工厂算出期末库存
with tmp as (
select t.*,case when t.stockcat='K' then 'K' else 'NK' end  as stock_k 
   from 
	dm.dm_rpt_zzt_kcmx_others_t t
		  where t.period_id_m='${period_id_m}'
		  and t.matl_type<>'FERT'
		  )
insert OVERWRITE table dm.dm_fin_stock_detail_accage_others_t_01
SELECT
	t.stock_k,
	t.matl_type,
	t.comp_code,
	t.plant,
	t.material,
	COALESCE(t.batch, '#') AS batch,
	SUM(t.quantity) AS quantity,
	SUM(t.zsjkcje) AS zsjkcje
FROM
	tmp t
GROUP BY
	t.stock_k,
	t.comp_code,
	t.plant,
	t.material,
	COALESCE(t.batch, '#'),
	t.matl_type
	;

-- 先进先出，拿到现有库存之前的所有入库时间，拼按时间排序降序
insert OVERWRITE table dm.dm_fin_stock_detail_accage_others_t_02
select t.*,A.budat,A.stock_qty,
sum(A.stock_qty) over(partition by t.stock_k,A.bukrs,A.werks,A.matnr, A.CHARG 
                     order by  case when (A.BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and A.BSTTYP_SG <> 'K')  then 1 else 0 end desc,  -- 优先以上入库类型
                       case when A.cancelled='X' then 0 else 1 end desc,  -- 被冲销的凭证往后
                      A.budat desc,A.LIFNR desc,A.WERKS desc,A.MBLNR desc,A.MJAHR desc,A.ZEILE desc,A.SHKZG desc) as  sum_stockin
from dm.dm_fin_stock_detail_accage_others_t_01   t
join   sdi.sdi_matdoc_1001  A  ON t.comp_code=  A.bukrs 
                               and t.plant=A.werks   
                               and t.material = A.matnr    
                               and coalesce(t.batch,'#') = coalesce(A.CHARG , '#')
                               and t.stock_k = case when substr(A.BSTTYP_SG,1,1)='K' then 'K' else 'NK' end  
where  A.stock_qty>0  and substr(A.budat,1,6)<='${period_id_m}';

-- 先进先出，截取刚好满足现有库存的入库时间
insert OVERWRITE table  dm.dm_fin_stock_detail_accage_others_t_03
select t1.matl_type,t1.material,t1.batch,t1.comp_code,t1.plant, t1.quantity as quantity_sum,t1.zsjkcje as zsjkcje_sum,t1.budat,t1.stock_qty,t1.sum_stockin,t1.stock_k,
      case when sum_stockin>=quantity then  t1.stock_qty-(sum_stockin - quantity) else   t1.stock_qty end tmp_stock_qty,
      case when sum_stockin>=quantity then  t1.stock_qty-(sum_stockin - quantity) else   t1.stock_qty end*(t1.zsjkcje/t1.quantity) tmp_zsjkcje,
      datediff(least(date_sub(current_date,1),date_sub(add_months(from_unixtime(to_unix_timestamp('${period_id_m}','yyyyMM')),1),1)),from_unixtime(to_unix_timestamp(budat,'yyyyMMdd'))) as stockage
  from dm.dm_fin_stock_detail_accage_others_t_02   t1  
where (sum_stockin>=quantity and sum_stockin-stock_qty<quantity) or sum_stockin<quantity;
