INSERT OVERWRITE TABLE DWROTD.DWR_WM_STOCK_PREPARE_FROM_RES
SELECT 
     T.MATERIAL_ID
	,a.matnr AS MATERIAL_NUM
	,T.MATERIAL_NAME
	,T1.FACTORY_ID
	,a.werks AS FACTORY_CODE
	,T1.FACTORY_NAME
	,a.lgort AS INV_LOCATION
	,a.CHARG AS BATCH_NUM
	,a.AUFNR AS RES_ORDER_NO --销售凭证
	,a.AUFPS AS RES_ORDER_ITEM_NO--销售凭证项目
	,a.PSPEL AS WBS_ELEMENT --WBS要素
	,D.SUPPLIER_OR_CREDITOR_ID
    ,D.SUPPLIER_OR_CREDITOR_NUM -- LIFNR
    ,D.SUPPLIER_OR_CREDITOR_NAME
	,SUM(COALESCE(a.erfmg,0)) erfmg
	,SUM(COALESCE(a.enmng,0)) enmng
	,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_RESB_1001 a
left join DWIMD.DWI_MD_MATERIAL_GENERAL_T T	--关联物料主数据表,取MATERIAL_NAME
ON T.MATERIAL_NUM = a.matnr		--物料ID
AND T.GROUP_NUM = a.MANDT			--集团
AND T.END_DATE ='4712-12-31'				--结束时间'4712-12-31'
AND A.DEL_FLAG = 'N'					--删除标识'N'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T T1			--关联工厂主数据表,取FACTORY_NAME
ON  T1.FACTORY_CODE = a.werks		--工厂ID
AND T1.GROUP_NUM = a.MANDT			--集团
AND T1.END_DATE ='4712-12-31'				--结束时间'4712-12-31'
AND A.DEL_FLAG = 'N'					--删除标识'N'
left join SDI.SDI_RKPF_1001 b
on a.RSNUM = b.RSNUM
and COALESCE(b.NPLNR,'') <> ''
left join(
  SELECT PURCHASE_VOUCHER_NUM, -- EBELN
         PURCHASE_VCH_ITEM_NUM1, -- EBELP
		 CUST_NUM -- KUNNR
  FROM DWRPUR.DWR_PTP_PURCHASE_VCH_ITEM_F -- EKPO    可以干掉这段逻辑，因为关联是left join,也没有取出字段
  GROUP BY PURCHASE_VOUCHER_NUM,
           PURCHASE_VCH_ITEM_NUM1,
		   CUST_NUM) c
on a.BANFN = c.PURCHASE_VOUCHER_NUM
and a.BNFPO = c.PURCHASE_VCH_ITEM_NUM1
left join(
	SELECT PURCHASE_VOUCHER_NUM, -- EBELN
		   SUPPLIER_OR_CREDITOR_ID,
		   SUPPLIER_OR_CREDITOR_NUM, -- LIFNR
		   SUPPLIER_OR_CREDITOR_NAME
	FROM DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F -- EKkO
	GROUP BY PURCHASE_VOUCHER_NUM,
	         SUPPLIER_OR_CREDITOR_ID,
		     SUPPLIER_OR_CREDITOR_NUM, -- LIFNR
		     SUPPLIER_OR_CREDITOR_NAME
) d
on a.BANFN = d.PURCHASE_VOUCHER_NUM
WHERE COALESCE(a.xloek,'') = ''
AND COALESCE(a.kzear,'') = ''
AND COALESCE(a.erfmg,0) > 0 
and b.NPLNR is null
group by T.MATERIAL_ID
	,a.matnr
	,T.MATERIAL_NAME
	,T1.FACTORY_ID
	,a.werks
	,T1.FACTORY_NAME
	,a.lgort
	,a.CHARG
	,a.AUFNR
	,a.AUFPS
	,a.PSPEL
	,D.SUPPLIER_OR_CREDITOR_ID
    ,D.SUPPLIER_OR_CREDITOR_NUM -- LIFNR
    ,D.SUPPLIER_OR_CREDITOR_NAME;
