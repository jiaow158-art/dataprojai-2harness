INSERT OVERWRITE TABLE DWROTD.DWR_WM_LADINGBILL_ITEM_TMP001
SELECT
	DWLI.GROUP_NUM			AS	GROUP_NUM
	,DWLI.DELIVERY          AS	DELIVERY
	,DWLI.DELIVERY_PROJECT  AS	DELIVERY_PROJECT
	,T.DOMAIN_VALUE_NAME	AS	GIVEN_INV_REVALUED_NAME
	,T1.DOMAIN_VALUE_NAME	AS	TITLE_ROUTINE_STATUS_NAME
	,T2.DOMAIN_VALUE_NAME	AS	PROJ_TYPE_NAME
	,T3.DOMAIN_VALUE_NAME	AS	DELIVERY_NOWHOLE_STATUS_NAME
	,T4.DOMAIN_VALUE_NAME	AS	ITEM_DEAL_STATUS_NAME
	,T5.DOMAIN_VALUE_NAME	AS	GOODS_MOVE_STATUS_NAME
	,T6.DOMAIN_VALUE_NAME	AS	CONSUMPTION_ENTRY_NAME
	,T7.DOMAIN_VALUE_NAME	AS	PREVIOUS_VCH_TYPE_NAME
FROM DWIOTD.DWI_WM_LADINGBILL_ITEM_T DWLI	---SAP提货单行表（关联DD07T配置表取中文名称）
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T		---特定库存的评估中文名称
	 ON T.DOMAIN_VALUE_CODE = DWLI.GIVEN_INV_REVALUED
	 AND T.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T1		---一般项目的不完整状态中文名称
	 ON T1.DOMAIN_VALUE_CODE = DWLI.TITLE_ROUTINE_STATUS
	 AND T1.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T2		---项目类型中文名称
	 ON T2.DOMAIN_VALUE_CODE = DWLI.PROJ_TYPE
	 AND T2.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T3		---关于交货的项目不完整状态中文名称
	 ON T3.DOMAIN_VALUE_CODE = DWLI.DELIVERY_NOWHOLE_STATUS
	 AND T3.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T4		---销售与分销凭证项目的总体处理状态中文名称
	 ON T4.DOMAIN_VALUE_CODE = DWLI.ITEM_DEAL_STATUS
	 AND T4.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T5		---货物移动状态中文名称
	 ON T5.DOMAIN_VALUE_CODE = DWLI.GOODS_MOVE_STATUS
	 AND T5.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T6		---消耗记帐名称中文名称
	 ON T6.DOMAIN_VALUE_CODE = DWLI.CONSUMPTION_ENTRY
	 AND T6.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R T7		---先前销售和分销凭证的凭证类别中文名称
	 ON T7.DOMAIN_VALUE_CODE = DWLI.PREVIOUS_VCH_TYPE
	 AND T7.LANG_CODE = '1'
WHERE DWLI.DEL_FLAG = 'N'
