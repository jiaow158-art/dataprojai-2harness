set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_LADINGBILL_ITEM_TMP003
SELECT
		DWLI.GROUP_NUM			AS	GROUP_NUM
		,DWLI.DELIVERY			AS	DELIVERY
		,DWLI.DELIVERY_PROJECT	AS	DELIVERY_PROJECT
		,R14.CONFIG_NAME		AS	CUST_GROUP3_NAME
		,R15.CONFIG_NAME		AS	SALES_OFFICE_NAME
		,R16.CONFIG_NAME		AS	WBS_ELEMENT_NAME
		,R17.CONFIG_NAME		AS	AVAIL_CHECK_GROUP_NAME
		,R18.CONFIG_NAME		AS	CUST_GROUP1_NAME
		,R19.CONFIG_NAME		AS	CUST_GROUP2_NAME
		,R20.CONFIG_NAME		AS	MATERIAL_GROUP1_NAME
		,R21.CONFIG_NAME		AS	MATERIAL_GROUP2_NAME
		,R22.CONFIG_NAME		AS	MATERIAL_GROUP3_NAME
		,R23.CONFIG_NAME		AS	MATERIAL_GROUP4_NAME
		,R24.CONFIG_NAME		AS	MATERIAL_GROUP5_NAME
		,R25.CONFIG_NAME		AS	LOAD_GROUP_NAME
		,R26.CONFIG_NAME 		AS	BUSINESS_SECTION_NAME
		,R27.CONFIG_NAME		AS	INTEGRAL_RELATED_NAME
FROM DWIOTD.DWI_WM_LADINGBILL_ITEM_T DWLI
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R14				---客户组 3名称中文名称
	 ON R14.CONFIG_CODE = DWLI.CUST_GROUP3
	 AND R14.GROUP_NUM = DWLI.GROUP_NUM
	 AND R14.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R15				---销售办事处名称中文名称
	 ON R15.CONFIG_CODE = DWLI.SALES_OFFICE_CODE
	 AND R15.GROUP_NUM = DWLI.GROUP_NUM
	 AND R15.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R16				---工作分解结构元素 (WBS 元素)中文名称
	 ON R16.CONFIG_CODE = DWLI.WBS_ELEMENT
	 AND R16.GROUP_NUM = DWLI.GROUP_NUM
	 AND R16.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R17				---可用性检查的检查组名称中文名称
	 ON R17.CONFIG_CODE = DWLI.AVAIL_CHECK_GROUP
	 AND R17.GROUP_NUM = DWLI.GROUP_NUM
	 AND R17.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R18				---客户组 1名称中文名称
	 ON R18.CONFIG_CODE = DWLI.CUST_GROUP1
	 AND R18.GROUP_NUM = DWLI.GROUP_NUM
	 AND R18.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R19				---客户组 2名称中文名称
	 ON R19.CONFIG_CODE = DWLI.CUST_GROUP2
	 AND R19.GROUP_NUM = DWLI.GROUP_NUM
	 AND R19.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R20				---物料组1名称中文名称
	 ON R20.CONFIG_CODE = DWLI.MATERIAL_GROUP1
	 AND R20.GROUP_NUM = DWLI.GROUP_NUM
	 AND R20.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R21				---物料组2名称中文名称
	 ON R21.CONFIG_CODE = DWLI.MATERIAL_GROUP2
	 AND R21.GROUP_NUM = DWLI.GROUP_NUM
	 AND R21.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R22				---物料组3名称中文名称
	 ON R22.CONFIG_CODE = DWLI.MATERIAL_GROUP3
	 AND R22.GROUP_NUM = DWLI.GROUP_NUM
	 AND R22.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R23				---物料组4名称中文名称
	 ON R23.CONFIG_CODE = DWLI.MATERIAL_GROUP4
	 AND R23.GROUP_NUM = DWLI.GROUP_NUM
	 AND R23.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R24			---物料组5名称中文名称
	 ON R24.CONFIG_CODE = DWLI.MATERIAL_GROUP5
	 AND R24.GROUP_NUM = DWLI.GROUP_NUM
	 AND R24.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R25			---装载组名称中文名称
	 ON R25.CONFIG_CODE = DWLI.LOAD_GROUP
	 AND R25.GROUP_NUM = DWLI.GROUP_NUM
	 AND R25.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R26			---业务版块名称中文名称
	 ON R26.CONFIG_CODE = DWLI.BUSINESS_SECTION
	 AND R26.GROUP_NUM = DWLI.GROUP_NUM
	 AND R26.LANG_CODE = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R27			---积分相关名称中文名称
	 ON R27.CONFIG_CODE = DWLI.INTEGRAL_RELATED
	 AND R27.GROUP_NUM = DWLI.GROUP_NUM
	 AND R27.LANG_CODE = '1'
WHERE DWLI.DEL_FLAG = 'N'
