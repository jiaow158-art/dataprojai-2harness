--alter table DWROTD.DWR_WM_STOCK_RES_FROM_PURCHASE add columns(stockout_qty decimal(38,10));
with ekbe as (
  select t1.ebeln ,t1.ebelp ,
		   sum(case when t1.bewtp='L' then t1.menge else 0 end) as lmeng,
		   sum(case when t1.bewtp='E' AND t1.bwart = '101' then t1.menge 
					when t1.bewtp='E' AND t1.bwart = '102' then 0-t1.menge else 0 end) as emeng,
		   max(case when t1.bewtp='L' then t1.budat end) thdat,
		   max(case when t1.bewtp='E' then t1.budat end) shdat
  from sdi.sdi_ekbe_1001 t1
  where 1=1
   and t1.menge >0
  group by t1.ebeln ,ebelp 
  )
INSERT  overwrite table DWROTD.DWR_WM_STOCK_RES_FROM_PURCHASE 
SELECT 
ekpo.matnr as MATERIAL_NUM
,ekko.reswk  as  stockout_FACTORY_CODE     --A.FACTORY_CODE
,ekpo.reslo as INV_LOCATION -- LGORT
,'' BATCH_NUM -- CHARG
,ekpo.ebeln as purchase_order_num --销售凭证 VBELN
,ekpo.ebelp purchase_order_item_num --销售凭证项目 POSNR
,'' as WBS_ELEMENT --WBS 要素 PSPEL
, concat('00000Z',substr(ekpo.werks,1,2),'00') CUST_NUM--客户  CUSTOMER KUNNR	
,ekpo.menge as purchase_qty
--,COALESCE(ekbe.lmeng,0) as confirm_qty
,COALESCE(ekbe.emeng,0) as stockin_qty
,ekpo.menge - greatest(COALESCE(lmeng,0), COALESCE(emeng,0)) as RES_STOCK_QTY
,ekpo.werks as stockin_factory_code --库存收获的工厂
,'' as SALES_ORG_CODE  
,t001l.lgobe  as cust_name
,ekko.ernam as creator_id
,concat(substr(ekko.AEDAT,1,4),'-',substr(ekko.AEDAT,5,2),'-',substr(ekko.AEDAT,7,2))  as res_stock_date
,current_timestamp() as dw_last_update_date
,ekko.ekorg as purchase_org_code --新增采购组织
,case 
	when ekpo.lgort = '0028' then '西南运营中心'
	when ekpo.lgort = '0020' then '粤东运营中心'
	when ekpo.lgort = 'GFZ' then '粤西运营中心'
	when ekpo.lgort = '0030' then '华东运营中心'
	when ekpo.lgort = '0018' then '西北运营中心'
	when ekpo.lgort = '0026' then '西南运营中心'
	when ekpo.lgort='7F21' then '闽琼运营中心'
end as belong_company, -- 新增归属组织字段
case when COALESCE(ekbe.lmeng,0)>=ekpo.menge then concat(substr(ekbe.thdat,1,4),'-',substr(ekbe.thdat,5,2),'-',substr(ekbe.thdat,7,2)) end as stockout_date ,
ekbe.lmeng as  stockout_qty,
ekpo.lgort as stockin_LOCATION,
ekpo.txz01 as item_desc, --订单行描述
concat(substr(ekpo.aedat,1,4),'-',substr(ekpo.aedat,5,2),'-',substr(ekpo.aedat,7,2)) as max_res_stock_date, --sto中订单行最后的修改日期，卫浴最后留货日期逻辑
concat(substr(ekko.BEDAT,1,4),'-',substr(ekko.BEDAT,5,2),'-',substr(ekko.BEDAT,7,2)) as BEDAT, --凭证日期
COALESCE(ekpo.volum,busi_volume) as unit_VOLUM
FROM sdi.sdi_ekko_1001 as ekko 
INNER JOIN sdi.sdi_ekpo_1001 as  ekpo ON ekko.ebeln = ekpo.ebeln
LEFT JOIN  ekbe AS ekbe on ekpo.ebeln  = ekbe.ebeln and ekpo.ebelp  = ekbe.ebelp
LEFT JOIN  sdi.sdi_t001l_1001 t001l ON t001l.werks  = ekpo.werks  and t001l.lgort  = ekpo.lgort 
JOIN  dwimd.dwi_md_data_material_general_t mat on  ekpo.matnr = mat.material_num 
LEFT JOIN (
  		select distinct t1.WERKS,--工厂
			t1.NAME1,--工厂名称
			t3.bukrs,--公司代码
			t3.butxt --公司名称
		from sdi.sdi_T001W_1001 t1
		left join sdi.sdi_t001k_1001 t2 on t1.BWKEY = t2.BWKEY
		left join sdi.sdi_t001_1001 t3 on t2.bukrs = t3.bukrs
		)q1 on ekpo.werks = q1.WERKS
WHERE 1=1
AND ( ekko.bsart = 'ZSTO' OR ekko.bsart = 'ZSUB' )
AND ekpo.menge > 0
and ekko.loekz = ''
AND ekpo.loekz =''
and ekpo.RETPO = ''
and ekpo.ELIKZ = ''
AND coalesce(ekpo.ELIKZ,'')<>'X'
AND COALESCE (ekko.memory,' ')<>'X';  -- 暂存的不完整的
