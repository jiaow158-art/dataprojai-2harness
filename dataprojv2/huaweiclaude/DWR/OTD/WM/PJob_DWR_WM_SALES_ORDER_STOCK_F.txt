---SAP销售订单库存
set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWROTD.DWR_WM_SALES_ORDER_STOCK_F	--DWR.SAP销售订单库存
SELECT
	 DWSOS.GROUP_NUM					  AS	GROUP_NUM
	,DWSOS.MATERIAL_ID                    AS	MATERIAL_ID
	,DWSOS.MATERIAL_NUM                   AS	MATERIAL_NUM
	,T.MATERIAL_NAME                  	  AS	MATERIAL_NAME
	,DWSOS.FACTORY_ID                     AS	FACTORY_ID
	,DWSOS.FACTORY_CODE                   AS	FACTORY_CODE
	,T1.FACTORY_NAME                      AS	FACTORY_NAME
	,DWSOS.STOCK_PLACE                    AS	STOCK_PLACE
	,DWSOS.BATCH_NUM                      AS	BATCH_NUM
	,DWSOS.SPECIAL_STOCK_FLAG             AS	SPECIAL_STOCK_FLAG
	,T2.CONFIG_NAME                    	  AS	SPECIAL_STOCK_FLAG_NAME
	,DWSOS.RETAIL_VOUCHER_NO              AS	RETAIL_VOUCHER_NO
	,DWSOS.RETAIL_VOUCHER_PROJECT_NO      AS	RETAIL_VOUCHER_PROJECT_NO
	,DWSOS.CURRENT_STOCK_FIN_YEAR         AS	CURRENT_STOCK_FIN_YEAR
	,DWSOS.CREATE_DATE                    AS	CREATE_DATE
	,DWSOS.CURRENT_PERIOD                 AS	CURRENT_PERIOD
	,DWSOS.CURRENT_PERIOD_FISCAL_YEAR     AS	CURRENT_PERIOD_FISCAL_YEAR
	,DWSOS.FROZEN_STOCK                   AS	FROZEN_STOCK
	,DWSOS.LAST_FROZEN_STOCK              AS	LAST_FROZEN_STOCK
	,DWSOS.DSTRT_STOCK_QTY                AS	DSTRT_STOCK_QTY
	,DWSOS.CHECK_STOCK                    AS	CHECK_STOCK
	,DWSOS.LAST_QUALITY_CHECK_STOCK       AS	LAST_QUALITY_CHECK_STOCK
	,DWSOS.LAST_PASS_CHECK_DATE           AS	LAST_PASS_CHECK_DATE
	,DWSOS.LAST_LIMIT_USE_STOCK           AS	LAST_LIMIT_USE_STOCK
    ,DWSOS.DEL_FLAG                       AS    DEL_FLAG
    ,CURRENT_TIMESTAMP()            	  AS    DW_LAST_UPDATE_DATE
FROM DWIOTD.DWI_WM_SALES_ORDER_STOCK_T DWSOS		--DWI.SAP销售订单库存
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T T		--关联物料主数据取物料名称
	 ON T.MATERIAL_ID = DWSOS.MATERIAL_ID		--物料ID
	 AND T.GROUP_NUM = DWSOS.GROUP_NUM			--集团
	 AND T.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWSOS.DEL_FLAG = 'N'					--删除标识为'N'
INNER JOIN DWIMD.DWI_MD_FACTORY_T T1				--关联工厂主数据取工厂名称
	 ON  T1.FACTORY_ID = DWSOS.FACTORY_ID		--工厂ID
	 AND T1.GROUP_NUM = DWSOS.GROUP_NUM			--集团
	 AND T1.END_DATE ='4712-12-31'				--结束时间为'4712-12-31'
	 AND DWSOS.DEL_FLAG = 'N'					--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T2		--关联通用配置表
	 ON T2.CONFIG_CODE = DWSOS.SPECIAL_STOCK_FLAG	--取特殊库存标识名称中文名称
	 AND T2.GROUP_NUM = DWSOS.GROUP_NUM				--集团
	 AND T2.LANG_CODE = '1'
WHERE DWSOS.DEL_FLAG = 'N'
