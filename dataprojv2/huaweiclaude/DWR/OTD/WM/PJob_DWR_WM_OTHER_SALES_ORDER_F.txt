INSERT OVERWRITE TABLE DWROTD.DWR_WM_OTHER_SALES_ORDER_F
select 'CT' AS customer_type, -- 客户类型 CT:瓷砖  SW:洁具
	   T.dealer_code, -- 经销商编码
	   CASE WHEN T.out_return_goods IN ('售后销售','样品销售','正常销售') THEN '出货'
	   		ELSE '退货' 
	   END AS out_return_goods, -- 出退货	   
	   T.doc_num ,   -- 单号
	   T.dealer_code AS cust_code,  -- 客户编码
	   T1.customer_name AS cust_name_cn, -- 客户名称
	   T.channel_name_cn, -- 渠道
	   T.cust_address,  -- 客户地址
	   T.doc_date, -- 单据日期
	   CONCAT(t.material_num,'_A') as material_num, -- 物料编码
	   T2.material_name,  --物料描述
	   T2.matl_group_code, --物料组代码
	   T2.matl_group_name, 		--物料组名称
	   T2.external_matl_group_code, -- 外部物料组
	   T2.external_matl_group_name, -- 外部物料组名称
       T2.matl_type_code   , -- 物料类型
  	   T2.matl_type_name   , --物料类型名称
       T2.product_group_code  ,-- 产品组
  	   T2.product_group_name  , --产品组名称 
       T2.product_classify_code  ,-- 产品分级
       T2.product_classify_name  ,-- 产品分级名称
       T2.product_level_code ,--  '产品层次' 
       T2.product_level_name ,--  '产品层次名称' 
  	   T2.product_level1_code,--   '产品层次1' 
       T2.product_level1_name,--   '产品层次1名称' 
       T2.product_level2_code,--   '产品层次2' 
       T2.product_level2_name,--   '产品层次2名称' 
       T2.product_level3_code,--   '产品层次3' 
       T2.product_level3_name,--   '产品层次3名称'
       T2.product_level4_code,--   '产品层次4'
       T2.product_level4_name,--   '产品层次4名称' 
       T.material_qty, --  数量
       T.batch_num, -- 批次
       CASE WHEN T.out_return_goods IN ('售后销售','样品销售','正常销售') THEN T.price
	   		ELSE T.price * -1
	   END as Price,					--单价
	   T.line_total_amt, 	--行合计
	   '' AS unit,				--单位
	   '' as bar_code,				--条形码
	   0 as discount_price_ago_amt,			--折扣前价格
	   0 as discount_price_amt,			--折扣
	   T.cost_price_amt,			--成本价
	   T.cost_total_amt,	--成本合计
	   T.gross_profit_amt,				--毛利
	   T.no_clear_qty,					--未清数量
	   0 as line_num,					--行号
	   T.wm_name ,				-- 仓库
	   '' as batch_num_stock,		-- 批次库存
	   T.group_name,					--客户组
	   '' as b1_material_group_name,					--b1物料组
	   T.cust_from, 	-- 客户来源
	   T.bill_address, -- 单据地址
	   '' as cloud_order_flag,-- 云订货下单
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	   T2.MATERIAL_ID, -- 物料代理键
   	   T.sales_channel,   -- 销售渠道
	   NVL(T2.length_1,0) AS length_1,  -- 长度
	   NVL(T2.width,0) AS width,    -- 宽度
	   NVL(T2.height,0) AS height,     --  高度
	   T2.DIMENSION,         -- 规格(大小量纲)
	   CASE WHEN T.out_return_goods IN ('售后销售','样品销售','正常销售') THEN T.material_qty * T.price
	   		ELSE (abs(T.material_qty) * T.price) * -1
	   END as total_sales_amt,			--销售总额
	   '' as U_jtOrderId --云订货单号
from SDI.SDI_ct_dealer_sales_order_1037 T
JOIN sdi.sdi_ct_customer_data_1037 T1
ON T.dealer_code = T1.customer_code 
JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T2
ON CONCAT(t.material_num,'_A') = T2.material_num
where T.out_return_goods in ('售后销售','样品销售','正常销售','正常销售退货','样品销售退货')
AND T1.data_source_falg = '其它';
