set hive.mapjoin.smalltable.filesize=200000000;

with operation_center_data as(  -- 客户数据
     select case when length(cust_code) <=6 then concat('0000',cust_code) else cust_code end as cust_code  ,
        abbreviation as cust_short_name
     from sdi.sdi_b1_customer_data_1037
     where cust_type = 'SW'
),
dim_data as(   -- 数据发散，从2021年1月开始
  select day_dt
  from DM.dm_dim_date_d T
  where day_dt between '2023-01-01'--ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3)  
  and  '${PERIOD_ID_D}' --'2023-01-01' and  '${PERIOD_ID_D}'
  --group by day_dt
),
hn_jx_fj_gx_pick_data as( -- 湖南 0000310635 江西 0000321104 福建 0000353879 广西 0000318683 提货数
	SELECT t1.pickup_goods_date as begin_pickup_goods_date,
		   date_sub(pickup_goods_date,-89) as end_pickup_goods_date, -- 90后天的提货时间
		   t1.cust_code,
		   t1.material_num,
		   sum(t1.pickup_goods_qty) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) as pickup_goods_amt
	FROM dwrotd.DWR_WM_CUST_PICK_DETAIL_F t1
	where t1.cust_code in ('0000310635','0000353879','0000321104','0000318683')
	--and t1.pickup_goods_date >= ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3)
	group by t1.pickup_goods_date,
		   	 t1.cust_code,
		   	 t1.material_num
),
hn_jx_fj_gx_pick_data01 as(  -- 湖南 0000310635 江西 0000321104 福建 0000353879 广西 0000318683 提货数
	SELECT t1.pickup_goods_date,
		   t1.cust_code,
		   t1.material_num,
		   sum(t1.pickup_goods_qty) over(PARTITION by t1.cust_code,t1.material_num order by t1.pickup_goods_date rows  BETWEEN 89 PRECEDING AND CURRENT ROW) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) over(PARTITION by t1.cust_code,t1.material_num order by t1.pickup_goods_date rows  BETWEEN 89 PRECEDING AND CURRENT ROW) as pickup_goods_amt
	from(
	   select t1.day_dt as pickup_goods_date,
			  t2.cust_code,
			  t2.material_num,
			  nvl(t3.pickup_goods_qty,0) as pickup_goods_qty,
			  nvl(t3.pickup_goods_amt,0) as pickup_goods_amt
	   from dim_data t1
	   left join (select distinct cust_code, material_num from hn_jx_fj_gx_pick_data) t2
	   on 1 = 1
	   left join hn_jx_fj_gx_pick_data t3
	   on t1.day_dt = t3.begin_pickup_goods_date
	   and t2.cust_code = t3.cust_code
	   and t2.material_num = t3.material_num
	   )t1
),
hn_jx_fj_gx_pick_data02 as(
	select t1.pickup_goods_date, 
		   t1.cust_code, 
		   t1.material_num, 
		   max(t1.pickup_goods_qty) as pickup_goods_qty,
		   max(t1.pickup_goods_amt) as pickup_goods_amt,
		  -- max(t1.pickup_goods_amt) / max(t1.pickup_goods_qty) as pickup_goods_price,
		   '' as begin_pickup_goods_date --t2.begin_pickup_goods_date--,t2.end_pickup_goods_date
	from hn_jx_fj_gx_pick_data01 t1
	left join hn_jx_fj_gx_pick_data t2
	on t1.cust_code = t2.cust_code
	and t1.material_num = t2.material_num
	where t1.pickup_goods_date >= t2.begin_pickup_goods_date--ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3)
	and t1.pickup_goods_date <= t2.end_pickup_goods_date
	group by t1.pickup_goods_date, 
		     t1.cust_code, 
		     t1.material_num
),
no_hn_jx_fj_gx_pick_data as( -- 非 湖南 0000310635 江西 0000321104 福建 0000353879 广西 0000318683 提货数
	SELECT t1.pickup_goods_date as begin_pickup_goods_date, -- 提货时间
		   date_sub(pickup_goods_date,-89) as end_pickup_goods_date, -- 90后天的提货时间
		   t1.cust_code,
		   t1.sap_delivery_num,
		   t1.material_num,
		   sum(t1.pickup_goods_qty) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) as pickup_goods_amt
	FROM dwrotd.DWR_WM_CUST_PICK_DETAIL_F t1
	where t1.cust_code not in ('0000310635','0000353879','0000321104','0000318683')
	--and t1.pickup_goods_date >= ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3)
	group by t1.pickup_goods_date,
		   	 t1.cust_code,
		   	 t1.sap_delivery_num,
		   	 t1.material_num
),
no_hn_jx_fj_gx_pick_data01 as(  -- 湖南 0000310635 江西 0000321104 福建 0000353879 广西 0000318683 提货数
	SELECT t1.pickup_goods_date,
		   t1.cust_code,
		   t1.sap_delivery_num,
		   t1.material_num,
		   sum(t1.pickup_goods_qty) over(PARTITION by t1.cust_code,t1.sap_delivery_num,t1.material_num order by t1.pickup_goods_date rows  BETWEEN 89 PRECEDING AND CURRENT ROW) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) over(PARTITION by t1.cust_code,t1.sap_delivery_num,t1.material_num order by t1.pickup_goods_date rows  BETWEEN 89 PRECEDING AND CURRENT ROW) as pickup_goods_amt
	from(
	   select t1.day_dt as pickup_goods_date,
			  t2.cust_code,
			  t2.material_num,
			  t2.sap_delivery_num,
			  nvl(t3.pickup_goods_qty,0) as pickup_goods_qty,
			  nvl(t3.pickup_goods_amt,0) as pickup_goods_amt
	   from dim_data t1
	   left join (select distinct cust_code, sap_delivery_num,material_num from no_hn_jx_fj_gx_pick_data) t2
	   on 1 = 1
	   left join no_hn_jx_fj_gx_pick_data t3
	   on t1.day_dt = t3.begin_pickup_goods_date
	   and t2.cust_code = t3.cust_code
	   and t2.sap_delivery_num = t3.sap_delivery_num
	   and t2.material_num = t3.material_num
	   )t1
),
no_hn_jx_fj_gx_pick_data02 as(
	select t1.pickup_goods_date,
		   t1.cust_code,
		   t1.sap_delivery_num,
		   t1.material_num,
		   max(t1.pickup_goods_qty) as pickup_goods_qty,
		   max(t1.pickup_goods_amt) as pickup_goods_amt,
		   max(t2.begin_pickup_goods_date) as begin_pickup_goods_date--,t2.end_pickup_goods_date
	from no_hn_jx_fj_gx_pick_data01 t1
	left join no_hn_jx_fj_gx_pick_data t2
	on t1.cust_code = t2.cust_code
	and t1.sap_delivery_num = t2.sap_delivery_num
	and t1.material_num = t2.material_num
	where t1.pickup_goods_date >= t2.begin_pickup_goods_date--ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3)
	and t1.pickup_goods_date <= t2.end_pickup_goods_date
	group by   t1.pickup_goods_date,
			   t1.cust_code,
			   t1.sap_delivery_num,
			   t1.material_num
),
cust_pick_data as(
	select t1.pickup_goods_date, 
		   t1.begin_pickup_goods_date,
		   t2.cust_short_name as cust_code, 
		   '' as sap_delivery_num, 
		   t1.material_num, 
		   sum(t1.pickup_goods_qty) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) as pickup_goods_amt
	from hn_jx_fj_gx_pick_data02 t1
	left join operation_center_data t2
	on t1.cust_code = t2.cust_code
	group by t1.pickup_goods_date,
			 t1.begin_pickup_goods_date,
			 t2.cust_short_name,
			 t1.material_num
	union all
	select t1.pickup_goods_date, 
		   t1.begin_pickup_goods_date,
		   t2.cust_short_name as cust_code, 
		   t1.sap_delivery_num, 
		   t1.material_num, 
		   sum(t1.pickup_goods_qty) as pickup_goods_qty,
		   sum(t1.pickup_goods_amt) as pickup_goods_amt
	from no_hn_jx_fj_gx_pick_data02 t1
	left join operation_center_data t2
	on t1.cust_code = t2.cust_code
	group by t1.pickup_goods_date,
			 t1.begin_pickup_goods_date,
			 t2.cust_short_name,
			 t1.sap_delivery_num,
			 t1.material_num
)
INSERT OVERWRITE TABLE DWROTD.DWR_WM_CUST_PICK_DATA_F PARTITION(STAT_MONTH)
select t.pickup_goods_date,
	   t.cust_code,
	   t.sap_delivery_num,
	   t.material_num,
	   t.pickup_goods_qty,
	   current_timestamp() as dw_last_update_date ,
	   t.begin_pickup_goods_date,
	   t.pickup_goods_amt,
	   substring(t.pickup_goods_date,1,7) as stat_month
from cust_pick_data t
where pickup_goods_date >= CASE WHEN substring('${PERIOD_ID_D}',-2) IN ('01','02','03') THEN 
                                   ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3) -- 传昨天，所在月份1号的前3个月
                               ELSE
                        		   CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01')  -- 传昨天，当月
                            END;
