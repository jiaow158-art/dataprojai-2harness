with mat as (  --存在批次为空的物料读不到入库时间
SELECT 
      MATNR AS MATERIAL_NUM,
      MIN(CASE WHEN ((BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG <> 'K') 
                or (BSTTYP_SG = 'K' and BWART ='411')) and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30'  THEN BUDAT 
               WHEN ((BWART in ('101','911','913','Z01','131','531','561','653') and BSTTYP_SG <> 'K') or (BSTTYP_SG = 'K' and BWART = '101')) 
               THEN BUDAT END) AS EARLIEST_VOUCHER_POST_DATE_BW, ---批次最早入库日期(BW通用)
      MIN(CASE WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and WERKS = '7C20' THEN BUDAT 
               WHEN BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and BSTTYP_SG <> 'K' and EBELN not like '47%'
               THEN BUDAT END) AS earliest_voucher_post_date_fc --批次最早入库日期(辅材专用)      
  FROM SDI.SDI_MATDOC_1001
  where charg = '' and charg_cid = ''
  GROUP BY
  MATNR
)
,mat0 as (
select MATNR,CHARG,LIFNR,WERKS,min(BUDAT) as BUDAT
from SDI.SDI_MATDOC_1001
where BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG = 'K'
and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30'
group by MATNR,CHARG,LIFNR,WERKS
)
,MC AS (  ---物料凭证与采购信息记录表关联获取生产日期期间的日期记录
select * from (
select A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG,substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2) as HSDAT,
c.MATNR,c.LIFNR,c.WERKS,c.EKORG,c.ESOKZ,c.KNUMH,c.KSCHL,c.DATAB,c.DATBI
,row_number() over(partition by c.MATNR,c.LIFNR,c.WERKS,B.PRODUCE_DATE,A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG order by c.DATBI desc ) AS RN
from SDI.SDI_MATDOC_1001 A
inner join DWIMD.DWI_MD_MATL_BATCH_INFO_T B
on A.MATNR = B.MATERIAL_NUM and A.CHARG = B.BATCH_NUM
inner join SDI.SDI_A017_1001 c 
on A.WERKS = c.WERKS and A.LIFNR = c.LIFNR and A.MATNR = c.MATNR and ESOKZ = '2' and KSCHL = 'PB00'
and c.DATBI >= substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2) 
and c.DATAB <= substr(B.PRODUCE_DATE,1,4)||substr(B.PRODUCE_DATE,6,2)||substr(B.PRODUCE_DATE,9,2)
where BSTTYP_SG = 'K' AND B.PRODUCE_DATE IS NOT NULL
) a where RN = 1
)
,MC1 AS (  --最小过账日期与采购信息记录表关联记录
select * from (
select A.MATNR,A.CHARG,A.LIFNR,A.WERKS,A.BUDAT
,B.EKORG,B.ESOKZ,B.KNUMH,B.KSCHL,B.DATAB,B.DATBI
,row_number() over(partition by A.MATNR,A.LIFNR,A.WERKS,A.BUDAT,A.MATNR,A.CHARG,A.LIFNR,A.WERKS order by B.DATBI desc ) AS RN
from MAT0 A
inner join SDI.SDI_A017_1001 B 
on A.MATNR = B.MATNR and A.LIFNR = B.LIFNR and A.WERKS = B.WERKS and ESOKZ = '2' and KSCHL = 'PB00'
and B.DATBI >= A.BUDAT and B.DATAB <= A.BUDAT
-- where A.BSTTYP_SG = 'K'
) a where RN = 1
)
,MC2 AS (  ---物料凭证与物料账目表关联获取过账日期期间的日期记录
select * from (
select A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG,A.BUDAT,
B.MATNR,B.BWKEY,B.BWTAR,B.KALNR,C.VPRSV,C.PEINH,C.STPRS,C.PVPRS,C.BDATJ,C.POPER
,row_number() over(partition by B.MATNR,B.BWKEY,B.BWTAR,A.BUDAT,A.MBLNR,A.MJAHR,A.ZEILE,A.SHKZG order by C.BDATJ||substr(C.POPER,2,2) desc) AS RN 
from SDI.SDI_MATDOC_1001 A
inner join SDI.SDI_CKMLHD_1001 B 
on A.MATNR = B.MATNR and A.WERKS = B.BWKEY and A.BWTAR = B.BWTAR 
inner join SDI.SDI_CKMLCR_1001 C 
on B.KALNR = C.KALNR and C.BDATJ = substr(A.BUDAT,1,4) and substr(C.POPER,2,2) = substr(A.BUDAT,5,2) 
where A.BSTTYP_SG = 'K'
) a where RN = 1
)

,A.LGORT as stor_loc --"库存地点",	
,F.LGOBE as stor_loc___t --库存地点_描述
,A.MEINS as unit --"计量单位",
,N.MSEHL as unit___t --计量单位_描述
,W.BKLAS as val_class --"评估类",
,H.BKBEZ as val_class___t --评估类_描述
,A.MAT_PSPNR as wbs_elemt -- "WBS",
,P.POST1 as wbs_elemt___t --Wbs描述
,case when A.MAT_PSPNR = '00000000' then '' else P.POSID end as wbs_elm_ex -- "WBS长号",
,coalesce(substr(BB.earliest_voucher_post_date_bw,1,4)||substr(BB.earliest_voucher_post_date_bw,6,2)||substr(BB.earliest_voucher_post_date_bw,9,2),MAT.earliest_voucher_post_date_bw) as zmmm_o017_zbatch_date --"入库日期",

,case when substr(A.WERKS,1,1) = '3' and B.external_matl_group_code <> '10090' and substr(A.WERKS,1,2) <> '3A' and substr(A.WERKS,1,2) <> '39' then '11000002' --卫浴事业部
  	 when substr(A.WERKS,1,1) = '3' and B.external_matl_group_code <> '10090' and substr(A.WERKS,1,2) = '39' then '11000012' --卫浴鹏嘉家居
  	 when substr(A.WERKS,1,1) = '3' and B.external_matl_group_code <> '10090' and substr(A.WERKS,1,2) = '3A' then '11000011' --卫浴艺耐品牌
	 when A.LGORT = 'BE' and B.external_matl_group_code = '10060' then '11000003' --家居
	 when B.external_matl_group_code = '10060' and A.BUKRS <> '1000' then 
(case when substr(A.WERKS,1,1) = '3' or (substr(A.WERKS,1,1) = '8' and substr(A.WERKS,1,2) <> '8A')
      or substr(A.WERKS,1,2) = '71' or substr(A.WERKS,1,2) = '7D' 
      or substr(A.WERKS,1,2) = '7C' or substr(A.WERKS,1,2) = '73' then ''

	 when B.external_matl_group_code = '10090' or A.WERKS = '7B20' or A.LGORT = 'BH' then '11000001'
	 when B.external_matl_group_code in ('10020','10021','10022','10023','10024') and A.WERKS = '7920' then 
         (case when C.loc_code is not null  then '' else '11000003' end)  
     when A.WERKS = '7120' then '11000003'
     when B.external_matl_group_code = '10070' and substr(A.WERKS,1,2) <> '21' and A.LGORT in ('BD','BA','DG','BX','DL') then '11000010'
	 when B.external_matl_group_code = '10070' and A.WERKS = '1520' and A.LGORT = 'SA' then '11000010'     
	 when substr(A.WERKS,1,2) = '21' and A.LGORT <> '0020' then '11000010' --生态新材  	      
	  end as zdpsyb --事业部编码
--,'' as zdpsyb___t --事业部_描述
,case when A.LGORT = 'A006' then '33000033' --"ART+
	 when B.external_matl_group_code = '10060' and A.BUKRS <> '1000' then
(case when substr(A.WERKS,1,1) = '3' or (substr(A.WERKS,1,1) = '8' and substr(A.WERKS,1,2) <> '8A')
       or substr(A.WERKS,1,2) = '71' or substr(A.WERKS,1,2) = '7D' 
       or substr(A.WERKS,1,2) = '7C' or substr(A.WERKS,1,2) = '73' then '' 
     else (case when C.loc_code is not null  then '' else '33000033' end) end) 
     when A.LGORT = 'BE' and B.external_matl_group_code = '10060' then '33000033'
     when substr(A.WERKS,1,2) = '73' then '33000016' --绿家
     when substr(A.WERKS,1,2) = '7C' or (A.WERKS = '7220' and A.LGORT in ('TY','FC','QA','TA')) then '33000018'-- "辅材
     when B.external_matl_group_code = '10090' or A.WERKS = '7B20' or A.LGORT = 'BH' then '33000019'-- "整装
	 when B.external_matl_group_code in ('10020','10021','10022','10023','10024') and A.WERKS = '7920' then 
       (case when C.loc_code is not null  then '' else '33000014' end)  --DPI
     when A.WERKS = '7120' then '33000001'-- "木业
end as zdpzgsdq --"子公司大区"
--,'' as zdpzgsdq___t --子公司大区_描述
,case when substr(A.WERKS,3,2) = '80' then 'Y' 
	  when substr(A.WERKS,1,1) = '3' and substr(A.WERKS,1,2) <> '37' and (substr(A.LGORT,3,2) = '90' or B.reserved_field_4 = '00003')
	  then 'Y' else 'N' end as zisqc --"是否清仓"
,'' as ziswx --"是否外协"
,case when W.BKLAS in ('7920','7940') then 'CCP' --产成品
      when W.BKLAS in ('7900','7910') then 'BCP' --半成品
      when W.BKLAS in ('7930','7950') then 'WGSP' --外购商品
      when substr(W.BKLAS,1,1) = '3' then 'YCL' --原材料
      else 'QT' end as zmatltype --"物料类别" ,--其他类型
--,'' as zmatltype___t --物料类别_描述
,coalesce(B.product_level3_code,B.product_level2_code,B.product_level1_code) as zprodh3 
,coalesce(B.product_level3_name,B.product_level2_name,B.product_level1_name) as zprodh3___t
,BB.product_area_code_bw AS zww010 --"产区",
,'' as zww010___t --产区_描述
,B.reserved_field_1 as zyl01 --"产品渠道",
,B.reserved_field1_name as zyl01___t  --产品渠道名称
,B.reserved_field_6 as zyl06 --"产品等级",
,B.reserved_field6_name as zyl06___t --"产品等级_描述,
,SUM(A.STOCK_QTY) as quantity--"库存数量" 
,SUM(CASE WHEN A.BSTTYP_SG = 'K' AND W.BWTTY <> 'X'   --k库存且评估类别<>X (非批次成本)
	     THEN COALESCE(y1.KBETR,y3.KBETR,yy.KBETR) / COALESCE(y1.KPEIN,y3.KPEIN,yy.KPEIN) * A.STOCK_QTY --取当期采购价，取不到再取小于入库/过账日期的最新日期采购价
	     WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'S' --为标准价
	     THEN COALESCE(y5.STPRS,y7.STPRS) / COALESCE(y5.PEINH,y7.PEINH) * A.STOCK_QTY  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期标准价，取不到拿小于入库/过账日期往期最大标准价
		 WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'V' --为移动平均价
		 THEN COALESCE(y5.PVPRS,y7.PVPRS) / COALESCE(y5.PEINH,y7.PEINH) * A.STOCK_QTY  --k库存逻辑 正常按成本估算号找不到，按工厂物料评估类取当期移动平均价，取不到拿小于入库/过账日期往期最大移动平均价 
	     WHEN A.BSTTYP_SG <> 'K' AND COALESCE(x.VPRSV,y.VPRSV) = 'S' 
	     THEN COALESCE(x.STPRS,y.STPRS) / COALESCE(x.PEINH,y.PEINH) * A.STOCK_QTY --非K库存按S价取当期标准价，取不到拿往期最大期标准价
		 WHEN A.BSTTYP_SG <> 'K' AND COALESCE(x.VPRSV,y.VPRSV) = 'V' 
		 THEN COALESCE(x.PVPRS,y.PVPRS) / COALESCE(x.PEINH,y.PEINH) * A.STOCK_QTY --非K库存按V价取当期移动平均价，取不到拿往期最大期移动平均价 
		 ELSE 0 END) zsjkcje --实际库存金额
,BB.SHELF_end_DATE as z_vfdat --"货架寿命到期日",
,B.product_level1_code as zprodh1 
,B.product_level1_name as zprodh1___t
,coalesce(B.product_level2_code,B.product_level1_code) as zprodh2 
,coalesce(B.product_level2_name,B.product_level1_name) as zprodh2___t
,coalesce(B.product_level4_code,B.product_level3_code,B.product_level2_code,B.product_level1_code) as zprodh4 
,coalesce(B.product_level4_name,B.product_level3_name,B.product_level2_name,B.product_level1_name) as zprodh4___t
,coalesce(B.product_level5_code,B.product_level4_code,B.product_level3_code,B.product_level2_code,B.product_level1_code) as zprodh5 
,coalesce(B.product_level5_name,B.product_level4_name,B.product_level3_name,B.product_level2_name,B.product_level1_name)  as zprodh5___t

,G.MATL_GROUP3_CODE as matl_grp_3 --物料组3
,G.MATL_GROUP3_NAME as matl_grp_3___t --物料组3描述 
,G.MATL_GROUP4_CODE as matl_grp_4 --物料组4 
,G.MATL_GROUP4_NAME as matl_grp_4___t --物料组4描述

	     THEN COALESCE(y1.KBETR,y3.KBETR,yy.KBETR) / COALESCE(y1.KPEIN,y3.KPEIN,yy.KPEIN)  --取当期入库/过账日期采购价，保底期初数取所在信息记录最小一条采购价

		 WHEN A.BSTTYP_SG = 'K' AND W.BWTTY = 'X' AND COALESCE(y5.VPRSV,y7.VPRSV) = 'V' --为移动平均价
		 THEN COALESCE(y5.PVPRS,y7.PVPRS) / COALESCE(y5.PEINH,y7.PEINH)   --k库存逻辑  正常按成本估算号找不到，按工厂物料评估类取当期入库/过账日期移动平均价
	     WHEN A.BSTTYP_SG <> 'K' AND COALESCE(x.VPRSV,y.VPRSV) = 'S' 
	     THEN COALESCE(x.STPRS,y.STPRS) / COALESCE(x.PEINH,y.PEINH)  --非K库存按S价取当期标准价，取不到拿往期最大期标准价
		 WHEN A.BSTTYP_SG <> 'K' AND COALESCE(x.VPRSV,y.VPRSV) = 'V'

left join dwimd.dwi_md_data_material_general_t B

left join SDI.SDI_JJ_LOC_EXCLUDE_1037 C
on A.LGORT = C.loc_code

on A.WERKS = F.WERKS and A.LGORT = F.LGORT and F.MANDT = '800'
