INSERT OVERWRITE TABLE DWRBIT.DWR_OPR_LOG_USER_TMS_F
SELECT
       A.ID
      ,A.COMPANY_ID
      ,A.EMPLOYEE_ID AS EMP_ID
      ,A.LOGIN_NAME
      ,A.STATUS AS LOGIN_STATUS
      ,A.ENTRY_ID AS CREATE_USER_ID
      ,A.ENTRY_DT AS CREATE_DATE
      ,A.UPDATE_ID AS UPDATE_USER_ID
      ,A.UPDATE_DT AS UPDATE_DATE
      ,A.IS_OFFLINE
      ,A.NICKNAME AS NICK_NAME
	  ,B.CATEGORY AS EMP_CATEGORY
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_PLAT_LOGIN_USER_1005 A
  LEFT JOIN SDI.SDI_PLAT_EMPLOYEE_1005 B
    ON A.LOGIN_NAME = B.EMPLOYEE_NO
