INSERT OVERWRITE TABLE DWRBIT.DWR_OPR_LOG_TMS_F
SELECT ID
      ,A.OPERATOR_ID
      ,A.OPERATOR_DT AS OPERATOR_DATE
      ,A.OPERATION AS OPERATOR_STATUS
      ,A.OPERATION_NO AS OPERATOR_NUM
      ,A.OPERATION_TYPE
      ,A.OPERATOR_NAME
      ,A.IP
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM SDI.SDI_PLAT_OPERATION_LOG_1005 A
