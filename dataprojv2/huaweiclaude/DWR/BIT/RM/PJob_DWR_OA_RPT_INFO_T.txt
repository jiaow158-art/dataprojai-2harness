WITH T AS(
SELECT T.FIELDNAME
      ,H.SELECTVALUE
	  ,H.SELECTNAME
	  ,T.BILLID
  FROM SDI.SDI_WORKFLOW_BILLFIELD_1004 T
 INNER JOIN SDI.SDI_WORKFLOW_SELECTITEM_1004 H
    ON H. FIELDID = T.ID
 WHERE T.BILLID = '-1479'
)
INSERT OVERWRITE TABLE DWRBIT.DWR_OA_RPT_INFO_T
SELECT  A.ID --ID
       ,A.REQUESTID AS RQST_ID --请求ID
       ,A.YYXT AS SYS_CODE --应用系统编码
	   ,T1.SELECTNAME AS SYS_NAME --应用系统描述
       ,A.BBBM AS RPT_CODE --报表编码
       ,A.BBMC AS RPT_NAME --报表名称
       ,A.ML AS RPT_DRCTY --目录
       ,A.BBFL AS RPT_TYPE_CODE --报表分类编码
	   ,T2.SELECTNAME AS RPT_TYPE_NAME --报表分类描述
       ,A.AQDJ AS SCTY_LVL_CODE --安全等级编码
	   ,T3.SELECTNAME AS SCTY_LVL_NAME --安全等级描述
       ,A.YWZRR AS BUSI_EMP_ID --业务责任人ID
	   ,T4.SAPID AS BUSI_EMP_NUM --业务责任人工号
	   ,T4.LASTNAME AS BUSI_EMP_NAME --业务责任人姓名
       ,A.JSZRR AS DEV_EMP_ID --技术责任人ID
	   ,T5.SAPID AS DEV_EMP_NUM --技术责任人工号
	   ,T5.LASTNAME AS DEV_EMP_NAME --技术责任人姓名
       ,A.SJTJZQ AS DATA_PER_CODE --数据统计周期编码
	   ,T6.SELECTNAME AS DATA_PER_NAME --数据统计周期描述
       ,A.XQLX AS RQST_TYPE_CODE --需求类型编码
	   ,T7.SELECTNAME AS RQST_TYPE_NAME --需求类型描述
       ,A.SQBZJSBM AS ROLE_CODE --授权标准角色编码
       ,A.SQBZJSMC AS ROLE_NAME --授权标准角色名称
       ,A.SFXX AS IS_DOWN_CODE --是否下线编码
	   ,T8.SELECTNAME AS IS_DOWN_NAME --是否下线描述
       ,A.SXSJ AS UP_DATE --上线日期
       ,A.XXSJ AS DOWN_DATE --下线日期
	   ,CASE WHEN A.BBMC LIKE '%填报%' THEN '是' ELSE '否' END AS IS_UPLOAD --是否填报
	   ,CASE WHEN A.BBMC LIKE '%预警%' THEN '是' ELSE '否' END AS IS_WARN --是否预警
       ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	   ,COALESCE(B.UP_CNT,1) AS UP_CNT --上线次数
  FROM SDI.SDI_UF_SJBBQD_1004 A
  LEFT JOIN T AS T1
    ON A.YYXT = T1.SELECTVALUE
   AND T1.FIELDNAME = 'YYXT'
  LEFT JOIN T AS T2
    ON A.BBFL = T2.SELECTVALUE
   AND T2.FIELDNAME = 'BBFL'
  LEFT JOIN T AS T3
    ON A.AQDJ = T3.SELECTVALUE
   AND T3.FIELDNAME = 'AQDJ'
  LEFT JOIN SDI.SDI_HRMRESOURCE_1004 T4
	ON A.YWZRR = T4.ID  
   AND T4.DEL_FLAG = 'N'
  LEFT JOIN SDI.SDI_HRMRESOURCE_1004 T5
	ON A.JSZRR = T5.ID  
   AND T5.DEL_FLAG = 'N'
  LEFT JOIN T AS T6
    ON A.SJTJZQ = T6.SELECTVALUE
   AND T6.FIELDNAME = 'SJTJZQ'
  LEFT JOIN T AS T7
    ON A.XQLX = T7.SELECTVALUE
   AND T7.FIELDNAME = 'XQLX'
  LEFT JOIN T AS T8
    ON A.SFXX = T8.SELECTVALUE
   AND T8.FIELDNAME = 'SFXX'
  LEFT JOIN (SELECT BBBMGXXX_MX1 --模板编码
                   ,COUNT(1) AS UP_CNT --上线次数
               FROM SDI.SDI_FORMTABLE_MAIN_1482_DT1_1004
			  GROUP BY BBBMGXXX_MX1
			) B
	ON B.BBBMGXXX_MX1 = A.BBBM
