WITH T1 AS(
SELECT B.FLOW_ID
--非自动评价评分星级在2024-03-05 16:00前是满分5星，1星20分，后修改为满分10星，1星10分
--自动评价评分星级在2024-03-26 00:00前是满分5星，1星20分，后修改为满分10星，1星10分
      ,CASE WHEN SUBSTR(B.CREATE_TIME,1,16) <= '2024-03-05 16:00' AND COALESCE(C.PJ_TYPE,0) = 0
	        THEN ROUND(SUM(A.ANSWER_TEXT)/3,1) * 10
			WHEN SUBSTR(B.CREATE_TIME,1,16) <= '2024-03-26 00:00' AND COALESCE(C.PJ_TYPE,0) = 1
	        THEN ROUND(SUM(A.ANSWER_TEXT)/3,1) * 10
			ELSE ROUND(SUM(A.ANSWER_TEXT)/3,1) * 10
			 END AS PF 
  FROM SDI.SDI_AA_PAPER_ANSWER_OPTION_1013 A
 INNER JOIN SDI.SDI_AA_PAPER_ANSWER_1013 B
    ON A.ANSWER_ID = B.ANSWER_ID
 INNER JOIN SDI.SDI_AA_FLOW_1013 C
    ON B.FLOW_ID = C.FLOW_ID
 WHERE A.QUESTION_ID IN (81,82,83)
 GROUP BY B.FLOW_ID,B.CREATE_TIME,COALESCE(C.PJ_TYPE,0)
),T2 AS(
SELECT DISTINCT R1.USERID
  FROM SDI.SDI_ITSM_SYS_USER_ROLE_1013 R1
 INNER JOIN SDI.SDI_ITSM_SYS_ROLE_1013 R2
    ON R1.ROLEID = R2.ROLEID
   AND R2.ROLENAME = '二线工程师'
 WHERE R1.USERID NOT IN(SELECT USERID FROM SDI.SDI_ITSM_SYS_USER_ROLE_1013 
                         WHERE ROLEID = 'dfa6c6344e114b1280ab0363580e228a')
),T3 AS(
--一条工单存在多个当前处理人，需要转成合并
SELECT FLOW_ID
      ,CONCAT_WS(',',COLLECT_LIST(USER_NAME)) AS USER_NAME
      ,CONCAT_WS(',', COLLECT_LIST(USER_ID)) AS USER_ID
  FROM SDI.SDI_AA_FLOW_USER_HANDING_1013
 GROUP BY FLOW_ID
)

    ON SPLIT(A.CTI_NAME,'-')[2] = C.SYSTEM_NAME
   AND SPLIT(A.CTI_NAME,'-')[0] NOT IN('桌面终端问题','基础架构问题','信息安全') --目录1为3个系统直接取目录1
  LEFT JOIN SDI.SDI_ITIL_SYSTEM_OP_DATE_1037 D
    ON SPLIT(A.CTI_NAME,'-')[3] = D.SYSTEM_NAME
   AND SPLIT(A.CTI_NAME,'-')[0] NOT IN('桌面终端问题','基础架构问题','信息安全') --目录1为3个系统直接取目录1	
  LEFT JOIN SDI.SDI_ITSM_CONFIG_DICTIONARIES_1013 E
    ON E.DICTIONARIESCODE = A.BUSINESS_STATUS

    ON A.FLOW_ID = F.EVENT_ID  
  LEFT JOIN SDI.SDI_ITSM_SYS_USER_1013 G
    ON G.USERID = CASE WHEN COALESCE(F.EVENT_REQUEST_ID,'') = '' THEN F.EVENT_WRITEUSER_ID

					    END

    ON A.FLOW_ID = H.FLOW_ID
  LEFT JOIN T2 AS I
    ON A.BUSINESS_HANDLE_USERID = I.USERID

    ON J.EMP_NAME = A.BUSINESS_HANDLE_USERNAME
  LEFT JOIN SDI.SDI_V_XYGDSH_1013 K
    ON A.FLOW_NO = K.FLOW_NO
  LEFT JOIN T3 O
    ON A.FLOW_ID = O.FLOW_ID
  LEFT JOIN SDI.SDI_IT_IMPLEMENTER_1037 P 
    ON A.BUSINESS_START_USERNAME = P.EMP_NAME
  LEFT JOIN SDI.SDI_IT_IMPLEMENTER_1037 Q 
    ON A.BUSINESS_CLOSE_USERNAME = Q.EMP_NAME
  LEFT JOIN SDI.SDI_IT_IMPLEMENTER_1037 R 
    ON O.USER_NAME = R.EMP_NAME 	
  LEFT JOIN SDI.SDI_ITIL_SYSTEM_REL_1037 U1
    ON CASE WHEN A.CTI_NAME LIKE '%法大大系统%' THEN 'OA系统'
	        WHEN SPLIT(A.CTI_NAME,'-')[2] = 'B1' THEN 'B1系统'
			ELSE COALESCE(COALESCE(D.SYSTEM_NAME,C.SYSTEM_NAME),B.SYSTEM_NAME)
			 END = COALESCE(u1.old_system_name,'SNULL')		 
  LEFT JOIN SDI.SDI_STA_ORDER_1013 ST 
    ON ST.ORDER_ID= A.FLOW_ID
  LEFT JOIN SDI.SDI_AA_FLOW_CLSC_1013 S
    ON A.FLOW_ID = S.FLOW_ID
