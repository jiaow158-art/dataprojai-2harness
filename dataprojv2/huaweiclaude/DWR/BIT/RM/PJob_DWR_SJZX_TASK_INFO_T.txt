WITH T1 AS(
SELECT ID
      ,ROLE_LIST
	  ,ROLE_ID
  FROM SDI.SDI_FINE_SCHEDULE_TASK_1035 A
LATERAL VIEW EXPLODE(SPLIT(REPLACE(REPLACE(REPLACE(ROLE_LIST,'"',''),'[',''),']',''),',')) TABLE_VIEW AS ROLE_ID
),T2 AS(
SELECT DISTINCT 
       T1.ID
      ,T1.ROLE_ID
	  ,C.USERNAME AS USER_NUM
	  ,C.REALNAME AS USER_NAME
  FROM T1
 INNER JOIN SDI.SDI_FR_FINE_USER_ROLE_MIDDLE_1035 B
    ON T1.ROLE_ID = B.ROLEID
 INNER JOIN SDI.SDI_FR_FINE_USER_1035 C
    ON B.USERID = C.ID
   AND C.ENABLE = 1
),T3 AS(
SELECT ID
      ,CONCAT_WS(',',COLLECT_SET(CAST(USER_NUM AS STRING))) AS USER_NUM
	  ,CONCAT_WS(',',COLLECT_SET(CAST(USER_NAME AS STRING))) AS USER_NAME
  FROM T2
 GROUP BY ID
),T4 AS(
SELECT ID
      ,USER_LIST
	  ,USER_ID
  FROM SDI.SDI_FINE_SCHEDULE_TASK_1035 A
LATERAL VIEW EXPLODE(SPLIT(REGEXP_REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(USER_LIST,'"',''),'[',''),']',''),'(',''),')',''),'[^0-9,]*',''),',')) TABLE_VIEW AS USER_ID
),T5 AS(
SELECT T4.ID
      ,T4.USER_ID
	  ,C.USERNAME AS USER_NUM
	  ,C.REALNAME AS USER_NAME
  FROM T4
 INNER JOIN SDI.SDI_FR_FINE_USER_1035 C
    ON T4.USER_ID = C.USERNAME
   AND C.ENABLE = 1
),T6 AS(
SELECT ID
      ,CONCAT_WS(',',COLLECT_SET(CAST(USER_NUM AS STRING))) AS USER_NUM
	  ,CONCAT_WS(',',COLLECT_SET(CAST(USER_NAME AS STRING))) AS USER_NAME
  FROM T5
 GROUP BY ID
)
INSERT OVERWRITE TABLE DWRBIT.DWR_SJZX_TASK_INFO_T
SELECT A.ID AS TASK_ID --任务ID
      ,A.TASKNAME AS TASK_NAME --任务名称
      ,A.CREATOR AS CREATE_EMP_NUM --创建人工号
	  ,A.TEMPLATEPATH AS TEMPL_PATH_RTE --模板路径
	  ,SPLIT(TEMPLATEPATH,'/')[LENGTH(TEMPLATEPATH) - LENGTH(REPLACE(TEMPLATEPATH,'/',''))] AS TEMPL_NAME --模板名称
	  ,CASE WHEN SUBSTR(SPLIT(TEMPLATEPATH,'/')[LENGTH(TEMPLATEPATH) - LENGTH(REPLACE(TEMPLATEPATH,'/',''))],1,1) = 'M'
	        THEN SUBSTR(SPLIT(TEMPLATEPATH,'/')[LENGTH(TEMPLATEPATH) - LENGTH(REPLACE(TEMPLATEPATH,'/',''))],1,6)
			WHEN SUBSTR(SPLIT(TEMPLATEPATH,'/')[LENGTH(TEMPLATEPATH) - LENGTH(REPLACE(TEMPLATEPATH,'/',''))],1,1) IN('F','S','P','H')
			THEN SUBSTR(SPLIT(TEMPLATEPATH,'/')[LENGTH(TEMPLATEPATH) - LENGTH(REPLACE(TEMPLATEPATH,'/',''))],1,5)
			 END AS TEMPL_CODE --模板编码
	  ,A.FILECLEARCOUNT AS FILE_SAVE_CNT --文件保留次数
	  ,A.PREFIRETIME AS RUN_TIME --最近的运行时间
	  ,A.NEXTFIRETIME AS NEXT_RUN_TIME --下一次运行时间
	  ,A.TASKSTATE AS TASK_ST_CODE --任务状态编码
	  ,CASE WHEN A.TASKSTATE = 0 THEN '启动'
	        WHEN A.TASKSTATE = 1 THEN '暂停'
			WHEN A.TASKSTATE = 2 THEN '已结束'
			 END AS TASK_ST_NAME --任务状态描述
	  ,A.TASKTYPE AS TASK_TYPE_CODE --任务类型编码
	  ,CASE WHEN A.TASKTYPE = 0 THEN '无调度对象'
	        WHEN A.TASKTYPE = 1 THEN '报表模板'
			WHEN A.TASKTYPE = 2 THEN 'BI模板'
			 END AS TASK_TYPE_NAME --任务类型描述
	  ,A.TRIGGERGROUP AS TRIGR_GRP_NAME --触发器组
	  ,A.USERGROUP AS USER_GRP_NAME --用户组
	  ,REPLACE(REPLACE(REPLACE(ROLE_LIST,'"',''),'[',''),']','') AS ROLE_ID_LIST --角色ID列表
	  ,REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(USER_LIST,'"',''),'[',''),']',''),'(',''),')','') AS USER_LIST --用户列表
	  ,COALESCE(B.USER_NUM,C.USER_NUM) AS USER_NUM --用户账号
	  ,COALESCE(B.USER_NAME,C.USER_NAME) AS USER_NAME --用户姓名
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN SUBSTR(A.OUTPUTSTR,1,1) = '1' THEN '邮件'
	        WHEN SUBSTR(A.OUTPUTSTR,1,1) = '9' THEN '企微'
			 END AS TRML_TYPE_NAME --终端类型
  FROM SDI.SDI_FINE_SCHEDULE_TASK_1035 A
  LEFT JOIN T3 AS B
    ON A.ID = B.ID
  LEFT JOIN T6 AS C
    ON A.ID = C.ID
