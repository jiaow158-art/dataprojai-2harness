insert overwrite TABLE DWRFIN.DWR_COST_TRVAPP_F
select 
t1.INTEGRATION_ID          AS INTEGRATION_ID,
t1.BUSINESS_CODE           AS BUSINESS_CODE,
t1.FORM_NAME               AS FORM_NAME,
t1.RELATED_BUSINESS_CODE   AS RELATED_BUSINESS_CODE,
t1.SUBMITTED_BY_NAME       AS SUBMITTED_BY_NAME,
t1.APPLICANT_NAME          AS APPLICANT_NAME,
(case 
when T1.APPLICANT_COM_NAME ='' then -999999
WHEN COM1.company_id IS NULL   THEN -999998 
ELSE COM1.company_id END)  AS APPLICANT_COM_ID,
(case 
when T1.APPLICANT_COM_NAME =''  then 'SNULL'
WHEN COM1.company_code     =''  THEN 'NOT EXISTS' 
ELSE COM1.company_code END) AS APPLICANT_COM_CODE,
t1.APPLICANT_COM_NAME      AS APPLICANT_COM_NAME,
t1.APPLICANT_DEPT_NAME     AS APPLICANT_DEPT_NAME,
(case 
when T1.company_name  = ''    then -999999
WHEN COM2.company_id IS NULL  THEN -999998 
ELSE COM2.company_id END)  AS COMPANY_ID,
(case 
when T1.company_name   =''  then 'SNULL'
WHEN COM2.company_code =''  THEN 'NOT EXISTS' 
ELSE COM2.company_code END)    AS COMPANY_CODE,
t1.COMPANY_NAME            AS COMPANY_NAME,
t1.DEPARTMENT_NAME         AS DEPARTMENT_NAME,
t1.INTERNAL_PARTICIPANT    AS INTERNAL_PARTICIPANT,
t1.PARTICIPANT_EMP_ID      AS PARTICIPANT_EMP_ID,
t1.START_DATE              AS START_DATE,
t1.END_DATE                AS END_DATE,
t1.TITLE                   AS TITLE,
t1.CURRENCY_CODE           AS CURRENCY_CODE,
t1.TOTAL_BUDGET            AS TOTAL_BUDGET,
t1.BOOKER                  AS BOOKER,
t1.APP_SUBMITTED_BY_DATE   AS APP_SUBMITTED_BY_DATE,
t1.APP_STATUS_DESC         AS APP_STATUS_DESC,
t1.APP_APPROVAL_DATE       AS APP_APPROVAL_DATE,
t1.LAST_APPROVER           AS LAST_APPROVER,
t1.TRV_ITINERARY_TYPE_DESC AS TRV_ITINERARY_TYPE_DESC,
t1.SUPPLIER_NAME           AS SUPPLIER_NAME,
t1.IS_ROUND_TRIP           AS IS_ROUND_TRIP,
t1.TRV_ITINERARY_CITY      AS TRV_ITINERARY_CITY,
t1.TRV_ITINERARY_DATE      AS TRV_ITINERARY_DATE,
t1.ROOM_QTY                AS ROOM_QTY,
t1.TRV_ITINERARY_REMARK    AS TRV_ITINERARY_REMARK,
t1.ATTRIBUTE1              AS ATTRIBUTE1,
t1.ATTRIBUTE2              AS ATTRIBUTE2,
t1.ATTRIBUTE3              AS ATTRIBUTE3,
t1.ATTRIBUTE4              AS ATTRIBUTE4,
t1.DEL_FLAG,
t1.DW_LAST_UPDATE_DATE
from DWIFIN.DWI_COST_TRVAPP_T  T1
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM1 
ON T1.APPLICANT_COM_NAME = COM1.company_name
 AND COM1.end_date = '4712-12-31'
 AND COM1.del_flag = 'N' 
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM2 
ON T1.company_name = COM2.company_name
 AND COM2.end_date = '4712-12-31'
 AND COM2.del_flag = 'N';
