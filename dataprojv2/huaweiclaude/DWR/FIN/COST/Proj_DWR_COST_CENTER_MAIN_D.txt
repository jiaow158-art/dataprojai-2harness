--插入数据到临时表，对语言编码进行补充
INSERT OVERWRITE TABLE DWRFIN.DWR_COST_CENTER_MAIN_D_TMP
SELECT
   T1.SEQ_ID
  ,T1.COST_CENTER
  ,T2.COST_CENTER_DESC
  ,T1.SRC_END_EFFECT_DATE
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,T1.CURRENCY_CODE
  ,T1.START_EFFECT_DATE
  ,T1.LEADER
  ,T1.COMPANY_CODE
  ,T1.REC_CONS_QUA_FLAG
  ,CASE WHEN COALESCE(T1.LANG_CODE,'') = '' THEN T2.LANG_CODE
        ELSE T1.LANG_CODE
        END AS LANG_CODE
  ,T1.END_EFFECT_DATE
  ,T1.COST_CENTER_MSTRREC_INT_FLAG
  ,T1.BUSINESS_SCOPE
  ,T1.STANDARD_HIERARCHY_AREA
  ,T1.ALLOWED_ALLOCTION_METHOD_FLAG
  ,T1.PROFIT_CENTER
  ,T1.FUNC_SCOPE
  ,T1.COST_CENTER_TYPE
  ,T1.REALINCOME_CHECKIN_FREEZE_FLAG
  ,T1.REAL_SUBCOST_FREEZE_FLAG
  ,T1.PLAN_INCOME_FREEZE_FLAG
  ,T1.PLAN_ETARY_COST_FREEZE_FLAG
  ,T1.PLAN_SUB_COST_FREEZE_FLAG
  ,T1.OPEN_ITEMS_FREEZE_FLAG
  ,T1.REAL_ETARY_COST_FREEZE_FLAG
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,T1.DEL_FLAG
FROM DWIFIN.DWI_COST_CENTER_MAIN_T T1
LEFT JOIN DWIFIN.DWI_COST_CENTER_DESC_T T2
ON T1.COST_CENTER = T2.COST_CENTER
AND T1.GROUP_NUM = T2.GROUP_NUM
AND T1.CONTROL_SCOPE = T2.CONTROL_SCOPE
AND T1.END_EFFECT_DATE = T2.END_EFFECT_DATE
AND T2.LANG_CODE = '1'
;

--关联维表或主数据，拓展维度的信息
INSERT OVERWRITE TABLE DWRFIN.DWR_COST_CENTER_MAIN_D
SELECT
   T1.SEQ_ID
  ,T1.COST_CENTER
  ,T1.COST_CENTER_DESC
  ,T1.SRC_END_EFFECT_DATE
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,T1.CURRENCY_CODE
  ,T1.START_EFFECT_DATE
  ,T1.LEADER
  ,T1.COMPANY_CODE
  ,T2.COMPANY_ID
  ,T2.COMPANY_NAME
  ,T1.REC_CONS_QUA_FLAG
  ,T1.LANG_CODE
  ,T3.CONFIG_NAME AS LANG_NAME
  ,T1.END_EFFECT_DATE
  ,T1.COST_CENTER_MSTRREC_INT_FLAG
  ,T1.BUSINESS_SCOPE
  ,T4.CONFIG_NAME AS BUSINESS_SCOPE_Name
  ,T1.STANDARD_HIERARCHY_AREA
  ,T1.ALLOWED_ALLOCTION_METHOD_FLAG
  ,T1.PROFIT_CENTER
  ,T5.PROFIT_CENTER_NAME
  ,T1.FUNC_SCOPE
  ,T6.CONFIG_NAME AS FUNC_SCOPE_Name
  ,T1.COST_CENTER_TYPE
  ,T7.CONFIG_NAME AS COST_CENTER_TYPE_Name
  ,T1.REALINCOME_CHECKIN_FREEZE_FLAG
  ,T1.REAL_SUBCOST_FREEZE_FLAG
  ,T1.PLAN_INCOME_FREEZE_FLAG
  ,T1.PLAN_ETARY_COST_FREEZE_FLAG
  ,T1.PLAN_SUB_COST_FREEZE_FLAG
  ,T1.OPEN_ITEMS_FREEZE_FLAG
  ,T1.REAL_ETARY_COST_FREEZE_FLAG
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,T1.DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWRFIN.DWR_COST_CENTER_MAIN_D_TMP T1
LEFT JOIN DWIMD.DWI_MD_COMPANY_T T2
ON T1.COMPANY_CODE = T2.COMPANY_CODE
AND T1.GROUP_NUM = T2.GROUP_NUM
AND T2.DEL_FLAG = 'N'
AND T2.END_DATE = '4712-12-31'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T3
ON T1.LANG_CODE = T3.SRC_SYS_CONFIG_CODE
AND T3.LANG_CODE = '1'
AND T3.SOURCE_CODE_D = 'T002T'
AND T3.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T4
ON T1.BUSINESS_SCOPE = T4.SRC_SYS_CONFIG_CODE
AND T1.GROUP_NUM = T4.GROUP_NUM
AND T4.LANG_CODE = '1'
AND T4.SOURCE_CODE_D = 'TGSBT'
AND T4.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_CEPCT_SAP_T T5
ON T1.PROFIT_CENTER = T5.PROFIT_CENTER_CODE
AND T5.LANGUAGE_CODE = '1'
AND T5.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T6
ON T1.FUNC_SCOPE = T6.SRC_SYS_CONFIG_CODE
AND T6.LANG_CODE = '1'
AND T1.GROUP_NUM = T6.GROUP_NUM
AND T6.SOURCE_CODE_D = 'TFKBT'
AND T6.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T7
ON T1.COST_CENTER_TYPE = T7.SRC_SYS_CONFIG_CODE
AND T7.LANG_CODE = '1'
AND T1.GROUP_NUM = T7.GROUP_NUM
AND T7.SOURCE_CODE_D = 'TKT05'
AND T7.DEL_FLAG = 'N'
;
