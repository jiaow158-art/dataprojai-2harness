with  FIXEDASSETS_LAST_EDIT
as (
SELECT 
*
FROM DWRFIN.DWR_TR_FIXEDASSETS_F 
WHERE (from_unixtime(unix_timestamp(creation_date, 'yyyyMMdd'), 'yyyy-MM-dd') = substr(date_sub(FROM_UNIXTIME(UNIX_TIMESTAMP()),1),1,10)  
or   '最后更新时间' = substr(date_sub(FROM_UNIXTIME(UNIX_TIMESTAMP()),1),1,10 ))
)
