with wbs as (
    select t0.PSPID,t0.post1 ,t0.objnr ,t2.anln1 ,t2.anln2,t2.bukrs,t0.ZVKGRP
    from sdi.sdi_proj_1001   t0
    join sdi.sdi_PRPS_1001  t1 on t0.pspnr = t1.PSPHI
    join sdi.sdi_cobrb_1001  t2 on t1.objnr = t2.objnr
    group by t0.PSPID,t0.post1 ,t0.objnr ,t2.anln1 ,t2.anln2,t2.bukrs,t0.ZVKGRP
)
INSERT OVERWRITE TABLE DWRFIN.DWR_TR_FIXEDASSETS_F
SELECT
      aa.BUKRS         company                       -- 公司
     ,aa.ANLN1         asset_code                    -- 资产号
     ,aa.txt50         asset_code_desc               -- 资产号描述
     ,aa.TXA50         asset_code_desc_ext           -- 资产号附加描述
     ,aa.ANLN2         asset_sub_code                -- 资产子编号
     ,aa.ktogr         asset_category                -- 资产类别
     ,t095.ktgrtx      asset_category_desc           -- 资产类别描述
     ,aa.SERNR         serial_num                    -- 序列号
     ,aa.INVNR         inventory_num                 -- 存货号
     ,aa.MENGE         asset_qty                     -- 数量
     ,aa.MEINS         asset_unit                    -- 单位
     ,t006.mseht       asset_unit_desc               -- 单位描述
     ,aa.INVZU         inventory_remark              -- 库存标注
     ,concat(substr(aa.ERDAT, 1, 4), '-', substr(aa.ERDAT, 5, 2), '-', substr(aa.ERDAT, 7, 2))        asset_creation_date           -- 资产创建日期
     ,concat(substr(aa.ZUGDT, 1, 4), '-', substr(aa.ZUGDT, 5, 2), '-', substr(aa.ZUGDT, 7, 2))         first_purchase_date           -- 首次购置日期
     ,concat(substr(aa.AKTIV, 1, 4), '-', substr(aa.AKTIV, 5, 2), '-', substr(aa.AKTIV, 7, 2))        asset_capitalization_date     -- 资产化日期
     ,concat(substr(aa.DEAKT, 1, 4), '-', substr(aa.DEAKT, 5, 2), '-', substr(aa.DEAKT, 7, 2))         asset_inactive_date           -- 不活动日期
     ,concat(substr(ab.AFABG, 1, 4), '-', substr(ab.AFABG, 5, 2), '-', substr(ab.AFABG, 7, 2))        depreciation_start_date       -- 折旧开始日期
     ,ab.NDJAR         useful_life_years             -- 使用年限
     ,az.kostl         cost_center                   -- 成本中心
     ,CT.LTEXT         cost_center_desc              -- 成本中心描述
     ,az.KOSTLV        duty_cost_center              -- 责任成本中心
     ,ct1.LTEXT        duty_cost_center_desc         -- 责任成本中心描述
     ,az.WERKS         factory                       -- 工厂
     ,t001.name2       factory_desc                  -- 工厂描述
     ,az.RAUMN         asset_room                    -- 房间
     ,az.PERNR         personnel                     -- 人员
     ,az.GSBER         business_scope                -- 业务范围
     ,aa.LIFNR         supplier                      -- 供应商
     ,aa.LIEFE         supplier_desc                 -- 供应商描述
     ,aa.HERST         manufacturer                  -- 制造商
     ,ab.AFASL         depreciation_num              -- 折旧码
     ,t090.AFATXT      depreciation_num_desc         -- 折旧码描述
     ,case when ab.afasl = 'Z001' THEN 0.03
           when ab.afasl = 'GL20' THEN 0.02
           when ab.afasl = 'GL20' THEN 0.025
         ELSE 0 END  net_salvage_ratio -- 残值比例
     ,coalesce(wbs.ZVKGRP,cs.sales_grp) sales_grp   -- 销售组
     ,tv.bezei       sales_grp_desc                 -- 销售组描述
     ,wbs.PSPID      WBS_top_level -- WBS 号
     ,wbs.post1      WBS_top_level_desc  -- WBS描述
     , 'N' AS del_flag
     ,current_timestamp() as dw_last_update_date
     ,aa.IVDAT as last_stock_location  --最后库存地点
     ,az.stort as position  --位置
     FROM sdi.sdi_ANLA_1001 aa
     LEFT JOIN sdi.sdi_anlb_1001 ab
               ON aa.MANDT = ab.MANDT AND aa.BUKRS = ab.BUKRS AND aa.ANLN1 = ab.ANLN1 AND
                  aa.ANLN2 = ab.ANLN2 AND ab.AFABE = '01'
     LEFT JOIN sdi.sdi_anlz_1001 az
               ON aa.MANDT = az.mandt AND aa.BUKRS = az.bukrs AND aa.ANLN1 = az.anln1 AND
                  aa.ANLN2 = az.ANLN2 AND az.bdatu = '99991231'
     LEFT JOIN sdi.sdi_CSKT_1001 ct
               ON az.kostl = ct.KOSTL AND ct.KOKRS = 'DP00' AND ct.DATBI = '99991231' AND
                  ct.MANDT = '800' AND ct.SPRAS = '1'
  LEFT JOIN sdi.sdi_CSKT_1001 ct1
               ON az.KOSTLV = ct1.KOSTL AND ct1.KOKRS = 'DP00' AND ct1.DATBI = '99991231' AND
                  ct1.MANDT = '800' AND ct1.SPRAS = '1'
     LEFT JOIN SDI.SDI_ANKT_1001 at ON at.MANDT = '800' AND at.spras = '1' AND aa.ANLKL = at.ANLKL
     left join sdi.SDI_BI0_QCOSTCENTER_1009 cs
               on az.kostl = cs.costcenter and aa.bukrs = cs.comp_code and cs.dateto = '99991231' and
                  cs.del_flag = 'N' and cs.co_area = 'DP00'

     left join SDI.SDI_T095T_1001 t095 on aa.mandt = t095.mandt and aa.ktogr = t095.ktogr and t095.spras='1'
     left join SDI.SDI_T006A_1001 t006 on aa.MEINS = t006.msehi and aa.mandt = t006.mandt and t006.spras='1'
     left join wbs on aa.anln1 = wbs.anln1 and aa.anln2 = wbs.anln2  and aa.bukrs = wbs.bukrs
     left join sdi.sdi_TVGRT_1001 tv
               on coalesce(wbs.ZVKGRP,cs.sales_grp) = tv.vkgrp and tv.mandt = '800' and tv.del_flag = 'N' AND
                  TV.spras = '1'
    left join SDI.SDI_T090NAt_1001 t090 on ab.AFASL = t090.AFASL and aa.mandt = t090.mandt and t090.spras='1' and t090.afapl ='CNDP'
     left join sdi.sdi_t001w_1001 t001   on  az.WERKS = t001.werks and t001.mandt = '800' and t001.spras='1'
              WHERE 1 = 1
                and aa.mandt = '800'
