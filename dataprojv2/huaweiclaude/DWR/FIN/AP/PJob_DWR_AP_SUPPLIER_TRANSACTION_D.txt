INSERT OVERWRITE TABLE DWRFIN.DWR_AP_SUPPLIER_TRANSACTION_D
SELECT 
    T.GJAHR                 AS FISAL_YEAR                       -- 财年
    ,T.RBUKRS               AS COMP_CODE                        -- 公司代码
    ,T.RACCT                AS GENERAL_LEDGER_ACCOUNT           -- 总账科目
    ,T5.TXT20               AS GENERAL_LEDGER_ACCOUNT_DESC      -- 总账科目描述
    ,T.UMSKZ                AS SPECIAL_GENERAL_LEDGER           -- 特别总账
    ,T3.BPKIND              AS SUPPLIER_ATTRIBUTE_CODE          -- 供应商属性代码
    ,T4.TEXT40              AS SUPPLIER_ATTRIBUTE_DESC          -- 供应商属性描述
    ,T.LIFNR                AS SUPPLIER_CODE                    -- 供应商代码
    ,CASE WHEN T3.NAME_ORG1 <> '' THEN T3.NAME_ORG1 ELSE T3.BU_SORT1 END
                            AS SUPPLIER_NAME                    -- 供应商全称
    ,T1.ZTERM               AS PAY_TERM_CODE                    -- 付款条款
    ,T2.TEXT1               AS PAY_TERM_DESC                    -- 付款条款描述
    ,T.BLART                AS VOUCHER_TYPE                     -- 凭证类型
    ,T.BELNR                AS VOUCHER_CODE                     -- 凭证编码
    ,T.DOCLN                AS VOUCHER_LINE_CODE                -- 凭证行编码
    ,T.BLDAT                AS VOUCHER_DATE                     -- 凭证日期
    ,T.BUDAT                AS POTING_DATE                      -- 过账日期
    ,date_add(to_date(from_unixtime(unix_timestamp(NVL(T7.ZFBDT,47121231),'yyyyMMdd'))),CAST(NVL(T7.ZBD1T,0) AS INT)+CAST(NVL(T7.ZBD2T,0) AS INT) ) 
                            AS DUE_DATE                         -- 到期日
    ,T.AUGDT                AS CLEARED_DATE                     -- 清账日期
    ,T.WSL                  AS ORIGIN_CU_END_BALANCE_AMT        -- 原币期末余额
    ,T.RWCUR                AS VOUCHER_CURRENCY                 -- 凭证货币
    ,T.HSL                  AS LOCAL_CU_END_BALANCE_AMT         -- 本币期末余额
    ,T.RHCUR                AS LOCAL_CURRENCY                   -- 本币货币
    ,T6.KURSF               AS ACTUAL_EXCHANGE_RATE             -- 实际汇率
    ,CURRENT_TIMESTAMP()	AS DW_LAST_UPDATE_DATE				-- 数据同步时间
FROM 
    SDI.SDI_ACDOCA_1001 T
    LEFT JOIN SDI.SDI_LFB1_1001 T1 ON T.LIFNR = T1.LIFNR AND T.RBUKRS = T1.BUKRS
    LEFT JOIN SDI.SDI_T052U_1001 T2 ON T1.ZTERM = T2.ZTERM AND T2.SPRAS = '1'
    LEFT JOIN SDI.SDI_BUT000_1001 T3 ON T.LIFNR = T3.PARTNER
    LEFT JOIN SDI.SDI_TB004T_1001 T4 ON T3.BPKIND = T4.BPKIND AND T4.SPRAS = '1'
    LEFT JOIN SDI.SDI_SKAT_1001 T5 ON T.RACCT = T5.SAKNR AND T5.KTOPL = 'CNDP'
    LEFT JOIN SDI.SDI_BKPF_1001 T6 ON T.RBUKRS = T6.BUKRS AND T.GJAHR = T6.GJAHR AND T.BELNR = T6.BELNR
    LEFT JOIN SDI.SDI_BSEG_1001 T7 ON  T.RBUKRS = T7.BUKRS AND T.GJAHR  = T7.GJAHR AND T.BELNR  = T7.BELNR AND T.BUZEI  = T7.BUZEI
WHERE T.KOART='K'
