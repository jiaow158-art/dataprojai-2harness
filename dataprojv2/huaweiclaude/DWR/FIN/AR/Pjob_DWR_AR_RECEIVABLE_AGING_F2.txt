-- 取本币累计余额、回款金额、逾期应收、1-90天逾期应收、91-275天逾期应收、276天-2年逾期应收、2-4年逾期应收、4年以上逾期应收
WITH ACDOCA4 AS(
SELECT
	T.COMP_CODE		    			    		AS COMP_CODE							-- 公司代码
	,T.CUST_CODE		    					AS CUST_CODE							-- 客户编码
	,T.CUST_NAME						    	AS CUST_NAME							-- 客户全称
	,T.SALES_GROUP_MASTER_FILE		        	AS SALES_GROUP_MASTER_FILE				-- 销售组-主档
	,T.WBS							        	AS WBS									-- WBS
	,T.WBS_SALES_GROUP							AS WBS_SALES_GROUP						-- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    		AS SPECIAL_GENERAL_LEDGER				-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    		AS GENERAL_LEDGER_ACCOUNT				-- 总账科目
	,T.CURRENCY		    				    	AS CURRENCY								-- 货币
	,T.SO_GROUP_CODE							AS SO_GROUP_CODE						-- 订单组编码 
	,T.LOCAL_CURRENCY_BALANCE_SUM           	AS LOCAL_CURRENCY_BALANCE_SUM			-- 本币累计余额
	,T.REPAY_AMT                            	AS REPAY_AMT							-- 回款金额
	,greatest(T.OVERDUE_RECEIVABLES,0) 			AS OVERDUE_RECEIVABLES                	-- 逾期应收
	,CASE WHEN T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT >= 0 THEN T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT
		ELSE 0
	END 										AS OVERDUE_RECEIVABLES_1461	        	-- 4年以上逾期应收
	,CASE WHEN T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT >= 0 THEN T.OVERDUE_RECEIVABLES_731_1460
		ELSE 
			greatest(T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT,0)
	END 										AS OVERDUE_RECEIVABLES_731_1460	    	-- 2年-4年逾期应收
	,CASE WHEN T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT >= 0 THEN T.OVERDUE_RECEIVABLES_276_730
		ELSE 
			greatest(T.OVERDUE_RECEIVABLES_276_730 + T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT,0)
	END 										AS OVERDUE_RECEIVABLES_276_730	    	-- 276-2年天逾期应收
	,CASE WHEN T.OVERDUE_RECEIVABLES_276_730 + T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT >= 0 THEN T.OVERDUE_RECEIVABLES_91_275
		ELSE 
			greatest(T.OVERDUE_RECEIVABLES_91_275 + T.OVERDUE_RECEIVABLES_276_730 + T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT,0)
	END 										AS OVERDUE_RECEIVABLES_91_275			-- 91-275天逾期应收
	,CASE WHEN T.OVERDUE_RECEIVABLES_91_275 + T.OVERDUE_RECEIVABLES_276_730 + T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT >= 0 THEN T.OVERDUE_RECEIVABLES_1_90
		ELSE 
			greatest(T.OVERDUE_RECEIVABLES_1_90 + T.OVERDUE_RECEIVABLES_91_275 + T.OVERDUE_RECEIVABLES_276_730 + T.OVERDUE_RECEIVABLES_731_1460 + T.OVERDUE_RECEIVABLES_1461 + T.REPAY_AMT,0)
	END 										AS OVERDUE_RECEIVABLES_1_90   			-- 1-90天逾期应收余额
FROM
-- 取本币累计余额：按公司代码、客户编码、销售组-主档、WBS、特别总账、总账科目维度，过账日期（ACDOCA-BUDAT）小于等于‘查询日期’的本币金额汇总
(SELECT
	T.COMP_CODE		    			    		AS COMP_CODE							-- 公司代码
	,T.CUST_CODE		    					AS CUST_CODE							-- 客户编码
	,T.CUST_NAME						    	AS CUST_NAME							-- 客户全称
	,T.SALES_GROUP_MASTER_FILE		        	AS SALES_GROUP_MASTER_FILE				-- 销售组-主档
	,T.WBS							        	AS WBS									-- WBS
	,T.WBS_SALES_GROUP							AS WBS_SALES_GROUP						-- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    		AS SPECIAL_GENERAL_LEDGER				-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    		AS GENERAL_LEDGER_ACCOUNT				-- 总账科目
	,T.CURRENCY		    				    	AS CURRENCY								-- 货币
	,T.SO_GROUP_CODE							AS SO_GROUP_CODE						-- 订单组编码
	,SUM(T.AMT) 								AS LOCAL_CURRENCY_BALANCE_SUM          	-- 本币累计余额
	,SUM(LEAST(NVL(T.AMT,0),0)) 				AS REPAY_AMT                			-- 回款金额
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 THEN NVL(T.AMT,0) ELSE 0 END)
	                   +SUM(LEAST(NVL(T.AMT,0),0)) 	AS OVERDUE_RECEIVABLES                  -- 逾期应收
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 AND T.DUE_DAYS BETWEEN 0 AND 90 
	               THEN NVL(T.AMT,0) ELSE 0 END)    AS OVERDUE_RECEIVABLES_1_90				-- 1-90天逾期应收余额
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 AND T.DUE_DAYS BETWEEN 91 AND 275 
	               THEN NVL(T.AMT,0) ELSE 0 END)    AS OVERDUE_RECEIVABLES_91_275			-- 91-275天逾期应收
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 AND T.DUE_DAYS BETWEEN 276 AND 730 
	               THEN NVL(T.AMT,0) ELSE 0 END)    AS OVERDUE_RECEIVABLES_276_730			-- 276-2年逾期应收
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 AND T.DUE_DAYS BETWEEN 731 AND 1460 
	               THEN NVL(T.AMT,0) ELSE 0 END)    AS OVERDUE_RECEIVABLES_731_1460			-- 2年-4年逾期应收
	,SUM(CASE WHEN T.AMT>0 AND T.DUE_DATE <T.DATE1 AND T.DUE_DAYS >1460 
	               THEN NVL(T.AMT,0) ELSE 0 END)    AS OVERDUE_RECEIVABLES_1461				-- 4年以上逾期应收
FROM 
	DWRFIN.DWR_AR_CUST_EXTEND_F T
	WHERE T.SPECIAL_GENERAL_LEDGER IN('','U','Q','M') 
GROUP BY T.COMP_CODE ,T.CUST_CODE ,T.CUST_NAME ,T.SALES_GROUP_MASTER_FILE ,T.WBS ,T.WBS_SALES_GROUP ,T.SPECIAL_GENERAL_LEDGER ,T.GENERAL_LEDGER_ACCOUNT ,T.CURRENCY,T.SO_GROUP_CODE
) T
)
INSERT OVERWRITE TABLE DWRFIN.DWR_AR_RECEIVABLE_AGING_F2
SELECT 
    '${PERIOD_ID_D}' 					    	AS QUERY_DATE							-- 查询日期
	,T.COMP_CODE		    					AS COMP_CODE							-- 公司代码
	,T.CUST_CODE		    					AS CUST_CODE							-- 客户编码
	,T.CUST_NAME								AS CUST_NAME							-- 客户全称
	,T.SALES_GROUP_MASTER_FILE					AS SALES_GROUP_MASTER_FILE				-- 销售组-主档
	,T.WBS										AS WBS									-- WBS
	,T.WBS_SALES_GROUP							AS WBS_SALES_GROUP						-- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    		AS SPECIAL_GENERAL_LEDGER				-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    		AS GENERAL_LEDGER_ACCOUNT				-- 总账科目
	,T.CURRENCY		    						AS CURRENCY								-- 货币
	,T.LOCAL_CURRENCY_BALANCE_SUM				AS LOCAL_CURRENCY_BALANCE_SUM			-- 本币累计余额
	,T.REPAY_AMT                            	AS REPAY_AMT                        	-- 回款金额
	,T.OVERDUE_RECEIVABLES 						AS OVERDUE_RECEIVABLES					-- 逾期应收
	,T.OVERDUE_RECEIVABLES_1_90					AS OVERDUE_RECEIVABLES_1_90				-- 1-90天逾期应收
	,T.OVERDUE_RECEIVABLES_91_275				AS OVERDUE_RECEIVABLES_91_275			-- 91-275天逾期应收
	,T.OVERDUE_RECEIVABLES_276_730				AS OVERDUE_RECEIVABLES_276_730			-- 276-2年天逾期应收
	,T.OVERDUE_RECEIVABLES_731_1460				AS OVERDUE_RECEIVABLES_731_1460			-- 2年-4年逾期应收
	,T.OVERDUE_RECEIVABLES_1461					AS OVERDUE_RECEIVABLES_1461				-- 4年以上逾期应收
	,CASE WHEN T.WBS_SALES_GROUP IS NULL OR T.WBS_SALES_GROUP = '' THEN T6.NODE_NAME2
		ELSE T5.NODE_NAME2 
	END                                    		AS BUSINESS_DEPARTMENT			    	-- 事业部
	,CURRENT_TIMESTAMP()						AS DW_LAST_UPDATE_DATE					-- 数据同步时间
	,T.SO_GROUP_CODE							AS SO_GROUP_CODE						-- 订单组编码
	,SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) AS period_id_m
FROM 
	ACDOCA4 T
LEFT JOIN DM.DM_RPT_SALES_GROUP_T T5
ON T.WBS_SALES_GROUP = T5.NODE_NAME9
LEFT JOIN DM.DM_RPT_SALES_GROUP_T T6
ON T.SALES_GROUP_MASTER_FILE = T6.NODE_NAME9
 ;
