INSERT OVERWRITE TABLE DWRFIN.DWR_FP_COMMITMENT_PROJ_F           --承诺项目主数据
SELECT
S.COMMITMENT_ITEM_ID               AS      COMMITMENT_ITEM_ID,
S.FINANCIAL_MANAGEMENT_SCOPE       AS      FIN_MANAGEMENT_SCOPE,
T.CONFIG_NAME                      AS      FIN_MANAGEMENT_SCOPE_NAME,
S.FISCAL_YEAR                      AS      FISCAL_YEAR,
S.GROUP_NUM                        AS      GROUP_NUM,
S.INNER_COMMITMENT_ITEM_NUM        AS      INNER_COMMITMENT_ITEM_NUM,
S.CAN_BE_POST_ID                   AS      CAN_BE_POST_ID,
S.COMMITMENT_ITEM_TYPE             AS      COMMITMENT_ITEM_TYPE,
S.STANDARD_VARIANT                 AS      STANDARD_VARIANT,
S.FINANCIAL_TRANSACTIONS           AS      FINANCIAL_TRANSACTIONS,
S.DEL_FLAG                         AS      DEL_FLAG,
CURRENT_TIMESTAMP()                AS      DW_LAST_UPDATE_DATE
FROM  DWIFIN.DWI_FP_COMMITMENT_PROJ_T S
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R T        --关联配置表
ON T.CONFIG_CODE = S.FINANCIAL_MANAGEMENT_SCOPE 
AND T.GROUP_NUM = S.GROUP_NUM 
AND T.LANG_CODE = '1'
