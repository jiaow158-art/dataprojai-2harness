,base_data as 
(

t1.posting_date --过账日期
,t1.clear_date --清账日期
,t1.voucher_date --凭证日期
,t1.general_ledger_account_desc  `总账科目描述`
,t1.general_ledger_account as `总账科目`
,case 
	when substr(t1.general_ledger_account,1,6) = '002131' then '预收账款'
	when substr(t1.general_ledger_account,1,6) = '001131' then '应收账款'
	when substr(t1.general_ledger_account,1,6) = '001133' then '其他应收账款'
	when substr(t1.general_ledger_account,1,6) = '002181' then '其他应付账款'
	when substr(t1.general_ledger_account,1,6) = '002192' then '预提费用'
	when substr(t1.general_ledger_account,1,6) = '002121' then '应付账款'
	when substr(t1.general_ledger_account,1,6) = '001151' then '预付账款'
	when substr(t1.general_ledger_account,1,6) = '001001' then '现金科目'
	when substr(t1.general_ledger_account,1,6) = '001002' then '银行科目'
	when substr(t1.general_ledger_account,1,6) = '001003' then '银行科目（外币）'
	when substr(t1.general_ledger_account,1,6) = '001009' then '其他货币资金'
	when t1.general_ledger_account = '0016030900' then '预付账款'
 end as `款项分类`
,case 
	when t2.MITKZ = 'A' then '资产'
	when t2.MITKZ = 'D' then '客户'
	when t2.MITKZ = 'K' then '供应商'
	when t2.MITKZ = 'V' then '应收合同科目'
	else '总账科目'
 end as `科目分类描述`
,case when COALESCE(t2.MITKZ,'NAL') in ('NAL','') then 'S' else t2.MITKZ end as `科目分类`
,case 
	when t1.general_ledger_account in ('0011330110','0011330100') then 'Y'--其他应收款-客户往来款、其他应收款-备用金
	when t1.general_ledger_account in ('0021810150','0021810117') then '-Y'--其他应付款-客户往来款、其他应付款-个人
	else 'N'
 end as `冲抵项目`
,case 
	when t1.general_ledger_account in ('0021810150','0011330110') then '0021810150' 
	when t1.general_ledger_account in ('0011330100','0021810117') then '0021810117' 
 end  as `冲抵总账科目`
,case 
	when t2.MITKZ = 'D' then concat(t1.comp_code,D1.NAME_ORG2) --当科目分类为D时：值为：公司代码+客户全称；
	when t2.MITKZ = 'K' then concat(t1.comp_code,
	            case when COALESCE(D11.NAME_ORG1,'') = '' then D11.name1_text else D11.NAME_ORG1 end ) --当科目分类为K时，值为：公司代码+供应商名称；
	when COALESCE(t2.MITKZ,'NAL') in ('NAL','') then concat(t1.comp_code,t1.general_ledger_account_desc) --当科目分类为S时：值为：公司代码+总账科目描述；
 end as `公司代码+全称`
,t1.comp_code as `公司代码`
,case when t2.MITKZ = 'D' then t1.cust_code else null end as `客户编码`
,case when t2.MITKZ = 'D' then D1.NAME_ORG2 else null end as `客户全称`
,t3.kdgrp as `客户组`
,t4.ktext as `客户组描述`
,case when t2.MITKZ = 'K' then t1.supplier_code else null end as `供应商编码`
,case when t2.MITKZ = 'K' then  case when COALESCE(D11.NAME_ORG1,'') = '' then D11.name1_text else D11.NAME_ORG1 end  else null end as `供应商全称`
,t6.BPKIND as `供应商属性编码`
,t7.text40 as `供应商属性描述`
,t5.TAXKD as `税分类`
,t1.voucher_currency as `币种`
,t1.ori_cu_clos_balance_amt as `0.SAP原币期末余额`
,t1.cur_cu_clos_balance_amt as `1.SAP本币期末余额`
,case 
	when datediff(last_day('${PERIOD_ID_D}'),from_unixtime(unix_timestamp(t1.voucher_date, 'yyyyMMdd'),'yyyy-MM-dd')) <= 365 then '0~365日合计'
	when datediff(last_day('${PERIOD_ID_D}'),from_unixtime(unix_timestamp(t1.voucher_date, 'yyyyMMdd'),'yyyy-MM-dd')) <= 730 then '366~730日合计'
	when datediff(last_day('${PERIOD_ID_D}'),from_unixtime(unix_timestamp(t1.voucher_date, 'yyyyMMdd'),'yyyy-MM-dd')) <= 1095 then '731~1095日合计'
	when datediff(last_day('${PERIOD_ID_D}'),from_unixtime(unix_timestamp(t1.voucher_date, 'yyyyMMdd'),'yyyy-MM-dd')) <= 1460 then '1096~1460日合计'
	when datediff(last_day('${PERIOD_ID_D}'),from_unixtime(unix_timestamp(t1.voucher_date, 'yyyyMMdd'),'yyyy-MM-dd')) <= 1825 then '1461~1825日合计'
	else  '1826以上日合计'
 end as seg_flag
from dwrfin.DWR_AR_ACCOUNT_DETAIL_F t1
left join sdi.sdi_skb1_1001 t2 on t1.comp_code = t2.BUKRS and  t1.general_ledger_account = t2.SAKNR and t2.mandt = '800'
left join dwr_dim_cust_partner_d t3 on substr(t1.comp_code,1,2) = t3.sales_org_str2 and t1.cust_code = t3.cust_num and t3.RK = 1
left join sdi.sdi_t151t_1001 t4 on t3.kdgrp = t4.kdgrp and t4.mandt = '800' and t4.spras = '1'
left join sdi.sdi_knvi_1001 t5 on t1.cust_code = t5.kunnr --新增入湖表

LEFT JOIN SDI.SDI_BUT000_1001 D11 ON t1.supplier_code = D11.PARTNER	
LEFT JOIN SDI.SDI_BUT000_1001 D1 ON t1.cust_code = D1.PARTNER
where 1=1
and t1.comp_code <> 'Z000'

and t1.general_ledger_account not in ('0011330200','0021810200')
and t1.posting_date <= last_day( '${PERIOD_ID_D}') --过账日期:自定义参数
and t1.clear_date > last_day( '${PERIOD_ID_D}')--清账日期:自定义参数
)
