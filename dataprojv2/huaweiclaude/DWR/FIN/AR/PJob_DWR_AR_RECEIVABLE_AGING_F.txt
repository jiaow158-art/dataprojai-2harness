-- 创建中间表 DWRFIN.DWR_AR_M1_F
DROP TABLE IF EXISTS `DWRFIN.DWR_AR_M1_F`;
CREATE TABLE IF NOT EXISTS `DWRFIN.DWR_AR_M1_F`(
	`comp_code` string COMMENT '公司代码',
	`cust_code` string COMMENT '客户编码',
	`cust_name` string COMMENT '客户全称',
	`sales_group_master_file` string COMMENT '销售组-主档',
	`wbs` string COMMENT 'WBS',
	`wbs_sales_group` string COMMENT 'WBS-销售组',
	`special_general_ledger` string COMMENT '特别总账',
	`general_ledger_account` string COMMENT '总账科目',
	`currency` string COMMENT '货币',
	`amt` decimal(38,2) COMMENT '金额',
	`due_date` timestamp COMMENT '到期日',
	`due_days` decimal(38,2) COMMENT '到期天数',
	`so_group_code` string COMMENT '订单组编码',
	`VOUCHER_DATE`     String  COMMENT '凭证日期',   
	`natural_due_days`  decimal(38,2)  comment '自然账龄天数'
	)
COMMENT '应收账龄中间表'
;



-- 将数据插入中间表 DWRFIN.DWR_AR_M1_F
WITH AGING AS (
SELECT
	T.COMP_CODE					    AS COMP_CODE					-- 公司代码                	
	,T.CUST_CODE                    AS CUST_CODE                    -- 客户编码
	,T.CUST_NAME                    AS CUST_NAME                    -- 客户全称
	,T.SALES_GROUP_MASTER_FILE      AS SALES_GROUP_MASTER_FILE      -- 销售组-主档
	,T.WBS                          AS WBS                          -- WBS
	,T.WBS_SALES_GROUP              AS WBS_SALES_GROUP              -- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER       AS SPECIAL_GENERAL_LEDGER       -- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT       AS GENERAL_LEDGER_ACCOUNT       -- 总账科目
	,T.CUR_CURRENCY                 AS CURRENCY                     -- 货币
	,T.CUR_CU_CLOS_BALANCE_AMT      AS AMT                          -- 金额
	,T.DUE_DATE                     AS DUE_DATE                     -- 到期日
	,to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd')))
                                    AS DATE1						-- 查询日期
	,T.SO_GROUP_CODE                AS SO_GROUP_CODE                -- 订单组编码
	,T.VOUCHER_DATE					AS VOUCHER_DATE					-- 凭证日期
	,row_number ()over(partition by COMP_CODE,CUST_CODE,WBS,SPECIAL_GENERAL_LEDGER,GENERAL_LEDGER_ACCOUNT,CUR_CURRENCY,SO_GROUP_CODE
	 order by DW_LAST_UPDATE_DATE desc) as rn
FROM 
	DWRFIN.DWR_AR_CUST_DETAIL_F T
WHERE T.POSTING_DATE <= to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd')))
AND T.CLEAR_DATE > to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd')))
AND T.SPECIAL_GENERAL_LEDGER IN('','U','Q','M','&')
)

INSERT OVERWRITE TABLE DWRFIN.DWR_AR_M1_F
SELECT
	T1.COMP_CODE					AS COMP_CODE					-- 公司代码                	
	,T1.CUST_CODE                   AS CUST_CODE                   	-- 客户编码
	,T2.CUST_NAME                   AS CUST_NAME                   	-- 客户全称
	,T2.SALES_GROUP_MASTER_FILE     AS SALES_GROUP_MASTER_FILE     	-- 销售组-主档
	,T1.WBS                         AS WBS                         	-- WBS
	,T2.WBS_SALES_GROUP             AS WBS_SALES_GROUP             	-- WBS-销售组
	,T1.SPECIAL_GENERAL_LEDGER      AS SPECIAL_GENERAL_LEDGER      	-- 特别总账
	,T1.GENERAL_LEDGER_ACCOUNT      AS GENERAL_LEDGER_ACCOUNT      	-- 总账科目
	,T1.CURRENCY                	AS CURRENCY                    	-- 货币
	,T1.AMT     					AS AMT                       	-- 金额
	,T1.DUE_DATE                    AS DUE_DATE                    	-- 到期日
	,CASE WHEN day(T1.DATE1 - T1.DUE_DATE) >= 0 THEN day(T1.DATE1 - T1.DUE_DATE)
		ELSE 0
	END 							AS DUE_DAYS						-- 到期天数
	,T1.SO_GROUP_CODE				AS SO_GROUP_CODE				-- 订单组编码
	,T1.VOUCHER_DATE                AS VOUCHER_DATE                 -- 凭证日期
	,CASE WHEN day(T1.DATE1 - T1.VOUCHER_DATE) >= 0 THEN day(T1.DATE1 - T1.VOUCHER_DATE)
	    ELSE 0 
	 END                            AS natural_due_days             --  自然账龄天数
FROM 
	AGING T1
join AGING t2 on T1.COMP_CODE   = T2.COMP_CODE		    
AND T1.CUST_CODE		        = T2.CUST_CODE
AND NVL(T1.WBS,'SNULL')			= NVL(T2.WBS,'SNULL')
AND T1.SPECIAL_GENERAL_LEDGER   = T2.SPECIAL_GENERAL_LEDGER
AND T1.GENERAL_LEDGER_ACCOUNT   = T2.GENERAL_LEDGER_ACCOUNT
AND NVL(T1.SO_GROUP_CODE,'SNULL')		    = NVL(T2.SO_GROUP_CODE,'SNULL')
and NVL(T1.CURRENCY,'SNULL')		    = NVL(T2.CURRENCY,'SNULL')
and t2.rn=1
;

-- 取本币累计余额、回款金额、逾期应收、1-90天逾期应收、91-275天逾期应收、276天-2年逾期应收、2-4年逾期应收、4年以上逾期应收
WITH ACDOCA4 AS(
SELECT
	T.COMP_CODE		    			    		AS COMP_CODE							-- 公司代码
	,T.CUST_CODE		    					AS CUST_CODE							-- 客户编码
	,T.CUST_NAME						    	AS CUST_NAME							-- 客户全称
	,T.SALES_GROUP_MASTER_FILE		        	AS SALES_GROUP_MASTER_FILE				-- 销售组-主档
	,T.WBS							        	AS WBS									-- WBS
	,T.WBS_SALES_GROUP							AS WBS_SALES_GROUP						-- WBS-销售组
	,T.SPECIAL_GENERAL_LEDGER		    		AS SPECIAL_GENERAL_LEDGER				-- 特别总账
	,T.GENERAL_LEDGER_ACCOUNT		    		AS GENERAL_LEDGER_ACCOUNT				-- 总账科目
	,T.CURRENCY		    				    	AS CURRENCY								-- 货币
	,T.SO_GROUP_CODE							AS SO_GROUP_CODE						-- 订单组编码
	,SUM(T.AMT) 								AS LOCAL_CURRENCY_BALANCE_SUM          	-- 本币累计余额
	,NVL(SUM(case when T.AMT < 0 then T.AMT else 0 end ),0)                       	AS REPAY_AMT                            -- 回款金额
	,NVL(SUM(CASE WHEN T.AMT > 0 AND T.DUE_DATE < to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd'))) THEN T.AMT  ELSE 0 END) ,0)             	AS OVERDUE_RECEIVABLES                  -- 逾期应收
	,NVL(SUM(CASE WHEN  T.DUE_DATE < to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd'))) 
         AND 1 <= T.DUE_DAYS AND T.DUE_DAYS <= 90 THEN T.AMT  ELSE 0 END),0)	       	AS OVERDUE_RECEIVABLES_1_90             -- 1-90天逾期应收余额
	,NVL(SUM(CASE WHEN  T.DUE_DATE < to_date(from_unixtime(unix_timestamp(CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) ELSE '${PERIOD_ID_D}' END,'yyyy-MM-dd'))) 
         AND 91 <= T.DUE_DAYS AND T.DUE_DAYS <= 275 THEN T.AMT  ELSE 0 END),0)	   	AS OVERDUE_RECEIVABLES_91_275           -- 91-275天逾期应收
