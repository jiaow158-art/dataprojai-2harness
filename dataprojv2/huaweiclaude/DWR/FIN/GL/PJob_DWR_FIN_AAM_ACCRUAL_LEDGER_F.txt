INSERT OVERWRITE TABLE dwrfin.dwr_fin_aam_accrual_ledger_f 
SELECT
  a.order_no AS `ledger_number`, -- 台账编号
  CASE 
    WHEN a.accrual_status="Reversed" THEN "已冲销"
    WHEN a.accrual_status="Accrued" THEN "已计提"
    WHEN a.accrual_status="Part_Reversed" THEN "部分冲销"
    ELSE a.accrual_status 
  END AS `accrual_status`, -- 计提状态
  a.corporation_code AS `corporation_code`, -- 公司代码
  a.corporation_name AS `corporation_name`, -- 公司名称
  a.currency_code AS `currency_code`, -- 业务货币代码
  a.currency_name AS `currency_name`, -- 业务货币名称
  a.accrual_period AS `accrual_period`, -- 计提期间
  a.accrual_start_date AS `accrual_start_date`, -- 计提开始时间
  a.accrual_end_date AS `accrual_end_date`, -- 计提结束时间
  a.account_date AS `account_date`, -- 记账日期
  a.expected_reversal_date AS `expected_reversal_date`, -- 预计冲销日期
  a.ledger_account_code AS `ledger_account_code`, -- 总账科目代码
  a.ledger_account_name AS `ledger_account_name`, -- 总账科目名称
  a.cost_center_code AS `cost_center_code`, -- 成本中心代码
  a.cost_center_name AS `cost_center_name`, -- 成本中心名称
  c.business_range AS `advance_reimbursement_number`, -- 费用报销预提来源报销单号
  c.special_code AS `special_ledger_code`, -- 特殊总账标识
  c.lend_ledger_account_name AS `credit_ledger_account_name`, -- 贷方总账科目名称
  c.lend_ledger_account_code AS `credit_ledger_account_code`, -- 贷方总账科目代码
  c.lend_cost_center_code AS `credit_cost_center_code`, -- 贷方成本中心代码
  c.lend_cost_center_name AS `credit_cost_center_name`, -- 贷方成本中心名称
  c.lend_business_range AS `credit_business_range`, -- 贷方业务范围
  c.lend_special_code AS `credit_special_ledger_code`, -- 贷方特殊总账标识
  c.supplier_code AS `credit_supplier_code`, -- 贷方供应商代码
  c.supplier_name AS `credit_supplier_name`, -- 贷方供应商名称
  c.sales_group_code AS `sales_group_code`, -- 销售组代码
  c.sales_group_name AS `sales_group_description`, -- 销售组描述
  c.project_code AS `project_code`, -- 项目代码
  c.project_name AS `project_name`, -- 项目名称
  a.amount AS `amount`, -- 金额
  a.pre_reversal_amount AS `pre_reversal_amount`, -- 预冲销金额
  a.reversal_amount AS `reversal_amount`, -- 已冲销金额
  a.item_note AS `business_explanation`, -- 业务说明
  a.erp_account_no AS `voucher_number`, -- 记账凭证号
  a.accounting_date AS `voucher_accounting_date`, -- 凭证记账日期
  CASE 
    WHEN a.reversal_method="Automatic" THEN "自动冲销"
    WHEN a.reversal_method="Manual" THEN "手工冲销"
    WHEN a.reversal_method="Not_Reversal" THEN "不冲销"  
    ELSE a.reversal_method 
  END AS `reversal_method`, -- 冲销方式
  a.detail_blsign AS `detail_debit_credit_indicator`, -- 明细所属借贷方
  a.reversal_erp_account_no AS `automatic_reversal_voucher_number`, -- 自动冲销凭证号
  a.reversal_accounting_date AS `automatic_reversal_voucher_date`, -- 自动冲销凭证日期
  a.accrual_request_order_no AS `accrual_request_number`, -- 计提申请单编号
  a.accrual_request_item_no AS `accrual_request_item_number`, -- 计提申请行号
  a.agent_account AS `agent_account`, -- 经办人账号
  a.agent_name AS `agent_name`, -- 经办人名称
  a.application_order_no AS `expense_application_advance_number`, -- 费用申请预提对应费用申请单号
  a.creator_account AS `creator_account`, -- 创建人账号
  a.creator_name AS `creator_name`, -- 创建人名称
  a.created_date AS `creation_date`, -- 创建时间
  a.last_editor_account AS `last_editor_account`, -- 最后修改人账号
  a.last_edited_date AS `last_edited_date`, -- 最后修改时间
  a.last_editor_name AS `last_editor_name`, -- 最后修改人名称
  current_timestamp()  dw_last_update_time,
    b.`name` AS `accrual_type`, -- 计提类型
  d.request_note AS `accrual_request_header_note`, -- 计提申请单抬头说明
  d.amount AS `accrual_request_total_amount` -- 计提申请单总金额
FROM sdi.sdi_aam_accrual_ledger_1039 a
LEFT JOIN sdi.sdi_aam_accrual_type_1039 b ON a.accrual_type_id = b.id
LEFT JOIN sdi.sdi_aam_accrual_request_item_1039 c ON c.id = a.accrual_request_item_id
LEFT JOIN sdi.sdi_aam_accrual_request_head_1039 d ON d.id = a.accrual_request_id
