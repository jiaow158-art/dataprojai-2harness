								   when  (T1.material <> ''  and  T1.material is not null ) and (T1.FACTORY_CODE is not null and  T1.FACTORY_CODE <> '')   
								   and	(T1.batch_num <> '' and 	T1.batch_num is not null )
								   then '外协'
		  else null  end  as ziswx,   	-- 自制外协
case when T1.sales_division_1n_code = '11000002' and (SUBSTR(T1.inv_location,-2) = '90' or T1.CLEARANCE_FLAG = '03' ) then '是' --卫浴事业部：库位为特惠品仓(库存后两位为90)或清库存标识为03洁具清仓产品
      when T1.sales_division_1n_code = '11240102' and T1.order_type = 'ZA02' then '是'          --国贸：订单类型属于ZA02的清仓订单
      when T1.sales_division_1n_code = '11000001' and SUBSTR(T1.FACTORY_CODE,3,2) = '80' then '是'   --瓷砖事业部:工厂80结尾
      when T1.sales_division_1n_code = '11000010' then '否'   --生态:是否清仓均为否
      when T1.sales_division_1n_code not in ('11000002','11240102','11000001','11000010') and (SUBSTR(T1.inv_location,-2) = '90' 
      or T1.CLEARANCE_FLAG = '03' or T1.order_type = 'ZA02' or SUBSTR(T1.FACTORY_CODE,3,2) = '80')  then '是' 
      else '否' end as zsfqc,  --是否清仓
    T1.period_id_m  as period_id_m 
FROM 
TMP1 T1  
LEFT JOIN 
DWRDIM.DWR_DIM_CUST_SALES_AREA_D T2  
ON  T1.SALESORG=T2.sales_org_code 
AND T1.CUSTOMER=T2.src_cust_num
AND T1.DISTR_CHAN=T2.distribution_chnl_code 
AND T2.end_date ='4712-12-31' AND T2.del_flag = 'N'
LEFT JOIN SDI.SDI_ZJFCON3_1001 T3
ON T1.ZJFBGLX = T3.ZJFBGLX
LEFT JOIN SDI.SDI_PROJ_1001 T4
ON T1.PSPHI = T4.PSPNR 
LEFT JOIN DWIMD.DWI_MD_MATL_BATCH_INFO_T T5 
ON  T1.MATERIAL = T5.material_num 
AND T1.batch_num = T5.batch_num
left join SDI.SDI_RPT_PROD_CATEGORY_T_1037 D 
on T1.product_level3_code = D.prod_level3_code 
and D.prod_level3_code is not null
