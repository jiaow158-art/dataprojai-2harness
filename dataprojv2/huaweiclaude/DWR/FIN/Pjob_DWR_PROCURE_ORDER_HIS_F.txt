WITH PROCURE_ORDER_HIS AS(
 SELECT 
  substr(T.BUDAT,1,4)||'-'||substr(T.BUDAT,5,2) AS MONTH,--年月
  T.EBELN AS PURCHASE_VOUCHER,--采购凭证
  T.EBELP AS PURCHASE_VOUCHER_ITEM,--采购凭证项目
  NULL AS PURCHASE_NUM,--步骤编号
  NULL AS ELEMENT_NUM,--条件计数器
  T.ZEKKN AS ACCOUNT_NUM,--账户分配的顺序编号
  T.VGABE AS EVENT_TYPE,--事件类型
  T.GJAHR AS MATERIAL_VOUCHER_YEAR,--物料凭证年度
  T.BELNR AS MATERIAL_VOUCHER,--物料凭证
  T.BUZEI AS MATERIAL_VOUCHER_ITEM,--物料凭证行项目
  NULL AS ACCOUNTING_VOUCHER,--会计凭证
  NULL AS ACCOUNTING_VOUCHER_ITEM,--会计凭证项目
  T2.BUKRS AS COMP_CODE,--公司代码
  NULL AS GENERAL_LEDGER_ACCOUNT,--总账科目
  T2.BSTYP AS PURCHASE_VOUCHER_CATE,--采购凭证类别
  T2.BSART AS PURCHASE_VOUCHER_TYPE,--采购凭证类型
  T2.BSAKZ AS CONTROL_FLAG,--控制标识
  T2.LOEKZ AS DEL_FLAG,--删除标识
  T2.STATU AS VOUCHER_STATUS,--凭证状态
  T2.AEDAT AS CREATE_DATE,--创建时间
  T2.LIFNR AS SUPPLIER_CODE,--供应商编码
  T2.EKORG AS PURCHASE_ORG_CODE,--采购组织
  T2.EKGRP AS PURCHASE_GROUP_CODE,--采购组
  T.BEWTP AS RECORD_CODE_HIS,--历史记录编码
  T.BWART AS MOVE_TYPE,--移动类型
  T.BUDAT AS VOUCHER_POST_DATE,--过账日期
  T.WAERS AS VOUCHER_CUR,--凭证货币
  T.DMBTR AS LOCAL_CUR_AMT,--本币金额
  T.SHKZG AS BORROW_OR_LEND,--借贷方
  T.BWTAR AS ESTIMATE_TYPE,--评估类型
  NULL AS ELEMENT_TYPE,--条件类型
  T.ELIKZ AS DELIVERY_FLAG,--交货类型
  T.XBLNR AS REFER_VOUCHER,--参考凭证号
  NULL AS ORDER_NUM,--提单好
  T.LFGJA AS REFER_VOUCHER_YEAR,--参考凭证年度
  T.LFBNR AS REFER_VOUCHER_NUM,--参考凭证的凭证号
  T.LFPOS AS REFER_VOUCHER_ITEM,--参考凭证的项目
  T.GRUND AS NOVE_REASON,--移动原因
  T.CPUDT AS INPUT_DATE,--输入日期
  T.CPUTM AS INPUT_TIME,--输入时间
  T.EVERE AS SHIPMENT_NOTICE,--装运通知
  T.MATNR AS MATERIAL_NUM,--物料编码
  T.WERKS AS FACTORY,--工厂
  T.XWSBR AS COVER_FLAG,--冲销标识
  T.ETENS AS CONFIRM_NUM,--确认序号
  T.KNUMV AS VOUCHER_ELEMENT_NUM,--凭证条件号
  T.MWSKZ AS TAX_CODE,--税码
  T.LSMNG AS UNIT_NUM,--单位数量
  T.LSMEH AS MEASUREMENT_UNIT,--计量单位
  T.EMATN AS MATERIAL_NUMBER,--物料号
  T.HSWAE AS LOCAL_CUR_NUM,--本币号
  T.CHARG AS BATCH,--批次
  T.BLDAT AS VOUCHER_DATE,--凭证日期
  T.XWOFF AS VALUE_COUNT,--开发值计算
  T.XUNPL AS SUBJECT_ALLOCATION,--发票确认中未计划的科目分配
  T.ERNAM AS CREATE_BY,--创建者
  T.SRVPOS AS SERIVE_NUM,--服务号
  T.PACKNO AS PAKG_NUM,--服务的包编码
  T.INTROW AS SERVICE_LINE_NUM,--服务行号
  T.BEKKN AS ALLOCATION_NUM,--采购订单帐户分配编号
  T.LEMIN AS RETURN_FLAG,--返回标识
  T.SAPRL AS SAP_VERSION,--SAP版本
  T.MENGE AS NUM,--数量
  T3.SIZE_UNIT AS MATERIAL_UNIT,--物料单位
  T4.KALNR AS COST_GAUGE_NUM,--成本估算号

  T.WRBTR AS VOUCHER_CUR_AMT,--凭证货币金额

  T.REEWR AS FOREIGN_INVOICE_AMT,--以外币计的发票价值
  T.BAMNG AS NUM2,--数量
  T.LSMNG AS LIST_UNIT_NUM,--交货单的计量单位数量 
  T.LSMEH AS LIST_MEASUREMENT_UNIT,--交货单的计量单位
  T.AREWW AS TMP_SETTLEMENT_VALUE,--在GR/IR暂记待结帐户的结算价值（业务货币） 
  T.AREWB AS SETTLEMENT_VALUE,--PO货币的GR/IR帐户的结算值 
  T.REWRB AS PO_INVOICE_AMT,--PO货币的发票金额 
  T.MENGE_POP AS POP_NUM,--数量(POP) 
  T.BPMNG_POP AS POP_UNIT_NUM,--使用采购订单价格单位的数量 (POP)  
  T.DMBTR_POP AS POP_LOCAL_AMT,--按本位币计的金额(POP)
  T.WRBTR_POP AS POP_VOUCHER_AMT,--凭证货币金额(POP)
  T.AREWR_POP AS POP_CHECK_OUT_VALUE,--用本位币计算的GR/IR帐户结算价值
  T.WESBB AS ORDER_FREEZE_INVENTORY,--以订单为单位的评估收货的锁定库存
  T.BPWEB AS ORDER_FREEZE_INVENTORY_NUM,--以订单价格为单位的评估收货锁定库存的数量
  T.KUDIF AS EXCHANGE_RATE,--汇率
  case when T.BWART in('101','102','122','123','161','162') and T.VGABE=1 THEN 
       CASE WHEN T.SHKZG='H' THEN -1*T.DMBTR
	        WHEN T.SHKZG='S' THEN T.DMBTR
			ELSE 0 END
	   END AS NON_SALE_RECEIPT_AMOUNT,--收货金额（非寄售）
  0 AS SALE_RECEIPT_AMOUNT,--收货金额（寄售） 
  case when T.BWART in('101','102','122','123','161','162') and T.VGABE=1 THEN 
       CASE WHEN T.SHKZG='H' THEN -1*T.DMBTR
	        WHEN T.SHKZG='S' THEN T.DMBTR
			ELSE 0 END
	   END AS NON_SALE_RECEIPT_AMOUNT2,--收货金额（非寄售-不含运费）
  0 as SALE_RECEIPT_AMOUNT2,--收货金额（非寄售-运费）
  case when T.BWART in('101','102','122','123','161','162') and T.VGABE=1 THEN 
       CASE WHEN T.SHKZG='H' THEN -1*T.MENGE
	        WHEN T.SHKZG='S' THEN T.MENGE
			ELSE 0 END
	   END AS NON_SALE_RECEIPT_NUM,--收货数量（非寄售）
   0 AS SALE_RECEIPT_NUM,--收货数量（寄售）
  T3.WIDTH*T3.HEIGHT AS UNIT_AREA,--单位面积
  NULL AS RECEIPT_AREA,--收货面积（含寄售）
  CASE WHEN T.VGABE='2' AND T.SHKZG='H' THEN -1*T.AREWR
       WHEN T.VGABE='2' AND T.SHKZG='S' THEN  T.AREWR
       ELSE 0 END AS NON_SALE_INVOCE_AMT,--发票校验金额（非寄售）
  0 AS SALE_INVOCE_AMT,--发票校验金额（非寄售）
  0 AS SALE_INVOCE_NUM,--发票校验数量（非寄售）
  'EKBE' AS SOURCE_TABLE
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE   
 FROM SDI.SDI_EKBE_1001 T
 LEFT JOIN SDI.SDI_EKKO_1001 T2 ON T.EBELN= T2.EBELN
 LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T T3 ON T.MATNR=T3.MATERIAL_NUM
 LEFT JOIN SDI.SDI_CKMLHD_1001 T4 ON T4.MATNR = T.MATNR AND T4.BWKEY = T.WERKS AND T4.BWTAR= T.BWTAR
                                  AND T4.VBELN='' AND T4.POSNR='000000' AND T4.PSPNR='00000000'
 LEFT JOIN (SELECT KALNR,BDATJ,POPER,
                   MAX(CASE WHEN A.VPRSV IN('V') THEN PVPRS 
					   ELSE NULL END) AS PVPRS,
				   MAX(CASE WHEN A.VPRSV IN('S') THEN STPRS 
					   ELSE NULL END) AS STPRS,
                   MAX(CASE WHEN A.VPRSV IN('V') THEN 2 
				            WHEN A.VPRSV IN('S') THEN 1
					   ELSE 1 END) AS FLAG FROM SDI.SDI_CKMLCR_1001 A 
             WHERE A.VPRSV IN('V','S') GROUP BY KALNR,BDATJ,POPER
 ) T5 ON T5.KALNR=T4.KALNR AND T5.BDATJ = T.GJAHR AND substr(T5.POPER,2,2) = substr(T.BUDAT,5,2)
)
