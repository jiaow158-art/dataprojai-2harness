INSERT OVERWRITE TABLE DWRFIN.DWR_CRE_SW_INTEGRAL_REPORT_F
--总额度
SELECT 
 A.YEAR                    							AS YEAR     					-- 年度
,A.COST_ATTRIBUTION         						AS COST_BELONG   				-- 费用归属
,H.SELECTNAME   									AS CHANNEL       				-- 渠道
,A.INTEGRAL                 						AS INTEGRAL    					-- 积分类型
,A.JAN_Y                    						AS INTEGRAL_TYPE    			-- 一月额度
,A.JAN_T                    						AS JAN_LIMIT_AMT    			-- 一月调整
,A.FEB_Y                    						AS FEB_LIMIT_AMT    			-- 二月额度
,A.FEB_T                    						AS FEB_ADJUST_AMT    			-- 二月调整
,A.MAR_Y                    						AS MAR_LIMIT_AMT    			-- 三月额度
,A.MAR_T                    						AS MAR_ADJUST_AMT    			-- 三月调整
,A.APR_Y                    						AS APR_LIMIT_AMT    			-- 四月额度
,A.APR_T                    						AS APR_ADJUST_AMT    			-- 四月调整
,A.MAY_Y                    						AS MAY_LIMIT_AMT    			-- 五月额度
,A.MAY_T                    						AS MAY_ADJUST_AMT    			-- 五月调整
,A.JUNE_Y                   						AS JUNE_LIMIT_AMT    			-- 六月额度
,A.JUNE_T                   						AS JUNE_ADJUST_AMT    			-- 六月调整
,A.JULY_Y                   						AS JULY_LIMIT_AMT    			-- 七月额度
,A.JULY_T                   						AS JULY_ADJUST_AMT    			-- 七月调整
,A.AUG_Y                    						AS AUG_LIMIT_AMT    			-- 八月额度
,A.AUG_T                    						AS AUG_ADJUST_AMT    			-- 八月调整
,A.SEPT_Y                   						AS SEPT_LIMIT_AMT    			-- 九月额度
,A.SEPT_T                   						AS SEPT_ADJUST_AMT    			-- 九月调整
,A.OCT_Y                    						AS OCT_LIMIT_AMT    			-- 十月额度
,A.OCT_T                    						AS OCT_ADJUST_AMT    			-- 十月调整
,A.NOV_Y                    						AS NOV_LIMIT_AMT    			-- 十一月额度
,A.NOV_T                    						AS NOV_ADJUST_AMT    			-- 十一月调整
,A.DEC_Y                    						AS DEC_LIMIT_AMT    			-- 十二月额度
,A.DEC_T                    						AS DEC_ADJUST_AMT    			-- 十二月调整
,( JAN_Y+JAN_T+FEB_Y+FEB_T+MAR_Y+MAR_T+APR_Y+APR_T+MAY_Y+MAY_T+JUNE_Y+JUNE_T+JULY_Y+JULY_T+AUG_Y+AUG_T+SEPT_Y+SEPT_T+OCT_Y+OCT_T+NOV_Y+NOV_T+DEC_Y+DEC_T )  
													AS CURRENT_YEAR_SUM_AMT         --本年累计额度
,NVL(B.REFUND_AMOUNT,0)   							AS REIMBURSED_LIMIT_AMT         -- 已报销额度
,NVL(C.TAKEUP_AMOUNT,0)   							AS OCCUPIED_LIMIT_AMT           -- 已占用额度
,(NVL(B.REFUND_AMOUNT,0)  + NVL(C.TAKEUP_AMOUNT,0)) AS OCCUPIED_LIMIT_SUM_AMT  		-- 总已占用额度
,( JAN_Y+JAN_T+FEB_Y+FEB_T+MAR_Y+MAR_T+APR_Y+APR_T+MAY_Y+MAY_T+JUNE_Y+JUNE_T+JULY_Y+JULY_T+AUG_Y+AUG_T+SEPT_Y+SEPT_T+OCT_Y+OCT_T+NOV_Y+NOV_T+DEC_Y+DEC_T - NVL(B.REFUND_AMOUNT,0) - NVL(C.TAKEUP_AMOUNT,0)) 
													AS REMAINDE_LIMIT_AMT 			--剩余额度
,CURRENT_TIMESTAMP() 								AS DW_LAST_UPDATE_DATE			-- 数据同步时间
FROM 
(
SELECT
NF   AS YEAR                                                                    --年度
,FYGS AS COST_ATTRIBUTION                                                       --费用归属
,QD   AS CHANNEL                                                                --渠道
,JFLX AS INTEGRAL                                                               --积分类型
,SUM(CASE WHEN (YF = '1' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS JAN_Y       --一月额度
,SUM(CASE WHEN (YF = '1' AND LX = '1' ) THEN JE ELSE 0 END) AS JAN_T            --一月调整
,SUM(CASE WHEN (YF = '2' AND NVL(LX,0) = 0  ) THEN JE ELSE 0 END) AS FEB_Y      --二月额度
,SUM(CASE WHEN (YF = '2' AND LX = '1' ) THEN JE ELSE 0 END) AS FEB_T            --二月调整
,SUM(CASE WHEN (YF = '3' AND NVL(LX,0) = 0  ) THEN JE ELSE 0 END) AS MAR_Y      --三月额度
,SUM(CASE WHEN (YF = '3' AND LX = '1' ) THEN JE ELSE 0 END) AS MAR_T            --三月调整
,SUM(CASE WHEN (YF = '4' AND NVL(LX,0) = 0  ) THEN JE ELSE 0 END) AS APR_Y      --四月额度
,SUM(CASE WHEN (YF = '4' AND LX = '1' ) THEN JE ELSE 0 END) AS APR_T            --四月调整
,SUM(CASE WHEN (YF = '5' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS MAY_Y       --五月额度
,SUM(CASE WHEN (YF = '5' AND LX = '1' ) THEN JE ELSE 0 END) AS MAY_T            --五月调整
,SUM(CASE WHEN (YF = '6' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS JUNE_Y      --六月额度
,SUM(CASE WHEN (YF = '6' AND LX = '1' ) THEN JE ELSE 0 END) AS JUNE_T           --六月调整
,SUM(CASE WHEN (YF = '7' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS JULY_Y      --七月额度
,SUM(CASE WHEN (YF = '7' AND LX = '1' ) THEN JE ELSE 0 END) AS JULY_T           --七月调整
,SUM(CASE WHEN (YF = '8' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS AUG_Y       --八月额度
,SUM(CASE WHEN (YF = '8' AND LX = '1' ) THEN JE ELSE 0 END) AS AUG_T            --八月调整
,SUM(CASE WHEN (YF = '9' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS SEPT_Y      --九月额度
,SUM(CASE WHEN (YF = '9' AND LX = '1' ) THEN JE ELSE 0 END) AS SEPT_T           --九月调整
,SUM(CASE WHEN (YF = '10' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS OCT_Y      --十月额度
,SUM(CASE WHEN (YF = '10' AND LX = '1' ) THEN JE ELSE 0 END) AS OCT_T           --十月调整
,SUM(CASE WHEN (YF = '11' AND NVL(LX,0) = 0 ) THEN JE ELSE 0 END) AS NOV_Y      --十一月额度
,SUM(CASE WHEN (YF = '11' AND LX = '1' ) THEN JE ELSE 0 END) AS NOV_T           --十一月调整
,SUM(CASE WHEN (YF = '12' AND NVL(LX,0) = 0  ) THEN JE ELSE 0 END) AS DEC_Y     --十二月额度
,SUM(CASE WHEN (YF = '12' AND LX = '1' ) THEN JE ELSE 0 END) AS DEC_T           --十二月调整
--,( JAN_Y+JAN_T+FEB_Y+FEB_T+MAR_Y+MAR_T+APR_Y+APR_T+MAY_Y+MAY_T+JUNE_Y+JUNE_T+JULY_Y+JULY_T+AUG_Y+AUG_T+SEPT_Y+SEPT_T+OCT_Y+OCT_T+NOV_Y+NOV_T+DEC_Y+DEC_T ) AS AMOUNT_SUM                                                                                      
FROM SDI.SDI_UF_WYJFZED_1004
GROUP BY NF,FYGS,QD,JFLX
) 
AS A 
--读取渠道
LEFT JOIN SDI.SDI_WORKFLOW_SELECTITEM_1004 H
ON A.CHANNEL = H.SELECTVALUE AND  H.FIELDID = '286941'
--已报销合并
LEFT JOIN
(
--已报销：子流程-非营销中心总经理（按渠道）
SELECT 
YEAR                                       --年度
,FYGS AS COST_ATTRIBUTION                  --费用归属
,QD   AS CHANNEL                           --渠道
,JFLXBH AS INTEGRAL                          --积分类型
,sum(bxje)  AS REFUND_AMOUNT               --报销金额
FROM SDI.SDI_FORMTABLE_MAIN_1120_1004
WHERE NVL(SFHX,0) = 0 AND NVL(ZLCSFGD,0)= 1 AND FYGS <> '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH
UNION ALL
--已报销：子流程-营销中心总经理（不按渠道）
SELECT 
YEAR                                       --年度
,FYGS AS COST_ATTRIBUTION                  --费用归属
,''   AS CHANNEL                           --渠道
,JFLXBH AS INTEGRAL                          --积分类型
,sum(bxje)  AS REFUND_AMOUNT               --报销金额
FROM SDI.SDI_FORMTABLE_MAIN_1120_1004
WHERE NVL(SFHX,0) = 0 AND NVL(ZLCSFGD,0)= 1 and FYGS = '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH
)
AS B 
ON A.YEAR = B.YEAR AND A.COST_ATTRIBUTION = B.COST_ATTRIBUTION AND A.CHANNEL = B.CHANNEL AND A.INTEGRAL = B.INTEGRAL
LEFT JOIN 
--已占用合并
(
SELECT 
year
,COST_ATTRIBUTION
,CHANNEL
,INTEGRAL
,SUM(TAKEUP_AMOUNT) AS  TAKEUP_AMOUNT
FROM
(
--已占用：主流程-非营销中心总经理（按渠道）
SELECT 
YEAR                                        --年度
,FYGS AS COST_ATTRIBUTION                   --费用归属
,QD   AS CHANNEL                            --渠道
,JFLXBH AS INTEGRAL                           --积分类型
,sum(trje)  AS TAKEUP_AMOUNT                --占用金额
FROM SDI.SDI_FORMTABLE_MAIN_1119_1004
WHERE  NVL(LCSFGD,0) = 1 AND FYGS <> '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH

UNION ALL
--已占用：主流程-营销中心总经理（按渠道）
SELECT 
YEAR                                         --年度
,FYGS AS COST_ATTRIBUTION                    --费用归属
,''   AS CHANNEL                             --渠道
,JFLXBH AS INTEGRAL                            --积分类型
,sum(trje)  AS TAKEUP_AMOUNT                  --占用金额
FROM SDI.SDI_FORMTABLE_MAIN_1119_1004
WHERE NVL(LCSFGD,0) = 1 AND FYGS = '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH

UNION ALL
--已占用：子流程-非营销中心总经理（按渠道）
SELECT 
YEAR                                       --年度
,FYGS AS COST_ATTRIBUTION                  --费用归属
,QD   AS CHANNEL                           --渠道
,JFLXBH AS INTEGRAL                          --积分类型
,sum(bxje)  AS TAKEUP_AMOUNT               --占用金额
FROM SDI.SDI_FORMTABLE_MAIN_1120_1004
WHERE NVL(SFHX,0) = 0 AND NVL(ZLCSFGD,0)= 0 AND FYGS <> '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH

UNION ALL 
--已占用：子流程-营销中心总经理（不按渠道）
SELECT 
YEAR                                       --年度
,FYGS AS COST_ATTRIBUTION                  --费用归属
,''   AS CHANNEL                           --渠道
,JFLXBH AS INTEGRAL                         --积分类型
,sum(bxje)  AS TAKEUP_AMOUNT               --占用金额
FROM SDI.SDI_FORMTABLE_MAIN_1120_1004
WHERE NVL(SFHX,0) = 0 AND NVL(ZLCSFGD,0)= 0 and FYGS = '营销中心总经理'
GROUP BY YEAR,FYGS,QD,JFLXBH
) AS C1
GROUP BY YEAR,COST_ATTRIBUTION ,CHANNEL,INTEGRAL 
)
AS C
ON A.YEAR = C.YEAR AND A.COST_ATTRIBUTION = C.COST_ATTRIBUTION AND A.CHANNEL = C.CHANNEL AND A.INTEGRAL = C.INTEGRAL
