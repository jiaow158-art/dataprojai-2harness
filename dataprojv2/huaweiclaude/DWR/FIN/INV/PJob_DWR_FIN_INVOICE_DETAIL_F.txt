--DWRFIN.DWR_FIN_INVOICE_DETAIL_F  发票明细报表

INSERT OVERWRITE TABLE  DWRFIN.DWR_FIN_INVOICE_DETAIL_F 
SELECT
	a.invoice_key AS invoice_key,--发票KEY,
	a.owner_corp_code AS CO_CODE, --所属公司代码,
	CASE
		
		WHEN a.type = "VatInvoice01" THEN
		"增值税专用发票" 
		WHEN a.type = "VatInvoice02" THEN
		"增值税货运发票" 
		WHEN a.type = "VatInvoice03" THEN
		"机动车销售统一发票(纸质)" 
		WHEN a.type = "VatInvoice04" THEN
		"增值税普通发票" 
		WHEN a.type = "VatInvoice08" THEN
		"增值税专用发票(电子)" 
		WHEN a.type = "VatInvoice10" THEN
		"增值税普票发票(电子)" 
		WHEN a.type = "VatInvoice11" THEN
		"增值税普票发票(卷票)" 
		WHEN a.type = "VatInvoice14" THEN
		"增值税通行费发票" 
		WHEN a.type = "VatInvoice15" THEN
		"二手车统一销售发票(纸质)" 
		WHEN a.type = "VatInvoice17" THEN
		"海关进口增值税专用缴款书" 
		WHEN a.type = "Blockchain" THEN
		"区块链发票" 
		WHEN a.type = "VatInvoice24" THEN
		"出口转内销发票" 
		WHEN a.type = "VatInvoice30" THEN
		"海关缴款书出口转内销" 
		WHEN a.type = "VatInvoice80" THEN
		"出口转内销电子专票" 
		WHEN a.type = "VatInvoice31" THEN
		"电子发票（增值税专用发票）" 
		WHEN a.type = "VatInvoice32" THEN
		"电子发票（普通发票）" 
		WHEN a.type = "VatInvoice81" THEN
		"电子发票（增值税专用发票）" 
		WHEN a.type = "VatInvoice82" THEN
		"电子发票（普通发票）" 
		WHEN a.type = "VatInvoice83" THEN
		"机动车销售电子统一发票" 
		WHEN a.type = "VatInvoice84" THEN
		"二手车销售电子统一发票" 
		WHEN a.type = "VatInvoice85" THEN
		"电子发票（增值税专用发票）" 
		WHEN a.type = "VatInvoice86" THEN
		"电子发票（普通发票）" 
		WHEN a.type = "VatInvoiceSalesList" THEN
		"增值税销货清单" 
		WHEN a.type = "GeneralElectronicInvoice" THEN
		"通用电子发票" 
		WHEN a.type = "GeneralInvoice" THEN
		"通用机打发票" 
		WHEN a.type = "QuotaInvoice" THEN
		"定额发票" 
		WHEN a.type = "VehicleTollTicket" THEN
		"过路过桥费发票" 
		WHEN a.type = "TrainTicket" THEN
		"火车票" 
		WHEN a.type = "TaxiTicket" THEN
		"出租车发票" 
		WHEN a.type = "FlightTicket" THEN
		"机票行程单" 
		WHEN a.type = "BusTicket" THEN
		"汽车票" 
		WHEN a.type = "ShipTicket" THEN
		"轮船票" 
		WHEN a.type = "ParkingInvoice" THEN
		"停车费发票" 
		WHEN a.type = "NonTaxInvoice" THEN
		"非税收入票据" 
		WHEN a.type = "Other" THEN
		"其他" ELSE a.type 
	END AS invoice_type, --"发票类型",
	a.invoice_code AS invoice_code,--发票代码,
	a.invoice_no AS invoice_no,--发票号码,
	a.invoice_date AS invoice_date,--开票日期,
	a.check_code AS check_code,--校验码,
	a.tax_period AS tax_period,--购方所属税期,
	CASE
		
		WHEN a.state = "INVALID" THEN
		"作废" 
		WHEN a.state = "NORMAL" THEN
		"正常" 
		WHEN a.state = "OFFSET" THEN
		"已红冲" 
		WHEN a.state = "UNKNOWN" THEN
		"未查验" 
	else a.state end AS invoice_state,-- 发票状态,
	CASE
		
		WHEN a.deduct_state = "UN_DEDUCT" THEN
		"未认证" 
		WHEN a.deduct_state = "DEDUCTED" THEN
		"已认证" 
		WHEN a.deduct_state = "DONT_DEDUCT" THEN
		"不认证" 
		WHEN a.deduct_state = "DEDUCTING" THEN
		"认证中" 
		WHEN a.deduct_state = "CANCELING" THEN
		"撤销认证中" 
	else a.deduct_state end AS deduct_state,--认证状态,
	CASE
		
		WHEN a.accounting_state = "UNACCOUNTED" THEN
		"未记账" 
		WHEN a.accounting_state = "ACCOUNTING" THEN
		"记账中" 
		WHEN a.accounting_state = "ACCOUNTED" THEN
		"已记账" 
	else a.accounting_state end AS accounting_state,--记账状态,
	a.buyer_tax_code AS buyer_tax_code,--买方纳税人识别号,
	a.buyer_name AS buyer_name,--买方名称,
	a.seller_tax_code AS seller_tax_code,--销售方识别号,
	a.seller_name AS seller_name,--销售方名称,
	a.remark AS remark,--发票备注,
	a.tax_scheme AS tax_scheme,--税率,
	a.tax_rate AS tax_rate,--税率（小数点）,
	a.tax_amount AS tax_amount,--税额,
	a.valid_tax AS valid_tax,--有效税额,
	a.total_amount AS total_amount,--金额（不含税）,
	a.total_tax_amount AS total_tax_amount,--价税合计（含税）,
	a.use_amount AS use_amount,--使用金额,
	a.rest_money AS rest_money,--冲销剩余金额,
	a.check_verify AS check_verify,--是否需要查验,
	a.is_vat AS is_vat,--是否是增值税票（区别火车票、机票等）,
	CASE
		
		WHEN a.cross_check_state = "NONE" THEN
		"不涉及" 
		WHEN a.cross_check_state = "UNCHECK" THEN
		"未勾稽" 
		WHEN a.cross_check_state = "CHECKED" THEN
		"已勾稽" 
	else a.cross_check_state end AS cross_check_state,--进项勾稽状态,
	a.created_date AS created_date,--创建时间,
	CASE
		
		WHEN a.sap_type = "NONE" THEN
		"无" 
		WHEN a.sap_type = "COMPANY" THEN
		"指定工厂" 
		WHEN a.sap_type = "STO" THEN
		"STO" 
	else a.sap_type end AS sap_type,--SAP发票类型,
	a.sync_sap AS sync_sap,--是否同步SAP,
	c.biz_code AS biz_code,--使用业务单号,
c.source_ AS data_source,--使用业务来源系统,
c.use_amount AS order_use_amount,--业务单使用金额,
c.use_time AS use_time,--使用时间,
c.remark AS use_remark,--使用业务描述,
c.voucher_no AS voucher_no,--记账凭证号,
c.voucher_date AS voucher_date,--记账凭据日期,
	CASE
		
		WHEN c.use_state = "USED" THEN
		"记账占用" 
		WHEN c.use_state = "PRE_USE" THEN
		"启动流程占用" 
else c.use_state end AS use_state,--使用状态,
	b.line_num AS line_num,--发票明细行号,
	b.tax_code AS tax_code,--税局编码、税收分类编码,
	b.commodity_abbreviation AS commodity_abbreviation,--商品编码简称,
	b.item_name AS item_name,--货物或应税劳务、服务名称,
	b.project_name AS project_name,--项目名称,
	b.spec_model AS spec_model,--规格型号、类型,
	b.unit AS unit,--计量单位或车牌号,
	b.unit_price AS unit_price,--行单价,
	b.qty AS line_qty,--行数量,
	b.rest_amount AS rest_amount,--行剩余数量（红字信息单创建时回写）,
	b.tax_rate AS line_tax_rate,--行税率,
	b.amount AS line_amount,--行金额,
	b.rest_money AS line_rest_money,--行剩余金额（由红字信息表回写）,
	b.tax AS line_tax,--行税额,
	b.rest_tax AS line_rest_tax,--行剩余税额（由红字信息表回写）
	current_timestamp() as dw_last_update_date
FROM
SDI.SDI_invoice_info_1039 a
	LEFT JOIN SDI.SDI_invoice_item_detail_1039 b ON a.id = b.invoice_id
	left join SDI.SDI_invoice_use_record_1039 c on a.id=c.invoice_id
	where c.source_="BDM-CorporateSettlement"
