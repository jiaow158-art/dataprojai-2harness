insert overwrite into dm.dm_sfa_cust_mes_main_t ---20240417初始化
with tmp as (  --存在crm创建的意向无客编客户，后续需要根据名称去把sap对应的编码覆盖过去
select  row_number() over(partition by cust_name) as rn ,cust_name,data_id from api_log.api_to_dws_cus_id where cust_code is null 
)
, tmp_table as ( 
--新建
select 
	 a.cust_num
	,a.cust_name
	,f.lastname as ctr_object_person_name --申请人姓名
	,a.address_name  
	,a.legal_person_name --企业法人  
	,EXTRACT(EPOCH FROM (a.rec_crt_date)) * 1000 as rec_crt_date
	,substr(a.rec_crt_date,1,10) as create_date
	,a.phone_1   
	,coalesce(b.data_id,c.data_id) as data_id  --客户唯一性id
	,current_timestamp as dw_last_update_date
	,EXTRACT(EPOCH FROM (a.last_chg_time)) * 1000 as t_update_date
	,substr(a.last_chg_time,1,10) as update_date
	,1 as is_collaborate --是否合作
	,case when e1.name is null then '系统' else e1.name end as owner  -- 负责人
	,case when e2.user_id is null then 'FSUID_0038FBDA010A42F8B1A43C1ADC04546B' else e2.user_id end as owner_id--负责人id
	,f2.user_id as ctr_object_person_id --申请人id
	,CASE WHEN ATTRI_CODE6 is not null THEN 1 ELSE 0 END  as is_share_cust --是否共享仓
	,b.owner as sfa_owner
	,e1.employee_id as sap_owner
	,b.cust_name as sfa_cust_name--sfa
from DWRDIM.DWR_DIM_CUST_GENERAL_D a
left join api_log.api_to_dws_cus_id  b
on a.cust_num = b.cust_code and b.cust_code is not null 
left join (select * from tmp where rn = 1)c
on a.cust_name = c.cust_name 
left join (select row_number() over(partition by cust_num order by distribution_chnl_code,sales_org_code asc,partner_counter desc) as RN,* 
from DWRDIM.DWR_DIM_CUST_PARTNER_D  --取11个大区零售业务员
where end_date = '4712-12-31' and del_flag = 'N'  and partner_func_code = 'VE' AND cust_group_7 is null 
and sale_region_code in ('S00001','S00003','S00004','S00005','S00006','S00008','S00009','S00010','S00011','S00014','B00018') ) e 
on a.cust_num = e.cust_num and e.rn = 1
left join (select row_number() over(partition by cust_num order by sales_org_code asc,partner_counter desc) as RN,* 
from DWRDIM.DWR_DIM_CUST_PARTNER_D  --优先读取工程业务员
where end_date = '4712-12-31' and del_flag = 'N'  and partner_func_code = 'VE'  AND cust_group_7 is null 
and distribution_chnl_code = '06' and sales_org_code not like '3%' and sales_org_code not in ('7G00','7B50','7B30','7B20','7910','7100','2100','1030','1020')) e0 
on a.cust_num = e0.cust_num and e0.rn = 1
left join api_log.dws_sfa_org_employee_info_t e1
on coalesce(e0.src_emp_num,e.src_emp_num) = e1.employee_number
left join api_log.api_to_dws_people_information e2
on e1.employee_id = e2."_id"
left join (select lpad(x.khbm,10,0) as khbm,y.lastname,y.id,y.loginid,row_number() over(partition by lpad(x.khbm,10,0) order by modedatacreatedate||modedatacreatetime desc) as rn 
from dwipi.dwi_wf_uf_htgl_t x
left join dwrhr.dwr_hrmresource_f y
on x.sqr = y.id where substr(x.htyxrq,1,4) >= '2023' ) f
on a.cust_num = f.khbm and f.rn = 1
left join api_log.dws_sfa_org_employee_info_t f1
on f.loginid = f1.employee_number
left join api_log.api_to_dws_people_information f2
on f1.employee_id = f2."_id" 
where a.end_date = '4712-12-31' and a.del_flag = 'N' and a.cust_num <> 'NOT EXISTS' 
and a.cust_num in (select cust_num from dwrdim.dwr_dim_cust_sales_area_d where end_date = '4712-12-31' 
and del_flag = 'N' and CUST_GROUP_7 is null and substr(sales_org_code,3,2) <> '10' group by cust_num )
and coalesce(b.data_id,c.data_id) is null --新建
and (select count(1) from api_log.api_sfa_data_dict_id) > 50 --数据字典表不为空
and (select count(1) from api_log.api_to_dws_people_information) > 200 --人员表不为空
union all
--修改
select 
	 a.cust_num
	,a.cust_name
	,f.lastname as ctr_object_person_name --申请人姓名
	,a.address_name  
	,a.legal_person_name --企业法人  
	,EXTRACT(EPOCH FROM (a.rec_crt_date)) * 1000 as rec_crt_date
	,substr(a.rec_crt_date,1,10) as create_date
	,a.phone_1   
	,coalesce(b.data_id,c.data_id) as data_id  --客户唯一性id
	,current_timestamp as dw_last_update_date
	,EXTRACT(EPOCH FROM (a.last_chg_time)) * 1000 as t_update_date
	,substr(a.last_chg_time,1,10) as update_date
	,case when d.cust_num is not null then 1 else 0 end as is_collaborate --是否合作
	,case when e1.name is null then '系统' else e1.name end as owner  -- 负责人
	,case when e2.user_id is null then 'FSUID_0038FBDA010A42F8B1A43C1ADC04546B' else e2.user_id end as owner_id--负责人id
	,f2.user_id as ctr_object_person_id --申请人id
	,CASE WHEN ATTRI_CODE6 is not null THEN 1 ELSE 0 END  as is_share_cust --是否共享仓
	,b.owner as sfa_owner
	,e1.employee_id as sap_owner
	,b.cust_name as sfa_cust_name--sfa
from DWRDIM.DWR_DIM_CUST_GENERAL_D a
left join api_log.api_to_dws_cus_id  b
on a.cust_num = b.cust_code
left join (select * from tmp where rn = 1)c
on a.cust_name = c.cust_name 
left join (select cust_num from dwrdim.dwr_dim_cust_sales_area_d where end_date = '4712-12-31' 
and del_flag = 'N' and CUST_GROUP_7 is null group by cust_num ) d
on a.cust_num = d.cust_num 
left join (select row_number() over(partition by cust_num order by distribution_chnl_code,sales_org_code asc,partner_counter desc ) as RN,* 
from DWRDIM.DWR_DIM_CUST_PARTNER_D  --优先取11个大区零售业务员
where end_date = '4712-12-31' and del_flag = 'N'  and partner_func_code = 'VE' 
and sale_region_code in ('S00001','S00003','S00004','S00005','S00006','S00008','S00009','S00010','S00011','S00014','B00018')) e 
on a.cust_num = e.cust_num and e.rn = 1
left join (select row_number() over(partition by cust_num order by sales_org_code asc,partner_counter desc) as RN,* 
from DWRDIM.DWR_DIM_CUST_PARTNER_D  --优先读取工程业务员
where end_date = '4712-12-31' and del_flag = 'N'  and partner_func_code = 'VE' 
and distribution_chnl_code = '06' and sales_org_code not like '3%' and sales_org_code not in ('7G00','7B50','7B30','7B20','7910','7100','2100','1030','1020')) e0 
on a.cust_num = e0.cust_num and e0.rn = 1
left join api_log.dws_sfa_org_employee_info_t e1
on coalesce(e0.src_emp_num,e.src_emp_num) = e1.employee_number
left join api_log.api_to_dws_people_information e2
on e1.employee_id = e2."_id"
left join (select lpad(x.khbm,10,0) as khbm,y.lastname,y.id,y.loginid,row_number() over(partition by lpad(x.khbm,10,0) order by modedatacreatedate||modedatacreatetime desc) as rn 
from dwipi.dwi_wf_uf_htgl_t x
left join dwrhr.dwr_hrmresource_f y
on x.sqr = y.id where substr(x.htyxrq,1,4) >= '2023' ) f
on a.cust_num = f.khbm and f.rn = 1
left join api_log.dws_sfa_org_employee_info_t f1
on f.loginid = f1.employee_number
left join api_log.api_to_dws_people_information f2
on f1.employee_id = f2."_id" 
where a.end_date = '4712-12-31' and a.del_flag = 'N' and a.cust_num <> 'NOT EXISTS' 
--and a.cust_num in (select cust_num from dwrdim.dwr_dim_cust_sales_area_d where end_date = '4712-12-31' 
--and del_flag = 'N' and sales_org_code <> '1020' 
--and sale_region_code in ('S00001','S00003','S00004','S00005','S00006','S00008','S00009','S00010','S00011','S00014','B00018') group by cust_num )
and coalesce(b.data_id,c.data_id) is not null --修改
and (select count(1) from api_log.api_sfa_data_dict_id) > 50 --数据字典表不为空
and (select count(1) from api_log.api_to_dws_people_information) > 200 --人员表不为空
)
,update_table as (
 select distinct
 	 t1.cust_num
 	,t1.data_id
 	,t1.owner_id
 	,t1.ctr_object_person_id
-- 	,concat(t1.cust_name,t1.ctr_object_person_name,t1.address_name,t1.legal_person_name,t1.rec_crt_date,t1.phone_1,t1.is_collaborate,t1.owner,t1.is_share_cust) as t1_concat
-- 	,concat(t2.cust_name,t2.ctr_object_person_name,t2.address_name,t2.legal_person_name,t2.rec_crt_date,t2.phone_1,t2.is_collaborate,t2.owner,t2.is_share_cust) as t2_concat
	 ,case when t1.sfa_owner <> t1.sap_owner or t1.cust_name <> t1.sfa_cust_name then 'Y'  --sfa跟sap对比负责人差异
		when concat(t1.cust_name,coalesce(t1.ctr_object_person_name,'#'),coalesce(t1.address_name,'#'),coalesce(t1.legal_person_name,'#'),coalesce(t1.phone_1,'#'),t1.is_collaborate,coalesce(t1.owner,'#'),t1.is_share_cust,t1.ctr_object_person_id,t1.owner) 
		   = concat(t2.cust_name,coalesce(t2.ctr_object_person_name,'#'),coalesce(t2.address_name,'#'),coalesce(t2.legal_person_name,'#'),coalesce(t2.phone_1,'#'),t2.is_collaborate,coalesce(t2.owner,'#'),t2.is_share_cust,t2.ctr_object_person_id,t2.owner) 
		then 'N' else 'Y' end as update_flag
from tmp_table t1
left join dm.dm_sfa_cust_mes_main_t t2 on t1.cust_num = t2.cust_num
)
,result_table as (
select 
	t1.cust_num
	,t1.cust_name
	,t1.ctr_object_person_name
	,t1.address_name
	,t1.legal_person_name
	,cast(t1.rec_crt_date as text) as rec_crt_date
	,cast(t1.create_date as text) as create_date
	,t1.phone_1
	,t2.data_id
	,current_timestamp as dw_last_update_date
	,cast(t1.t_update_date as text) as t_update_date
	,cast(t1.update_date as text) as update_date
	,cast(t1.is_collaborate as text) as is_collaborate
	,t1.owner
	,t2.owner_id
	,t2.ctr_object_person_id
	,cast(t1.is_share_cust as text) as is_share_cust
	,t2.update_flag
from dm.dm_sfa_cust_mes_main_t t1
left join update_table t2 on t1.cust_num = t2.cust_num
where t2.update_flag = 'N'
union all 
select 
	t1.cust_num
	,t1.cust_name
	,t1.ctr_object_person_name
	,t1.address_name
	,t1.legal_person_name
	,cast(t1.rec_crt_date as text) as rec_crt_date
	,cast(t1.create_date  as text) as create_date
	,t1.phone_1
	,t1.data_id
	,t1.dw_last_update_date
	,cast(EXTRACT(EPOCH FROM (current_timestamp)) * 1000 as text) as t_update_date
	,cast(current_timestamp as text) as update_date
	,cast(t1.is_collaborate as text) as is_collaborate
	,t1.owner
	,t1.owner_id
	,t1.ctr_object_person_id
	,cast(t1.is_share_cust as text) as is_share_cust
	,t2.update_flag
from tmp_table t1
left join update_table t2 on t1.cust_num = t2.cust_num
where t2.update_flag = 'Y'
)
