with get_parent_cust_num as (
select distinct 
-- SALES_GROUP_CODE as sale_grp,--线组
case 
	when COALESCE(PARENT_ACCNT_NUMBER,'NAL') not in ('','NAL') then lpad(PARENT_ACCNT_NUMBER,10,'0')
	else lpad(ACCOUNT_NUMBER,10,'0')
end as parent_CUST_NUM, --母

case 
	when SALES_CHANNEL_CODE = '06' then 'GD03'
	when SALES_CHANNEL_CODE = '08' then 'GD02'
	else 'GD01'
end as zh_channel_code4,--整合渠道
SALES_CHANNEL_CODE as distribution_channel_code,--渠道
SALES_ORGANIZATION_CODE as SALES_ORG_CODE --组织
From SDI.sdi_zmaster_cus_1001
where PARVW = 'AG'
)
,get_curyear_nopick as (
select
    t1.sales_grp,
	t1.KUNNR_OLD as cust_num,
	case 
	    when t1.VTWEG_OLD in ('06') then 'GD03'
	    when t1.VTWEG_OLD in ('08') then 'GD02'
	    else 'GD01'
	end as integrate_channel,
	t3.parent_CUST_NUM,
    sum(XS_NETWR) as nopick_amount_curyear
from dwrotd.dwr_accumulate_overdue_nopick_f t1
left join dm.dm_rpt_sales_group_t t2 on t1.sales_grp = t2.node_name9
left join get_parent_cust_num t3 on t1.VKORG_OLD = t3.SALES_ORG_CODE and t1.VTWEG_OLD = t3.SALES_ORG_CODE and t1.kunnr_old = t3.CUST_NUM
where t1.CREATEDATE <= replace('${PERIOD_ID_D}','-','')
and t1.CREATEDATE >= concat(substring('${PERIOD_ID_D}',1,4),'0101')
and t1.sale_region_name like '%运营中心'
group by
t1.sales_grp,
t1.KUNNR_OLD,
case 
    when t1.VTWEG_OLD in ('06') then 'GD03'
    when t1.VTWEG_OLD in ('08') then 'GD02'
    else 'GD01'
end,
t3.parent_CUST_NUM
)

--新增销售订单留备货A：逾期、未逾期 + 新增销售订单留备货B：逾期、未逾期

	-->当前时间
	sum(case when  distribution_channel_code in ('06') and acc_time_date <= SUBSTR(date_format(date_sub('${PERIOD_ID_D}',0),'yyyyMMdd'),1,8) then sal_pre_res_area else 0 end ) as lbh_before_gc_area,

	-->当前时间 and <lastday
	sum(case when  distribution_channel_code in ('06') 
		and acc_time_date > SUBSTR(date_format(date_sub('${PERIOD_ID_D}',0),'yyyyMMdd'),1,8) 
		and acc_time_date <= cur_month_last_day
	then sal_pre_res_area else 0 end ) as lbh_mid_gc_area,
	sum(case when  distribution_channel_code not in ('06') 
		and acc_time_date > SUBSTR(date_format(date_sub('${PERIOD_ID_D}',0),'yyyyMMdd'),1,8) 
		and acc_time_date <= cur_month_last_day
	then sal_pre_res_area else 0 end ) as lbh_mid_ls_area,
	
	-->=当月最后一天
	sum(case when  distribution_channel_code in ('06') 
		and acc_time_date > cur_month_last_day
	then sal_pre_res_area else 0 end ) as lbh_after_gc_area,
