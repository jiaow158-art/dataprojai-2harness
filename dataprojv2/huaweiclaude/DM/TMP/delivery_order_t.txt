WITH LIPS_TMP AS (
SELECT 
 T1.POSNR                -- 行
,T1.VBELN                 -- 交货单
,T1.PSTYV                 -- 交货项目类别
,T2.VTEXT                --描述
,T1.MATNR                 -- 物料编码
,T1.LGORT                 -- 库存地点
,T1.LFIMG                 -- 发货数量
,T1.VGBEL                -- 来源
,T1.VGPOS                 -- 来源项目号
FROM sdi.sdi_lips_1001 T1 LEFT JOIN (SELECT PSTYV,VTEXT FROM sdi.sdi_tvapt_1001 WHERE SPRAS = '1' AND MANDT = '800' GROUP BY PSTYV,VTEXT) T2 ON T1.PSTYV =T2.PSTYV
)

,LIKP_TMP AS (
SELECT 
 T1.VBELN             -- 单号
,T1.LFART                 -- 交货类型
,T2.VTEXT                --描述
,T1.WADAT_IST         -- 交货日期
,T1.KUNAG                  --售达方
FROM sdi.sdi_LIKP_1001 T1 
LEFT JOIN  (SELECT LFART,VTEXT from sdi.sdi_TVLKT_1001 where SPRAS = '1'AND MANDT = '800'  group by LFART,VTEXT) T2 ON T1.LFART =T2.LFART
)
