delete from dm.dm_huilianyi_application_form_history;
insert into  dm.dm_huilianyi_application_form_history
with tem as  
(
select 
	SPLIT_PART(SPLIT_PART(docpdfattachment,':',3),E'\'',2) as sqd_pdf_id -- 附件ID
	,SPLIT_PART(SPLIT_PART(docpdfattachment,':',5),E'\'',2) as sqd_pdf_name -- 附件名字
	,businesscode -- 申请单号
	
	,replace(replace(approvaldate,'T',' '),'Z','') as  approvaldate-- 提交日期
	,companycode -- 公司代码
	,SPLIT_PART(SPLIT_PART(createduser,'fullName',2),E'\'',3) as sqr_name-- 申请人姓名
	,SPLIT_PART(SPLIT_PART(createduser,'employeeID',2),E'\'',3) sqr_num-- 申请人工号
	,SPLIT_PART(SPLIT_PART(createduser,'path',2),E'\'',3) as sqr_bm -- 申请部门
	,substring(SPLIT_PART(SPLIT_PART(custformvalues,'开始日期',2),':',4),3,10) as start_date -- 开始日期
	,substring(SPLIT_PART(SPLIT_PART(custformvalues,'结束日期',2),':',4),3,10) as end_date -- 结束日期
	,formname	-- 申请单类型
	,loanableamount	-- 申请金额
	,regexp_replace(  
        replace(replace(replace(simpleapplicationparticipants,E'\'',''),', department: None',''),', employeeID',''),   
        'userOID: [a-f0-9-]+[,] fullName[:]',   
        '' ,
        'g'
    ) as part_people -- 参与人
	,title -- 出差事由
from upload.upload_huilianyi_application_form_history
)

select 
    distinct
	sqd_pdf_id
	,sqd_pdf_name
	,businesscode
	,approvaldate
	,companycode
	,sqr_name
	,sqr_num
	,sqr_bm
	,start_date
	,end_date
	,formname
	,loanableamount 
	,part_people
	,title
	,current_timestamp() as dw_last_update_date
from tem
