		   when dayofweek(c.operate_date) = 7 and coalesce(c.operate_time,'08:00:00') > '11:30:00'

     ,CASE WHEN c.operation_type = '0' AND C.examine_flag = '-2' THEN '已查看'

           WHEN c.operation_type = '1' THEN '1:转发'

		   WHEN c.operation_type = '8' THEN '8:抄送(不需提交)'

     ,CASE WHEN c.operation_type = '0' THEN '0:未操作'
           WHEN c.before_change_operation_type = '1' THEN '1:转发'
           WHEN c.before_change_operation_type = '2' THEN '2:已操作'
           WHEN c.before_change_operation_type = '5' THEN '5:超时'
           WHEN c.before_change_operation_type = '9' THEN '9:抄送(需提交)'
		   WHEN c.before_change_operation_type = '4' THEN '4:归档'
		   WHEN c.before_change_operation_type = '8' THEN '8:抄送(不需提交)'
		   WHEN c.before_change_operation_type = '9' THEN '9:抄送(需提交)'
		   WHEN c.before_change_operation_type = 'a' THEN 'a:意见征询'
		   WHEN c.before_change_operation_type = 'b' THEN 'b:回复'
		   WHEN c.before_change_operation_type = 'h' THEN 'h:转办'
		   WHEN c.before_change_operation_type = 'j' THEN 'j:转办提交'
		   WHEN c.before_change_operation_type = '11' THEN '11:传阅'

     ,A.last_operate_type
	 ,A.hs

	 ,A.receive_time_hs

	 ,A.JS_COUNT --接收日期在节假日数
	 ,A.CZ_COUNT --操作日期在节假日数		   
     ,A.dept_name
     ,A.request_explain 
     ,A.dept_id
     ,A.belonging_branch
     ,A.node_type
     ,A.examine_flag
	 ,A.l1lc
	 ,A.current_node_type
	 ,A.EMP_POST_NAME

	 ,CASE WHEN JS_COUNT > 0 OR dayofweek(A.receive_date) = 1 --当接收日期在节假日或者周日，接受日不计算
	       THEN unix_timestamp(A.receive_date || ' ' || '23:59:59') - unix_timestamp(A.receive_date || ' ' || A.receive_time_hs)
	       ELSE 0
            END as HS_JS --接收日期耗时
     ,CASE WHEN CZ_COUNT > 0 OR dayofweek(A.operate_date) = 1 --当操作日期在节假日或者周日，操作日不计算
	       THEN unix_timestamp(A.operate_date || ' ' || operate_time_hs) - unix_timestamp(A.operate_date || ' ' || '00:00:00')
	       ELSE 0
            END as CZ_JS --操作日期耗时
	 ,unix_timestamp(A.operate_date || ' ' || A.operate_time_hs) - unix_timestamp(A.receive_date || ' ' || A.receive_time_hs)
      -
	  CASE WHEN JS_COUNT > 0 OR dayofweek(A.receive_date) = 1 OR (dayofweek(A.receive_date) = 7 and A.receive_time_hs > '11:30:00')
	       THEN unix_timestamp(A.receive_date || ' ' || '23:59:59') - unix_timestamp(A.receive_date || ' ' || A.receive_time_hs)

            END
	  -
	  CASE WHEN CZ_COUNT > 0 OR dayofweek(A.operate_date) = 1
	       THEN unix_timestamp(A.operate_date || ' ' || operate_time_hs) - unix_timestamp(A.operate_date || ' ' || '00:00:00')
	       ELSE 0

	 ,CASE WHEN A.HS_GZ1 < 0 THEN 0

     ,case when datediff(cast(A.operate_date as date),cast(A.receive_date as date))/7 < 1
           then 
                case when (dayofweek(date_add(cast(A.receive_date as date),1)) in(1) and pmod(datediff(cast(A.operate_date as date),cast(A.receive_date as date)),7) >= 2) --开始日期
				       or (dayofweek(date_add(cast(A.receive_date as date),2)) in(1) and pmod(datediff(cast(A.operate_date as date),cast(A.receive_date as date)),7) >= 3)
				       or (dayofweek(date_add(cast(A.receive_date as date),3)) in(1) and pmod(datediff(cast(A.operate_date as date),cast(A.receive_date as date)),7) >= 4)

            else 0
             end as ZR_COUNT --计算从接收日到操作日共经历多少个周日

           then 
                case when (dayofweek(date_add(cast(A.receive_date as date),1)) in(7) and pmod(datediff(cast(A.operate_date as date),cast(A.receive_date as date)),7) >= 2) --开始日期
