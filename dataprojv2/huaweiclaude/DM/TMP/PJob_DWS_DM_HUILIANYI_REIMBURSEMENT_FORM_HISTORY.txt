delete from dm.dm_huilianyi_reimbursement_form_history;
insert into  dm.dm_huilianyi_reimbursement_form_history
with pxd_billingCode as -- 发票代码列表
(
SELECT 
	id
	,string_agg(distinct replace(SPLIT_PART(billingCode_ele,':',2),E'\'',''),';') as billingCode_list
FROM (
  SELECT 
	  id
	  ,unnest(string_to_array(receiptvolist, ',')) AS billingCode_ele
  FROM upload.upload_huilianyi_reimbursement_form_history
) AS subq
WHERE billingCode_ele LIKE '%billingCode%'
group by id
)

,pxd_billingNo as -- 发票号码
(
SELECT 
	id
	,string_agg(distinct replace(SPLIT_PART(billingNo_ele,':',2),E'\'',''),';') as billingNo_list
FROM (
  SELECT 
	  id
	  ,unnest(string_to_array(receiptvolist, ',')) AS billingNo_ele
  FROM upload.upload_huilianyi_reimbursement_form_history
) AS subq
WHERE billingNo_ele LIKE '%billingNo%'
group by id
)

,lc_sp0_date as -- 创建时间
(
select 
	id
	,replace(replace(string_agg(distinct SPLIT_PART(SPLIT_PART(sp_date_ele,'lastModifiedDate',2),E'\'',3),';'),'T',' '),'Z','')  as sp0_date
	,string_agg(distinct SPLIT_PART(SPLIT_PART(sp_date_ele,'operatorName',2),E'\'',3),';') as sp0_name
from 
(
select 
	id
	,approvalhistoryvolist
	,unnest(string_to_array(approvalhistoryvolist, '}')) AS sp_date_ele -- 上级领导审批时间
from upload.upload_huilianyi_reimbursement_form_history
)
where replace(sp_date_ele,E'\'','') like '%operation: 1001%' 
group by id
)

,lc_sp1_date as -- 上级领导审批时间
(
select 
	id
	,replace(replace(string_agg(SPLIT_PART(SPLIT_PART(sp_date_ele,'createdDate',2),E'\'',3),';'),'T',' '),'Z','')  as sp1_date
	,string_agg(SPLIT_PART(SPLIT_PART(sp_date_ele,'operatorName',2),E'\'',3),';') as sp1_name
from 
(
select 
	id
	,approvalhistoryvolist
	,unnest(string_to_array(approvalhistoryvolist, '}')) AS sp_date_ele 
from upload.upload_huilianyi_reimbursement_form_history
)
where sp_date_ele like '%上级领导审批%' and sp_date_ele not like '%节点重复%' and sp_date_ele not like '%节点为空%'
group by id
)

	,string_agg(distinct SPLIT_PART(SPLIT_PART(sp_date_ele,'operatorName',2),E'\'',3),';') as sp2_name

(
select 
	id
	,approvalhistoryvolist
	,unnest(string_to_array(approvalhistoryvolist, '}')) AS sp_date_ele -- 上级领导审批时间
from upload.upload_huilianyi_reimbursement_form_history
)

group by id
)

,lc_sp3_date as -- 财务总监审批

,base_information as 
(
select 
	t1.id
	,t1.billingCode_list -- 发票代码列表
	,t2.billingNo_list -- 发票号码列表
	,t0.sp0_date -- 创建人时间
	,t0.sp0_name -- 创建人姓名
	,t3.sp1_date -- 上级领导审批时间
	,t3.sp1_name -- 上级领导名称
	,t4.sp2_date --  财务环节-费用审核时间
	,t4.sp2_name -- 财务环节审批人
	,t5.sp3_date --  财务总监审批时间
	,t5.sp3_name -- 财务总监审批人
	,t6.sp4_date --  单据接受时间
	,t6.sp4_name -- 单据接受人
	,t7.sp5_date -- 财务环节-收单审核时间
	,t7.sp5_name -- 财务环节-收单审核人
	,t8.sp6_date -- 信用事件生成时间
	,t8.sp6_name -- 信用事件生成审核人
	,t9.sp7_date -- 已付款时间
	,t9.sp7_name -- 操作人
from pxd_billingCode t1
left join pxd_billingNo t2 on t1.id = t2.id
left join lc_sp1_date t3 on t1.id = t3.id
left join lc_sp2_date t4 on t1.id = t4.id
left join lc_sp3_date t5 on t1.id = t5.id
left join lc_sp4_date t6 on t1.id = t6.id
left join lc_sp5_date t7 on t1.id = t7.id
left join lc_sp6_date t8 on t1.id = t8.id
left join lc_sp7_date t9 on t1.id = t9.id
left join lc_sp0_date t0 on t1.id = t0.id
)

,receiptVOList_all as -- 附件ID1
(
select 
	id
	,SPLIT_PART(receiptVOList,E'\'',3) receiptVOList_all_oid
FROM 
(
	SELECT 
		  id
		  ,unnest(string_to_array(receiptVOList, 'attachmentOID')) AS receiptVOList -- attachmentOID
	FROM upload.upload_huilianyi_reimbursement_form_history
	union all 
		SELECT 
		  id
		  ,unnest(string_to_array(receiptVOList, 'slicingAttachmentOID')) AS receiptVOList -- slicingAttachmentOID
	FROM upload.upload_huilianyi_reimbursement_form_history
	union all 
		SELECT 
		  id
		  ,unnest(string_to_array(receiptVOList, 'originAttachmentOID')) AS receiptVOLis_t -- originAttachmentOID
	FROM upload.upload_huilianyi_reimbursement_form_history
)
where receiptVOList like E'\':%'
)

,receiptAttachmentVOList_all as -- 附件ID2
(
select 
	id
	,SPLIT_PART(receiptAttachmentVOList,E'\'',3) receiptAttachmentVOList_all_oid
FROM 
(
	SELECT 
		  id
		  ,unnest(string_to_array(receiptAttachmentVOList, 'attachmentOID')) AS receiptAttachmentVOList -- attachmentOID
	FROM upload.upload_huilianyi_reimbursement_form_history
	union all 
		SELECT 
		  id
		  ,unnest(string_to_array(receiptVOList, 'resourceOID')) AS receiptAttachmentVOList -- resourceOID
	FROM upload.upload_huilianyi_reimbursement_form_history
)
where receiptAttachmentVOList like E'\':%'
)

,fj_all as -- 附件汇总

	id 
	,receiptAttachmentVOList_all_oid as  fj_all_AttachmentOID

)

,fj_all_list as

	id

)
