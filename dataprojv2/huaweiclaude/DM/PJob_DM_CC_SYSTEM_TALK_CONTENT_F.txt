set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
-- 极低版本Hive兼容版：无transform，纯数值排序+纯对话内容
with chat_base as (
    select
        cbbo.ORDER_NO,
        COALESCE(cbdc.c_complaint_type,cbdc1.c_work_type) as c_complaint_type,
        COALESCE(cbdc.c_complaint_type_two,cbdc1.c_work_type_two) as c_complaint_type_two,
        COALESCE(cbdc.c_complaint_type_three,cbdc1.c_work_type_three) as c_complaint_type_three,
        ccr.BILL_BEGIN_TIME,
        concat(case when cqt.TYPE = 1 then '坐席' else '客户' end, ':', cqt.CONTENT) as chat_content,
        cast(cqt.start_offset as double) as start_offset  -- 数值型偏移量
    from sdi.SDI_C_BO_BASE_ORDE_1017 cbbo
    left join sdi.SDI_CC_CALL_RECORD ccr on cbbo.SOURCE_RECORD_ID = ccr.SERIAL_ID
    left join sdi.SDI_CC_QC_TRANSFER_1017 cqt on cbbo.SOURCE_RECORD_ID = cqt.SERIAL_ID
    left join sdi.sdi_c_box_dpcz_complaint_work_1017 cbdc on cbbo.id = cbdc.m_id
    left join sdi.sdi_c_box_dpcz_consulting_work_1017 cbdc1 on cbbo.id = cbdc1.m_id
    where 1=1
--        and cbbo.ORDER_NO = 'TS2025112717-6512'
--		  and cbbo.ORDER_NO = 'TS2025111808-1588'
        and ccr.BILL_BEGIN_TIME >= '2025-03-01 00:00:00'
		and ccr.BILL_BEGIN_TIME <= '2025-08-31 00:00:00'
		and substring(cbbo.ORDER_NO,1,2) not in ('20')
),
-- 按工单+偏移量排序，生成行号
chat_rn as (
    select
        ORDER_NO,
        BILL_BEGIN_TIME,
        c_complaint_type,
        c_complaint_type_two,
        c_complaint_type_three,
        chat_content,
        row_number() over (partition by ORDER_NO order by start_offset asc) as rn  -- 按偏移量升序编行号
    from chat_base
)
,tem as 
(
-- 按行号拼接，保证顺序
select
    ORDER_NO,
    max(BILL_BEGIN_TIME) as BILL_BEGIN_TIME,
    max(c_complaint_type) as c_complaint_type,
    max(c_complaint_type_two) as c_complaint_type_two,
    max(c_complaint_type_three) as c_complaint_type_three,
    concat_ws('\n', collect_list(chat_content) over (
        partition by ORDER_NO 
        order by rn asc 
        rows between unbounded preceding and unbounded following
    )) as chat_record
from chat_rn
group by ORDER_NO, rn, chat_content
)

,tem1 as 
(
select 
* ,
ROW_NUMBER() over(partition by ORDER_NO order by BILL_BEGIN_TIME ) rn

from tem
)

insert OVERWRITE TABLE dm.dm_cc_system_talk_content_f
select 
	ORDER_NO,
	BILL_BEGIN_TIME,
	c_complaint_type,
	c_complaint_type_two,
	c_complaint_type_three,
	chat_record
from tem1 
where rn = 1
