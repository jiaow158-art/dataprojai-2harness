out_fact as(
-- 实际发票数据
select
	substr(a.bill_date,1,10) as stat_date_d,
	a.sale_org_code,
	a.distribution_channel_code , 
	a.cust_sal,
	case when a.sale_org_code = '3830' then '津冀子公司'
	     when a.sale_org_code = '3831' then '贵州子公司'
		 when a.sale_org_code = '3832' then '苏沪子公司'
	 end as operation_center_name,
	a.material_num,
	sum(nvl(a.actual_billed_qty,0)) as actual_billed_qty,   -- 开票数量
    sum(a.subtotal_price_process1) as actual_billed_amt -- 开票金额
from DM_GL_OUT_INVOICE_T_DATA a  -- DM.DM_GL_OUT_INVOICE_T
where a.sale_org_code between '3210' and '3920'  -- 卫浴供应链
and a.payer_id not in ('0000003200','0000003300','0000003500','0000003600','0000003700','0000003800','0000003900')  -- 去除内部发票
and bill_date >=concat('${PERIOD_ID_M}','-01')   -- 跑昨日所在月份1号数据   2022-11-01
and a.sale_org_code IN('3830','3831','3832')
and a.inv_location not in ('WX','GYCK','TJ') -- 排除 WX 苏沪  GYCK 贵州  TJ  津冀

a.sale_org_code,
a.distribution_channel_code , 
a.cust_sal,
case when a.sale_org_code = '3830' then '津冀子公司'
	     when a.sale_org_code = '3831' then '贵州子公司'
		 when a.sale_org_code = '3832' then '苏沪子公司'
 end,
a.material_num

ekbe as  
(
select t1.ebeln ,
	   t1.ebelp , 
	   t1.budat,   -- 凭证日期
	   sum(case when t1.bewtp='L' then t1.menge else 0 end) as lmeng, -- 生产提货单数量
	   sum(case when t1.bewtp='U' AND t1.shkzg = 'H' then t1.menge 
			    when t1.bewtp='U' AND t1.shkzg = 'S' then 0-t1.menge 
			    else 0 end) as out_emeng,   -- 发货数量
	   sum(case when t1.bewtp='E' AND t1.shkzg = 'S' then t1.menge 
			    when t1.bewtp='E' AND t1.shkzg = 'H' then 0-t1.menge 
			    else 0 end) as in_emeng   -- 收货数量
from sdi.sdi_ekbe_1001 t1
where 1=1
group by t1.ebeln ,
	   t1.ebelp ,
	   t1.budat
),
sub_data_01 as(
 select concat(substring(cast(t3.budat as string),1,4),'-',substring(cast(t3.budat as string),5,2),'-',substring(cast(t3.budat as string),7,2)) as stat_date_d,
         case when t2.lgort = 'WX' then '苏沪子公司'
       		  when t2.lgort = 'GYCK' then '贵州子公司'
       		  when t2.lgort = 'TJ' then '津冀子公司'
       	   else '' end as operation_center_name,
       	T2.matnr AS material_num,
       	SUM(NVL(T3.in_emeng,0)) AS actual_billed_qty
  FROM sdi.sdi_ekko_1001 as T1 
  INNER JOIN sdi.sdi_ekpo_1001 as  T2 
  ON T1.ebeln = T2.ebeln
  JOIN  ekbe AS T3 
  on T2.ebeln  = T3.ebeln 
  and T2.ebelp  = T3.ebelp
  where T2.lgort in('WX','GYCK','TJ')   -- 库存地点
  and t3.budat >= replace(concat('${PERIOD_ID_M}','-01'),'-','')
  GROUP BY concat(substring(cast(t3.budat as string),1,4),'-',substring(cast(t3.budat as string),5,2),'-',substring(cast(t3.budat as string),7,2)),
  		   case when t2.lgort = 'WX' then '苏沪子公司'
       		  when t2.lgort = 'GYCK' then '贵州子公司'
       		  when t2.lgort = 'TJ' then '津冀子公司'
       	   else '' end,
       	   T2.matnr
)
,sub_com_2 as(
-- 子公司入库
select stat_date_d,
	   operation_center_name,
	   material_num,
	   sum(INSTOCK_QTY) as INSTOCK_QTY,
	   0 as OUTSTOCK_QTY
from(
	select stat_date_d,
		   operation_center_name,
		   material_num,
		   sum(actual_billed_qty) as INSTOCK_QTY
	from out_fact t
	where operation_center_name is not null
	group by stat_date_d,
			 operation_center_name,
			 material_num
	union all
	select t.stat_date_d,
		   t.operation_center_name,
		   t.material_num,
		   t.actual_billed_qty as INSTOCK_QTY
	from sub_data_01 t
) t
group by stat_date_d,
	     operation_center_name,
	     material_num
union all
-- 子公司出库数
select stat_date_d,
	   operation_center_name,
	   material_num,
	   0 as INSTOCK_QTY,
	   sum(actual_billed_qty) as OUTSTOCK_QTY
from out_fact t
where operation_center_name is not null
group by stat_date_d,
		 operation_center_name,
		 material_num
)
INSERT OVERWRITE TABLE dm_sw.DM_WM_CENTER_OUTINSTOCK_T_D PARTITION (stat_month)
select operation_center_name,
	   material_num,
	   sum(nvl(INSTOCK_QTY,0)) as INSTOCK_QTY,
	   sum(nvl(OUTSTOCK_QTY,0)) as OUTSTOCK_QTY,
	   current_timestamp() as dw_last_update_date ,
	   stat_date_d,
	   substring(stat_date_d,1,7) as stat_month
from sub_com_2 t
group by stat_date_d,
		 operation_center_name,
		 material_num
