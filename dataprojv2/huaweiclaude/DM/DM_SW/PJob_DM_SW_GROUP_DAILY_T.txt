SELECT A.SALE_GRP_NAME AS SALE_GRP_CODE
      ,A.PROD_CHANNEL_NAME AS OLD_INTEGRATE_CHANNEL_CODE
      ,D.CHANNEL_NAME AS OLD_INTEGRATE_CHANNEL_T
	  ,CASE WHEN A.PROD_CHANNEL_NAME = 'GD99' THEN 'GD03'
            ELSE A.PROD_CHANNEL_NAME
	         END AS INTEGRATE_CHANNEL_CODE
      ,CASE WHEN A.PROD_CHANNEL_NAME = 'GD99' THEN '工程渠道'
            ELSE D.CHANNEL_NAME
	         END AS INTEGRATE_CHANNEL_T
      ,B.NODE_NAME9
      ,B.NODE_NAME2
      ,B.NODE_DESC2
      ,B.NODE_NAME3
      ,B.NODE_DESC3
      ,B.NODE_NAME4
      ,B.NODE_DESC4
      ,B.NODE_NAME5
      ,B.NODE_DESC5
      ,B.NODE_NAME6
      ,B.NODE_DESC6	  
      ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS MONTH_BUDGET_ACHIEVEMENT --当月预算业绩
	  ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') >= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
	             AND TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS ACC_BUDGET_ACHIEVEMENT --当年累计预算业绩
	  ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') >= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
	             AND TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-12') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS YEAR_BUDGET_ACHIEVEMENT --全年预算业绩	  
	  ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') = TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEAR','yyyy-MM') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS LY_MONTH_BUDGET_ACHIEVEMENT --去年当月预算业绩
	  ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') >= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEAR','yyyy-01')
	             AND TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') <= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEAR','yyyy-MM') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS LY_ACC_BUDGET_ACHIEVEMENT --去年累计预算业绩
	  ,SUM(CASE WHEN TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') >= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEAR','yyyy-01')
	             AND TO_CHAR(TO_DATE(A.STAT_MONTH),'yyyy-MM') <= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEAR','yyyy-12') 
	            THEN A.PROD_IT_AMT1
				ELSE 0
				 END) / 10000 AS LY_YEAR_BUDGET_ACHIEVEMENT --去年预算业绩
  FROM UPLOAD.UPLOAD_ACHIEVEMENT_BUDGET_T A
  JOIN DM.DM_RPT_SALES_GROUP_T B
    ON A.SALE_GRP_NAME = B.NODE_NAME9
  LEFT JOIN DM.DM_DIM_INTEGRATE_CHANNEL_D D
    ON A.PROD_CHANNEL_NAME = D.CHANNEL_CODE
   AND D.CHANNEL_TYPE = '整合渠道1'
 WHERE B.NODE_NAME2 = 'H11000002'
 --剔除销售组代码为R2T，管理渠道非"GD06"
   AND CASE WHEN A.SALE_GRP_NAME = 'R2T' AND COALESCE(A.PROD_CHANNEL_NAME,'$$$') = 'GD06' THEN 1
            WHEN A.SALE_GRP_NAME = 'R2T' THEN 0
			ELSE 1
			 END = 1

      ,A.PROD_CHANNEL_NAME

	  ,CASE WHEN A.PROD_CHANNEL_NAME = 'GD99' THEN 'GD03'
            ELSE A.PROD_CHANNEL_NAME
	         END

            ELSE D.CHANNEL_NAME
	         END

      ,B.NODE_DESC4

),BASE AS( --业绩、成本、收入
SELECT  A.NODE_NAME9 AS SALE_GRP_CODE
       ,A.INTEGRATE_CHANNEL AS OLD_INTEGRATE_CHANNEL_CODE
       ,D.CHANNEL_NAME AS OLD_INTEGRATE_CHANNEL_T
	   ,CASE WHEN A.INTEGRATE_CHANNEL = 'GD99' THEN 'GD03'
             ELSE A.INTEGRATE_CHANNEL
	          END AS INTEGRATE_CHANNEL_CODE
       ,CASE WHEN A.INTEGRATE_CHANNEL = 'GD99' THEN '工程渠道'
             ELSE D.CHANNEL_NAME
	          END AS INTEGRATE_CHANNEL_T
       ,A.NODE_NAME9
       ,A.NODE_NAME2
       ,A.NODE_DESC2
       ,A.NODE_NAME3
       ,A.NODE_DESC3
       ,A.NODE_NAME4
       ,A.NODE_DESC4
       ,A.NODE_NAME5
       ,A.NODE_DESC5
       ,A.NODE_NAME6
       ,A.NODE_DESC6
       ,SUM(CASE WHEN CALDAY = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 DAY_ACHIEVEMENT --日业绩
	   ,SUM(CASE WHEN CALDAY = TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 DAY','yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 LAST_DAY_ACHIEVEMENT --昨日业绩
       ,SUM(CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 MONTH_ACHIEVEMENT --月累计业绩
       ,SUM(CASE WHEN CALMONTH BETWEEN TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
	              AND TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 YEAR_ACHIEVEMENT --年累计业绩
       ,SUM(CASE WHEN CALDAY = TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 LAST_YEAR_DAY_ACHIEVEMENT --去年同期日业绩
       ,SUM(CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 LAST_YEAR_MONTH_ACHIEVEMENT --去年同期月累计业绩
       ,SUM(CASE WHEN CALMONTH BETWEEN TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyy-01')
	              AND TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyyMMdd')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 LAST_YEAR_ACHIEVEMENT --去年同期年累计业绩
	   ,SUM(CASE WHEN CALDAY = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN NOTAX
                 ELSE 0 END) / 10000 DAY_NOTAX --日不含税收入 
       ,SUM(CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN NOTAX
                 ELSE 0 END) / 10000 MONTH_NOTAX --月累计不含税收入
       ,SUM(CASE WHEN CALMONTH BETWEEN TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
	              AND TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN NOTAX
                 ELSE 0 END) / 10000 YEAR_NOTAX --年累计不含税收入				 
	   ,SUM(CASE WHEN CALDAY = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN SALESCOST
                 ELSE 0 END) / 10000 DAY_SALESCOST --日销售成本
       ,SUM(CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN SALESCOST
                 ELSE 0 END) / 10000 MONTH_SALESCOST --月累计销售成本	 
       ,SUM(CASE WHEN CALMONTH BETWEEN TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
	              AND TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                  AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                 THEN (CASE WHEN CALMONTH BETWEEN TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-01')
				             AND TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 MONTHS','yyyy-MM')
                             AND LEFT(CALMONTH, 4) = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy')
                             AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                            THEN A_SALES_COST
                            ELSE 0 END --截止到上个月末的实际A成本
					 + CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyy-MM')
                             AND CALDAY <= TO_CHAR(TO_DATE('${PERIOD_ID_D}'),'yyyyMMdd')
                            THEN SALESCOST
                            ELSE 0 END --当月销售成本
                       )
                 ELSE 0 END) / 10000 YEAR_SALESCOST --年累计销售成本
       ,SUM(CASE WHEN CALMONTH = TO_CHAR(TO_DATE('${PERIOD_ID_D}') - INTERVAL '1 YEARS','yyyy-MM')
                 THEN AMBPERFORMANCE
                 ELSE 0 END) / 10000 LAST_YEAR_ALL_MONTH_ACHIEVEMENT --去年同期整月累计业绩
  FROM DM.DM_FIN_GROUP_DAILY_DETAIL A
  LEFT JOIN DM.DM_DIM_INTEGRATE_CHANNEL_D D

   AND D.CHANNEL_TYPE = '整合渠道1'
 WHERE A.NODE_NAME2 IN('H11000002')

	          END
       ,CASE WHEN A.INTEGRATE_CHANNEL = 'GD99' THEN '工程渠道'
             ELSE D.CHANNEL_NAME
	          END
