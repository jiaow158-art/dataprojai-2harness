delete from dm_sw.dm_bath_storage_age_t WHERE stat_month = substr('${PERIOD_ID_D}',1,7);
insert into dm_sw.dm_bath_storage_age_t
with tmp as (
select 
bath_category_name as 品类
,case when (b.matl_group2_code = 'J10') then '退市库存' else '常规库存' end as 库存状态
,a.物料编码
,a.物料编码描述
,c.bathroom_channal as 产品渠道
,a.库位分类
,sum(a.月末库存数量+a.月末外协寄售数量) as 库存数量
,sum(a.月末库存金额+a.月末外协寄售金额)/10000 as 库存金额
,sum(a."1-6个月数量") as 半年以下数量  
,sum(a."7-12个月数量") as 半年至一年数量
,sum(a."1-2年数量") as 一年至两年数量
,sum(a."2年以上数量") as 两年以上数量
,sum(a."1-6个月金额")/10000  as 半年以下金额
,sum(a."7-12个月金额")/10000 as 半年至一年金额
,sum(a."1-2年金额")/10000  as 一年至两年金额
,sum(a."2年以上金额")/10000 as 两年以上金额
from dm.dm_own_inventory_t as a 
left join upload.upload_wm_sw_sap_prod_grp_t as k on a.产品层次3 = k.prod_code3 
left join (select t.material_num,max(t.matl_group2_code) as matl_group2_code 
from dwimd.dwi_md_data_material_sales_t_his as t 
where start_date < last_day(to_date( concat( substr('${PERIOD_ID_D}',1,4),'-',substr('${PERIOD_ID_D}',6,2),'-','01'),'YYYY-MM-DD'))
and end_date >= last_day(to_date( concat( substr('${PERIOD_ID_D}',1,4),'-',substr('${PERIOD_ID_D}',6,2),'-','01'),'YYYY-MM-DD'))
group by t.material_num ) b on a.物料编码 = b.material_num
left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T c on a.物料编码 = c.material_num 
where concat(substr(a.库存月份,1,4),'-',substr(a.库存月份,5,2)) = substr('${PERIOD_ID_D}',1,7)  and a.工厂 not like '39%'
group by 
bath_category_name
,case when (b.matl_group2_code = 'J10') then '退市库存' else '常规库存' end
,a.物料编码
,a.物料编码描述
,c.bathroom_channal 
,a.库位分类
)

select
 substr('${PERIOD_ID_D}',1,7) as stat_month
,品类 as bath_category_name
,库存状态 as inv_status
,物料编码 as material_num
,物料编码描述 as material_name
,产品渠道 as bathroom_channal
,库位分类 as loc_class
,库存数量 as inv_qty
,库存金额 as inv_amount
,半年以下数量 as inv_qty1
,半年至一年数量 as inv_qty2
,一年至两年数量 as inv_qty3
,两年以上数量 as inv_qty4
,半年以下金额 as inv_amount1
,半年至一年金额 as inv_amount2
,一年至两年金额 as inv_amount3
,两年以上金额 as inv_amount4
,CURRENT_TIMESTAMP() AS LAST_UPDATE_DATE 
from tmp
where 1=1
