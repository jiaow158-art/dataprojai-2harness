with A as (
select *
	,case 
		when t.org_lev3_name = '供应链中心' or t.org_lev4_name = '供应链中心' then '供应链中心'
		when t.org_lev3_name = '卫浴国际营销' then '卫浴国际营销'
		when t.org_lev3_name = '品类运营部' then '品类运营部'
		when (t.org_lev3_name = '商用中心' or t.org_lev4_name = '商用中心')  then '商用中心' --这里需要区分自营、经销
		when t.org_lev3_name = '国内营销中心' then '卫浴子公司-STO转储订单' --卫浴子公司-STO转储订单
		when (t.org_lev3_name = '营销中心' or  t.org_lev4_name = '营销中心') and t.channel_code <> '06' then '营销中心'
		when (t.org_lev3_name = '营销中心' or  t.org_lev4_name = '营销中心') and t.channel_code = '06' then '商用中心'
		when t.org_lev3_name = '客户服务部' then '客户服务部'
		when t.org_lev3_name = '鹏嘉家居' then '鹏嘉家居'
		when t.org_lev3_name = '国内直营板块' then '瓷砖事业部'
		when t.org_lev3_name = '国内经销板块' then '瓷砖事业部'
		when t.org_lev3_name = '国内营销系统' then '瓷砖事业部'
		when t.org_lev3_name = '国际营销系统' then '瓷砖事业部'
		when t.org_lev3_name = '瓷砖新零售中心' then '瓷砖事业部'
		when t.org_lev3_name = '瓷砖营销中心' then '瓷砖事业部'
		when t.org_lev3_name = '其他' then '其他'
		else org_lev2_name
	end as sub_grp,
	case 
		when t.org_lev3_name = '供应链中心' or t.org_lev4_name = '供应链中心' then '供应链中心'
		when t.org_lev3_name = '卫浴国际营销' then '卫浴国际营销'
		when t.org_lev3_name = '品类运营部' then t.org_lev5_name
		when (t.org_lev3_name = '商用中心' or t.org_lev4_name = '商用中心') and t.cust_group_code in ('A0','A4','A1') then '商用中心-自营' 
		when (t.org_lev3_name = '商用中心' or t.org_lev4_name = '商用中心') and t.cust_group_code not in ('A0','A4','A1') then '商用中心-经销'--这里需要区分自营、经销
		when t.org_lev3_name = '国内营销中心' then t.org_lev4_name --卫浴子公司-STO转储订单
		when (t.org_lev3_name = '营销中心' or  t.org_lev4_name = '营销中心') and t.channel_code <> '06' then t.org_lev5_name
		when (t.org_lev3_name = '营销中心' or  t.org_lev4_name = '营销中心') and t.channel_code = '06' and t.cust_group_code in ('A0','A4','A1')  then '商用中心-自营'
		when (t.org_lev3_name = '营销中心' or  t.org_lev4_name = '营销中心') and t.channel_code = '06' and t.cust_group_code not in ('A0','A4','A1')  then '商用中心-经销'
		when t.org_lev3_name = '客户服务部' then '客户服务部'
		when t.org_lev3_name = '鹏嘉家居' then '鹏嘉家居'
		when t.org_lev3_name = '国内直营板块' then t.org_lev4_name
		when t.org_lev3_name = '国内经销板块' then t.org_lev4_name
		when t.org_lev3_name = '国内营销系统' then t.org_lev5_name
		when t.org_lev3_name = '国际营销系统' then '国际营销系统'
		when t.org_lev3_name = '瓷砖新零售中心' then '瓷砖新零售中心'
		when t.org_lev3_name = '瓷砖营销中心' then t.org_lev5_name
		when t.org_lev3_name = '其他' then '其他'
		else concat(org_lev2_name,'-其他部门')
	end as sub_grp1
	,case 
		when t.dz_flag = '是' then '是'
		when t.is_have_group4 = '否' and substring(t.order_num,1,3) in (133,139) then '是'
		else '否'
	end as new_is_dz_dingzhi
	,case 
		when channel_code = '06' then  channel_code -- 工程
		when COALESCE(posnr_zcd,'NAL') in ('NAL','000000') and COALESCE(bom_material_num,'NAL') <> 'NAL' then bom_material_num
		else null
	end as bom_material_key -- BOM场景：正常下单A，订单行间存在BOM关系
	,case 
		when channel_code = '06' then  channel_code -- 工程场景
		when COALESCE(posnr_zcd,'NAL') not in ('NAL','000000') then posnr_zcd -- 拆单场景
		else order_item_num
	end as ipk -- 拆单场景+正常订单场景：非BOM拆单+BOM拆单+正常销售订单
from dm_sw.dm_otd_sales_order_det_sw t
where 1=1
and (inv_location in('GR','GR90','GD','GD90','GT','GT90','GQ','GA','GO','GK','GB','GZ','GF','GX') or inv_location is null)
and factory_code in('3220','3320','3520','3620','3720','3820')
and least(promise_stockout_date,COALESCE(stockout_date,'4712-12-31'))<substring(current_timestamp(),1,10)
and res_stock_qty>=order_qty
and COALESCE(cancal_code,'NAL') in ('','NAL')
and bsark <> 'CS'
)

,B as (
select order_num
	,COALESCE(bom_material_key,ipk) as bomipk
	,sum(order_qty) as bomipk_order_qty
	,sum(res_stock_qty) as bomipk_res_stock_qty
	,sum(stockout_qty) as bomipk_stockout_qty
	,max(COALESCE(stockout_date,'4712-12-31')) as max_bomipk_stockout_date
	,max(COALESCE(promise_stockout_date,'4712-12-31')) as max_bomipk_promise_stockout_date
from A
group by order_num,COALESCE(bom_material_key,ipk)
order by order_num
)
