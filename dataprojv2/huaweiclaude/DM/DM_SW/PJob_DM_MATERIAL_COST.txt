with tmp as (
select t1.matnr, t1.knumh, row_number() over(partition by matnr order by flag) rn from 
(
select matnr,knumh , 1 as flag from sdi.sdi_a963_1001 t1 
where kschl ='PR00'
and konda='JI'
and datbi='99991231'  
UNION  ALL 
select matnr,knumh , 4 as flag from sdi.sdi_a930_1001 t1 
where kschl ='PR00'
and konda='JI'
and datbi='99991231' 
UNION  ALL 
select matnr,knumh , 5 as flag from sdi.sdi_a941_1001 t1 
where kschl ='PR00'
and konda='JI'
and datbi='99991231' 
UNION  ALL 
select matnr,knumh , 6 as flag from sdi.sdi_a930_1001 t1 
where kschl ='PR00'
and konda='JI'
and datbi='99991231' 

) t1 
)
INSERT OVERWRITE TABLE DM_SW.DM_MATERIAL_COST
select t1.matnr as material_num,
	   t1.knumh ,
	   t2.KBETR/t2.KPEIN as unit_price,
       t2.KBETR/t2.KPEIN*0.31 as unit_cost,
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
from tmp t1
left join sdi.sdi_konp_1001  t2 ON t1.knumh = t2.knumh and t2.kopos ='01'
where t1.rn=1;
