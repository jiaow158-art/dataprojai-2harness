with base_information as (
	select 
		coalesce(a1.product_level4_code,a1.product_level_code) as product_level4_code,
		c1.bath_category_name, --外挂表：大品类
        c1.product_level3_name, --外挂表:小品类

        a1.material_num_short,
        case when c1.product_level3_name in ('浴室柜', '晾衣架','洗衣柜') then a1.material_bom else '' end as material_bom,--物料的bom
        --简码和物料编码的返回处理

        	when c1.bath_category_name = '智能' and c1.product_level3_name = '智能坐便器' then a1.material_num_short
        	when c1.bath_category_name = '智能' and c1.product_level3_name = '智能盖板' then a1.material_num
        	when c1.bath_category_name = '陶瓷' and a1.material_num LIKE  'W%' then a1.material_num_short
        	else a1.material_num 
        end as re_material,
        b1.matl_group1_code,
        b1.matl_group1_name, --获取的是ABC对应的等级
        b1.matl_group2_code, 
        b1.matl_group2_name, -- 是否在市
        b1.matl_group3_code, 
        b1.matl_group3_name --中高端
	from dm_sw.dm_sw_sale_rank_structure_t a1
	left join 
		(
		select 
			distinct material_num,matl_group1_code,matl_group1_name,matl_group2_code,
			matl_group2_name,matl_group3_code,matl_group3_name
		from dwimd.dwi_md_data_material_sales_t 
		where matl_group1_code is not null and sales_org_code like '3%'
		) b1 on a1.material_num = b1.material_num 
	left join upload.upload_sw_sales_rank_category_structure c1 
		on coalesce(a1.product_level4_code,a1.product_level_code) = c1.product_level4_code
	where  1=1
),

		t1.bill_date, 
		t1.distribution_channel_code, 
		t1.actual_billed_qty
	from DM.DM_GL_OUT_INVOICE_T t1
	join base_information t2 on t1.material_num = t2.material_num
	where

	       ('0000003200', '0000003300', '0000003500', '0000003600', '0000003700', '0000003800',
	        '0000003900', '0000003A00')
),
material_channel as (
	select 
		A.material_num
		,B.distribution_channel_code
	from (select distinct material_num from DM_GL_OUT_INVOICE_T ) A
	left join (SELECT DISTINCT distribution_channel_code FROM DM.DM_GL_OUT_INVOICE_T) B on 1=1
),
--查询月的销售情况
maintableCurrerntMonth as (
	select material_num,
	 	YearMonth,
	 	distribution_channel_code,
	 	sum(actual_billed_qty) as actual_billed_qty
	from (
		select t.material_num,
	       substr(t.bill_date, 1, 7) as YearMonth,
	       t.distribution_channel_code,
	       sum(IFNULL(t.actual_billed_qty,0)) as actual_billed_qty
	    from DM_GL_OUT_INVOICE_T t
	    where 1 = 1
	    	and substr(t.bill_date, 1, 7) = substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'),1,7)
	    group by t.material_num, substr(t.bill_date, 1, 7),t.distribution_channel_code
	    union all 
		select 
			material_num,
			substr(TO_DATE('${PERIOD_ID_D}','YYYY-MM-DD'), 1, 7) as YearMonth,
			distribution_channel_code,
			'0' as actual_billed_qty
		from material_channel
		
		)q1
	group by material_num,YearMonth,distribution_channel_code
),
--查询月的上个月销售情况
maintableLastMonth as (
	select 
		material_num,
		YearMonth,
		distribution_channel_code,
		sum(actual_billed_qty) as actual_billed_qty
	from(
		select t.material_num,
			substr(t.bill_date, 1, 7) as YearMonth,
			t.distribution_channel_code,
			sum(IFNULL(t.actual_billed_qty,0)) as actual_billed_qty
        from DM_GL_OUT_INVOICE_T t
        where 1 = 1
          and substr(t.bill_date, 1, 7) = SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7)
        group by t.material_num,substr(t.bill_date, 1, 7),t.distribution_channel_code
        union all 
		select 
			material_num,
			SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7) as YearMonth,
			distribution_channel_code,
			'0' as actual_billed_qty
		from material_channel)q2
	group by material_num,YearMonth,distribution_channel_code
),
--去年同月销售情况
maintableLastYearMonth as (
	select 
		material_num,
		YearMonth,
		distribution_channel_code,
		sum(actual_billed_qty) as actual_billed_qty
	from (		
		select t.material_num,
               substr(t.bill_date, 1, 7) as YearMonth,
               t.distribution_channel_code,
               sum(ifnull(t.actual_billed_qty,0)) as actual_billed_qty
        from DM_GL_OUT_INVOICE_T t
        where 1 = 1
          and substr(t.bill_date, 1, 7) = SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 year', 1, 7)
        group by t.material_num, substr(t.bill_date, 1, 7),t.distribution_channel_code
        union all 
		select 
		material_num,
		SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 year', 1, 7) as YearMonth,
		distribution_channel_code,
		'0' as actual_billed_qty
	from material_channel)q3
	group by material_num,YearMonth,distribution_channel_code
),
--当年年同期累计销量
 maintableCurrerntYear as (
	select 
		material_num,
		Year,
		distribution_channel_code,
		sum(actual_billed_qty) as actual_billed_qty
	from (
	    select t.material_num,
              substr(t.bill_date, 1, 4) as Year,
              t.distribution_channel_code,
              sum(IFNULL(t.actual_billed_qty,0)) as actual_billed_qty
        from DM_GL_OUT_INVOICE_T t
        where 1 = 1
          and substr(t.bill_date, 1, 4) = substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'), 1, 4)
          and substr(t.bill_date, 1, 7) <= substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'), 1, 7)
        group by t.material_num, substr(t.bill_date, 1, 4),t.distribution_channel_code
        union all 
		select 
			material_num,
			substr(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD'), 1, 4) as YearMonth,
			distribution_channel_code,
			'0' as actual_billed_qty
	from material_channel)q4
	group by material_num,Year,distribution_channel_code
),
--去年年同期累计销量
maintableLastYear as (
	select 
		material_num,
		Year,
		distribution_channel_code,
		sum(actual_billed_qty) as actual_billed_qty
	from(
		select t.material_num,
              substr(t.bill_date, 1, 4) as Year,
              t.distribution_channel_code,
              sum(IFNULL(t.actual_billed_qty,0)) as actual_billed_qty
        from DM_GL_OUT_INVOICE_T t
        where 1 = 1
          and substr(t.bill_date, 1, 4) >=
              SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 year', 1, 4)
          and substr(t.bill_date, 1, 7) <=
              SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 year', 1, 7)
        group by t.material_num, substr(t.bill_date, 1, 4),t.distribution_channel_code
		union all 
		select 
			material_num,
			SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 year', 1, 4) as YearMonth,
			distribution_channel_code,
			'0' as actual_billed_qty
		from material_channel)q5
	group by material_num,Year,distribution_channel_code
),
--查询月、查询月的上个月、去年同月、当年年、去年年、去年年同期 
all_time_tem as (
	select 
			COALESCE(maintableCurrerntMonth.material_num,maintableCurrerntMonth.material_num,maintableCurrerntMonth.material_num,maintableCurrerntMonth.material_num) as all_material_num, 
			maintableLastMonth.YearMonth                  as LM_YearMonth,
             maintableLastMonth.distribution_channel_code as LM_channel_code,
             maintableLastMonth.actual_billed_qty          as LM_actual_billed_qty,
             maintableCurrerntMonth.YearMonth              as CM_YearMonth,
             maintableCurrerntMonth.distribution_channel_code as CM_channel_code,
             maintableCurrerntMonth.actual_billed_qty      as CM_actual_billed_qty,
             maintableCurrerntYear.Year                    as CY_Year,
             maintableCurrerntYear.distribution_channel_code as CY_channel_code,
             maintableCurrerntYear.actual_billed_qty     as CY_actual_billed_qty,
             maintableLastYear.Year                        as LY_Year,
             maintableLastYear.distribution_channel_code as LY_channel_code,
             maintableLastYear.actual_billed_qty         as LY_actual_billed_qty,
             maintableLastYearMonth.YearMonth                        as LYM_YearMonth,
             maintableLastYearMonth.distribution_channel_code as LYM_channel_code,
             maintableLastYearMonth.actual_billed_qty             as LYM_actual_billed_qty
	from maintableCurrerntMonth
	join maintableLastMonth on maintableCurrerntMonth.material_num = maintableLastMonth.material_num 
		and maintableCurrerntMonth.distribution_channel_code = maintableLastMonth.distribution_channel_code 
	join maintableLastYearMonth on maintableCurrerntMonth.material_num = maintableLastYearMonth.material_num
		and maintableCurrerntMonth.distribution_channel_code = maintableLastYearMonth.distribution_channel_code 
	join maintableCurrerntYear on maintableCurrerntMonth.material_num = maintableCurrerntYear.material_num
		and maintableCurrerntMonth.distribution_channel_code = maintableCurrerntYear.distribution_channel_code 
	join maintableLastYear on maintableCurrerntMonth.material_num = maintableLastYear.material_num
		and maintableCurrerntMonth.distribution_channel_code = maintableLastYear.distribution_channel_code),

--对21-JTN4033AD、21-JTN4019AD、21-JTN4005ADQ、21-JTN4000ADQ这4款感应器，且这4款感应器销量需再加上对应配套的小便器的销量进行处理
tem21_to_product as 
	(
	select 
		tem21_3.bom_component, --子项
		tem21_2.material_num,--母项
		row_number() over(partition by tem21_3.bom_component,tem21_2.material_num order by tem21_2.material_num desc ) as RN -- 多个含有该材料配件的产品-去重
	from 
	    (
	    select matl_list,material_num
	    from 
	            (
	            select 
	            *,row_number() over(partition by material_num order by rec_crt_date desc ) as RN
	            from DWRDIM.DWR_DIM_MATERIAL_LIST_LINK_F 
	            ) as tem21_1
--		    使用的是销售的bom是5，生产还有一个bom为1
	    where RN = 1 and bom_use_code = '1' 
	    ) tem21_2
	left join DWRDIM.DWR_DIM_BOM_ITEM_D tem21_3 on tem21_3.matl_list = tem21_2.matl_list 
	where tem21_3.bom_component in('21-JTN4033AD','21-JTN4019AD', '21-JTN4005ADQ','21-JTN4000ADQ')
	union all
	select  distinct
		all_material_num as bom_component, 
		all_material_num as material_num,
		1 as RN
	from all_time_tem
	),
	
--加上21-JTN4033AD、21-JTN4019AD、21-JTN4005ADQ、21-JTN4000ADQ上对应配套的小便器的销量
all_time as 
	(
	select 
--			coalesce(tem21_to_product.bom_component,all_time_tem.all_material_num) as all_material_num,
				all_time_tem.all_material_num,
			lm_yearmonth,
			lm_channel_code,
			sum(IFNULL(lm_actual_billed_qty,0)) as lm_actual_billed_qty,
			cm_yearmonth,
			cm_channel_code,
			sum(IFNULL(cm_actual_billed_qty,0)) as cm_actual_billed_qty,
			cy_year,
			cy_channel_code,
			sum(IFNULL(cy_actual_billed_qty,0)) as cy_actual_billed_qty,
			ly_year,
			ly_channel_code,
			sum(IFNULL(ly_actual_billed_qty,0)) as ly_actual_billed_qty,
			lym_yearmonth,
			lym_channel_code,
			sum(IFNULL(lym_actual_billed_qty,0)) as lym_actual_billed_qty
		from tem21_to_product
		left  join all_time_tem  on tem21_to_product.material_num = all_time_tem.all_material_num
		where tem21_to_product.RN = 1 
		group by --coalesce(tem21_to_product.bom_component,all_time_tem.all_material_num),
		all_time_tem.all_material_num,
			lm_yearmonth,
			lm_channel_code,
			cm_yearmonth,
			cm_channel_code,
			cy_year,
			cy_channel_code,
			ly_year,
			ly_channel_code,
			lym_yearmonth,
			lym_channel_code
	)
