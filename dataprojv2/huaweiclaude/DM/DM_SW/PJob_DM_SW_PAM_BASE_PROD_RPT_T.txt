with jm_zj_data_detail as( -- 注浆明细数据(jm_np_fact_num) 
select t.stat_date,   -- 生产日期
       sum(t.jm_zj_qty) as jm_zj_qty  -- 注浆数 
from(
	select substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) AS stat_date,  -- 生产日期
			  a.jm_np_code,   -- 泥坯编码
			  b.JM_LINENO_CODE , -- 生产线号编码
			  b.jm_lineno_name,  -- 生产线号名称 
			  b.jm_zj_class_name,  -- 注浆班组
			  a.jm_main_id,     -- 主表ID
			  sum(a.jm_np_num) jm_np_num, -- 模具数量
			  sum(a.jm_np_fact_num) as jm_zj_qty, -- 模具实际数量     注浆数量
			  sum(case when b.JM_IS_COMPLEMENT=1 then a.jm_np_fact_num else 0 end ) jm_num1 -- JM_IS_COMPLEMENT 是否补码
		 from SDI.SDI_jm_t_wy_zj_plan_detail_1038 a -- 注浆计划产品明细
		join SDI.SDI_jm_t_wy_zj_plan_1038 b -- 注浆计划
		on b.jm_uuid=a.jm_main_id   
		where a.jm_is_del=0 
		and b.jm_is_del=0
		and substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) >= '2022-06'
		group by substring(from_unixtime(unix_timestamp(b.jm_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) ,
				 a.jm_np_code,
				 b.JM_LINENO_CODE ,
				 b.jm_lineno_name,
				 b.jm_zj_class_code,
				 b.jm_zj_class_name, 
				 a.jm_main_id 
    ) t
    group by t.stat_date
)
,jm_zj_data as(  -- -- 注浆数
	select t1.day_dt as stat_date,
		   nvl(t2.jm_zj_qty,0) as jm_zj_qty
	from dm.dm_dim_date_d t1
	left join jm_zj_data_detail t2
	on t1.day_dt = t2.stat_date
	where t1.day_dt >= '2022-06-01'
	and t1.day_dt <= date_sub(CURRENT_DATE() ,1)--current_date()
),
jm_actual_receipt_01 as(  -- 收呸数  
	SELECT STAT_DATE,
		   SUM(jm_hg_qty) AS jm_hg_qty
	FROM(
		SELECT t.STAT_DATE, -- 普通半检
			   sum(t.jm_hg_qty) as jm_hg_qty
		FROM(
			select substring(from_unixtime(unix_timestamp(a.jm_gx_pj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) AS STAT_DATE, -- 评级日期
				   a.jm_np_code,  -- JM_NP_CODE
				   sum(case when a.jm_uuid is not null then 1 else 0 end ) jm_num,  -- jm_uuid 唯一标识
				   sum(case when   a.jm_is_standard=1 then 1 else 0 end ) jm_hg_qty,  -- 是否合格('1:合格 2:返修') JM_GX_HG
				   sum(case when   a.jm_is_standard=2 then 1 else 0 end ) jm_gb_num,
				   sum(case when   a.JM_IS_AL_GB=1 then 1 else 0 end ) jm_num1  -- 是否已干补
			from SDI.SDI_jm_t_gx_online_1038 a -- 改洗上线检验
			where a.jm_is_del=0   
			and a.jm_is_judge=1  
			and a.JM_IS_COMPLEMENT =0 
			and substring(from_unixtime(unix_timestamp(a.jm_gx_pj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) >= '2022-06'
			group by a.jm_np_code,
					 substring(from_unixtime(unix_timestamp(a.jm_gx_pj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10)--SUBSTRING(CAST(a.jm_gx_pj_date AS STRING),1,10)
			) t
		GROUP BY t.STAT_DATE
		union all   -- 高压半检
		select stat_date,
		   sum(jm_hg_num) as jm_hg_qty
		from(
			select  substring(from_unixtime(unix_timestamp(a.JM_GY_PJ_DATE) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) AS STAT_DATE,
					a.jm_np_code,
					a.jm_lineno_code,
					a.jm_lineno_name,
					a.jm_zj_class_code,
					b.jm_main_id,
					a.jm_is_gy,
					sum(case when a.jm_uuid is not null then 1 else 0 end ) jm_num,
					sum(case when   a.jm_is_standard=1 then 1 else 0 end ) jm_hg_num,
					sum(case when   a.jm_is_standard=2 then 1 else 0 end ) jm_gb_num
			 from SDI.SDI_JM_T_GY_PJ_1038 a --STG.STG_jm_t_gy_pj_1038 a
			 left join SDI.SDI_JM_T_WY_ZJ_PLAN_BARCODE_1038 b -- STG.STG_jm_t_wy_zj_plan_barcode_1038 b
			 on a.jm_uuid=b.jm_gx_id
			where a.jm_is_del=0
			 group by substring(from_unixtime(unix_timestamp(a.JM_GY_PJ_DATE) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) ,
					  a.jm_np_code,
					  a.jm_lineno_code,
					  a.jm_lineno_name,
					  a.jm_zj_class_code,
					  b.jm_main_id,
					  a.jm_is_gy 
			) t
			group by stat_date
			union all    -- 半检报废数
			select stat_date,
			   -sum(jm_num1) as jm_hg_qty
			from(
				select  substring(from_unixtime(unix_timestamp(a.JM_DUMPINGER_DATE) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) stat_date  ,
						a.jm_np_code   ,
						a.jm_lineno_code,
						a.jm_lineno_name,
						a.jm_zj_class_code,
						a.jm_zj_main_id,
						a.JM_IS_GY   ,
						sum(case when a.jm_uuid is not null then 1 else 0 end )   ,
						sum(case when a.jm_is_standard=1 and a.JM_IS_CQ=0  then 1 else 0 end ) as jm_num1
					 from sdi.sdi_jm_t_gx_online_dumping_1038 a
					 left join SDI.SDI_JM_T_GX_ONLINE_1038 b --stg.stg_jm_t_gx_online_1038 b
					 on a.jm_barcode =b.jm_barcode and b.jm_is_del=0
					where a.jm_is_del=0 and a.jm_is_flow=2
					 and b.JM_IS_COMPLEMENT =0  
					 --and exists (  select 1 from table(splitstr('',',')) where column_value=a.jm_factory_id  ) 
					 --and a.JM_DUMPINGER_DATE between :1 and :2   
					 group by substring(from_unixtime(unix_timestamp(a.JM_DUMPINGER_DATE) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10)  ,
								a.jm_np_code,
								a.jm_lineno_code,
								a.jm_lineno_name,
								a.jm_zj_class_code,
								a.jm_zj_main_id,
								a.JM_IS_GY  
			  ) t
			  group by stat_date
		)t
		group by stat_date
),

,jm_first_data_01 as(  -- 一烧分级数&优等数		 fjnum、ydnum
  select t.stat_date,
         t.jm_gc_code,
         t.jm_category_name,
         t.jm_weight,
       sum(fj_qty) as jm_first_fj_qty,  
       sum(yd_qty) as jm_first_yd_qty
  from (
	select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) AS stat_date,  -- 分级时间
			a.jm_gc_code,  -- 光瓷型号编码
			a.jm_np_code,  -- 泥坯编码

			b.jm_weight,
			sum(case when a.jm_uuid is not null then 1 else 0 end ) fj_qty, -- 分级数  -- JM_FJ_NUM
			sum(case when a.jm_fj_grade_frist=1    then 1 else 0 end ) yd_qty -- 优等数  -- JM_YD_NUM  jm_fj_grade_frist 一烧分级等级

	   on a.jm_gc_id=b.jm_uuid
	   where a.jm_is_del=0 
	   and a.JM_IS_BL_COMPLEMENT=0 
	   and a.JM_IS_BM=0   
	   and substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) >= '2022-06'
	   group by --substring(cast(a.jm_fj_date as string),1,10),
	            substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10),
	            a.jm_gc_code,
				a.jm_np_code,
				b.jm_category_name,
				b.jm_weight
   ) t
   group by t.stat_date,
            t.jm_gc_code,
         t.jm_category_name,
         t.jm_weight

jm_sj_data as(   -- 升降级日报表 升级数据		  -- 1 合格/优等品  2 不良品  3  废品

			t.jm_gc_code,

	       3 as jm_bf_grade,  -- 分级前等级

	  on  a.jm_gc_id=b.jm_uuid
