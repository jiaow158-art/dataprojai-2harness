	coalesce(LEAD(99999999- gdatu, 1) OVER (partition by kurst,fcurr,tcurr ORDER BY 99999999- gdatu),47121231) AS end_time,

from sdi.SDI_TCURR_1001
)

SELECT
      a.VBELN
	 ,a.POSNR
	 ,max(e.VMENG) as VMENG

	 ,max(case when e.bmeng>0  then e.edatu end) as max_edatu
FROM SDI.SDI_VBAP_1001 a
inner join(
          SELECT VBELN
		        ,POSNR
				,edatu
				,BMENG
				,plart
		        ,SUM(BMENG) OVER(PARTITION BY VBELN,POSNR ORDER BY edatu rows between unbounded preceding and current row) as VMENG
            FROM SDI.SDI_VBEP_1001
		   WHERE DEL_FLAG = 'N'
         ) e
on a.VBELN = e.VBELN
and a.POSNR = e.POSNR
where a.DEL_FLAG = 'N'
group by
      a.VBELN
	 ,a.POSNR
