--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with 
material_data_01 as(   -- 物料明细数据
	select t.period_id_d as stat_date_d,
		   T3.operation_center_code,
		   T.b1_material_num as material_num,
		   sum(T.qj_pp_del_qty) as INSTOCK_QTY, 		-- 期间采购收货数量
		   sum(t.qj_sale_outstock_qty) as OUTSTOCK_QTY,  -- 期间销售出库数量
		   sum(case when period_id_d = '${PERIOD_ID_D}' then t.qm_qty else 0 end) as qm_qty   -- 期末数量
	FROM DWRCS.DWR_DEA_B1_INVENTORY_SW_T T 
	JOIN  DM_SW.DM_SW_PREDICT_CUST_ORG T3 ON t.CUSTOMER_CODE = T3.cust_num
	where T.period_id_d >= concat(substring(add_months('${PERIOD_ID_D}',-23),1,7),'-01')
	AND T.period_id_d  <= '${PERIOD_ID_D}'   -- 每月1号，传昨天，如：5月1日跑4月30日的数据
	GROUP BY T.period_id_d,
			 T3.operation_center_code ,
			 --T.b1_material_num
			 t.b1_material_num
union ALL 
    select  stat_date_d,
            operation_center_name,
            material_num, 
            INSTOCK_QTY,
            OUTSTOCK_QTY,
            0
    from  dm_sw.DM_WM_CENTER_OUTINSTOCK_T_D T
    where T.stat_date_d >= concat(substring(add_months('${PERIOD_ID_D}',-23),1,7),'-01')
	AND T.stat_date_d  <= '${PERIOD_ID_D}'   -- 每月1号，传昨天，如：5月1日跑4月30日的数据
),
material_data_02 as(
select t1.stat_date_d,
	   t1.operation_center_code,
       t1.material_num,
       round(t1.instock_qty * nvl(t2.material_ratio,1)) as instock_qty,
       round(t1.outstock_qty * nvl(t2.material_ratio,1)) as outstock_qty,
       round(t1.qm_qty * nvl(t2.material_ratio,1)) as qm_qty
from material_data_01 t1
left join DM_SW.DM_SW_PERDICT_BOM_RATIO  t2
on t1.material_num = t2.small_material_num
and t1.operation_center_code = t2.operation_center_code
)
INSERT OVERWRITE TABLE DM_SW.DM_SALES_FORECAST_DISTRIBUTION_DETAIL_TMP PARTITION(stat_month)
select t.stat_date_d,
       t.operation_center_code,
	   t.material_num,
	   sum(nvl(t.INSTOCK_QTY,0)) INSTOCK_QTY,
	   sum(nvl(t.OUTSTOCK_QTY,0)) OUTSTOCK_QTY,
	   sum(nvl(t.qm_qty,0)) as qm_qty,
	   current_timestamp() dw_last_update_date,
	   substring(t.stat_date_d,1,7) as stat_month
from material_data_02 t
 join dwimd.dwi_md_data_material_general_t t1
 on t.material_num = t1.material_num
 group by t.stat_date_d,
		  t.operation_center_code,
		  t.material_num;
