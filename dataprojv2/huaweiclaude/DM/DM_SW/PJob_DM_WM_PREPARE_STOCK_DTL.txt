 --alter table  DM_SW.DM_WM_PREPARE_STOCK_DTL add columns(require_date string comment '客户交期');
 --alter table DM_SW.DM_WM_PREPARE_STOCK_DTL add columns(promise_date string comment '承诺交期');
with trans_money as(
select 
	KURST,FCURR,TCURR,
	99999999- gdatu  as start_time,
	coalesce(LEAD(99999999- gdatu, 1) OVER (partition by KURST,FCURR,TCURR ORDER BY 99999999- gdatu),47121231) AS end_time,
	UKURS 
from sdi.SDI_TCURR_1001
)
,tem as(
SELECT 
	A.MATERIAL_NUM ,
	A.FACTORY_CODE ,
    A.INV_LOCATION ,
    A.BATCH_NUM ,
    A.STOCKOUT_ORDER_NO,
    A.stockout_order_item_no,
    A.WBS_ELEMENT ,
	A.cust_num,
	A.cust_name,
	A.SALES_ORG_CODE , 
    A.SUBJECT_DISTR_TYPE,
	A.PRO_SKU_PREPARE_QTY,
	A.TOTAL_PREPARE_QTY,
	A.ORDER_SOURCE,
	A.REFERENCE_BILL_NUM,
	A.REFERENCE_PROJ_NUM,
	A.prepare_stock_date,
	A.prepare_stock_days,
	A.distribution_channel_code,
	A.creator_id,
    A.res_stock_date,
	A.unit_VOLUM as busi_volume,
-- 	COALESCE(vbap.NETPR*tr.ukurs,vbap.NETPR,cost.unit_price*0.31) as unit_price,
    COALESCE(((sorder.NO_TAX_AMT+sorder.TAX_AMT)*tr.ukurs)/sorder.ORDR_QTY,(sorder.NO_TAX_AMT+sorder.TAX_AMT)/sorder.ORDR_QTY,cost.unit_price*0.31) as unit_price,
	A.TOTAL_PREPARE_QTY*A.unit_VOLUM as TOTAL_PREPARE_VOLUMe,
	A.TOTAL_PREPARE_QTY*COALESCE(((sorder.NO_TAX_AMT+sorder.TAX_AMT)*tr.ukurs)/sorder.ORDR_QTY,(sorder.NO_TAX_AMT+sorder.TAX_AMT)/sorder.ORDR_QTY,cost.unit_price*0.31)  as TOTAL_PREPARE_AMT,
	org.lev2_code as org_lev2_code,
	COALESCE(org.lev2_name,oaorg.short_objid_name_t3) as org_lev2_name,
	org.lev3_code as org_lev3_code,
	COALESCE(org.lev3_name,oaorg.short_objid_name_t4) as org_lev3_name,
	org.lev4_code as org_lev4_code,
	COALESCE(org.lev4_name,oaorg.short_objid_name_t5) as org_lev4_name,
	org.lev5_code as org_lev5_code,
	COALESCE(org.lev5_name,oaorg.short_objid_name_t6) as org_lev5_name,
	CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
	A.require_date,
	A.promise_date,
	zhr.nachn as creator_name
	,stockin_LOCATION
	,stockout_LOCATION
	,stockout_FACTORY_CODE
	,stockin_factory_code
FROM dwrotd.dwr_wm_stock_prepare_from_stockout A
JOIN  dwimd.dwi_md_data_material_general_t mat on  A.material_num = mat.material_num 
LEFT JOIN dm_sw.dm_material_cost cost ON A.material_num  =  cost.material_num 
left join dwrdim.dwr_dim_cust_sales_area_d carea 
on a.cust_num  = carea.cust_num  and a.distribution_channel_code  = carea.distribution_chnl_code 
and a.sales_org_code  = carea.sales_org_code and  carea.product_group = '00' and carea.end_date ='4712-12-31'
left join dm.dm_rpt_sale_grp_t  org On carea.sales_group_code  = org.sale_grp
left join DWIMD.DWI_ORG_PERNR_STRUCTURE_INFO_T oaorg On A.creator_id  = oaorg.employee_code
left join (select distinct pernr,nachn from sdi.sdi_pa0002_1001) zhr on zhr.pernr = A.creator_id 

left join DWROTD.DWR_SAP_ORDER_INFO_T sorder on sorder.order_num = A.REFERENCE_BILL_NUM and sorder.order_item_num = A.REFERENCE_PROJ_NUM
-- left join sdi.sdi_vbap_1001 vbap on vbap.vbeln= A.REFERENCE_BILL_NUM and vbap.posnr=A.REFERENCE_PROJ_NUM
-- LEFT JOIN SDI.sdi_vbak_1001 B ON A.reference_bill_num  = B.vbeln 

left join sdi.sdi_vbkd_1001  as q1 on sorder.order_num = q1.VBELN and q1.POSNR = '000000'
left join trans_money  tr on sorder.CURY_UNIT = tr.fcurr and tr.start_time <= cast(q1.PRSDT as int) and cast(q1.PRSDT as int) < tr.end_time and tr.tcurr = 'RMB'
WHERE A.factory_code LIKE '3%'
AND mat.matl_group_code  in('3000010','4000010')
)
,run as
(
select concat(reference_bill_num,reference_proj_num) id
from tem
where 1=1
and order_source = 'PURCHASE'
and stockout_LOCATION in ('GR','GR90','GD','GD90','GT','GT90','GQ','GA','GO','GK','GB','GZ','GF','GX')
and stockout_FACTORY_CODE in ('3220','3320','3520','3620','3720','3820')
and stockin_LOCATION in ('GR','GR90','GD','GD90','GT','GT90','GQ','GA','GO','GK','GB','GZ','GF','GX','GG','GP')
and stockin_factory_code in ('3220','3320','3520','3620','3720','3820')
)
insert overwrite table DM_SW.DM_WM_PREPARE_STOCK_DTL
select 
	MATERIAL_NUM ,
	FACTORY_CODE ,
    INV_LOCATION ,
    BATCH_NUM ,
    STOCKOUT_ORDER_NO,
    stockout_order_item_no,
    WBS_ELEMENT ,
	cust_num,
	cust_name,
	SALES_ORG_CODE , 
    SUBJECT_DISTR_TYPE,
	PRO_SKU_PREPARE_QTY,
	TOTAL_PREPARE_QTY,
	ORDER_SOURCE,
	REFERENCE_BILL_NUM,
	REFERENCE_PROJ_NUM,
	prepare_stock_date,
	prepare_stock_days,
	distribution_channel_code,
	creator_id,
    res_stock_date,
	busi_volume,
	unit_price,
	TOTAL_PREPARE_VOLUMe,
	TOTAL_PREPARE_AMT,
	org_lev2_code,
	org_lev2_name,
	org_lev3_code,
	org_lev3_name,
	org_lev4_code,
	org_lev4_name,
	org_lev5_code,
	org_lev5_name,
	DW_LAST_UPDATE_DATE,
	require_date,
	promise_date,
	creator_name
from tem t1
left join run t2 on concat(t1.reference_bill_num,t1.reference_proj_num) = t2.id
where 1=1
and t2.id is null
