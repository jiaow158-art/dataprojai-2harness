insert overwrite table  DM_SW.DM_SW_PREDICT_DEF_CHANNEL_D 
SELECT a.cust_num,
       a.sales_org_code,
       a.distribution_chnl_code,
      case when a.distribution_chnl_code = '01' then '经销'
         when a.distribution_chnl_code = '02' then '展厅'    --  
		 when a.distribution_chnl_code = '03' and a.cust_num in('0000003A00') then '艺耐'
		 when a.distribution_chnl_code = '03' and a.cust_num in('0000003900','0000003B00') then '电商'  
		 when a.distribution_chnl_code = '03' then '内部交易'
		 when a.distribution_chnl_code = '05' and b.node_name4='R03230630'  then 'OEM'  
		 when a.distribution_chnl_code = '05' and b.node_name2='H11240102' then '国贸'
		 when a.distribution_chnl_code = '06' and a.cust_group_code in('A6','A7','B0') then '经销工程'   -- 林妙珍提需求改动
		 when a.distribution_chnl_code = '06' then '自营工程'
		  when a.distribution_chnl_code = '08' then '家装'
		   when a.distribution_chnl_code = '10' then '流通'end  as  def_channel
from dwimd.dwi_md_cust_sales_area_t a 
left join  dm.dm_rpt_sales_group_t  b ON a.sales_group_code = b.node_name10
where  a.product_group ='00' 
and  end_date = '4712-12-31';
