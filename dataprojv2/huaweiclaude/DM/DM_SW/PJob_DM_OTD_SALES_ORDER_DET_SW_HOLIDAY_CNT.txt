/*
节假日顺延逻辑
*/

with tem as(
select
	VBAK.VBELN as order_num,  
	VBAP.POSNR as order_item_num, 
	concat(substr(VBAK.AUDAT,1,4),'-',substr(VBAK.AUDAT,5,2),'-',substr(VBAK.AUDAT,7,2))  as order_date,--
	case 
	    --20260122新增准交逻辑
	    --A类
	    when matext.ZX_flag > 0 then 7
	    --B类
	    when matext.ZY_flag_all > 0  and mat.bath_category_name = '浴室柜' then 30
	    when matext.ZY_flag_all > 0  and mat.bath_category_name <> '浴室柜' then 20
	    --C类
	    when matext.zszdtx_flag > 0  and mat.bath_category_name = '大五金' then 45
	    when matext.zszdtx_flag > 0  and mat.bath_category_name <> '大五金' then 40
	    --D类
	    when matext.ts_flag > 0  and mat.bath_category_name = '浴室柜' then 30
	    when matext.ts_flag > 0  and mat.bath_category_name <> '浴室柜' then 50
	    --j10 D类
	    when matext.j10_flag > 0  and mat.bath_category_name = '浴室柜' then 30
	    when matext.j10_flag > 0  and mat.bath_category_name <> '浴室柜' then 50
	    --j9 C类
	    when matext.j9_flag > 0  and mat.bath_category_name = '大五金' then 45
	    when matext.j9_flag > 0  and mat.bath_category_name <> '大五金' then 40
	    --兜底C类
	    when mat.bath_category_name = '大五金' then 45
	    when mat.bath_category_name <> '大五金' then 40
	    
	    --20260122新增准交逻辑
        when tvkbt.bezei in ('苏沪子公司','津冀子公司','贵州子公司')  then 7 --初始下单部门
    	when VBAK.VTWEG = '01' then 
			case 
				when mat.product_level4_code like '4014001%' and VBAP.KWMENG < 5 then 60
	            when mat.bath_category_name ='浴室家居' and matext.zy_flag>0 then 7
	            when mat.bath_category_name = '浴室家居' and (matext.j2_flag>0 or mat.material_name  like '%仿古%') then 45 
	            when mat.bath_category_name = '浴室家居' then 35 
	            when mat.bath_category_name in( '陶瓷','智能') and matext.zy_flag>0  then 7
	            when mat.bath_category_name in( '陶瓷','智能') and matext.zy_flag=0  then 35
	            when mat.bath_category_name in('大五金','小五金') and matext.zy_flag>0 then 7
	            when mat.bath_category_name in('大五金','小五金') and matext.zy_flag=0  then 35 
	            when mat.bath_category_name = '休闲'  then 35
	        end 
	     when VBAK.VTWEG = '06'  then 
	     	case 
		     	when  sw.sales_order_num is not null
	        	then 35 
	        	else 5 
	        end 
	     when VBAK.VTWEG in('02','10','08')  then 35
	     when VBAK.VTWEG in('05','07')  then 35 
	end as restock_ctrl_days
from
	sdi.sdi_VBAK_1001 as VBAK
inner join sdi.sdi_VBAP_1001 as VBAP on VBAK.VBELN = VBAP.VBELN
left join sdi.sdi_vbpa_1001 vbpa On vbak.vbeln = vbpa.vbeln and vbpa.posnr='000000' and vbpa.parvw='VE'
left join  dwimd.dwi_md_data_material_general_t mat on  vbap.matnr = mat.material_num 
left join (	select 
		t.material_num,sales_org_code,distribution_channel_code,t.start_date,t.end_date, 
		sum(case when t.distribution_channel_code = '01' and t.matl_group4_code ='ZY' then 1 else 0 end ) as zy_flag,
		sum(case when t.matl_group2_code ='J9' then 1 else 0 end ) as j9_flag,--如上 判断不用
		sum(case when t.matl_group2_code ='J2' then 1 end ) as j2_flag,
		sum(case when t.matl_group2_code ='J10' then 1 else 0 end ) as j10_flag,
		
		sum(case when  t.matl_group4_code in('ZX') then 1 else 0 end) as ZX_flag, --A
    	sum(case when  t.matl_group4_code in('ZY') then 1 else 0 end) as ZY_flag_all, --B
    	sum(case when  t.matl_group4_code in('ZS','ZD','TX') then 1 else 0 end) as ZSZDTX_flag, --C
    	sum(case when  t.matl_group4_code in('TS') then 1 else 0 end) as TS_flag --D
from dwimd.dwi_md_data_material_sales_t_his  t
group by t.material_num,sales_org_code,distribution_channel_code,t.start_date,t.end_date
) matext ON vbap.matnr = matext.material_num 
and concat(substr(VBAK.AUDAT,1,4),'-',substr(VBAK.AUDAT,5,2),'-',substr(VBAK.AUDAT,7,2))>=matext.start_date 
and concat(substr(VBAK.AUDAT,1,4),'-',substr(VBAK.AUDAT,5,2),'-',substr(VBAK.AUDAT,7,2))<matext.end_date 
left join sdi.sdi_vbkd_1001  as q1 on VBAP.vbeln = q1.VBELN and VBAP.POSNR = q1.POSNR
left join SDI.SDI_OTD_PROJECT_ORDER_NOPREPARE_SW_1037 sw on VBAK.VBELN = sw.sales_order_num and VBAP.POSNR = lpad(sw.sales_order_item_num,6,0) 
left join sdi.sdi_tvkbt_1001 tvkbt on VBAK.vkbur = tvkbt.vkbur
where vbak.vbtyp in('C','J')  
and mat.matl_group_code in('3000010','4000010')
and vbak.audat>='20180101'
)

,dm_otd_sales_order_det_sw as (
select 
	t.order_num
	,t.order_item_num
	,t.order_date
	,t.restock_ctrl_days
	,date_add(t.order_date,restock_ctrl_days) as promise_deliver_date
from tem  t
)

,date_table AS (
	select date_add('2018-01-01', pos) as curr_date
	from (
		select posexplode(split(space(datediff('2030-12-31', '2018-01-01')), ' ')) as (pos, val)
		) t
),

A as (
select a1.order_num  --订单
	,a1.order_item_num --订单行
	,a1.order_date --下单日期
	,a1.restock_ctrl_days --考核日期
from dm_otd_sales_order_det_sw a1
join sdi.sdi_otd_deliver_holiday_cfg_1037 a2 on 1=1
where to_date(a2.holiday_date) between  to_date(a1.order_date) and to_date(a1.promise_deliver_date)
group by a1.order_num  --订单
	,a1.order_item_num --订单行
	,a1.order_date --下单日期
	,a1.restock_ctrl_days --考核日期
),

B as (
select 
	b1.order_num  --订单
	,b1.order_item_num --订单行
	,b1.order_date --下单日期
	,b1.restock_ctrl_days --考核日期
	,b2.curr_date --日历
	,b2.holiday_desc --假期描述
	,sum(b2.date_falg) over (partition by b1.order_num,b1.order_item_num order by b2.curr_date) as num --累计num
	,sum(case when b2.date_falg = 0 then 1 else 1 end) over (partition by b1.order_num,b1.order_item_num order by b2.curr_date) as num1 --假期顺延逻辑num1
from A b1
join (
	select 
		tt1.curr_date,
		to_date(tt2.holiday_date) as holiday_date,
		tt2.holiday_desc,
		case when tt2.holiday_date is null then 1 else 0 end as date_falg
	from date_table tt1
	left join sdi.sdi_otd_deliver_holiday_cfg_1037 tt2 on to_date(tt1.curr_date) = tt2.holiday_date
	 ) b2
on 1=1 
where b2.curr_date between to_date(b1.order_date) and date_add( to_date(b1.order_date),restock_ctrl_days + 100)
)

insert overwrite table dm_sw.dm_otd_sales_order_det_sw_holiday_cnt
select order_num 
 	,order_item_num
 	,curr_date as holiday_date
 	,current_timestamp() as dw_last_update_date
from B
where num <= restock_ctrl_days+1  and holiday_desc is not null
--and order_num = '1112314207' and order_item_num = '000012'
