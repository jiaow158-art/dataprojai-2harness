DELETE FROM dm_sw.dm_sw_rpt_wm_cxc_day_sum
WHERE stat_date = replace('${PERIOD_ID_D}','-','');
insert into dm_sw.dm_sw_rpt_wm_cxc_day_sum
with dz_chose as (
select t.material_num,
	sum(case when  t.matl_group4_code in('DZ') then 1 else 0 end) as dz_flag --DZ定制过滤标识
from dwimd.dwi_md_data_material_sales_t_his  t
join dwimd.dwi_md_data_material_general_t t2 on t.material_num = t2.material_num and t2.matl_group_code in ('3000010','4000010')
where 1=1
and TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD')>= t.start_date
and TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD')< t.end_date
group by t.material_num
)
,mat as (
select t1.material_num,t1.product_level2_name  as cat,t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        t1.width * t1.length_1/1000/1000 as area,
        t1.product_brand_code,t1.product_brand_name,
        t1.product_level2_code,
        t1.busi_volume,
        t1.reserved_field_1,
        case 
            when t1.reserved_field_1 in ('G101','G111','G108') then '零售'
            when t1.reserved_field_1 in ('G102','G112') then '工程'
            when t1.reserved_field_1 in ('G103','G113') then '家装'
            when t1.reserved_field_1 in ('G104','G116') then '流通'
            when t1.reserved_field_1 in ('G107') then '国贸'
            when t1.reserved_field_1 in ('G109OEM') then 'OEM'
            when t1.reserved_field_1 in ('G110') then '电商'
            else t1.reserved_field1_name 
        end reserved_field1_name,
        t1.bathroom_channal,
        t1.matl_group_code,t1.matl_group_name,t1.external_matl_group_code,t1.external_matl_group_name,
        case
	        when t1.product_level3_code in ('4012009') then '小五金'
	        when t1.product_level1_code='30' and t1.product_level2_code in ('3005','3007') then '智能坐便器'
		    when t1.product_level1_code='30' and t1.product_level2_code not in ('3005','3007') then '陶瓷'
		    when t1.product_level2_code in ('4002','4007') and t2.dz_flag <= 0 then '浴室柜（标品）'
		    when t1.product_level2_code in ('4002','4007') and t2.dz_flag > 0 then '浴室柜（特（客）定）'
		    when t1.product_level2_code in ('4004','4005','4012')  then '大五金'
		    when t1.product_level2_code in ('4006','4009','4010','4011','4013','4109','4103')  then '小五金'
		    when t1.product_level2_code in ('4003','4014')  then '休闲类'
        end as category_name,
        basic_calc_unit_name  --物料单位
from dwimd.dwi_md_data_material_general_t t1
left join dz_chose t2 on t1.material_num = t2.material_num
where 1=1 
and (t1.matl_group_code in ('3000010','4000010') or (t1.matl_group_code = '9000010' and external_matl_group_code = '10050'))
and 1=
case 
    when t1.product_level1_code in ('30','40')  then 1
    when t1.product_level1_code in ('41') and t1.product_level2_code in ('4109','4103') then 1 
    else 0
end
)
,all_data as 
(
select 
case 
	when t2.material_source = 'OEM' then '外购'
	when t1.plant = '3220' then '江西基地'
	when t1.plant = '3520' then '江门基地'
	when t1.plant in ('3620','3603') and t3.product_level2_code like '30%' then '江门基地'
    when t1.plant = '3720' then '江门基地'
	when t1.plant in ('3320','3820','3620','3603') then '外购'
	else '其他基地'
end jd_name
,case 
    when t3.category_name = '小五金' and t1.plant = '3520' and coalesce(t2.material_source,'NAL') <> 'OEM' then '大五金'
    else coalesce(t3.category_name,'其他品类')
end category_name
,0 stock_qty_month_end--本月库存
,0 stock_qty_month_start--月初库存
,0 stock_qty_year_start--年初库存
,0 in_stock_qty_d--入库数量 当日
,0 out_stock_qty_d--出库数量 当日
,0 in_stock_qty_m--入库数量 当月
,0 out_stock_qty_m--出库数量 当月
,0 in_stock_qty_y--入库数量 当年
,0 out_stock_qty_y--出库数量 当年
,0 cur_stock_qty --实时库存（件）
,0 cur_stock_volume --实时库存（立方米）
,sum(case when  t1.calday = replace('${PERIOD_ID_D}','-','') then t1.ambperformance else 0 end)/10000 as amount_d --当日业绩
,sum(case when  t1.calday >= replace(concat(substring('${PERIOD_ID_D}',1,6),'-01'),'-','')
	and t1.calday <= replace('${PERIOD_ID_D}','-','') then t1.ambperformance else 0 end)/10000 as amount_m --当月业绩
,sum(case when  t1.calday >= concat(substring('${PERIOD_ID_D}',1,4),'0101')
	and t1.calday <= replace('${PERIOD_ID_D}','-','') then t1.ambperformance else 0 end)/10000 as amount_y --当年业绩
,replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','') as stat_date --分区字段(按天)
from DM.DM_AMBV2_ZZT_XSSR_T t1
left join dm_sw.dm_otd_material_source_t t2 on t1.plant = t2.factory_werks_code and t1.material = t2.material_num
join mat t3 on t1.material = t3.material_num
where 1=1 
and t1.stor_loc in  ('GR','GX','GB','GK','GO','GR90','GR99','GC','GD','GZ','GD90','GQ','GF','GA')
and t1.plant in  ('3220','3520','3620','3603','3320','3820','3720')
and substring(t1.calmonth,1,4) = substring('${PERIOD_ID_D}',1,4)
group by 
case 
	when t2.material_source = 'OEM' then '外购'
	when t1.plant = '3220' then '江西基地'
	when t1.plant = '3520' then '江门基地'
	when t1.plant in ('3620','3603') and t3.product_level2_code like '30%' then '江门基地'
	when t1.plant = '3720' then '江门基地'
	when t1.plant in ('3320','3820','3620','3603') then '外购'
	else '其他基地'
end
,case 
    when t3.category_name = '小五金' and t1.plant = '3520' and coalesce(t2.material_source,'NAL') <> 'OEM' then '大五金'
    else coalesce(t3.category_name,'其他品类')
end 

union all 

select
jd_name
,category_name
,stock_qty_month_end--本月库存
,stock_qty_month_start--月初库存
,stock_qty_year_start--年初库存
,in_stock_qty_d--入库数量 当日
,out_stock_qty_d--出库数量 当日
,in_stock_qty_m--入库数量 当月
,out_stock_qty_m--出库数量 当月
,in_stock_qty_y--入库数量 当年
,out_stock_qty_y--出库数量 当年
,cur_stock_qty --实时库存（件）
,cur_stock_volume --实时库存（立方米）
,0 amount_d
,0 amount_m
,0 amount_y
,stat_date --分区字段(按天)
from dm_sw.dm_sw_rpt_wm_cxc_day_sum_tem 
WHERE stat_date = replace('${PERIOD_ID_D}','-','')
)

select 
jd_name
,category_name
,sum(stock_qty_month_end) stock_qty_month_end--本月库存
,sum(stock_qty_month_start) stock_qty_month_start--月初库存
,sum(stock_qty_year_start) stock_qty_year_start--年初库存
,sum(in_stock_qty_d) in_stock_qty_d--入库数量 当日
,sum(out_stock_qty_d) out_stock_qty_d--出库数量 当日
,sum(in_stock_qty_m) in_stock_qty_m--入库数量 当月
,sum(out_stock_qty_m) out_stock_qty_m--出库数量 当月
,sum(in_stock_qty_y)in_stock_qty_y--入库数量 当年
,sum(out_stock_qty_y) out_stock_qty_y--出库数量 当年
,sum(cur_stock_qty) cur_stock_qty --实时库存（件）
,sum(cur_stock_volume) cur_stock_volume --实时库存（立方米）
,sum(amount_d) amount_d
,sum(amount_m) amount_m
,sum(amount_y) amount_y
,current_timestamp() as dw_last_update_date
,stat_date --分区字段(按天)
from all_data
group by 
jd_name
,category_name
,stat_date
