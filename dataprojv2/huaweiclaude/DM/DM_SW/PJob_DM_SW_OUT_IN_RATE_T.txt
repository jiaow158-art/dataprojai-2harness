with mat as (
select 
 b.material_num
,c.matl_group2_code
,c.matl_group4_code
,k.prod_type
,d.pur_type_code
,b.product_level_code
,b.material_name 
,case when prod_type = '浴室柜' and matl_group4_code <> 'DZ' and ( SUBSTR(product_level_code,1,7) = '4002001'
 or ( SUBSTR(product_level_code,1,7) = '4002005' and material_name like '%主柜%' ) ) and external_matl_group_code <> '10050' then '主浴室柜'  ---物料主档中 物料组4中维护了DZ的为定制组产品，其余为标品（取）,排除艺耐
 	when prod_type = '陶瓷' and pur_type_code = 'E' then '自制陶瓷'
   	when prod_type = '陶瓷' and pur_type_code = 'F' then '外协陶瓷'
   	else prod_type end as category_name
from dwimd.dwi_md_data_material_general_t as b 
left join (select material_num,max(matl_group2_code) matl_group2_code,max(matl_group4_code) as matl_group4_code 
from dwimd.dwi_md_data_material_sales_t group by material_num ) c 
on b.material_num = c.material_num
left join (select material_num ,max(pur_type_code) as pur_type_code 
from dwimd.dwi_md_data_material_factory_t where pur_type_code is null or pur_type_code in ('','E','F') group by material_num ) d 
on b.material_num = d.material_num
inner join SDI.SDI_WM_SW_SAP_PROD_EXT_T_1037  as k on b.product_level3_code = k.prod_code3 
where b.external_matl_group_code <> '10030'
) 
,

tmp as (
   select 
   substr(voucher_post_date,1,7) as DOC_YEAR_MONTH
  ,a.material_num 
  ,m.material_name
  ,m.prod_type
  ,m.product_level_code
  ,m.pur_type_code
  ,m.matl_group2_code
  ,m.matl_group4_code
  ,m.category_name
  ,sum(case when move_type in ('101','911','162','123','131') and (substr(supplier_or_creditor_acct,1,6) <> '000000' or supplier_or_creditor_acct is null ) and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) then quantity  
            when move_type in ('102','912','161','122','132') and (substr(supplier_or_creditor_acct,1,6) <> '000000' or supplier_or_creditor_acct is null ) and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) then -quantity
            else 0 end ) as product_num
  ,sum(case when move_type in ('601','654') then quantity when move_type in('602','653') then -quantity else 0 end ) as sales_num
   from DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F a 
   inner join mat m on a.material_num = m.material_num
   left join (select distinct purchase_voucher_num,purchase_voucher_type from DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F) c on  a.purchase_voucher_num = c.purchase_voucher_num
   where factory_werks_code not like '3A%' and factory_werks_code like '3%' and substr(voucher_post_date,1,4) >= '2021' 
   group by 
   substr(voucher_post_date,1,7)
  ,a.material_num 
  ,m.material_name
  ,m.prod_type
  ,m.product_level_code
  ,m.pur_type_code
  ,m.matl_group2_code
  ,m.matl_group4_code
  ,m.category_name
), 
tmp1 as (
   select
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) as DOC_YEAR
   ,'产总数' as DETAIL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '01' then product_num else 0 end ) as NUM_JAN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '02' then product_num else 0 end ) as NUM_FEB
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '03' then product_num else 0 end ) as NUM_MAR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '04' then product_num else 0 end ) as NUM_APR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '05' then product_num else 0 end ) as NUM_MAY
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '06' then product_num else 0 end ) as NUM_JUN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '07' then product_num else 0 end ) as NUM_JUL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '08' then product_num else 0 end ) as NUM_AUG
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '09' then product_num else 0 end ) as NUM_SEP
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '10' then product_num else 0 end ) as NUM_OCT
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '11' then product_num else 0 end ) as NUM_NOV
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '12' then product_num else 0 end ) as NUM_DEC
   from tmp
--   where material_num in ('WFC0050X' , 'JJH1955D')
   group by 
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) 
   union all 
   select
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) as DOC_YEAR
   ,'销总数' as DETAIL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '01' then sales_num else 0 end ) as NUM_JAN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '02' then sales_num else 0 end ) as NUM_FEB
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '03' then sales_num else 0 end ) as NUM_MAR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '04' then sales_num else 0 end ) as NUM_APR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '05' then sales_num else 0 end ) as NUM_MAY
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '06' then sales_num else 0 end ) as NUM_JUN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '07' then sales_num else 0 end ) as NUM_JUL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '08' then sales_num else 0 end ) as NUM_AUG
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '09' then sales_num else 0 end ) as NUM_SEP
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '10' then sales_num else 0 end ) as NUM_OCT
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '11' then sales_num else 0 end ) as NUM_NOV
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '12' then sales_num else 0 end ) as NUM_DEC
   from tmp
--   where material_num in ('WFC0050X' , 'JJH1955D')
   group by 
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) 
   union all 
   select
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) as DOC_YEAR
   ,'产销率' as DETAIL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '01' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_JAN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '02' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_FEB
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '03' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_MAR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '04' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_APR
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '05' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_MAY
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '06' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_JUN
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '07' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_JUL
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '08' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_AUG
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '09' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_SEP
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '10' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_OCT
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '11' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_NOV
   ,sum(case when substr(DOC_YEAR_MONTH,6,2) = '12' and product_num <> 0 then sales_num / product_num  else 0 end ) as NUM_DEC
   from tmp
   group by 
    material_num
   ,material_name
   ,category_name
   ,substr(DOC_YEAR_MONTH,1,4) 
)
