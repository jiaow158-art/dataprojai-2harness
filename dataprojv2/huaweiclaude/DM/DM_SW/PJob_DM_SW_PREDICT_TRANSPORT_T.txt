--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
with FACT_DETAIL_DATA01 as(   -- 提货分销的明细数据
	select 
	       t1.area_code, -- 大区编码
		   t1.area_name,  -- 大区名称
		   t1.operation_center_code ,   -- 客户编码
		   t1.operation_center_name ,  -- 客户名称
		   t1.material_num,  -- 物料编码
   		   t1.pack_num,  -- 箱容
		   t1.near12_avg_pickup_goods_qty, -- 近12个月月均提货数
		   t1.near12_avg_distribution_qty,  -- 近12个月月均分销
		   t1.b1_lm_qm_stock_qty, -- B1上月期末库存
		   CASE WHEN t1.in_transit_qty < 0 THEN 0 ELSE t1.in_transit_qty END  in_transit_qty,-- N月在途订单
		   t2.not_delivered_qty,
		   --t1.n1_pickup_forecast_suggest_qty, -- N+1月系统提货预测建议值 
		   --t1.n2_pickup_forecast_suggest_qty,-- N+2月系统提货预测建议值
		   --t1.n3_pickup_forecast_suggest_qty,-- N+3月系统提货预测建议值
		   t2.n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
		   t2.n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
		   t2.n3_distribution_forecast_suggest_qty -- N+3月系统分销预测建议值
	from DM_SW.DM_SW_PREDICT_TRANSPORT_PICKUP_T t1  -- 运中提货预测建议
	left join  DM_SW.DM_SW_PREDICT_TRANSPORT_DISTRIBUTION_T t2 -- 运中分销预测建议
	on t1.stat_month = t2.stat_month 
	and t1.operation_center_code = t2.operation_center_code
	and t1.material_num = t2.material_num
	where t1.stat_month = substring('${PERIOD_ID_D}',1,7) -- 每月1号跑上月月末的数据
),
tb_result as (
SELECT b.busi_region,c.product_num,a.nadd1_m_take_qty,a.nadd1_m_distribution_qty from 
sdi.SDI_SALES_FORECAST_ORDER_ITEM_1029  a 
join sdi.SDI_SALES_FORECAST_ORDER_1029 b On a.order_id = b.id    
join sdi.SDI_CENTER_RESULT_1029 c ON a.middle_result_id  = c.id
where b.middle_result_type ='CENTER'
and a.del_flag='0'
and b.status='CHANNEL_APPROVED'
and  b.start_month = substring('${PERIOD_ID_D}',1,7)
),
n1_pickup_forecast as (
select   t2.operation_center_code,
         t2.material_num,
        nvl(t2.b1_lm_qm_stock_qty,0)+nvl(t2.in_transit_qty,0) - nvl(t2.not_delivered_qty,0) + nvl(nadd1_m_take_qty,0) - nvl(nadd1_m_distribution_qty,0)  as n_yj_qm_stock_qty,
        ceil((case when nvl(t2.b1_lm_qm_stock_qty,0)+nvl(t2.in_transit_qty,0) - nvl(t2.not_delivered_qty,0) +  nvl(nadd1_m_take_qty,0) - nvl(nadd1_m_distribution_qty,0) -   nvl(t2.n1_distribution_forecast_suggest_qty,0) >= near12_avg_distribution_qty/30*45 then  0    -- 如果n月预计库存-n+1月预计分销数 大于等于45天日均销，则不补
        else near12_avg_distribution_qty/30*60 - (nvl(t2.b1_lm_qm_stock_qty,0)+nvl(t2.in_transit_qty,0) - nvl(t2.not_delivered_qty,0) +  nvl(nadd1_m_take_qty,0) - nvl(nadd1_m_distribution_qty,0) ) + nvl(t2.n1_distribution_forecast_suggest_qty,0) -- 否则补货数：60天日均销（目标库存天数） - n月预计库存 + n+1月预计分销数 
        end)
              /nvl(t2.pack_num,1))*nvl(t2.pack_num,1) as n1_pickup_forecast_suggest_qty,  -- ceil做箱容校验
          t2.n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
		   t2.n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
		   t2.n3_distribution_forecast_suggest_qty,
		   t2.pack_num,
		   t2.near12_avg_distribution_qty
from  FACT_DETAIL_DATA01 t2 
left join  tb_result t3 ON t2.operation_center_code = t3.busi_region and t2.material_num = t3.product_num
),
n2_pickup_forecast as (
select  t2.operation_center_code,
        t2.material_num,
        n_yj_qm_stock_qty + n1_pickup_forecast_suggest_qty - n1_distribution_forecast_suggest_qty as n1_yj_qm_stock_qty,
       ceil((case when (n_yj_qm_stock_qty + n1_pickup_forecast_suggest_qty - n1_distribution_forecast_suggest_qty) - nvl(t2.n2_distribution_forecast_suggest_qty,0) >= near12_avg_distribution_qty/30*45 then  0    -- 如果n+1月预计库存-n+2月预计分销数 大于等于45天日均销，则不补
        else near12_avg_distribution_qty/30*60 - (n_yj_qm_stock_qty + n1_pickup_forecast_suggest_qty - n1_distribution_forecast_suggest_qty) + nvl(t2.n2_distribution_forecast_suggest_qty,0) -- 否则补货数：60天日均销（目标库存天数） - n+1月预计库存 + n+2月预计分销数 
        end)
              /nvl(t2.pack_num,1))*nvl(t2.pack_num,1) as n2_pickup_forecast_suggest_qty,  -- ceil做箱容校验
          t2.n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
		   t2.n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
		   t2.n3_distribution_forecast_suggest_qty,
		   t2.pack_num,
		   t2.near12_avg_distribution_qty,
		   T2.n1_pickup_forecast_suggest_qty
from n1_pickup_forecast t2 
),
n3_pickup_forecast as (
select t2.operation_center_code,
         t2.material_num,
         n1_yj_qm_stock_qty + n2_pickup_forecast_suggest_qty - n2_distribution_forecast_suggest_qty as n2_yj_qm_stock_qty,
       ceil((case when (n1_yj_qm_stock_qty + n2_pickup_forecast_suggest_qty - n2_distribution_forecast_suggest_qty) - nvl(t2.n3_distribution_forecast_suggest_qty,0) >= near12_avg_distribution_qty/30*45 then  0    -- 如果n+2月预计库存-n+3月预计分销数 大于等于45天日均销，则不补
        else near12_avg_distribution_qty/30*60 - (n1_yj_qm_stock_qty + n2_pickup_forecast_suggest_qty - n2_distribution_forecast_suggest_qty) + nvl(t2.n3_distribution_forecast_suggest_qty,0) -- 否则补货数：60天日均销（目标库存天数） - n+1月预计库存 + n+2月预计分销数 
        end)
              /nvl(t2.pack_num,1))*nvl(t2.pack_num,1) as n3_pickup_forecast_suggest_qty,  -- ceil做箱容校验
          t2.n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
		   t2.n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
		   t2.n3_distribution_forecast_suggest_qty,
		   t2.pack_num,
		   t2.near12_avg_distribution_qty,
		   T2.n1_pickup_forecast_suggest_qty,
		   T2.n2_pickup_forecast_suggest_qty
from n2_pickup_forecast t2 
),
DIM_DATA AS(    -- 获取维度，用于后面字段发散
	select 
		   t.area_code,
		   t.area_name,
		   t.operation_center_code,
		   t.operation_center_name
	from FACT_DETAIL_DATA01 t
	group by
		     t.area_code,
		     t.area_name,
		     t.operation_center_code,
		     t.operation_center_name
),
DIM_DETAIL_DATA as(  -- 数据发散，用型谱的数据按月/物料发散,有型谱无事实数据，指标字段给0
	SELECT 
		   t2.area_code,
		   t2.area_name,
		   t2.operation_center_code,
		   t2.operation_center_name,
		   t1.material_num 
	FROM DWIOTD.DWI_PLAN_PRODUCT_SPECTRUM_T T1
	JOIN (SELECT DISTINCT t1.MATERIAL_NUM FROM DWIMD.dwi_md_data_material_SALES_t t1 
	     LEFT JOIN sdi.SDI_ZTMM004_1001 t2 ON t1.material_num = t2.matnr and t1.distribution_channel_code = t2.vtweg and t2.flag='X'
	      join  DWIMD.dwi_md_data_material_general_t t3 ON t1.material_num = t3.material_num
	      WHERE t1.distribution_channel_code IN('01') 
	      AND (not t1.matl_group3_code in('PJ'))
          AND ( t1.matl_group4_code NOT IN('DZ','DJ','TZ','TS',''))
          AND (not t3.matl_texture in('A001','A002'))
          AND ((t3.matl_type_code ='FERT' AND coalesce(t1.matl_group2_code,'') IN('J9','J11')) or (t3.matl_type_code ='HALB' AND coalesce(t1.matl_group2_code,'') IN('J9')))
          AND t3.product_level1_code IN('30','40','31','41')
          AND t3.reserved_field_1 IN('G101','G111','G112','G113','G116','G108','G114','G116')
          AND (not t3.x_dstrb_c_mm_status_code='GG')
          AND (not t3.X_plant_matl_status='99')
	      AND t2.matnr is null
	) t3 ON T1.MATERIAL_NUM = T3.MATERIAL_NUM  -- 在01可分销
	LEFT JOIN DIM_DATA T2
	where t1.stat_month = substring(add_months('${PERIOD_ID_D}',2),1,7)   -- 传5月，获取7月型谱的数据
),
material_sales_data as(   -- 物料销售数据
   SELECT T1.material_num,
	   T1.material_name,
	   t2.prod_line_name ,    -- 营销大类
	   t2.second_level_name   -- 营销小类
	FROM DWIMD.dwi_md_data_material_general_t t1
	left join DWIMD.DWI_MD_PRODUCT_LEVEL_APS_T t2
	on case when NVL(t1.product_level3_code,'NULL') <> NVL(t2.prod_level_code,'NULL') then t1.product_level2_code = t2.prod_level_code 
	else t1.product_level3_code = t2.prod_level_code end
),
price_data as(   -- 产品价格数据  
	select /*case when t1.konda = 'JK' then '零售' 
            	when t1.konda = 'JL' then '流通'
            	when t1.konda = 'JE' then '鹏嘉'
            else '' end channnel_name,*/
	       --T1.mandt as group_code,  -- 集团
		   --t1.konda as price_group_code, -- 价格组
		   t1.matnr as material_num, -- 物料编码
		   MAX(CASE WHEN t1.konda='JK' THEN t2.KBETR/t2.KPEIN  ELSE 0 END) as unit_price, -- 单价
		   MAX(CASE WHEN t1.konda='JM' THEN t2.KBETR/t2.KPEIN  ELSE 0 END) as dist_price  -- 分销价
	from sdi.sdi_a963_1001 t1
	LEFT JOIN SDI.sdi_konp_1001 t2
	on t1.knumh = t2.knumh  -- 条件记录号
	and t1.kschl = t2.kschl  
	where t1.mandt = '800' -- 集团
	and t1.kschl = 'PR00' -- 条件类型
	and t1.konda IN('JK', 'JM')-- in ('JK','JE','JL') -- 客户价格组
	and REPLACE(CAST('${PERIOD_ID_D}' AS STRING),'-','') BETWEEN t1.datab  and t1.datbi 
	and t2.kopos = '01' -- 条件的序列号
	GROUP BY t1.matnr
),
sales_code_data as(   -- 销售代码及产品状态数据
	SELECT T2.code as material_num,  -- 物料编码
	       T2.name as material_name,  -- 物料名称
	       T1.product_line,			  -- 产品线
	       T1.series ,				  -- 系列
	       T1.code as sales_code,	  -- 销售代码
	       T3.PROD_STATUS_NAME  	  -- 产品状态
	FROM SDI.SDI_WYAPS_SALES_CODE_MAINTAIN_1007 T1
	JOIN SDI.SDI_ITEM_1007 T2
	ON T1.item_id = T2.ID
	left join (
	   select t1.code AS PROD_STATUS_CODE,
			   t1.name AS PROD_STATUS_NAME
		from SDI.SDI_BD_CATALOG_1007 t1
		join SDI.SDI_BD_CATALOG_TYPE_1007 t2
		on t1.catalog_type_id = t2.id
		where t2.CODE = 'PRODUCT_STATE'
		and t1.inv_org_id = '2'
		and t1.is_phantom = '0'
	) T3
	ON T1.product_state = T3.PROD_STATUS_CODE
	WHERE T1.inv_org_id = '2'   -- 库存组织
	and T1.is_phantom = '0'     -- 幽灵属性
)
---alter table  DM_SW.DM_SW_PREDICT_TRANSPORT_T  add columns( dist_price decimal(38,10))
insert overwrite table dm_sw.dm_sw_predict_transport_t partition(stat_month)
select CONCAT(replace( substring(add_months('${PERIOD_ID_D}',1),1,7) ,'-',''),ROW_NUMBER() OVER()) SEQ_NO, -- 数据同步ID
	   t1.operation_center_code,  -- 
       t1.operation_center_name, -- 
       '01' as channnel_num, -- 渠道编码
       '经销' as channnel_name, -- 渠道名称
       t1.area_code,   -- 大区编码
       t1.area_name,   -- 大区名称
       t1.material_num , -- 物料编码
       t3.material_name,  -- 物料名称
       t5.sales_code,	  -- 销售代码
	   t5.PROD_STATUS_NAME , 	  -- 产品状态
       t3.prod_line_name , -- 营销大类
	   t3.second_level_name ,  -- 营销小类
       nvl(t7.pack_num,1) as pack_num,  -- 箱容
       nvl(t4.unit_price,0) as unit_price,  -- 产品单价
	   nvl(t2.near12_avg_pickup_goods_qty,0) as near12_avg_pickup_goods_qty, -- 近12个月月均提货数
	   nvl(t2.near12_avg_distribution_qty,0) as near12_avg_distribution_qty,  -- 近12个月月均分销
	   nvl(t2.b1_lm_qm_stock_qty,0) as b1_lm_qm_stock_qty, -- B1上月期末库存
	   nvl(t2.in_transit_qty,0) as in_transit_qty, -- N月在途订单
	   nvl(t6.n1_pickup_forecast_suggest_qty,0) as n1_pickup_forecast_suggest_qty, -- N+1月系统提货预测建议值 
	   nvl(t2.n1_distribution_forecast_suggest_qty,0) as n1_distribution_forecast_suggest_qty, -- N+1月系统分销预测建议值
	   nvl(t6.n2_pickup_forecast_suggest_qty,0) as n2_pickup_forecast_suggest_qty,-- N+2月系统提货预测建议值
	   nvl(t2.n2_distribution_forecast_suggest_qty,0) as n2_distribution_forecast_suggest_qty, -- N+2月系统分销预测建议值
	   nvl(t6.n3_pickup_forecast_suggest_qty,0)  as n3_pickup_forecast_suggest_qty,-- N+3月系统提货预测建议值
	   nvl(t2.n3_distribution_forecast_suggest_qty,0) as n3_distribution_forecast_suggest_qty, -- N+3月系统分销预测建议值
	   current_timestamp() as dw_last_update_date, -- 数据同步时间
	   nvl(t4.dist_price,0) as dist_price,  
	   substring(add_months('${PERIOD_ID_D}',1),1,7) 
from DIM_DETAIL_DATA t1
left join FACT_DETAIL_DATA01 t2
on  t1.operation_center_code = t2.operation_center_code
and t1.material_num = t2.material_num
LEFT JOIN material_sales_data t3   -- 物料
on t1.material_num =  t3.material_num
LEFT JOIN price_data t4        -- 产品数据
on T1.material_num = t4.material_num
LEFT JOIN sales_code_data t5   -- 销售组织数据
on T1.material_num = t5.material_num
left join n3_pickup_forecast t6 ON t1.operation_center_code = t6.operation_center_code and t1.material_num = t6.material_num 
left join (select product_code AS material_num,max(pack_num) as pack_num from SDI.SDI_ZTMM001_1001 group by product_code)  t7 ON t1.material_num = t7.material_num;
