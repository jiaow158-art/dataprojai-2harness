with operation_center_data as(  -- 客户数据
     select case when length(cust_code) <=6 then concat('0000',cust_code) else cust_code end as cust_code  ,
        abbreviation as cust_short_name
     from sdi.sdi_b1_customer_data_1037
     where cust_type = 'SW'
),
dim_date as(
	select distinct day_mend_dt 
	from dm.dm_dim_date_d
	where day_dt >= '2023-01-01'
	and day_dt < CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01')   -- 传昨天，小于当月1号
	union all
	select '${PERIOD_ID_D}'  -- 传昨天
),
DWR_WM_CUST_PICK_DATA_F_DATA AS(
    SELECT t1.*
	from DWROTD.DWR_WM_CUST_PICK_DATA_F t1
	join dim_date t4
	on t1.pickup_goods_date = t4.day_mend_dt
),
DWR_WM_CUST_INSTOCK_DATA_F_DATA AS(
    SELECT t1.*
	from DWROTD.DWR_WM_CUST_INSTOCK_DATA_F t1
	join dim_date t4
	on t1.in_stock_date = t4.day_mend_dt
),
hn_fj_jx_gx_data as(  -- 湖南江西广西福建数据
	select t1.cust_code, -- 客编
		   t1.cust_code as cust_short_name,  -- 客编简称
		   '' as sap_delivery_num, -- SAP提货单号
		   t1.material_num,         -- 物料
		   t1.begin_pickup_goods_date AS pickup_goods_date,    -- 总部出库时间
		   t2.begin_in_stock_date AS in_stock_date,     -- 运营中心入库时间
		   NVL(t1.pickup_goods_qty,0) AS pickup_goods_qty,            -- 总部出库数量
		   NVL(t2.in_stock_qty,0) AS in_stock_qty,				--运营中心入库数量
		   NVL(t1.pickup_goods_qty,0) - NVL(t2.in_stock_qty,0) as in_transit_qty,  -- 在途数量
		   current_timestamp() as dw_last_update_date, -- 最后同步数据时间,
		   (NVL(t1.pickup_goods_amt,0) / NVL(t1.pickup_goods_qty,0)) * (NVL(t1.pickup_goods_qty,0) - NVL(t2.in_stock_qty,0)) as  in_transit_amt,  -- 在途金额
		   t2.return_date,
		   NVL(t2.return_qty,0) AS return_qty,
		   t1.stat_month AS STAT_MONTH 
	from DWR_WM_CUST_PICK_DATA_F_DATA t1
	left join DWR_WM_CUST_INSTOCK_DATA_F_DATA t2
	on t1.pickup_goods_date = t2.in_stock_date
	and t1.cust_code = t2.cust_code
	and t1.material_num = t2.material_num
	--join operation_center_data t3
	--on t1.cust_code = t3.cust_short_name --cust_code
	WHERE t1.cust_code in ('湖南锦辉铭','广西升钻','福建万艺祥','江西高时')
	--in ('0000310635','0000353879','0000321104','0000318683')
	--and t1.pickup_goods_date >= ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3) --'2023-01-01'
),
no_hn_fj_jx_gx_data as(  -- 非 湖南江西广西福建数据
	select t1.cust_code, -- 客编
		   t1.cust_code as cust_short_name,  -- 客编简称
		   t1.sap_delivery_num, -- SAP提货单号
		   t1.material_num,         -- 物料
		   t1.begin_pickup_goods_date pickup_goods_date,    -- 总部出库时间
		   t2.begin_in_stock_date as in_stock_date,     -- 运营中心入库时间
		   NVL(t1.pickup_goods_qty,0) AS pickup_goods_qty,            -- 总部出库数量
		   NVL(t2.in_stock_qty,0) AS in_stock_qty,				--运营中心入库数量
		   NVL(t1.pickup_goods_qty,0) - NVL(t2.in_stock_qty,0) as in_transit_qty,  -- 在途数量
		   current_timestamp() as dw_last_update_date, -- 最后同步数据时间
		   (NVL(t1.pickup_goods_amt,0) / NVL(t1.pickup_goods_qty,0)) * (NVL(t1.pickup_goods_qty,0) - NVL(t2.in_stock_qty,0)) as  in_transit_amt,  -- 在途金额
		   t2.return_date,
		   NVL(t2.return_qty,0) AS return_qty,
		   t1.stat_month AS STAT_MONTH 
	from DWR_WM_CUST_PICK_DATA_F_DATA t1
	left join DWR_WM_CUST_INSTOCK_DATA_F_DATA t2
	on t1.pickup_goods_date = t2.in_stock_date
	and t1.cust_code = t2.cust_code
	and t1.sap_delivery_num = t2.sap_delivery_num 
	and t1.material_num = t2.material_num
	--join operation_center_data t3
	--on t1.cust_code = t3.cust_short_name --cust_code
	WHERE t1.cust_code not in ('湖南锦辉铭','广西升钻','福建万艺祥','江西高时')
	--not in ('0000310635','0000353879','0000321104','0000318683')
	--and t1.pickup_goods_date >= ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3) -- '2023-01-01'
),
tmp_data as(
	select t.*
	from hn_fj_jx_gx_data t
	where abs(pickup_goods_qty) + abs(in_stock_qty) + abs(in_transit_qty) <> 0
	union all
	select t.*
	from no_hn_fj_jx_gx_data t
	where abs(pickup_goods_qty) + abs(in_stock_qty) + abs(in_transit_qty) <> 0
)
INSERT OVERWRITE TABLE DM_SW.DM_SO_TRANSIT_INVENTORY_T PARTITION(STAT_MONTH)
select cust_code, 
	   cust_short_name, 
	   sap_delivery_num, 
	   material_num, 
	   pickup_goods_date, 
	   case when in_stock_qty <= 0 then null else in_stock_date end as in_stock_date, 
	   pickup_goods_qty, 
	   in_stock_qty, 
	   in_transit_qty, 
	   dw_last_update_date, 
	   in_transit_amt, 
	   case when return_qty > 0 then COALESCE(return_date,in_stock_date) else null end as return_date,
	   return_qty,
	   stat_month
from tmp_data t1
where t1.STAT_MONTH >= CASE WHEN substring('${PERIOD_ID_D}',-2) IN ('01','02','03') THEN 
                               SUBSTRING(ADD_MONTHS(CONCAT(SUBSTRING('${PERIOD_ID_D}',1,7),'-01'),-3),1,7) -- 传昨天，所在月份1号的前3个月
                           ELSE
                    		   SUBSTRING('${PERIOD_ID_D}',1,7)  -- 传昨天，当月
                       END;
