set hive.cbo.enable=true;

INSERT OVERWRITE TABLE  DM_SW.DM_SW_PREDICT_CUST_ORG 
select t1.org_code as cust_num,  
       t2.org_code as operation_center_code ,
       t2.org_name as operation_center_name,
       t3.org_code as area_code,
       t3.org_name as area_name
from sdi.SDI_MAIN_ORG_1029 t1 
left join sdi.SDI_MAIN_ORG_1029 t2 ON t1.parent_id = t2.id  and t2.del_flag=0  and t2.is_freeze = 'N'-- 运中
left join sdi.SDI_MAIN_ORG_1029 t3 ON t2.parent_id = t3.id   and t3.del_flag=0  and t3.is_freeze = 'N' --- 大区
where t1.level = 6
and t1.del_flag=0
and t1.is_freeze = 'N'
union all      
select null as cust_num,  
       t1.org_code as operation_center_code ,
       t1.org_name as operation_center_name,
       t2.org_code as area_code,
       t2.org_name as area_name
from sdi.SDI_MAIN_ORG_1029 t1 
left join sdi.SDI_MAIN_ORG_1029 t2 ON t1.parent_id = t2.id  and t2.del_flag=0 and t2.is_freeze = 'N' -- 运中
where t1.level = 5
and t1.del_flag=0
and t2.org_code = '子公司'
and t1.is_freeze = 'N'
;
