with detail_data_01 as(   -- 成本还原明细数据
	SELECT NVL(T1.STAT_MONTH,T2.prod_date) AS STAT_MONTH,     -- 月份
		   NVL(T1.factory_code,T2.factory_code) AS FACTORY_CODE, -- 工厂
		   NVL(T1.material_num,T2.material_num) AS MATERIAL_NUM, --物料编码
		   nvl(t2.line_name,'普通线') AS line_name, -- 高压线&普通线
		   NVL(T1.bz_cost_qty,0) AS bz_cost_qty,  -- 数量
		   NVL(T1.bz_cost_price,0) AS bz_cost_price,  -- 单价
			NVL(T1.bz_total_amt,0) AS bz_total_amt,  -- 合计
			NVL(T1.bz_cl_ns_amt,0) AS bz_cl_ns_amt, -- 材料 - 沙泥 
			NVL(T1.bz_cl_hg_amt,0) AS bz_cl_hg_amt, -- 材料 - 化工
			NVL(T1.bz_cl_pj_amt,0) AS bz_cl_pj_amt, -- 材料－配件
			NVL(T1.bz_cl_bz_amt,0) AS bz_cl_bz_amt, -- 材料 -- 包装
			NVL(T1.bz_cl_qtcl_amt,0) AS bz_cl_qtcl_amt,-- 材料－其他材料

			NVL(T1.bz_ccp_amt,0) AS bz_ccp_amt,-- 产成品
			NVL(T1.bz_rgf_amt,0) AS bz_rgf_amt,-- 人工费
			NVL(T1.bz_lq_m_amt,0) AS bz_lq_m_amt,  -- 燃料-煤
			NVL(T1.bz_lq_q_amt,0) AS bz_lq_q_amt, -- 燃料-气
			NVL(T1.bz_df_amt,0) AS bz_df_amt, -- 电费
			NVL(T1.bz_zj_tx_amt,0) AS bz_zj_tx_amt,  -- 折旧及摊销
			NVL(T1.bz_wwjg_pgf_amt,0) AS bz_wwjg_pgf_amt, -- 委外加工/抛光费
			NVL(T1.bz_sf_amt,0) AS bz_sf_amt,  -- 水费
			NVL(T1.bz_wlxh_amt,0) AS bz_wlxh_amt, -- 物料消耗
			NVL(T1.bz_wxf_amt,0) AS bz_wxf_amt, -- 维修费
			NVL(T1.bz_cy_amt,0) AS bz_cy_amt, -- 柴油
			NVL(T1.bz_wscl_amt,0) AS bz_wscl_amt, -- 污水处理
			NVL(T1.bz_qtfy_amt,0) AS bz_qtfy_amt, -- 其他费用
			NVL(T1.sj_cost_qty,0) AS sj_cost_qty, -- 数量
			NVL(T1.sj_cost_price,0) AS sj_cost_price, -- 单价
			NVL(T1.sj_total_amt,0) AS sj_total_amt, -- 合计
			NVL(T1.sj_cl_ns_amt,0) AS sj_cl_ns_amt, -- 材料 - 沙泥
			NVL(T1.sj_cl_hg_amt,0) AS sj_cl_hg_amt, -- 材料 - 化工
			NVL(T1.sj_cl_pj_amt,0) AS sj_cl_pj_amt, -- 材料－配件
			NVL(T1.sj_cl_bz_amt,0) AS sj_cl_bz_amt, -- 材料 -- 包装
			NVL(T1.sj_cl_qtcl_amt,0) AS sj_cl_qtcl_amt, -- 材料－其他材料
			NVL(T1.sj_bcp_amt,0) AS sj_bcp_amt, -- 半成品
			NVL(T1.sj_ccp_amt,0) AS sj_ccp_amt, -- 产成品
			NVL(T1.sj_rgf_amt,0) AS sj_rgf_amt, -- 人工费
			NVL(T1.sj_rq_m_amt,0) AS sj_lq_m_amt, -- 燃料-煤
			NVL(T1.sj_rq_q_amt,0) AS sj_lq_q_amt, -- 燃料-气
			NVL(T1.sj_df_amt,0) AS sj_df_amt, -- 电费
			NVL(T1.sj_zjtx_amt,0) AS sj_zjtx_amt, -- 折旧及摊销
			NVL(T1.sj_wyjg_amt,0) AS sj_wyjg_amt, -- 委外加工/抛光费
			NVL(T1.sj_sf_amt,0) AS sj_sf_amt, -- 水费
			NVL(T1.sj_wlxh_amt,0) AS sj_wlxh_amt, -- 物料消耗
			NVL(T1.sj_wyf_amt,0) AS sj_wyf_amt, -- 维修费
			NVL(T1.sj_cy_amt,0) AS sj_cy_amt, -- 柴油
			NVL(T1.sj_wscl_amt,0) AS sj_wscl_amt, -- 污水处理
			NVL(T1.sj_qtfy_amt,0) AS sj_qtfy_amt, -- 其他费用
			NVL(T2.zj_qty,0) AS zj_qty,  -- 注浆数
			NVL(T2.ss_qty,0) AS ss_qty,  -- 实收数
			NVL(T2.ysfj_qty,0) AS ysfj_qty, -- 一烧分级数
			NVL(T2.ysyd_qty,0) AS ysyd_qty, -- 一烧优等数
			NVL(T2.csyd_qty,0) AS csyd_qty, -- 重烧优等数
			NVL(T2.sy_jj_qty,0) AS sy_jj_qty,  -- 升降数
			NVL(T2.py_qty,0) AS py_qty -- 喷釉数
	FROM DWIOTD.DWI_PAM_COST_REDUCTION_SAP_T T1  -- 成本还原SAP部分(标准成本&实际成本)
	FULL OUTER JOIN DWIOTD.DWI_PAM_COST_REDUCTION_BCS_T T2 -- 成本还原BCS部分(收呸率%综合合格率&当年投入产出比)
	ON NVL(T1.STAT_MONTH,'NULL') = NVL(T2.prod_date,'NULL')
	AND NVL(T1.factory_code,'NULL') = NVL(T2.factory_code,'NULL')
	AND NVL(T1.material_num,'NULL') = NVL(T2.material_num,'NULL')
)
INSERT OVERWRITE TABLE DM_SW.DM_OTD_PAM_COST_REDUCTION_T PARTITION(STAT_MONTH)
SELECT T1.factory_code, -- 工厂
       T1.material_num, -- 物料编码
       T2.material_NAME, -- 物料名称
       T2.product_level3_name AS category_name, -- 品类
       T1.line_name,-- 普通线&高压线
       T2.gross_weight, -- 单重
       nvl(t1.bz_cost_qty,t1.sj_cost_qty) as cost_qty, -- 产量
       --NVL(nvl(t1.bz_cost_qty,t1.sj_cost_qty) * T2.gross_weight,0) as prod_weight_qty, -- 产重
       NVL(t3.base_spi_rate,0) AS base_spi_rate, --收坯率(标准)   
       NVL(t1.zj_qty,0) as zj_qty, -- 注浆数
       NVL(t1.ss_qty,0) as ss_qty, -- 实收数
       --NVL(t1.zj_qty / nvl(t1.ss_qty,1),0) as actual_spi_rate,  --收坯率(实际)
       NVL(t3.base_comp_hg_rate,0) AS base_comp_hg_rate, -- 综合合格率(标准)
       NVL(t1.ysyd_qty,0) AS ysyd_qty, -- 一烧优等
       NVL(t1.csyd_qty,0) AS csyd_qty, -- 重烧优等
       NVL(t1.sy_jj_qty,0) AS sy_jj_qty, -- 升降级数
       NVL(t1.ysfj_qty,0) AS ysfj_qty, -- 一烧分级
       --NVL((t1.ysyd_qty + t1.csyd_qty + t1.sy_jj_qty) / nvl(t1.ysfj_qty,1),0) as actual_comp_hg_rate, -- 综合合格率(实际)
        NVL(T4.jx_ysfj_qty,0) as  jx_ysfj_qty,    -- 江西一烧分级
		NVL(T4.jx_actual_receipt_qty,0) as jx_actual_receipt_qty,-- 江西实收数
		NVL(T4.jx_pc_qc_qty,0) as jx_pc_qc_qty,  -- 江西坯仓期初数
		NVL(T4.jx_sy_qc_qty,0) as jx_sy_qc_qty,  -- 江西试釉期初数
		NVL(T4.jx_sy_np_qc_qty,0) as jx_sy_np_qc_qty, -- 江西试釉泥坯期初数
		NVL(T4.jx_sc_qc_qty,0) as  jx_sc_qc_qty, -- 江西烧成期初数
		NVL(T4.jx_pc_pd_qty,0) as jx_pc_pd_qty,-- 江西坯仓盘点数
		NVL(T4.jx_sy_pd_qty,0) as jx_sy_pd_qty,-- 江西试釉盘点数
		NVL(T4.jx_sy_np_pd_qty,0) as jx_sy_np_pd_qty,-- 江西试釉泥坯盘点数
		NVL(T4.jx_sc_pd_qty,0) as jx_sc_pd_qty,	-- 江西烧成盘点数
		NVL(T4.jm_qp_hg_qty,0) as jm_qp_hg_qty, -- 江门青坯合格数
		NVL(T4.jm_ysfj_qty,0) as jm_ysfj_qty, -- 江门一烧分级数
       --NVL(t4.gcsh_qty,0) as base_Process_loss_qty, -- 过程损耗
		NVL(T1.PY_QTY,0) as PY_QTY, --  喷釉数
       --NVL(T1.PY_QTY / (T1.ysyd_qty + T1.csyd_qty),0)  as input_prod_ratio, -- 当年投入产占比
       t1.bz_cost_price, -- 标准成本(产品成本)
       NVL((t1.bz_total_amt - t1.bz_cl_pj_amt - t1.bz_cl_bz_amt) / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_gc_unit_cost, --标准成本(光瓷单位成本)
       NVL(t1.bz_cl_ns_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_np_unit_cost, -- 标准成本(泥坯单位成本)
       0 as bz_ns_dh,-- 标准成本(泥沙-单耗)
       NVL(t1.bz_cl_ns_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_ns_unit, -- 标准成本(泥沙-单价)
       0 as bz_hg_dh, -- 标准成本(化工-单耗)
       NVL(t1.bz_cl_hg_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_hg_unit, -- 标准成本(化工-单价)
       NVL(t1.bz_rgf_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_rg_unit, -- 标准成本(人工单价)
       NVL((t1.bz_lq_m_amt + t1.bz_lq_q_amt) / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_lq_q_unit, -- 标准成本(气价)
       NVL((t1.bz_total_amt - t1.bz_cl_pj_amt - t1.bz_cl_bz_amt) / (nvl(t1.bz_cost_qty,t1.sj_cost_qty) * T2.gross_weight),0) as bz_gjc_unit_cost, --标准成本(公斤瓷单位成本) 
       NVL(t1.bz_cl_bz_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_bz_cost, -- 标准成本(包装成本)
       NVL(t1.bz_cl_pj_amt / nvl(t1.bz_cost_qty,t1.sj_cost_qty),0) as bz_pj_cost, -- 标准成本(配件成本)
       t1.bz_cost_qty, -- 标准成本
       t1.sj_cost_qty, -- 实际成本
       t1.sj_cost_price, -- 实际成本(产品成本)
       --NVL((t1.sj_total_amt - t1.sj_cl_pj_amt - t1.sj_cl_bz_amt) / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_gc_unit_cost, --实际成本(光瓷单位成本)
       --NVL(t1.sj_cl_ns_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_np_unit_cost, -- 实际成本(泥坯单位成本)
       0 as sj_ns_dh,-- 实际成本(泥沙-单耗)
       --NVL(t1.sj_cl_ns_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_ns_unit, -- 实际成本(泥沙-单价)
       0 as sj_hg_dh, -- 实际成本(化工-单耗)
       --NVL(t1.sj_cl_hg_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_hg_unit, -- 实际成本(化工-单价)
       --NVL(t1.sj_rgf_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_rg_unit, -- 实际成本(人工单价)
       --NVL(t1.sj_lq_q_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_lq_q_unit, -- 实际成本(气价)
       --NVL((t1.sj_total_amt - t1.sj_cl_pj_amt - t1.sj_cl_bz_amt) / (nvl(t1.sj_cost_qty,t1.bz_cost_qty) * T2.gross_weight),0) as sj_gjc_unit_cost, --实际成本(公斤瓷单位成本) 
       --NVL(t1.sj_cl_bz_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_bz_cost, -- 实际成本(包装成本)
       --NVL(t1.sj_cl_pj_amt / nvl(t1.sj_cost_qty,t1.bz_cost_qty),0) as sj_pj_cost, -- 实际成本(配件成本)
        T1.sj_total_amt, -- 合计
		T1.sj_cl_ns_amt, -- 材料 - 沙泥
		T1.sj_cl_hg_amt, -- 材料 - 化工
		T1.sj_cl_pj_amt, -- 材料－配件
		T1.sj_cl_bz_amt, -- 材料 -- 包装
		T1.sj_cl_qtcl_amt, -- 材料－其他材料
		T1.sj_bcp_amt, -- 半成品
		T1.sj_ccp_amt, -- 产成品
		T1.sj_rgf_amt, -- 人工费
		T1.sj_lq_m_amt, -- 燃料-煤
		T1.sj_lq_q_amt, -- 燃料-气
		T1.sj_df_amt, -- 电费
		T1.sj_zjtx_amt, -- 折旧及摊销
		T1.sj_wyjg_amt, -- 委外加工/抛光费
		T1.sj_sf_amt, -- 水费
		T1.sj_wlxh_amt, -- 物料消耗
		T1.sj_wyf_amt, -- 维修费
		T1.sj_cy_amt, -- 柴油
		T1.sj_wscl_amt, -- 污水处理
		T1.sj_qtfy_amt, -- 其他费用
       current_timestamp() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
       T2.matl_type_name, -- 物料类型名称
       T1.STAT_MONTH  --月份
FROM detail_data_01 T1
JOIN DWIMD.dwi_md_data_material_general_t T2 -- 物料主数据
ON T1.material_num = T2.material_num 
LEFT JOIN DWIOTD.DWI_PAM_BASE_STANDARD_VALUE_T T3 -- 基地标准产值损指标
ON T1.factory_code  = T3.factory_code
AND T1.material_num = T3.material_num
AND T1.STAT_MONTH = T3.STAT_MONTH
LEFT JOIN DWIOTD.DWI_PAM_COST_GCSH_T T4 -- 过程损耗数据
ON T1.factory_code  = T4.factory_code
AND T1.material_num = T4.material_num
AND T1.STAT_MONTH = T4.prod_date
where T1.STAT_MONTH = SUBSTRING('${PERIOD_ID_D}',1,7);
