insert overwrite table dm_sw.dm_ptp_purchase_transit_t  
select purchase_order_num,
purchase_item_num,
purchase_order_type,
comp_code,
purchase_org_code,
purchase_group_code,
purchase_order_status,
purchase_voucher_date,
order_create_date,
t1.ernam,
t1.supplier_num,
t1.material_num,
t1.inv_location,
t1.factory_code,
item_txt,
purchase_qty,
stockin_qty,
transit_qty,
purchaser,
apply_obj,
approved,
MIN_DELV_DATE,
MAX_DELV_DATE,
case when purchase_order_type='NB' then '标准采购订单'  
     when purchase_order_type='ZNB1' then 'OEM采购订单'
     when purchase_order_type='ZNB2' then '委外加工采购订单' end purchase_order_type_t,
T4.supplier_or_creditor_name AS supplier_name,
t2.product_level2_name,
t1.purchase_order_txt
from dwrpur.dwr_ptp_purchase_transit_f t1 
join dwimd.dwi_md_data_material_general_t t2 
on t1.material_num = t2.material_num
join  CFG  t3 ON t1.factory_code = t3.werks and t1.inv_location = t3.lgort
LEFT JOIN (select material_num,
                  MAX(CASE WHEN T.matl_group4_code IN('DJ' ,'TX' ,'TS' ,'ZS' ,'ZX' ,'ZY','TD', 'ZD','DZ') THEN 1 ELSE 0 END) AS DZ1,
                  MAX(CASE WHEN T.matl_group3_code ='PJ' THEN 1 ELSE 0 END) AS PJ1
           from DWIMD.dwi_md_data_material_sales_t T 
           GROUP BY material_num
          )  T9 ON T1.material_num = T9.material_num
LEFT JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T T4 ON T1.supplier_num = T4.supplier_or_creditor_num AND T4.end_date='4712-12-31'
where   
 ((T2.matl_group_code in('3000010','4000010') and T2.external_matl_group_code = '10010') or  
 (T2.matl_group_code in('3000010','4000010') and T2.prod_property = 'B0001' and T2.external_matl_group_code = '90010') or 
 ( T2.matl_group_code in('7000010','9000010') and T2.prod_property = 'B0001' and T2.external_matl_group_code = '10010'))
--AND coalesce(T9.DZ1,0) = 1
AND T1.purchase_order_type IN('NB','ZNB1','ZNB2')
and  t1.stat_month = '${PERIOD_ID_M}'
;
