         where is_zdjfzj in ('是', '否')

                                              when sale_region_name like '%子公司' then sales_office_name
                                              else sale_region_name end
                , werks, kunnr
                , b.category_name
                , b.dimension
         union all
         select tj_month
              , '工程'                         channel_text
              , department
              , kunnr
              , werks
              , count(case
                          when SFZDJF = '是' then zjhd
                          else zjhdid
             end)
              , count(case
                          when is_zdjfzj = '是' then case
                                                        when SFZDJF = '是' then zjhd
                                                        else zjhdid
                              end end)
              , sum(zj_qty * a.unit_area) as zj_area
              , sum(menge * a.unit_area)  as order_area
              , b.category_name
              , b.dimension
         from dm.DM_SO_ZL_ORDER_ZJ_PRE_T a
                  left join dwimd.dwi_md_data_material_general_t b on a.matnr = b.material_num
         where is_zdjfzj in ('是', '否')
           and bstdk <= replace(CURRENT_DATE + INTERVAL '-1 day', '-', '')
         group by tj_month

                , b.dimension
     ),

                    when (vtweg not in ('08', '06') or vtweg is null) then '零售'
                    when vtweg = '08' then '整装'
                    else '' end              channel_text
              , department
              , kunnr
              , werks
              , count(case
                          when SFZDJF = 'Y' then VBELN
                          else concat(VBELN, POSNR)
             end)                            order_cnt

                          when is_zdjfzt = '是' then case
                                                        when SFZDJF = 'Y' then VBELN
                                                        else concat(VBELN, POSNR)
                              end end)       zt_cnt
              , sum(zt_qty * a.unit_area) as zt_area
              , sum(kwmeng * a.unit_area) as order_area
              , b.category_name
              , b.dimension
         from dm.DM_SO_FZL_ORDER_PRE_T a
                  left join dwimd.dwi_md_data_material_general_t b on a.matnr = b.material_num
         where is_zdjfzt in ('是', '否')

         group by tj_month --统计月份
                , channel_text
                , department
                , werks
                , kunnr
                , b.category_name
                , b.dimension

         select tj_month

                          when SFZDJF = '是' then zjhd
                          else zjhdid
             end)
              , count(case
                          when is_zdjfzt = '是' then case
                                                        when SFZDJF = '是' then zjhd
                                                        else zjhdid
                              end end)
              , sum(zt_qty * a.unit_area) as zt_area

                , b.category_name
                , b.dimension

           when substring(plant_code, 1, 2) in ('16') then '湖南'
           when substring(plant_code, 1, 2) in ('17') then '华盛昌'
           when substring(plant_code, 1, 2) in ('15') then '江西'
           when substring(plant_code, 1, 2) in ('13') then '清远'
           when substring(plant_code, 1, 2) in ('19') then '山西'
           when substring(plant_code, 1, 2) in ('8A', '72', '12', '7D', '77', '75', '79') then '外协'
           when substring(plant_code, 1, 2) in ('18') then '重庆'
           else concat('其他', substring(plant_code, 1, 2))
           end as       base_jd,
       category_name,
       dimension,
       sum(当月逾期未交付)     当月逾期未交付,
       sum(当月逾期未提面积)    当月逾期未提面积,
       sum(上月及之前逾期未交付)  上月及之前逾期未交付,
       sum(上月及之前逾期未提面积) 上月及之前逾期未提面积

         select to_char((stat_month || '01')::date, 'yyyy-MM') stat_month,
                case when channel_name= '大包' then '整装' else channel_name end                    channel_text,
                case
                    when dept_lv4_name = '工程市场中心' then '工程市场中心'
                    when dept_lv4_name in ('直营经营管理板块', '区域运营中心') then dept_lv5_name
                    else '其他'
                    end                                        org,
                cust_num                                       cust_code,
                sum(coalesce(nodeliver_area_aps, '0'))         未交付面积,
                sum(case
                        when to_char(expect_date_aps::date, 'yyyyMMdd') <= to_char(c.month_last_day::date, 'yyyyMMdd')
                            then nodeliver_area_aps
                        else 0 end)                            逾期未交付,
                0                                              留备货面积,
                0                                              逾期未提面积,
                0                                              本月准交面积,
                0                                              本月准提面积,
                0                                              本月准交订单面积,
                0                                              本月准提订单面积,
                plant_code,
                b.category_name,
                b.dimension,
                sum(case
                        when to_char(expect_date_aps::date, 'yyyyMMdd') <=
                             to_char(c.month_last_day::date, 'yyyyMMdd')
                            and to_char(expect_date_aps::date, 'yyyyMMdd') >=
                                to_char(c.day_mbegin_dt::date, 'yyyyMMdd')
                            then nodeliver_area_aps
                        else 0 end)                            当月逾期未交付,
                0                                              当月逾期未提面积,
                sum(case
                        when to_char(expect_date_aps::date, 'yyyyMMdd') <
                             to_char(c.day_mbegin_dt::date, 'yyyyMMdd') then nodeliver_area_aps
                        else 0 end)                            上月及之前逾期未交付,
                0                                              上月及之前逾期未提面积
         from DM.DM_RPT_SALES_DELIVER_T A
                  left join dwimd.dwi_md_data_material_general_t b on a.material_num = b.material_num
                  join monthseies c on a.stat_month = c.month_id
         where 1 = 1
           and nodeliver_area_aps > 0
         group by stat_month, channel_text, cust_code, org, plant_code,
                  b.category_name,
                  b.dimension
         union all
         --准交准提
         select tj_month,
                channel_text,
                department       org,
                kunnr            cust_code,
                0                未交付面积,
                0                逾期未交付,
                0                留备货面积,
                0                逾期未提面积,
                sum(zj_cnt)      本月准交面积,
                sum(zt_cnt)      本月准提面积,
                sum(order_cnt_1) 本月准交订单面积,
                sum(order_cnt_2) 本月准提订单面积,
                werks as         factory_code,
                category_name,
                dimension,
                0                当月逾期未交付,
                0                当月逾期未提面积,
                0                上月及之前逾期未交付,
                0                上月及之前逾期未提面积
         from mid
         where tj_month = '${PERIOD_ID_M}'
         group by tj_month, channel_text, department, kunnr, werks,
                  category_name,
                  dimension
         union all
         --逾期未提
         select to_char((stat_month || '01')::date, 'yyyy-MM') stat_month,
                case
                    when distribution_channel_code = '06' then '工程'

                    when lev4_name = '工程市场中心' then lev4_name
                    when lev5_name like '%运营中心%' then lev5_name
                    else '其他' end                              org,
                cust_num                                       cust_code,
                0                                              未交付面积,
                0                                              逾期未交付,
                sum(sal_pre_res_area)                          留备货面积,
                sum(case
                        when to_char(acc_time_date::date, 'yyyyMMdd') <=
                             to_char(c.month_last_day::date, 'yyyyMMdd') then sal_pre_res_area
                        else 0 end)                            逾期未提面积,
                0                                              本月准交面积,
                0                                              本月准提面积,
                0                                              本月准交订单面积,
                0                                              本月准提订单面积,
                factory_code,
                b.category_name,
                b.dimension,
                0                                              当月逾期未交付,
                sum(case
                        when to_char(acc_time_date::date, 'yyyyMMdd') >=
                             to_char(c.day_mbegin_dt::date, 'yyyyMMdd')
                            and to_char(acc_time_date::date, 'yyyyMMdd') <=
                                to_char(c.month_last_day::date, 'yyyyMMdd') then sal_pre_res_area
                        else 0 end)                            当月逾期未提面积,
                0                                              上月及之前逾期未交付,
                sum(case
                        when to_char(acc_time_date::date, 'yyyyMMdd') <
                             to_char(c.day_mbegin_dt::date, 'yyyyMMdd') then sal_pre_res_area
                        else 0 end)                            上月及之前逾期未提面积
         from DM.DM_RPT_SALES_DELIVER_2023_T A
                  left join dwimd.dwi_md_data_material_general_t b on a.material_num = b.material_num
                  join monthseies c on a.stat_month = c.month_id
         where 1 = 1
           and lb_type in ('销售订单留货', '专属客编留货', '销售订单备货')
         group by stat_month, channel_text, cust_code, org, factory_code,
                  b.category_name,
                  b.dimension
         union all
         --留备货
         select to_char((start_month || '01')::date, 'yyyy-MM')           start_month,

                    when distribution_channel_code = '06' then '工程'

                        then '零售'
                    when distribution_channel_code = '08' then '整装'
                    else '' end                                           channel_text,
                case
                    when lev4_name = '工程市场中心' then lev4_name
                    when lev5_name like '%运营中心%' then lev5_name
                    else '其他' end                                         org,

                             to_char(c.month_last_day::date, 'yyyyMMdd')
                            then available_stock_qty * b.length_1 * b.width / 1000000
                        else 0 end)                                       逾期未提面积,

                0                                                         本月准提面积,
                0                                                         本月准交订单面积,

                b.category_name,
                b.dimension,
                0                                                         当月逾期未交付,

                        when to_char(promise_stock_date::date, 'yyyyMMdd') >=
                             to_char(c.day_mbegin_dt::date, 'yyyyMMdd')
                            and to_char(promise_stock_date::date, 'yyyyMMdd') <=
                                to_char(c.month_last_day::date, 'yyyyMMdd')

                0                                                         上月及之前逾期未交付,

                        when to_char(promise_stock_date::date, 'yyyyMMdd') <
                             to_char(c.day_mbegin_dt::date, 'yyyyMMdd')
                            then available_stock_qty * b.length_1 * b.width / 1000000
                        else 0 end)                                       上月及之前逾期未提面积
         from dwrotd.dwr_wm_wbs_stock_f A
                  left join dwimd.dwi_md_data_material_general_t b
                            on A.material_num = b.material_num
                  join monthseies c on a.start_month = c.month_id
         where 1 = 1
           and available_stock_qty > 0
           and promise_stock_date is not null
         group by start_month, channel_text, cust_code, org, factory_code,
                  b.category_name,
                  b.dimension
     ) a
where 1 = 1
group by stat_month, channel_text, org,
         cust_code, base_jd,
         category_name,
         dimension
