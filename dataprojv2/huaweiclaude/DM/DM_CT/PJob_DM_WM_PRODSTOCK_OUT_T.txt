--OEM出库量
INSERT OVERWRITE TABLE DM.DM_WM_PRODSTOCK_OUT_T
SELECT  
 a.FACTORY_WERKS_ID --工厂
,a.FACTORY_WERKS_CODE --工厂
,a.FACTORY_WERKS_NAME --工厂
,a.VOUCHER_DATE --凭证日期
,a.VOUCHER_POST_DATE --过帐日期
,a.COST_CENTER --成本中心
,a.SUPPLIER_OR_CREDITOR_ID --供应商
,a.SUPPLIER_OR_CREDITOR_ACCT --供应商
,A.SUPPLIER_OR_CREDITOR_NAME --供应商
,a.MATERIAL_VCH_NO_MBLNR --物料凭证
,a.MATERIAL_ID --物料
,A.MATERIAL_NUM --物料
,A.MATERIAL_NAME --物料
,a.BATCH_NUM --批次
,a.ORDER_NUM --订单号
,a.PRODUCTION_DATE --生产日期
,case when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY
      when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' then a.QUANTITY
	  else 0
	   end as QUANTITY_K --出库数量K
,case when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY
      when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' then a.QUANTITY
	  else 0
	   end as QUANTITY_Q --出库数量Q
,case when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY
      when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' then a.QUANTITY
	  else 0
	   end -
case when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY
      when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' then a.QUANTITY
	  else 0
	   end as QUANTITY --出库数量
,case when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
      when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' then a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
	  else 0
	   end as AREA_K --出库面积K（单位平方毫米）
,case when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
      when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' then a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
	  else 0
	   end as AREA_Q --出库面积Q（单位平方毫米）
,case when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
      when a.SPECIAL_STOCK_FLAG_2 = 'K_T148T' then a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
	  else 0
	   end - 
case when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' and a.DR_CR_ID = 'H_SHKZG' then 0 - a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
      when a.SPECIAL_STOCK_FLAG_2 = 'Q_T148T' then a.QUANTITY * B.LENGTH_1 * B.WIDTH /1000000
	  else 0
	   end AS AREA --出库面积   
,a.INV_LOCATION --库存地点
,a.MOVE_TYPE --移动类型
,a.REFERENCE_VOUCHER_NUM --参照
,a.CLEARANCE_POINT --卸货点
,a.ACCOUNT_VOUCHER_INPUT_DATE --输入日期
,a.INPUT_TIME --输入时间
,a.PURCHASE_VOUCHER_NUM --采购订单
,a.RESERVE_DEMAND_NO --预留编号
,a.USER_NAME --用户名称
,a.DELIVERY_NO --收货/发货单编号
,a.BASE_MEASURE_UNIT --基本计量单位
,a.INPUT_UNIT_QTY --Quantity in UnE
,a.SPECIAL_STOCK_FLAG_2 --特殊库存
,a.ENTRY_UNIT_CODE --OPUn 中的数量
,a.PO_PRICE_UNIT --订单价格单位
,a.PURCHASE_ORDER_UNIT --订单单位
,a.COMPANY_ID --公司ID
,a.COMPANY_CODE --公司代码
,a.COMPANY_NAME --公司名称
,a.PURCHASE_VOUCHER_ITEM_NUM --项目
,a.GOODS_MOVE_REASON --移动原因
,a.SALES_ORDER_DELIVERY_PLAN --销售订单计划
,a.SALES_ORDER_CLAUSE_QTY --销售订单项目
,a.CUST_ID --客户
,a.CUST_NUM --客户
,a.CUST_NAME --客户
,a.MATERIAL_VCH_YEAR_MJAHR --物料凭证的年份
,a.RESERVE_DEMAND_PROJECT_NO --项目编号库存转储预留
,a.TRADE_TYPE --交易/事件类型
,a.ASSET_SUB_NUM --次级编号
,a.AUTO_CREATE_LINE_ITEM_FLAG --项目自动创建
,a.MATERIAL_VCH_PROJ_ZEILE --项目
,a.MAIN_ASSET_NUMBER --资产
,a.PROCESS_TASK_LIST_NUM --工序的工艺路线编号
,a.RECEIVE_QTY --按订单单位计的数量
,a.ASSESSMENT_TYPE --评估类型
,a.ORDER_UNIVERSAL_COUNTER --计数器
,a.EXTERNAL_INPUT_SALES_VALUE --销售值
,a.SALES_ORDER_QTY --销售订单
,a.MOVE_FLAG --移动标识
,a.RECEIVE_FLAG --收货标识
,a.WBS_ELEMENT --WBS 元素
,a.PROJECT_TEXT --文本
,a.DR_CR_ID --借/贷标识
,a.CONSUMPTION_ENTRY --消耗过账
,a.VOUCHER_HEADER_TEXT --凭证抬头文本
,a.REFER_VCH_NO --参考凭证的凭证号
,a.REFER_VCH_PROJECT --参考凭证项目
,B.EXTERNAL_MATL_GROUP_CODE --外部物料组
,B.EXTERNAL_MATL_GROUP_NAME --外部物料组
,B.MATL_GROUP_CODE --物料组
,B.MATL_GROUP_NAME --物料组
,B.PRODUCT_LEVEL_CODE --产品层次
,B.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE --产品渠道编码
,B2.CONFIG_NAME AS PRODUCT_CHANNEL_NAME --产品渠道名称
,B1.VTEXT AS PRODUCT_STAGE2 --产品层次2
,B.LENGTH_1--长度
,B.WIDTH--宽度
,B.DIMENSION--规格
,C.PRODUCT_AREA_ID --产区
,C.PRODUCT_AREA_CODE --产区
,C.PRODUCT_AREA_NAME --产区
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
,A.LC_AMT AS OUTWAREHOUSE_AMOUNT --出库金额
,A.CURRENCY_CODE --货币单位
,CASE WHEN COALESCE(BD.BRAND_NAME,'') <> '' THEN BD.BRAND_NAME
      WHEN COALESCE(BD1.BRAND_NAME,'') <> '' THEN BD1.BRAND_NAME
      WHEN COALESCE(BD2.BRAND_NAME,'') <> '' THEN BD2.BRAND_NAME
      ELSE 'NA'
       END AS BRAND_NAME --品牌名称
FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D B
ON A.MATERIAL_ID = B.MATERIAL_ID
AND B.END_DATE = '4712-12-31'
AND B.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_T179T_1001 B1 --产品品类
ON B1.MANDT = B.GROUP_NUM
AND B1.SPRAS = '1'
AND B1.DEL_FLAG = 'N'
AND SUBSTR(B.PRODUCT_LEVEL_CODE,1,4) = B1.PRODH
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R B2
ON B.RESERVED_FIELD_1 = B2.SRC_SYS_CONFIG_CODE
AND B2.SOURCE_CODE = 'ZTYL01'
LEFT JOIN (SELECT A.SUPPLIER_OR_CREDITOR_ID
                 ,A.PRODUCT_AREA_ID
	             ,A.PRODUCT_AREA_CODE
                 ,A.PRODUCT_AREA_NAME
             FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D A
			GROUP BY 
			      A.SUPPLIER_OR_CREDITOR_ID
                 ,A.PRODUCT_AREA_ID
	             ,A.PRODUCT_AREA_CODE
                 ,A.PRODUCT_AREA_NAME
		  ) C
ON A.SUPPLIER_OR_CREDITOR_ID = C.SUPPLIER_OR_CREDITOR_ID
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD
    ON A.FACTORY_WERKS_CODE = BD.FACTORY_WERKS_CODE
   AND B.EXTERNAL_MATL_GROUP_CODE = BD.EXTERNAL_MATL_GROUP_CODE
   AND B.MATL_GROUP_CODE = BD.MATL_GROUP_CODE
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD1
    ON A.FACTORY_WERKS_CODE = BD1.FACTORY_WERKS_CODE
   AND BD1.EXTERNAL_MATL_GROUP_CODE = ''
   AND B.MATL_GROUP_CODE = BD1.MATL_GROUP_CODE 
  LEFT JOIN (SELECT DISTINCT FACTORY_WERKS_CODE
                            ,BRAND_NAME
               FROM DWRDIM.DWR_DIM_MATERIAL_BRAND_D
			  WHERE MATL_GROUP_CODE = ''
			) BD2
    ON A.FACTORY_WERKS_CODE = BD2.FACTORY_WERKS_CODE
/*业务对象：
会计凭证-物料凭证：移动类型411K/412K
物料库存-特别库存标识Q
供应商-供应商编码、省份城市（对应区分产区），BP查看
物料主数据-产品渠道、产品层次2、大小/量纲、
外部物料组：10010，物料组1000010*/
WHERE a.MOVE_TYPE IN('411','412')
AND a.SPECIAL_STOCK_FLAG_2 IN('K_T148T','Q_T148T')
