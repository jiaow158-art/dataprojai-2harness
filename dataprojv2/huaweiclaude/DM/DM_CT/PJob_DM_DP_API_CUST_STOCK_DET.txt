with C as(
select
	t3.sales_center_code as sales_center_code,
	t1.sales_org_name as sales_center,
	t1.customer_code as cust_num,
	t4.cust_name as cust_name,
	coalesce(case when t5.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t5.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t5.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t5.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t5.product_level2_code = '1003' AND t5.dimension = '1500X750' OR t5.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t5.product_level2_code = '1003' AND (t5.dimension != '1500X750' AND t5.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t5.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t5.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t5.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t5.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,
	t1.dimension as dimension,
	t1.material_num as material_num,
	t1.material_name as material_name,
	qm_area_qty as stock_area,
	if(t2.share_flag in ('A','Y'),'Y','N') as is_share
from DWROTD.DWR_WM_INVENTORY_CT_T t1
left join SDI.SDI_CT_CUSTOMER_DATA_1037 t2 on t1.data_source_falg = t2.data_source_falg and  t1.customer_code = t2.customer_code
left join (select distinct sales_center_code,sales_center from dm.dm_dp_api_sales_group_struct) t3 on t1.sales_org_name = t3.sales_center
left join dwrdim.dwr_dim_cust_general_d  t4 on t1.customer_code = t4.cust_num and t4.end_date = '4712-12-31'
left join DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T t5 on t1.material_num = t5.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t6 on coalesce(t5.product_level3_code,t5.product_level2_code,t5.product_level1_code) = t6.prod_code3 and t6.PROD_CODE3 IS NOT NULL  
where 1=1
and t1.period_id_m = substring(date_sub (CURRENT_TIMESTAMP(),1),1,7)
and t5.prod_property  = 'A0001' --仅需要瓷砖产品
and t2.share_flag in ('N','A') --经销商库存看('N','A') 
)

INSERT OVERWRITE TABLE  dm.dm_dp_api_cust_stock_det
select 
	sales_center_code,
	sales_center,
	cust_num,
	cust_name,
	fin_cate,
	dimension,
	material_num,
	material_name,
	sum(stock_area) as stock_area,
	is_share,
	current_timestamp() as dw_last_update_date
from C
group by 
	sales_center_code,
	sales_center,
	cust_num,
	cust_name,
	fin_cate,
	dimension,
	material_num,
	material_name,
	is_share
