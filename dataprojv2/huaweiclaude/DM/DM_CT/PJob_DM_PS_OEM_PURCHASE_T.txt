--出/入库取值范围:   工厂：8A*/72*/7D*/12*     库存地点：BE*/BJ01/YX03
--入库：移动类型：101、102、161
--出库：移动类型排除：101、102、313、344、411、412、415、301、311、991、161、162
--出/入库日期：月初第一日到日报前一日
--库存取值一样：工厂：8A*/72*/7D*/12*     库存地点：BE*/BJ01/YX03

WITH MAT AS(
SELECT 
       A.MATERIAL_NUM ,A.BATCH_NUM, A.INV_LOCATION ,A.FACTORY_WERKS_CODE ,A.FACTORY_WERKS_NAME ,A.MOVE_TYPE ,
       A.SUPPLIER_OR_CREDITOR_ID,   A.SUPPLIER_OR_CREDITOR_ACCT,  A.SUPPLIER_OR_CREDITOR_NAME,
       ROW_NUMBER() OVER(PARTITION BY  A.MATERIAL_NUM ,A.BATCH_NUM ORDER BY A.VOUCHER_POST_DATE,A.MATERIAL_VCH_NO_MBLNR  ASC) RN
 FROM  DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A 
)
,KC AS( --库存
SELECT B.SUPPLIER_OR_CREDITOR_ACCT AS SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,SUM(QUANTITY) AS STOCK_QTY --库存数量
      ,SUM(ZKCMJ) AS STOCK_AREA --库存面积
  FROM DM.DM_FIN_STOCK_DETAIL_ACCAGE_T_2023 A
  LEFT JOIN MAT B
    ON A.MATERIAL = B.MATERIAL_NUM
   AND A.BATCH = B.BATCH_NUM
   AND B.RN = 1
 WHERE CALMONTH = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
   AND SUBSTR(PLANT,1,2) IN('8A','72','7D','12')
   AND (SUBSTR(STOR_LOC,1,2) = 'BE' OR STOR_LOC IN('BJ01','YX03'))
 GROUP BY B.SUPPLIER_OR_CREDITOR_ACCT
),WJSL AS( --未交付数
SELECT SUPPLIER_OR_CREDITOR_ACCT AS SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,SUPPLIER_OR_CREDITOR_NAME
      ,SUM(COALESCE(NOT_LEVITE_QUANTITY,0)) AS NOT_LEVITE_QUANTITY --未交付数量
	  ,SUM(COALESCE(NOT_LEVITE_AREA,0)) AS NOT_LEVITE_AREA --未交付面积
  FROM DM.DM_PS_PURCHASE_PLAN_DTL_T
 WHERE PURCHASE_VOUCHER_NUM IS NOT NULL
   AND STATUS NOT IN('已完成交付','已关闭')   
 GROUP BY SUPPLIER_OR_CREDITOR_ACCT
      ,SUPPLIER_OR_CREDITOR_NAME
),IN_OUT AS(
SELECT C.SUPPLIER_OR_CREDITOR_ACCT AS SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,SUM(CASE WHEN A.MOVE_TYPE IN('101') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY
				WHEN A.MOVE_TYPE IN('102','161') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY * -1
				ELSE 0
				 END) AS MONTH_IN_QTY --月入库量
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	             AND A.PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
	            THEN QUANTITY
				ELSE 0
				 END) AS MONTH_OUT_QTY --月出库量
	  ,SUM(CASE WHEN A.MOVE_TYPE IN('101')
				THEN A.QUANTITY
				WHEN A.MOVE_TYPE IN('102','161')
				THEN A.QUANTITY * -1
				ELSE 0
				 END) AS TOT_IN_QTY --年累计入库量
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	            THEN A.QUANTITY
				ELSE 0
				 END) AS TOT_OUT_QTY --年累计出库量
      ,SUM(CASE WHEN A.MOVE_TYPE IN('101') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY * B.UNIT_AREA
				WHEN A.MOVE_TYPE IN('102','161') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY * B.UNIT_AREA * -1
				ELSE 0
				 END) AS MONTH_IN_AREA --月入库面积
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	             AND A.PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
	            THEN QUANTITY * B.UNIT_AREA
				ELSE 0
				 END) AS MONTH_OUT_AREA --月出库面积
      ,SUM(CASE WHEN A.MOVE_TYPE IN('101')
				THEN A.QUANTITY * B.UNIT_AREA
				WHEN A.MOVE_TYPE IN('102','161')
				THEN A.QUANTITY * B.UNIT_AREA * -1
				ELSE 0
				 END) AS TOT_IN_AREA --年累计入库面积
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	            THEN A.QUANTITY * B.UNIT_AREA
				ELSE 0
				 END) AS TOT_OUT_AREA --年累计出库面积
	  ,SUM(CASE WHEN A.MOVE_TYPE IN('101') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
	             AND SUBSTR(A.VOUCHER_POST_DATE,1,10) = '${PERIOD_ID_D}'
				THEN A.QUANTITY
				WHEN A.MOVE_TYPE IN('102','161') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY * -1
				ELSE 0
				 END) AS DAY_IN_QTY --日入库量
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	             AND A.PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				 AND SUBSTR(A.VOUCHER_POST_DATE,1,10) = '${PERIOD_ID_D}'
	            THEN QUANTITY
				ELSE 0
				 END) AS DAY_OUT_QTY --日出库量
      ,SUM(CASE WHEN A.MOVE_TYPE IN('101') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
	             AND SUBSTR(A.VOUCHER_POST_DATE,1,10) = '${PERIOD_ID_D}'
				THEN A.QUANTITY * B.UNIT_AREA
				WHEN A.MOVE_TYPE IN('102','161') AND PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				THEN A.QUANTITY * -1 * B.UNIT_AREA
				ELSE 0
				 END) AS DAY_IN_AREA --日入库面积
      ,SUM(CASE WHEN A.MOVE_TYPE NOT IN('101','102','313','344','411','412','415','301','311','991','161','162')
	             AND A.PERIOD_ID_M = SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
				 AND SUBSTR(A.VOUCHER_POST_DATE,1,10) = '${PERIOD_ID_D}'
	            THEN QUANTITY * B.UNIT_AREA
				ELSE 0
				 END) AS DAY_OUT_AREA --日出库面积
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
    ON A.MATERIAL_NUM = B.MATERIAL_NUM
  LEFT JOIN MAT C
    ON A.MATERIAL_NUM = C.MATERIAL_NUM
   AND A.BATCH_NUM = C.BATCH_NUM
   AND C.RN = 1
 WHERE SUBSTR(A.FACTORY_WERKS_CODE,1,2) IN('8A','72','7D','12')
   AND (SUBSTR(A.INV_LOCATION,1,2) = 'BE' OR A.INV_LOCATION IN('BJ01','YX03'))
   AND A.PERIOD_ID_M >= CONCAT(SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,4),'01')
   AND A.PERIOD_ID_M <= SUBSTR(REPLACE('${PERIOD_ID_D}','-',''),1,6)
   AND C.SUPPLIER_OR_CREDITOR_ACCT IS NOT NULL
  GROUP BY C.SUPPLIER_OR_CREDITOR_ACCT --供应商编码
),ALL_T AS(
SELECT SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,MONTH_IN_QTY --月入库量
      ,MONTH_OUT_QTY --月出库量
	  ,TOT_IN_QTY --年累计入库量
      ,TOT_OUT_QTY --年累计出库量
      ,MONTH_IN_AREA --月入库面积
      ,MONTH_OUT_AREA --月出库面积
      ,TOT_IN_AREA --年累计入库面积
      ,TOT_OUT_AREA --年累计出库面积
	  ,0 AS STOCK_QTY --库存数量
      ,0 AS STOCK_AREA --库存面积
	  ,0 AS NOT_LEVITE_QUANTITY --未交付数量
	  ,0 AS NOT_LEVITE_AREA --未交付面积
	  ,DAY_IN_QTY --日入库量
      ,DAY_OUT_QTY --日出库量
      ,DAY_IN_AREA --日入库面积
      ,DAY_OUT_AREA --日出库面积
  FROM IN_OUT
  
 UNION ALL
 
SELECT SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,0 AS MONTH_IN_QTY --月入库量
      ,0 AS MONTH_OUT_QTY --月出库量
	  ,0 AS TOT_IN_QTY --年累计入库量
      ,0 AS TOT_OUT_QTY --年累计出库量
      ,0 AS MONTH_IN_AREA --月入库面积
      ,0 AS MONTH_OUT_AREA --月出库面积
      ,0 AS TOT_IN_AREA --年累计入库面积
      ,0 AS TOT_OUT_AREA --年累计出库面积
	  ,0 AS STOCK_QTY --库存数量
      ,0 AS STOCK_AREA --库存面积
	  ,NOT_LEVITE_QUANTITY --未交付数量
	  ,NOT_LEVITE_AREA --未交付面积
	  ,0 AS DAY_IN_QTY --日入库量
      ,0 AS DAY_OUT_QTY --日出库量
      ,0 AS DAY_IN_AREA --日入库面积
      ,0 AS DAY_OUT_AREA --日出库面积
  FROM WJSL
  
  UNION ALL
 
SELECT SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,0 AS MONTH_IN_QTY --月入库量
      ,0 AS MONTH_OUT_QTY --月出库量
	  ,0 AS TOT_IN_QTY --年累计入库量
      ,0 AS TOT_OUT_QTY --年累计出库量
      ,0 AS MONTH_IN_AREA --月入库面积
      ,0 AS MONTH_OUT_AREA --月出库面积
      ,0 AS TOT_IN_AREA --年累计入库面积
      ,0 AS TOT_OUT_AREA --年累计出库面积
	  ,STOCK_QTY --库存数量
      ,STOCK_AREA --库存面积
	  ,0 AS NOT_LEVITE_QUANTITY --未交付数量
	  ,0 AS NOT_LEVITE_AREA --未交付面积
	  ,0 AS DAY_IN_QTY --日入库量
      ,0 AS DAY_OUT_QTY --日出库量
      ,0 AS DAY_IN_AREA --日入库面积
      ,0 AS DAY_OUT_AREA --日出库面积	  
  FROM KC 
)
INSERT OVERWRITE TABLE DM.DM_PS_OEM_PURCHASE_T
SELECT A.SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,B.SUPPLIER_OR_CREDITOR_NAME --供应商名称
      ,SUM(COALESCE(A.MONTH_IN_QTY,0)) AS MONTH_IN_QTY --月入库量
      ,SUM(COALESCE(A.MONTH_OUT_QTY,0)) AS MONTH_OUT_QTY --月出库量
	  ,SUM(COALESCE(A.TOT_IN_QTY,0)) AS TOT_IN_QTY --年累计入库量
      ,SUM(COALESCE(A.TOT_OUT_QTY,0)) AS TOT_OUT_QTY --年累计出库量
      ,SUM(COALESCE(A.MONTH_IN_AREA,0)) AS MONTH_IN_AREA --月入库面积
      ,SUM(COALESCE(A.MONTH_OUT_AREA,0)) AS MONTH_OUT_AREA --月出库面积
      ,SUM(COALESCE(A.TOT_IN_AREA,0)) AS TOT_IN_AREA --年累计入库面积
      ,SUM(COALESCE(A.TOT_OUT_AREA,0)) AS TOT_OUT_AREA --年累计出库面积
	  ,SUM(COALESCE(A.STOCK_QTY,0)) AS STOCK_QTY --库存数量
      ,SUM(COALESCE(A.STOCK_AREA,0)) AS STOCK_AREA --库存面积
	  ,SUM(COALESCE(A.NOT_LEVITE_QUANTITY,0)) AS NOT_LEVITE_QUANTITY --未交付数量
	  ,SUM(COALESCE(A.NOT_LEVITE_AREA,0)) AS NOT_LEVITE_AREA --未交付面积
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SUM(COALESCE(A.DAY_IN_QTY,0)) AS DAY_IN_QTY --日入库量
      ,SUM(COALESCE(A.DAY_OUT_QTY,0)) AS DAY_OUT_QTY --日出库量
      ,SUM(COALESCE(A.DAY_IN_AREA,0)) AS DAY_IN_AREA --日入库面积
      ,SUM(COALESCE(A.DAY_OUT_AREA,0)) AS DAY_OUT_AREA --日出库面积
  FROM ALL_T A
  LEFT JOIN DWRDIM.DWR_DIM_SUPPLIER_GENERAL_D B
    ON A.SUPPLIER_OR_CREDITOR_NUM = B.SUPPLIER_OR_CREDITOR_NUM 
   AND B.END_DATE = '4712-12-31'
 GROUP BY A.SUPPLIER_OR_CREDITOR_NUM --供应商编码
      ,B.SUPPLIER_OR_CREDITOR_NAME --供应商名称
;
