with tem as
(
select 
	case when substr(t1.factory_werks_code,length(t1.factory_werks_code)-1,2) in ('10') then 'H03230617'
	    when substr(t1.factory_werks_code,length(t1.factory_werks_code)-1,2) in ('20') and t4.external_matl_group_code = '10080' then 'H03230628'
	    when t1.inv_location in ('GCQA','BJ01','BJ','GCSA','GCCA','GCTA','GCSD') then 'H04230685' --战略就是工程市场中心？
	    when t1.inv_location in ('20') then 'H05230626' --深圳子公司中心仓
	    when t1.inv_location in ('0026') then 'H05230608' --广西子公司中心仓
	    when t1.inv_location in ('0030') then 'H05230624' --上海子公司中心仓
	    when t1.inv_location in ('0018') then 'H05230623' --陕西子公司中心仓
	    when t1.inv_location in ('GFZ','BC') then 'H05230607' --广佛子公司中心仓
	    else '' end as sales_center_code,
	case when substr(t1.factory_werks_code,length(t1.factory_werks_code)-1,2) in ('10') then '专属库存'-- 国际营销系统
	    when substr(t1.factory_werks_code,length(t1.factory_werks_code)-1,2) in ('20') and t4.external_matl_group_code  = '10080' then '专属库存' -- 梦之家事业部
	    when t1.inv_location in ('GCQA','BJ01','BJ','GCSA','GCCA','GCTA','GCSD') then '专属库存'
	    when t1.inv_location in ('20') then '专属库存' --深圳子公司中心仓
	    when t1.inv_location in ('0026') then '专属库存' --广西子公司中心仓
	    when t1.inv_location in ('0030') then '专属库存' --上海子公司中心仓
	    when t1.inv_location in ('0018') then '专属库存' --陕西子公司中心仓
	    when t1.inv_location in ('GFZ','BC') then '专属库存' --广佛子公司中心仓
	    else '工厂可用库存' end as warehouse_type,
	coalesce(case when t4.prod_property IN('A0001', 'A0002', 'C0002') then   -- 只针对 产品所有权 =瓷砖产品、梦之家产品、ART+ 定义
         case when t4.tow_lev_channel_code in('A003','A004') then '世界印象'   -- 二级渠道=art+/iw
               WHEN t4.product_level2_code IN ('1001') THEN '玻化砖'   --产品层次2=玻化砖
               WHEN t4.product_level2_code IN ('1006', '1007', '1104') THEN '抛釉砖'  -- 产品层次2=晶理石/晶玉瓷/抛釉、晶玉配套
               WHEN t4.product_level2_code = '1003' AND t4.dimension = '1500X750' OR t4.dimension = '1800X900' THEN '大板'  -- 产品层次2=岩板且规格=1500X750或者1800X900
               WHEN t4.product_level2_code = '1003' AND (t4.dimension != '1500X750' AND t4.dimension != '1800X900') THEN '超大板' -- 产品层次2=岩板且规格≠1500X750或者1800X900
               WHEN t4.product_level2_code IN ('1002', '1103') THEN '瓷片'  -- 产品层次2=瓷片/瓷片配套
               WHEN t4.product_level2_code = '1017' THEN '精瓷中板'   -- 产品层次2=精瓷中板
               WHEN t4.product_level2_code IN ('1004', '1102') THEN '仿古砖'  -- 产品层次2=仿古砖/仿古砖配套
               WHEN t4.product_level2_code = '1018' THEN '景观砖'  -- 产品层次2=景观砖
               ELSE  '其他瓷砖' END 
              END,'其他品类') as fin_cate,
	t4.dimension as dimension,
	t1.material_num as material_num,
	t4.material_name as material_name,
	t1.available_inventory_area as stock_area,
	current_timestamp() as dw_last_update_date
from dm.DM_WM_ALL_TYPE_STOCK_T t1
left join dwimd.dwi_md_data_material_general_t t4 on t1.material_num = t4.material_num
--left join DM.DM_RPT_PROD_LEVEL_T t5 on  coalesce(t4.product_level3_code,t4.product_level2_code,t4.product_level1_code) = t5.prod_code3 and t5.PROD_CODE3 IS NOT NULL 
where 1=1
and t4.external_matl_group_code IN('10010','10080')
and t4.matl_group_code IN('1000010')
and t4.matl_type_code = 'FERT'
and substr(t1.factory_werks_code,length(t1.factory_werks_code)-1,2) not in ('00','80')
and substr(t1.batch_num,length(t1.batch_num),1) not in ('7','9')
and t1.inv_location not in ('0028','7F21')
and t1.available_inventory_area > 0
)

INSERT OVERWRITE TABLE dm.dm_dp_api_warehouse_stock
select 
    t1.warehouse_type,
    t1.fin_cate,
    t1.dimension,
    t1.material_num,
    t1.material_name,
    sum(t1.stock_area) as stock_area,
    t1.dw_last_update_date,
    t1.sales_center_code,
    t2.sales_center
from tem t1
left join (select distinct sales_center_code,sales_center from dm.dm_dp_api_pred_org) t2 on t1.sales_center_code = t2.sales_center_code
GROUP BY 
    t1.sales_center_code,
    t2.sales_center,
    t1.warehouse_type,
    t1.fin_cate,
    t1.dimension,
    t1.material_num,
    t1.material_name,
    t1.dw_last_update_date
