delete dm.dm_expense_prediction_fsreclass_t;
insert into dm.dm_expense_prediction_fsreclass_t
select
t1.frdelete
,t1.division
,t1.division_description
,t1.son_division
,t1.stat_date
,t1.functional_scope
,t1.cost_center
,t1.cost_center_description
,t1.ledger_account
,t1.ledger_account_description
,t1.amount
,case when substr(lpad(t1.ledger_account,10,'0') ,1,3) = '005' then '' else  coalesce(t2.actual_post_function_scope,F1.func_area) end functional_scope_reclass
,current_timestamp  dw_last_update_date
from 
upload.upload_expense_prediction_t t1
left join DWRDIM.DWR_DIM_CSKS_F F1 
on T1.cost_center  = F1.kostl
and F1.kokrs = 'DP00'
left join upload.upload_account_reclassification_t t2  
on F1.func_area  = t2.function_scope
and  lpad(t1.ledger_account,10,'0')    =  lpad(t2.gl_account,10,'0')
