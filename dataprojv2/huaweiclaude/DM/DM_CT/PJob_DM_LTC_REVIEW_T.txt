with tmp as (
SELECT DISTINCT
        fl.factory_code,
        fl.factory_name,
        p.delivery_province,
        p.delivery_city,
        p.delivery_area as delivery_area1,

        (
        SELECT
                fp.factory_code,
                fp.factory_name,
                fp.delivery_area,
                fp.delivery_area_name,
                l.first_choice_transport_way,
                l.DAY,
                l.logistics_transport_way_id,
                fp.province,
                fp.city,
                fp.area,
                fp.product_specification,
                fp.product_hierarchy,
                l.send_province,
                l.send_city,
                fp.supply_code
        FROM
                (
                SELECT  distinct 
                        p.factory_code,
                        p.factory_name,
                        p.delivery_area,
                        p.delivery_area_name,
                        p.product_specification,
                        p.product_hierarchy,
                        f.province,
                        f.city,
                        f.area,
                        f.supply_code
                FROM sdi.sdi_product_area_1033 p
                LEFT JOIN sdi.sdi_factory_area_1033 f         
                ON p.factory_code = f.factory_code
                AND p.factory_name = f.factory_name     
                
  			    WHERE p.delete_flag = 0
                  AND f.delete_flag = 0
--                  AND p.product_specification = '600X300'
--                  AND p.product_hierarchy = '1002'
----                  AND p.province =  '重庆市'            
--                  and p.factory_code like concat('%8A20%')
--                  and p.factory_name like concat('%四川省达州市大竹县%')            
                ) fp
                LEFT JOIN sdi.sdi_logistics_transport_way_1033 l ON fp.province = l.send_province
                AND l.delete_flag = 0
                AND IF(l.send_city IS NOT NULL and l.send_city!='', l.send_city = fp.city,1 = 1)
--                AND l.bms_sku_type_name = '瓷砖'
--                AND l.destination_province =  '陕西省'
--                AND IF(l.destination_city IS NOT NULL and l.destination_city!='',l.destination_city = '宝鸡市',1 = 1)
        ) fl
        LEFT JOIN sdi.sdi_logistics_transport_price_1033 p ON fl.province = p.take_delivery_province
    	AND p.delete_flag = 0
        AND fl.city = p.take_delivery_city
        AND IF(p.take_delivery_area IS NOT NULL and p.take_delivery_area != '', fl.area = p.take_delivery_area,1 = 1)
--        AND p.delivery_province = '陕西省'
--        AND p.delivery_city = '宝鸡市'
--        AND IF(p.delivery_area IS NOT NULL and p.delivery_area!='', p.delivery_area = '',1 = 1)
--        AND p.bms_sku_type_name = '瓷砖'
        and fl.first_choice_transport_way = p.ship_method_name
    	AND IF(p.start_date IS NOT NULL and p.start_date!='', DATE_FORMAT(p.start_date,'%Y-%m-%d') <= DATE_FORMAT(CURRENT_DATE() ,'%Y-%m-%d'),1 = 1)
    	AND IF(p.end_date IS NOT NULL and p.end_date!='', DATE_FORMAT(p.end_date,'%Y-%m-%d')>= DATE_FORMAT(CURRENT_DATE(),'%Y-%m-%d'),1 = 1)
--    	order by price desc, factory_code asc
)
