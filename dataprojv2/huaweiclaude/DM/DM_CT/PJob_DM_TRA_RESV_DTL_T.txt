--性能优化SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH T AS(
SELECT DISTINCT
       B.DMAND_NUM --要货单号
      ,A.DELV_NUM --提货单号
  FROM DWRLTC.DWR_BO_DSPT_ORDER_F A
 INNER JOIN DWRLTC.DWR_BO_DMAND_ORDER_F B
    ON A.DMAND_ID = B.DMAND_ID
   AND A.DMAND_LINE_ID = B.DMAND_LINE_ID	
),T1 AS(
SELECT DISTINCT 
       DELV_NUM
	  ,SALES_NUM
  FROM DWROTD.DWR_OTD_LOADING_PLAN_T
 WHERE DELV_NUM <> '' 
)
INSERT OVERWRITE TABLE DM.DM_TRA_RESV_DTL_T PARTITION(ORDER_CAT_NAME)
SELECT T1.SALES_NUM AS SAP_ORDER_NUM --SAP订单号
      ,B.DMAND_NUM --要货单号
      ,A.DELIVERY_ORDER_NUMBER AS DELV_NUM --提货单号
      ,A.RESERVATION_DATE AS RESV_DATE --预约时间
	  ,CASE WHEN E.RECV_TIME IS NOT NULL THEN '已接单' ELSE '未接单' END AS ACC_ST_NAME --仓库接单状态
	  ,E.RECV_TIME AS ACC_TIME --接单时间
	  ,A.RESERVATION_STATE_DESC AS RESV_ST_NAME --预约状态
	  ,A.DRIVER_PRESENCE_STATUS AS DRVR_PRES_ST_NAME --司机到厂状态
	  ,A.SIGN_TIME --司机到场时间
	  ,C.PICKING_CONFIRM_STATUS_NAME AS PICK_ST_NAME --系统拣配状态
	  ,C.PICKING_DATE||' '||SUBSTR(C.PICKING_TIME,1,2)||':'||SUBSTR(C.PICKING_TIME,3,2)||':'||SUBSTR(C.PICKING_TIME,5,2) AS PICK_CREATE_DATE --生成拣配单时间
	  ,A.CAR_LOADING_STATUS AS LOAD_ST_NAME --装车状态
	  ,A.LOAD_END_TIME --装车完成时间
	  ,A.LEAVE_TIME --离场时间
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.LOAD_END_TIME) - UNIX_TIMESTAMP(A.SIGN_TIME)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.LOAD_END_TIME) - UNIX_TIMESTAMP(A.SIGN_TIME)),86400)) / 3600,0) || '小时','.0','') AS DELV_CNSUM_TIME --发货耗时 等于装车完成时间-司机到厂时间
	  ,D.ACTUAL_DELIVERY_QTY AS ACTL_DELV_QTY --出库数量
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN B.DMAND_NUM IS NOT NULL THEN '自营工程'
	        ELSE '零售订单'
			 END AS ORDER_CAT_NAME --订单分类
  FROM DWROTD.DWR_TM_RESERVATION_RECORD_DETAIL_F A
  LEFT JOIN T AS B
    ON A.DELIVERY_ORDER_NUMBER = B.DELV_NUM
  LEFT JOIN DWIOTD.DWI_WM_LADINGBILL_HEADER_T C
    ON A.DELIVERY_ORDER_NUMBER = C.DELIVERY
  LEFT JOIN(SELECT DELIVERY
                  ,SUM(ACTUAL_DELIVERY_QTY) AS ACTUAL_DELIVERY_QTY
              FROM DWROTD.DWR_WM_LADINGBILL_ITEM_F
			 GROUP BY DELIVERY
		   ) D
	ON A.DELIVERY_ORDER_NUMBER = D.DELIVERY
  LEFT JOIN(SELECT DISTINCT DELV_NUM
                           ,RECV_TIME --仓库接单时间
              FROM DWROTD.DWR_TRA_TMS_SHIP_DTL_F
		   ) E
	ON A.DELIVERY_ORDER_NUMBER = E.DELV_NUM
  LEFT JOIN T1
    ON A.DELIVERY_ORDER_NUMBER = T1.DELV_NUM
 WHERE COALESCE(A.RESERVATION_STATE,'') NOT IN('CANCELED','EXIT')
