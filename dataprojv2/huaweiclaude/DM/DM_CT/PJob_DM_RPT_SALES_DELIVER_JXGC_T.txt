DELETE FROM DM.DM_RPT_SALES_DELIVER_jxgc_t;
insert into DM.DM_RPT_SALES_DELIVER_jxgc_t
with get_ys_price as
(
select  operation_desc,coalesce(prod_level3_name,'其他品类') as prod_level3_name,coalesce(dimension,'其他规格') as dimension,
datatool.division_by_zero(sum(coalesce(prod_it_amt1,0)),sum(coalesce(sales_area,0))) as ys_price--预算全年品类规格单价
FROM UPLOAD.UPLOAD_ACHIEVEMENT_BUDGET_T 
where substring(stat_month,1,4) = substring(current_date(),1,4)
group by operation_desc,coalesce(prod_level3_name,'其他品类'),coalesce(dimension,'其他规格')
)
select 
case when t1.dept_lv3_name = '国际营销' or t1.lev2_name in ('国际营销中心') then '国际营销' 
    when t1.dept_lv3_name in ('梦之家事业部','梦之家') then '梦之家'
    when t1.dept_lv5_name in ( 'Z031','幕墙工程中心') then '幕墙'
	when t1.dept_lv5_name in ('国内工程', '战略工程中心','战略地产','战略工程') then '战略地产'
	when t1.dept_lv5_name in ('商机资源开发部','战略客户部','资源开发运营部') then '战略客户部' 
    when t1.dept_lv4_name in('直营经营管理板块','区域运营中心','国内直营中心') then t1.dept_lv5_name
    else t1.dept_lv5_name
end as sub_grp_name,--销售区域	
t1.cust_num,--客编	
t1.cust_name,--客编名称	
case when expect_date_aps>current_date -1	then '否' else '是' end as isdue,--是否逾期	
t1.order_no,--订单号	
t1.order_item_no,--订单行	
t1.business_opportunity,--商机号	
t1.jhddlx,--计划订单类型	
t1.submi,--备货类型	
t1.order_type,--订单类型	
t1.werks,--下单工厂	
t1.audat,--订单下单时间	
t1.creator_name,--下单人	
t1.product_area_new_name,--产区名称	
t1.material_num,--物料编码	
t1.material_name,--物料名称	
t1.distribution_chnl_code,--渠道	
t1.operate_depart,--运营部	
t1.prod_name,--产品分类	
t1.product_level2_name,--品类名称	
t1.dimension,--规格	
ysp.ys_price as ys_price,--平均单价(平方米)	
t1.nodeliver_qty_aps,--未交付数量	
t1.nodeliver_area_aps,--未交付面积	
t1.expect_date_aps,--客户期望交期	
t1.ESTIMATED_DELIVERY_TIME,--预计交货时间	
t1.jhddlx, --订单流程名称	,
t2.wbs_top_level_desc,	--WBS工程名称 ： 使用顶层
t1.promise_delivery,--默认交期:使用的是工厂交期
t1.matl_group1_code,--物料组1
t7."name" as bus_gjr,--商机跟踪人
t1.creator_name as gc_creator_name	,
t1.zvkgrp,--销售组
t2.wbs_level3_desc,--WBS元素描述
DATE_PART('day',  current_date() -1 - to_date(expect_date_aps)) as over_days,--逾期天数
t1.gcdzs,--工程地址-省
t1.gcdzcs,--工程地址-市
t1.gcjhrq,--确认排产交付时间
t1.salesman_name_list,--业务员名称	
t1.vkgrp,--线组	
t1.dept_lv4_name as lev4_name,--销售区域L4	
t1.dept_lv5_name as lev5_name,--销售部门L5	
t1.lev6_name,--销售部门L6	
t1.lev7_name,--销售部门L7
NOW() as dw_last_update_date, --更新日期
t1.operate_group,--运营线组
t5.xmmc --工程名称
from DM.DM_RPT_SALES_DELIVER_T t1
left join dwimd.dwi_md_data_material_general_t mat on t1.material_num = mat.material_num
left join DWRDIM.DWR_DIM_WBS_BASIS_INFO_F t2 on t1.order_item_no = t2.wbs_level3
left join dm.dm_rpt_sales_group_t t4 on t1.vkgrp = t4.node_name9
left join get_ys_price ysp on t4.node_name5 = ysp.operation_desc and mat.category = ysp.prod_level3_name and mat.dimension = ysp.dimension
left join dm.dm_SDI_NEW_OPPORTUNITY_1046 t6 on t1.business_opportunity = t6.value54
left join api_log.api_to_dws_people_information t7 on t6.value88 = t7."_id"
left join dm.dm_sdi_zsd_sjxx_1001_t t5 on t1.business_opportunity = t5.sjh
where 1=1
and t1.stat_month = replace(substring(current_date()-1,1,7),'-','')
and t1.distribution_chnl_code = '06'
and mat.external_matl_group_code in ('10010')
and mat.matl_group_code in ('1000010')
and t1.nodeliver_qty_aps <> 0
