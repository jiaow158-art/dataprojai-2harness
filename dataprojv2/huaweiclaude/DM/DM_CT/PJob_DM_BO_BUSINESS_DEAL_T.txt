set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;

WITH MAT_DATA AS(	 --- 物料数据
    select t1.material_num,
           t1.width * t1.length_1/1000/1000 as area
    from dwimd.dwi_md_data_material_general_t t1 
    where matl_group_code ='1000010'
 ),
DH_DATA_01 as(   -- 商机数据
	SELECT T2.DH,
		   T1.MATNR,
		   SUM(T3.LFIMG) AS LFIMG
	FROM SDI.SDI_VBAP_1001 T1
	LEFT JOIN SDI.SDI_VBAK_1001 T2
	ON T1.VBELN = T2.VBELN
	AND T2.del_flag = 'N'
	LEFT JOIN SDI.SDI_LIPS_1001 T3
	ON T1.VBELN = T3.VGBEL 
	AND T1.POSNR = T3.VGPOS 
	AND T3.del_flag = 'N'
	LEFT JOIN SDI.SDI_LIKP_1001 T4
	ON T3.VBELN = T4.VBELN 
	AND T4.del_flag = 'N'
	WHERE T4.WBSTK = 'C'
	AND T2.DH <> ''
	AND T1.del_flag = 'N'
	GROUP BY T2.DH,
		 	 T1.MATNR
),
DH_DATA_02 as(
	select t1.DH AS business_code, 
	       case when nvl(sum(t1.LFIMG * t2.area),0) > 1000 then 'Y' else 'N' END as change_flag
	from DH_DATA_01 t1
	left join MAT_DATA t2
	on t1.MATNR = t2.material_num
	group by t1.DH
)
INSERT OVERWRITE TABLE DM.DM_BO_BUSINESS_DEAL_T
SELECT T.GSDM as company_code, -- 公司代码
       t3.company_name, -- 公司名称
       t2.operation_code, -- 运营中心编码
       t2.operation_name, -- 运营中心名称
       CASE WHEN SUBSTRING(T.KHBM,1,4) = '0000' THEN SUBSTRING(T.KHBM,5)
	       ELSE T.KHBM
	       END CUSTOMER_CODE, -- 客户编码
	   t4.cust_name as customer_name, -- 客户名称
	   T.SJJD as sales_stage ,-- 销售阶段
	   T.SQRQ as apply_date, -- 申请时间
	   T.SJYXQ as business_Valid, -- 商机有效期
	   T.SHENG as province, -- 省
	   T.SHI as city, -- 市
	   T.SJLX as business_type, -- 商机类型
	   T.LCLX as flow_type,   -- 流程类型
	   T.SJH as business_code, -- 商机号
	   NVL(T1.change_flag,'N') as change_flag, -- 是否已转换  Y:是  N：否
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM SDI.SDI_ZSD_SJXX_1001 T
left join DH_DATA_02 t1
on T.SJH = t1.business_code
left join DWIMD.DWI_MD_DATA_OPERATION_CUST_R t2
on CASE WHEN SUBSTRING(T.KHBM,1,4) = '0000' THEN SUBSTRING(T.KHBM,5)
	       ELSE T.KHBM
	       END = t2.customer_code
left join 
   (SELECT T.company_code ,t.company_name
	  FROM DWRDIM.dwr_dim_company_d T 
	  WHERE t.group_num = '800'
	  and T.end_date = '4712-12-31'
   ) t3
on T.GSDM = t3.company_code
LEFT JOIN 
    (SELECT T.cust_num ,T.cust_name 
	  FROM DWRDIM.dwr_dim_cust_general_d T
	  WHERE T.group_num = '800'
	  AND T.end_date = '4712-12-31'
    ) t4
on CASE WHEN SUBSTRING(T.KHBM,1,4) = '0000' THEN SUBSTRING(T.KHBM,5)
	       ELSE T.KHBM
	       END = t4.cust_num
WHERE T.del_flag = 'N'
and T.SQRQ >='20200101';
