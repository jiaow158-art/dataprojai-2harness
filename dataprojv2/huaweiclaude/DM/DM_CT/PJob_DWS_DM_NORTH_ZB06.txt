--来源表dm.dm_ambv2_profit_t  毛利率-标A毛利率（工程渠道）
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB06';

with ambv2_profit as (
select 
T1.calmonth,
sum(
                                                                case
                                                                      when T1.calmonth = SUBSTRING(current_date,1,7)
                                                                          then
                                                                          case
                                                                              when coalesce(T1.prod_property, '#') = 'A0001' and
                                                                                   coalesce(T1.cust_group, '#') != 'Z0'
                                                                                  THEN
                                                                                  case
                                                                                      when T1.data_source in ('S', 'T')
                                                                                          then coalesce(T1.standard_price_a, 0)
                                                                                      else coalesce(T1.last_period_sum_price, 0) + coalesce(zzyywcb_azf, 0) end
                                                                              else coalesce(T1.last_period_sum_price, 0) + coalesce(zzyywcb_azf, 0) end
                                                                      else
                                                                          case
                                                                              when coalesce(T1.prod_property, '#') = 'A0001' and
                                                                                   coalesce(T1.cust_group, '#') != 'Z0'
                                                                                  THEN
                                                                                  case
                                                                                      when T1.data_source in ('S', 'T')
                                                                                          then coalesce(T1.standard_price_a, 0)
                                                                                      else T1.act_cost_sum_amt - T1.zyywcb_zxf - T1.zyywcb_yf - T1.zcbtzz - T1.zzyywcb_azf end
                                                                              else T1.act_cost_sum_amt - T1.zyywcb_zxf - T1.zyywcb_yf - T1.zcbtzz - T1.zzyywcb_azf end end
                                                             ) as ZBACBHJ,
SUM(notax_sales_net_amt) as    notax_sales_net_amt                                     
from dm.dm_ambv2_profit_t T1
left join  
dm.dm_rpt_sales_group_t  t2 
on t1.sales_grp = t2.node_name9
where T2.node_name3 = 'H03230618'
and  T1.integrate_channel = 'GD03'
and  T1.data_source in ('S','T','D','U')
and  T2.node_name5 not in ('H03230605','H03230612','H03230604','H03230627')
and  T2.node_name6 not in ('H03230605','H03230627')
group by        
calmonth
)
,ambv2_profit2 as (
select 
calmonth ,
'ZB06' as zb_code,
case when notax_sales_net_amt = 0 then 0 else (notax_sales_net_amt - ZBACBHJ)/notax_sales_net_amt end as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
ambv2_profit
where calmonth = '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB06' as zb_code,
0 as 当月实绩,
case when SUM(notax_sales_net_amt) = 0 then 0 else(SUM(notax_sales_net_amt) - SUM(ZBACBHJ))/SUM(notax_sales_net_amt) end as 累计实绩,
0 as 季度累计实绩
from 
ambv2_profit
where CALMONTH between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB06' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
case when SUM(notax_sales_net_amt) = 0 then 0 else(SUM(notax_sales_net_amt) - SUM(ZBACBHJ))/SUM(notax_sales_net_amt) end as 季度累计实绩
from 
ambv2_profit
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)*100 as CM_STAT_AMT,
sum(累计实绩)*100 as ACC_STAT_AMT,
sum(季度累计实绩)*100 as QTR_TGT_AMT
from 
ambv2_profit2
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.PJFX
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
