with A as(
select  
	case when  substr(stat_month,-2,2)='01' then  substr(stat_month,1,4)||'02' else stat_month end as start_month,
	case 
		when sub_name in ('国贸','国际营销') then  '国际营销'
		else sub_name
	end as sub_name,
	case 
		when sub_grp_name in ('国贸','国际营销') then  '国际营销'
		else sub_grp_name
	end as sub_grp_name,
	sum(zxsyj_dy) as ndn_amount_dy, --实际金额
	sum(zxsmj_dy) as ndn_area_dy, --实际面积
	sum(yc_area) as ndn_area_yc, --预测面积
	sum(yc_amount) as ndn_amount_yc--预测金额
from DM.DM_RPT_DEMAND_FORECAST_T
where 1=1
-- and sub_grp_name not in ('ART+')
and case when  substr(stat_month,-2,2)='01' then  substr(stat_month,1,4)||'02' else stat_month end =
	case 
		when  substr(REPLACE(SUBSTR(date('${PERIOD_ID_D}')  ,1,7),'-',''),-2,2)='01' 
		then  substr(REPLACE(SUBSTR(date('${PERIOD_ID_D}')  ,1,7),'-',''),1,4)||'02' 
		else REPLACE(SUBSTR(date('${PERIOD_ID_D}')  ,1,7),'-','')
	end
group by start_month,sub_name,sub_grp_name
)

where stat_month >= replace(SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7),'-','')

		else REPLACE(SUBSTR(date('${PERIOD_ID_D}')  ,1,7),'-','')

from dm.dm_rpt_sales_group_t
where node_name5 in ('H03230604','H03230605','H03230612'))
)
