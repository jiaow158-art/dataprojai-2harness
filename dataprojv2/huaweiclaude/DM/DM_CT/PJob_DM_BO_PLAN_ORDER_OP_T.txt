WITH T AS(
SELECT MATERIAL_NUM
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '01A' THEN OP_TIME END) AS OP_DATE_01A --O1A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '06A' THEN OP_TIME END) AS OP_DATE_06A --06A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '07A' THEN OP_TIME END) AS OP_DATE_07A --07A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '15A' THEN OP_TIME END) AS OP_DATE_15A --15A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '05A' THEN OP_TIME END) AS OP_DATE_05A --05A节点操作日期
      ,MAX(CASE WHEN SUBSTR(OP_NODE_NAME,1,3) = '04B' THEN OP_TIME END) AS OP_DATE_04B --04B节点操作日期
  FROM DWRPI.DWR_WF_CSTM_PROD_NODE_F
 WHERE MATERIAL_NUM IS NOT NULL
 GROUP BY MATERIAL_NUM
)
INSERT OVERWRITE TABLE DM.DM_BO_PLAN_ORDER_OP_T
SELECT 
       REPLACE(TRUNC((UNIX_TIMESTAMP(D.OP_DATE_06A) - UNIX_TIMESTAMP(D.OP_DATE_01A)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(D.OP_DATE_06A) - UNIX_TIMESTAMP(D.OP_DATE_01A)),86400)) / 3600,0) || '小时','.0','') AS TOT_EDIT_CNSUM_TIME --总对版时长
      ,REPLACE(TRUNC((UNIX_TIMESTAMP(D.OP_DATE_07A) - UNIX_TIMESTAMP(D.OP_DATE_15A)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(D.OP_DATE_07A) - UNIX_TIMESTAMP(D.OP_DATE_15A)),86400)) / 3600,0) || '小时','.0','') AS CNFRM_CNSUM_TIME --业务确认样板耗时
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(D.OP_DATE_05A) - UNIX_TIMESTAMP(D.OP_DATE_04B)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(D.OP_DATE_05A) - UNIX_TIMESTAMP(D.OP_DATE_04B)),86400)) / 3600,0) || '小时','.0','') AS FR_CNSUM_TIME --基地仿版耗时
	  ,CASE WHEN E.MATERIAL_NUM IS NOT NULL THEN '是' ELSE '否' END AS IS_TEMPL_NAME --有无样板
	  ,C.PAR_BUSI_CODE AS MTH_BUSI_CODE --母商机编号
	  ,A.WBS_NUM --工程WBS号（首层）
	  ,A.PLAN_NUM --计划订单流水号
	  ,A.PLAN_ITEM_NUM --计划单行号（三层WBS号）
	  ,A.WBS_NAME --WBS元素描述
	  ,A.MATERIAL_NUM --物料编码
	  ,A.MATERIAL_NAME --物料名称
	  ,A.FACTORY_CODE --工厂
	  ,A.SPLD_AREA_NAME --供货产区
	  ,A.ORDER_QTY --评审后需求数量
	  ,COALESCE(T4.RSDU_PLAN_QTY,0) AS RSDU_PLAN_QTY --项目剩余需求数量
	  ,COALESCE(T4.RSDU_PLAN_AREA) AS RSDU_PLAN_AREA --剩余需求数量面积
	  ,A.CNFRM_PROD_DATE --确认排产交付日期
	  ,A.EXPCT_DELV_DATE --到达项目地日期
	  ,A.APPR_STATUS_NAME --头业务审批状态
      ,A.PLAN_STATUS_NAME --头计划单状态
	  ,A.APPR_START_TIME --业务审批开始时间
	  ,A.APPR_END_TIME --业务审批结束时间
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.APPR_END_TIME) - UNIX_TIMESTAMP(A.APPR_START_TIME)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.APPR_END_TIME) - UNIX_TIMESTAMP(A.APPR_START_TIME)),86400)) / 3600,0) || '小时','.0','') AS APPR_CNSUM_TIME --业务部门审批时效
	  ,A.IS_SPEC_PROJ_NAME --是否特殊工程
	  ,A.IS_CHAIN_NAME --是否连锁网点
	  ,A.PLAN_ORDER_TYPE_NAME --计划订单类型
	  ,A.SUB_BUSI_CODE --子商机编码
	  ,A.MD_SALE_GRP_NAME AS SALE_GRP_CODE --销售组
	  ,A.SALE_EMP_NAME --WBS销售人员
	  ,A.CREATE_EMP_NAME --计划订单创建者名称
	  ,A.CREATE_DATE --计划订单创建日期
	  ,A.SUBMIT_TIME --计划订单提交日期
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.SUBMIT_TIME) - UNIX_TIMESTAMP(A.CREATE_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.SUBMIT_TIME) - UNIX_TIMESTAMP(A.CREATE_DATE)),86400)) / 3600,0) || '小时','.0','') AS ORDER_CNSUM_TIME --业务部门下单时效
	  ,A.OP_DEPT_NAME --运营部
	  ,A.OP_GRP_NAME --运营线组
	  ,A.RFUS_RSN --拒绝原因
	  ,A.APPR_TIME --评审通过日期
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.APPR_TIME) - UNIX_TIMESTAMP(A.APPR_END_TIME)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.APPR_TIME) - UNIX_TIMESTAMP(A.APPR_END_TIME)),86400)) / 3600,0) || '小时','.0','') AS SCM_CNSUM_TIME --供应链评审时效
	  ,A.CUST_NUM --客户编号
	  ,A.PAY_CUST_NUM --付款方编号
	  ,A.SALE_CHCL_CODE --分销渠道
	  ,A.SALE_ORG_CODE --销售组织
	  ,F.EST_DELV_TIME --预计交付时间
	  ,F.PROG_ST_NAME --订单进度状态
	  ,COALESCE(B.RMN_STOCK_QTY,0) AS RMN_STOCK_QTY --留货数量汇总
	  ,CASE WHEN A.APPR_QTY > COALESCE(B.RMN_STOCK_QTY,0) + COALESCE(B.OUT_STOCK_ZT_QTY,0) THEN '待留货'
	        WHEN COALESCE(A.APPR_QTY,0) - COALESCE(B.RMN_STOCK_QTY,0) - COALESCE(B.OUT_STOCK_ZT_QTY,0) <= 0
			THEN '已留货'
	         END AS RMN_ST_NAME --留货状态 当 评审后需求数量-留货数量汇总-出库数<=0  已留货
                                --评审后需求数量-留货数量汇总-出库数>0  待留货
	  ,B.OUT_STOCK_ZT_QTY AS TOT_OUT_QTY --累计出库数量
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,A.PLAN_ITEM_STATUS_NAME --计划单行状态
      ,A.ENGR_NAME --工程名称
  FROM DWRLTC.DWR_BO_PLAN_ORDER_F A
  LEFT JOIN(SELECT WBS_LEVEL3
                  ,SUM(CASE WHEN RMN_STOCK_QTY > 0 THEN RMN_STOCK_QTY ELSE 0 END) AS RMN_STOCK_QTY
				  ,SUM(OUT_STOCK_ZT_QTY) AS OUT_STOCK_ZT_QTY
              FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T
             GROUP BY WBS_LEVEL3
           ) B
	ON A.PLAN_ITEM_NUM = B.WBS_LEVEL3
  LEFT JOIN DWRLTC.DWR_BO_BUSI_INFO_F C
    ON A.SUB_BUSI_CODE = C.BUSI_CODE
  LEFT JOIN T AS D
    ON A.MATERIAL_NUM = D.MATERIAL_NUM
  LEFT JOIN (SELECT DISTINCT MATERIAL_NUM,SITE_CODE
               FROM DWROTD.DWR_PAM_GENE_BOARD_INFO_T
			) AS E
    ON A.MATERIAL_NUM = E.MATERIAL_NUM
   AND A.FACTORY_CODE = E.SITE_CODE
  LEFT JOIN (SELECT DISTINCT WBS_LINE_NO
                            ,PLAN_DELV_TIME AS EST_DELV_TIME
							,PROG_ST_NAME
			   FROM DWROTD.DWR_PS_SALE_ORDER_DTL_F
              WHERE WBS_LINE_NO <> ''
			    AND WBS_LINE_NO IS NOT NULL
            ) F
	ON A.PLAN_ITEM_NUM = F.WBS_LINE_NO
  LEFT JOIN DWROTD.DWR_OTD_RSDU_PLAN_F T4
    ON A.PLAN_ITEM_NUM = T4.PLAN_ITEM_NUM
 WHERE A.DIV_NAME = '瓷砖事业部'
;
