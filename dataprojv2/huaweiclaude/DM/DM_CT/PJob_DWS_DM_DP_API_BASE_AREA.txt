DELETE FROM dm.dm_dp_api_base_area;
INSERT INTO dm.dm_dp_api_base_area
with pre_cate_dimension as 
(
select 
	dp_cate -- 预测品类
	,dp_dimension -- 预测规格
	,fin_cate -- 财务品类
	,fin_dimension -- 财务规格
from dwrotd.dwr_czdp_basic_cate_t
where 1=1
and delete_flag = 0
group by 
	dp_cate -- 预测品类
	,dp_dimension -- 预测规格
	,fin_cate -- 财务品类
	,fin_dimension -- 财务规格
)
,base_f as (
select 
    batch_start
    ,product_area_name
    ,FACTORY_TYPE
from DWRDIM.DWR_DIM_PRODUCT_AREA_D
group by 
    batch_start
    ,product_area_name
    ,FACTORY_TYPE
)

,tem as (
select 
    substring(a.calmonth,1,4) as stat_year    -- 年月
    ,a.calmonth as stat_month    -- 年月
    ,e.sales_center_code -- 业务部门代码
    ,e.sales_center -- 业务部门
    ,e.sales_dep_code   -- 营销部代码
    ,e.sales_dep   -- 营销部
    ,e.sales_region_code   -- 销区代码
    ,e.sales_region   -- 销区
    ,coalesce(c.lve2_code,'其他品类') as fin_cate2   -- 财务品类 转为其它瓷砖？
    ,coalesce(a.size_dim, '其他规格')  as dimension   -- 财务规格
    ,f.dp_cate -- 预测品类
    ,f.dp_dimension -- 预测规格
    ,zxsmj          -- 面积（平方米）
	,a.data_source
	,a.batch -- 批次
	,a.material
	,coalesce(g2.product_area_name,g1.product_area_name,'广东基地') as base
	,case 
		when coalesce(g2.FACTORY_TYPE,g1.FACTORY_TYPE) in ('自制') then 'N'
		when coalesce(g2.FACTORY_TYPE,g1.FACTORY_TYPE,'NAL') in ('NAL','NA','外协') then 'Y'
	end as oem_flag
from dm.dm_ambv2_zzt_xssr_t a
join dm.dm_dp_api_sales_group_struct e on a.sales_grp = e.sales_grp
    left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
    left join dwimd.dwi_md_data_material_general_t c0 on a.material = c0.material_num
    left join DM.DM_RPT_PROD_LEVEL_T c on coalesce(c0.product_level3_code,c0.product_level2_code,c0.product_level1_code) = c.prod_code3 and c.PROD_CODE3 IS NOT NULL
    left join DWROTD.DWR_SO_HEAD_SAP_F d on a.doc_number = d.sal_vch_num
    left join pre_cate_dimension f on coalesce(c.lve2_code,'其他品类')  = f.fin_cate and coalesce(a.size_dim, '其他规格') = f.fin_dimension
    left join base_f g1 on replace(substring(a.batch,1,2),'OB','0B') = g1.batch_start and  length(g1.batch_start)=2 and g1.batch_start is not null
    left join base_f g2 on replace(substring(a.batch,1,3),'OB','0B') = g2.batch_start and  length(g2.batch_start)=3 and g2.batch_start is not null
where 1=1 
and (NODE_NAME2 = 'H11000001' or (coalesce(b.node_name2, '#') = '#' AND A.PROD_PROPERTY_NAME IN ('整装产品', '瓷砖产品')))
and a.calmonth = SUBSTR(TO_DATE('${PERIOD_ID_D}', 'YYYY-MM-DD') - INTERVAL '1 month', 1, 7)
)

select 
	stat_year
	,stat_month
	,sales_center_code
	,sales_center
	,dp_cate
	,dp_dimension
	,base
	,oem_flag
	,sum(zxsmj) as area
	,current_timestamp() as dw_last_update_date
    ,stat_month as stat_month_ym
from tem
where 1=1 
and data_source = 'S'
group by 
	stat_year
	,stat_month
	,sales_center_code
	,sales_center
	,dp_cate
	,dp_dimension
	,base
	,oem_flag
