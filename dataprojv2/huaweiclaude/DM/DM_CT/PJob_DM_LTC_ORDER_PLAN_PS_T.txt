--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
WITH BASE_COMPLETE_QTY AS(  -- 基地完成数量
	select t.plan_task_id,  -- 计划任务
	       t.process_tech_order_id, -- 制程工艺单
	       sum(t.qty) qty --基地完成数量
	from sdi.sdi_APS_DAILY_SCHEDULE_1007 t  -- 生产日进度
	where del_flag = 'N'
	group  by t.plan_task_id ,
	        t.process_tech_order_id
),
OEM_COMPLETE_QTY AS( -- OEM完成数量,SAP获取
   SELECT AN.BANFN,
		sum(CASE WHEN BE.SHKZG = 'S' THEN BE.MENGE 
		        ELSE 0 - BE.MENGE 
		     END) AS MENGE -- OEM完成数量
	FROM SDI.SDI_EBAN_1001 AN  
	left JOIN SDI.SDI_EKPO_1001 PO 
	  ON PO.BANFN = AN.BANFN 
	  AND PO.BNFPO = AN.BNFPO 
	  AND PO.del_flag = 'N'
	INNER JOIN SDI.SDI_EKKO_1001 KO 
	  ON KO.EBELN = PO.EBELN 
     AND KO.del_flag = 'N'	  
	LEFT JOIN SDI.SDI_EKBE_1001 BE 
	  ON PO.EBELN = BE.EBELN 
	  AND PO.EBELP = BE.EBELP 
      AND BE.del_flag = 'N'	  
	WHERE AN.BNFPO = '00010'
	AND AN.BSART = 'ZNB1' 
	AND BE.BEWTP = 'E' 
	AND AN.del_flag = 'N'
	group by AN.BANFN 
),
PRODUCT_LEVEL2_DATA AS( -- 产品层级2数据
   SELECT X5.MATERIAL_NUM
      ,X5.MATERIAL_ID
      ,X6.VTEXT
  FROM DWRDIM.DWR_DIM_MATERIAL_GENERAL_D X5 --物料
  LEFT JOIN SDI.SDI_T179T_1001 X6 --产品品类
    ON X6.MANDT = X5.GROUP_NUM
   AND X6.SPRAS = '1'
   AND X6.del_flag = 'N'
   AND SUBSTR(X5.PRODUCT_LEVEL_CODE,1,4) = X6.PRODH
 WHERE X5.END_DATE = '4712-12-31'
   AND X5.DEL_FLAG = 'N'
),
ORDER_DETAIL_DATA AS(    -- 订单明细数据
select  T7.SALE_NO AS SALE_ORDER_HEAD_NO,--销售订单号
		T7.SALE_LINE_NO AS SALE_ORDER_LINE_NO,--销售订单行号
        b.PLAN_NO,  -- 计划单号
        srws1.NAME AS WIP_RESOURCE_NAME, --生产资源名称
		srws1.ID AS WIP_RESOURCE_ID, --生产资源ID
		sre1.NAME AS WORK_SHOP_NAME,  --车间名称
		sre1.ID AS WORK_SHOP_ID,  --车间ID
		sre2.NAME AS BASE_AREA, --工厂名称
		sre2.ID AS BASE_AREA_ID, --工厂ID
		sre3.NAME AS PRODUCT_AREA,  --产区名称
		sre3.ID AS PRODUCT_AREA_ID,  --产区ID
		T7.IS_CUSTOM AS CUSTOM_PROPERTY, --定制标识
		T7.CODE AS PRODUCT_ORDER_CODE, --生产工单号
		T7.REMARK AS PRODUCT_ORDER_REMARK,--生产订单备注
		b.START_TIME as PLANT_START_TIME,  -- 计划开始时间
		b.END_TIME as PLANT_END_TIME, -- 计划完成时间
		b.STATUS as PLANT_STATUS, -- 计划状态
		T7.REQUIRE_DELIVERY AS  DEMAND_DELIVERY_DATE, -- 客户交期
		a.RELEASE_START_TIME, --下单时间
		COALESCE(T15.QTY,T16.MENGE)  as COMPLETE_QTY,--已完成数量
		CASE WHEN a.PLAN_AMOUNT - COALESCE(T15.QTY,T16.MENGE) < 0 THEN 0
      		ELSE a.PLAN_AMOUNT - COALESCE(COALESCE(T15.QTY,T16.MENGE),0)
	   	END  as NO_COMPLETE_QTY,--未完成数量
	   	a.PLAN_AMOUNT,--计划数量
		t7.item_id,
		t7.order_type_set_id,
		t7.customer_id,
		case when b.RE_STATE = 0 then '否'
             when b.RE_STATE = 1 then '是'
			  end as RE_STATE, --复版状态
		b.RE_DATE --复版日期
from SDI.SDI_APS_PROTECH_ORDER_1007 t6  --制程工艺单
 left join SDI.SDI_APS_PRO_ORDER_1007 t7 --生产工单
 on t6.product_order_id = t7.id 
 and t7.is_phantom = 0 
 AND T7.del_flag = 'N'
 left join SDI.SDI_APS_PLAN_TASK_dtl_1007 a  -- 任务计划明细
 on  t6.id = a.process_tech_task_id
 and t6.is_phantom = 0 
 AND A.del_flag = 'N'
 join SDI.SDI_APS_PLAN_TASK_1007 b  -- 计划任务
 on a.Plan_Task_Id=b.id -- 计划任务
 and b.is_cancel = 0 
 AND B.del_flag = 'N'
 left join sdi.sdi_res_wip_sche_1007 srws1  -- 生产资源
 on b.resource_id = srws1.id 
 and srws1.is_phantom = 0
 AND SRWS1.del_flag = 'N'
 left join sdi.sdi_RES_ENTERPRISE_1007 sre1  --车间
 on srws1.work_shop_id = sre1.id 
 and sre1.is_phantom = 0
 AND SRE1.del_flag = 'N'
 left join sdi.sdi_RES_ENTERPRISE_1007 sre2  --工厂
 on sre1.tree_pid = sre2.id  
 and sre2.is_phantom = 0
 AND SRE2.del_flag = 'N'
 left join sdi.sdi_RES_ENTERPRISE_1007 sre3  --产区
 on sre2.tree_pid = sre3.id 
 and sre3.is_phantom = 0 
 AND SRE3.del_flag = 'N'
 left join BASE_COMPLETE_QTY t15  --OEM完成数量 从SAP获取
	on a.plan_task_id = t15.plan_task_id
    and a.PROCESS_TECH_TASK_ID = t15.process_tech_order_id 
 left join OEM_COMPLETE_QTY t16 
	on a.WORK_ORDER = t16.BANFN
),TT AS(
SELECT A.POSNR AS SAP_ITEM_NUM
 ,D.POSID AS PLAN_ITEM_NUM
 ,B.POSNR AS SAP_PLAN_NUM
 ,E.POSID AS PLAN_NUM 
 ,C.POSNR AS SAP_WBS_NUM
 ,F.POSID AS WBS_NUM  
 FROM SDI.SDI_PRHI_1001 A
 LEFT JOIN SDI.SDI_PRHI_1001 B
 ON A.UP = B.POSNR
 AND A.MANDT = B.MANDT
 AND B.DEL_FLAG = 'N'
 LEFT JOIN SDI.SDI_PRHI_1001 C
 ON B.UP = C.POSNR
 AND C.MANDT = B.MANDT
 AND C.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 D
 ON A.POSNR = D.PSPNR
 AND A.MANDT = D.MANDT
 AND D.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 E
 ON B.POSNR = E.PSPNR 
 AND E.MANDT = B.MANDT
 AND E.DEL_FLAG = 'N' 
 LEFT JOIN SDI.SDI_PRPS_1001 F
 ON C.POSNR = F.PSPNR  
 AND C.MANDT = F.MANDT
 AND F.DEL_FLAG = 'N' 
 WHERE A.MANDT = '800'
 AND A.DEL_FLAG = 'N'
)
INSERT OVERWRITE TABLE DM.DM_LTC_ORDER_PLAN_PS_T
select  t1.SALE_NO AS SALE_ORDER_HEAD_NO, -- 销售单号（或者计划单号）
        t1.line_no AS SALE_ORDER_LINE_NO, -- 销售单行号
        TT.WBS_NUM,   -- WBS编码
        sp1.post1 AS WBS_NAME_CN, -- WBS名称
        t.PLAN_NO, -- 报工号
		t.PRODUCT_AREA_ID,  --产区ID
		t.PRODUCT_AREA as PRODUCT_AREA_NAME_CN,  --产区名称
		t.BASE_AREA_ID as FACTORY_ID, --工厂ID
		t.BASE_AREA as FACTORY_NAME_CN, --工厂名称
		t.WORK_SHOP_ID as WORKSHOP_ID,  --车间ID
		t.WORK_SHOP_NAME as WORKSHOP_NAME_CN,  --车间名称
		t.WIP_RESOURCE_ID as PRODUCT_LINE_ID, --生产资源ID(产线)
		t.WIP_RESOURCE_NAME as PRODUCT_LINE_NAME_CN, --生产资源名称(产线)
		t.CUSTOM_PROPERTY, --定制标识
		T11.CODE AS MATERIAL_NUM,--物料编码
		T11.NAME AS MATERIAL_NAME_CN,--物料名称
		T11.SPECIFICATION_MODEL as MATERIAL_SPECIFICATIONS,--物料规格
		T11.PRODUCT_PROPERTY as MATERIAL_GROUP1,  --物料组1
		T1.MAT_GRP2 as MATERIAL_GROUP2,--物料组2
		T11.PRODUCT_PROPERTY AS PRODUCT_PROPERTY,  --产品属性
		T17.VTEXT AS PRODUCT_LEVEL2_NAME_CN, --  产品层次2
		t.PRODUCT_ORDER_CODE, --生产工单号
		t.PRODUCT_ORDER_REMARK,--生产订单备注
		T1.REMARK AS SALE_REMARK,--销售订单行备注
		t.PLANT_START_TIME,  -- 计划开始时间
		t.PLANT_END_TIME, -- 计划完成时间
		case when t.PLANT_STATUS = '10' then '未完成'
		     when t.PLANT_STATUS = '20' then '已下达'
		     when t.PLANT_STATUS = '30' then '已完成'
		end as PLANT_STATUS, -- 计划状态
		T12.CODE AS CUSTOMER_NUM, --客户编码
		T12.NAME AS CUSTOMER_NAME_CN,--客户名称
		t.DEMAND_DELIVERY_DATE as CUSTOMER_DELIVERY_DATE, -- 客户交期
		T14.NAME AS ORDER_TYPE, --订单类型
		T11.EMBRYO,--胚料
		T11.GLAZE,--釉面
		T11.GRAYSCALE,--灰度
		T11.TECHNOLOGY,--工艺
		T11.MOULD,  --模具
		T1.SPECIAL_REQUIREMENT,--特殊要求
		T1.PACKAGING_CODE,--包装箱打特殊编码
		T1.SPECIAL_PACKAGING,--特殊包装
		T1.TRADEMARK,--特殊砖底商标
		T1.SPECIAL_SIZE,--尺寸
		T1.SPECIAL_THICKNESS,--厚度
		T1.COLORS,--颜色
		T1.CUSTOMIZED_REQUIREMENTS,--其他定制要求
		T13.UPLOAD_OSS_DATE,  -- 上传OSS交期
		T11.CATEGORY, --品类
		T11.PACKAGE_MSG, --包装信息
		T.RELEASE_START_TIME, --下单时间
		T1.ESTIMATED_DELIVERY_TIME, -- 预计交货时间
		T.COMPLETE_QTY, -- 已完成数量
		(T.COMPLETE_QTY * T11.AREA) AS COMPLETE_QTY_AREA,  -- 已完成数量面积
		T.NO_COMPLETE_QTY,--未完成数量
		(T.NO_COMPLETE_QTY * T11.AREA) AS NO_COMPLETE_QTY_AREA,--未完成数量面积
		T.PLAN_AMOUNT as PLAN_QTY, -- 计划数量
		(T.PLAN_AMOUNT * T11.AREA) AS PLAN_QTY_AREA, -- 计划数量面积
		CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
		T.RE_STATE, --复版状态
		T.RE_DATE, --复版日期
		TT.PLAN_ITEM_NUM --计划单行号
from ORDER_DETAIL_DATA t  
 left join SDI.SDI_sale_order_dtl_1007 t1 -- 销售计划单明细
 on t.SALE_ORDER_HEAD_NO = t1.sale_no 
 and t.SALE_ORDER_LINE_NO = t1.line_no
 AND T1.del_flag = 'N'
 LEFT JOIN TT
 ON T1.wbs_line_no = TT.PLAN_ITEM_NUM
 LEFT JOIN SDI.sdi_prps_1001 SP1 -- WBS
 ON TT.WBS_NUM = SP1.posid 
 AND SP1.del_flag = 'N'
 left join sdi.sdi_PLAN_REVIEW_1007 t13  --计划评审表
 on t13.sale_order_detail_id = t1.id 
 and t13.is_phantom = 0
 AND T13.del_flag = 'N'
 left join SDI.SDI_ITEM_1007 t11   --物料
 on t.item_id = t11.id 
 and t11.is_phantom = 0 
 AND T11.del_flag = 'N'
 left join sdi.sdi_ORDER_TYPE_SETTING_1007 t14  --订单类型
 on t.order_type_set_id = t14.id 
 and t14.is_phantom = 0 
 AND T14.del_flag = 'N'
 left join sdi.sdi_CSM_CUSTOMER_1007 t12  --客户
 on t.customer_id = t12.id 
 and t12.is_phantom = 0
 AND T12.del_flag = 'N'
 LEFT JOIN PRODUCT_LEVEL2_DATA  T17 -- 产品层级2
 ON T11.CODE = T17.MATERIAL_NUM
 WHERE t1.Detail_State = 0 
 and t1.is_phantom = 0;
