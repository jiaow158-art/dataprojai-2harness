DELETE FROM DM.DM_PUR_ORDER_ZJ_PRE_T
WHERE STAT_MONTH = SUBSTR(CAST(CURRENT_DATE() + INTERVAL '-1 DAY' AS VARCHAR),1,7)
;

WITH T AS(
SELECT 
 PURCHASE_VOUCHER_NUM
,PURCHASE_VOUCHER_ITEM_NUM
,CONCAT(PURCHASE_VOUCHER_NUM,PURCHASE_VOUCHER_ITEM_NUM) AS PURCHASE_NAME
,REPLY_DELIVERY
,PURCHASE_ORDER_QUANTITY --采购量
,MAX(TOTAL_INTOWAREHOUSE_QUANTITY) AS TOTAL_INTOWAREHOUSE_QUANTITY --累计入库量
,A.PRODUCT_AREA_ID --产区
,A.PRODUCT_AREA_CODE --产区编码
,A.PRODUCT_AREA_NAME --产区名称
,A.SUPPLIER_OR_CREDITOR_ID --供应商ID
,A.SUPPLIER_OR_CREDITOR_ACCT --供应商编码
,A.SUPPLIER_OR_CREDITOR_NAME --供应商名称
,A.PRODUCT_CHANNEL_CODE --产品渠道编码
,A.PRODUCT_CHANNEL_NAME --产品渠道名称
,MIN(CASE WHEN A.PRODUCT_PROPERTY IN('A','B') THEN VOUCHER_POST_DATE
          WHEN A.PRODUCT_PROPERTY NOT IN('A','B') AND TOTAL_INTOWAREHOUSE_QUANTITY >= PURCHASE_ORDER_QUANTITY*0.95 THEN VOUCHER_POST_DATE
		   END
    ) VOUCHER_POST_DATE --最小足量交付日期
FROM(
SELECT A.PURCHASE_VOUCHER_NUM
,A.PURCHASE_VOUCHER_ITEM_NUM
,A.REPLY_DELIVERY
,A.VOUCHER_POST_DATE
,A.PURCHASE_ORDER_QUANTITY
,A.INTOWAREHOUSE_QUANTITY
,A.PRODUCT_AREA_ID --产区
,A.PRODUCT_AREA_CODE --产区编码
,A.PRODUCT_AREA_NAME --产区名称
,A.SUPPLIER_OR_CREDITOR_ID --供应商ID
,A.SUPPLIER_OR_CREDITOR_ACCT --供应商编码
,A.SUPPLIER_OR_CREDITOR_NAME --供应商名称
,A.PRODUCT_CHANNEL_CODE --产品渠道编码
,A.PRODUCT_CHANNEL_NAME --产品渠道名称
,CASE WHEN A.PRODUCT_PROPERTY NOT IN('A','B') OR A.PRODUCT_PROPERTY IS NULL THEN 'D'
      ELSE A.PRODUCT_PROPERTY
 	   END AS PRODUCT_PROPERTY
,SUM(INTOWAREHOUSE_QUANTITY) OVER(PARTITION BY A.PURCHASE_VOUCHER_NUM
,A.PURCHASE_VOUCHER_ITEM_NUM ORDER BY A.VOUCHER_POST_DATE ASC) TOTAL_INTOWAREHOUSE_QUANTITY  --累计入库量
FROM DM.DM_WM_PRODSTOCK_IN_T A
WHERE A.REPLY_DELIVERY IS NOT NULL
--AND A.MATL_GROUP_CODE = '1000010'
AND COALESCE(A.SPECIAL_STOCK_FLAG_2,'') = 'K_T148T'
--AND SUBSTR(VOUCHER_POST_DATE,1,7) <= SUBSTR((NOW() - INTERVAL '1 MONTHS'),1,7)
--AND SUBSTR(REPLY_DELIVERY,1,7) = TO_CHAR(to_date('2022-05') + INTERVAL '-1 MONTH','YYYY-MM') 
--and substr(reply_delivery,1,10) < substr(to_date('2022-05'||'-'||extract(day from now())),1,10)
--and substr(reply_delivery,1,7) <= '2022-05'
and substr(A.reply_delivery,1,7) >= '2022-01'
and substr(A.reply_delivery,1,10) < cast(current_date() as varchar)
and A.material_num NOT LIKE '%_B'
and A.batch_num not like '%7'
and A.batch_num not like '%9'
) A
WHERE TOTAL_INTOWAREHOUSE_QUANTITY >= PURCHASE_ORDER_QUANTITY * 0.9
--准交数量要求： AB类大于等于90% CD类大于等于100%
--日期：回复交期+5
GROUP BY PURCHASE_VOUCHER_NUM
,PURCHASE_VOUCHER_ITEM_NUM
,REPLY_DELIVERY
,PURCHASE_ORDER_QUANTITY --采购量
,A.PRODUCT_AREA_ID --产区
,A.PRODUCT_AREA_CODE --产区编码
,A.PRODUCT_AREA_NAME --产区名称
,A.SUPPLIER_OR_CREDITOR_ID --供应商ID
,A.SUPPLIER_OR_CREDITOR_ACCT --供应商编码
,A.SUPPLIER_OR_CREDITOR_NAME --供应商名称
,A.PRODUCT_CHANNEL_CODE --产品渠道编码
,A.PRODUCT_CHANNEL_NAME --产品渠道名称
)
,T11 AS(
SELECT DISTINCT
 A.PURCHASE_VOUCHER_NUM
,A.PURCHASE_VOUCHER_ITEM_NUM
,CONCAT(A.PURCHASE_VOUCHER_NUM,A.PURCHASE_VOUCHER_ITEM_NUM) AS PURCHASE_NAME
,SUBSTR(REPLY_DELIVERY + INTERVAL '5 DAY',1,7) AS REPLY_MONTH --回复交期月份
,A.MATERIAL_NUM
,A.MATERIAL_NAME
,A.DIMENSION
,A.PRODUCT_STAGE2
,REPLY_DELIVERY
,A.PRODUCT_AREA_ID --产区
,A.PRODUCT_AREA_CODE --产区编码
,A.PRODUCT_AREA_NAME --产区名称
,A.SUPPLIER_OR_CREDITOR_ID --供应商ID
,A.SUPPLIER_OR_CREDITOR_ACCT --供应商编码
,A.SUPPLIER_OR_CREDITOR_NAME --供应商名称
,A.PRODUCT_CHANNEL_CODE --产品渠道编码
,A.PRODUCT_CHANNEL_NAME --产品渠道名称
,A.PLAN_AMOUNT AS PURCHASE_ORDER_QUANTITY --下单量
,A.PURCHASING_AREA AS PURCHASE_ORDER_AREA --采购面积（平方米）
,CASE WHEN A.PRODUCT_PROPERTY NOT IN('A','B') OR A.PRODUCT_PROPERTY IS NULL THEN 'D'
      ELSE A.PRODUCT_PROPERTY
 	   END AS PRODUCT_PROPERTY --物料组1
FROM DM.DM_PS_PURCHASE_PLAN_DTL_T A
WHERE A.reply_delivery IS NOT NULL
AND (COALESCE(A.CLOSE_FLAG,' ') <> '是' OR COALESCE(A.PREPARE_CLOSE_FLAG,' ') <> '是'
     OR COALESCE(A.ALREADY_LEVITE_QUANTITY,0) > 0)
AND A.PURCHASE_VOUCHER_NUM IS NOT NULL
--AND SUBSTR(REPLY_DELIVERY,1,7) = TO_CHAR(to_date('2022-05') + INTERVAL '-1 MONTH','YYYY-MM') 
and substr(A.REPLY_DELIVERY + INTERVAL '5 DAY',1,10) < cast(current_date() as varchar)
--AND substr(reply_delivery,1,7) <= '2022-05'
AND substr(reply_delivery,1,7) >= '2022-01'
AND A.material_num not like '%_B'
AND A.SUPPLIER_OR_CREDITOR_ACCT <>'0001008160'
AND A.MATL_GROUP_CODE = '1000010'
)
, T2 AS(
SELECT A.REPLY_MONTH --回复交期月份 
      ,A.PURCHASE_NAME
      ,a.TOTAL_INTOWAREHOUSE_QUANTITY
      ,a.VOUCHER_POST_DATE
--准交数量要求： AB类大于等于90% CD类大于等于100% ①非工程订单：入库数量范围在下单数量+/-10%，视为足量。②工程订单：入库数量不超过下单数量的+10%，视为足量
--日期：回复交期+5
	  ,CASE WHEN A.TOTAL_INTOWAREHOUSE_QUANTITY <= B.PURCHASE_ORDER_QUANTITY * 1.1
	                        AND B.PRODUCT_PROPERTY IN('A','B') THEN '准交'
						   WHEN A.TOTAL_INTOWAREHOUSE_QUANTITY <= B.PURCHASE_ORDER_QUANTITY * 1.1
						    AND A.TOTAL_INTOWAREHOUSE_QUANTITY >= B.PURCHASE_ORDER_QUANTITY*0.95
	                        AND B.PRODUCT_PROPERTY NOT IN('A','B') THEN '准交'
	                        else '不准交'
							END is_zj --准交订单数
FROM(SELECT DISTINCT SUBSTR(REPLY_DELIVERY + INTERVAL '5 DAY',1,7) AS REPLY_MONTH
                    ,PURCHASE_NAME
					,TOTAL_INTOWAREHOUSE_QUANTITY
					,VOUCHER_POST_DATE
       FROM  T A
WHERE DATE_PART('DAY',A.VOUCHER_POST_DATE - A.REPLY_DELIVERY) <= 5) A
INNER JOIN (SELECT DISTINCT
                  PURCHASE_NAME
                 ,PRODUCT_PROPERTY
				 ,PURCHASE_ORDER_QUANTITY
             FROM T11
          ) B
  ON A.PURCHASE_NAME = B.PURCHASE_NAME
)
INSERT INTO DM.DM_PUR_ORDER_ZJ_PRE_T(
       STAT_MONTH
	  ,IS_ZJ
      ,TOTAL_INTOWAREHOUSE_QUANTITY
      ,VOUCHER_POST_DATE
      ,PURCHASE_VOUCHER_NUM
      ,PURCHASE_VOUCHER_ITEM_NUM
	  ,MATERIAL_NUM
      ,MATERIAL_NAME
	  ,DIMENSION
	  ,PRODUCT_STAGE2
      ,REPLY_DELIVERY
      ,PRODUCT_AREA_CODE
      ,PRODUCT_AREA_NAME
      ,SUPPLIER_OR_CREDITOR_ACCT
      ,SUPPLIER_OR_CREDITOR_NAME
      ,PRODUCT_CHANNEL_CODE
      ,PRODUCT_CHANNEL_NAME
      ,PURCHASE_ORDER_QUANTITY
      ,PURCHASE_ORDER_AREA
      ,PRODUCT_PROPERTY
	  ,DW_LAST_UPDATE_DATE
) 
SELECT SUBSTR(A.REPLY_DELIVERY + INTERVAL '5 DAY',1,7) AS STAT_MONTH
      ,COALESCE(B.IS_ZJ,'不准交') AS IS_ZJ
      ,B.TOTAL_INTOWAREHOUSE_QUANTITY
      ,B.VOUCHER_POST_DATE
      ,A.PURCHASE_VOUCHER_NUM
      ,A.PURCHASE_VOUCHER_ITEM_NUM
	  ,A.MATERIAL_NUM
	  ,A.MATERIAL_NAME
	  ,A.DIMENSION
	  ,A.PRODUCT_STAGE2
      ,A.REPLY_DELIVERY
      ,A.PRODUCT_AREA_CODE
      ,A.PRODUCT_AREA_NAME
      ,A.SUPPLIER_OR_CREDITOR_ACCT
      ,A.SUPPLIER_OR_CREDITOR_NAME
      ,A.PRODUCT_CHANNEL_CODE
      ,A.PRODUCT_CHANNEL_NAME
      ,A.PURCHASE_ORDER_QUANTITY
      ,A.PURCHASE_ORDER_AREA
      ,A.PRODUCT_PROPERTY
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM T11 A
LEFT JOIN T2 B
ON A.PURCHASE_NAME = B.PURCHASE_NAME
WHERE SUBSTR(A.REPLY_DELIVERY + INTERVAL '5 DAY',1,7) = SUBSTR(CAST(CURRENT_DATE() + INTERVAL '-1 DAY' AS VARCHAR),1,7)
;
