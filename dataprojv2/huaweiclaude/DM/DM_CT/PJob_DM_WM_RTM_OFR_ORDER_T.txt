DROP TABLE IF EXISTS DM.DM_WM_RTM_OFR_ORDER_T_TEMP01;
CREATE TABLE DM.DM_WM_RTM_OFR_ORDER_T_TEMP01
AS
SELECT ZUONR --报价单号
      --,SGTXT 
      ,SUM(TSL) AS TSL
  FROM SDI.SDI_ACDOCA_1001
 WHERE PERIOD_ID_M >= '202303'
   AND DRCRK = 'S'
   AND (SGTXT = 'OMS订货会提货定金回款' --退定金
    OR SGTXT LIKE '%释放OMS订货会报价单定金%')
   AND ZUONR IN(SELECT QUOTATION_CODE
                  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F
			   )
 GROUP BY ZUONR
;

DROP TABLE IF EXISTS DM.DM_WM_RTM_OFR_ORDER_T_TEMP02;
CREATE TABLE DM.DM_WM_RTM_OFR_ORDER_T_TEMP02
AS
SELECT ZUONR --报价单号
      --,SGTXT 
      ,SUM(TSL) AS TSL
  FROM SDI.SDI_ACDOCA_1001
 WHERE PERIOD_ID_M >= '202303'
   AND DRCRK = 'S'
   AND (SUBSTR(SGTXT,1,13) IN('OMS订货会报价单定金扣款')
    OR SGTXT = 'OMS订货会提货定金回款')
   AND ZUONR IN(SELECT QUOTATION_CODE
                  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F
			   )
 GROUP BY ZUONR
;


INSERT OVERWRITE TABLE DM.DM_WM_RTM_OFR_ORDER_T
SELECT  C.SALES_AREA AS SALE_DEPT_NAME --运营中心
       ,A.SALE_CHANNEL_CODE --渠道编码
	   ,CASE WHEN A.SALE_CHANNEL_CODE = '01' THEN '经销'
	         WHEN A.SALE_CHANNEL_CODE = '02' THEN '零售'
			 WHEN A.SALE_CHANNEL_CODE = '03' THEN '内部'
			 WHEN A.SALE_CHANNEL_CODE = '05' THEN 'OEM'
			 WHEN A.SALE_CHANNEL_CODE = '06' THEN '工程'
			 WHEN A.SALE_CHANNEL_CODE = '07' THEN '电商'
			 WHEN A.SALE_CHANNEL_CODE = '08' THEN '家装'
			 WHEN A.SALE_CHANNEL_CODE = '09' THEN '设计师'
			 WHEN A.SALE_CHANNEL_CODE = '10' THEN '流通'
			  END AS SALE_CHANNEL_NAME --渠道名称
       ,C.SALES_GROUP AS SALE_SUB_DEPT_NAME --线组			  
	   ,A.ACCOUNT_NUMBER --客户名称
       ,A.ACCOUNT_NAME --客户编码
	   ,A.CREATION_DATE AS OFR_DATE --报价单时间	   
	   ,A.QUOTATION_CODE AS OFR_ORDER_NUM --报价单号
       ,B.PRODUCT_CODE AS MATERIAL_CODE --物料编码
	   ,B.PRODUCT_NAME AS MATERIAL_NAME --物料名称
	   ,B.UNIT_PRICE --单价
	   ,SUM(B.QUANTITY) AS SALE_QTY --单据数量
       ,SUM(B.SALES_AREA) AS SALE_AREA --单据面积
       ,SUM(B.QUANTITY * B.UNIT_PRICE) AS TOTAL_AMT --单据金额
	   ,D.CREATION_DATE AS ORDER_DATE --转订单时间	   
       ,F.ORDER_NUMBER AS ORDER_NUM --订单编码
       ,E.QUANTITY AS ORDER_QTY --转订单数量
	   ,E.ITEM_AMOUNT AS ORDER_AMT --转订单金额
	   ,COALESCE(J.TSL,0) AS DEPOSIT_AMT --定金金额（初始定金）
	   ,COALESCE(I.TSL,0) AS RETURN_AMT --退回定金
	   ,A.SALESMAN_NUMBER AS SALE_EMP_NUM
	   ,G.EMP_NAME AS SALE_EMP_NAME
	   ,A.QUOTATION_TYPE AS OFR_ORDER_TYPE_CODE
	   ,CASE WHEN A.QUOTATION_TYPE = 'SPECIAL_QUOTATION' THEN '特惠品价审报价单'
	         WHEN A.QUOTATION_TYPE = 'ORDER_PLACING_MEETING_QUOTATION' THEN '订货会报价单'
			 WHEN A.QUOTATION_TYPE = 'DIRECTIONAL_QUOTATION' THEN '定向促销报价单'
			 WHEN A.QUOTATION_TYPE = 'BIG_PLACING_MEETING_QUOTATION' THEN '订货会报价单(大包渠道)'
              END AS OFR_ORDER_TYPE_NAME
	   ,H.MENGE	AS PICK_QTY --提货数量
       ,C.SALES_DEPARTMENT AS SALE_AREA_NAME --销区
       ,A.SALE_ORG_CODE --销售组织编码
       ,C.SALES_ORGANIZATION AS SALE_ORG_NAME --销售组织名称
       ,E.SAP_ORDER_NUMBER AS SAP_ORDER_NUM --SAP订单编码
       ,CASE WHEN COALESCE(H.ERDAT,'00000000') NOT IN('','00000000')
	         THEN from_unixtime(unix_timestamp(H.ERDAT,'yyyyMMdd'),'yyyy-MM-dd')
			  END AS SAP_ORDER_DATE --SAP订单日期
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
       ,B.POLICY_NAME AS POLICY_INFO --政策信息   
       ,F.CREATION_DATE AS TURN_OFR_DATE --转订单日期
  FROM DWRCS.DWR_CS_QUOTATION_HEADER_F A
  LEFT JOIN DWRCS.DWR_CS_QUOTATION_ITEM_F B
    ON A.ID = B.QUOTATION_ID
  LEFT JOIN SDI.SDI_ZMASTER_CUS_1001 C
    ON A.ACCOUNT_NUMBER = C.ACCOUNT_NUMBER
   AND A.SALE_ORG_CODE = C.SALES_ORGANIZATION_CODE
   AND A.SALE_CHANNEL_CODE = C.SALES_CHANNEL_CODE
   AND C.PARVW = 'AG'
  LEFT JOIN (SELECT QUOTATION_ITEM_ID
                   ,MATERIAL_NUMBER
				   ,HEAD_ID
				   ,SAP_ORDER_NUMBER
				   ,SUM(CASE WHEN COALESCE(STATUS,'') <> 'CANCEL' THEN QUANTITY
				             ELSE QUANTITY * -1
							  END
					   ) AS QUANTITY
				   ,SUM(CASE WHEN COALESCE(STATUS,'') <> 'CANCEL' THEN QUANTITY * UNIT_PRICE
				             ELSE QUANTITY * UNIT_PRICE * -1
							  END
				       ) AS ITEM_AMOUNT
               FROM DWRCS.DWR_CS_ENGIN_P_ORDER_LINE_F E
			  --WHERE COALESCE(STATUS,'') <> 'CANCEL'
			  GROUP BY
			        QUOTATION_ITEM_ID
				   ,MATERIAL_NUMBER
				   ,HEAD_ID
				   ,SAP_ORDER_NUMBER
			) E
    ON B.ID = E.QUOTATION_ITEM_ID
  LEFT JOIN DWRCS.DWR_CS_ENGIN_P_ORDER_HEADER_F D
    ON E.HEAD_ID = D.ID		
  LEFT JOIN DWRCS.DWR_CS_ORDER_ALL_BUSINESS_F F
    ON D.ID = F.ID
  LEFT JOIN (SELECT DISTINCT EMP_NUM,EMP_NAME FROM DM.DM_WF_EMP_INFO_T WHERE STATUS IN(0,1,2,3)) G
    ON A.SALESMAN_NUMBER = G.EMP_NUM
  LEFT JOIN (SELECT A.QUOTATION_ITEM_ID
                   ,SUM(B.MENGE) AS MENGE
				   ,MAX(ERDAT) AS ERDAT
               FROM DWRCS.DWR_CS_ENGIN_P_ORDER_LINE_F A
			   LEFT JOIN DM.DM_OTD_SALES_ORDER_DET_T B
			     ON A.SAP_ORDER_NUMBER = B.VBELN
				AND REPLACE(A.SAP_ORDER_ITEM_NUMBER,A.SAP_ORDER_NUMBER,'') = B.POSNR
			 GROUP BY
			        A.QUOTATION_ITEM_ID
			) H
    ON B.ID = H.QUOTATION_ITEM_ID
  LEFT JOIN DM.DM_WM_RTM_OFR_ORDER_T_TEMP01 I
    ON I.ZUONR = A.QUOTATION_CODE
  LEFT JOIN DM.DM_WM_RTM_OFR_ORDER_T_TEMP02 J
    ON J.ZUONR = A.QUOTATION_CODE  
 WHERE A.STATUS NOT IN('DRAFT','CANCEL')
 --增加过滤条件：报价单行状态为草稿、无效、取消的进行过滤
   AND B.STATUS NOT IN('DRAFT','CANCEL','INVALID')
 GROUP BY
        C.SALES_AREA --运营中心
       ,A.SALE_CHANNEL_CODE --渠道编码
	   ,CASE WHEN A.SALE_CHANNEL_CODE = '01' THEN '经销'
	         WHEN A.SALE_CHANNEL_CODE = '02' THEN '零售'
			 WHEN A.SALE_CHANNEL_CODE = '03' THEN '内部'
			 WHEN A.SALE_CHANNEL_CODE = '05' THEN 'OEM'
			 WHEN A.SALE_CHANNEL_CODE = '06' THEN '工程'
			 WHEN A.SALE_CHANNEL_CODE = '07' THEN '电商'
			 WHEN A.SALE_CHANNEL_CODE = '08' THEN '家装'
			 WHEN A.SALE_CHANNEL_CODE = '09' THEN '设计师'
			 WHEN A.SALE_CHANNEL_CODE = '10' THEN '流通'
			  END --渠道名称
       ,C.SALES_GROUP --线组			  
	   ,A.ACCOUNT_NUMBER --客户名称
       ,A.ACCOUNT_NAME --客户编码
	   ,A.CREATION_DATE --报价单时间	   
	   ,A.QUOTATION_CODE --报价单号
       ,B.PRODUCT_CODE --物料编码
	   ,B.PRODUCT_NAME --物料名称
	   ,B.UNIT_PRICE --单价
	   ,D.CREATION_DATE --转订单时间	   
       ,F.ORDER_NUMBER --订单编码
       ,E.QUANTITY --转订单数量
	   ,E.ITEM_AMOUNT --转订单金额
	   ,COALESCE(J.TSL,0) --定金金额（初始定金）
	   ,COALESCE(I.TSL,0) --退回定金
	   ,A.SALESMAN_NUMBER
	   ,G.EMP_NAME
	   ,A.QUOTATION_TYPE
	   ,CASE WHEN A.QUOTATION_TYPE = 'SPECIAL_QUOTATION' THEN '特惠品价审报价单'
	         WHEN A.QUOTATION_TYPE = 'ORDER_PLACING_MEETING_QUOTATION' THEN '订货会报价单'
			 WHEN A.QUOTATION_TYPE = 'DIRECTIONAL_QUOTATION' THEN '定向促销报价单'
			 WHEN A.QUOTATION_TYPE = 'BIG_PLACING_MEETING_QUOTATION' THEN '订货会报价单(大包渠道)'
              END
	   ,H.MENGE	--提货数量
       ,C.SALES_DEPARTMENT --销区
       ,A.SALE_ORG_CODE --销售组织编码
       ,C.SALES_ORGANIZATION --销售组织名称
       ,E.SAP_ORDER_NUMBER --SAP订单编码
       ,CASE WHEN COALESCE(H.ERDAT,'00000000') NOT IN('','00000000')
	         THEN from_unixtime(unix_timestamp(H.ERDAT,'yyyyMMdd'),'yyyy-MM-dd')
			  END --SAP订单日期
       ,B.POLICY_NAME --政策信息
       ,F.CREATION_DATE --转订单日期
;
