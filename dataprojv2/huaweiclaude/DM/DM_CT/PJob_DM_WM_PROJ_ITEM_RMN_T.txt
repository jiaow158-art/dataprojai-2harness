WITH T0 AS( --有出过库的留货
SELECT WBS_LEVEL3
      ,BATCH_NUM
	  ,PLANT_CODE
	  ,LOC_CODE
	  ,SUM(RMN_STOCK_QTY) AS RMN_STOCK_QTY
  FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T
 WHERE RMN_STOCK_QTY < 0
 GROUP BY 
       WBS_LEVEL3
      ,BATCH_NUM
	  ,PLANT_CODE
	  ,LOC_CODE
),T1 AS( --剔除总留货数为0的
SELECT WBS_LEVEL3
      ,BATCH_NUM
	  ,PLANT_CODE
	  ,LOC_CODE
	  ,SUM(RMN_STOCK_QTY) AS RMN_STOCK_QTY
  FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T
 GROUP BY 
       WBS_LEVEL3
      ,BATCH_NUM
	  ,PLANT_CODE
	  ,LOC_CODE
 HAVING SUM(RMN_STOCK_QTY) > 0
),T2 AS( 
--取没有出过库的留货
SELECT A.WBS_LEVEL3
      ,A.BATCH_NUM
	  ,A.PLANT_CODE
	  ,A.LOC_CODE
	  ,A.RMN_DATE
	  ,SUM(A.RMN_STOCK_QTY) AS RMN_STOCK_QTY
	  ,SUM(COALESCE(OUT_STOCK_ZT_QTY,0)) AS OUT_STOCK_ZT_QTY --累计出库数量
  FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T A
 INNER JOIN T1
    ON A.WBS_LEVEL3 = T1.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(T1.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(T1.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(T1.LOC_CODE,'SNULL')
  LEFT JOIN T0
    ON A.WBS_LEVEL3 = T0.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(T0.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(T0.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(T0.LOC_CODE,'SNULL')
 WHERE T0.WBS_LEVEL3 IS NULL
 GROUP BY
       A.WBS_LEVEL3
      ,A.BATCH_NUM
	  ,A.PLANT_CODE
	  ,A.LOC_CODE
	  ,A.RMN_DATE
 
 UNION ALL 
 
 --取有出过库的留货且没有完全出库
SELECT A.WBS_LEVEL3
      ,A.BATCH_NUM
	  ,A.PLANT_CODE
	  ,A.LOC_CODE
	  ,MAX(CASE WHEN A.RMN_STOCK_QTY > 0 THEN A.RMN_DATE END) AS RMN_DATE
	  ,SUM(A.RMN_STOCK_QTY) AS RMN_STOCK_QTY
	  ,SUM(COALESCE(OUT_STOCK_ZT_QTY,0)) AS OUT_STOCK_ZT_QTY --累计出库数量
  FROM DWROTD.DWR_WM_PROJ_ITEM_STOCK_T A
 INNER JOIN T1
    ON A.WBS_LEVEL3 = T1.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(T1.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(T1.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(T1.LOC_CODE,'SNULL')
  LEFT JOIN T0
    ON A.WBS_LEVEL3 = T0.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(T0.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(T0.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(T0.LOC_CODE,'SNULL')
 WHERE T0.WBS_LEVEL3 IS NOT NULL
 GROUP BY
       A.WBS_LEVEL3
      ,A.BATCH_NUM
	  ,A.PLANT_CODE
	  ,A.LOC_CODE
),T3 AS(
SELECT DISTINCT A.WBS_LEVEL3
      ,A.BATCH_NUM
	  ,A.PLANT_CODE
	  ,A.LOC_CODE
	  ,A.RMN_DATE
	  ,C.YQTS
	  ,C.SFYQ
  FROM T2
 INNER JOIN DWROTD.DWR_WM_PROJ_ITEM_STOCK_T A
    ON A.WBS_LEVEL3 = T2.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(T2.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(T2.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(T2.LOC_CODE,'SNULL')
 INNER JOIN SDI.SDI_ZTSD015_1001 C
    ON A.WBS_LEVEL3 = C.POSID
   AND A.MAT_VOUCHER = C.MBLNR
   AND C.MANDT = 800
)

INSERT OVERWRITE TABLE DM.DM_WM_PROJ_ITEM_RMN_T
SELECT B.PLAN_NUM --计划订单号
      ,B.PLAN_ITEM_NUM --计划单行号
      ,A.RMN_DATE --留货日期
	  ,A.RMN_STOCK_QTY AS RMN_QTY --留货数量
	  ,A.LOC_CODE AS WAREHOUSE_CODE --库存地点
	  ,A.BATCH_NUM --批次
	  ,CASE WHEN B.APPR_QTY > A.OUT_STOCK_ZT_QTY THEN DATEDIFF(CURRENT_DATE,A.RMN_DATE) - 1
             END AS RMN_DAY --留货天数
	  ,CASE WHEN B.APPR_QTY > COALESCE(A.RMN_STOCK_QTY,0) + COALESCE(A.OUT_STOCK_ZT_QTY,0) THEN '待留货'
	        WHEN COALESCE(B.APPR_QTY,0) - COALESCE(A.RMN_STOCK_QTY,0) - COALESCE(A.OUT_STOCK_ZT_QTY,0) <= 0
			THEN '已留货'
	         END AS RMN_ST_NAME  --留货状态 当 评审后需求数量-留货数量汇总-出库数<=0  已留货
                                 --评审后需求数量-留货数量汇总-出库数>0  待留货
	  ,CASE WHEN C.YQTS LIKE '%-' THEN REPLACE(C.YQTS,'-','') * -1 ELSE C.YQTS END AS OVER_DAY --逾期天数
      ,CASE WHEN C.SFYQ = 'Y' THEN '已逾期' ELSE '未逾期' END AS RMN_WARN --留货预警
	  ,NULL AS SEAL_QTY --封仓数量
	  ,NULL AS SEAL_TIME --冻结时间
	  ,NULL AS CHK_TIME --质量检查库存时间
	  ,NULL AS SEAL_ST_NAME --封仓状态
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(A.RMN_DATE) - UNIX_TIMESTAMP(B.CREATE_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(A.RMN_DATE) - UNIX_TIMESTAMP(B.CREATE_DATE)),86400)) / 3600,0) || '小时','.0','') AS RMN_CNSUM_TIME --留货耗时
	  ,NULL AS SEAL_RSN --封仓原因
	  ,NULL AS UN_SEAL_TIME --预计解封时间
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T2 AS A
 INNER JOIN DWRLTC.DWR_BO_PLAN_ORDER_F B
    ON A.WBS_LEVEL3 = B.PLAN_ITEM_NUM
  LEFT JOIN T3 C
    ON A.WBS_LEVEL3 = C.WBS_LEVEL3
   AND COALESCE(A.BATCH_NUM,'SNULL') = COALESCE(C.BATCH_NUM,'SNULL')
   AND COALESCE(A.PLANT_CODE,'SNULL') = COALESCE(C.PLANT_CODE,'SNULL')
   AND COALESCE(A.LOC_CODE,'SNULL') = COALESCE(C.LOC_CODE,'SNULL')
   AND A.RMN_DATE = C.RMN_DATE
;
