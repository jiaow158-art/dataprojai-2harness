INSERT OVERWRITE TABLE DM.DM_LTC_CUSTOMER_LIST_T
SELECT T1.kunnr as customer_code,  -- 客户编码
	   T2.name1 as customer_name, -- 客户名称
	   T1.vkbur as sales_office_code, -- 销售办事处
	   T3.bezei AS sales_office_name,  -- 销售办事处名称
	   T1.vkgrp as sales_group_code, --  销售组
	   T4.bezei AS sales_group_name, -- 销售组名称
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM SDI.SDI_knvv_1001 T1
LEFT JOIN SDI.SDI_kna1_1001 T2
ON T1.kunnr = T2.kunnr
AND T2.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_tvkbt_1001 T3
ON T3.vkbur = T1.vkbur 
AND T3.spras = '1'
AND T3.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_tvgrt_1001 T4
ON T4.vkgrp = T1.vkgrp 
AND T4.spras = '1'
AND T4.DEL_FLAG = 'N'
WHERE T1.vkbur IN ('Z075','Z076')
AND T1.DEL_FLAG = 'N';
