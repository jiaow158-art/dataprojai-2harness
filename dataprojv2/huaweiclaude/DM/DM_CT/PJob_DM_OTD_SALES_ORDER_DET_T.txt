DROP TABLE IF EXISTS  default.nopro_sales_order;
create table default.nopro_sales_order as 
WITH T0 AS(
SELECT
      a.VBELN
	 ,a.POSNR
	 ,max(e.VMENG) as VMENG
	 ,min(case when a.KWMENG <= e.VMENG then e.edatu end) as edatu
	 ,max(case when bmeng>0  then e.edatu end) as max_edatu
FROM SDI.SDI_VBAP_1001 a
inner join(
          SELECT VBELN
		        ,POSNR
				,edatu
				,BMENG
				,plart
		        ,SUM(BMENG) OVER(PARTITION BY VBELN,POSNR ORDER BY edatu rows between unbounded preceding and current row) as VMENG
            FROM SDI.SDI_VBEP_1001
		   WHERE DEL_FLAG = 'N'
         ) e
on a.VBELN = e.VBELN
and a.POSNR = e.POSNR
where a.DEL_FLAG = 'N'
group by
      a.VBELN
	 ,a.POSNR
)

,c.BZIRK --营销大区
,b.AUART --订单类型

,a.MVGR2
