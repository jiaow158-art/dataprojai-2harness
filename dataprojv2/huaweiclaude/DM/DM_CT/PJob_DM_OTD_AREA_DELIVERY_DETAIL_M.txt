set hive.cbo.enable=true;				--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

with batch_area  as(
	select batch_start,
		   max(belong_area_code) as belong_area_code ,
		   max(belong_area_name) as belong_area_name
	from DWRDIM.DWR_DIM_PRODUCT_AREA_D
	group by batch_start 
),
tms as(
	select externno ,
		   externlineno ,
		   t.receiving_province ,
		   row_number() over(partition by externno ,externlineno order by t.actual_ship_date desc ) as rn
   from DM.DM_LTC_ORDER_PLAN_SIGN_T T
),
detail as
(
  select t1.VOUCHER_POST_DATE ,
		 t1.material_num,
		 t1.batch_num , 
		 t1.material_vch_no_mblnr ,
		 t1.delivery_project ,
		 t1.delivery_im,
		 t8.sale_order_num ,
		 t8.sale_order_item_num ,
		 coalesce(t3.belong_area_code,t4.belong_area_code, 'GD') as belong_area_code,  
		 coalesce(t3.belong_area_name,t4.belong_area_name, '广东基地') as belong_area_name,
		 coalesce(t7.province,t6.REGION_PROVINCE_NAME) as receiving_province,
		 DR_CR_ID,
		 t2.material_name,
		 t2.dimension,
		 t2.product_level3_name ,
		 case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end as sales_stock_out_qty,
		 case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end*t2.length_1*t2.width/1000/1000 as sales_stock_out_area,
		 t8.dis_channel_name,
		 coalesce(t10.partment,t11.lev4_name,t11.lev3_name) as op_partment,
		 substr(t1.VOUCHER_POST_DATE,1,7) as stat_month,
		 t2.product_level2_name,
		 current_timestamp() as dw_last_update_date
	from dwiotd.DWI_WM_PRODSTOCK_OUT_IN_T t1
	join dwimd.dwi_md_data_material_general_t t2 
	     on t1.material_num  = t2.material_num 
	left join batch_area t3 
	     on t3.batch_start = substr(t1.batch_num,1,3) 
	left join batch_area t4 
	    on t4.batch_start = substr(t1.batch_num,1,2) 
	left join tms t5 
	    on t1.delivery_im = t5.externno 
		and t1.delivery_project  = t5.externlineno and t5.rn=1
	left join dwrdim.dwr_dim_province t7 
	    on t5.receiving_province = t7.province_all_name 
	left join DWRDIM.DWR_DIM_CUST_GENERAL_D t6 
	    on t1.cust_num  = t6.cust_num  
		and t6.group_num ='800' 
		and t6.end_date ='4712-12-31'
	left join DM.dm_ltc_order_plan_stockup_t t8 
	   on t8.delivery_no = t1.delivery_im 
	   and t8.delivery_project = t1.delivery_project
	left join dwrdim.dwr_dim_cust_sales_area_d t9 
	    on substr(t8.sales_org_code,1,4) = t9.sales_org_code 
		and t8.cust_sal_num  = t9.cust_num  
		and substr(t8.dis_channel,1,2) = t9.distribution_chnl_code 
		and t9.product_group='00' 
		and t9.end_date ='4712-12-31'
	left join default.sales_grp_partment t10 
	   ON t10.node_name = t9.sales_group_code
	left join dm.DM_RPT_SALE_GRP_T  t11 
	    On  t9.sales_group_code = t11.sale_grp
	where 1=1
		--and VOUCHER_POST_DATE>='2022-08-15' and  VOUCHER_POST_DATE<='2022-08-16'
		and t11.lev2_name in ('瓷砖事业部')
		and t2.matl_group_code ='1000010'
		and t2.external_matl_group_code in('10010','10080')
		and t1.MOVE_TYPE IN('601','602','643','644','653','654')   -- 销售出库
		and t1.record_type ='MDOC'
	)
insert OVERWRITE table dm.dm_otd_area_delivery_detail_m
select  a.*
from detail a;
