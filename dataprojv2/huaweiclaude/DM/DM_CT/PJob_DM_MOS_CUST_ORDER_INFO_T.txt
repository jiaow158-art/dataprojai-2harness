INSERT OVERWRITE TABLE DM.DM_MOS_CUST_ORDER_INFO_T
SELECT A.CUST_ID AS CUST_ID --客户ID      
      ,A.CUST_NAME --客户名称
      ,A.CUST_PHONE_NUM --手机号
      ,A.AGE --年龄层
	  ,CASE WHEN A.ORDER_ST_NAME IN('交易成功','已完成','已评价','待评价','已发货','待发货') THEN '交易成功'
	        WHEN A.ORDER_ST_NAME IN('已小定','已大定','已交定') THEN '已交定'
			WHEN COALESCE(A.CUST_PHONE_NUM,'') <> '' OR A.ORDER_ST_NAME IN('已流失','已关闭') THEN '已留资'
			WHEN COALESCE(A.WECHAT_NUM,'') <> '' OR COALESCE(A.CUST_PHONE_NUM,'') <> '' THEN '已加微'
	         END AS ORDER_ST_NAME--订单状态
	  ,CASE WHEN A.DMT_SRC_NAME = '万家灯火' THEN A.DMT_SRC_NAME ELSE A.PARNT_CUST_SRC_NAME END AS DMT_SRC_NAME --客户来源
	  ,A.ADDR AS CUST_ADDR --客户详细地址
	  ,A.ACC_EMP_NUM --导购编码
      ,A.ACC_EMP_NAME --导购名称
	  ,COALESCE(B.CUST_NUM,A.DEALER_NUM) AS DEALER_NUM  --经销商编码
      ,B.DEALER_NAME --经销商名称
      ,COALESCE(B.OA_STORE_CODE,B.STORE_CODE) AS STORE_CODE --门店编码
      ,B.STORE_NAME --门店名称
      ,B.MKT_NAME --运营中心
	  ,B.SALE_GRP_NAME --线组名称
	  ,E.CREATE_TIME AS FST_CREATE_TIME --客户首次创建日期
	  ,NULL AS LEAD_DATE --留资日期
	  ,A.ADD_WECHAT_TIME --加微日期
      ,A.LITL_DEP_AMT --小定金额
      ,A.LITL_DEP_TIME --小定时间
	  ,A.BIG_DEP_AMT --大定金额
	  ,A.BIG_DEP_TIME --大定时间
	  ,A.DOWNPAY_AMT --已交定金额
	  ,A.DEP_TIME --已交定日期
      ,A.FIN_TIME AS TILE_FIN_TIME --交易成功日期
      ,A.MEAS_TIME --已量尺日期
      ,A.DSGN_TIME AS EXAM_TIME --已出图日期
	  ,A.SURE_PLOT_TIME --已确图日期
	  ,A.ORDER_NUM --订单编码
	  ,CASE WHEN A.NATL_SBSDZ_NAME = '是' THEN '国补订单'
	        ELSE '非国补订单'
			 END AS ORDER_TYPE_NAME --订单类型
	  ,A.ORDER_AMT --订单金额 
	  --判断条件：客户状态=“已量尺”日期在所有“已交定/已小定/已大定/交易成功/已完成/已评价/待评价/已发货/待发货”时间之前
	  ,CASE WHEN A.MEAS_TIME <= LEAST(COALESCE(A.DEP_TIME,CURRENT_DATE),COALESCE(A.LITL_DEP_TIME,CURRENT_DATE),COALESCE(A.BIG_DEP_TIME,CURRENT_DATE),COALESCE(A.TILE_FIN_TIME,CURRENT_DATE),COALESCE(A.FIN_TIME,CURRENT_DATE),COALESCE(A.SHIP_TIME,CURRENT_DATE)) THEN '是'
	        ELSE '否'
			 END AS IS_NOT_MEAS_NAME --是否测未定
	  ,CASE WHEN A.ADD_WECHAT_TIME IS NOT NULL THEN '是'
	        ELSE '否'
			 END AS IS_ADD_WECHAT_NAME --是否已加微
	  ,CASE WHEN A.ORDER_ST_NAME IN('交易成功','已完成','已评价','待评价','已发货','待发货') AND F.ORDER_ID IS NOT NULL
	        THEN COALESCE(F.SALE_AMT,0)
	        WHEN A.ORDER_ST_NAME IN('交易成功','已完成','已评价','待评价','已发货','待发货') THEN COALESCE(A.ORDER_AMT,0)
            ELSE 0
             END AS SALE_AMT --销售总金额
      ,CASE WHEN A.ORDER_ST_NAME IN('已流失','已关闭') THEN 0
	        WHEN A.ORDER_ST_NAME IN('交易成功','已完成','已评价','待评价','已发货','待发货','已小定','已大定','已交定') AND F.ORDER_ID IS NOT NULL
			THEN COALESCE(F.RECV_AMT,0)
	        ELSE COALESCE(A.ORDER_AMT,0) + COALESCE(A.BIG_DEP_AMT,0) + COALESCE(A.LITL_DEP_AMT,0) + COALESCE(A.DOWNPAY_AMT,0)
             END AS RECV_AMT --收款总金额			 
	  ,CASE WHEN SUBSTR(E.CREATE_TIME,1,10) = SUBSTR(A.FIN_TIME,1,10) THEN '是'
	        ELSE '否'
			 END AS IS_ONE_TIME_FIN_NAME --是否为一次成交客户
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN A.SURE_PLOT_TIME IS NOT NULL THEN '已确图'
	        WHEN A.DSGN_TIME IS NOT NULL THEN '已出图'
			WHEN A.MEAS_TIME IS NOT NULL THEN '已量尺'
			 END AS DSGN_ST_NAME --客户设计状态
	  ,A.ORDER_GRP_NUM --订单组号
	  ,B.LARGE_AREA_NAME --销区名称
  FROM DWROTD.DWR_SO_MOS_ORDER_INFO_F A
  LEFT JOIN DWRDIM.DWR_DIM_MOS_STORE_INFO_T B
    ON A.STORE_ID = B.ID
  LEFT JOIN DWRDIM.DWR_DIM_MOS_CUST_INFO_D E
    ON A.CUST_ID = E.CUST_ID
/*
单店模式
收款总金额：“交易成功”+“已交定”状态客户的卖卡+小定金额+大定金额+补货金额+合同款
销售总金额：“交易成功”状态客户的卖卡+小定金额+大定金额+补货金额+合同款之和
*/
  LEFT JOIN (SELECT ORDER_ID
                   ,SUM(CASE WHEN CHRG_TYPE_NAME IN('合同款','卖卡','小定','大定','补货') THEN CHRG_AMT - RTN_AMT END) AS RECV_AMT
				   ,SUM(CASE WHEN CHRG_TYPE_NAME IN('合同款','卖卡','小定','大定','补货') THEN CHRG_AMT - RTN_AMT END) AS SALE_AMT
               FROM DWROTD.DWR_SO_MOS_CHRG_ORDER_INFO_F
			  WHERE CHRG_TYPE_NAME IN('合同款','卖卡','小定','大定','补货')
			  GROUP BY ORDER_ID
            ) F
	ON A.ORDER_ID = F.ORDER_ID
 WHERE B.MKT_NAME LIKE '%运营中心%'
   AND B.MKT_NAME NOT IN('华南运营中心','江浙运营中心')
;
