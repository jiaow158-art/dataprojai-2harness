--税后净利润（高端品牌营销中心）ZB18
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB18';
with  LRZE as (
SELECT 
stat_month,
SUM(COALESCE(extr_sales_notax_amt, 0) + 
    COALESCE(extr_sel_prty_1n_amt, 0) +  
    COALESCE(cross_div_extr_income_amt, 0) + 
    COALESCE(cross_div_splr_income_amt, 0) + 
    COALESCE(div_extr_income_amt, 0) + 
    COALESCE(div_splr_income_amt, 0) + 
    COALESCE(income_other_amt, 0) + 
    COALESCE(income_adj_amt, 0) +  
    COALESCE(incld_point_amt, 0) - 
    COALESCE(noincld_wthl_man_point_amt, 0) - 
    COALESCE(noincld_wthl_sys_point_amt, 0) )as  "不含税销售净额（管理）",
SUM(COALESCE(sale_cost_amt, 0) + 
    COALESCE(lgstcs_cost_amt, 0) + 
    COALESCE(total_1n_cost_amt, 0) + 
    COALESCE(oth_cost_amt, 0) + 
    COALESCE(cross_div_extr_cost_amt, 0) + 
    COALESCE(cross_div_splr_cost_amt, 0) + 
    COALESCE(div_extr_cost_amt, 0) + 
    COALESCE(div_splr_cost_amt, 0) + 
    COALESCE(cost_adj_amt, 0))
 AS 销售成本合计,

SUM(COALESCE(ext_profit_1n_amt, 0) +
    COALESCE(ext_cross_profit_amt, 0) +
    COALESCE(ext_dept_inner_profit_amt, 0))     AS 对外分成毛利额合计,

SUM(COALESCE(addl_tax_amt, 0) +
    COALESCE(hs_prpty_tax_amt, 0) +
    COALESCE(land_used_tax_amt, 0))  AS 税金及附加合计,

SUM(COALESCE(noincld_sale_fee_amt, 0) +
    COALESCE(lgstcs_fee_amt, 0) +
    COALESCE(mng_fee_amt, 0) +
    COALESCE(dev_fee_amt, 0) +
    COALESCE(noincld_div_shr_fee_amt, 0) +
    COALESCE(fin_fee_amt, 0))  AS 总费用合计,

SUM(COALESCE(noincld_div_shr_fee_amt, 0) ) AS "费用-分摊费用(事业部内部)",

SUM(COALESCE(asset_impar_inv_dpre_2_amt, 0) +
    COALESCE(asset_impar_inv_dpre_3_amt, 0) +
    COALESCE(asset_impar_fixed_assets_amt, 0) ) AS 资产减值损失合计,

SUM(COALESCE(cr_impar_loss_1_amt, 0) +
    COALESCE(cr_impar_loss_2_amt, 0) +
    COALESCE(cr_impar_loss_3_amt, 0) +
    COALESCE(cr_impar_loss_4_amt, 0) +
    COALESCE(cr_impar_loss_5_amt, 0))  AS 信用减值损失合计,

SUM(COALESCE(fr_val_chg_income_amt, 0) ) AS 公允价值变动收益,

SUM(COALESCE(oth_income_amt, 0))  AS 其他收益,

SUM(COALESCE(inv_income_amt, 0))  AS "投资收益(损失负)",

SUM(COALESCE(asset_dspsl_income_amt, 0))  AS 资产处置损益,

SUM(COALESCE(sale_income_amt, 0) +
    COALESCE(pj_coll_fee_amt, 0))  AS 其他业务收入合计,

SUM(COALESCE(extr_sale_cost_amt, 0) +
    COALESCE(pj_pmt_fee_amt, 0))  AS 其他业务支出合计,

SUM(COALESCE(incld_oth_income_prft_amt, 0))  AS 营业外收入,

SUM(COALESCE(incld_oth_expd_prft_amt, 0)) AS 营业外支出,
    
    
SUM(coalesce(amb_acc_rcvbl_int_amt,0) +
    COALESCE(inv_int_amt, 0) +
    COALESCE(fund_cost_1_amt, 0) +
    COALESCE(fund_cost_2_amt, 0) +
    COALESCE(fund_cost_3_amt, 0) +
    COALESCE(fund_cost_4_amt, 0) +
    COALESCE(fund_cost_5_amt, 0) +
    COALESCE(splr_pay_cash_int_amt, 0))  AS 资金成本合计
    
  FROM 
    dm.dm_fi_ambv2_profit_sum_t  
left join
dm.dm_rpt_sales_group_t  t2 
on dm_fi_ambv2_profit_sum_t.sales_grp = t2.node_name9
where  T2.NODE_DESC2 = '瓷砖事业部'
and T2.NODE_NAME4  = 'H04240301'
 group by  stat_month
)
,LRZE2 as (
select 
stat_month,
(不含税销售净额（管理）
-销售成本合计
-对外分成毛利额合计
-税金及附加合计
-总费用合计
-"费用-分摊费用(事业部内部)"
-资产减值损失合计
-信用减值损失合计
+公允价值变动收益
+其他收益
+"投资收益(损失负)"
+资产处置损益
+其他业务收入合计
-其他业务支出合计
+营业外收入
-营业外支出
-资金成本合计) as 高端品牌营销中心利润总额
from 
LRZE
)
,LJ_LRZE as (
select 
'${PERIOD_ID_M}' AS CALMONTH,
SUM(
case when stat_month between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
then  高端品牌营销中心利润总额 else 0 end 
) as 高端品牌营销中心利润总额累计,
SUM (
case when  stat_month between substr('${PERIOD_ID_M}',1,4)||'-01' and to_char(to_date('${PERIOD_ID_M}', 'YYYY-MM') - interval '1 month', 'YYYY-MM')
then  高端品牌营销中心利润总额 else 0 end 
) as 上期高端品牌营销中心利润总额累计,
SUM(case when stat_month ='${PERIOD_ID_M}' then  高端品牌营销中心利润总额 else 0 end )as 当月高端品牌营销中心利润总额
from LRZE2
)
,SHJYL as (
select 
CALMONTH,   
case when 高端品牌营销中心利润总额累计 > 0 then 高端品牌营销中心利润总额累计 *0.25 else 0 end 
- 
case when 上期高端品牌营销中心利润总额累计 > 0 then 上期高端品牌营销中心利润总额累计 *0.25 else 0 end as 当月所得税,
当月高端品牌营销中心利润总额 as 高端品牌营销中心利润总额,
高端品牌营销中心利润总额累计 - (case when 高端品牌营销中心利润总额累计 > 0 then 高端品牌营销中心利润总额累计 *0.25 else 0 end)  as 月累计高端品牌营销中心利润总额,
高端品牌营销中心利润总额累计,
case when 高端品牌营销中心利润总额累计 > 0 then 高端品牌营销中心利润总额累计 *0.25 else 0 end  as 累计所得税
from 
LJ_LRZE
)
,SHJYL2 as (
select 
CALMONTH,
'ZB18' as zb_code,
SUM(高端品牌营销中心利润总额)- SUM(当月所得税) as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
SHJYL
where CALMONTH = '${PERIOD_ID_M}'
group by 
CALMONTH
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB18' as zb_code,
0 as 当月实绩,
sum(高端品牌营销中心利润总额累计-累计所得税) as 累计实绩,
0 as 季度累计实绩
from 
SHJYL
where  CALMONTH = '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB18' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
sum(月累计高端品牌营销中心利润总额) as 季度累计实绩
from 
SHJYL
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)/10000 as CM_STAT_AMT,
sum(累计实绩)/10000 as ACC_STAT_AMT,
sum(季度累计实绩)/10000 as QTR_TGT_AMT
from 
SHJYL2
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
t2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
