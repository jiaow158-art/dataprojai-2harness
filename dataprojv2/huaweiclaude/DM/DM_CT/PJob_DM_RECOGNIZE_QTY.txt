with TMP AS (
SELECT 
T1.PACK_date AS RECOGNIZE_DATE,
SUBSTRING(T1.type,1,2) as base,
T1.type AS PROCESS,
COUNT(T1.PACK_code) - COUNT(CASE WHEN T1.PACK_code = 'NULL' THEN 1 ELSE NULL END) AS ALL_QTY, --打包总数
COUNT(T1.PACK_code) - COUNT(CASE WHEN T1.PACK_code = 'NULL' OR T1.PACK_code = 'NOREAD' THEN 1 ELSE NULL END) AS RECOGNIZE_QTY,--打包识别数
COUNT(CASE WHEN T1.PACK_code = 'NOREAD' THEN 1 ELSE NULL END) AS UNRECOGNIZE_QTY,--打包未识别数
COUNT(CASE WHEN T1.PACK_code = 'NULL' THEN 1 ELSE NULL END) AS EMPTY_QTY --打包空砖数
FROM SDI.SDI_PACK_RELATION3_1043 T1 
WHERE T1.PACK_date = '${PERIOD_ID_D}'
GROUP BY PACK_date, T1.type ,SUBSTRING(T1.type,1,2)
UNION ALL 
SELECT 
T1.FRACTIONATION_date AS RECOGNIZE_DATE,
SUBSTRING(T1.type,1,2) as base,
T1.type AS PROCESS,
COUNT(T1.FRACTIONATION_code) - COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' THEN 1 ELSE NULL END) AS ALL_QTY, --分级总数
COUNT(T1.FRACTIONATION_code) - COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' OR T1.FRACTIONATION_code = 'NOREAD' THEN 1 ELSE NULL END) AS RECOGNIZE_QTY, --分级识别数
COUNT(CASE WHEN T1.FRACTIONATION_code = 'NOREAD' THEN 1 ELSE NULL END) AS UNRECOGNIZE_QTY, --分级未识别数
COUNT(CASE WHEN T1.FRACTIONATION_code = 'NULL' THEN 1 ELSE NULL END) AS EMPTY_QTY --分级空砖数      
FROM SDI.SDI_FRACTIONATION_RELATION_1043 T1
LEFT JOIN dm.dm_stock_fill T2
ON substr(t1.FRACTIONATION_code, 1, 2) = t2.invisible_code
WHERE  T1.FRACTIONATION_date = '${PERIOD_ID_D}'
GROUP BY 
T1.fractionation_date,T1.type ,SUBSTRING(T1.type,1,2)
UNION ALL 
SELECT 
T1.INKJET_date AS RECOGNIZE_DATE, --识别日期
SUBSTRING(T1.type,1,2) as base,
T1.type AS PROCESS,
COUNT(T1.INKJET_code) - COUNT(CASE WHEN T1.INKJET_code = 'NULL' THEN 1 ELSE NULL END) AS ALL_QTY, --喷墨总数
COUNT(T1.INKJET_code) - COUNT(CASE WHEN T1.INKJET_code = 'NULL' OR T1.INKJET_code = 'NOREAD' THEN 1 ELSE NULL END) AS RECOGNIZE_QTY, --喷墨识别数
COUNT(CASE WHEN T1.INKJET_code = 'NOREAD' THEN 1 ELSE NULL END) AS UNRECOGNIZE_QTY, --喷墨未识别数
COUNT(CASE WHEN T1.INKJET_code = 'NULL' THEN 1 ELSE NULL END) AS EMPTY_QTY --喷墨空砖数
FROM SDI.SDI_INKJET_RELATION_1043 T1
LEFT JOIN dm.dm_stock_fill T2
ON substr(t1.INKJET_code, 1, 2) = t2.invisible_code
WHERE T1.INKJET_date = '${PERIOD_ID_D}'
GROUP BY 
T1.INKJET_date,T1.type,SUBSTRING(T1.type,1,2)
)
INSERT OVERWRITE TABLE  DM.DM_RECOGNIZE_QTY
SELECT 
RECOGNIZE_DATE,
BASE,
PROCESS,
SUM(ALL_QTY) AS ALL_QTY,--总数
SUM(RECOGNIZE_QTY) AS RECOGNIZE_QTY,--识别数
SUM(UNRECOGNIZE_QTY) AS UNRECOGNIZE_QTY ,--未识别数
SUM(EMPTY_QTY) AS EMPTY_QTY, --空砖数
SUM(RECOGNIZE_QTY)/SUM(ALL_QTY) AS RATE, --打包识别率
'N' AS del_flag,
CURRENT_TIMESTAMP AS dw_last_update_date
FROM TMP 
GROUP BY RECOGNIZE_DATE ,BASE,
PROCESS
UNION
ALL 
SELECT 
recognize_date,
base,
process,
all_qty,
recognize_qty,
unrecognize_qty,
empty_qty,
rate,
del_flag,
dw_last_update_date
FROM 
DM.DM_RECOGNIZE_QTY
WHERE 
RECOGNIZE_DATE <> '${PERIOD_ID_D}'
