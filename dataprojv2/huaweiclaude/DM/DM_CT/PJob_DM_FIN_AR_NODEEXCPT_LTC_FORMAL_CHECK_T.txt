INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_FORMAL_CHECK_T
SELECT
           B.WBS
	      ,A.FORMAL_CHECK_ID --正式验收单ID
		  ,B.FORMAL_CHECK_CODE --正式验收单号
		  ,B.AFFIRM_DATE --验收日期
		  ,E.ACCEPTANCE_REQUEST_TIME2 --验收到请款时间（天）
		  ,SUM(COALESCE(A.CURRENT_FORMAL_AMOUNT,0)) AS CURRENT_FORMAL_AMOUNT--已办验收金额
		  ,SUM(COALESCE(A.CURRENT_REQUEST_AMOUNT,0)) AS TOTAL_REQUEST_AMOUNT --请款金额
		  ,B.AFFIRM_RESULT --确认结果编码
          ,B1.MEANING AS AFFIRM_RESULT_NAME --确认结果描述
		  ,B.REQUEST_STATUS --请款状态
		  ,B2.MEANING AS REQUEST_STATUS_NAME --请款状态描述		  
          ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM SDI.SDI_SETTLEMENT_FORMAL_CHECK_DETAIL_1033 A
 INNER JOIN SDI.SDI_SETTLEMENT_FORMAL_CHECK_1033 B
    ON A.FORMAL_CHECK_ID = B.FORMAL_CHECK_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B1
    ON B.AFFIRM_RESULT = B1.LOOKUP_CODE
   AND B1.LOOKUP_TYPE = 'SETTLE_FORMAL_CHECK_AFFIRM'
   AND B1.ENABLED_FLAG = 'Y'
  LEFT JOIN SDI.SDI_CONTRACT_1033 D
    ON B.CONTRACT_CODE = D.CONTRACT_CODE
   AND D.DELETE_FLAG = 0    
  LEFT JOIN(SELECT DISTINCT
                           CONTRACT_ID
						  ,ACCEPTANCE_REQUEST_TIME2 
              FROM SDI.SDI_CONTRACT_SETTLEMENT_HEAD_1033 
             WHERE DELETE_FLAG = 0 
           ) E
	ON D.CONTRACT_ID = E.CONTRACT_ID
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 B2
    ON B.REQUEST_STATUS = B2.LOOKUP_CODE
   AND B2.LOOKUP_TYPE = 'SETTLE_REQUEST_PAYOUT_STATUS'
   AND B2.ENABLED_FLAG = 'Y'
 WHERE A.DELETE_FLAG = 0  
   AND B.DELETE_FLAG = 0 
 GROUP BY B.WBS
	      ,A.FORMAL_CHECK_ID --正式验收单ID
		  ,B.FORMAL_CHECK_CODE --正式验收单号
		  ,B.AFFIRM_DATE --验收日期
		  ,E.ACCEPTANCE_REQUEST_TIME2 --验收到请款时间（天）
		  ,B.AFFIRM_RESULT --确认结果编码
          ,B1.MEANING --确认结果描述
		  ,B.REQUEST_STATUS --请款状态
		  ,B2.MEANING --请款状态描述
