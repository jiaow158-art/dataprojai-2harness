--一年以上库存面积占比
--一年以上库存面积占比ZB21
delete from 
DM.DM_NORTH_SM_T 
where  calmonth = '${PERIOD_ID_M}' 
and zb_code = 'ZB21';
with  detailaccage as (
select 
CONCAT (SUBSTRING (T1.calmonth,1,4),'-',SUBSTRING (T1.calmonth,5,2)) as calmonth
,SUM(wbzq_12_24_area +  wbzq_24_area
)as "一年以上库存面积"
,SUM(ZKCMJ)as "全部库存面积"
from 
dm.dm_fin_stock_detail_accage_t_2023 t1
left join 
dwimd.dwi_md_data_material_general_t mt
on t1.material  = mt.material_num 
where 
1=1
and t1.zdpsyb in ('11000001','11000003')
and coalesce(t1.zdpzgsdq,'NAL') not in ('33000033','33000014','33000016','33000018','33000001')
and  CONCAT (SUBSTRING (T1.calmonth,1,4),'-',SUBSTRING (T1.calmonth,5,2)) = '${PERIOD_ID_M}'
group by CONCAT (SUBSTRING (T1.calmonth,1,4),'-',SUBSTRING (T1.calmonth,5,2))
)
,detailaccage2 as (
select 
calmonth,
'ZB21' as zb_code,
case when 全部库存面积 = 0 then 0 else  一年以上库存面积/全部库存面积 end as 当月实绩,
0 as 累计实绩,
0 as 季度累计实绩
from 
detailaccage
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB21' as zb_code,
0 as 当月实绩,
case when SUM(全部库存面积) = 0 then 0 else SUM(一年以上库存面积)/SUM(全部库存面积) end  as 累计实绩,
0 as 季度累计实绩
from 
detailaccage
where CALMONTH between substr('${PERIOD_ID_M}',1,4)||'-01' and '${PERIOD_ID_M}'
union all 
select 
'${PERIOD_ID_M}' as calmonth,
'ZB21' as zb_code,
0 as 当月实绩,
0 as 累计实绩,
case when SUM(全部库存面积) = 0 then 0 else SUM(一年以上库存面积)/SUM(全部库存面积) end  as 季度累计实绩
from 
detailaccage
where    
to_char((SUBSTRING(calmonth,1,7) || '-01')::date, 'yyyyQ') = to_char(('${PERIOD_ID_M}' || '-01')::date, 'yyyyQ') 
and (SUBSTRING(calmonth,1,7) || '-01') <= SUBSTRING(concat('${PERIOD_ID_M}' ,'-01')+ INTERVAL '1 MONTH' - INTERVAL '1 day', 1, 10)
)
,AMT as (
select 
calmonth,
zb_code,
sum(当月实绩)*100 as CM_STAT_AMT,
sum(累计实绩)*100 as ACC_STAT_AMT,
sum(季度累计实绩)*100 as QTR_TGT_AMT
from 
detailaccage2
group by 
calmonth,
zb_code
)
insert into DM.DM_NORTH_SM_T
select 
T1.CALMONTH,
T1.ZB_CODE,
T2.indicator_name,
T2.definition,
T2.zb_stat_desc,
T2.indicator_owner,
T2.target_unit,
T2.monthly_target_amount,
T2.monthly_cumulative_amount,
T1.CM_STAT_AMT,
T1.ACC_STAT_AMT,
T2.quarterly_target,
T1.QTR_TGT_AMT,
T2.annual_target,
T1.ACC_STAT_AMT as CY_STAT_AMT,
CURRENT_TIMESTAMP as DW_LAST_UPDATE_DATE,
T2.pjfx
from 
AMT T1
left join
dwrfin.DWR_FIN_POLARIS_TARGET_F T2
on 
T1.ZB_CODE  =  T2.indicator_code 
and T1.CALMONTH = T2.cal_month
