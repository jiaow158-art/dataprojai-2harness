with order_pick as ( 
select distinct olt.opportunity_number --商机号 
,olp.pick_number --提货单 
from SDI.SDI_S_ORDER_LT_PICK_test_1034 olp 
left join SDI.SDI_S_ORDER_LT_test_1029 olt on olt.id = olp.order_id --and olt.DELETE_FLAG != 1
where olp.pick_number IS NOT NULL 
and olt.opportunity_number is not null -- and olp.DELETE_FLAG != 1 order表用status控制失效 
) 
--商机信息 
,business_opp as ( 
select business_code 
,business_name 
,customer_code 
,customer_name 
,sale_dept 
,sale_dept_describe 
from sdi.sdi_business_opportunity_1033_test bot 
join ( select DISTINCT opportunity_number from order_pick ) t1 
on t1.opportunity_number = bot.business_code 
) 
,view_mat_price as ( select p.material --物料 
,p.material_description --物料描述 
,p.review_quotation --评审报价 
,p.review_dosage --评审用量 
,p.support_point_rate / 100 AS   support_point_rate --积分比率
,p.review_quotation * p.review_dosage as total_amount --总金额 
,vi.business_code --商机号 
,vi.business_name --商机名称 
,vi.project_id --项目ID 
,vi.project_code --项目编号
,order_pick.pick_number --提货单 
from sDI.SDI_project_review_jjjxs_material_price_TEST_1033 p 
join sDI.SDI_PROJECT_REVIEW_jjjxs_TEST_1033 vi on vi.project_id = p.project_id AND vi.DELETE_FLAG = 0 
inner join order_pick on order_pick.opportunity_number = vi.business_code 
where p.delete_flag = 0 --ltc表用delete标识控制失效 
order by vi.business_code,p.material 
) 
--发货单信息 
, delivery as( 
select 
    lips.vbeln --发货单号 
    ,case lips.UECHA WHEN '' then lips.posnr else lips.UECHA end as POSNR --发货单行号 
    ,lips.vgbel 
    ,lips.vgpos 
    ,LIKP.WADAT_IST --实际发货日期 
    ,order_pick.opportunity_number 
    ,LIPS.LFIMG --发货数量
from sdi.SDI_likp_TEST_1001 likp 
join sdi.SDI_lips_TEST_1001 lips on lips.vbeln = likp.vbeln 
inner join order_pick on order_pick.pick_number = likp.vbeln 
where likp.WBSTK = 'C' 
-- and lips.vbeln in ( -- '8000677960','8000677962','8000677961','8000677963' -- ) 
and lips.LFIMG != 0 
) 
,sales_order as ( 
select vbap.VBELN --订单号 
    ,vbap.posnr --行项目号 
    ,vbap.MATNR --物料 
    ,vbap.KWMENG --数量 
    ,vbap.KZWI1 --金额 
    ,vbap.KZWI1 / KWMENG AS PRICE --单价 
,delivery.opportunity_number 
from sdi.SDI_VBAP_TEST_1001 vbap 
INNER join delivery on delivery.vgbel = vbap.vbeln and delivery.vgpos = vbap.posnr 
) 
--发货单数量、金额汇总 
,DELIVERY_AMOUNT AS ( 
select delivery.vbeln --发货单号 
,delivery.POSNR --发货单行号 
,sales_order.MATNR --物料 
,delivery.LFIMG --发货数量 
,sales_order.PRICE --发货单价 
,delivery.WADAT_IST
,sales_order.PRICE * delivery.LFIMG as total_amount_dl --发货金额 
,delivery.opportunity_number 
from delivery JOIN sales_order ON sales_order.VBELN = delivery.VGBEL and sales_order.posnr = delivery.vgpos 
) 
--发货单数量金额汇总 
,sales_order_amount as (
SELECT DELIVERY_AMOUNT.opportunity_number 
,DELIVERY_AMOUNT.vbeln --销售订单号 
,DELIVERY_AMOUNT.POSNR --行项目号 
,DELIVERY_AMOUNT.MATNR --物料 
,DELIVERY_AMOUNT.PRICE --发货单价 
,DELIVERY_AMOUNT.WADAT_IST  --发货日期
,SUM(DELIVERY_AMOUNT.LFIMG) as LFIMG --总发货数量 
,SUM(DELIVERY_AMOUNT.total_amount_dl) as total_amount_dl --总发货金额 
from DELIVERY_AMOUNT 
group by DELIVERY_AMOUNT.vbeln,DELIVERY_AMOUNT.POSNR,DELIVERY_AMOUNT.MATNR, DELIVERY_AMOUNT.PRICE,DELIVERY_AMOUNT.opportunity_number ,DELIVERY_AMOUNT.WADAT_IST
)
--商机、物料维度计算发货比例
,delivery_rate as (
select 
    opportunity_number 
    ,MATNR
    ,sum(LFIMG) as LFIMG
    ,sum(total_amount_dl) as total_amount_dl
    from sales_order_amount
    group by opportunity_number,MATNR
)
--OA积分使用流程数据
,oa_jfsy as (
select
    TRJE * -1 AS TRJE   --使用额度
    ,YEAR               --年度
    ,KHBM  as customer_code              --客户编码
    ,KHJC  as customer_name             --客户简称
    ,'积分使用' AS leixing  --使用类型
    ,substring(FYGS,3) as sale_dept_describe  --大区
    ,LSH                --流水号
    ,SQRQ  as WADAT_IST     --申请日期
    ,'0.85' as delivery_rate --默认0.85发货百分比用于后续计算总积分
from sdi.SDI_formtable_main_1131_TEST1004 
where  qd = '2'
  and lsh IS NOT NULL
)
--OA期初积分导入
, oa_qcdr as(
SELECT
modedatacreatedate as WADAT_IST --创建日期
,fygs as sale_dept_describe --大区
,je as total_amount_PT--金额
,nf as YEAR --年份
,'期初导入' as leixing 
,'0.85' as delivery_rate --默认0.85发货百分比用于后续计算总积分
FROM sdi.SDI_UF_WYJFZED_1004 
where qd = '2'
)


INSERT OVERWRITE TABLE DM.DM_LTC_Points_pool_TEST
--支持积分明细 
SELECT 
substr(DELIVERY_AMOUNT.WADAT_IST,1,4) as YEAR               --年度
,bo.sale_dept_describe  --大区名称
,bo.sale_dept --大区编码
,bo.customer_code --客户编码
,bo.customer_name --客户名称
,DELIVERY_AMOUNT.opportunity_number --商机编号
,bo.business_name   --商机名称
,T1.project_code    --项目评审号
,DELIVERY_AMOUNT.MATNR --物料 
,t1.material_description --物料名称
,t1.review_dosage --评审用量 
,t1.total_amount --评审总金额 
,DELIVERY_AMOUNT.vbeln --销售订单号 
,DELIVERY_AMOUNT.POSNR --行项目号 
,DELIVERY_AMOUNT.WADAT_IST  --发货日期
,DELIVERY_AMOUNT.PRICE --发货单价 
,DELIVERY_AMOUNT.LFIMG --总发货数量 
,DELIVERY_AMOUNT.total_amount_DL --总发货金额 
,t1.support_point_rate --支持积分率
,DELIVERY_AMOUNT.total_amount_dl * t1.support_point_rate as total_amount_PT --支持积分 
,delivery_rate.total_amount_dl / t1.total_amount as delivery_rate--发货百分比 
,'积分计提' as leixing
,'' as LSH                --流水号
from DELIVERY_AMOUNT 
join ( select DISTINCT material 
,business_code 
,review_dosage 
,total_amount 
,material_description 
,support_point_rate
,project_code
from view_mat_price ) t1 
on t1.business_code = DELIVERY_AMOUNT.opportunity_number and t1.material = DELIVERY_AMOUNT.MATNR
join business_opp bo on bo.business_code = DELIVERY_AMOUNT.opportunity_number 
join delivery_rate on delivery_rate.opportunity_number = DELIVERY_AMOUNT .opportunity_number and delivery_rate.MATNR = DELIVERY_AMOUNT .MATNR
--order by bo.sale_dept_describe,bo.customer_code,DELIVERY_AMOUNT.opportunity_number,DELIVERY_AMOUNT.vbeln

union all 

SELECT 
    YEAR               --年度
    ,sale_dept_describe  --大区名称
    ,'' as sale_dept --大区编码
    ,customer_code --客户编码
    ,customer_name --客户名称
    ,'' as opportunity_number --商机编号
    ,'' as business_name   --商机名称
    ,'' as project_code    --项目评审号
    ,'' as MATNR --物料 
    ,'' as material_description --物料名称
    ,'' as review_dosage --评审用量 
    ,'' as total_amount --评审总金额 
    ,'' as vbeln --销售订单号 
    ,'' as POSNR --行项目号 
    ,WADAT_IST  --发货日期
    ,'' as PRICE --发货单价 
    ,'' as LFIMG --总发货数量 
    ,'' as total_amount_DL --总发货金额 
    ,'' as support_point_rate --支持积分率
    ,TRJE as total_amount_PT --支持积分 
    ,delivery_rate --默认0.85发货百分比用于后续计算总积分
    , leixing  --使用类型
    ,LSH                --流水号
from oa_jfsy

union all 

SELECT 
    YEAR               --年度
    ,sale_dept_describe  --大区名称
    ,'' as sale_dept --大区编码
    ,'' as customer_code --客户编码
    ,'' as customer_name --客户名称
    ,'' as opportunity_number --商机编号
    ,'' as business_name   --商机名称
    ,'' as project_code    --项目评审号
    ,'' as MATNR --物料 
    ,'' as material_description --物料名称
    ,'' as review_dosage --评审用量 
    ,'' as total_amount --评审总金额 
    ,'' as vbeln --销售订单号 
    ,'' as POSNR --行项目号 
    ,WADAT_IST  --发货日期
    ,'' as PRICE --发货单价 
    ,'' as LFIMG --总发货数量 
    ,'' as total_amount_DL --总发货金额 
    ,'' as support_point_rate --支持积分率
    ,total_amount_PT --支持积分 
    ,delivery_rate --默认0.85发货百分比用于后续计算总积分
    , leixing  --使用类型
    ,'' as LSH                --流水号
from oa_qcdr
