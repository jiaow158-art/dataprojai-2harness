set hive.cbo.enable=true;
-- set hive.mapjoin.smalltable.filesize=200000000;
-- set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH BASE_LOCKOUT AS (
SELECT 
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
-- ,T1.WORK_SHOP_CODE
-- ,T1.WORK_SHOP_NAME
-- ,T1.PRODUCT_LINE_CODE
-- ,T1.PRODUCT_LINE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
,MIN(T.START_DATE) AS START_DATE
,MAX(T.END_DATE) AS END_DATE
,MAX(T.LOCKOUT_DAYS) AS LOCKOUT_DAYS
FROM DWROTD.DWR_PAM_BASE_LOCKOUT_DATE_T T
WHERE 1=1  
AND PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
GROUP BY 
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
)

,ENERGYCONSUME AS (
SELECT 
 CASE WHEN T.BUSINESS_TYPE_CODE = '1' THEN 'C' WHEN T.BUSINESS_TYPE_CODE = '2' THEN 'B' ELSE '#' END BUSINESS_TYPE_CODE
,T.START_DATE
,T.PERIOD_ID_M
,T.SITE_CODE
,T.WORK_SHOP_CODE
,T1.M_PRODUCT_LINE_CODE AS PRODUCT_LINE_CODE
,T.COST_CENTER_CODE
,SUM(T.USE_QTY) AS USE_QTY
FROM DWROTD.DWR_PAM_BASE_ENERGYCONSUME_D_F T
LEFT JOIN SDI.SDI_BASE_LINES_COST_CENTER_T_1037 T1 
ON T.SITE_CODE = T1.SITE_CODE 
AND T.COST_CENTER_CODE = T1.COST_CENTER_CODE
WHERE T.BUSINESS_TYPE_CODE IN ('1','2') --气、电
AND T.PERIOD_ID_M BETWEEN SUBSTR(DATE_ADD('${PERIOD_ID_M1}'||'-01',-1),1,7) AND '${PERIOD_ID_M2}'--为拿到上月最后一天的数
GROUP BY 
 T.SITE_CODE
,T.START_DATE
,T.WORK_SHOP_CODE
,T1.M_PRODUCT_LINE_CODE
,T.PERIOD_ID_M
,CASE WHEN T.BUSINESS_TYPE_CODE = '1' THEN 'C' WHEN T.BUSINESS_TYPE_CODE = '2' THEN 'B' ELSE '#' END 
,T.COST_CENTER_CODE
)
,TMP AS (
SELECT  
 T1.STAT_MONTH
,T1.SITE_CODE 
,T1.SITE_NAME
-- ,T1.WORK_SHOP_CODE
-- ,T1.WORK_SHOP_NAME
-- ,T1.PRODUCT_LINE_CODE
-- ,T1.PRODUCT_LINE_NAME
,T1.S_PRODUCT_LINE_CODE
,T1.S_PRODUCT_LINE_NAME
,T1.PCS_SEG_CODE
,T1.PCS_SEG_NAME
,T1.START_DATE
,T1.END_DATE
,T1.LOCKOUT_DAYS --停工天数
,T2.COST_ELE AS COST_ELEMENT_CODE --成本要素
,T2.COST_ELE_DESC AS COST_ELEMENT_NAME
,T2.VALUE_RULE AS LOCKOUT_RULE  --停工取值规则
,T2.VALUE_RULE_DESC AS LOCKOUT_RULE_NAME --停工取值规则描述
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.COST_CENTER_CODE 
      ELSE '' END AS COST_CENTER_CODE --成本中心
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.COST_CENTER_DESC
      ELSE '' END AS COST_CENTER_NAME --成本中心描述
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.GENERAL_LEDGER_ACCOUNT_CODE 
      ELSE '' END AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.GENERAL_LEDGER_ACCOUNT_NAME 
      ELSE '' END AS GENERAL_LEDGER_ACCOUNT_NAME --总账科目描述
,SUM(CASE WHEN T2.VALUE_RULE = 'L1' THEN NVL(T3.LOCAL_CURRENCY_AMT,0) 
      ELSE 0 END) AS SOURCE_AMT -- 原金额
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.ITEM_TEXT 
      ELSE '' END AS LOCKOUT_END  --停工结转      
,0 AS LOCKOUT_USE_QTY  --停工用量
,0 AS LOCKOUT_AVG_AMT --停工单价
,SUM(CASE WHEN T2.VALUE_RULE = 'L1' THEN NVL(T3.LOCAL_CURRENCY_AMT,0) * NVL(T1.LOCKOUT_DAYS,0) / DAY(LAST_DAY(T1.STAT_MONTH||'-01'))
      ELSE 0 END) AS LOCKOUT_AMT --停工金额
FROM BASE_LOCKOUT T1  -- 基地停工天数表
LEFT JOIN SDI.SDI_LINESTOP_RULEVALUES_T_1037 T2  -- 产线停工取值规则表
ON T1.SITE_CODE = T2.BASE_CODE
AND T1.PCS_SEG_CODE = T2.SEGMENTATION
AND T1.STAT_MONTH BETWEEN SUBSTR(T2.DW_EFFECT_START_DATE,1,7) AND SUBSTR(T2.DW_EFFECT_END_DATE,1,7)
LEFT JOIN DWROTD.DWR_PAM_BASE_MFG_COST_T T3  --制造费用明细表
ON T2.BASE_CODE = T3.SITE_CODE
AND T2.SEGMENTATION = T3.PCS_SEG_CODE
AND T2.COST_ELE = T3.COST_ELEMENT_CODE
AND T1.S_PRODUCT_LINE_CODE = T3.M_PRODUCT_LINE_CODE
AND T1.STAT_MONTH = T3.PERIOD_ID_M  
WHERE T1.LOCKOUT_DAYS >= 15 AND T3.FEE_FLAG = '' AND T3.ITEM_TEXT = ''
GROUP BY  
 T1.STAT_MONTH
,T1.SITE_CODE 
,T1.SITE_NAME
-- ,T1.WORK_SHOP_CODE
-- ,T1.WORK_SHOP_NAME
-- ,T1.PRODUCT_LINE_CODE
-- ,T1.PRODUCT_LINE_NAME
,T1.S_PRODUCT_LINE_CODE
,T1.S_PRODUCT_LINE_NAME
,T1.PCS_SEG_CODE
,T1.PCS_SEG_NAME
,T1.START_DATE
,T1.END_DATE
,T1.LOCKOUT_DAYS --停工天数
,T2.COST_ELE 
,T2.COST_ELE_DESC 
,T2.VALUE_RULE 
,T2.VALUE_RULE_DESC
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.COST_CENTER_CODE 
      ELSE '' END 
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.COST_CENTER_DESC
      ELSE '' END 
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.GENERAL_LEDGER_ACCOUNT_CODE 
      ELSE '' END 
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.GENERAL_LEDGER_ACCOUNT_NAME 
      ELSE '' END 
,CASE WHEN T2.VALUE_RULE = 'L1' THEN T3.ITEM_TEXT 
      ELSE '' END 
UNION ALL  
SELECT  
 T1.STAT_MONTH
,T1.SITE_CODE 
,T1.SITE_NAME
-- ,T1.WORK_SHOP_CODE
-- ,T1.WORK_SHOP_NAME
-- ,T1.PRODUCT_LINE_CODE
-- ,T1.PRODUCT_LINE_NAME
,T1.S_PRODUCT_LINE_CODE
,T1.S_PRODUCT_LINE_NAME
,T1.PCS_SEG_CODE
,T1.PCS_SEG_NAME
,T1.START_DATE
,T1.END_DATE
,T1.LOCKOUT_DAYS --停工天数
,T2.COST_ELE AS COST_ELEMENT_CODE --成本要素
,T2.COST_ELE_DESC AS COST_ELEMENT_NAME
,T2.VALUE_RULE AS LOCKOUT_RULE  --停工取值规则
,T2.VALUE_RULE_DESC AS LOCKOUT_RULE_NAME --停工取值规则描述
,CASE WHEN T2.VALUE_RULE = 'L2' AND (T2.COST_ELE = 'B' OR T2.COST_ELE = 'C') THEN T4.COST_CENTER_CODE
      ELSE '' END AS COST_CENTER_CODE --成本中心
,CASE WHEN T2.VALUE_RULE = 'L2' AND (T2.COST_ELE = 'B' OR T2.COST_ELE = 'C') THEN T3.KTEXT
      ELSE '' END AS COST_CENTER_NAME --成本中心描述      
,'' AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,'' AS GENERAL_LEDGER_ACCOUNT_NAME --总账科目描述
,SUM(CASE WHEN T2.VALUE_RULE = 'L3' THEN NVL(T5.FIXEDAMOUNT_SHARE,0)
          ELSE 0 END) AS SOURCE_AMT -- 原金额
,'' AS LOCKOUT_END  --停工结转      
,SUM(CASE WHEN T2.VALUE_RULE = 'L2' THEN NVL(T4.USE_QTY,0)
          ELSE 0 END) AS LOCKOUT_USE_QTY  --停工用量
,CASE WHEN T2.VALUE_RULE = 'L2' THEN NVL(T6.PRICE,0)    
      ELSE 0 END AS LOCKOUT_AVG_AMT --停工单价
,SUM(CASE WHEN T2.VALUE_RULE = 'L2' THEN NVL(T4.USE_QTY,0) * NVL(T6.PRICE,0)
          WHEN T2.VALUE_RULE = 'L3' THEN T5.FIXEDAMOUNT_SHARE 
          ELSE 0 END) AS LOCKOUT_AMT --停工金额
FROM DWROTD.DWR_PAM_BASE_LOCKOUT_DATE_T T1  -- 基地停工天数表
LEFT JOIN SDI.SDI_LINESTOP_RULEVALUES_T_1037 T2  -- 产线停工取值规则表  
ON T1.SITE_CODE = T2.BASE_CODE
AND T1.PCS_SEG_CODE = T2.SEGMENTATION
AND T1.STAT_MONTH BETWEEN SUBSTR(T2.DW_EFFECT_START_DATE,1,7) AND SUBSTR(T2.DW_EFFECT_END_DATE,1,7)
LEFT JOIN ENERGYCONSUME T4 --电、气表
ON T1.SITE_CODE = T4.SITE_CODE
AND T1.WORK_SHOP_CODE = T4.WORK_SHOP_CODE
AND T1.S_PRODUCT_LINE_CODE = T4.PRODUCT_LINE_CODE
-- AND T1.STAT_MONTH = T4.PERIOD_ID_M
AND T4.START_DATE BETWEEN T1.START_DATE AND T1.END_DATE
AND T2.COST_ELE = T4.BUSINESS_TYPE_CODE
LEFT JOIN SDI.SDI_CSKT_1001 T3 
ON T3.KOKRS = 'DP00' AND T3.KOSTL = T4.COST_CENTER_CODE AND T3.DATBI = '99991231' AND T3.SPRAS = '1'
LEFT JOIN SDI.SDI_FIXEDVALUES_SHUTDOWN_T_1037 T5  --停工固定分摊值表
ON T1.STAT_MONTH = T5.YEAR_MONTH
AND T2.BASE_CODE = T5.BASE_CODE
AND T2.COST_ELE = T5.COST_ELE
LEFT JOIN SDI.SDI_ZTCO07PRIC_1001 T6
ON T1.PERIOD_ID_M = T6.LFGJA||'-'||T6.LFMON
AND T4.BUSINESS_TYPE_CODE = T6.ZPRICT
AND T4.COST_CENTER_CODE = T6.RCNTR
WHERE T1.LOCKOUT_DAYS >= 15
AND T1.PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
GROUP BY  
 T1.STAT_MONTH
,T1.SITE_CODE 
,T1.SITE_NAME
-- ,T1.WORK_SHOP_CODE
-- ,T1.WORK_SHOP_NAME
-- ,T1.PRODUCT_LINE_CODE
-- ,T1.PRODUCT_LINE_NAME
,T1.S_PRODUCT_LINE_CODE
,T1.S_PRODUCT_LINE_NAME
,T1.PCS_SEG_CODE
,T1.PCS_SEG_NAME
,T1.START_DATE
,T1.END_DATE
,T1.LOCKOUT_DAYS --停工天数
,T2.COST_ELE 
,T2.COST_ELE_DESC 
,T2.VALUE_RULE 
,T2.VALUE_RULE_DESC 
,CASE WHEN T2.VALUE_RULE = 'L2' AND (T2.COST_ELE = 'B' OR T2.COST_ELE = 'C') THEN T4.COST_CENTER_CODE
      ELSE '' END
,CASE WHEN T2.VALUE_RULE = 'L2' AND (T2.COST_ELE = 'B' OR T2.COST_ELE = 'C') THEN T3.KTEXT
      ELSE '' END
,CASE WHEN T2.VALUE_RULE = 'L2' THEN NVL(T6.PRICE,0)    
      ELSE 0 END
)
INSERT OVERWRITE TABLE DM.DM_PAM_BASE_PROD_LOCKOUT_FEE_T PARTITION(PERIOD_ID_M)
SELECT  
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
,T.S_PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.START_DATE
,T.END_DATE
,T.LOCKOUT_DAYS --停工天数
,T.COST_ELEMENT_CODE --成本要素
,T.COST_ELEMENT_NAME
,T.LOCKOUT_RULE  --停工取值规则
,T.LOCKOUT_RULE_NAME --停工取值规则描述
,NVL(T.COST_CENTER_CODE,'') AS COST_CENTER_CODE --成本中心
,NVL(T.COST_CENTER_NAME,'') AS COST_CENTER_NAME --成本中心描述
,NVL(T.GENERAL_LEDGER_ACCOUNT_CODE,'') AS GENERAL_LEDGER_ACCOUNT_CODE--总账科目
,NVL(T.GENERAL_LEDGER_ACCOUNT_NAME,'') AS GENERAL_LEDGER_ACCOUNT_NAME--总账科目描述
,SUM(T.SOURCE_AMT) AS SOURCE_AMT -- 原金额
,NVL(T.LOCKOUT_END,'') AS LOCKOUT_END   --停工结转      
,SUM(T.LOCKOUT_USE_QTY) AS LOCKOUT_USE_QTY  --停工用量
,NVL(T.LOCKOUT_AVG_AMT,'') AS LOCKOUT_AVG_AMT --停工单价
,SUM(T.LOCKOUT_AMT) AS LOCKOUT_AMT --停工金额
,CURRENT_TIMESTAMP () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,SUBSTR(T.STAT_MONTH,1,7) AS PERIOD_ID_M
FROM TMP T
GROUP BY 
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
,T.S_PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.START_DATE
,T.END_DATE
,T.LOCKOUT_DAYS --停工天数
,NVL(T.COST_CENTER_CODE,'') 
,NVL(T.COST_CENTER_NAME,'') 
,NVL(T.GENERAL_LEDGER_ACCOUNT_CODE,'') 
,NVL(T.GENERAL_LEDGER_ACCOUNT_NAME,'') 
,NVL(T.LOCKOUT_END,'') 
,NVL(T.LOCKOUT_AVG_AMT,'')
,T.COST_ELEMENT_CODE --成本要素
,T.COST_ELEMENT_NAME
,T.LOCKOUT_RULE  --停工取值规则
,T.LOCKOUT_RULE_NAME --停工取值规则描述
