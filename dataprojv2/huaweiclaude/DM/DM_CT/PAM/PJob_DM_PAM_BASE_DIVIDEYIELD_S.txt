--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;
INSERT OVERWRITE TABLE DM.DM_PAM_BASE_DIVIDEYIELD_S PARTITION(PERIOD_ID_M)
SELECT START_DATE, -- 日期
	   SITE_CODE, -- 基地编码
	   WORK_SHOP_CODE, -- 车间编码
	   PRODUCT_LINE_CODE, -- 产线编码
       PRODUCT_MODEL,   -- 产品规则
       MATERIAL_CODE, -- 物料编码
	   NVL(SUM(DIVIDEYIELD_AREA * ((MATERIAL_LENGTH * MATERIAL_WIDTH)/1000000)),0) AS GRADE_QTY, -- 分级数量
	   'N' AS DEL_FLAG , -- 删除数据标识
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 数据同步时间
	   NVL(SUM(FIRST_RATE_AREA * MATERIAL_LENGTH * MATERIAL_WIDTH /1000000),0) AS FIRST_RATE_QTY, -- 优等
	   NVL(SUM(ONE_GRADE_AREA * MATERIAL_LENGTH * MATERIAL_WIDTH /1000000),0) AS ONE_GRADE_QTY, -- 一级
	   NVL(SUM(SECOND_AREA * MATERIAL_LENGTH * MATERIAL_WIDTH /1000000),0) AS SECOND_QTY, -- 次品
	   NVL(SUM(WASTE_AREA * MATERIAL_LENGTH * MATERIAL_WIDTH /1000000),0) AS WASTE_QTY, -- 废品
	   BATCH_NUM, --批次号
	   SHIFT_CODE, --班次代码
	   GROUP_CODE,--班组代码
	   MATERIAL_LENGTH,--物料长度
	   MATERIAL_WIDTH,--物料宽度
	   PERIOD_ID_M -- 分区字段
FROM DWROTD.DWR_PAM_BASE_DIVIDEYIELD_F
WHERE PERIOD_ID_M >= SUBSTRING(ADD_MONTHS('${PERIOD_ID_D}',-2),1,7)  -- 3月前的1号开始，调度传昨天,格式：YYYY-MM
GROUP BY START_DATE, -- 日期
	     SITE_CODE, -- 基地编码
	     WORK_SHOP_CODE, -- 车间编码
	     PRODUCT_LINE_CODE, -- 产线编码
         PRODUCT_MODEL,   -- 产品规则
         MATERIAL_CODE, -- 物料编码
         BATCH_NUM, --批次号
         SHIFT_CODE, --班次代码
	     GROUP_CODE,--班组代码
	     MATERIAL_LENGTH,--物料长度
	     MATERIAL_WIDTH,--物料宽度
	     PERIOD_ID_M
;
