,press_prod_tmp1 as ( --压机数摊分
select 
 T.site_code
,T.work_shop_code
,T.start_date
,T.product_line_code
,T1.material_num
,T.shift_code
,T.unit_area
,round(SUM(T1.inkjet_counter_brick_num) over(partition by T1.site_code,T1.work_shop_code,T1.start_date,T1.shift_code,T1.material_num)  --喷墨机产量(分子)
/ SUM(T1.inkjet_counter_brick_num) over(partition by T1.site_code,T1.work_shop_code,T1.start_date,T1.shift_code,T1.unit_area)  --喷墨机产量(分母)

from press_prod_sum T
left join prod_glaze_line_sum T1
on T.work_shop_code = T1.work_shop_code
and T.shift_code = T1.shift_code
and T.start_date = T1.start_date
and T.unit_area = T1.unit_area
--and t.product_line_code = T1.product_line_code
-- where T.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}'
-- and T.site_code = '1800'

,DAILY_PROD AS ( --分级产量
select 
 T.start_date
,T.site_code
,T.work_shop_code
,T.product_line_code
,T.material_code as material_num 
,sum(T.DIVIDEYIELD_AREA * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000) as grade_area --产量（㎡）
,sum(T.DIVIDEYIELD_AREA) as grade_qty --产量数
,sum(T.first_rate_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as superior_area --优等品产量（㎡）
,max(T3.daily_average_production) AS daily_avg_prod_goat --日均产量目标（㎡）
,sum(T.one_grade_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as one_grade_area --一级品产量㎡
,sum(T.second_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as second_area -- 次品产量㎡
,sum(T.waste_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000) as waste_area -- 废品产量㎡
,sum(case when T.form = '产品升降单' then (T.SECOND_AREA+T.WASTE_AREA)*T.MATERIAL_LENGTH*T.MATERIAL_WIDTH / 1000000 else 0 end) as product_defect_total_area --产品降次废面积
,sum(case when T.form <>'产品升降单' then (T.first_rate_area + T.one_grade_area) else 0 end ) as superior_one_fjd_qty--分级优一产量
,sum(case when T.form = '产品升降单' then (T.SECOND_AREA+T.WASTE_AREA) else 0 end) as product_defect_total_qty ---产品降次废数量
from DWROTD.DWR_PAM_BASE_DIVIDEYIELD_F T
LEFT JOIN DM.DM_PAM_PRODUCTION_TARGET_MAINTENANCE_TABLE_T T3
ON T.site_code = T3.site_code 
AND case when T.site_code = '1300' then T.product_line_code else T.work_shop_code end = T3.product_line
AND T.start_date BETWEEN T3.start_date AND T3.end_date
AND T.product_model = T3.specification
AND T3.delete_flag = '1'
where  T.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}' 
and T.site_code = '1800'
group by 
 T.start_date
,T.site_code
,T.work_shop_code
,T.product_line_code
,T.material_code
)
