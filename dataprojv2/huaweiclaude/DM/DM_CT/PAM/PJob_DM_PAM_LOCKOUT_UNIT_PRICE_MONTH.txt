CREATE TABLE IF NOT EXISTS DM.DM_PAM_LOCKOUT_UNIT_PRICE_MONTH (
    stat_month        VARCHAR(200) COMMENT '月份',
    site_code        VARCHAR(200) COMMENT '工厂名称',
    site_name        VARCHAR(200) COMMENT '工厂代码',
    pcs_seg_name        VARCHAR(200) COMMENT '成本段',
    cur_month           NUMERIC(38,10) COMMENT '当期月份单价',
    cur_cum_month       NUMERIC(38,10) COMMENT '当期累计单价',
    cur_year_total      NUMERIC(38,10) COMMENT '当年合计单价',
    DW_LAST_UPDATE_DATE TIMESTAMP comment '最后更新日期'
) ;
DELETE FROM DM.DM_PAM_LOCKOUT_UNIT_PRICE_MONTH 
WHERE substr(stat_month,1,4) = '${PERIOD_ID_Y}';

INSERT INTO  DM.DM_PAM_LOCKOUT_UNIT_PRICE_MONTH
WITH 
month_series AS (
    SELECT distinct substr(day_dt,1,7) as stat_month from DM.DM_DIM_DATE_D where year_id = '${PERIOD_ID_Y}'
),
dim_all AS (
    SELECT DISTINCT
        site_code,
        site_name,
        pcs_seg_name
    FROM DM.DM_PAM_BASE_LOCKOUT_COST_REC_T
    WHERE substr(stat_month,1,4) = '${PERIOD_ID_Y}'
),
full_month_dim AS (
    SELECT
        d.site_code,
        d.site_name,
        d.pcs_seg_name,
        m.stat_month
    FROM dim_all d
    CROSS JOIN month_series m
    where d.site_code <> '1900'
),
base as (
    SELECT
        A.site_code,
        A.site_name,
        A.pcs_seg_name,
        A.stat_month,
        coalesce(SUM(CASE WHEN flag=0 THEN 0 ELSE -pcs_cost_total END),0) AS total_cost,
        coalesce(SUM(CASE WHEN prod_lev='优等品' THEN sum_area ELSE 0 END),0) AS total_area
    FROM full_month_dim A
    left join DM.DM_PAM_BASE_LOCKOUT_COST_REC_T B
    on A.site_code = B.site_code
    and A.site_name = B.site_name
    and A.pcs_seg_name = B.pcs_seg_name
    and A.stat_month = B.stat_month
    WHERE substr(A.stat_month,1,4) = '${PERIOD_ID_Y}' 
    GROUP BY A.site_code, A.site_name, A.pcs_seg_name, A.stat_month
),
base_c AS (
    SELECT 
        site_code,
        site_name,
        pcs_seg_name,
        stat_month,
        total_cost,
        MAX(CASE WHEN pcs_seg_name='成型段' THEN total_area END) 
            OVER (PARTITION BY site_code, stat_month) AS total_area
    FROM base
),

------------------------------------------------------
-- 各成本段 单月单价
------------------------------------------------------
month_price AS (
    SELECT
        site_code, site_name, pcs_seg_name, stat_month,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS unit_price
    FROM base_c
    GROUP BY site_code, site_name, pcs_seg_name, stat_month
),

------------------------------------------------------
-- 各成本段 累计
------------------------------------------------------
cum_calc AS (
    SELECT
        site_code, site_name, pcs_seg_name, stat_month,
        SUM(total_cost) OVER (PARTITION BY site_code, pcs_seg_name ORDER BY stat_month) AS cum_cost,
        SUM(total_area) OVER (PARTITION BY site_code, pcs_seg_name ORDER BY stat_month) AS cum_area
    FROM base_c
    WHERE substr(stat_month,1,4)::int = '${PERIOD_ID_Y}'
),
cum_price AS (
    SELECT
        site_code, site_name, pcs_seg_name, stat_month,
        CASE WHEN cum_area=0 THEN 0 ELSE cum_cost/cum_area END AS cum_price
    FROM cum_calc
),

------------------------------------------------------
-- 工厂级【所有成本段】单月 + 累计（关键：只出1行）
------------------------------------------------------
factory_all_seg AS (
    SELECT
        site_code,
        site_name,
        '所有成本段' AS pcs_seg_name,
        stat_month,
        SUM(total_cost) AS total_cost,
        SUM(total_area) AS total_area,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS unit_price
    FROM base
    WHERE substr(stat_month,1,4)::int = '${PERIOD_ID_Y}'
    GROUP BY site_code, site_name, stat_month
),
factory_cum AS (
    SELECT
        site_code,
        site_name,
        '所有成本段' AS pcs_seg_name,
        stat_month,
--        SUM(total_cost) OVER (PARTITION BY site_code ORDER BY stat_month) AS cum_cost,
--        SUM(total_area) OVER (PARTITION BY site_code ORDER BY stat_month) AS cum_area,
        CASE WHEN SUM(total_area) OVER (PARTITION BY site_code ORDER BY stat_month)=0 
             THEN 0 
             ELSE SUM(total_cost) OVER (PARTITION BY site_code ORDER BY stat_month)/SUM(total_area) OVER (PARTITION BY site_code ORDER BY stat_month)
        END AS cum_price
    FROM factory_all_seg
),

------------------------------------------------------
-- 集团级【合计+所有成本段】单月 + 累计
------------------------------------------------------
total_all_seg AS (
    SELECT
        '合计' AS site_code,
        '合计' AS site_name,
        '所有成本段' AS pcs_seg_name,
        stat_month,
        SUM(total_cost) AS total_cost,
        SUM(total_area) AS total_area,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS unit_price
    FROM base
    WHERE substr(stat_month,1,4)::int = '${PERIOD_ID_Y}'
    GROUP BY stat_month
),
total_cum AS (
    SELECT
        '合计' AS site_code,
        '合计' AS site_name,
        '所有成本段' AS pcs_seg_name,
        stat_month,
--        SUM(total_cost) OVER (ORDER BY stat_month) AS cum_cost,
--        SUM(total_area) OVER (ORDER BY stat_month) AS cum_area,
        CASE WHEN SUM(total_area) OVER (ORDER BY stat_month)=0 
             THEN 0 
             ELSE SUM(total_cost) OVER (ORDER BY stat_month)/SUM(total_area) OVER (ORDER BY stat_month)
        END AS cum_price
    FROM total_all_seg
),

------------------------------------------------------
--合并所有数据（无扩散、无重复）
------------------------------------------------------
all_data AS (
    SELECT * FROM month_price
    UNION ALL
    SELECT site_code,site_name,pcs_seg_name,stat_month,unit_price FROM factory_all_seg
    UNION ALL
    SELECT site_code,site_name,pcs_seg_name,stat_month,unit_price FROM total_all_seg
),
all_cum AS (
    SELECT * FROM cum_price
    UNION ALL
    SELECT * FROM factory_cum
    UNION ALL
    SELECT * FROM total_cum
),

------------------------------------------------------
--年度合计
------------------------------------------------------
year_total AS (
    SELECT
        site_code, site_name, pcs_seg_name,
        substr(stat_month,1,4) AS year,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS year_price
    FROM base_c
    GROUP BY site_code, site_name, pcs_seg_name, substr(stat_month,1,4)
    UNION ALL
    SELECT
        site_code, site_name, '所有成本段' AS pcs_seg_name,
        substr(stat_month,1,4) AS year,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS year_price
    FROM base
    GROUP BY site_code, site_name, substr(stat_month,1,4)
    UNION ALL
    SELECT
        '合计' AS site_code, '合计' AS site_name, '所有成本段' AS pcs_seg_name,
        substr(stat_month,1,4) AS year,
        CASE WHEN SUM(total_area)=0 THEN 0 ELSE SUM(total_cost)/SUM(total_area) END AS year_price
    FROM base
    GROUP BY substr(stat_month,1,4)
)

SELECT
    a.stat_month,      --年月
    a.site_code,      --工厂代码
    a.site_name,      --工厂名称    
    a.pcs_seg_name,   --成本段
    COALESCE(a.unit_price,0)            AS cur_month,  --当期月份单价
    COALESCE(c.cum_price,0)             AS cur_cum_month, --当期累计单价
    COALESCE(y.year_price,0)            AS cur_year_total,--当年合计单价
    CURRENT_DATE                        AS dw_last_update_date
FROM all_data a
LEFT JOIN all_cum c 
    ON a.site_code = c.site_code 
    AND a.pcs_seg_name = c.pcs_seg_name 
    AND a.stat_month = c.stat_month
LEFT JOIN year_total y 
    ON a.site_code = y.site_code 
    AND a.pcs_seg_name = y.pcs_seg_name 
    AND y.year = '${PERIOD_ID_Y}'::text
--WHERE substr(a.stat_month,1,4)::int = '${PERIOD_ID_Y}'
ORDER BY
    CASE WHEN a.site_code='合计' THEN 0 ELSE 1 END,
    CASE WHEN a.pcs_seg_name='所有成本段' THEN 0 ELSE 1 END,
    a.site_code, a.pcs_seg_name;
