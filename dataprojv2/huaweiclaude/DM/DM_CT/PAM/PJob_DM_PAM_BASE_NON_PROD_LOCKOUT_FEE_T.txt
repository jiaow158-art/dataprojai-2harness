-- SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=200000000;
set hive.auto.convert.join.noconditionaltask.size=400000000;
set hive.mapred.local.mem=4096;
-- 设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

WITH MAIN_DATA AS ( 
SELECT
 SUBSTR(T1.DAY_DT,1,7) AS STAT_MONTH
,T.SITE_CODE --'基地编码'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.COST_CENTER_CODE  --'成本中心'
,T.COST_CENTER_NAME  --'成本中心描述
,T.PRODUCT_LINE_CODE --产线编码
,T.PRODUCT_LINE_NAME --产线描述
,T.M_PRODUCT_LINE_CODE --修正产线编码
,T.M_PRODUCT_LINE_NAME --修正产线描述
FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 T --基地产线成本中心表
JOIN (SELECT DAY_DT FROM DM.DM_DIM_DATE_D 
	  WHERE SUBSTR(DAY_DT,1,7) BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}') T1 ON 1=1
WHERE T.PCS_SEG_CODE <> 'C' OR (T.PCS_SEG_CODE = 'C' AND T.M_PRODUCT_LINE_NAME IN ('辅助车间','分级车间') )  ---釉料、原料
GROUP BY  
 SUBSTR(T1.DAY_DT,1,7)
,T.SITE_CODE --'基地编码'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.COST_CENTER_CODE  --'成本中心'
,T.COST_CENTER_NAME  --'成本中心描述
,T.PRODUCT_LINE_CODE --产线编码
,T.PRODUCT_LINE_NAME --产线描述
,T.M_PRODUCT_LINE_CODE --修正产线编码
,T.M_PRODUCT_LINE_NAME --修正产线描述
)
,RAW_DATA AS (
SELECT 
 T.SITE_CODE
,T.PRODUCT_LINE_A
,T1.M_PRODUCT_LINE_CODE AS M_PRODUCT_LINE_CODE
,T1.M_PRODUCT_LINE_NAME AS M_PRODUCT_LINE_NAME
,SUBSTR(T.START_DATE,1,10) AS START_DATE
,SUBSTR(T.END_DATE,1,10) AS END_DATE
,T.DAILY_AVERAGE_PRODUCTION
FROM DM.DM_PAM_PRODUCTION_TARGET_MAINTENANCE_TABLE_T T
LEFT JOIN (SELECT PRODUCT_LINE_CODE,M_PRODUCT_LINE_CODE,M_PRODUCT_LINE_NAME FROM SDI.SDI_BASE_LINES_COST_CENTER_T_1037 
WHERE PCS_SEG_CODE = 'C' GROUP BY PRODUCT_LINE_CODE,M_PRODUCT_LINE_CODE,M_PRODUCT_LINE_NAME) T1 --基地产线成本中心表
ON T.PRODUCT_LINE_A = T1.PRODUCT_LINE_CODE
WHERE T.DELETE_FLAG = 1 
AND ('${PERIOD_ID_M1}' BETWEEN SUBSTR(T.START_DATE,1,7)  AND SUBSTR(T.END_DATE,1,7) OR '${PERIOD_ID_M2}' BETWEEN SUBSTR(T.START_DATE,1,7) AND SUBSTR(T.END_DATE,1,7))
)
-- 2. 生成足够大的数字序列（覆盖最大可能的月份跨度，这里取100足够）
,NUM_SEQ AS (
    SELECT POSEXPLODE(SPLIT(SPACE(100), ' ')) AS (SEQ, VAL)
)
-- 3. 生成区间内的所有年月（按月份递增）
,DATE_RANGE AS (
SELECT 
 SITE_CODE
,PRODUCT_LINE_A
,M_PRODUCT_LINE_CODE
,M_PRODUCT_LINE_NAME
,DATE_FORMAT(ADD_MONTHS(TO_DATE(START_DATE), SEQ), 'YYYY-MM') AS YEAR_MONTH    -- 从开始日期累加月份
,DAILY_AVERAGE_PRODUCTION
FROM RAW_DATA
CROSS JOIN NUM_SEQ
    -- 过滤条件：生成的年月 <= 结束日期的年月（忽略日，只比对年月）
WHERE ADD_MONTHS(TO_DATE(START_DATE), SEQ) <= LAST_DAY(TO_DATE(END_DATE))
)
,AVG_PROD AS (
SELECT 
 T.SITE_CODE
,T.PRODUCT_LINE_A
,T.M_PRODUCT_LINE_CODE
,T.M_PRODUCT_LINE_NAME
,T.YEAR_MONTH AS STAT_MONTH
,AVG(T.DAILY_AVERAGE_PRODUCTION) AS AVG_PORD_NUM
FROM DATE_RANGE T
GROUP BY 
 T.SITE_CODE
,T.PRODUCT_LINE_A
,T.YEAR_MONTH
,T.M_PRODUCT_LINE_CODE
,T.M_PRODUCT_LINE_NAME
)

,BASE_LOCKOUT AS (
SELECT 
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
,MIN(T.START_DATE) AS START_DATE
,MAX(T.END_DATE) AS END_DATE
,MAX(T.LOCKOUT_DAYS) AS LOCKOUT_DAYS
FROM DWROTD.DWR_PAM_BASE_LOCKOUT_DATE_T T
WHERE 1=1
AND PERIOD_ID_M BETWEEN '${PERIOD_ID_M1}' AND '${PERIOD_ID_M2}'
GROUP BY 
 T.STAT_MONTH
,T.SITE_CODE 
,T.SITE_NAME
,T.PCS_SEG_CODE
,T.PCS_SEG_NAME
,T.PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线名称
)
,RELAT_LINE AS ( 
SELECT   --特殊辅助产线对应成型产线关系
 '0' AS FLAG --‘区分是否特殊辐射车间’
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T.COST_CENTER_CODE  --'成本中心'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.PRODUCT_LINE_CODE --产线编码
,T.PRODUCT_LINE_NAME --产线描述
,T.M_PRODUCT_LINE_CODE  --修正产线编码
,T.M_PRODUCT_LINE_NAME --修正产线描述
,T0.S_PRODUCT_LINE_CODE  --服务产线编码
,T0.S_PRODUCT_LINE_NAME  --服务产线描述
,T2.START_DATE --停工开始日期
,T2.END_DATE --停工结束日期
,T2.LOCKOUT_DAYS --停工天数
,SUM(NVL(T1.AVG_PORD_NUM,0)) OVER(PARTITION BY T.SITE_CODE,T.COST_CENTER_CODE,T.STAT_MONTH) AS TOATL_PROD_NUM  --覆盖产线总理论月产量
,SUM(NVL(T1.AVG_PORD_NUM,0)) OVER(PARTITION BY T.SITE_CODE,T.STAT_MONTH,T.COST_CENTER_CODE,T0.S_PRODUCT_LINE_CODE) AS PROD_NUM  --停工产线理论月产量
FROM MAIN_DATA T
JOIN SDI.SDI_RADIATION_WORKSHOP_T_1037 T0 --辅助车间辐射车间
ON T.STAT_MONTH BETWEEN SUBSTR(T0.EFFECT_START_DATE,1,7) AND SUBSTR(T0.EFFECT_END_DATE,1,7)
AND T.SITE_CODE = T0.SITE_CODE 
AND T.M_PRODUCT_LINE_CODE = T0.M_PRODUCT_LINE_CODE 
AND T.COST_CENTER_CODE = T0.COST_CENTER_CODE
LEFT JOIN AVG_PROD T1 -- 理论日产量表 (只包含成型车间的产线)
ON T.SITE_CODE = T1.SITE_CODE
AND T0.S_PRODUCT_LINE_CODE = T1.M_PRODUCT_LINE_CODE
AND T.STAT_MONTH = T1.STAT_MONTH
LEFT JOIN BASE_LOCKOUT T2 
ON T.SITE_CODE = T2.SITE_CODE
AND T.STAT_MONTH = T2.STAT_MONTH
AND T0.S_PRODUCT_LINE_CODE = T2.S_PRODUCT_LINE_CODE 
UNION ALL  
SELECT   --常规辅助产线对应成型产线关系
'1' AS FLAG --‘区分是否特殊辐射车间’
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T.COST_CENTER_CODE  --'成本中心'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.PRODUCT_LINE_CODE --产线编码
,T.PRODUCT_LINE_NAME --产线描述
,T.M_PRODUCT_LINE_CODE  --修正产线编码
,T.M_PRODUCT_LINE_NAME  --修正产线描述
,T2.S_PRODUCT_LINE_CODE  --服务产线编码
,T2.S_PRODUCT_LINE_NAME  --服务产线描述
,T2.START_DATE --停工开始日期
,T2.END_DATE --停工结束日期
,T2.LOCKOUT_DAYS --停工天数
,SUM(NVL(T1.AVG_PORD_NUM,0)) OVER(PARTITION BY T.SITE_CODE,T.STAT_MONTH) AS TOATL_PROD_NUM  --覆盖产线总理论月产量
,SUM(NVL(T1.AVG_PORD_NUM,0)) OVER(PARTITION BY T.SITE_CODE,T.STAT_MONTH,T1.PRODUCT_LINE_A) AS PROD_NUM  --停工产线理论月产量
FROM MAIN_DATA T
LEFT JOIN AVG_PROD T1 -- 理论日产量表 (只包含成型车间的产线)
ON T.SITE_CODE = T1.SITE_CODE
AND T.STAT_MONTH = T1.STAT_MONTH
LEFT JOIN BASE_LOCKOUT T2 
ON T.SITE_CODE = T2.SITE_CODE
AND T.STAT_MONTH = T2.STAT_MONTH
AND T1.PRODUCT_LINE_A = T2.PRODUCT_LINE_CODE 
WHERE T.COST_CENTER_CODE NOT IN (SELECT DISTINCT COST_CENTER_CODE FROM SDI.SDI_RADIATION_WORKSHOP_T_1037 WHERE COST_CENTER_CODE <> '')--成本中心不在辅助车间映射表

)
,TMP AS (
SELECT DISTINCT 
 T.FLAG --‘区分是否特殊辐射车间’
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T.COST_CENTER_CODE  --'成本中心'
,T.M_PRODUCT_LINE_CODE  --修正产线编码
,T.M_PRODUCT_LINE_NAME  --修正产线描述
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,T.TOATL_PROD_NUM  --覆盖产线总理论月产量
,T.PROD_NUM  --停工产线理论月产量
-- ,T1.PRODUCT_LINE_A,T2.PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_CODE --服务产线编码
,T.S_PRODUCT_LINE_NAME --服务产线描述
FROM RELAT_LINE T
-- WHERE T.SITE_CODE = '1500'
)
,TMP0 AS (
SELECT DISTINCT 
 T.FLAG --‘区分是否特殊辐射车间’
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
-- ,T.COST_CENTER_CODE  --'成本中心'
,T.M_PRODUCT_LINE_CODE  --修正产线编码
,T.M_PRODUCT_LINE_NAME  --修正产线描述
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,T.TOATL_PROD_NUM  --覆盖产线总理论月产量
,T.PROD_NUM  --停工产线理论月产量
-- ,T1.PRODUCT_LINE_A,T2.PRODUCT_LINE_CODE
,T.S_PRODUCT_LINE_CODE --服务产线编码
,T.S_PRODUCT_LINE_NAME --服务产线描述
FROM RELAT_LINE T
WHERE T.FLAG = '1'
)
,TMP1 AS (
SELECT 
 T.flag 
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T1.SITE_NAME --'基地名称'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T4.COST_ELE AS COST_ELEMENT_CODE --成本要素
,T4.COST_ELE_DESC AS COST_ELEMENT_NAME
,T4.VALUE_RULE AS LOCKOUT_RULE  --停工取值规则
,T4.VALUE_RULE_DESC AS LOCKOUT_RULE_NAME --停工取值规则描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.COST_CENTER_CODE 
      ELSE '' END AS COST_CENTER_CODE --成本中心
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.COST_CENTER_DESC 
      ELSE '' END AS COST_CENTER_NAME --成本中心描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.GENERAL_LEDGER_ACCOUNT_CODE 
      ELSE '' END AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.GENERAL_LEDGER_ACCOUNT_NAME 
      ELSE '' END AS GENERAL_LEDGER_ACCOUNT_NAME --总账科目描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.ITEM_TEXT 
      ELSE '' END AS LOCKOUT_END  --停工结转  
,SUM(CASE WHEN T4.VALUE_RULE = 'L1' THEN NVL(T5.LOCAL_CURRENCY_AMT,0)
      ELSE 0 END) AS SOURCE_AMT -- 原金额
-- ,SUM(T.PROD_NUM) AS PROD_NUM  --停工产线理论月产量
,T.PROD_NUM  --停工产线理论月产量
,T.TOATL_PROD_NUM  --覆盖产线总理论月产量
-- ,SUM(T.TOATL_PROD_NUM) AS TOATL_PROD_NUM  --覆盖产线总理论月产量
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线描述
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,DAY(LAST_DAY(T.STAT_MONTH||'-01')) AS DAYS_IN_MONTH  
FROM TMP T
LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D T1 -- 基地维表
ON T.SITE_CODE = T1.SITE_CODE
-- LEFT JOIN BASE_LOCKOUT T2
-- -- ON T.S_PRODUCT_LINE_CODE = T2.S_PRODUCT_LINE_CODE
-- ON T.SITE_CODE = T2.SITE_CODE
-- AND T.PCS_SEG_CODE = T2.PCS_SEG_CODE
-- AND T.STAT_MONTH = T2.STAT_MONTH
LEFT JOIN SDI.SDI_LINESTOP_RULEVALUES_T_1037 T4  -- 产线停工取值规则表
ON T.SITE_CODE = T4.BASE_CODE
AND T.PCS_SEG_CODE = T4.SEGMENTATION
AND CASE WHEN T4.SEGMENTATION = 'C' THEN T4.FLAG = 'Y' ELSE 1=1 END
AND T.STAT_MONTH BETWEEN SUBSTR(T4.DW_EFFECT_START_DATE,1,7) AND SUBSTR(T4.DW_EFFECT_END_DATE,1,7)
LEFT JOIN DWROTD.DWR_PAM_BASE_MFG_COST_T T5  --制造费用明细表
ON T4.BASE_CODE = T5.SITE_CODE
AND T4.SEGMENTATION = T5.PCS_SEG_CODE
AND T4.COST_ELE = T5.COST_ELEMENT_CODE
AND T.STAT_MONTH = T5.PERIOD_ID_M
AND T.M_PRODUCT_LINE_CODE = T5.M_PRODUCT_LINE_CODE 
AND T.COST_CENTER_CODE = T5.COST_CENTER_CODE 
WHERE T5.FEE_FLAG = '' AND T5.ITEM_TEXT = ''
AND T.LOCKOUT_DAYS >= 15 --只取停工大于等于15天的成型产线
GROUP BY  
 T.FLAG
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T1.SITE_NAME --'基地名称'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T4.COST_ELE  --成本要素
,T4.COST_ELE_DESC
,T4.VALUE_RULE   --停工取值规则
,T4.VALUE_RULE_DESC  --停工取值规则描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.COST_CENTER_CODE 
      ELSE '' END  --成本中心
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.COST_CENTER_DESC 
      ELSE '' END  --成本中心描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.GENERAL_LEDGER_ACCOUNT_CODE 
      ELSE '' END  --总账科目
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.GENERAL_LEDGER_ACCOUNT_NAME 
      ELSE '' END  --总账科目描述
,CASE WHEN T4.VALUE_RULE = 'L1' THEN T5.ITEM_TEXT 
      ELSE '' END   --停工结转      
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线描述
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,T.PROD_NUM  --停工产线理论月产量
,T.TOATL_PROD_NUM  --覆盖产线总理论月产量
UNION ALL  
SELECT 
 T.flag 
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T1.SITE_NAME --'基地名称'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T4.COST_ELE AS COST_ELEMENT_CODE --成本要素
,T4.COST_ELE_DESC AS COST_ELEMENT_NAME
,T4.VALUE_RULE AS LOCKOUT_RULE  --停工取值规则
,T4.VALUE_RULE_DESC AS LOCKOUT_RULE_NAME --停工取值规则描述
,'' AS COST_CENTER_CODE --成本中心
,'' AS COST_CENTER_NAME --成本中心描述
,'' AS GENERAL_LEDGER_ACCOUNT_CODE --总账科目
,'' AS GENERAL_LEDGER_ACCOUNT_NAME --总账科目描述
,'' AS LOCKOUT_END  --停工结转    
,SUM(CASE WHEN T4.VALUE_RULE = 'L3' THEN NVL(T6.FIXEDAMOUNT_SHARE,0) ELSE 0 END) AS SOURCE_AMT -- 原金额
,T.PROD_NUM AS PROD_NUM  --停工产线理论月产量
,T.TOATL_PROD_NUM AS TOATL_PROD_NUM  --覆盖产线总理论月产量
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线描述
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,DAY(LAST_DAY(T.STAT_MONTH||'-01')) AS DAYS_IN_MONTH  
FROM TMP0 T
-- LEFT JOIN BASE_LOCKOUT T2 
-- ON T.S_PRODUCT_LINE_CODE = T2.S_PRODUCT_LINE_CODE
-- AND T.SITE_CODE = T2.SITE_CODE
-- AND T.PCS_SEG_CODE = T2.PCS_SEG_CODE
-- AND T.STAT_MONTH = T2.STAT_MONTH
LEFT JOIN DWRDIM.DWR_DIM_PAM_SITE_D T1 -- 基地维表
ON T.SITE_CODE = T1.SITE_CODE
LEFT JOIN SDI.SDI_LINESTOP_RULEVALUES_T_1037 T4  -- 产线停工取值规则表
ON T.SITE_CODE = T4.BASE_CODE
AND T.PCS_SEG_CODE = T4.SEGMENTATION
AND CASE WHEN T4.SEGMENTATION = 'C' THEN T4.FLAG = 'Y' ELSE 1=1 END
AND T.STAT_MONTH BETWEEN SUBSTR(T4.DW_EFFECT_START_DATE,1,7) AND SUBSTR(T4.DW_EFFECT_END_DATE,1,7)
LEFT JOIN SDI.SDI_FIXEDVALUES_SHUTDOWN_T_1037 T6  --停工固定分摊值表
ON T.STAT_MONTH = T6.YEAR_MONTH
AND T4.BASE_CODE = T6.BASE_CODE
AND T4.COST_ELE = T6.COST_ELE
WHERE T.LOCKOUT_DAYS >= 15 --只取停工大于等于15天的成型产线
GROUP BY  
 T.FLAG
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T1.SITE_NAME --'基地名称'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T4.COST_ELE  --成本要素
,T4.COST_ELE_DESC
,T4.VALUE_RULE   --停工取值规则
,T4.VALUE_RULE_DESC  --停工取值规则描述
,T.S_PRODUCT_LINE_CODE  --服务产线编码
,T.S_PRODUCT_LINE_NAME  --服务产线描述
,T.START_DATE --停工开始日期
,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,T.PROD_NUM  --停工产线理论月产量
,T.TOATL_PROD_NUM   --覆盖产线总理论月产量
)
INSERT OVERWRITE TABLE DM.DM_PAM_BASE_NON_PROD_LOCKOUT_FEE_T PARTITION(PERIOD_ID_M)
SELECT  
 T.FLAG -- 0特殊辅助产线 1常规辅助产线
,T.STAT_MONTH --'年月'
,T.SITE_CODE --'基地编码'
,T.SITE_NAME --'基地名称'
,T.PCS_SEG_CODE --'工序段编码'
,T.PCS_SEG_NAME --'工序段名称'
,T.COST_ELEMENT_CODE --成本要素编码
,T.COST_ELEMENT_NAME --成本要素名称
,T.LOCKOUT_RULE  --停工取值规则
,T.LOCKOUT_RULE_NAME --停工取值规则描述
,T.COST_CENTER_CODE --成本中心
,T.COST_CENTER_NAME --成本中心描述

,T.END_DATE --停工结束日期
,T.LOCKOUT_DAYS --停工天数
,T.SOURCE_AMT * (CASE WHEN T.TOATL_PROD_NUM <> 0 THEN T.PROD_NUM / T.TOATL_PROD_NUM ELSE 0 END) * (T.LOCKOUT_DAYS / T.DAYS_IN_MONTH) AS LOCKOUT_AMT --停工金额
,CURRENT_TIMESTAMP () AS DW_LAST_UPDATE_DATE -- '最后更新时间'
,SUBSTR(T.STAT_MONTH,1,7) AS PERIOD_ID_M
FROM TMP1 T
WHERE T.SOURCE_AMT <> 0
