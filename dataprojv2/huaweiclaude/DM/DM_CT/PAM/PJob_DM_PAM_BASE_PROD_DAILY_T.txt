WITH DALIY_USE AS ( --能源消耗
select 
 A.start_date
,A.site_code
,A.product_line_code
,case when A.s_work_shop_code in ('CJ_YCYL1','CJ_YCYL2','DB_QY01') then 'CJ_YCYL1' --重庆原料车间
      when A.work_shop_code in ('CJ_HSE','CJ_HNYL','CJ_HNMQZ') then 'CJ_HNYLCJ' --湖南原料车间
      when A.work_shop_code = 'CJ_YLFZ' then 'WS_YL' --清远原料车间
      else A.work_shop_code end work_shop_code
,sum(case when A.business_type_code = 1 then A.use_qty else 0 end)  as ELEC_USE_QTY --用电量
,sum(case when A.business_type_code = 2 then A.use_qty else 0 end ) as GAS_USE_QTY --用气量
,sum(case when A.business_type_code = 3 then A.use_qty else 0 end ) as COAL_USE_QTY --用煤量
,sum(case when A.business_type_code = 4 then A.use_qty else 0 end ) as MC_USE_QTY --用物耗量
,max(B1.conversion_coefficient) as gas_coefficient --天然气折标系数
,max(B2.conversion_coefficient) as elec_coefficient --电力折标系数
,max(B3.conversion_coefficient) as coal_coefficient --煤折标系数
from DWROTD.DWR_PAM_BASE_ENERGYCONSUME_D_F A
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B1
on A.site_code = B1.site_code
and A.work_shop_code = B1.work_shop_code
and A.start_date between B1.start_date and B1.end_date
and B1.delete_flag = 1 
and B1.fuel_category = '天然气'
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B2
on A.site_code = B2.site_code
and A.work_shop_code = B2.work_shop_code
and A.start_date between B2.start_date and B2.end_date
and B2.delete_flag = 1 
and B2.fuel_category = '电力'
left join DM.DM_PAM_ENERGY_CONVERSION_COEFFICIENT_T B3
on A.site_code = B3.site_code
and A.work_shop_code = B3.work_shop_code
and A.start_date between B3.start_date and B3.end_date
and B3.delete_flag = 1 
and B3.fuel_category = '煤'
where  A.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}' 
group by 
 A.start_date
,A.site_code
,A.product_line_code
,case when A.s_work_shop_code in ('CJ_YCYL1','CJ_YCYL2','DB_QY01') then 'CJ_YCYL1' 
      when A.work_shop_code in ('CJ_HSE','CJ_HNYL','CJ_HNMQZ') then 'CJ_HNYLCJ'
      when A.work_shop_code = 'CJ_YLFZ' then 'WS_YL' --清远原料车间
      else A.work_shop_code end
)

,DAILY_PROD AS ( --分级产量
select

,max(T3.daily_average_production) AS daily_avg_prod_goat --日均产量目标（㎡）
,sum(T.DIVIDEYIELD_AREA * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000) as grade_area --产量（㎡）
,sum(T.DIVIDEYIELD_AREA) as grade_qty --产量数
,sum(T.first_rate_area * T.MATERIAL_LENGTH * T.MATERIAL_WIDTH / 1000000 ) as superior_area --优等品产量（㎡）
from DWROTD.DWR_PAM_BASE_DIVIDEYIELD_F T
LEFT JOIN DM.DM_PAM_PRODUCTION_TARGET_MAINTENANCE_TABLE_T T3
ON T.site_code = T3.site_code 
AND case when T.site_code = '1300' then T.product_line_code else T.work_shop_code end = T3.product_line
AND T.start_date BETWEEN substr(T3.start_date,1,10) AND substr(T3.end_date,1,10)  --不取前十位会导致类型不一致而取不到
AND T.product_model = T3.specification
AND T3.delete_flag = '1'
where  T.start_date between '${PERIOD_ID_D1}' and '${PERIOD_ID_D2}' 
group by 
 T.start_date
,T.site_code
,T.work_shop_code
,T.product_line_code
)
