create  TABLE DM.DM_LTC_ORDER_TRANSPORT_FEE_T as 
with SIGN_data01 as(
SELECT
     TO1.ORDERTYPE AS ORDER_TYPE , -- 订单类型
	 T1.NAME AS ORDER_TYPE_NAME, -- 订单类型名称
	 DPS.OSS_ORDER_NUM, -- 销售订单头编码
     DPS.PLAN_ORDER_NUM, -- 计划订单头编码
     DPS.PLAN_ORDER_ITEM_NUM, -- 计划订单行编码
	 TD.DO_NO AS TMS_NO , -- 运单号
	 TD.STATUS AS TMS_STATUS, --运单状态
	 T2.NAME AS TMS_STATUS_NAME, --运单状态名称
	 TDD.DO_LINENO AS TMS_LINE_NO, --运单明细
	 TD.ORDER_NO,	--订单单号
	 TO1.EXTERNAL_ORDER_CODE, -- 订单的外部订单号
	 TDD.ORDER_LINENO, --订单明细行号
	 TDD.EXTERNNO,	--SAP单号
	 TDD.EXTERNLINENO,	--SAP提货单行号
	 TDD.PLAN_NO , -- 发运计划号
	 TDD.SAP_PLAN_NO, --SAP装载计划号
	 TDD.SALEORDER_NO,	--SAP销售订单号
	 TDD.PROJECT_PLAN,	--SAP项目计划号
	 TD.WBS AS WBS_NUM, -- WBS编号
	 T3.POST1 AS WBS_NAME, --WBS项目名称
	 TDD.QUANTITY AS DEMAND_QTY, -- 需求数量
	 TO1.REQUEST_DELIVERY_DATE AS DEMAND_DATE, -- 需求时间
	 T6.MATERIAL_ID, --物料ID
	 TDD.SKU_CODE AS MATERIAL_NUM, -- 物料编码
	 T6.MATERIAL_NAME ,--物料名称
	 T6.LENGTH_1, --长度
	 T6.WIDTH, --宽度
	 T6.DIMENSION, --规格
	 T6.EXTERNAL_MATL_GROUP_CODE, --外部物料组
	 T6.MATL_GROUP_CODE, --物料组
	 T6.PRODUCT_LEVEL_CODE, --产品层次
	 T6.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE, --产品渠道编码
	 T7.CONFIG_NAME AS PRODUCT_CHANNEL_NAME, --产品渠道名称
	 TD.WMS_PICKING_STATUS, --WMS拣货状态
	 TD.ENTRYDT AS CREATE_DATE, --创建时间
	 TD.CARGO_OWNER_CODE,	--运营中心编码
	 TD.SHIPPER_CODE,	--发货方编码
	 TD.CARRIER_NAME,	--承运商
	 TD.CARRIER_CODE,	--承运商编码
	 TD.LOADING_SITE_CODE, --提货站点
	 TD.LOADING_SITE_NAME, --提货站点名称
	 TD.LOADING_ADDR,	--提货详实地址
	 TD.UNLOADING_SITE_CODE, --卸货站点编码
	 TD.UNLOADING_SITE_NAME, --卸货站点名称
	 TD.SHIP_METHOD, --运输方式编码
	 TD.OUTSOURCING, --三方订单
	 TD.PICKING_LOCATION,	--拣货地点
	 TD.DESCRIPTION, --备注
	 TD.RMS_FLAG,	--退货订单
	 TD.SUSR7 AS CABINET_NUMBER, --柜号
	 TD.SUSR8 AS WOODEN_FRAME_QTY, --木架数量
	 TD.SUSR9 AS DELIVER_REQUIREMENT, --发货要求
	 TD.EXPRESS_NO, --承运商面单号
	 TD.LOADING_CONTACT, --提货联系人
	 TD.LOADING_PHONE, --提货联系人电话
	 s3.STREET_NAME AS LOADING_PROVINCE,	--提货省
	 s4.STREET_NAME AS LOADING_CITY,	--提货市
	 s5.STREET_NAME AS LOADING_COUNTY, --提货区
	 s6.STREET_NAME AS LOADING_TOWN,	--提货镇
	 TD.UNLOADING_CONTACT, --卸货联系人1
	 TD.UNLOADING_PHONE, --卸货联系电话1
	 s1.STREET_NAME AS UNLOADING_PROVINCE, --卸货省份
	 s2.STREET_NAME AS UNLOADING_CITY, --卸货城市
	 s9.STREET_NAME AS UNLOADING_COUNTY,	--卸货区县
	 s10.STREET_NAME AS UNLOADING_TOWN, --卸货乡镇
	 TD.PLAN_SHIP_DATE, --计划提货时间
	 TD.DEMAND_ARRIVE_DATE, --计划到货时间
	 TD.SHIP_LINE, --路线编码
	 TD.COST_CENTER, --成本中心
	 TD.ACTUAL_SHIP_DATE,	--实际到货时间
	 TD.ACTUAL_ARRIVE_DATE, --实际送达时间
	 TD.REQUEST_DELIVERY_DATE, --要求到货日期
	 TD.TOTAL_GOODS_PIECE, --总件数
	 TD.ACTUAL_GOODS_PIECE, --实际件数
	 TD.TOTAL_GOODS_COLUME, --总体积
	 TD.ACTUAL_GOODS_COLUME, --实际总体积
	 TD.TOTAL_ROUGH_WEIGHT, --总体重
	 TD.ACTUAL_ROUGH_WEIGHT, --实际总毛重
	 TD.TOTAL_NET_WEIGHT,	--总净重
	 TD.ACTUAL_NET_WEIGHT, --实际总净重
	 TD.TOTAL_BOXES, --总箱数
	 TD.ACTUAL_BOXES,	--实际总箱数
	 TD.TOTAL_GOODS_AMOUNTS, --总金额
	 TD.ACTUAL_GOODS_AMOUNTS,	--实际金额
	 TD.OWNER_CODE, --货主编码
	 TD.OWNER_NAME, --货主名称
	 TD.SETTLEMENT_METHOD, --结算方式
	 T4.CUST_ID, -- 客户ID
	 TD.CUSTOMER_CODE, --客户编码
	 T4.CUST_NAME, --客户名称
	 T4.ADDRESS_NAME, --客户所在地址
	 TD.SUSPENDDESCRIPTION, --暂挂描述
	 TD.SUSPENDREASON, --暂挂原因编码
	 TD.DELIVERY_TAG, --发货标记
	 TD.ALLOCATECODE, --分配策略
	 TD.WAREHOUSEKEY, --仓库编码
	 TD.COMPANY_ID, --公司ID
	 TD.REGIONAL, --收货所属大区
	 TD.CAR_TYPE,	--包车车型
	 TD.SPECIAL_CAR, --是否专车
	 TD.REQUETSTID, --OA流程ID
	 TD.MACHINING_NOTES, --加工备注
	 TD.MACHINING_NO,	--加工单号 
	 TD.ACTUAL_REVEIVED_QUANTITY,	--实际运输量
	 TDD.WEIGHT, --净重
	 TDD.ACTUAL_FREIGHT_QUANTITY, --实运量
	 TDD.TENANT_ID, --租户号
	 TDD.SKU_TYPE_CODE, --物料类型名称
	 TDD.VOLUME, --体积
	 TDD.PRICE, --价格
	 TDD.AMOUNT, --金额
	 TDD.GROSS_WEIGHT,	--重量
	 TDD.BOXES, --箱数
	 TDD.LOTNO, --批次
	 TDD.COST_CENTER_CODE,	--成本中心编码
	 T5.FACTORY_ID AS FACTORY_WERKS_ID, --工厂ID
	 TDD.FACTORY_CODE, -- 工厂代码
	 T5.FACTORY_NAME AS FACTORY_WERKS_NAME, --工厂名称
	 TDD.STORAGE_LOCATION, -- 库存地
	 TDD.WEIGHTUNIT, -- 单位
	 TD.UNLOADING_ADDR AS RECEIVING_ADDR, -- 收货地址
	 S1.STREET_NAME  AS RECEIVING_PROVINCE, -- 收货省
	 S2.STREET_NAME AS RECEIVING_CITY, -- 收货市
	 S3.STREET_NAME AS RECEIVING_COUNTY, -- 收货县
	 TDD.LICENSE_PLATE_NO , -- 车牌号
	 TDD.DRIVER_NAME , -- 司机名字
	 TDD.DRIVER_PHONE , -- 司机电话
	 TDD.LICENSENO , -- 司机证件
	 TRD.RECEIPT_NO,-- 回单号
	 TRD.OPERATOR AS SIGN_BY, -- 签收人 （确认什么状态下为签收）
	 TRD.OPER_DATETIME AS SIGN_DATE,  -- 签收时间    （确认什么状态下为签收）
	 TRD.ERP_NO AS TMS_PICK_NO, -- TMS提货单号
	 TO1.MANUAL_SINGLE_CATEGORY, --手工单类别
	 TO1.MANUAL_SMALL_CLASS, --手工单小类
	 TO1.STOCK_UP_AREA, --备货区
	 TO1.SALE_LINEGROUP, --销售组编码
	 TO1.PAYER_METHOD, --付款方方式
	 TO1.DRAWER_CODE, --开票公司编码
	 TO1.PAYER, --付款方名称
	 TO1.MACHININGER, --加工方式
	 TO1.DEPARTMENT_CODE, --部门编码
	 TO1.LOADING_METHOD, --提货方式
	 CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE, -- 同步数据时间
	 (CASE
                 WHEN ((td.mileage / 1000) > 1000) THEN
                  '>1000KM'
                 WHEN (((td.mileage / 1000) >= 500) AND
                      ((td.mileage / 1000) <= 1000)) THEN
                  '500-1000KM'
                 ELSE
                  '<500KM'
               END) AS transportation_radius_type,--运输半径类型
      T8.descr AS ship_methodname,--运输方式名称
      T9.descr AS order_department,--运营部名称
	  T10.costcenter_name AS cost_center_name,--成本中心名称
	  to1.TMS_PAY_METHOD,
	  td.CARRIER_DATA as CARRIER_DATE, -- 承运日期
	  T11.BILL_TOTAL AS TOTAL_FREIGHT_FEE, -- 总运费
	  td.susr3 as merge_do_no ,
	  t12.PLAN_NO_merge as merge_plan_no,
	  T11.fee_item_name
FROM
 SDI.SDI_tms_do_detail_1005 tdd  -- 运单行表
JOIN SDI.SDI_tms_do_1005 td     -- 运单头表
ON td.do_no = tdd.do_no
and td.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_order_1005 to1   -- 运单订单头表
ON td.order_no = to1.code
and to1.del_flag = 'N'
--LEFT JOIN TMS_ORDER_DETAIL_DATA TOD -- 运单订单数据
--ON to1.code = tod.order_code
LEFT JOIN DM.DM_LTC_ORDER_PLAN_STOCKUP_T DPS -- 提货单数据
ON TDD.EXTERNNO = DPS.DELIVERY_NO
AND TDD.EXTERNLINENO = DPS.DELIVERY_PROJECT
left join (
select trd.do_no,trd.do_lineno, 
     TR1.RECEIPT_NO,-- 回单号
	 TR1.OPERATOR, -- 签收人 （确认什么状态下为签收）
	 TR1.OPER_DATETIME ,  -- 签收时间    （确认什么状态下为签收）
	 TRD.ERP_NO,  -- TMS提货单号
	 row_number() over(partition by trd.do_no,trd.do_lineno order by tr1.oper_datetime desc ) as rn
from SDI.sdi_tms_receipt_detail_1005 trd -- 回单行表
left join SDI.SDI_tms_receipt_1005 TR1   -- 回单主表
on TR1.receipt_no = trd.receipt_no
and TR1.del_flag = 'N'
where trd.del_flag ='N'
) TRD ON TRD.do_no = TDD.do_no and TRD.do_lineno  = TDD.do_lineno and TRD.rn=1
LEFT JOIN SDI.SDI_tms_site_street_1005 s1   -- 省市区街道四及联动基础数据表
ON td.unloading_province = s1.street_code
and S1.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s2   
ON td.unloading_city = s2.street_code
and S2.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s3   
ON td.loading_province = s3.street_code
and S3.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s4   
ON td.loading_city = s4.street_code
and S4.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s5   
ON td.loading_county = s5.street_code
and S5.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s6
ON td.loading_town = s6.street_code
and S6.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s9
ON td.UNLOADING_COUNTY = s9.street_code
and S9.del_flag = 'N'
LEFT JOIN SDI.SDI_tms_site_street_1005 s10
ON td.UNLOADING_TOWN = s10.street_code
and s10.del_flag = 'N'
LEFT JOIN sdi.sdi_plat_code_1005 T1 -- TMS配置表
ON to1.ordertype = T1.CODE
AND t1.typename='TMS_ORDER_TYPE'
and t1.del_flag = 'N'
LEFT JOIN sdi.sdi_plat_code_1005 T2   -- TMS配置表
ON td.STATUS = T2.CODE
AND t2.typename='TRANSORDERSTATUS'
and t2.del_flag = 'N'
LEFT JOIN SDI.SDI_PRPS_1001 T3   -- wbs
ON td.WBS = t3.posid
AND t3.DEL_FLAG = 'N'
LEFT JOIN DWRDIM.DWR_DIM_CUST_GENERAL_D T4   -- 客户
ON td.CUSTOMER_CODE = T4.cust_num 
and t4.end_date = '4712-12-31'
LEFT JOIN DWRDIM.DWR_DIM_FACTORY_D t5   -- 工厂
ON tdd.FACTORY_CODE = t5.factory_code
and t5.end_date = '4712-12-31'
LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D t6 -- 物料
ON tdd.sku_code = t6.MATERIAL_NUM
AND t6.END_DATE = '4712-12-31'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R t7 -- 配置表
ON t6.RESERVED_FIELD_1 = t7.SRC_SYS_CONFIG_CODE
AND t7.SOURCE_CODE = 'ZTYL01'
LEFT JOIN sdi.sdi_plat_code_1005 T8   -- TMS配置表
ON td.ship_method = T8.CODE
AND T8.typename='TMS_SHIPMETHOD'
and T8.del_flag = 'N'
LEFT JOIN SDI.SDI_plat_partner_1005 T9
ON TD.susr4 = T9.partner_key
AND T9.trade_type = 'ORDER_DEPARTMENT'
AND T9.del_flag = 'N'
LEFT JOIN SDI.SDI_bms_costcenter_1005 T10
ON tdd.cost_center_code = T10.costcenter_code
AND T10.del_flag = 'N'
JOIN (
SELECT BILL_ORDER_NO,SOURCE_ORDER_LINENO,fee_item_name,SUM(BILL_TOTAL) AS BILL_TOTAL
FROM SDI.SDI_BMS_BILL_RESULT_TMS_1005 
GROUP BY BILL_ORDER_NO,SOURCE_ORDER_LINENO,fee_item_name
) T11 ON TDD.DO_NO = T11.BILL_ORDER_NO AND TDD.DO_LINENO = T11.SOURCE_ORDER_LINENO
LEFT JOIN sdi.sdi_tms_transport_plan_1005 t12 ON td.plan_no = t12.plan_no
where tdd.del_flag = 'N'
)

select t.* from SIGN_data01 t;
