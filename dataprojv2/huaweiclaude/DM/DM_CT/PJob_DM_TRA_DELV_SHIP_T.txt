WITH T AS(
SELECT DELV_NUM
      ,DELV_LINE_NUM
	  ,SUM(ACTL_DELV_QTY) AS SIGN_QTY --签收数量
  FROM DWROTD.DWR_TRA_TMS_SHIP_DTL_F A
 WHERE DO_NUM IN(SELECT DO_NUM FROM DWROTD.DWR_TRA_RCPT_ORDER_F)
 GROUP BY DELV_NUM
      ,DELV_LINE_NUM
),T1 AS(
SELECT DMAND_ID
      ,DMAND_LINE_ID
      ,DEPT_ST_NAME
      ,DELV_NUM
      ,DELV_LINE_NUM
      ,SUM(SHIP_QYT) AS SHIP_QYT
  FROM DWRLTC.DWR_BO_DSPT_ORDER_F
 WHERE IS_PAR_NAME = '否'
 GROUP BY DMAND_ID
      ,DMAND_LINE_ID
      ,DEPT_ST_NAME
      ,DELV_NUM
      ,DELV_LINE_NUM
),T2 AS(
SELECT A.DELIVERY_IM AS DELV_NUM
      ,A.DELIVERY_PROJECT AS DELV_LINE_NUM
      ,SUM(CASE WHEN DR_CR_ID = 'H_SHKZG' THEN QUANTITY ELSE 0 END) AS OUT_STOCK_QTY --出库数量
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
 WHERE SPECIAL_STOCK_FLAG_2 = 'Q_T148T'
   AND MOVE_TYPE <> '313'
 GROUP BY A.DELIVERY_IM
      ,A.DELIVERY_PROJECT
),T3 AS(
SELECT T1.DMAND_ID
      ,T1.DMAND_LINE_ID
	  ,T1.DEPT_ST_NAME
	  ,SUM(T1.SHIP_QYT) AS SHIP_QYT
	  ,SUM(COALESCE(T2.OUT_STOCK_QTY,0)) AS OUT_STOCK_QTY
	  ,SUM(COALESCE(T.SIGN_QTY,0)) AS SIGN_QTY
  FROM T1
  LEFT JOIN T2
    ON T1.DELV_NUM = T2.DELV_NUM
   AND T1.DELV_LINE_NUM = T2.DELV_LINE_NUM
  LEFT JOIN T
    ON T1.DELV_NUM = T.DELV_NUM
   AND T1.DELV_LINE_NUM = T.DELV_LINE_NUM 
 GROUP BY T1.DMAND_ID
      ,T1.DMAND_LINE_ID
	  ,T1.DEPT_ST_NAME
),T4 AS(
SELECT DISTINCT DMAND_LINE_ID,PLAN_ITEM_NUM
  FROM DWROTD.DWR_BO_DELV_SHIP_BILL_F)
,T5 AS(
SELECT T4.DMAND_LINE_ID
      ,D.SUB_BUSI_CODE
	  ,D.MD_SALE_GRP_CODE
	  ,SUM(COALESCE(D.ORDER_QTY,0)) AS ORDER_QTY
  FROM T4
  LEFT JOIN DWRLTC.DWR_BO_PLAN_ORDER_F D --自营工程
    ON T4.PLAN_ITEM_NUM = D.PLAN_ITEM_NUM
 GROUP BY T4.DMAND_LINE_ID
      ,D.SUB_BUSI_CODE
	  ,D.MD_SALE_GRP_CODE
)
INSERT OVERWRITE TABLE DM.DM_TRA_DELV_SHIP_T
SELECT A.PLAN_TYPE_NAME --计划订单类型
      ,A.BUSI_NAME AS ENGR_NAME --工程名称
	  ,A.SALE_ORG_CODE --销售组织编码
	  ,A.SALE_ORG_NAME --销售组织名称
	  ,A.MATERIAL_NUM --物料编码
	  ,A.WBS_NUM --WBS号（首层）
	  ,CASE WHEN COALESCE(D.ORDER_QTY,0) > COALESCE(E.OUT_STOCK_QTY,0) THEN '待出库'
	        ELSE '已出库'
	         END AS OUT_WH_ST_NAME --出库状态 订单数量-已出库数量>0：待出库 订单数量-已出库数量=0：已出库
	  ,E.DEPT_ST_NAME AS LGSTCS_ST_NAME --物流状态
	  ,A.DMAND_LINE_ID --要货单行ID
	  ,A.BUSI_CODE AS ENGR_CODE --工程报备号
	  ,A.DMAND_NUM --要货单号
      ,A.DMAND_ST_NAME --要货单状态描述
      ,A.DMAND_QTY --本次要货单数量
	  ,A.CREATE_DATE --要货单创建日期
	  ,COALESCE(A.DMAND_QTY,0) - COALESCE(A.SHIP_QTY,0) AS RSDU_DMAND_QTY --要货单剩余未开单数
	  ,COALESCE(B.SHIP_PLAN_QTY,0) AS SHIP_PLAN_QTY --出货计划数汇总
	  ,COALESCE(E.SHIP_QYT,0) AS SHIP_QYT --提货申请数汇总
	  ,COALESCE(E.OUT_STOCK_QTY,0) AS OUT_STOCK_QTY --出库数汇总
	  ,COALESCE(E.SIGN_QTY,0) AS SIGN_QTY --签收数量
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(B.CREATE_DATE) - UNIX_TIMESTAMP(A.CREATE_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(B.CREATE_DATE) - UNIX_TIMESTAMP(A.CREATE_DATE)),86400)) / 3600,0) || '小时','.0','') AS DMAND_CNSUM_TIME --要货单耗时 最小出货计划创建时间-要货单创建时间
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,CASE WHEN B.RSDU_SHIP_QTY > 0 THEN '待申请提货'
	        ELSE '已申请提货'
			 END AS SHIP_ST_NAME --提货状态
  FROM DWRLTC.DWR_BO_DMAND_ORDER_F A --要货单
    LEFT JOIN (SELECT DMAND_ID
                   ,DELV_DMAND_LINE_ID
				   ,MIN(CREATE_DATE) AS CREATE_DATE
				   ,SUM(CASE WHEN COALESCE(DELV_SHIP_ST_NAME,'') <> '已失效' THEN COALESCE(SHIP_PLAN_QTY,0) ELSE 0 END) AS SHIP_PLAN_QTY
				   ,SUM(CASE WHEN COALESCE(DELV_SHIP_ST_NAME,'') <> '已失效' THEN COALESCE(RSDU_SHIP_QTY,0) ELSE 0 END) AS RSDU_SHIP_QTY
               FROM DWRLTC.DWR_BO_DELV_SHIP_ORDER_F
			  GROUP BY DMAND_ID
                   ,DELV_DMAND_LINE_ID
			) B --出货计划
    ON A.DMAND_ID = B.DMAND_ID
   AND A.DMAND_LINE_ID = B.DELV_DMAND_LINE_ID
  LEFT JOIN T5 AS D --自营工程
    ON A.DMAND_LINE_ID = D.DMAND_LINE_ID 
  LEFT JOIN T3 AS E --发运委托
    ON A.DMAND_ID = E.DMAND_ID
   AND A.DMAND_LINE_ID = E.DMAND_LINE_ID
 WHERE A.DIV_NAME = '瓷砖事业部'
