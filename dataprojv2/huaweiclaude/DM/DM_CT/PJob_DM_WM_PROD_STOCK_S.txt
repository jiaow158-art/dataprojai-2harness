--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--设置动态分区为非严格模式
set hive.exec.dynamic.partition=true; 
set hive.exec.dynamic.partition.mode=nonstrict;

INSERT OVERWRITE TABLE DM.DM_WM_PROD_STOCK_S PARTITION(STAT_MONTH)
SELECT
 MATERIAL_ID  --物料ID
,MATERIAL_NUM  --物料编码
,MATERIAL_NAME  --物料名称
,EXTERNAL_MATL_GROUP_CODE  --外部物料组
,MATL_GROUP_CODE  --物料组
,SIZE_UNIT  --规格单位
,LENGTH_1  --长度
,WIDTH  --宽度
,PRODUCT_CHANNEL_CODE  --产品渠道编码
,PRODUCT_CHANNEL_NAME  --产品渠道名称
,PRODUCT_STAGE2  --产品层次2
,BRAND_NAME  --品牌名称
,FACTORY_WERKS_ID  --工厂ID
,FACTORY_WERKS_CODE  --工厂编码
,FACTORY_WERKS_NAME  --工厂名称
,INV_LOCATION  --库存地点
,SUPPLIER_OR_CREDITOR_ID  --供应商ID
,SUPPLIER_OR_CREDITOR_ACCT  --供应商编码
,SUPPLIER_OR_CREDITOR_NAME  --供应商名称
,PRODUCT_AREA_ID  --产区ID
,PRODUCT_AREA_CODE  --产区编码
,PRODUCT_AREA_NAME  --产区名称
,SUM(COALESCE(WAREHOUSE_QUANTITY,0)) AS WAREHOUSE_QUANTITY --库存量
,SUM(COALESCE(WAREHOUSE_AREA,0)) AS WAREHOUSE_AREA --库存面积（平方米）
,SUM(COALESCE(WAREHOUSE_PRICE,0)) AS WAREHOUSE_PRICE --库存金额（元）
,SUM(COALESCE(STOCK_QTY,0)) AS STOCK_QTY --库存数量
,SUM(CASE WHEN SPECIAL_STOCK_FLAG_2 IN('Q_T148T') THEN COALESCE(WAREHOUSE_QUANTITY,0) ELSE 0 END) AS PROJ_WAREHOUSE_QUANTITY --库存量
,SUM(CASE WHEN SPECIAL_STOCK_FLAG_2 IN('Q_T148T') THEN COALESCE(WAREHOUSE_AREA,0) ELSE 0 END) AS PROJ_WAREHOUSE_AREA --库存面积（平方米）
,SUM(CASE WHEN SPECIAL_STOCK_FLAG_2 IN('Q_T148T') THEN COALESCE(WAREHOUSE_PRICE,0) ELSE 0 END) AS PROJ_WAREHOUSE_PRICE --库存金额（元）
,SUM(CASE WHEN SPECIAL_STOCK_FLAG_2 IN('Q_T148T') THEN COALESCE(STOCK_QTY,0) ELSE 0 END) AS PROJ_STOCK_QTY --库存数量
,datediff(date_format('${PERIOD_ID_D}','yyyy-MM-dd'), MIN_VOUCHER_POST_DATE) AS MIN_WAREHOUSE_DATE  --批次最早过账日期
,datediff(date_format('${PERIOD_ID_D}','yyyy-MM-dd'), MAX_VOUCHER_POST_DATE) AS MAX_WAREHOUSE_DATE --批次最晚过账日期
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
,STAT_MONTH
FROM DM.DM_WM_PRODSTOCK_OUT_IN_T
WHERE STAT_MONTH = REPLACE(SUBSTR(date_format('${PERIOD_ID_D}','yyyy-MM-dd'),1,7),'-','')
GROUP BY
MATERIAL_ID  --物料ID
,MATERIAL_NUM  --物料编码
,MATERIAL_NAME  --物料名称
,EXTERNAL_MATL_GROUP_CODE  --外部物料组
,MATL_GROUP_CODE  --物料组
,SIZE_UNIT  --规格单位
,LENGTH_1  --长度
,WIDTH  --宽度
,PRODUCT_CHANNEL_CODE  --产品渠道编码
,PRODUCT_CHANNEL_NAME  --产品渠道名称
,PRODUCT_STAGE2  --产品层次2
,BRAND_NAME  --品牌名称
,FACTORY_WERKS_ID  --工厂ID
,FACTORY_WERKS_CODE  --工厂编码
,FACTORY_WERKS_NAME  --工厂名称
,INV_LOCATION  --库存地点
,SUPPLIER_OR_CREDITOR_ID  --供应商ID
,SUPPLIER_OR_CREDITOR_ACCT  --供应商编码
,SUPPLIER_OR_CREDITOR_NAME  --供应商名称
,PRODUCT_AREA_ID  --产区ID
,PRODUCT_AREA_CODE  --产区编码
,PRODUCT_AREA_NAME  --产区名称
,datediff(date_format('${PERIOD_ID_D}','yyyy-MM-dd'), MIN_VOUCHER_POST_DATE)
,datediff(date_format('${PERIOD_ID_D}','yyyy-MM-dd'), MAX_VOUCHER_POST_DATE)
,STAT_MONTH
