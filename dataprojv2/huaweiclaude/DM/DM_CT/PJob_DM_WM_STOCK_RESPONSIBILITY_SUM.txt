with tmp as(
	select resp_system,
		   material_num,
		   sum(q_stock) as q_stock, 
		   0  as res_prepare_stock_qty,
		   0 as exclusive_stock, 
		   0 as dz1_stock,
		   0 as stock_qty
	from dm.dm_wm_stock_responsibility_total_tmp
	where stat_month = substr('${PERIOD_ID_D}',1,7)
	group by resp_system ,material_num
	union all    
	select resp_system,
		   material_num,
		   0,
		   sum(res_prepare_stock_qty),
		   0 as exclusive_stock ,
		   0 as dz1_stock,
		   0 as stock_qty
	from dm.dm_wm_stock_responsibility_res_prepare_tmp 
	group by resp_system,material_num
	union all 
	select resp_system,
		   material_num,
		   0,
		   0,
		   sum(exclusive_stock) as exclusive_stock,
		   0 as dz1_stock,
		   0 as stock_qty
	from dm.dm_wm_stock_responsibility_exclusive_tmp 
	group by resp_system,material_num
	union all
	select resp_system,
		   material_num,
		   0,
		   0,
		   0 as exclusive_stock,
		   sum(dz1_stock) as dz1_stock,
		   0 as stock_qty
	from dm.dm_wm_stock_responsibility_dz1_tmp 
	group by resp_system,material_num
	union all  
	select '基地溢出',
		   material_num,
		   0,
		   0,
		   0,
		   0,
		   sum(dist_overflow_stock_qty) as stock_qty
	from  dm.dm_wm_stock_responsibility_overproduct_tmp
	group by material_num
	union all 
	select '冻结+质检',
		   material_num,
		   0,
		   0,
		   0,
		   0,
		   sum(stock_qty) as stock_qty
	from  dm.dm_wm_stock_responsibility_djzjkc_tmp
	group by material_num
	union all 
	select item_type,
		   material_num,
		   0,
		   0,
		   0,
		   0,
		   sum(stock_qty) as stock_qty
	from  dm.dm_wm_stock_responsibility_others_tmp
	group by item_type,material_num
)
insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_SUM
select resp_system, 
       sum(q_stock) as q_stock_qty, 
       sum(res_prepare_stock_qty)  as res_prepare_stock_qty,
       sum(exclusive_stock) as exclusive_stock_qty, 
       sum(dz1_stock) as dz1_stock_qty,
       sum(stock_qty) as stock_qty,
       sum(q_stock*b.area) as q_stock_area, 
       sum(res_prepare_stock_qty*b.area)  as res_prepare_stock_area,
       sum(exclusive_stock*b.area) as exclusive_stock_area, 
       sum(dz1_stock*b.area) as dz1_stock_area,
       sum(stock_qty*b.area) as stock_area,
       current_timestamp() as dw_last_update_date
from tmp  a 
join dm.dm_wm_stock_responsibility_mat_tmp b 
On a.material_num = b.material_num 
group by resp_system;
