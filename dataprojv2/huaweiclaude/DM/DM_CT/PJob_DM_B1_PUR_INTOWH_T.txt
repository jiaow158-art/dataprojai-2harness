WITH T1 AS(
SELECT TH.DOCDATE      AS DELIVERY_DATE,		-- 提货日期
       TH.DOCNUM       AS DELIVERY_BILL_NUM,	-- 提货单据编号
       T0.DOCDATE      AS RECEIPT_DATE,			-- 收货日期
       T0.DOCNUM       AS RECEIPT_BILL_NUM,		-- 收货单据编号
       T0.CARDCODE     AS SUPPLIER_CODE,		-- 供应商编码
       T0.CARDNAME     AS SUPPLIER_NAME,		-- 供应商名称
       DD.ITEMCODE     AS MATERIAL_CODE,		-- 物料编码
       DD.DSCRIPTION   AS MATERIAL_DESC,		-- 物料描述
       --DD.QUANTITY     AS QUANTITY,				-- 数量
       T1.QUANTITY     AS QUANTITY,				-- 数量
       DD.PRICE        AS PRICE,				-- 单价
       DD.LINETOTAL    AS LINE_ACCOUNT,			-- 行合计
       DD.WHSCODE      AS WAREHOUSE_NUM,		-- 仓库编号
       DD.U_DISTNUMBER AS BATCH_NUM,			-- 批次
       DD.LINENUM      AS LINE_NUM,             --行号
       TH.INVNTSTTUS   AS DELIVERY_STATUS,		-- 交货状态 C：交货，O：未清
       DD.U_A1_BASENUM AS SAP_DELIVERY_NUM,		-- SAP提货单号
	   DD.U_A1_BASELINE AS SAP_DELIVERY_LINE_NUM, --SAP提货单行号
       DD.CUSTOMER_CODE, --客编
       DD.OpenQty as outstand_qty --剩余未清数量 TH1.OpenQty
  FROM DWIOTD.DWI_WM_B1_POR1_T DD 
  LEFT JOIN DWIOTD.DWI_WM_B1_PCH1_T TH1 
    ON DD.DOCENTRY = TH1.BASEENTRY
   AND DD.LINENUM = TH1.BASELINE
   AND DD.CUSTOMER_CODE = TH1.CUSTOMER_CODE
   AND DD.CUSTOMER_TYPE = TH1.CUSTOMER_TYPE
  LEFT JOIN DWIOTD.DWI_WM_B1_OPCH_T TH 
    ON TH1.DOCENTRY = TH.DOCENTRY
   AND TH.CUSTOMER_CODE = TH1.CUSTOMER_CODE
   AND TH.CUSTOMER_TYPE = TH1.CUSTOMER_TYPE
  LEFT JOIN DWIOTD.DWI_WM_B1_PDN1_T T1 
    ON TH1.DOCENTRY = T1.BASEENTRY
   AND TH1.LINENUM = T1.BASELINE
   AND T1.BASETYPE = 18   
   AND T1.CUSTOMER_CODE = TH1.CUSTOMER_CODE
   AND T1.CUSTOMER_TYPE = TH1.CUSTOMER_TYPE
  LEFT JOIN DWIOTD.DWI_WM_B1_OPDN_T T0
    ON T0.DOCENTRY = T1.DOCENTRY	   
   AND T1.CUSTOMER_CODE = T0.CUSTOMER_CODE
   AND T1.CUSTOMER_TYPE = T0.CUSTOMER_TYPE
 WHERE DD.CUSTOMER_TYPE = 'SW'
),T2 AS(
SELECT A.VBELV
      ,A.ERDAT
	  ,A.POSNV
  FROM SDI.SDI_VBFA_1001 A
 WHERE A.DEL_FLAG = 'N'
   AND A.VBTYP_N = 'M'
   AND A.MANDT = '800'
   AND A.VBELN NOT IN(SELECT VBELV
                        FROM SDI.SDI_VBFA_1001
					   WHERE VBTYP_N = 'N'
					     AND MANDT = '800'
					     AND DEL_FLAG = 'N'
					 )
)
INSERT OVERWRITE TABLE DM.DM_B1_PUR_INTOWH_T
SELECT  A.DELIVERY_DATE		-- 提货日期
       ,A.DELIVERY_BILL_NUM	-- 提货单据编号
       ,A.RECEIPT_DATE			-- 收货日期
       ,A.RECEIPT_BILL_NUM		-- 收货单据编号
       ,A.SUPPLIER_CODE		-- 供应商编码
       ,A.SUPPLIER_NAME		-- 供应商名称
       ,A.MATERIAL_CODE		-- 物料编码
       ,A.MATERIAL_DESC		-- 物料描述
       ,A.QUANTITY				-- 数量
       ,A.PRICE				-- 单价
       ,A.LINE_ACCOUNT			-- 行合计
       ,A.WAREHOUSE_NUM		-- 仓库编号
       ,A.BATCH_NUM			-- 批次
       ,A.LINE_NUM             --行号
       ,CASE WHEN A.DELIVERY_STATUS = 'C' THEN '交货'
 	         WHEN A.DELIVERY_STATUS = 'O' THEN '未清'
			  END AS DELIVERY_STATUS -- 交货状态 C：交货，O：未清
       ,B.DELIVERY AS SAP_DELIVERY_NUM		-- SAP提货单号
	   ,B.DELIVERY_PROJECT AS SAP_DELIVERY_LINE_NUM --SAP提货单行号
	   ,B.MATERIAL_NUM     --SAP单对应物料编码
	   ,B.MATERIAL_NAME    --SAP单对应物料名称
	   ,B.ACTUAL_DELIVERY_QTY --发货数量
	   ,D.LGOBE AS INV_LOCATION_NAME --发货仓库
	   ,from_unixtime(unix_timestamp(F.ERDAT,'yyyyMMdd'),'yyyy-MM-dd') AS SAP_DELIVERY_DATE
	   --,C.BILL_DATE AS SAP_DELIVERY_DATE --发货时间
	   ,C.CUST_DEL_NUM AS CUST_CODE --客人编号
	   ,C.CUST_DEL_NAME AS CUST_NAME --客人名称
	   ,E.RECEIVING_ADDR --收货地址
	   ,A.CUSTOMER_CODE --客编
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	   ,A.outstand_qty --剩余未清数量
  FROM DWROTD.DWR_WM_LADINGBILL_ITEM_F B
 INNER JOIN (
      select DWLH.CUST_DEL_NUM, --客人编号
		   t9.CUST_NAME AS CUST_DEL_NAME, --客人名称
		   DWLH.DELIVERY,
		   DWLH.DELIVERY_TYPE
	from DWIOTD.DWI_WM_LADINGBILL_HEADER_T DWLH
	INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T T9	--关联客户主数据表,取CUST_NAME
		 ON T9.CUST_ID = DWLH.CUST_DEL_ID		--客户ID
		 AND T9.GROUP_NUM = DWLH.GROUP_NUM		--集团
		 AND T9.END_DATE ='4712-12-31'
		 AND DWLH.DEL_FLAG = 'N'
   ) C
    ON B.DELIVERY = C.DELIVERY
  LEFT JOIN T1 A
    ON A.SAP_DELIVERY_NUM = B.DELIVERY
   AND A.SAP_DELIVERY_LINE_NUM = CAST(B.DELIVERY_PROJECT AS INT)
   AND B.GROUP_NUM = '800'
  LEFT JOIN SDI.SDI_T001L_1001 D
    ON B.INV_LOCATION = D.LGORT
   AND B.FACTORY_CODE = D.WERKS
   AND D.MANDT = '800'
  LEFT JOIN (
            SELECT *
			FROM(
				SELECT E.EXTERNNO,
					   E.EXTERNLINENO,
					   E.RECEIVING_ADDR,
					   ROW_NUMBER() OVER(PARTITION BY EXTERNNO,EXTERNLINENO) AS RN
				FROM DM.DM_LTC_ORDER_PLAN_SIGN_T E
				) T
				WHERE T.RN = 1
     ) E
    ON B.DELIVERY = E.EXTERNNO
   AND B.DELIVERY_PROJECT = E.EXTERNLINENO
  LEFT JOIN T2 F
    ON F.VBELV = B.DELIVERY
   AND F.POSNV = B.DELIVERY_PROJECT
 WHERE CASE WHEN C.DELIVERY_TYPE = 'ZLR1' THEN 1
            ELSE 0
			 END = 0
