WITH T AS(
SELECT A.SALE_NO --销售订单号
      ,A.SALE_LINE_NO --销售订单行号
      ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.ORDER_TYPE ELSE A.ORDER_TYPE END AS ORDER_TYPE --生产订单类型
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.ORDER_NUM ELSE A.ORDER_NUM END AS ORDER_NUM --生产订单号
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.QTY ELSE A.QTY END AS QTY --生产订单数量
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.ORDER_ST_NAME ELSE A.ORDER_ST_NAME END AS ORDER_ST_NAME --生产订单状态
      ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.SELF_FRGN_FLAG_NAME ELSE A.SELF_FRGN_FLAG_NAME END AS SELF_FRGN_FLAG_NAME --自制/外协
      ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.ID ELSE A.ID END AS ID  
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.ORDER_DATE ELSE A.ORDER_DATE END AS ORDER_DATE
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.MATERIAL_NUM ELSE A.MATERIAL_NUM END AS MATERIAL_NUM
	  ,CASE WHEN A.RELEVANT_CODE IS NOT NULL THEN B.CLOSE_QTY ELSE A.CLOSE_QTY END AS CLOSE_QTY
  FROM DWROTD.DWR_PS_APS_PRO_ORDER_F A
  LEFT JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F B
    ON A.RELEVANT_CODE = B.ORDER_NUM
)
INSERT OVERWRITE TABLE DM.DM_PS_PRCS_PRCH_PLAN_T
SELECT A.SALE_NO AS SALE_NUM --销售订单号
      ,A.SALE_LINE_NO AS SALE_LINE_NUM --销售订单行号
      ,A.ORDER_TYPE AS ORDER_TYPE_NAME --生产订单类型
	  ,A.ORDER_NUM --生产订单号
	  ,A.QTY --生产订单数量
	  ,A.ORDER_ST_NAME --生产订单状态
	  ,C.PLAN_NO AS PLAN_NO --报工号
	  ,D.PROD_QTY --报工号数量
	  ,C.PLAN_ST_NAME --报工单状态
	  ,C.START_TIME --计划开始时间
	  ,C.END_TIME --计划结束时间
	  ,C.RLSE_START_TIME --计划下达时间
	  ,E.APLY_NUM --OEM申请号
	  ,E.PRCH_ST_NAME --OEM状态
	  ,E.REPLY_DELV_DATE --OEM回复交期
	  ,E.OA_RQST_DATE --OA申请日期
	  ,E.SPLR_NAME --供应商名称
	  ,D.PROD_QTY AS PRCH_IN_QTY --采购入库数量汇总
	  ,D.LAST_IN_DATE AS LAST_PRCH_IN_DATE --采购最晚入库时间
	  ,F.RECORD_CREATE_DATE AS PRCH_CREATE_DATE --采购订单创建时间
	  ,D.PROD_QTY AS PROD_IN_QTY --生产报工数量汇总
	  ,D.LAST_IN_DATE AS LAST_PROD_IN_DATE --生产最晚入库日期
	  ,E.PROD_LINE_NAME AS RSRS_NAME --资源名称
	  ,E.BASE_NAME --产区
	  ,A.SELF_FRGN_FLAG_NAME --自制/外协
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(C.RLSE_START_TIME) - UNIX_TIMESTAMP(A.ORDER_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(C.RLSE_START_TIME) - UNIX_TIMESTAMP(A.ORDER_DATE)),86400)) / 3600,0) || '小时','.0','') AS PLAN_CNSUM_TIME --排程耗时 计划下达时间（一对多取最早下达时间）-订单评审通过
	  ,G.START_DATE --开始复板时间
	  ,G.ONLINE_STATE --上线情况
	  ,G.CHECKIN_DATE --生产签板时间
	  ,G.DUR_DAY --复板天数
	  ,G.START_PRO_DATE --上线开始生产时间
	  ,G.END_PRO_DATE --上线结束生产时间
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(D.LAST_IN_DATE) - UNIX_TIMESTAMP(G.START_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(D.LAST_IN_DATE) - UNIX_TIMESTAMP(G.START_DATE)),86400)) / 3600,0) || '小时','.0','') AS PROD_CNSUM_TIME --生产耗时 最晚入库时间-最早复板开始时间
	  ,REPLACE(TRUNC((UNIX_TIMESTAMP(D.LAST_IN_DATE) - UNIX_TIMESTAMP(G.START_DATE)) / 86400,0) || '天'
	  || TRUNC((MOD((UNIX_TIMESTAMP(D.LAST_IN_DATE) - UNIX_TIMESTAMP(G.START_DATE)),86400)) / 3600,0) || '小时','.0','') AS PRCH_CNSUM_TIME --外协耗时 最晚入库时间-最早复板开始时间
      ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,E.PRCH_NUM AS PRCH_ORDER_NUM --采购订单号
	  ,CASE WHEN H.PLAN_NUM IS NOT NULL THEN '自营工程'
	        WHEN I.ORDER_NUM IS NOT NULL THEN '零售订单'
			 END AS ORDER_CAT_NAME --订单分类
	  ,A.MATERIAL_NUM --物料编码
	  ,A.CLOSE_QTY --关闭数量
	  ,C.CLOSE_TIME --关闭时间
  FROM T AS A
  LEFT JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F B
    ON B.PRODUCT_ORDER_ID = A.ID
  LEFT JOIN DWROTD.DWR_PS_APS_PLAN_TASK_F C
    ON B.ID = C.PRCS_TASK_ID
  LEFT JOIN(SELECT PLAN_TASK_ID
                  ,LAST_IN_DATE
                  ,SUM(COALESCE(PROD_QTY,0)) AS PROD_QTY
              FROM DWROTD.DWR_PS_PROD_SCHED_F
			 GROUP BY PLAN_TASK_ID
			      ,LAST_IN_DATE
		   ) D
    ON C.ID = D.PLAN_TASK_ID
  LEFT JOIN DWROTD.DWR_PS_PRCH_ORDER_F E
    ON C.PLAN_TASK_DTL_ID = E.PLAN_TASK_DTL_ID
  LEFT JOIN DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F F
    ON E.PRCH_NUM = F.PURCHASE_VOUCHER_NUM
  LEFT JOIN DWROTD.DWR_PAM_SAMPLE_CONFIRM_T G
    ON C.PLAN_NO = G.PLAN_NO
  LEFT JOIN (SELECT DISTINCT PLAN_NUM FROM DWRLTC.DWR_BO_PLAN_ORDER_F) H
    ON A.SALE_NO = H.PLAN_NUM
  LEFT JOIN (SELECT DISTINCT ORDER_NUM FROM DWROTD.DWR_OTD_SO_SALE_OFR_ORDER_F) I
    ON A.SALE_NO = I.ORDER_NUM
 WHERE CASE WHEN H.PLAN_NUM IS NOT NULL THEN '自营工程'
	        WHEN I.ORDER_NUM IS NOT NULL THEN '零售订单'
			 END IS NOT NULL
