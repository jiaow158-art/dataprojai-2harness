insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_MAT_TMP 
select t1.material_num,
		t1.product_level2_name  as cat,
		t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        t1.width * t1.length_1/1000/1000 as area,
        t1.external_matl_group_code,
        t2.matl_group2_code,
        t2.matl_group3_code,
		current_timestamp() as dw_last_update_date,
		t1.cstm_main_code   -- 新增字段定制主体
  from dwimd.dwi_md_data_material_general_t t1 
  join (select material_num,
               max(matl_group2_code) as matl_group2_code,
                max(matl_group2_code) as matl_group3_code
        from dwimd.dwi_md_data_material_sales_t
        group by material_num) t2 ON t1.material_num  = t2.material_num
  where t1.matl_group_code ='1000010'
	and t1.external_matl_group_code in('10010','10080')
    and t1.product_level2_code  like '1%'
    and t1.product_level2_code  not in('1011','1012','1013','1014','1016')
	and matl_type_code='FERT';
