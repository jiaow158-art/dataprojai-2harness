with DELIVER_DESPATCH_DATA AS(
--提货单 发运委托
	SELECT DISTINCT
           DDHS.business_id,
		   DDHS.contract_id,
		   DDHS.logistics_type,
		   DDL.delivery_no,
		   DDHS.despatch_id,
		   b2.MEANING as logistics_type_name --运输方式描述
	FROM SDI.SDI_DELIVER_DESPATCH_HEADER_1033 DDHS
	LEFT JOIN SDI.SDI_DELIVER_DESPATCH_LINE_1033 DDL
	ON DDHS.despatch_id = DDL.despatch_id
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b2
          on DDHS.logistics_type = b2.lookup_code
         and b2.lookup_type = 'DELIVERY_METHOD_C'
         and b2.enabled_flag = 'Y'
	WHERE DDHS.delivery_no IS NOT NULL
),
SETTLEMENT_REQUEST_PAYOUT_DATA AS(
--请款单
	SELECT SRP.business_id,
		   SRP.contract_code,
		   SRP.request_payout_code,
		   SRP.REQUEST_PAYOUT_ID,
		   sum(SRP.total_request_amount) as total_request_amount -- 请款金额
	FROM SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_1033 SRP
	group by SRP.business_id,
			 SRP.contract_code,
			 SRP.request_payout_code,
			 SRP.REQUEST_PAYOUT_ID
),
SETTLEMENT_REQUEST_PAYOUT_DETAIL_DATA AS(
--请款单明细
	SELECT REQUEST_PAYOUT_ID,
	        pick_code,
	        max(SRPD.pick_date) as pick_date
	FROM SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_DETAIL_1033 SRPD
	group by REQUEST_PAYOUT_ID,
			 pick_code
),
PROJECT_REVIEW_MATERIAL_PRICE_DATA AS(
--项目评审价格
	SELECT DISTINCT PR.BUSINESS_ID,
		   PR.project_code , -- 项目评审号
		   PR.estimated_complete_date,   -- 预计竣工日期
		   NULL AS customer_material -- 客户产品编码
	FROM SDI.SDI_PROJECT_REVIEW_1033 PR
	WHERE PR.contract_flag = 'Y'
	AND PR.delete_flag = 0
),
SETTLEMENT_VIRTUAL_CHECK_DETAIL_data as(
--验收单明细
	select svcd.VIRTUAL_CHECK_ID,
		   svcd.material_code,
		   svcd.pick_code, --提货单
		   product_check_status, --货物验收状态
		   case when product_check_status = 'WAIT_OVERDUE' then '待办逾期'
		        when product_check_status = 'WAIT_NOT_OVERDUE' then '待办未逾期'
				when product_check_status = 'COMPLETE_OVERDUE' then '已办逾期'
				when product_check_status = 'COMPLETE_NOT_OVERDUE' then '已办未逾期'
				 end as product_check_status_name, --货物验收状态描述
		   sum(SVCD.total_virtual_qty) as  total_virtual_qty,
		   sum(SVCD.total_formal_qty) as total_formal_qty,
		   sum(SVCD.total_virtual_amount) as total_virtual_amount,
		   sum(SVCD.total_formal_amount) as total_formal_amount
	from SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_DETAIL_1033 SVCD
	group by svcd.VIRTUAL_CHECK_ID,
			 svcd.material_code,
			 svcd.pick_code, --提货单
			 product_check_status, --货物验收状态
			 case when product_check_status = 'WAIT_OVERDUE' then '待办逾期'
		        when product_check_status = 'WAIT_NOT_OVERDUE' then '待办未逾期'
				when product_check_status = 'COMPLETE_OVERDUE' then '已办逾期'
				when product_check_status = 'COMPLETE_NOT_OVERDUE' then '已办未逾期'
				 end
),
SETTLEMENT_FORMAL_CHECK_DATA AS(
--正式验收单
		select SFC.formal_check_code,
			   SFC.affirm_date , -- 验收日期
			   SFC.collect_date ,-- 验收单回收时间
			   SFC.is_collect_source_file , -- 验收单是否回收
			   SFC.CONTRACT_CODE,
			   SFC.formal_check_id,
			   sfc.business_id,
			   sfc.request_status, --请款状态
			   b2.MEANING as request_status_name, --请款状态描述
			   sfc.affirm_Result, --确认结果编码
			   b1.MEANING as affirm_Result_name --确认结果描述
		from SDI.SDI_SETTLEMENT_FORMAL_CHECK_1033 sfc
		left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b1
          on sfc.affirm_Result = b1.lookup_code
         and b1.lookup_type = 'SETTLE_FORMAL_CHECK_AFFIRM'
         and b1.enabled_flag = 'Y'
		left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b2
          on sfc.request_status = b2.lookup_code
         and b2.lookup_type = 'SETTLE_REQUEST_PAYOUT_STATUS'
         and b2.enabled_flag = 'Y'
		where sfc.delete_flag <> 1
		  and sfc.affirm_result = 'GRANT_ACCEPTANCE' --审批通过
),
SETTLEMENT_FORMAL_CHECK_DETAIL_DATA AS(
--验收单列表
	select formal_check_id,
		   --material_code,
		   sum(current_formal_amount) current_formal_amount
	from SDI.SDI_SETTLEMENT_FORMAL_CHECK_DETAIL_1033 SEFCD
	group by formal_check_id
	         --material_code
),
SDI_SETTLEMENT_OPEN_TICKET_DATA AS(
--开票单
	select 
	      sot.business_id,
		  sot.contract_code,
		  sot.wbs,
		  sot.open_ticket_code,
		  sot.creation_date,
		  sot.pro_ticket_flag,
		  sot.verification_status, --核销状态
		  b.MEANING as verification_status_name, --核销状态描述
		  sum(SOT.current_ticket_amount) as current_ticket_amount,
		  sum(SOT.urplus_ticket_amount) AS urplus_ticket_amount,
		  SUM(SOT.un_check_amount) AS un_check_amount
	from SDI.SDI_SETTLEMENT_OPEN_TICKET_1033 sot
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b
      on sot.verification_status = b.lookup_code
     and b.lookup_type = 'SETTLE_VERIFICATIONSTATUS'
     and b.enabled_flag = 'Y'
	WHERE sot.delete_flag = 0
	group by sot.business_id,
			 sot.contract_code,
			 sot.wbs,
			 sot.open_ticket_code,
		     sot.creation_date,
		     sot.pro_ticket_flag,
		     sot.verification_status,
			 b.MEANING
),
tmp_data as(
SELECT
	  substring(cast(BO.creation_date as string),1,10) as business_create_date , -- 商机创建日期
	  BO.apply_user_area_code ,  -- 销售区域编码
	  BO.apply_user_area as apply_user_area_name, -- 销售区域名称
	  BO.apply_user_dept_code , -- 销售部门编码
	  BO.apply_user_dept as apply_user_dept_name, -- 销售部门名称
	  BO.strategic_customer_id ,  -- 大客户ID
	  BO.strategic_customer_name,  -- 大客户名称
	  BO.apply_user_id ,  -- 业务员ID
	  BO.apply_user as apply_user_name , -- 业务员名称
	  BO.business_id ,   -- 商机ID
	  BO.business_code ,  -- 商机编号
	  BO.business_name,   -- 商机名称（项目名称）
	  BO.customer_code , -- 客户编码
	  BO.customer_name,  -- 客户名称
	  BO.wbs,  -- WBS号
	  DDH.demand_code , -- 要货单号
	  DDH.demand_status,  --要货单状态
	  DDH.recove_flag,    -- 要货单是否回收 
	  C.contract_code ,   -- 合同编号
	  C.contract_name,    -- 合同名称
	  C.new_flag,      -- 合同版本
	  C.is_credit,     -- 是否授信项目
	  CSH.acceptance_recovery_time2 , -- 验收单回收时间（天）
	  CSH.acceptance_request_time2 , -- 验收到请款时间（天）
	  CSH.request_invoice_time2 , --请款到开票时间（天）
	  CSH.invoice_returned_time2,  -- 开票到回款时间
	  C.signed_date , -- 合同签收日期
	  DDD.delivery_no , -- 提货单号
	  SRPD.pick_date , -- 发货时间
	  DDD.logistics_type , -- 运输方式
	  DSH.sign_status,  -- 签收状态
	  CASE WHEN SZWE.eert = 'X' THEN '是' ELSE '否' END  as sign_no_status, -- 签收单是否回收
	  SVC.virtual_check_code,  -- 待办验收单号
	  PR.project_code , -- 项目评审号
	  PR.estimated_complete_date,   -- 预计竣工日期
	  PR.customer_material, -- 客户产品编码
	  SVCD.material_code, -- 物料编码
	  NVL(SVCD.total_virtual_qty - SVCD.total_formal_qty,0) AS db_accept_qty, -- 待办验收数量 --验收数量
	  NVL(SVCD.total_virtual_amount  - SVCD.total_formal_amount,0) AS db_accept_amt,   -- 待办验收金额 -- 验收金额
	  SFC.formal_check_code , -- 正式验收单号
	  SEFCD.current_formal_amount, -- 正式验收单已办验收金额
	  SFC.affirm_date , -- 验收日期
	  SFC.collect_date ,-- 验收单回收时间
	  SFC.is_collect_source_file , -- 验收单是否回收
	  SB.register_status , -- 结算状态
	  SB.back_date , -- 回单日期
	  SB.is_source_file,  -- 结算单是否回收 回单是否签收
	  SB.settlement_bill_code, -- 结算单号
	  SRP.request_payout_code , -- 请款单号
	  nvl(SRP.total_request_amount ,0) as total_request_amount, -- 请款金额
	  SOT.open_ticket_code,   -- 开票单号
	  SOT.creation_date , -- 开票申请日期
	  NVL(SOT.current_ticket_amount,0) AS  current_ticket_amount, -- 开票金额合计
	  NVL(SOT.urplus_ticket_amount,0) AS urplus_ticket_amount , -- 未核销金额
	  NVL(SOT.current_ticket_amount,0) - NVL(SOT.urplus_ticket_amount,0) as write_off_amt, -- 已核销金额
	  SOT.pro_ticket_flag, -- 是否预开票
	  NVL(SOT.un_check_amount,0) AS un_check_amount, -- 未勾稽金额
	  current_timestamp() as dw_last_update_date ,
	  c.receipts_status, -- 合同单据状态
	  DDH.creation_date  AS deliver_creation_date, -- 要货单创建日期
	  b1.meaning as receipts_status_name, -- 合同单据状态描述	  
	  C.orders_push_method, --要货单推送方式
	  b.meaning as orders_push_method_name, --要货单推送方式描述
	  sfc.request_status, --请款状态
	  sfc.request_status_name, --请款状态描述
	  sot.verification_status, --开票单：核销状态
	  sot.verification_status_name, --开票单：核销状态描述
	  product_check_status, --货物验收状态
	  product_check_status_name, --货物验收状态描述
	  B2.meaning as REGISTER_STATUS_name, -- 结算状态描述
	  DDH1.EARLT_CREATION_DATE, --最早要货单成功提交时间
	  DDD.logistics_type_name, --运输方式描述
	  SVC.VIRTUAL_CHECK_STATUS, --虚拟验收单状态编码
	  B3.meaning AS VIRTUAL_CHECK_STATUS_NAME, --虚拟验收单状态描述
	  sfc.affirm_Result as affirm_Result_code, --正式验收单确认结果编码	  
	  sfc.affirm_Result_name, --正式验收单确认结果描述
	  DDH.plan_type as plan_type_code, --要货单计划类型编码
	  b4.meaning as plan_type_NAME, --要货单计划类型描述
	  substring(cast(BO.creation_date as string),1,7) as period_id_m
	FROM SDI.SDI_BUSINESS_OPPORTUNITY_1033 BO
	LEFT JOIN SDI.SDI_DELIVER_DEMAND_HEADER_1033 DDH --要货单
	ON BO.BUSINESS_ID = DDH.BUSINESS_ID
	LEFT JOIN SDI.SDI_CONTRACT_1033 C
	ON BO.BUSINESS_ID = C.BUSINESS_ID
	LEFT JOIN SDI.SDI_CONTRACT_SETTLEMENT_HEAD_1033 CSH --验收单
	ON C.CONTRACT_ID = CSH.CONTRACT_ID
	LEFT JOIN DELIVER_DESPATCH_DATA DDD --发运委托
	ON BO.BUSINESS_ID = DDD.BUSINESS_ID
	AND C.CONTRACT_ID = DDD.CONTRACT_ID
	LEFT JOIN SETTLEMENT_REQUEST_PAYOUT_DATA SRP --请款单
	ON BO.BUSINESS_ID = SRP.business_id
	AND C.contract_code = SRP.contract_code
	LEFT JOIN SETTLEMENT_REQUEST_PAYOUT_DETAIL_DATA SRPD --请款单明细
	ON SRPD.PICK_CODE = DDD.DELIVERY_NO 
	AND SRP.REQUEST_PAYOUT_ID = SRPD.REQUEST_PAYOUT_ID
	LEFT JOIN SDI.SDI_DELIVER_SIGN_HEADER_1033 DSH --签收单
	ON BO.BUSINESS_ID = DSH.business_id
	AND DDD.despatch_id = DSH.despatch_id 
	and DDD.delivery_no =  DSH.delivery_no
	and BO.wbs = DSH.wbs
	and dsh.sign_status = 'ALREADY_SIGNED'
	LEFT JOIN PROJECT_REVIEW_MATERIAL_PRICE_DATA PR --项目评审价格
	ON BO.BUSINESS_ID = PR.BUSINESS_ID 
	LEFT JOIN SETTLEMENT_VIRTUAL_CHECK_DETAIL_DATA SVCD --验收单明细
	ON DDD.DELIVERY_NO = SVCD.PICK_CODE 
	--and pr.customer_material = svcd.material_code --？
	LEFT JOIN SDI.SDI_SETTLEMENT_VIRTUAL_CHECK_1033 SVC --虚拟验收单
	ON SVC.VIRTUAL_CHECK_ID = SVCD.VIRTUAL_CHECK_ID 
	LEFT JOIN SETTLEMENT_FORMAL_CHECK_DATA SFC --验收单
	ON BO.business_id = SFC.business_id
	AND SFC.CONTRACT_CODE = C.CONTRACT_CODE
	LEFT JOIN SETTLEMENT_FORMAL_CHECK_DETAIL_DATA SEFCD --验收单列表
	ON SFC.formal_check_id = SEFCD.formal_check_id
	--AND svcd.material_code = SEFCD.material_code --？
	LEFT JOIN SDI.SDI_SETTLEMENT_BILL_1033 SB --结算单
	ON BO.business_code = SB.business_code
	LEFT JOIN SDI.SDI_ZTSD_WBS_EERT_1001 SZWE
	ON BO.WBS = SZWE.posid
	LEFT JOIN SDI_SETTLEMENT_OPEN_TICKET_DATA SOT --开票单
	ON BO.business_id = SOT.business_id
	AND BO.wbs = SOT.wbs
	AND  C.CONTRACT_CODE = SOT.CONTRACT_CODE
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b
    on C.orders_push_method = b.lookup_code
    and b.lookup_type = 'ORDER_PUSH_METHOD'
    and b.enabled_flag = 'Y'
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b1
    on C.receipts_status = b1.lookup_code
    and b1.lookup_type = 'RECIPS_CODE'
    and b1.enabled_flag = 'Y'
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b2
    on SB.REGISTER_STATUS = b2.lookup_code
    and b2.lookup_type = 'REGISTER_STATUS'
    and b2.enabled_flag = 'Y'
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b3
    on SVC.VIRTUAL_CHECK_STATUS = b3.lookup_code
    and b3.lookup_type = 'SETTLE_VIRTUAL_CHECK_STATUS'
    and b3.enabled_flag = 'Y'
	left join SDI.SDI_BASE_LOOKUP_VALUES_1029 b4
    on DDH.plan_type = b4.lookup_code
    and b4.lookup_type = 'PLAN_TYPE'
    and b4.enabled_flag = 'Y'
	LEFT JOIN (SELECT BUSINESS_ID
	                 ,MIN(CREATION_DATE) AS EARLT_CREATION_DATE --最早要货单成功提交时间
	             FROM SDI.SDI_DELIVER_DEMAND_HEADER_1033
                WHERE demand_status IN('APPROVAL','SUBMIT','TO_SUBMIT')
				GROUP BY BUSINESS_ID) DDH1 --要货单
	ON BO.BUSINESS_ID = DDH1.BUSINESS_ID
	where 1 = 1 
	--and C.new_flag = 'NEW'
)
