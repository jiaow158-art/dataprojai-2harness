WITH T1 AS(
SELECT T0.BUSINESS_CODE /*商机编号，用于项目评审、合同登记变更的主表链接*/
      ,T0.BUSINESS_ID /*商机ID号，用于商机其他子表的链接*/
      ,T0.BUSINESS_NAME /*商机名称*/
      ,T0.DIVISION /*事业部，CERAMIC_TILE_DIVISION是瓷砖事业部，BATHROOM_DIVISION是卫浴事业部，NEW_ECOLOGICAL_TIMBER_DIVISION是新材事业部*/
      ,T0.CUSTOMER_CODE /*客户编码*/
      ,T0.CUSTOMER_NAME /*客户名称*/
      ,T0.APPLY_USER_ID /*申请人ID*/
      ,T0.APPLY_USER /*申请人姓名*/
      ,T0.APPLY_USER_AREA_CODE /*申请人所属区域编号*/
      ,T0.APPLY_USER_AREA /*申请人所属区域*/
      ,T0.APPLY_USER_DEPT_CODE /*申请人部门编码*/
      ,T0.APPLY_USER_DEPT /*申请人部门*/
      ,T0.BUSINESS_STAGE /*商机阶段*/
      ,T0.S0APPROVE_STATE/*S0商机审批状态编码，有效商机是此字段为S0APPROVE_STATE=ALLOW+BUSINESS_STAGE<>SX*/
      ,T1.MEANING AS S0APPROVE_STATE_NAME /*S0商机审批状态描述*/
  FROM SDI.SDI_BUSINESS_OPPORTUNITY_1033 T0 /*商机主表*/
  LEFT JOIN(SELECT LOOKUP_CODE,MEANING 
              FROM SDI.SDI_BASE_LOOKUP_VALUES_1027
			 WHERE LOOKUP_TYPE = 'BUSINESS_APPROVE_STATE'
           ) T1 /*获取BASE数据库表的S0商机审批状态对应代码和中文*/
    ON T0.S0APPROVE_STATE = T1.LOOKUP_CODE
),T2 AS(
SELECT 
       PR.PROJECT_CODE /*项目单据号*/
      ,PR.PROJECT_ID /*项目单ID，用于商机其他子表的链接*/
      ,PR.PROJECT_CLASS /*商机类型编码*/
	  ,F.MEANING AS PROJECT_CLASS_NAME /*商机类型描述*/
      ,PR.PROJECT_TYPE /*工程类型*/
      ,PR.PARENT_BUSINESS_CODE /*母商机编号*/
      ,PR.BUSINESS_CODE /*子商机编号*/
      ,PR.DELIVERY /*事业部，CERAMIC_TILE_DIVISION是瓷砖事业部，BATHROOM_DIVISION是卫浴事业部，NEW_ECOLOGICAL_TIMBER_DIVISION是新材事业部*/
      ,PR.WBS /*WBS号*/
      ,PR.ENGINEERING_BIG_CATEGORY /*工程大类*/
      ,T0.MEANING AS ENGINEERING_BIG_CATEGORY_NAME /*工程大类名称*/
      ,PR.ENGINEERING_SUB_CATEGORY /*工程小类*/
      ,T1.MEANING AS ENGINEERING_SUB_CATEGORY_NAME /*工程小类名称*/
      ,PR.IS_OTHER_DIVISION /*是否跨事业部*/
	  ,CASE WHEN PR.IS_OTHER_DIVISION = 'N' THEN '否'
            WHEN PR.IS_OTHER_DIVISION = 'Y' THEN '是'
			 END AS IS_OTHER_DIVISION_NAME /*是否跨事业部描述*/
      ,PR.IS_CREDIT /*是否授信项目*/
	  ,CASE WHEN PR.IS_CREDIT = 'N' THEN '否'
            WHEN PR.IS_CREDIT = 'Y' THEN '是'
			 END AS IS_CREDIT_NAME /*是否授信项目描述*/
      ,PR.CUSTOMER_CODE /*工程客编*/
      ,PR.CUSTOMER_NAME /*客户名称*/
  FROM SDI.SDI_PROJECT_REVIEW_1033 PR /*项目评审主表*/
  LEFT JOIN(SELECT LOOKUP_CODE,MEANING
              FROM SDI.SDI_BASE_LOOKUP_VALUES_1027
			 WHERE LOOKUP_TYPE = 'ENGINEERING_BIG_CATEGORY'
           ) T0 /*获取BASE数据库表的商机大类对应代码和中文*/
    ON PR.ENGINEERING_BIG_CATEGORY = T0.LOOKUP_CODE
  LEFT JOIN(SELECT LOOKUP_CODE,MEANING 
              FROM SDI.SDI_BASE_LOOKUP_VALUES_1027
			 WHERE LOOKUP_TYPE = 'ENGINEERING_SMALL_CATEGORY'
           ) T1 /*获取BASE数据库表的商机小类对应代码和中文*/
    ON PR.ENGINEERING_SUB_CATEGORY = T1.LOOKUP_CODE
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 F
    ON PR.PROJECT_CLASS = F.LOOKUP_CODE
   AND F.LOOKUP_TYPE = 'BUSINESS_CATEGORY'	
),T3 AS(
SELECT T0.LOGISTICS_ID /*物流页签ID*/
      ,T0.PROJECT_ID /*项目评审单据ID*/
      ,T0.DISCHARGE_CARGO /*卸货方编码*/
	  ,F.MEANING AS DISCHARGE_CARGO_NAME /*卸货方描述*/
      ,T0.FREIGHT_PAYERS /*物流支付方编码*/
	  ,H.MEANING AS FREIGHT_PAYERS_NAME /*物流支付方描述*/
      ,T0.DELIVERY /*配送方式编码*/
	  ,J.MEANING AS DELIVERY_NAME /*配送方式描述*/
      ,T0.LOGISTICS_FEE_QUOTATION_METHOD /*物流费报价方式编码*/
	  ,G.MEANING AS LOGISTICS_FEE_QUOTATION_METHOD_NAME /*物流费报价方式描述*/
  FROM SDI.SDI_PROJECT_REVIEW_LOGISTICS_1033 T0 /*项目评审-物信息页签*/
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 F
    ON T0.DISCHARGE_CARGO = F.LOOKUP_CODE
   AND F.LOOKUP_TYPE = 'DISCHARGER'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 H
    ON T0.FREIGHT_PAYERS = H.LOOKUP_CODE
   AND H.LOOKUP_TYPE = 'SHIPPING_METHODS'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 J
    ON T0.DELIVERY = J.LOOKUP_CODE
   AND J.LOOKUP_TYPE = 'DELIVERY'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 G
    ON T0.LOGISTICS_FEE_QUOTATION_METHOD = G.LOOKUP_CODE
   AND G.LOOKUP_TYPE = 'LOGISTICS_FEE_METHOD'   
),T4 AS(
SELECT 
       PR.BUSINESS_CODE /*商机编号*/
	  ,PR.CONTRACT_ID /*合同单号ID*/
      ,PR.CONTRACT_CODE /*合同号*/
      ,PR.PROJECT_CODE /*项目号*/
      ,PR.BUSINESS_NAME /*商机名称*/
      ,PR.SEAL_DATE /*盖章日期*/
      ,PR.NEW_FLAG /*合同最新标识，NEW当前最新合同.OLD历史合同纪录*/
      ,PR.RECEIPTS_STATUS /*审批状态编码，PASS为审批通过*/
	  ,K.MEANING AS RECEIPTS_STATUS_NAME /*审批状态描述*/
      ,PR.DELIVERY AS DIVISION_CODE /*事业部编码，CERAMIC_TILE_DIVISION是瓷砖事业部，BATHROOM_DIVISION是卫浴事业部，NEW_ECOLOGICAL_TIMBER_DIVISION是新材事业部*/
      ,N.MEANING AS DIVISION_NAME /*事业部描述*/
	  ,PR.CONTRACT_STATUS /*合同状态编码*/
	  ,I.MEANING AS CONTRACT_STATUS_NAME /*合同状态描述*/
	  ,PR.CONTRACT_VERSION /*合同版本号*/
	  ,ROW_NUMBER() OVER(PARTITION BY PR.BUSINESS_CODE ORDER BY PR.CONTRACT_VERSION DESC) AS RN
	  --,ROW_NUMBER() OVER(PARTITION BY PR.BUSINESS_CODE ORDER BY PR.CONTRACT_VERSION ASC) AS RN1
	  ,PR.EFFECTIVE_DATE AS DFF_DATE /*生效日期*/
      ,PR.INVALID_DATE AS INVAL_DATE /*失效日期*/
      ,PR.DELIVERY_PERIOD AS DLVR_PERIOD /*预计供货周期，它是字符串，时间之间用逗号区别*/
  FROM SDI.SDI_CONTRACT_1033 PR /*合同主表*/
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 I
    ON PR.CONTRACT_STATUS = I.LOOKUP_CODE
   AND I.LOOKUP_TYPE = 'CONTRACT_SEAL_STATUS'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 K
    ON PR.RECEIPTS_STATUS = K.LOOKUP_CODE
   AND K.LOOKUP_TYPE = 'RECIPS_CODE'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 N
    ON PR.DELIVERY = N.LOOKUP_CODE
   AND N.LOOKUP_TYPE = 'DIVISION' 
),T5 AS(
SELECT 
       PRMP.CONTRACT_ID /*合同单号ID*/
      ,PRMP.PRODUCT_ATTRIBUTE /*产品来源编码*/
	  ,K.MEANING AS PRODUCT_ATTRIBUTE_NAME /*产品来源描述*/
      ,PRMP.MATERIAL /*物料编码*/
      ,PRMP.MATERIAL_DESCRIPTION /*物料描述*/
      ,PRMP.CONTRACT_PRICE /*合同单价*/
      ,PRMP.UNIT /*单位*/
      ,PRMP.WEIGHT_VOLUME /*重量体积*/
      ,PRMP.DEALER_PICK_PRICE /*经销商提货价*/
      ,PRMP.REVIEW_QUOTATION /*评审报价*/
      ,PRMP.PRODUCT_NAKED_PRICE /*产品单价*/
      ,PRMP.EP_PRICE /*EP价*/
      ,PRMP.DISCOUNT_RATE /*折扣率*/
      ,PRMP.REVIEW_DOSAGE /*评审用量*/
      ,PRMP.CONTRACT_QUANTITY /*合同用量*/
      ,PRMP.CENTRALIZED_SERVICE_RATE /*集采服务费率*/
      ,PRMP.ORDER_SERVICE_RATE /*抢单服务费率*/
      ,PRMP.GROUND_SERVICE_RATE /*落地服务费率*/
      ,PRMP.ADVANCE_SERVICE_RATE /*垫资服务费率*/
      ,PRMP.SERVICE_RATE /*综合费率*/
	  ,PRMP.MATERIAL_STATUS /*物料行标签编码*/
	  ,CASE WHEN PRMP.MATERIAL_STATUS = 'INITIAL' THEN '原始'
	        WHEN PRMP.MATERIAL_STATUS = 'SUPPLEMENT' THEN '增补'
	        ELSE N.MEANING
			 END AS MATERIAL_STATUS_NAME /*物料行标签描述*/
  FROM SDI.SDI_CONTRACT_MATERIAL_1033 PRMP /*合同的物料价格清单页签*/
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 K
    ON PRMP.PRODUCT_ATTRIBUTE = K.LOOKUP_CODE
   AND K.LOOKUP_TYPE = 'PRODUCT_TYPE'
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1027 N
    ON PRMP.MATERIAL_STATUS = N.LOOKUP_CODE
   AND N.LOOKUP_TYPE = 'INFO_CUSTONMER_MATERIAL_STATUS' 
),T6 AS( --工作流
SELECT  BPM.BILL_NO
       ,MIN(TASK.START_TIME_) AS START_TIME
       ,MAX(TASK.END_TIME_) AS END_TIME
  FROM SDI.SDI_ACT_HI_TASKINST_1031 TASK
 INNER JOIN SDI.SDI_ACT_BPM_TASK_LEAVEL_1031 LEAVEL 
    ON LEAVEL.TASK_ID = TASK.ID_
 INNER JOIN SDI.SDI_ACT_BPM_LIST_1031 BPM
    ON BPM.PROC_INST_ID = LEAVEL.TOP_PROC_INST_ID
 WHERE ( TASK.DELETE_REASON_ IS NULL OR TASK.DELETE_REASON_ = 'completed') 
   AND TASK.END_TIME_ IS NOT NULL
   AND BPM.RESULT = 'ALLOW' /*流程归档标识*/
   AND SUBSTR(BPM.BILL_NO,1,2) IN ('HT','XM') /*框定合同和项目评审单据范围*/
 GROUP BY BPM.BILL_NO  
)
