INSERT OVERWRITE TABLE DM.DM_STORE_DATA_DETAIL_F
select
T1.MONTH AS CALMONTH,
T2.SHENG AS PROVINCE ,
T2.SHI AS CITY,
T2.qu AS AREA,
T1.outStoreId,
SUM(T2.MDDPZSM) AS operating_area,
COUNT(DISTINCT employeeId) AS EMPLOYEE_QTY,
SUM(finishedAmount/10000) AS FINISHEDAMOUNT,
current_timestamp AS DW_LAST_UPDATE_DATE,
T1.PERIOD_ID_M AS PERIOD_ID_M
from 
SDI.SDI_STORE_DATA_1037 T1 
join DWIMD.DWI_MD_SHOP_INFO_OA_T T2 
on T1.outStoreId = T2.xjmdbh 
where T1.period_id_m  = '${PERIOD_ID_M}'
GROUP BY 
T1.MONTH,
T2.SHENG,
T2.SHI,
T2.qu,
T1.outStoreId,
T1.PERIOD_ID_M
