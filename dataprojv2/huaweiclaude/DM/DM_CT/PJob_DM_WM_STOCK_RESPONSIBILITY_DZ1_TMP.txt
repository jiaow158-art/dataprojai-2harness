insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_DZ1_TMP
select t1.material_num ,
	   t1.batch_num ,
	   t1.FACTORY_CODE,
	   t1.inv_location,
       case when t4.cstm_main_code IN('KM')  or t4.matl_group2_code in ('KM') then '工程市场中心'
            when t4.cstm_main_code IN('ZY')  or t4.matl_group2_code in ('ZY')  then '直营经营管理板块'
            when t4.cstm_main_code IN('JX')  or t4.matl_group2_code in ('JX') then '区域运营中心'
            when t4.cstm_main_code IN('GM')  or t4.matl_group2_code in ('GM') then '国际营销系统' end 
            as resp_system,
       sum(left_zkkc) as dz1_stock,
	   current_timestamp() as dw_last_update_date
from dm.dm_wm_stock_responsibility_left_0_tmp  t1
join dm.dm_wm_stock_responsibility_mat_tmp t4 
On t1.material_num  = t4.material_num 
where case when t1.inv_location IN('GCQA','BJ01','GCSA','GCCA','GCPA','GCTA','GCSD','BJ','BY','HNDY','JXDY',          'SXDY','SDDY','QYDY','CQDY') then '工程市场'
            when t1.inv_location IN('0020','0018','0026','0030','0018','GFZ','0028','0019','BC','HA08','EA03','LA01','EA1','VA01','0034','0015','A005','MA01','FB73','FB62','GFZ') then '直营'
            when t1.factory_code like '%10' then '国际营销系统'
            when t4.external_matl_group_code = '10080' then '梦之家' end is  null
  and t1.material_num like '%_A'
  and t4.cstm_main_code  IN('KM','ZY','JX','GM') or  t4.matl_group2_code  IN('KM','ZY','JX','GM') 
group by t1.material_num ,
         t1.batch_num ,
		 t1.FACTORY_CODE,
		 t1.inv_location,
       case when t4.cstm_main_code IN('KM')  or t4.matl_group2_code in ('KM') then '工程市场中心'
            when t4.cstm_main_code IN('ZY')  or t4.matl_group2_code in ('ZY')  then '直营经营管理板块'
            when t4.cstm_main_code IN('JX')  or t4.matl_group2_code in ('JX') then '区域运营中心'
            when t4.cstm_main_code IN('GM')  or t4.matl_group2_code in ('GM') then '国际营销系统' end ;
