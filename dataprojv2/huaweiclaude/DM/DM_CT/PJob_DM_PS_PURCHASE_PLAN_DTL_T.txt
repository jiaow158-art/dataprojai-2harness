WITH 
QT AS(
SELECT A.PURCHASE_VOUCHER_NUM
      ,A.PURCHASE_VOUCHER_ITEM_NUM
      ,SUM(CASE WHEN A.MOVE_TYPE = '101' THEN A.QUANTITY
                WHEN A.MOVE_TYPE = '102' THEN -1 * A.QUANTITY
                ELSE 0
                 END) AS QUANTITY --已交付数量
	  ,SUM(CASE WHEN A.MOVE_TYPE = '101' AND SUBSTR(OEM.REPLY_DELIVERY,1,10) >= SUBSTR(A.VOUCHER_POST_DATE,1,10) THEN A.QUANTITY
                WHEN A.MOVE_TYPE = '102' AND SUBSTR(OEM.REPLY_DELIVERY,1,10) >= SUBSTR(A.VOUCHER_POST_DATE,1,10) THEN -1 * A.QUANTITY
                ELSE 0
                 END) AS NOOVERTIME_QUANTITY --未逾期已交付数量
	  ,SUM(CASE WHEN A.MOVE_TYPE = '101' AND SUBSTR(OEM.REPLY_DELIVERY,1,10) < SUBSTR(A.VOUCHER_POST_DATE,1,10) THEN A.QUANTITY
                WHEN A.MOVE_TYPE = '102' AND SUBSTR(OEM.REPLY_DELIVERY,1,10) < SUBSTR(A.VOUCHER_POST_DATE,1,10) THEN -1 * A.QUANTITY
                ELSE 0
                 END) AS OVERTIME_QUANTITY --逾期已交付数量				 
  FROM DWROTD.DWR_WM_PRODSTOCK_OUT_IN_F A
  INNER JOIN SDI.SDI_APS_OEM_PO_SUM_1007 OEM
  	ON A.PURCHASE_VOUCHER_NUM = OEM.PURCHASE_ORDER_NO
   AND A.PURCHASE_VOUCHER_ITEM_NUM = OEM.PURCHASE_ORDER_PROJ_NO
  INNER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F D
    ON OEM.APS_PLAN_TASK_DTL_ID = D.ID
 WHERE 
    A.MOVE_TYPE in('101','102')
   AND COALESCE(A.SPECIAL_STOCK_FLAG_2,'') = 'K_T148T'
 GROUP BY A.PURCHASE_VOUCHER_NUM
      ,A.PURCHASE_VOUCHER_ITEM_NUM
)
,PA AS(
SELECT A.SUPPLIER_OR_CREDITOR_ID
       ,A.SUPPLIER_OR_CREDITOR_NUM
                 ,A.PRODUCT_AREA_ID
	             ,A.PRODUCT_AREA_CODE
                 ,A.PRODUCT_AREA_NAME
                 ,row_number() over(partition by A.SUPPLIER_OR_CREDITOR_NUM order by a.update_date DESC) AS RN
             FROM DWRDIM.DWR_DIM_PRODUCT_AREA_D A
             WHERE IS_PHANTOM=0
),MT AS(
SELECT
       O.SRC_MATERIAL_ID, --物料
	   O.MATERIAL_ID, --物料
	   O.MATERIAL_NUM, --物料编码
	   O.MATERIAL_NAME, --物料名称
	   O.REQUIRE_DELIVERY, --要求交期
	   O.CANCEL_QTY,
	   O.ID,
       O.INV_ORG_ID,
	   B.LENGTH_1, --长度
	   B.WIDTH, --宽度
	   B.DIMENSION, --规格
	   B.EXTERNAL_MATL_GROUP_CODE, --外部物料组
	   B.MATL_GROUP_CODE, --物料组
	   B.PRODUCT_LEVEL_CODE, --产品层次
	   B1.VTEXT AS PRODUCT_STAGE2, --产品层次2
	   B.RESERVED_FIELD_1 AS PRODUCT_CHANNEL_CODE, --产品渠道编码
	   B2.config_name AS PRODUCT_CHANNEL_NAME, --产品渠道名称
	   O.ORDER_NUM
  FROM DWROTD.DWR_PS_APS_PRO_ORDER_F O 
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_GENERAL_D B
    ON O.MATERIAL_ID = B.MATERIAL_ID
   AND B.END_DATE = '4712-12-31'
  LEFT JOIN SDI.SDI_T179T_1001 B1 --产品品类
    ON B1.MANDT = B.GROUP_NUM
   AND B1.SPRAS = '1'
   AND SUBSTR(B.PRODUCT_LEVEL_CODE,1,4) = B1.PRODH
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R B2
    ON B.RESERVED_FIELD_1 = B2.src_sys_config_code
   AND B2.source_code = 'ZTYL01'
 WHERE O.INV_ORG_ID = '1'
), CT AS(
SELECT PURCHASE_VOUCHER_NUM,PURCHASE_VCH_ITEM_NUM1,
       CUST_ID,CUST_NUM,CUST_NAME
FROM DWRPUR.DWR_PTP_PURCHASE_VCH_ITEM_F
GROUP BY PURCHASE_VOUCHER_NUM,PURCHASE_VCH_ITEM_NUM1,
       CUST_ID,CUST_NUM,CUST_NAME
),SALE_EMP_INFO AS(
SELECT A.PRODUCT_ORDER_ID
      ,CONCAT_WS(',',COLLECT_SET(B.BUSINESS_BY)) AS SALE_EMP_NAME
  FROM SDI.SDI_APS_PRO_ORDER_SALE_REL_1007 A
  LEFT JOIN SDI.SDI_SALE_ORDER_1007 B
    ON A.SALE_ORDER_ID = B.ID
 GROUP BY A.PRODUCT_ORDER_ID
)
INSERT OVERWRITE TABLE DM.DM_PS_PURCHASE_PLAN_DTL_T
SELECT 
       D.ID,
       D.WORK_ORDER,
       '00010' AS SEC_NO,
       T.ID AS APS_PLAN_TASK_ID,
       O.SRC_MATERIAL_ID, --物料
	   O.MATERIAL_ID, --物料
	   O.MATERIAL_NUM, --物料编码
	   O.MATERIAL_NAME, --物料名称
	   O.LENGTH_1, --长度
	   O.WIDTH, --宽度   
       I.SPECIFICATION_MODEL AS DIMENSION, --规格
	   O.EXTERNAL_MATL_GROUP_CODE, --外部物料组
	   O.MATL_GROUP_CODE, --物料组
	   O.PRODUCT_LEVEL_CODE, --产品层次
	   O.PRODUCT_STAGE2, --产品层次2
	   O.PRODUCT_CHANNEL_CODE, --产品渠道编码
	   O.PRODUCT_CHANNEL_NAME, --产品渠道名称
       U.NAME AS UNIT_NAME,
       COALESCE(D.PLAN_AMOUNT,0) AS PLAN_AMOUNT, --实际下单数量
       I.AREA * COALESCE(D.PLAN_AMOUNT,0) AS PURCHASING_AREA, --实际下单面积（平方米）
       O.REQUIRE_DELIVERY, --要求交期
       OEM.REPLY_DELIVERY, --回复交期
	   OEM.SUPPLIER_ID AS SUPPLIER_OR_CREDITOR_ID,
       S.CODE AS SUPPLIER_OR_CREDITOR_ACCT, --供应商编码
       S.NAME AS SUPPLIER_OR_CREDITOR_NAME, --供应商名称
	   PA.PRODUCT_AREA_ID, --产区
	   PA.PRODUCT_AREA_CODE, --产区编码
       PA.PRODUCT_AREA_NAME, --产区名称
       OEM.REMARK, 
       CASE WHEN T.CLOSE_FLAG = '1' THEN '已关闭'
            WHEN D.PLAN_AMOUNT - QT.QUANTITY <= 0 THEN '已完成交付'
            WHEN COALESCE(QT.QUANTITY,0) > 0 THEN '部分交付'
            WHEN OEM.STATUS = '10' THEN '未处理'
            WHEN OEM.STATUS = '20' THEN '审批中'
	        WHEN OEM.STATUS = '30' THEN '审批退回'
	        WHEN OEM.STATUS = '40' THEN '待回复交期'
	        WHEN OEM.STATUS = '50' THEN '已回复交期'
	        WHEN OEM.STATUS = '60' THEN '已确认并转单'
	        WHEN OEM.STATUS = '70' THEN '已完成交付'
	        WHEN OEM.STATUS = '80' THEN '部分交付'
	        WHEN OEM.STATUS = '90' THEN '超期审批中'
	        WHEN OEM.STATUS = '95' THEN '特批退回'
	        WHEN OEM.STATUS = '110' THEN '已关闭'
	        ELSE '未处理'
	         END AS STATUS, --采购计划订单状态
       OEM.OA_NO,
       OEM.PURCHASE_ORDER_NO AS PURCHASE_VOUCHER_NUM, --采购订单
       OEM.PURCHASE_ORDER_PROJ_NO AS PURCHASE_VOUCHER_ITEM_NUM, --采购订单项目
       COALESCE(QT.QUANTITY,0) AS ALREADY_LEVITE_QUANTITY, --已交付数量
	   COALESCE(QT.QUANTITY,0) * O.LENGTH_1 * O.WIDTH /1000000 AS ALREADY_LEVITE_AREA, --已交付面积（平方米）
       CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        ELSE COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)
             END AS NOT_LEVITE_QUANTITY, --未交付数量
	   CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        ELSE (COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)) * O.LENGTH_1 * O.WIDTH /1000000
             END AS NOT_LEVITE_AREA, --未交付面积（平方米）
	   OEM.PURCHASE_VOUCHER_TYPE, --采购凭证类型
       OEM.PURCHASE_VOUCHER,
       D.OEM_ORDER_DATE, --下单日期
       '' AS SALE_ORDER_CODE,
       OEM.TMS_REMARK,
       OEM.PURCHASE_ORG_ID, 
	   ORG.EKORG AS PURCHASE_ORG_CODE, --采购组织编码
	   ORG.EKOTX AS PURCHASE_ORG_NAME, --采购组织名称
       OEM.PURCHASE_GROUP_ID,
	   GUP.EKGRP AS PURCHASE_GROUP_CODE, --采购组编码
	   GUP.EKNAM AS PURCHASE_GROUP_NAME, --采购组名称
       OEM.APS_INVENTORY_LOCATION_ID,
	   IL.LGORT AS INV_LOCATION, --库存地点编码
	   IL.LGOBE AS INV_LOCATION_NAME, --库存地点名称
	   OEM.SAP_COMPANY_ID AS COMPANY_ID, 
	   CA.BUKRS AS COMPANY_CODE, --公司编码
	   CA.BUTXT AS COMPANY_NAME, --公司名称
       OEM.SAP_STOCK_AREA_ID AS FACTORY_WERKS_ID,
	   SA.WERKS AS FACTORY_WERKS_CODE, --工厂编码
	   SA.NAME AS FACTORY_WERKS_NAME, --工厂名称
       D.CREATE_BY,
       D.UPDATE_BY,
       D.CREATE_DATE,
       D.UPDATE_DATE,
	   COALESCE(O.CANCEL_QTY,0) AS CANCEL_QUANTITY, --取消数量
	   COALESCE(O.CANCEL_QTY,0) * O.LENGTH_1 * O.WIDTH /1000000 AS CANCEL_AREA, --取消面积（平方米）
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
	   CT.CUST_ID, --客户
	   CT.CUST_NUM, --客户
	   CT.CUST_NAME, --客户 
       CASE WHEN COALESCE(BD.BRAND_NAME,'') <> '' THEN BD.BRAND_NAME
	        WHEN COALESCE(BD1.BRAND_NAME,'') <> '' THEN BD1.BRAND_NAME
	        WHEN COALESCE(BD2.BRAND_NAME,'') <> '' THEN BD2.BRAND_NAME
			ELSE 'NA'
			 END AS BRAND_NAME, --品牌名称
	   D.PLAN_AMOUNT + COALESCE(O.CANCEL_QTY,0) AS ORIGINAL_PLAN_AMOUNT, --原下单数量
       I.AREA * (D.PLAN_AMOUNT + COALESCE(O.CANCEL_QTY,0)) AS ORIGINAL_PURCHASING_AREA, --原下单面积（平方米）
       NULL AS VOUCHER_POST_DATE, --过账日期
       QT.NOOVERTIME_QUANTITY, --未逾期已交付数量
       QT.OVERTIME_QUANTITY, --逾期已交付数量
	   CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        WHEN SUBSTR(CURRENT_DATE - interval 1 days,1,10) <= SUBSTR(OEM.REPLY_DELIVERY,1,10) THEN COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)
             END AS NOOVERTIME_NOT_LEVITE_QUANTITY, --未逾期未交量
	   CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        WHEN SUBSTR(CURRENT_DATE - interval 1 days,1,10) > SUBSTR(OEM.REPLY_DELIVERY,1,10) THEN COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)
             END AS OVERTIME_NOT_LEVITE_QUANTITY, --逾期未交量
       QT.NOOVERTIME_QUANTITY * O.LENGTH_1 * O.WIDTH /1000000 AS NOOVERTIME_AREA, --未逾期已交付面积（平方米）
       QT.OVERTIME_QUANTITY * O.LENGTH_1 * O.WIDTH /1000000 AS OVERTIME_AREA, --逾期已交付面积（平方米）
	   CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        WHEN SUBSTR(CURRENT_DATE - interval 1 days,1,10) <= SUBSTR(OEM.REPLY_DELIVERY,1,10) THEN (COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)) * O.LENGTH_1 * O.WIDTH /1000000
             END AS NOOVERTIME_NOT_LEVITE_AREA, --未逾期未交面积（平方米）
	   CASE WHEN COALESCE(D.PLAN_AMOUNT,0) <= COALESCE(QT.QUANTITY,0) THEN 0
	        WHEN SUBSTR(CURRENT_DATE - interval 1 days,1,10) > SUBSTR(OEM.REPLY_DELIVERY,1,10) THEN (COALESCE(D.PLAN_AMOUNT,0) - COALESCE(QT.QUANTITY,0)) * O.LENGTH_1 * O.WIDTH /1000000
             END AS OVERTIME_NOT_LEVITE_AREA, --逾期未交面积（平方米）
	   CASE WHEN T.CLOSE_FLAG = '0' THEN '否'
	        WHEN T.CLOSE_FLAG = '1' THEN '是'
             END AS CLOSE_FLAG, --关闭标识
       CASE WHEN T.PREPARE_CLOSE_FLAG = '0' THEN '否'
	        WHEN T.PREPARE_CLOSE_FLAG = '1' THEN '是'
             END AS PREPARE_CLOSE_FLAG, --预关闭标识
       I.PRODUCT_PROPERTY, --物料组1
       O.ORDER_NUM as PRODUCT_ORDER_NUM, -- 生产工单号
	   T.PLAN_NO, --报工号
	   OEM.SECOND_REPLY_DELIVERY, --二次回复交期
	   OEM.BOTTOM_LABEL, --底标
	   OEM.FOLLOW_PERSON, --跟进人
	   OEM.BIBULOUS_RATE, --吸水率编码
	   CASE WHEN OEM.BIBULOUS_RATE = 10 THEN 'E>10%'
	        WHEN OEM.BIBULOUS_RATE = 20 THEN '3%<E≤6%'
			WHEN OEM.BIBULOUS_RATE = 30 THEN '6%<E≤10%'
			WHEN OEM.BIBULOUS_RATE = 40 THEN '0.5%<E≤3%'
			WHEN OEM.BIBULOUS_RATE = 50 THEN 'E≤0.5%'
             END AS BIBULOUS_RATE_NAME, --吸水率
	   OEM.PACKAGE_WAY, --包装方式编码
	   CASE WHEN OEM.PACKAGE_WAY = 10 THEN '东鹏常规纸箱'
	        WHEN OEM.PACKAGE_WAY = 20 THEN '东鹏旧版纸箱'
			WHEN OEM.PACKAGE_WAY = 30 THEN '世界印象纸箱'
			WHEN OEM.PACKAGE_WAY = 40 THEN '梦之家纸箱'
			WHEN OEM.PACKAGE_WAY = 50 THEN 'ART+纸箱'
			WHEN OEM.PACKAGE_WAY = 60 THEN '中性纸箱'
			WHEN OEM.PACKAGE_WAY = 70 THEN '供应商纸箱'
	         END AS PACKAGE_WAY_NAME, --包装方式
	   OEM.ACCEPTANCE_CRITERIA, --验收标准编码
       CASE WHEN OEM.ACCEPTANCE_CRITERIA = 10 THEN '东鹏A类标准'
	        WHEN OEM.ACCEPTANCE_CRITERIA = 20 THEN '东鹏C类标准'
			WHEN OEM.ACCEPTANCE_CRITERIA = 30 THEN '国际'
			WHEN OEM.ACCEPTANCE_CRITERIA = 40 THEN '供应商标准'
			 END AS ACCEPTANCE_CRITERIA_NAME, --验收标准
	   OEM.CALCULATE_PROPORTION, --计算标准
	   OEM.ADOBE_BODY, --胚体编码
	   CASE WHEN OEM.ADOBE_BODY = 10 THEN '普通坯'
	        WHEN OEM.ADOBE_BODY = 20 THEN '色坯'
			WHEN OEM.ADOBE_BODY = 30 THEN '通体'
			WHEN OEM.ADOBE_BODY = 40 THEN '通体布料'
			 END AS ADOBE_BODY_NAME, --坯体
	   OEM.CLOSE_BILL_REASON_TXT, --关单原因
	   D.CUSTOMIZED_REQUIREMENTS, --其他特殊定制要求
	   T.SPLIT_REASON, --减数原因
	   T.SPLIT_STATE, --减数状态编码
	   CASE WHEN T.SPLIT_STATE = 1 THEN '待确认'
	        WHEN T.SPLIT_STATE = 2 THEN '已确认'
			 END AS SPLIT_STATE_NAME, --减数状态
	   PRO.SALE_EMP_NAME, --订单联系人
	   TAS.FLOW_STATUS, --超期入库状态编码
	   CASE WHEN TAS.FLOW_STATUS = 10 THEN '新建'
	        WHEN TAS.FLOW_STATUS = 20 THEN '20已申请报检'
			WHEN TAS.FLOW_STATUS = 30 THEN '报检合格'
			WHEN TAS.FLOW_STATUS = 31 THEN '报检退回'
			WHEN TAS.FLOW_STATUS = 40 THEN '已完成'
	        WHEN TAS.FLOW_STATUS = 41 THEN '超期审批中'
	        WHEN TAS.FLOW_STATUS = 42 THEN '特批退回'
			 END AS FLOW_STATUS_NAME --超期入库状态
  FROM DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F D
 INNER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_F T 
    ON T.ID = D.PLAN_TASK_ID 
   AND T.INV_ORG_ID = D.INV_ORG_ID
 INNER JOIN DWROTD.DWR_PS_RES_WIP_SCHE_F R 
    ON R.ID = T.RESOURCE_ID 
   AND R.IS_OUT_MADE = '1'
 INNER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F P 
    ON P.ID = D.PROCESS_TECH_TASK_ID 
   AND P.IS_PHANTOM = '0'
 INNER JOIN MT O 
    ON O.ID = P.PRODUCT_ORDER_ID 
   AND P.INV_ORG_ID = O.INV_ORG_ID 
 INNER JOIN SDI.SDI_ITEM_1007 I 
    ON I.ID = O.SRC_MATERIAL_ID 
   AND I.IS_PHANTOM = '0' 
   AND I.INV_ORG_ID = O.INV_ORG_ID 
 INNER JOIN SDI.SDI_BD_UNIT_1007 U 
    ON U.ID = I.UNIT_ID 
   AND U.IS_PHANTOM = '0' 
   AND U.INV_ORG_ID = O.INV_ORG_ID 
  LEFT JOIN SDI.SDI_APS_OEM_PO_SUM_1007 OEM 
    ON OEM.APS_PLAN_TASK_DTL_ID = D.ID
  LEFT JOIN SDI.SDI_SUPPLIER_1007 S 
    ON OEM.SUPPLIER_ID = S.ID
  LEFT JOIN SDI.SDI_APS_PURCHASE_ORG_1007 ORG 
    ON OEM.PURCHASE_ORG_ID = ORG.ID
  LEFT JOIN SDI.SDI_APS_PURCHASE_GROUP_1007 GUP 
    ON OEM.PURCHASE_GROUP_ID = GUP.ID
  LEFT JOIN SDI.SDI_APS_INVENTORY_LOCATION_1007 IL 
    ON OEM.APS_INVENTORY_LOCATION_ID = IL.ID
  LEFT JOIN SDI.SDI_APS_COMPANY_1007 CA 
    ON OEM.SAP_COMPANY_ID = CA.ID
  LEFT JOIN SDI.SDI_APS_SAP_STOCK_AREA_1007 SA 
    ON OEM.SAP_STOCK_AREA_ID = SA.ID
  LEFT JOIN QT
	ON QT.PURCHASE_VOUCHER_NUM = OEM.PURCHASE_ORDER_NO
   AND QT.PURCHASE_VOUCHER_ITEM_NUM = OEM.PURCHASE_ORDER_PROJ_NO
  LEFT JOIN PA
    ON S.CODE = PA.SUPPLIER_OR_CREDITOR_NUM AND PA.rn=1
  LEFT JOIN CT
    ON OEM.PURCHASE_ORDER_NO = CT.PURCHASE_VOUCHER_NUM
   AND OEM.PURCHASE_ORDER_PROJ_NO = CT.PURCHASE_VCH_ITEM_NUM1 
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD
    ON SA.WERKS = BD.FACTORY_WERKS_CODE
   AND O.EXTERNAL_MATL_GROUP_CODE = BD.EXTERNAL_MATL_GROUP_CODE
   AND O.MATL_GROUP_CODE = BD.MATL_GROUP_CODE
  LEFT JOIN DWRDIM.DWR_DIM_MATERIAL_BRAND_D BD1
    ON SA.WERKS = BD1.FACTORY_WERKS_CODE
   AND BD1.EXTERNAL_MATL_GROUP_CODE = ''
   AND O.MATL_GROUP_CODE = BD1.MATL_GROUP_CODE 
  LEFT JOIN (SELECT DISTINCT FACTORY_WERKS_CODE
                            ,BRAND_NAME
               FROM DWRDIM.DWR_DIM_MATERIAL_BRAND_D
			  WHERE MATL_GROUP_CODE = ''
			) BD2
    ON SA.WERKS = BD2.FACTORY_WERKS_CODE			
  LEFT JOIN SALE_EMP_INFO PRO
    ON PRO.PRODUCT_ORDER_ID = O.ID
  LEFT JOIN (SELECT SAP_PLAN_ORDER_NO
                   ,SAP_PLAN_ORDER_LINE_NO
				   ,FLOW_STATUS
				   ,ROW_NUMBER() OVER(PARTITION BY SAP_PLAN_ORDER_NO,SAP_PLAN_ORDER_LINE_NO ORDER BY SAP_PLAN_ORDER_ORDER DESC) AS RN
               FROM SDI.SDI_TMS_APS_STOCK_IN_D_1005 TAS) TAS
    ON OEM.PURCHASE_ORDER_NO = TAS.SAP_PLAN_ORDER_NO
   AND OEM.PURCHASE_ORDER_PROJ_NO = TAS.SAP_PLAN_ORDER_LINE_NO
   AND TAS.RN = 1
 WHERE D.WORK_ORDER IS NOT NULL
   AND D.INV_ORG_ID = '1'
-- ORDER BY D.WORK_ORDER DESC
