WITH T AS(
SELECT B.FORMAL_CHECK_DETAIL_ID
  FROM SDI.SDI_SETTLEMENT_FORMAL_CHECK_1033 SFC
 INNER JOIN SDI.SDI_SETTLEMENT_FORMAL_CHECK_DETAIL_1033 B
    ON SFC.FORMAL_CHECK_ID = B.FORMAL_CHECK_ID 
 WHERE SFC.DELETE_FLAG = 0
   AND B.DELETE_FLAG = 0
   AND SFC.AFFIRM_RESULT = 'GRANT_ACCEPTANCE' --审批通过
   --AND B2.MEANING IN('未请款','部分请款')
)
INSERT OVERWRITE TABLE DM.DM_FIN_AR_NODEEXCPT_LTC_REQUEST_PAYOUT_T 
SELECT B.REQUEST_PAYOUT_CODE --请款单号
      ,B.WBS
      ,B.CREATION_DATE --请款日期
	  ,C.REQUEST_INVOICE_TIME2 --请款到开票时间（天）
      ,SUM(A.CURRENT_REQUEST_AMOUNT) AS TOTAL_REQUEST_AMOUNT --请款金额
      ,SUM(A.CURRENT_TICKET_AMOUNT) AS CURRENT_TICKET_AMOUNT --回款金额
      ,SUM(A.CURRENT_REQUEST_AMOUNT) - SUM(A.CURRENT_TICKET_AMOUNT) UN_TICKET_AMOUNT --请款未回款金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,B.TICKET_STATUS --请款状态
	  ,D.MEANING AS TICKET_STATUS_NAME --请款状态描述
  FROM SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_DETAIL_1033 A
  LEFT JOIN SDI.SDI_SETTLEMENT_REQUEST_PAYOUT_1033 B
    ON A.REQUEST_PAYOUT_ID = B.REQUEST_PAYOUT_ID
  LEFT JOIN (SELECT DISTINCT 
                             A.CONTRACT_ID
							,A.REQUEST_INVOICE_TIME2
							,B.CONTRACT_CODE
               FROM SDI.SDI_CONTRACT_SETTLEMENT_HEAD_1033 A
			   LEFT JOIN SDI.SDI_CONTRACT_1033 B
			     ON A.CONTRACT_ID = B.CONTRACT_ID
			    AND B.DELETE_FLAG = 0    
			  WHERE A.DELETE_FLAG = 0    
            ) C
	ON B.CONTRACT_CODE = C.CONTRACT_CODE		
  LEFT JOIN SDI.SDI_BASE_LOOKUP_VALUES_1029 D
    ON B.TICKET_STATUS = D.LOOKUP_CODE
   AND D.LOOKUP_TYPE = 'SETTLE_OPEN_TICKET_STATUS'
   AND D.ENABLED_FLAG = 'Y'
 WHERE A.DELETE_FLAG = 0
   AND B.DELETE_FLAG = 0
   AND A.FORMAL_CHECK_DETAIL_ID IN(SELECT FORMAL_CHECK_DETAIL_ID FROM T)
 GROUP BY B.REQUEST_PAYOUT_CODE --请款单号
      ,B.WBS
      ,B.CREATION_DATE --请款日期
	  ,C.REQUEST_INVOICE_TIME2 --请款到开票时间（天）
	  ,B.TICKET_STATUS --请款状态
	  ,D.MEANING --请款状态描述
