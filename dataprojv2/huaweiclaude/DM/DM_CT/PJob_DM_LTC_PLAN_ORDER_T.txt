INSERT OVERWRITE TABLE DM.DM_LTC_PLAN_ORDER_T
SELECT
 A.PLAN_NUM --计划单号
,A.WBS_NUM --WBS号
,Q.post1 AS WBS_NAME -- WBS名称
,C.ITEM_NUM AS PLAN_ITEM_NUM --计划单行号
,A.CHANGE_NUM --工程报备号
,A.PURCHASE_ID --要货单头id
,C.DP_REFERENCE_ORDER AS REFERENCE_ORDER_NUM --参照计划单行号
,A.CREATED AS CREATE_TIME --创建日期
,A.PLAN_ORDER_COMMENT --计划单备注
,I.OU_NUM AS CUST_NUM --客户编码
,I.DESC_TEXT AS CUST_NAME --客户名称
,A.END_TIME	--结束时间
,A.SERVICE_TIME	--送达时间
,A.BREAKDUTY AS BREAK_DUTY --违约责任
,A.SALE_DEPT --销售部门
,J.VAL AS STATUS --状态
,K.VAL AS PLAN_ORDER_TYPE --计划订单类型
,M.VAL AS PLAN_TYPE	--类型
,A.DISCHARGER --卸货方
,B.SALES_CHAN_CD AS SALES_CHANNEL_CODE --分销渠道编号
,B.SALES_CHAN AS SALES_CHANNEL_NAME --分销渠道名称
,P.VAL AS CARRIAGE --承运条件
,N.VAL AS STOCK_TYPE --备货类型
,A.ALLSUBMIT AS IS_ZDJF --是否整单交付
,O.VAL AS TAKE_DELIVERY_TYPE --提货方式
,C.APPROVAL_QTY AS APPROVAL_QUANTITY --计划单审批数量
,C.APPLY_QTY AS APPLY_QUANTITY --转订单数量
,C.CREATED_ORDER_QTY AS CREATED_ORDER_QUANTITY --计划单已转订单数量
,C.AREA	--面积
,H.NAME AS MATERIAL_NUM --物料编码
,H.DESC_TEXT AS MATERIAL_NAME --物料名称
,H.X_SIZE_DIMENSION AS DIMENSION --规格
,H.X_PROD_HRY AS PRODUCT_LEVEL_CODE --产品层次
,H.X_LENGHT AS LENGTH_1 --长度
,H.X_WIDE AS WIDE --宽度
,H.X_HEIGHT AS HEIGHT --高度
,H.X_PROD_LEVEL AS PROD_LEVEL --产品分级
,C.UNIT	--物料单位
,C.APP_ENTER_DATE --审批最早时间
,C.EXCEPT_DATE --希望最早交货日期
,C.DP_BASE_PRICE --单价
,C.BATCH AS BATCH_NUM --批次
,C.DELIVERY_DATE --交货日期
,C.DP_GO_UP AS GO_UP_PRICE --提价费
,C.AVL_AMT AS AVAILABLE_QUANTITY --可用数量
,C.DEFAULT_DELIVERY_TIME --默认交货时间
,C.CANCEL_REASON --评审拒绝
,C.DP_BILL_SUBMIT_TIME AS SUBMIT_TIME --提交时间
,C.X_ORDER_DESC AS ORDER_COMMENT --计划单备注
,C.MODIFICATION_NUM	AS MODIFY_QUANTITY --修改数量
,C.DP_SUPPLY_AREA AS PRODUCE_AREA --供货产区名称
,C.DP_MOQ_NUMBER AS MOQ_QUANTITY --起订量数量
,D.SALES_ORG_CD AS SALES_ORG_CODE --销售组织编码
,D.SALES_ORG AS SALES_ORG_NAME --销售组织名称
,G.FACTORY_CD AS PRODUCT_FACTORY_CODE --产品属地化工厂编码
,G.FACTORY_NAME AS PRODUCT_FACTORY_NAME --产品属地化工厂名称
,C.REVIEW_PRODUCT_TIME2 AS REVIEW_PRODUCT_TIME --工厂交货日期
,E.FACTORY_CD AS FACTORY_CODE --工厂编码
,E.FACTORY_NAME --工厂名称
,F.FACTORY_CD AS APPROVAL_FACTORY_CODE --审批工厂编码
,F.FACTORY_NAME AS APPROVAL_FACTORY_NAME --审批工厂名称
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
,R.VAL AS ITEM_STATUS --行状态
,CASE WHEN A.VOLUN_FLAG = 'Y' THEN '是'
      WHEN A.VOLUN_FLAG = 'N' THEN '否'
	  ELSE 'N/A'
	   END AS VOLUN_FLAG --是否为自动评审
,C.APPROVAL_INFO AS APPROVAL_REMARK --审批意见
,C.LAST_UPD AS APPROVAL_DATE --评审时间
,C.PRODUCTION_SCHEDULE --生产进度
,C.GROSS_MARGIN --毛利润额
,C.GROSS_MARGIN_FLAG --毛利润额分类
FROM SDI.SDI_CX_PLAN_ORDER_1002 A
LEFT JOIN SDI.SDI_CX_SALES_CHAN_1002 B
ON A.CHANNL_ID = B.ROW_ID
AND B.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_PLAN_O_ITEM_1002 C
ON A. ROW_ID = C.PLAN_ORDER_ID
AND C.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_SALES_ORG_1002 D
ON A.ORG_ID = D.ROW_ID
AND D.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_FACTORY_1002 E
ON C.FACTORY_ID = E.ROW_ID
AND E.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_FACTORY_1002 F
ON C.APPROVAL_FACTORY_ID = F.ROW_ID
AND F.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_CX_FACTORY_1002 G
ON C.PRODUCT_FACTORY = G.ROW_ID
AND G.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_PROD_INT_1002 H
ON C.PROD_ID = H.ROW_ID
AND H.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_ORG_EXT_1002 I
ON A.ACC_ID = I.PAR_ROW_ID
AND I.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 J
ON A.STATUS = J.NAME
AND J.`TYPE` = 'FS_ORDER_STATUS'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 K
ON A.PLAN_ORDER_TYPE = K.NAME
AND K.`TYPE` = 'ORDER_PLAN_TYPE'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 M
ON A.`TYPE` = M.NAME
AND M.`TYPE` = 'DP_PLAN_ORDERV2_TYPE'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 N
ON A.DP_STOCK_TYPE = N.NAME
AND N.`TYPE` = 'DP_PLAN_PROPERTY'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 O
ON A.PICKTYPE = O.NAME
AND O.`TYPE` = 'N_SHIP_TYPE'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 P
ON A.CARRIAGE = P.NAME
AND P.`TYPE` = 'DP_CARRIAGE_TERMS'
LEFT JOIN SDI.SDI_PRPS_1001 Q
ON A.WBS_NUM = Q.POSID
AND Q.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_S_LST_OF_VAL_1002 R
ON C.STAUTS = R.NAME
AND R.`TYPE` = 'FS_ORDER_STATUS'
WHERE A.DEL_FLAG = 'N'
AND SUBSTR(A.CREATED,1,4) >= '2022'
;
