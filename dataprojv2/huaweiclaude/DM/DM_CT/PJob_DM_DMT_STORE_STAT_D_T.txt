WITH T1 AS
(
SELECT SUBSTR(START_DATE,1,10) AS STAT_DATE
      ,A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
	  ,COUNT(DISTINCT A.STORE_ID) AS STORE_COUNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM DWIMD.DWI_MD_DMT_STORE_T A
  LEFT JOIN DWIMD.DWI_MD_DMT_DEPT_INFO_T B
    ON A.STORE_ID = B.STORE_ID
 WHERE A.STATUS = 1
 GROUP BY
       SUBSTR(START_DATE,1,10)
      ,A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
),T2 AS(
SELECT SUBSTR(UPDATE_DATE,1,10) AS STAT_DATE
      ,A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
	  ,COUNT(DISTINCT A.STORE_ID) AS STORE_COUNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM DWIMD.DWI_MD_DMT_STORE_T A
  LEFT JOIN DWIMD.DWI_MD_DMT_DEPT_INFO_T B
    ON A.STORE_ID = B.STORE_ID
 WHERE A.STATUS = 0
 GROUP BY
       SUBSTR(UPDATE_DATE,1,10)
      ,A.STORE_TAG
      ,A.PROVINCE_NAME
	  ,A.CITY_NAME
	  ,A.DISTRICT_NAME
	  ,B.LEV1_DEPT_NAME
	  ,B.LEV2_DEPT_NAME
	  ,B.LEV3_DEPT_NAME
	  ,B.LEV4_DEPT_NAME
	  ,B.LEV5_DEPT_NAME
	  ,B.LEV6_DEPT_NAME
),T3 AS(
SELECT
       STAT_DATE
      ,STORE_TAG
      ,PROVINCE_NAME
	  ,CITY_NAME
	  ,DISTRICT_NAME
	  ,LEV1_DEPT_NAME
	  ,LEV2_DEPT_NAME
	  ,LEV3_DEPT_NAME
	  ,LEV4_DEPT_NAME
	  ,LEV5_DEPT_NAME
	  ,LEV6_DEPT_NAME
	  ,STORE_COUNT AS OPEN_STORE_COUNT
	  ,0 AS CLOSE_STORE_COUNT
  FROM T1
  
 UNION ALL

SELECT
       STAT_DATE
      ,STORE_TAG
      ,PROVINCE_NAME
	  ,CITY_NAME
	  ,DISTRICT_NAME
	  ,LEV1_DEPT_NAME
	  ,LEV2_DEPT_NAME
	  ,LEV3_DEPT_NAME
	  ,LEV4_DEPT_NAME
	  ,LEV5_DEPT_NAME
	  ,LEV6_DEPT_NAME
	  ,0 AS OPEN_STORE_COUNT
	  ,STORE_COUNT AS CLOSE_STORE_COUNT
  FROM T2 
)
INSERT OVERWRITE TABLE DM.DM_DMT_STORE_STAT_D_T
SELECT
       STAT_DATE
      ,STORE_TAG
      ,PROVINCE_NAME
	  ,CITY_NAME
	  ,DISTRICT_NAME
	  ,LEV1_DEPT_NAME
	  ,LEV2_DEPT_NAME
	  ,LEV3_DEPT_NAME
	  ,LEV4_DEPT_NAME
	  ,LEV5_DEPT_NAME
	  ,LEV6_DEPT_NAME
	  ,SUM(OPEN_STORE_COUNT) AS OPEN_STORE_COUNT
	  ,SUM(CLOSE_STORE_COUNT) AS CLOSE_STORE_COUNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T3
 GROUP BY
       STAT_DATE
