-- 营销系统留备货 
with tmp as(
	select t1.material_num ,
		   t1.batch_num ,
		   t1.FACTORY_CODE,
		   t1.inv_location,
		   t1.lev1_name,
		   t1.lev2_name,
		   t1.lev3_name,
		   t1.lev4_name,
		   t1.lev5_name,
		   t1.lev6_name,
		   '' as belong_company,
		   sum(SKU_CONFIRM_QTY) as res_stock_qty,  -- 留货数(非项目库存)
		   sum(SKU_CONFIRM_QTY_PRO) as res_stock_qty_pro, -- 留货数(项目库存)
		   0 as sku_prepare_qty,0 as pro_sku_prepare_qty
	from  DWROTD.dwr_wm_stock_res_from_sales  t1
	join dm.dm_wm_stock_responsibility_mat_tmp t2 
	On t1.material_num  = t2.material_num
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.FACTORY_CODE,
			 t1.inv_location,
			 t1.lev1_name,
			 t1.lev2_name,
			 t1.lev3_name,
			 t1.lev4_name,
			 t1.lev5_name,
			 t1.lev6_name
union all 
	select  t1.material_num ,
			t1.batch_num ,
			t1.stockout_factory_code,
			t1.inv_location,
			'', 
			'', 
			'', 
			'',
			'',
			'',
			belong_company,
			sum(res_stock_qty) as res_stock_qty,
			0 as res_stock_qty_pro,
			0,
			0
	from DWROTD.DWR_WM_STOCK_RES_FROM_PURCHASE t1
	join dm.dm_wm_stock_responsibility_mat_tmp t2 
	On t1.material_num  = t2.material_num
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.stockout_factory_code,
			 t1.inv_location,
			 belong_company
union all  
-- 备货
	select t1.material_num ,
		   t1.batch_num ,
		   t1.FACTORY_CODE,
		   t1.inv_location,
		   t1.lev1_name,
		   t1.lev2_name,
		   t1.lev3_name,
		   t1.lev4_name,
		   t1.lev5_name,
		   t1.lev6_name,
		   case when t1.order_source ='PURCHASE' AND t1.butxt like '%子公司%' then t1.butxt
			    when t1.order_source ='PURCHASE' then '其他' end as belong_company,
		   0,
		   0,
		   sum(t1.sku_prepare_qty) as  sku_prepare_qty,
		   sum(t1.pro_sku_prepare_qty) as pro_sku_prepare_qty
	from dwrotd.dwr_wm_stock_prepare_from_stockout  t1
	join dm.dm_wm_stock_responsibility_mat_tmp t2 
	On t1.material_num  = t2.material_num 
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.FACTORY_CODE,
			 t1.inv_location,
			 t1.lev1_name,
			 t1.lev2_name,
			 t1.lev3_name,
			 t1.lev4_name,
			 t1.lev5_name,
			 t1.lev6_name,
			 case when t1.order_source ='PURCHASE' AND t1.butxt like '%子公司%' then t1.butxt
				   when t1.order_source ='PURCHASE' then '其他' end 
union all
	select t1.material_num ,
		   t1.batch_num ,
		   t1.FACTORY_CODE,
		   t1.inv_location,
		   0,
		   0,
		   0,
		   0,
		   0,
		   0,
		   '其他' as belong_company,
		   0,
		   0,
		   sum(erfmg) - sum(enmng) as sku_prepare_qty,
		   0
	from dwrotd.dwr_wm_stock_prepare_from_res t1 
	join dm.dm_wm_stock_responsibility_mat_tmp t2 
	On t1.material_num  = t2.material_num 
	group by t1.material_num ,
			 t1.batch_num ,
			 t1.FACTORY_CODE,
			 t1.inv_location
) 
insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_RES_PREPARE_TMP
select t1.material_num ,
	   t1.batch_num ,
	   t1.FACTORY_CODE,
	   t1.inv_location,
       t1.lev1_name,
	   t1.lev2_name,
	   t1.lev3_name,
	   t1.lev4_name,
	   t1.lev5_name,
	   t1.lev6_name,
	   belong_company,
       case   
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='总裁虚拟组' then '第三方包销'
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='国内营销系统' and t1.lev4_name='区域运营中心'        then '区域运营中心'
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='国内营销系统' and t1.lev4_name='直营经营管理板块' then '直营经营管理板块'
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='国内营销系统' and t1.lev4_name='工程市场中心' then '工程市场中心'
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='国际营销系统'  then '国际营销系统'
            when t1.lev2_name = '瓷砖事业部'  and t1.lev3_name='梦之家事业部' then '梦之家'
            when t1.belong_company like '%子公司%' then '直营经营管理板块'
            else '其他部门留备货' end as resp_system,
       sum(sku_prepare_qty) as sku_prepare_qty,
       sum(res_stock_qty) as res_stock_qty,
       sum(sku_prepare_qty) + sum(res_stock_qty) as res_prepare_stock_qty,
	   current_timestamp() as dw_last_update_date
from tmp t1
group by t1.material_num ,
		 t1.batch_num ,
		 t1.FACTORY_CODE,
		 t1.inv_location,
         t1.lev1_name,
		 t1.lev2_name,
		 t1.lev3_name,
		 t1.lev4_name,
		 t1.lev5_name,
		 t1.lev6_name,
		 belong_company;
