DELETE FROM DM.DM_AR_RCVBL_STAT_T
 WHERE STAT_MONTH BETWEEN LEFT('${PERIOD_ID_D}',7) AND LEFT('${PERIOD_ID_D1}',7)
;

WITH T AS(
SELECT LEFT(A.QUERY_DATE,7) AS STAT_MONTH --统计月份
      ,A.SALES_GROUP_MASTER_FILE AS SALE_GRP_CODE --销售组编码
	  ,SUM(A.OVERDUE_RECEIVABLES) AS OVERDUE_RECEIVABLES --逾期应收金额
	  ,SUM(A.N_OVERDUE_RECEIVABLES) AS N_OVERDUE_RECEIVABLES --未逾期应收金额
      ,SUM(A.OVERDUE_RECEIVABLES + A.N_OVERDUE_RECEIVABLES) AS TOT_RECEIVABLES_AMT --总应收金额
	  ,SUM(A.OVERDUE_RECEIVABLES_SUM) AS OVERDUE_RECEIVABLES_SUM --逾期账龄汇总金额
	  ,SUM(A.OVERDUE_1_MONTH_SUM) AS OVERDUE_1_MONTH_SUM --一个月内逾期账龄汇总金额
	  ,SUM(A.N_OVERDUE_RECEIVABLES_SUM) AS N_OVERDUE_RECEIVABLES_SUM --自然账龄汇总金额
	  ,SUM(A.OVERDUE_RECEIVABLES_SUM + A.OVERDUE_1_MONTH_SUM + A.N_OVERDUE_RECEIVABLES_SUM) AS TOT_IMPAIRMENT_AMT --总减值金额
	  ,0 AS NOTAX_SALES_NET_AMT --不含税收入净额
	  ,SUM(CASE WHEN A.GENERAL_LEDGER_ACCOUNT != '0011110100' THEN A.OVERDUE_RECEIVABLES + A.N_OVERDUE_RECEIVABLES ELSE 0 END) AS NO_TICKET_RECEIVABLES_AMT --不含票据总应收金额
	  ,SUM(CASE WHEN A.GENERAL_LEDGER_ACCOUNT != '0011110100' THEN A.OVERDUE_RECEIVABLES ELSE 0 END) AS NO_TICKET_OVERDUE_AMT --不含票据应收逾期金额
	  ,SUM(CASE WHEN A.GENERAL_LEDGER_ACCOUNT = '0011110100' THEN A.OVERDUE_RECEIVABLES + A.N_OVERDUE_RECEIVABLES ELSE 0 END) AS TICKET_RECEIVABLES_AMT --票据总应收金额
	  ,SUM(CASE WHEN A.GENERAL_LEDGER_ACCOUNT = '0011110100' THEN A.OVERDUE_RECEIVABLES ELSE 0 END) AS TICKET_OVERDUE_AMT --票据应收逾期金额
  FROM DWRFIN.DWR_AR_CREDIT_DEVALUE_F A
 WHERE LEFT(A.QUERY_DATE,7) BETWEEN LEFT('${PERIOD_ID_D}',7) AND LEFT('${PERIOD_ID_D1}',7)
   AND COMP_CODE != '8000'
   AND IS_OVERDUE_CUST = 'N'
   AND COALESCE(CUST_GROUP_CODE,'#') != 'Z0'
   AND GENERAL_LEDGER_ACCOUNT IN ('0011310100','0011310600','0011310700','0011310800','0011110100','0011310300')
 GROUP BY LEFT(A.QUERY_DATE, 7) --统计月份
      ,A.SALES_GROUP_MASTER_FILE --销售组编码
	  
 UNION ALL

SELECT A.CALMONTH AS STAT_MONTH --统计月份
      ,A.SALES_GRP AS SALE_GRP_CODE --销售组编码
	  ,0 AS OVERDUE_RECEIVABLES --逾期应收金额
	  ,0 AS N_OVERDUE_RECEIVABLES --未逾期应收金额
      ,0 AS TOT_RECEIVABLES_AMT --总应收金额
	  ,0 AS OVERDUE_RECEIVABLES_SUM --逾期账龄汇总金额
	  ,0 AS OVERDUE_1_MONTH_SUM --一个月内逾期账龄汇总金额
	  ,0 AS N_OVERDUE_RECEIVABLES_SUM --自然账龄汇总金额
	  ,0 AS TOT_IMPAIRMENT_AMT --总减值金额 
	  ,A.NOTAX_SALES_NET_AMT --不含税收入净额
	  ,0 AS NO_TICKET_RECEIVABLES_AMT --不含票据总应收金额
	  ,0 AS NO_TICKET_OVERDUE_AMT --不含票据应收逾期金额
	  ,0 AS TICKET_RECEIVABLES_AMT --票据总应收金额
	  ,0 AS TICKET_OVERDUE_AMT --票据应收逾期金额
  FROM DM.DM_AMBV2_PROFIT_T A
 WHERE A.CALMONTH BETWEEN LEFT('${PERIOD_ID_D}',7) AND LEFT('${PERIOD_ID_D1}',7)
   AND COALESCE(CUST_GROUP, '#') != 'Z0')
INSERT INTO DM.DM_AR_RCVBL_STAT_T
     (STAT_MONTH --统计月份
      ,SALE_GRP_CODE --销售组编码
	  ,OVERDUE_RECEIVABLES --逾期应收金额
	  ,N_OVERDUE_RECEIVABLES --未逾期应收金额
      ,TOT_RECEIVABLES_AMT --总应收金额
	  ,OVERDUE_RECEIVABLES_SUM --逾期账龄汇总金额
	  ,OVERDUE_1_MONTH_SUM --一个月内逾期账龄汇总金额
	  ,N_OVERDUE_RECEIVABLES_SUM --自然账龄汇总金额
	  ,TOT_IMPAIRMENT_AMT --总减值金额 
	  ,NOTAX_SALES_NET_AMT --不含税收入净额
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,NO_TICKET_RECEIVABLES_AMT --不含票据总应收金额
	  ,NO_TICKET_OVERDUE_AMT --不含票据应收逾期金额
	  ,TICKET_RECEIVABLES_AMT --票据总应收金额
	  ,TICKET_OVERDUE_AMT --票据应收逾期金额
	  )
SELECT A.STAT_MONTH --统计月份
      ,A.SALE_GRP_CODE --销售组编码
	  ,SUM(A.OVERDUE_RECEIVABLES) AS OVERDUE_RECEIVABLES --逾期应收金额
	  ,SUM(A.N_OVERDUE_RECEIVABLES) AS N_OVERDUE_RECEIVABLES --未逾期应收金额
      ,SUM(A.TOT_RECEIVABLES_AMT) AS TOT_RECEIVABLES_AMT --总应收金额
	  ,SUM(A.OVERDUE_RECEIVABLES_SUM) AS OVERDUE_RECEIVABLES_SUM --逾期账龄汇总金额
	  ,SUM(A.OVERDUE_1_MONTH_SUM) AS OVERDUE_1_MONTH_SUM --一个月内逾期账龄汇总金额
	  ,SUM(A.N_OVERDUE_RECEIVABLES_SUM) AS N_OVERDUE_RECEIVABLES_SUM --自然账龄汇总金额
	  ,SUM(A.TOT_IMPAIRMENT_AMT) AS TOT_IMPAIRMENT_AMT --总减值金额 
	  ,SUM(A.NOTAX_SALES_NET_AMT) AS NOTAX_SALES_NET_AMT --不含税收入净额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,SUM(A.NO_TICKET_RECEIVABLES_AMT) AS NO_TICKET_RECEIVABLES_AMT --不含票据总应收金额
	  ,SUM(A.NO_TICKET_OVERDUE_AMT) AS NO_TICKET_OVERDUE_AMT --不含票据应收逾期金额
	  ,SUM(A.TICKET_RECEIVABLES_AMT) AS TICKET_RECEIVABLES_AMT --票据总应收金额
	  ,SUM(TICKET_OVERDUE_AMT) AS TICKET_OVERDUE_AMT --票据应收逾期金额
  FROM T AS A
 GROUP BY A.STAT_MONTH --统计月份
      ,A.SALE_GRP_CODE --销售组编码
