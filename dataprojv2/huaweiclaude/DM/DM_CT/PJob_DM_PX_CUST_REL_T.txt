insert overwrite table dm.dm_px_cust_rel_t
select user_name_virtual_column as uuid,lpad(user_name,10,'0') as cust_num,user_full_name as cust_name,current_timestamp as dw_last_update_date
  from sdi.SDI_BASE_USERS_1029
 where user_type = 'OUT'
   and substr(user_name,1,1) in('0','1')
;
