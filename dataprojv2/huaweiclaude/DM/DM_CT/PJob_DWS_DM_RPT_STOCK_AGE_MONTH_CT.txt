delete from dm.dm_rpt_stock_age_month_ct ;
insert into  dm.dm_rpt_stock_age_month_ct
select calmonth,
        sum(zsjkcje) zsjkcje,   -- 库存总金额
        sum(ybzq_bzdq_amt) as ybzq_bzdq_amt,   -- 有保质期到期
sum(ybzq_bzdq_3_amt) as ybzq_bzdq_3_amt,  -- 有保质期三个月内到期
sum(wbzq_6_amt) as wbzq_6_amt,  --无保质期6个月以内
sum(wbzq_6_12_amt) as wbzq_6_12_amt,  -- 无保质期6-12月
sum(wbzq_12_24_amt) as  wbzq_12_24_amt,  -- 无保质期1年-2年
sum(wbzq_24_amt) as wbzq_24_amt,   -- 无保质期2年以上
sum(zsjkcje - ybzq_bzdq_amt-ybzq_bzdq_3_amt-wbzq_6_amt-wbzq_6_12_amt-wbzq_12_24_amt-wbzq_24_amt) as other_amt  --其他
        
         from  dm.dm_fin_stock_detail_accage_t_2023 t
    where zdpsyb IN( '11000001','11000003')
    --and calmonth >= '2024'
    and t.comp_code <>'7100'
    group by calmonth;
