INSERT OVERWRITE TABLE DM.DM_PS_ORDER_PLAN_T
SELECT
        T0.ID
       ,T0.END_TIME
       ,T0.FINISHED_AMOUNT
       ,T0.FINISHED_AREA
       ,T0.IS_CANCEL
       ,T0.ITEM_CODE
       ,T0.ITEM_NAME
       ,T1.PLAN_AMOUNT PLAN_AMOUNT
       ,T0.PLAN_AREA
       ,T0.PLAN_NO--报工号
       ,T0.PLAN_TASK_DETAIL_ID
       ,T0.PLAN_TASK_ID
       ,T0.PROCESS_TECH_TASK_ID
       ,T3.PROTECH_NUM
       ,T0.PRODUCT_AREA_CODE
       ,T0.PRODUCT_AREA_ID
       ,T0.PRODUCT_AREA_NAME
       ,T0.PRODUCT_ORDER_CODE
       ,T0.PRODUCT_ORDER_ID --生产订单
       ,T0.PROMISE_DELIVERY
       ,T0.RE_DATE
       ,T0.RELEASE_START_TIME
       ,T0.REQUIRE_DELIVERY
       ,T0.RESOURCE_CODE
       ,T0.RESOURCE_ID
       ,T0.RESOURCE_NAME
       ,T0.RE_STATE
       ,T0.START_TIME
       ,T0.STATUS
       ,T0.Is_Canceled
       ,T0.Release_End_Time
       ,T0.Area
       ,T0.Product_Property
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM(SELECT PLANTASKDETAIL.ID
           ,PLANTASKDETAIL.ID AS PLAN_TASK_DETAIL_ID
           ,TECHORDER.ID AS PROCESS_TECH_TASK_ID
           ,TASK.ID AS PLAN_TASK_ID
           ,PRODUCTORDER.ID AS PRODUCT_ORDER_ID
           ,PRODUCTORDER.ORDER_NUM AS PRODUCT_ORDER_CODE
           ,PRODUCTORDER.PROMISE_DELIVERY AS PROMISE_DELIVERY
           ,PRODUCTORDER.REQUIRE_DELIVERY AS REQUIRE_DELIVERY
           ,ITEM.ID AS ITEM_ID
           ,ITEM.CODE AS ITEM_CODE
           ,ITEM.NAME AS ITEM_NAME
           ,WIPRES.ID AS RESOURCE_ID
           ,WIPRES.CODE AS RESOURCE_CODE
           ,WIPRES.NAME AS RESOURCE_NAME
           ,T0.ID AS PRODUCT_AREA_ID
           ,T0.CODE AS PRODUCT_AREA_CODE
           ,T0.NAME AS PRODUCT_AREA_NAME
           ,FINISHEDVM.FINISHED_AMOUNT
           ,(PLANTASKDETAIL.PLAN_AMOUNT * ITEM.AREA) AS PLAN_AREA
           ,(FINISHEDVM.FINISHED_AMOUNT * ITEM.AREA) AS FINISHED_AREA
           ,TASK.PLAN_NO
           ,TASK.START_TIME
           ,TASK.END_TIME
           ,TASK.STATUS
           ,TASK.IS_CANCEL
           ,TASK.RE_STATE
           ,TASK.RE_DATE
           ,PLANTASKDETAIL.RELEASE_START_TIME
           ,TASK.Is_Cancel AS Is_Canceled
           ,PLANTASKDETAIL.Release_End_Time
           ,PRODUCTORDER.ACREAGE AS Area
           ,ITEM.Product_Property
    FROM DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F PLANTASKDETAIL
    LEFT OUTER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_F TASK 
	  ON PLANTASKDETAIL.PLAN_TASK_ID = TASK.ID 
	 AND TASK.INV_ORG_ID = '1'
    LEFT OUTER JOIN DWROTD.DWR_PS_RES_WIP_SCHE_F WIPRES 
	  ON TASK.RESOURCE_ID = WIPRES.ID
     AND WIPRES.INV_ORG_ID = '1'
	 AND WIPRES.IS_PHANTOM = '0'
    LEFT OUTER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F TECHORDER 
	  ON PLANTASKDETAIL.PROCESS_TECH_TASK_ID = TECHORDER.ID 
     AND TECHORDER.INV_ORG_ID = '1'
	 AND TECHORDER.IS_PHANTOM = '0'
    LEFT OUTER JOIN DWROTD.DWR_PS_APS_PRO_ORDER_F PRODUCTORDER 
	  ON TECHORDER.PRODUCT_ORDER_ID = PRODUCTORDER.ID 
     AND PRODUCTORDER.INV_ORG_ID = '1'  
	 AND PRODUCTORDER.IS_PHANTOM = '0'
    LEFT OUTER JOIN sdi.sdi_item_1007 ITEM
	  ON PRODUCTORDER.SRC_MATERIAL_ID = ITEM.ID
     AND ITEM.INV_ORG_ID = '1'  
	 AND ITEM.IS_PHANTOM = '0'
    LEFT OUTER JOIN (SELECT 
	                       DAILYSCHEDULE.PLAN_TASK_ID PLANTASKID
	                      ,DAILYSCHEDULE.PLAN_TASK_ID PLAN_TASK_ID
						  ,DAILYSCHEDULE.PROCESS_TECH_ORDER_ID PROCESS_TECH_TASK_ID
						  ,SUM(DAILYSCHEDULE.QTY) FINISHED_AMOUNT
                       FROM DWROTD.DWR_PS_DAILY_SCHEDULE_F DAILYSCHEDULE
                      WHERE DAILYSCHEDULE.INV_ORG_ID = '1'
                      GROUP BY DAILYSCHEDULE.PLAN_TASK_ID
					         , DAILYSCHEDULE.PROCESS_TECH_ORDER_ID) FINISHEDVM 
      ON PLANTASKDETAIL.PLAN_TASK_ID = FINISHEDVM.PLAN_TASK_ID 
     AND PLANTASKDETAIL.PROCESS_TECH_TASK_ID = FINISHEDVM.PROCESS_TECH_TASK_ID
    LEFT OUTER JOIN DWROTD.DWR_PS_RES_ENTERPRISE_F T0 
	  ON WIPRES.SRC_FACTORY_ID = T0.ID 
	 AND T0.IS_PHANTOM = '0'
   WHERE PLANTASKDETAIL.INV_ORG_ID = '1') T0
    LEFT OUTER JOIN DWROTD.DWR_PS_APS_PLAN_TASK_DTL_F T1
	  ON T0.PLAN_TASK_DETAIL_ID = T1.ID
    LEFT OUTER JOIN DWROTD.DWR_PS_APS_PROTECH_ORDER_F T3
	  ON T0.PROCESS_TECH_TASK_ID = T3.ID
