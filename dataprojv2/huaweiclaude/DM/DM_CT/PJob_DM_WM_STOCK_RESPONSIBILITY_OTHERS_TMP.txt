insert overwrite table DM.DM_WM_STOCK_RESPONSIBILITY_OTHERS_TMP
select a.material_num,
       a.batch_num,
       a.factory_code,
       a.inv_location ,
       b.matl_group2_code,
       D.PROMOTION_REASON_CODE,
       c.clearance_reason_code,
       b.matl_group3_code,
       c.batch_in_time,
       case when (b.matl_group2_code  in('DZ','GDZ')  or b.cstm_main_code in ('DZ','GDZ'))and a.material_num like '%_A' and 
                 (CASE WHEN A.batch_num  LIKE '%888888' or A.batch_num LIKE '%999999' THEN 0-- 虚拟入库批次不计算
                       WHEN A.factory_code LIKE '%80' THEN 1   -- 80全算
                       WHEN D.PROMOTION_REASON_CODE IN('AC01','AC02','AC03','AC04','AC05') THEN 1 ELSE 0
	              END) =0 then '公用定制品可用库存GDZ（非特惠品)'
	        when (CASE WHEN A.batch_num  LIKE '%888888' or A.batch_num LIKE '%999999' THEN 0 -- 虚拟入库批次不计算
                       WHEN A.factory_code LIKE '%80' THEN 1   -- 80全算
                       WHEN D.PROMOTION_REASON_CODE IN('AC01','AC02','AC03','AC04','AC05') THEN 1 ELSE 0
	              END) =1 then (case when a.factory_code LIKE '%80' and c.clearance_reason_code='QC19' then  '剩余特惠品-清仓小堆头' 
	                                else concat('剩余特惠品-',d.rejects_type) end)
	        when b.matl_group3_code in('YTT') then '预淘汰'
	        when c.batch_in_time>= DATE_SUB(CURRENT_date,181) then '新品可用库存（上市半年内）'
	        else '主菜单可用库存（非新品）' end as item_type, 
       left_zkkc as  stock_qty,
	   current_timestamp() as dw_last_update_date
from  dm.dm_wm_stock_responsibility_left_2_tmp a  
join  dm.dm_wm_stock_responsibility_mat_tmp b 
ON a.material_num  = b.material_num 
left join dwimd.dwi_md_matl_batch_info_t  c 
On a.material_num = c.material_num  and a.batch_num  = c.batch_num 
left join sdi.sdi_rejects_definition_1037 d 
ON c.promote_reason_code = d.promotion_reason_code
where left_zkkc >0;
