insert OVERWRITE table   DM.DM_SYSTEM_ROLE_MONTHLY_D    
select 
p_system_flag,
period_id_m,
count(distinct user_id) as login_lng,
role_name,
current_timestamp
from 
 dwrdim.dwr_system_role_capacity_d
 where datediff('${PERIOD_ID_D}' ,substring(login_date,1,10))<=365
  and is_froze = 'N'
group by 
p_system_flag,
period_id_m,
role_name
