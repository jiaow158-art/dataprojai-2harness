INSERT OVERWRITE TABLE DM.DM_SYSTEM_ROLE_ALL_D
--正式
select
system_flag,
count(distinct user_id) as login_lng,
role_name,
current_timestamp
from 
 dwrdim.dwr_system_role_capacity_d
 where datediff('${PERIOD_ID_D}' ,substring(login_date,1,10))<=365
 and is_froze = 'N'
 group by 
 system_flag,
role_name
