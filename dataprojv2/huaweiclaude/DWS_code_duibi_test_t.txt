with zjl_base as (
select 
        tj_month --统计月份
       ,vtweg 
       ,case when sale_region_name like '%子公司' then sales_office_name else sale_region_name end as department 
       ,werks
       ,jhddlx
       ,coalesce(new_material_flag,'NAL') as new_material_flag
       ,coalesce(grelight_material_flag,'NAL') as grelight_material_flag
       ,count(distinct case when SFZDJF = 'Y' then VBELN
            else concat(VBELN,POSNR)
             end) order_cnt
       ,count(distinct case when is_zdjfzj='是'  then case when SFZDJF = 'Y' then VBELN
            else concat(VBELN,POSNR)
             end  end) zj_cnt
       ,sum(zj_qty*unit_area) as zj_area
       ,sum(kwmeng*unit_area) as order_area
from dm.DM_SO_FZL_ORDER_ZJ_PRE_T a
 where is_zdjfzj in('是','否')
 and bstdk <= replace(CURRENT_DATE + INTERVAL '-1 day','-','') 
group by tj_month
	,vtweg
	,case when sale_region_name like '%子公司' then sales_office_name else sale_region_name end
	,werks
	,jhddlx
 	,coalesce(new_material_flag,'NAL')
    ,coalesce(grelight_material_flag,'NAL')
union all

select 
       tj_month
      ,'06' vtweg
     ,department
     ,werks
     ,ddzt as jhddlx
     ,coalesce(new_material_flag,'NAL') as new_material_flag
     ,coalesce(grelight_material_flag,'NAL') as grelight_material_flag
     ,count(distinct case when SFZDJF = '是' then zjhd
            else zjhdid
             end)
      ,count(distinct case when is_zdjfzj='是' then case when SFZDJF = '是' then zjhd
            else zjhdid
             end end ) 
      ,sum(zj_qty*unit_area) as zj_area
      ,sum(menge *unit_area) as order_area
from dm.DM_SO_ZL_ORDER_ZJ_PRE_T
 where is_zdjfzj in('是','否')
 and bstdk <= replace(CURRENT_DATE + INTERVAL '-1 day','-','') 
 group by tj_month
     ,department
     ,werks
  	 ,ddzt
  	 ,coalesce(new_material_flag,'NAL')
     ,coalesce(grelight_material_flag,'NAL')
)
,ztl_base as (
select
       tj_month --统计月份
      ,vtweg
      ,case when sale_region_name like '%子公司' then sales_office_name else sale_region_name end as department 
      ,werks
      ,jhddlx
      ,coalesce(new_material_flag,'NAL') as new_material_flag
      ,coalesce(grelight_material_flag,'NAL') as grelight_material_flag
      ,count(distinct  case when SFZDJF = 'Y' then VBELN
            else concat(VBELN,POSNR)
             end   ) order_cnt
      ,count(distinct case when is_zdjfzt='是' then case when SFZDJF = 'Y' then VBELN
            else concat(VBELN,POSNR)
             end  end ) zt_cnt
      ,sum(zt_qty*unit_area) as zt_area
      ,sum(kwmeng*unit_area) as order_area
from dm.DM_SO_FZL_ORDER_PRE_T a
where is_zdjfzt in('是','否')
--and yt_date <= replace(CURRENT_DATE + INTERVAL '-1 day','-','') 
group by tj_month --统计月份
      ,vtweg
      ,case when sale_region_name like '%子公司' then sales_office_name else sale_region_name end
      ,werks
      ,jhddlx
      ,coalesce(new_material_flag,'NAL')
      ,coalesce(grelight_material_flag,'NAL')
union all

select 
       tj_month
      ,'06' as vtweg
      ,department 
      ,werks
      ,ddzt as jhddlx
      ,coalesce(new_material_flag,'NAL') as new_material_flag
      ,coalesce(grelight_material_flag,'NAL') as grelight_material_flag
      ,count(distinct case when SFZDJF = '是' then zjhd
            else zjhdid
             end )
      ,count(distinct case when  is_zdjfzt='是' then case when SFZDJF = '是' then zjhd
            else zjhdid
             end end )
      ,sum(zt_qty*unit_area) as zt_area
      ,sum(menge *unit_area) as order_area
from dm.DM_SO_ZL_ORDER_PRE_T
where is_zdjfzt in('是','否')
group by
       tj_month
      ,department 
      ,werks
      ,ddzt
      ,coalesce(new_material_flag,'NAL')
      ,coalesce(grelight_material_flag,'NAL')
),
mid as(
select tj_month,vtweg,
		case when department = '战略工程-战略二部' then '工程市场中心' else department end department,
		werks,jhddlx,new_material_flag,grelight_material_flag,
        order_cnt as  order_cnt_1,zj_cnt,zj_area,order_area  as  order_area_1,
       0 as order_cnt_2,0 as zt_cnt,0 as zt_area, 0 as order_area_2
  from zjl_base
union all 
select tj_month,vtweg,
		case when department = '战略工程-战略二部' then '工程市场中心' else department end department,
		werks,jhddlx,new_material_flag,grelight_material_flag,
       0 ,0,0,0,
       order_cnt as order_cnt_2,zt_cnt,zt_area ,order_area as order_area_2 from ztl_base
)
select tj_month,vtweg,department,werks,jhddlx,new_material_flag,grelight_material_flag,
       sum(order_cnt_1) as order_cnt_1,
       sum(zj_cnt) as zj_cnt,
       sum(zj_area) as zj_area,
       sum(order_area_1) as order_area_1,
       sum(order_cnt_2) as order_cnt_2,
       sum(zt_cnt) as zt_cnt,
       sum(zt_area) as zt_area,
       sum(order_area_2) as order_area_2
from mid 
where tj_month >='${start_month}' and tj_month<='${end_month}'
${if(len(dis_channel) == 0 ,"","AND (vtweg IN ('"+ dis_channel +"' ) )")}
${if(len(werks) == 0 ,"","AND (werks IN ('"+ werks +"') ) ")}
--and (vtweg = '${dis_channel}'  or '${dis_channel}' is null)
--and (werks = '${werks}'  or '${werks}' is null)
${if(len(department) == 0 ,"","AND department IN ('"+ department +"' )")}
${if(len(jhddlx) == 0 ,"","AND jhddlx IN ('"+ jhddlx +"' )")}
${if(len(新品分类) == 0 ,"","AND new_material_flag IN ('"+ 新品分类 +"' )")}
${if(len(绿灯产品分类) == 0 ,"","AND grelight_material_flag IN ('"+ 绿灯产品分类 +"' )")}
group by tj_month,vtweg,department,werks,jhddlx,new_material_flag,grelight_material_flag
order by tj_month
