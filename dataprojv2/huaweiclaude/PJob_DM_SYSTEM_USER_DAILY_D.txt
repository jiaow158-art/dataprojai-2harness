INSERT OVERWRITE TABLE DM.DM_SYSTEM_USER_DAILY_D PARTITION(period_id_m)
select 
T1.system_flag,
period_id_d ,
daily_active_users,--(有效用户数)
T2.daily_active_users_all ,--总用户数
current_timestamp as  dw_last_update_date  ,
login_lng as login_date_cnt,--用户访问量
T2.daily_active_users_all - T2.daily_active_users_anomalous  AS daily_active_users_kt, --（开通用户数）
daily_active_users_login,--当日登录人数
daily_login_lng,--当日登录次数
role_name,
substr(period_id_d,1,7) as period_id_m
from  dwrltc.dwr_system_user_capacity_daily_d  T1
LEFT JOIN DWRBIT.DWR_SYSTEM_USER_SUMMARY_F T2
ON T1.system_flag = T2.system_flag
where  substr(period_id_d,1,7) = '${PERIOD_ID_M}'
