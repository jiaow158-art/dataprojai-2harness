INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_PURCHASE_ORG_T	--DWI.SAP采购组织数据表
SELECT
		SE1.GROUP_NUM			AS		GROUP_NUM,
		SE1.PURCHASE_INFO_RECORD_NO           AS		PURCHASE_INFO_RECORD_NO,
		SE1.PURCHASE_ORG           AS		PURCHASE_ORG,
		SE1.PURCHASE_INFO_RECORD_CODE           AS		PURCHASE_INFO_RECORD_CODE,
		CASE WHEN SE1.FACTORY_CODE IS NULL OR SE1.FACTORY_CODE = ''	--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)	--关联不上主数据时赋值-999998
        END 			AS      FACTORY_ID, 	--工厂ID
		CASE	
		WHEN SE1.FACTORY_CODE IS NULL OR SE1.FACTORY_CODE = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(SE1.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 			AS 	FACTORY_CODE,	--工厂编码
		SE1.CURRENCY_CODE           AS		CURRENCY_CODE,
		CASE WHEN SE1.RECORD_CREATE_DATE = '00000000' 	--判断SDI源表字段记录建立日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SE1.RECORD_CREATE_DATE, 1, 4) = '9999' --SDI源表字段记录建立日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.RECORD_CREATE_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS 		RECORD_CREATE_DATE,
		SE1.PURCHASE_VCH_ITEM_NUM           AS		PURCHASE_VCH_ITEM_NUM,
		SE1.CREATE_BY           AS		CREATE_BY,
		SE1.PURCHASE_VCH_TYPE           AS		PURCHASE_VCH_TYPE,
		SE1.RECENTLY_PO_DATE           AS		RECENTLY_PO_DATE,
		SE1.PURCHASE_ORG_INFO_FLAG           AS		PURCHASE_ORG_INFO_FLAG,
		SE1.REVALUED_RECEIPT_FLAG           AS		REVALUED_RECEIPT_FLAG,
		CASE WHEN SE1.PRICE_VALID_DATE = '00000000' 	--判断SDI源表字段价格有效至为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SE1.PRICE_VALID_DATE, 1, 4) = '9999' --SDI源表字段价格有效至为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.PRICE_VALID_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS 		PRICE_VALID_DATE,
		SE1.PRICE_UNIT           AS		PRICE_UNIT,
		SE1.PURCHASE_TAX_CODE           AS		PURCHASE_TAX_CODE,
		SE1.PURCHASE_INFO_NET_PRI           AS		PURCHASE_INFO_NET_PRI,
		SE1.ASSESSMENT_TYPE           AS		ASSESSMENT_TYPE,
		SE1.PURCHASE_VOUCHER_NUM           AS		PURCHASE_VOUCHER_NUM,
		SE1.PURCHASE_GROUP           AS		PURCHASE_GROUP,
		SE1.PLAN_DELIVERY_TIME_DAYS           AS		PLAN_DELIVERY_TIME_DAYS,
		SE1.PO_PRICE_UNIT           AS		PO_PRICE_UNIT,
		SE1.PURCHASE_INFO_VALID_PRICE           AS		PURCHASE_INFO_VALID_PRICE,
		SE1.RECEIVING_BILL_FLAG           AS		RECEIVING_BILL_FLAG,
		SE1.OVERDELIVERY_FLAG           AS		OVERDELIVERY_FLAG,
		SE1.STANDARD_PO_QTY           AS		STANDARD_PO_QTY,
		SE1.OVERDELIVERY_LIMIT           AS		OVERDELIVERY_LIMIT,
		SE1.SHORTEST_GOODS_SHELVES_QTY           AS		SHORTEST_GOODS_SHELVES_QTY,
		SE1.DISCOUNT_NO_ITEM_FLAG           AS		DISCOUNT_NO_ITEM_FLAG,
		SE1.MATERIAL_PO_FLAG           AS		MATERIAL_PO_FLAG,
		SE1.TRANS_PKG_COMPONENT_FLAG        AS		TRANS_PKG_COMPONENT_FLAG,
		SE1.RFQ_ITEM_NO           AS		RFQ_ITEM_NO,
		SE1.ORDER_UNIT_DENOMINATOR           AS		ORDER_UNIT_DENOMINATOR,
		SE1.ORDER_UNIT_NUMERATOR           AS		ORDER_UNIT_NUMERATOR,
		CASE WHEN SE1.QUOTE_VALID_DATE = '00000000' --判断SDI源表字段报价有效日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SE1.QUOTE_VALID_DATE, 1, 4) = '9999' --SDI源表字段记录建立日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.QUOTE_VALID_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS 		QUOTE_VALID_DATE,
		SE1.MAX_PO_QTY           AS		MAX_PO_QTY,
		SE1.MIN_PO_QTY           AS		MIN_PO_QTY,
		SE1.STOCKUP_TIME_DAYS    AS		STOCKUP_TIME_DAYS,
		SE1.DEL_FLAG			 	AS 		DEL_FLAG,
		CURRENT_TIMESTAMP()  AS   DW_LAST_UPDATE_DATE
		FROM SDI_AND_Y_FLAG_DATA SE1
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF	--关联工厂主数据取工厂ID
			 ON  SE1.GROUP_NUM = DMF.GROUP_NUM	--集团
			 AND SE1.FACTORY_CODE = DMF.FACTORY_CODE	--工厂编码
			 AND DMF.END_DATE = '4712-12-31'	--结束时间为'4712-12-31'
