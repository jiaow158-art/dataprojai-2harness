set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_PURCHASE_VCH_HIS_T	--DWI.SAP采购凭证历史表
 SELECT
		SE1.MANDT				AS		GROUP_NUM,
		SE1.EBELN               AS		PURCHASE_VOUCHER_NUM,
		SE1.EBELP               AS		PURCHASE_VCH_ITEM_NUM,
		SE1.ZEKKN               AS		PURCHASE_VCH_ACC_SERIALNO,
		SE1.VGABE               AS		PURCHASE_ORDER_HIS_RECORD,
		SE1.GJAHR               AS		MATERIAL_VCH_YEARS,
		SE1.BELNR               AS		MATERIAL_VCH_CODE,
		SE1.BUZEI               AS		MATERIAL_VCH_ITEM_CODE,
		SE1.WRBTR               AS		VCH_CURRENCY_AMT,
		SE1.WAERS               AS		CURRENCY_CODE,
		SE1.BAMNG               AS		QUANTITY1,
		SE1.MENGE               AS		QUANTITY2,
		SE1.VBELP_ST            AS		DELIVERY_ITEM,
		SE1.ERNAM               AS		CREATE_BY,
		CASE WHEN SE1.MATNR IS NULL OR SE1.MATNR = ''		--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)		--关联不上主数据时赋值-999998
        END 				AS 		MATERIAL_ID,	--物料ID
		CASE WHEN SE1.MATNR IS NULL OR SE1.MATNR = ''		--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')			--关联不上主数据时赋值'NOT EXISTS'
		END 				AS		MATERIAL_NUM,	--物料编号
		CASE WHEN SE1.WERKS IS NULL OR SE1.WERKS = ''		--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)		--关联不上主数据时赋值-999998
        END 				AS 		FACTORY_ID,		--工厂ID
		CASE WHEN SE1.WERKS IS NULL OR SE1.WERKS = ''		--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')		--关联不上主数据时赋值'NOT EXISTS'
		END 				AS		FACTORY_CODE,		--工厂编码
		SE1.CPUTM               AS		INPUT_TIME,
		SE1.SAPRL               AS		SAP_VERSION,
		SE1.LFPOS               AS		REFERENCE_VCH_ITEM,
		SE1.GRUND               AS		MOVE_REASON,
		CASE WHEN SE1.BLDAT = '00000000' 	--判断SDI源表字段凭证中的凭证日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.BLDAT, 1, 4) = '9999' --SDI源表字段凭证中的凭证日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.BLDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 				AS VOUCHER_DATE,
		SE1.HSWAE               AS		BASIC_CURRENCY_CODE,
		CASE WHEN SE1.BUDAT = '00000000' 	--判断SDI源表字段凭证中的过帐日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.BUDAT, 1, 4) = '9999' --SDI源表字段凭证中的凭证日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.BUDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 				AS VOUCHER_POST_DATE,
		SE1.LFBNR               AS		REFERENCE_VOUCHER_NUM1,
		SE1.VBELN_ST            AS		DELIVERY_CODE,
		SE1.XBLNR               AS		REFERENCE_VOUCHER_NUM2,
		SE1.MWSKZ               AS		PURCHASE_TAX_CODE,
		SE1.CHARG               AS		BATCH_NUM,
		CASE WHEN SE1.CPUDT = '00000000' 	--判断SDI源表字段会计凭证输入日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.CPUDT, 1, 4) = '9999' 	--SDI源表字段会计凭证输入日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.CPUDT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 				AS ACCOUNT_VOUCHER_INPUT_DATE,
		SE1.BWTAR               AS		ASSESSMENT_TYPE,
		SE1.REEWR               AS		INPUT_INVOICE_VALUE,
		SE1.ELIKZ               AS		DELIVERY_FLAG,
		SE1.BWART               AS		MOVE_TYPE,
		SE1.DMBTR               AS		BASIC_CURRENCY_AMT,
		SE1.REWRB               AS		PO_CURRENCY_INVOICE_AMT,
		SE1.AREWB               AS		PO_GR_IR_AMT,
		SE1.AREWW               AS		GR_IR_BUSINESS_AMT,
		SE1.REFWR               AS		INVOICE_VALUE_AMT,
		T.DOMAIN_VALUE_CODE AS		DEBIT_CREDIT_FLAG,
		SE1.AREWR               AS		GR_IR_VALUE_AMT1,
		SE1.AREWR_POP           AS		GR_IR_VALUE_AMT2,
		SE1.LSMNG               AS		DELIVERY_UNIT_QTY,
		SE1.BPMNG               AS		PO_PRICE_UNIT_QTY,
		SE1.LSMEH               AS		DELIVERY_UNIT,
		SE1.BEWTP               AS		PO_HIS_RECORD_TYPE,
		SE1.LFGJA               AS		REF_VOUCHER_FISCAL_YEAR,
		SE1.ETENS               AS		VENDOR_CONFIG_SEQ,
		SE1.WKURS               AS		EXCHANGE_RATE,
		SE1.PACKNO              AS		SERVICE_PACKAGE_CODE,
		SE1.INTROW              AS		SERVICE_ITEM_NO,
		SE1.BEKKN               AS		PO_ASSIGN_CODE,
		'N'				 	AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() AS      DW_LAST_UPDATE_DATE
		FROM SDI.SDI_EKBE_1001 SE1				--SDI.SAP采购凭证历史表
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG		--关联物料主数据取物料ID
			 ON  SE1.MANDT = DMMG.GROUP_NUM					--集团
			 AND SE1.MATNR = DMMG.MATERIAL_NUM				--物料编号
			 AND DMMG.END_DATE = '4712-12-31'				--结束日期为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF				--关联工厂主数据取工厂ID
			 ON  SE1.MANDT = DMF.GROUP_NUM					--集团
			 AND SE1.WERKS = DMF.FACTORY_CODE				--工厂编码
			 AND DMF.END_DATE = '4712-12-31'				--结束日期为'4712-12-31'
		LEFT JOIN  DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R T		--关联DD07T配置表
			 ON  T.DOMAIN_NAME  = 'SHKZG'					--取借方/贷方标识中文名称
			 AND T.SRC_SYS_DOMAIN_VALUE_CODE = SE1.SHKZG
		UNION ALL				--SDI_EKBE_1001.SAP采购凭证历史表 UNION ALL 拼接 DWI_PTP_PURCHASE_VCH_HIS_T.SAP采购凭证历史表
		SELECT
				 T.GROUP_NUM
				,T.PURCHASE_VOUCHER_NUM
				,T.PURCHASE_VCH_ITEM_NUM
				,T.PURCHASE_VCH_ACC_SERIALNO
				,T.PURCHASE_ORDER_HIS_RECORD
				,T.MATERIAL_VCH_YEARS
				,T.MATERIAL_VCH_CODE
				,T.MATERIAL_VCH_ITEM_CODE
				,T.VCH_CURRENCY_AMT
				,T.CURRENCY_CODE
				,T.QUANTITY1
				,T.QUANTITY2
				,T.DELIVERY_ITEM
				,T.CREATE_BY
				,T.MATERIAL_ID
				,T.MATERIAL_NUM
				,T.FACTORY_ID
				,T.FACTORY_CODE
				,T.INPUT_TIME
				,T.SAP_VERSION
				,T.REFERENCE_VCH_ITEM
				,T.MOVE_REASON
				,T.VOUCHER_DATE
				,T.BASIC_CURRENCY_CODE
				,T.VOUCHER_POST_DATE
				,T.REFERENCE_VOUCHER_NUM1
				,T.DELIVERY_CODE
				,T.REFERENCE_VOUCHER_NUM2
				,T.PURCHASE_TAX_CODE
				,T.BATCH_NUM
				,T.ACCOUNT_VOUCHER_INPUT_DATE
				,T.ASSESSMENT_TYPE
				,T.INPUT_INVOICE_VALUE
				,T.DELIVERY_FLAG
				,T.MOVE_TYPE
				,T.BASIC_CURRENCY_AMT
				,T.PO_CURRENCY_INVOICE_AMT
				,T.PO_GR_IR_AMT
				,T.GR_IR_BUSINESS_AMT
				,T.INVOICE_VALUE_AMT
				,T.DEBIT_CREDIT_FLAG
				,T.GR_IR_VALUE_AMT1
				,T.GR_IR_VALUE_AMT2
				,T.DELIVERY_UNIT_QTY
				,T.PO_PRICE_UNIT_QTY
				,T.DELIVERY_UNIT
				,T.PO_HIS_RECORD_TYPE
				,T.REF_VOUCHER_FISCAL_YEAR
				,T.VENDOR_CONFIG_SEQ
				,T.EXCHANGE_RATE
				,T.SERVICE_PACKAGE_CODE
				,T.SERVICE_ITEM_NO
				,T.PO_ASSIGN_CODE
				,'Y' AS DEL_FLAG
				,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
				FROM DWIPUR.DWI_PTP_PURCHASE_VCH_HIS_T T		--DWI.SAP采购凭证历史表
				LEFT JOIN SDI.SDI_EKBE_1001 T1				--SDI.SAP采购凭证历史表
				ON T.GROUP_NUM = T1.MANDT					--集团
				AND T.PURCHASE_VOUCHER_NUM = T1.EBELN		--采购凭证编号
				AND T.PURCHASE_VCH_ITEM_NUM = T1.EBELP		--采购凭证的项目编号
				AND T.PURCHASE_VCH_ACC_SERIALNO = T1.ZEKKN	--帐户分配的顺序编号
				AND T.PURCHASE_ORDER_HIS_RECORD = T1.VGABE	--业务处理/事件类型,采购订单的历史记录
				AND T.MATERIAL_VCH_YEARS = T1.GJAHR		--物料凭证的年份
				AND T.MATERIAL_VCH_CODE = T1.BELNR		--物料凭证编号
				AND T.MATERIAL_VCH_ITEM_CODE = T1.BUZEI		--物料凭证中的项目
				WHERE T1.MANDT  IS NULL
				AND T1.EBELN  IS NULL
				AND T1.EBELP  IS NULL
				AND T1.ZEKKN  IS NULL
				AND T1.VGABE  IS NULL
				AND T1.GJAHR  IS NULL
				AND T1.BELNR  IS NULL
				AND T1.BUZEI  IS NULL
				AND T.DEL_FLAG = 'N'
