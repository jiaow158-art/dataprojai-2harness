set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_CONDITION_HEADER_T	--DWI.SAP条件抬头表
SELECT 
		SK1.MANDT			AS		GROUP_NUM,
		SK1.KNUMH           AS		CONDITION_RECORD_NO,
		CASE WHEN SK1.ERDAT = '00000000' 	--判断SDI源表字段记录建立日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SK1.ERDAT, 1, 4) = '9999' --SDI源表字段记录建立日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SK1.ERDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS RECORD_CREATE_DATE,
		SK1.KAPPL           AS		APPLICATION_PROGRAM_CODE,
		SK1.ERNAM           AS		CREATE_BY,
		CASE WHEN SK1.DATAB = '00000000' 	--判断SDI源表字段开始生效日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SK1.DATAB, 1, 4) = '9999' 	--SDI源表字段开始生效日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SK1.DATAB, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS START_DATE,
		SK1.KOTABNR         AS		CONDITION_TABLE,
		CASE WHEN SK1.DATBI = '00000000' 	--判断SDI源表字段有效截至日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SK1.DATBI, 1, 4) = '9999' --SDI源表字段有效截至日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SK1.DATBI, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 			AS VALID_EXPIRY_DATE,
		NULL   AS		CONDITION_TYPE,
		SK1.REFERENCE       AS		REFER,
		SK1.KVEWE           AS		CONDITION_PURPOSE,
		CASE WHEN SK1.LICDT = '00000000' 	--判断SDI源表字段允许的许可日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SK1.LICDT, 1, 4) = '9999' --SDI源表字段允许的许可日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SK1.LICDT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END           AS		ALLOW_PERMIT_DATE,
		'N'			 	AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() 					 AS      DW_LAST_UPDATE_DATE
		FROM SDI.SDI_KONH_1001 SK1		--SDI.SAP条件抬头表
		UNION ALL			--SDI_KONH_1001.SAP条件抬头表 UNION ALL 拼接 DWI_PTP_CONDITION_HEADER_T.SAP条件抬头表
		SELECT
				 T.GROUP_NUM
				,T.CONDITION_RECORD_NO
				,T.RECORD_CREATE_DATE
				,T.APPLICATION_PROGRAM_CODE
				,T.CREATE_BY
				,T.START_DATE
				,T.CONDITION_TABLE
				,T.VALID_EXPIRY_DATE
				,T.CONDITION_TYPE
				,T.REFER
				,T.CONDITION_PURPOSE
				,T.ALLOW_PERMIT_DATE
				,'Y' AS DEL_FLAG
				,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
				FROM DWIPUR.DWI_PTP_CONDITION_HEADER_T T	--DWI.SAP条件抬头表
				LEFT JOIN SDI.SDI_KONH_1001 T1		--SDI.SAP条件抬头表
				ON T.GROUP_NUM = T1.MANDT		--集团
				AND T.CONDITION_RECORD_NO = T1.KNUMH	--条件记录号
				WHERE T1.MANDT  IS NULL
				AND T1.KNUMH IS NULL
				AND T.DEL_FLAG = 'N'
