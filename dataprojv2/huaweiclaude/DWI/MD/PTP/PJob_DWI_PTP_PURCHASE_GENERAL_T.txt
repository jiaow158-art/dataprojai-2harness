set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_PURCHASE_GENERAL_T	--DWI.SAP采购一般数据表
SELECT
		SE1.MANDT					AS		GROUP_NUM,
		SE1.INFNR                   AS      PURCHASE_INFO_RECORD_NO,
		CASE WHEN SE1.ERDAT = '00000000' 	--判断SDI源表字段记录建立日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.ERDAT, 1, 4) = '9999' 	--SDI源表字段记录建立日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.ERDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS RECORD_CREATE_DATE,
		SE1.ERNAM                   AS      CREATE_BY,
		SE1.LASTCHANGEDATETIME      AS      LONG_UTC_TIMESTAMP,
		CASE WHEN SE1.MATNR IS NULL OR SE1.MATNR = ''	--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)		--关联不上主数据时赋值-999998
        END 					AS 		MATERIAL_ID,	--物料ID
		CASE WHEN SE1.MATNR IS NULL OR SE1.MATNR = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 					AS      MATERIAL_NUM,	--物料编号
		CASE WHEN SE1.LIFNR IS NULL OR SE1.LIFNR = ''	--关联供应商主数据取供应商ID,判断SDI源表字段供应商帐户号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMSG.SUPPLIER_OR_CREDITOR_ID, -999998)	--关联不上主数据时赋值-999998
        END 					AS 		SUPPLIER_OR_CREDITOR_ID,	--供应商ID
		CASE WHEN SE1.LIFNR IS NULL OR SE1.LIFNR = ''	--关联供应商主数据,判断SDI源表字段供应商帐户号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMSG.SUPPLIER_OR_CREDITOR_NUM, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 					AS   SUPPLIER_OR_CREDITOR_NUM,	--供应商帐户号
		SE1.REGIO                   AS      REGION_CODE,
		SE1.LMEIN                   AS      BASE_UNIT,
		SE1.TXZ01                   AS      PURCHASE_INFO,
		SE1.LOEKZ                   AS      PURCHASE_INFO_FLAG,
		SE1.UMREN                   AS      BASE_UNTI_DENOMINATOR,
		SE1.TELF1                   AS      VENDOR_PHONE_NUMBER,
		T.CONFIG_CODE           AS      MATERIAL_GROUP,
		SE1.MEINS                   AS      PURCHASE_ORDER_UNIT,
		SE1.IDNLF                   AS      VENDOR_USE_MATERIAL_NUM,
		CASE WHEN SE1.LIFAB = '00000000' 	--判断SDI源表字段可用（可供货的）开始日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.LIFAB, 1, 4) = '9999' --SDI源表字段可用（可供货的）开始日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.LIFAB, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS START_DATE,
		SE1.SORTL                   AS      NO_STOCK_INFO,
		SE1.VERKF                   AS      SALES_USER,
		SE1.UMREZ                   AS      BASE_UNTI_NUMERATOR,
		SE1.MAHN1                   AS      REMINDER_DAY1,
		CASE WHEN SE1.LIFBI = '00000000' 	--判断SDI源表字段可用(可供货的)截至日期为00000000
	     THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
	     WHEN SUBSTR(SE1.LIFBI, 1, 4) = '9999' --SDI源表字段可用(可供货的)截至日期为9999
		 THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
	     ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SE1.LIFBI, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 					AS END_DATE,
		SE1.URZLA                   AS      DELIVERY_COUNTRY_CODE,
		SE1.LTSSF                   AS      SORT_SEQ_NUM,
		'N'						AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE
		FROM SDI.SDI_EINA_1001 SE1		--SDI.SAP采购一般数据表
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据取物料ID
			 ON  SE1.MANDT = DMMG.GROUP_NUM				--集团
			 AND SE1.MATNR = DMMG.MATERIAL_NUM			--物料编号
			 AND DMMG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T DMSG	--关联供应商主数据取供应商ID
			 ON  SE1.MANDT = DMSG.GROUP_NUM				--集团
			 AND SE1.LIFNR = DMSG.SUPPLIER_OR_CREDITOR_NUM	--供应商帐户号
			 AND DMSG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T	--关联通用配置表
			 ON T.SRC_SYS_CONFIG_CODE = SE1.MATKL		--取物料组描述中文名称
			 AND T.GROUP_NUM = SE1.MANDT				--集团
			 AND T.SOURCE_CODE = 'T023T'
		UNION ALL 				--SDI_EINA_1001.SAP采购一般数据表 UNION ALL 拼接 DWI_PTP_PURCHASE_GENERAL_T.SAP采购一般数据表
		SELECT
				 T.GROUP_NUM
				,T.PURCHASE_INFO_RECORD_NO
				,T.RECORD_CREATE_DATE
				,T.CREATE_BY
				,T.LONG_UTC_TIMESTAMP
				,T.MATERIAL_ID
				,T.MATERIAL_NUM
				,T.SUPPLIER_OR_CREDITOR_ID
				,T.SUPPLIER_OR_CREDITOR_NUM
				,T.REGION_CODE
				,T.BASE_UNIT
				,T.PURCHASE_INFO
				,T.PURCHASE_INFO_FLAG
				,T.BASE_UNTI_DENOMINATOR
				,T.VENDOR_PHONE_NUMBER
				,T.MATERIAL_GROUP
				,T.PURCHASE_ORDER_UNIT
				,T.VENDOR_USE_MATERIAL_NUM
				,T.START_DATE
				,T.NO_STOCK_INFO
				,T.SALES_USER
				,T.BASE_UNTI_NUMERATOR
				,T.REMINDER_DAY1
				,T.END_DATE
				,T.DELIVERY_COUNTRY_CODE
				,T.SORT_SEQ_NUM
				,'Y' AS DEL_FLAG
				,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
				FROM DWIPUR.DWI_PTP_PURCHASE_GENERAL_T T	--DWI.SAP采购一般数据表
				LEFT JOIN SDI.SDI_EINA_1001 T1		--SDI.SAP采购一般数据表
				ON T.GROUP_NUM = T1.MANDT 			--集团
				AND T.PURCHASE_INFO_RECORD_NO = T1.INFNR	--采购信息记录号
				WHERE T1.MANDT  IS NULL
				AND T1.INFNR IS NULL
				AND T.DEL_FLAG = 'N'
