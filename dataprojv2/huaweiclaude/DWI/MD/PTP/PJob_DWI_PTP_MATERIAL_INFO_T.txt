WITH Y_FALG_DATA AS(  -- 源表数据删了，打Y标识的数据
	SELECT
	DPMI.GROUP_NUM,
	DPMI.APPLICATION_PROGRAM_CODE,
	DPMI.CONDITION_TYPE,
	DPMI.VENDOR_ACCOUNT,
	DPMI.MATERIAL_NUM,
	DPMI.PURCHASE_ORG,
	DPMI.FACTORY_CODE,
	DPMI.PURCHASE_INFO_RECORD,
	DPMI.CONDITION_END_DATE,
	DPMI.CONDITION_START_DATE,
	DPMI.CONDITION_RECORD_NO,
	'Y' AS DEL_FLAG
FROM DWIPUR.DWI_PTP_MATERIAL_INFO_T DPMI	--DWI.SAP物料信息记录表
LEFT JOIN SDI.SDI_A017_1001 SA1		--SDI.SAP物料信息记录表
	ON DPMI.GROUP_NUM = SA1.MANDT	--集团
	AND DPMI.APPLICATION_PROGRAM_CODE = SA1.KAPPL	--应用程序
	AND DPMI.CONDITION_TYPE = SA1.KSCHL		--条件类型
	AND DPMI.VENDOR_ACCOUNT = SA1.LIFNR		--供应商帐户号
	AND DPMI.MATERIAL_NUM = SA1.MATNR	--物料编号
	AND DPMI.PURCHASE_ORG = SA1.EKORG	--采购组织
	AND DPMI.FACTORY_CODE = SA1.WERKS	--工厂编码
	AND DPMI.PURCHASE_INFO_RECORD = SA1.ESOKZ	--采购信息记录分类
	AND DPMI.CONDITION_END_DATE = SA1.DATBI	--条件记录有效截止日期
	WHERE SA1.MANDT  IS NULL
	AND SA1.KAPPL IS NULL
	AND SA1.KSCHL IS NULL
	AND SA1.LIFNR IS NULL
	AND SA1.MATNR IS NULL
	AND SA1.EKORG IS NULL
	AND SA1.WERKS IS NULL
	AND SA1.ESOKZ IS NULL
	AND SA1.DATBI IS NULL
	AND SA1.DEL_FLAG = 'N'),
SDI_AND_Y_FLAG_DATA AS( -- SDI与打Y标识数据
   SELECT
	SA1.MANDT	AS	GROUP_NUM,
	SA1.KAPPL	AS	APPLICATION_PROGRAM_CODE,
	SA1.KSCHL	AS	CONDITION_TYPE,
	SA1.LIFNR	AS	VENDOR_ACCOUNT,
	SA1.MATNR	AS	MATERIAL_NUM,
	SA1.EKORG	AS	PURCHASE_ORG,
	SA1.WERKS	AS	FACTORY_CODE,
	SA1.ESOKZ	AS	PURCHASE_INFO_RECORD,
	SA1.DATBI	AS	CONDITION_END_DATE,
	SA1.DATAB	AS	CONDITION_START_DATE,
	SA1.KNUMH	AS	CONDITION_RECORD_NO,
	SA1.DEL_FLAG AS DEL_FLAG
FROM SDI.SDI_A017_1001 SA1	--SDI.SAP物料信息记录表
UNION ALL
SELECT
	DPMI.GROUP_NUM,
	DPMI.APPLICATION_PROGRAM_CODE,
	DPMI.CONDITION_TYPE,
	DPMI.VENDOR_ACCOUNT,
	DPMI.MATERIAL_NUM,
	DPMI.PURCHASE_ORG,
	DPMI.FACTORY_CODE,
	DPMI.PURCHASE_INFO_RECORD,
	DPMI.CONDITION_END_DATE,
	DPMI.CONDITION_START_DATE,
	DPMI.CONDITION_RECORD_NO,
	DPMI.DEL_FLAG
FROM Y_FALG_DATA DPMI)
INSERT OVERWRITE TABLE DWIPUR.DWI_PTP_MATERIAL_INFO_T	--DWI.SAP物料信息记录表
SELECT      --- 关联主数据表
		SA1.GROUP_NUM		AS		GROUP_NUM,
		SA1.APPLICATION_PROGRAM_CODE       AS		APPLICATION_PROGRAM_CODE,
		T.CONFIG_CODE   AS		CONDITION_TYPE,
		SA1.VENDOR_ACCOUNT       AS		VENDOR_ACCOUNT,
		CASE WHEN SA1.MATERIAL_NUM IS NULL OR SA1.MATERIAL_NUM = ''	--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)	--关联不上主数据时赋值-999998
        END 		AS  	MATERIAL_ID, 	--物料ID
		CASE
		WHEN SA1.MATERIAL_NUM IS NULL OR SA1.MATERIAL_NUM = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')		--关联不上主数据时赋值'NOT EXISTS'
		END         AS		MATERIAL_NUM,	--物料编号
		SA1.PURCHASE_ORG       AS		PURCHASE_ORG,
		CASE WHEN SA1.FACTORY_CODE IS NULL OR SA1.FACTORY_CODE = ''	--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)	--关联不上主数据时赋值-999998
        END 		AS   	FACTORY_ID, 	--工厂ID
		CASE
		WHEN SA1.FACTORY_CODE IS NULL OR SA1.FACTORY_CODE = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 		AS		FACTORY_CODE,	--工厂编码
		SA1.PURCHASE_INFO_RECORD       AS		PURCHASE_INFO_RECORD,
		CASE WHEN SA1.CONDITION_END_DATE = '00000000' --判断SDI源表字段条件记录有效截止日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SA1.CONDITION_END_DATE, 1, 4) = '9999' 	--SDI源表字段条件记录有效截止日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SA1.CONDITION_END_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END			AS CONDITION_END_DATE,
		CASE WHEN SA1.CONDITION_START_DATE = '00000000' 	--判断SDI源表字段条件记录有效起始日为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SA1.CONDITION_START_DATE, 1, 4) = '9999' --SDI源表字段条件记录有效起始日为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SA1.CONDITION_START_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 		AS CONDITION_START_DATE,
		SA1.CONDITION_RECORD_NO       AS		CONDITION_RECORD_NO,
		SA1.DEL_FLAG			AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE
		FROM SDI_AND_Y_FLAG_DATA SA1	--SDI.SAP物料信息记录表
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据取物料ID
			 ON  SA1.GROUP_NUM = DMMG.GROUP_NUM			--集团
			 AND SA1.MATERIAL_NUM = DMMG.MATERIAL_NUM	--物料编号
			 AND DMMG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF			--关联工厂主数据取工厂ID
			 ON  SA1.GROUP_NUM = DMF.GROUP_NUM			--集团
			 AND SA1.FACTORY_CODE = DMF.FACTORY_CODE	--工厂编码
			 AND DMF.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T	--关联通用配置表
			 ON T.SRC_SYS_CONFIG_CODE = SA1.CONDITION_TYPE	--取条件类型： 文本中文名称
			 AND T.GROUP_NUM = SA1.GROUP_NUM			--集团
			 AND T.SOURCE_CODE = 'T685T'
