INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVLST') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN LIFSP ='' THEN NULL 
         ELSE LIFSP 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN LIFSP ='' THEN NULL 
         ELSE LIFSP 
         END,'SNULL'),'TVLST') AS CONFIG_CODE,  
VTEXT AS CONFIG_NAME,
'TVLST' AS SOURCE_CODE,
'交货：冻结原因/范围：文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVLST_1001
