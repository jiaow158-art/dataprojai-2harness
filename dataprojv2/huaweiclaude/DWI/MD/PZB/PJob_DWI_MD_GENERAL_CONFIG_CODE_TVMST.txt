INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TVMST') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN VMSTA ='' THEN NULL 
         ELSE VMSTA 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN VMSTA ='' THEN NULL 
         ELSE VMSTA 
         END,'SNULL'),'TVMST') AS CONFIG_CODE,
VMSTB AS CONFIG_NAME,
'TVMST' AS SOURCE_CODE,
'物料: 销售状态: 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVMST_1001
