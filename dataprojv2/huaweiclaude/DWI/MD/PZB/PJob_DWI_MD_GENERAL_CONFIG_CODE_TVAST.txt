SPRAS AS LANG_CODE,
COALESCE(CASE WHEN AUFSP ='' THEN NULL 
         ELSE AUFSP 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN AUFSP ='' THEN NULL 
         ELSE AUFSP 
         END,'SNULL'),'TVAST') AS CONFIG_CODE,      
VTEXT AS CONFIG_NAME,
'TVAST' AS SOURCE_CODE,
'销售凭证冻结原因： 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVAST_1001
