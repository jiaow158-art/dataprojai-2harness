INSERT OVERWRITE TABLE DWIMD.DWI_MD_ADDRESS_SERVICE_T
SELECT 
   RC.CLIENT AS GROUP_NUM,
   FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN RC.DATE_FROM = '00010101' THEN '19000101' END,'yyyymmdd'),'yyyy-mm-dd') AS VALID_BEGIN_DATE,
   RC.ADDRNUMBER AS ADDRESS_CODE,
   RC.STREET AS ADDRESS_NAME,
   RC.LANGU AS LANGUAGE_CODE,
   FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN RC.DATE_TO = '99991231' THEN '47121231' END,'yyyymmdd'),'yyyy-mm-dd') AS VALID_END_DATE,
   RC.COUNTRY AS CNTRY_REGION_CODE,
   RC.ADDR_GROUP AS ADDR_GROUP_CODE,
   RC.PERS_ADDR AS PERSON_ADDR_FLAG,
   RC.TIME_ZONE AS ADDR_TIME_ZONE,
   RC.ADRC_UUID AS ADRC_UUID,
   'N' AS DEL_FLAG,
   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ADRC_1001 RC;
