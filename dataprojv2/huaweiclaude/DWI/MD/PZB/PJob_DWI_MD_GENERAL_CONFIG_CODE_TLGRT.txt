COALESCE(CASE WHEN LADGR ='' THEN NULL

         ELSE LADGR	 
         END,'SNULL'),'TLGRT') AS CONFIG_CODE,
VTEXT AS CONFIG_NAME,
'TLGRT' AS SOURCE_CODE,
'途径： 装运组： 文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
