INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='ZTYL10') 
SELECT MANDT AS GROUP_NUM,                 
'1' AS LANG_CODE,
COALESCE(CASE WHEN value_code ='' THEN NULL 
         ELSE value_code 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN value_code ='' THEN NULL 
         ELSE value_code	 
         END,'SNULL'),'ZTYL10') AS CONFIG_CODE,
value_desc AS CONFIG_NAME,
'ZTYL10' AS SOURCE_CODE,
'品类'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZMMT014_1001
where fieldname = 'YL10'
