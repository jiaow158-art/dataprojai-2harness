INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TFKBT') 
SELECT MANDT AS GROUP_NUM,                 
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN FKBER ='' THEN NULL 
         ELSE FKBER 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN FKBER ='' THEN NULL 
         ELSE FKBER	 
         END,'SNULL'),'TFKBT') AS CONFIG_CODE,
FKBTX AS CONFIG_NAME,
'TFKBT' AS SOURCE_CODE,
'功能范围的名称'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TFKBT_1001
