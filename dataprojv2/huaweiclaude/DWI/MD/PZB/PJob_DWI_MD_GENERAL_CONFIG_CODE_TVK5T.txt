SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN KATR5 ='' THEN NULL 
         ELSE KATR5 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN KATR5 ='' THEN NULL 
         ELSE KATR5 
         END,'SNULL'),'TVK5T') AS CONFIG_CODE,  
VTEXT AS CONFIG_NAME,
'TVK5T' AS SOURCE_CODE,
'属性5文本（客户主文件）'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TVK5T_1001
