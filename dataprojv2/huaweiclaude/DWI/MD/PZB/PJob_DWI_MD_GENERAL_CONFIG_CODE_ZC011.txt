INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='ZC011') 
SELECT MANDT AS GROUP_NUM,                 
'1' AS LANG_CODE,
COALESCE(CASE WHEN WLPP ='' THEN NULL 
         ELSE WLPP 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN WLPP ='' THEN NULL 
         ELSE WLPP	 
         END,'SNULL'),'ZC011') AS CONFIG_CODE,
WLPPMC AS CONFIG_NAME,
'ZC011' AS SOURCE_CODE,
'产品系列'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZC011_1001
