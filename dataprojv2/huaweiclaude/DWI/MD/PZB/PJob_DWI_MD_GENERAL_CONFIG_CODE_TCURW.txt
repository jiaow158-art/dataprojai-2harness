INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TCURW') 
SELECT MANDT AS GROUP_NUM,
SPRAS AS LANG_CODE,
COALESCE(CASE WHEN KURST ='' THEN NULL 
         ELSE KURST 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN KURST ='' THEN NULL 
         ELSE KURST 
         END,'SNULL'),'TCURW') AS CONFIG_CODE,
CURVW AS CONFIG_NAME,
'TCURW' AS SOURCE_CODE, 
'汇率类型的使用'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TCURW_1001
