INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATERIAL_LIST_LINK_T
SELECT 
   T.MANDT AS GROUP_NUM
   ,T.STLNR AS MATL_LIST
   ,T.STLAL AS OPTIONAL_BOM
   ,T.MATNR AS MATERIAL_NUM
   ,T.WERKS AS FACTORY_CODE
   ,T.STLAN AS BOM_USE_CODE
   ,FROM_UNIXTIME(UNIX_TIMESTAMP(T.ANDAT,'yyyyMMdd'),'yyyy-MM-dd')  AS REC_CRT_DATE
   ,T.ANNAM AS REC_CRT_PERSON_NUM
   ,'N' AS DEL_FLAG
   ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_MAST_1001 T
