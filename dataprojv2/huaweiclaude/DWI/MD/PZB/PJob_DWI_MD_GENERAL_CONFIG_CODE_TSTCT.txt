INSERT OVERWRITE TABLE DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R PARTITION(SOURCE_CODE_D='TSTCT') 
SELECT NULL  AS GROUP_NUM,
SPRSL AS LANG_CODE,
COALESCE(CASE WHEN TCODE ='' THEN NULL 
         ELSE TCODE 
         END,'SNULL') AS SRC_SYS_CONFIG_CODE,   
CONCAT_WS('_',COALESCE(CASE WHEN TCODE ='' THEN NULL 
         ELSE TCODE 
         END,'SNULL'),'TSTCT') AS CONFIG_CODE,
TTEXT AS CONFIG_NAME,
'TSTCT' AS SOURCE_CODE,
'事务代码文本'AS SOURCE_DESC,
NULL AS EXTENDED_ATTRIBUTES1,
NULL AS EXTENDED_ATTRIBUTES2,
NULL AS EXTENDED_ATTRIBUTES3,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_TSTCT_1001
