INSERT OVERWRITE TABLE DWIMD.DWI_MD_CUST_CREDIT_MNGMT_TMP
SELECT   
        NVL(ST.CUST_ID,-999998),
        CMS.PARTNER AS SRC_CUST_NUM,                                                                                                                                  
        CMS.CLIENT AS GROUP_NUM,                                                                                                                                          
        T.CONFIG_CODE AS CALC_SCORE_CREDIT_L_RULE_CODE,                                                                                                                             
        FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN CMS.RATING_CHG_DATE ='00000000' OR CMS.RATING_CHG_DATE = ''THEN '19000101' 
			       ELSE CMS.RATING_CHG_DATE END , 'yyyymmdd'),'yyyy-mm-dd') AS SCORE_CHG_DATE,                     
        FROM_UNIXTIME(UNIX_TIMESTAMP (CASE WHEN CMS.RISK_CLASS_CHGDT  = '00000000' OR CMS.RISK_CLASS_CHGDT = '' THEN '19000101'                                                                      
                                           ELSE CMS.RISK_CLASS_CHGDT END , 'yyyymmdd'),'yyyy-mm-dd') AS RISK_CLASS_CHG_DATE,                                              
        FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN CMS.RATING_VAL_DATE ='00000000' OR  CMS.RATING_VAL_DATE = '' THEN '19000101'
			       ELSE CMS.RATING_VAL_DATE END , 'yyyymmdd'),'yyyy-mm-dd') AS EFFEC_END_DATE,                     
        T1.CONFIG_CODE AS RISK_CHASS_CODE,                                                                                                                                
        CMS.OWN_RATING  AS SCORE,                                                                                                                                         
        CMS.CREDIT_GROUP  AS CUST_CREDIT_GROUP,                                                                                                               
        T2.CONFIG_CODE AS CREDIT_CHECK_RULES,                                                                                                                        
        'N' AS DEL_FLAG ,
         NVL(ST.CUST_NUM,'NOT EXISTS') as CUST_NUM,
         CMS.CHECK_RULE AS CUST_CHECK_RULE,
        CMS.LIMIT_RULE AS CUST_LIMIT_RULE,
        CMS.RISK_CLASS AS CUST_RISK_CLASS
   FROM SDI.SDI_UKMBP_CMS_1001 CMS
   LEFT  JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T
    ON T.SRC_SYS_CONFIG_CODE = CMS.LIMIT_RULE
   AND T.GROUP_NUM = CMS.CLIENT
   AND T.SOURCE_CODE = 'UKM_LIMIT_RULE0T' 
   LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1
    ON T1.SRC_SYS_CONFIG_CODE =  CMS.RISK_CLASS
   AND T1.GROUP_NUM = CMS.CLIENT 
   AND T1.SOURCE_CODE = 'UKM_RISK_CL0T' 
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T2
    ON T2.SRC_SYS_CONFIG_CODE = CMS.CHECK_RULE 
   AND T2.GROUP_NUM = CMS.CLIENT 
   AND T2.SOURCE_CODE = 'UKM_CHCK_RULE0T' 
LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T ST
   ON CMS.PARTNER = ST.CUST_NUM
  AND CMS.CLIENT = ST.GROUP_NUM  
  AND ST.END_DATE = '4712-12-31'
  AND ST.DEL_FLAG = 'N';
