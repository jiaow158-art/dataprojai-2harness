      ON T.SRC_SYS_CONFIG_CODE = KNB1.BEGRU
	 AND T.GROUP_NUM = KNB1.MANDT 
     AND  T.SOURCE_CODE = 'TVGRT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1
      ON T1.SRC_SYS_CONFIG_CODE = KNB1.FDGRV
	 AND T1.GROUP_NUM = KNB1.MANDT 
     AND T1.SOURCE_CODE = 'T035T'
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COMPANY
    ON KNB1.MANDT = COMPANY.GROUP_NUM
   AND KNB1.BUKRS = COMPANY.COMPANY_CODE
   AND COMPANY.END_DATE = '4712-12-31'
   AND COMPANY.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T ST
   ON KNB1.KUNNR = ST.CUST_NUM
  AND KNB1.MANDT = ST.GROUP_NUM
  AND ST.END_DATE = '4712-12-31'
  AND ST.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_T035T_1001 T035T
   ON KNB1.MANDT = T035T.MANDT
  AND KNB1.FDGRV = T035T.GRUPP
  AND T035T.SPRAS = '1'
  ;
