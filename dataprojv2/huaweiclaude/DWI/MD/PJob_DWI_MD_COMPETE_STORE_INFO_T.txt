INSERT OVERWRITE TABLE DWIMD.DWI_MD_COMPETE_STORE_INFO_T
  SELECT ID, 					-- 代理主键
         STORE_ID, 				-- 门店id
         BRAND_ID, 				-- 品牌id
         STORE_NAME, 			-- 门店名称
         IS_OPEN, 				-- 是否营业 1-营业，0-打烊
         PROVINCE_ID, 			-- 省区域ID
         PROVINCE_NAME, 		-- 省
         CITY_ID, 				-- 市区域ID
         CITY_NAME, 			-- 市
         DISTRICT_ID, 			-- 区区域ID
         DISTRICT_NAME, 		-- 区
         ADDRESS, 				-- 详细地址
         LONGITUDE, 			-- 门店经度
         LATITUDE, 				-- 门店纬度
         CREATE_TIME, 			-- 创建时间
         UPDATE_TIME, 			-- 更新时间
         STRAT_DATE, 			-- 开始日期
         UPDATE_DATE, 			-- 结束日期
         END_DATE, 				-- 结束日期
         DATA_TYPE, 			-- 数据类型
         DEL_FLAG, 				-- 删除标识
         DW_LAST_UPDATE_DATE 	-- 同步数据时间
    FROM DWIMD.DWI_MD_COMPETE_STORE_INFO_T_HIS
   WHERE END_DATE = '4712-12-31' 
     AND DEL_FLAG = 'N';
