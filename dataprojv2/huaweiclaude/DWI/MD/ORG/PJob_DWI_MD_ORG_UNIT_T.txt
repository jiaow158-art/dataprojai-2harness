set hive.execution.engine=spark;



 with sdi_tmp as  (
--INSERT overwrite table  dwimd.dwi_md_org_unit_t
select 
 t1.mandt  GROUP_NUM
,t1.sprsl  LANGUAGE_CODE
,t1.orgeh ORG_UNIT_CODE
,t1.orgtx ORG_UNIT_NAME 
,t1.maint OM_maintain
,'N' del_flag 
,CURRENT_TIMESTAMP() AS dw_last_update_date
,FROM_UNIXTIME(UNIX_TIMESTAMP(t1.BEGDA,'yyyymmdd'),'yyyy-mm-dd') AS START_DATE
,case when  FROM_UNIXTIME(UNIX_TIMESTAMP(t1.ENDDA ,'yyyymmdd'),'yyyy-mm-dd') = '9999-12-31' then '4712-12-31' else  
 date_add(FROM_UNIXTIME(UNIX_TIMESTAMP(t1.ENDDA ,'yyyymmdd'),'yyyy-mm-dd'),1)
end end_date
from sdi.sdi_t527x_1001 t1 

)



,dwi_tmp as (
select
 org_unit_id
,GROUP_NUM
,LANGUAGE_CODE
,ORG_UNIT_CODE
,ORG_UNIT_NAME
,OM_maintain
,del_flag
,dw_last_update_date
,START_DATE
,END_DATE
from  dwimd.dwi_md_org_unit_t
)






,onlyid as (
select t1.GROUP_NUM,t1.LANGUAGE_CODE,t1.ORG_UNIT_CODE,ROW_NUMBER () over() org_unit_id 
from sdi_tmp t1 left join dwi_tmp t2 on t1.GROUP_NUM = t2.GROUP_NUM and t1.LANGUAGE_CODE = t2.LANGUAGE_CODE and t1.ORG_UNIT_CODE =t2.ORG_UNIT_CODE 
and  t1.END_DATE = t2.END_DATE 
where  t2.ORG_UNIT_CODE is null 
)






INSERT overwrite table  dwimd.dwi_md_org_unit_t
select
 t1.org_unit_id
,t1.GROUP_NUM
,t1.LANGUAGE_CODE
,t1.ORG_UNIT_CODE
,t1.ORG_UNIT_NAME
,t1.OM_maintain
,case when t2.del_flag is null and t1.org_unit_id not in (-999999,-999998)  then 'Y' else t1.del_flag end del_flag
,CURRENT_TIMESTAMP() dw_last_update_date
,t1.START_DATE
,t1.END_DATE
from  dwi_tmp t1 left join sdi_tmp t2 on t1.GROUP_NUM = t2.GROUP_NUM and t1.LANGUAGE_CODE = t2.LANGUAGE_CODE and t1.ORG_UNIT_CODE =t2.ORG_UNIT_CODE and t1.end_date =t2.end_date
union 
select
 (t4.org_unit_id + t3.org_unit_id ) ord_unit_id 
,t1.GROUP_NUM
,t1.LANGUAGE_CODE
,t1.ORG_UNIT_CODE
,t1.ORG_UNIT_NAME
,t1.OM_maintain 
,t1.del_flag
,CURRENT_TIMESTAMP() dw_last_update_date
,t1.START_DATE
,t1.END_DATE
from  sdi_tmp t1 
left join dwi_tmp t2 on t1.GROUP_NUM = t2.GROUP_NUM and t1.LANGUAGE_CODE = t2.LANGUAGE_CODE and t1.ORG_UNIT_CODE =t2.ORG_UNIT_CODE and t1.end_date =t2.end_date
 join onlyid t3 on  t1.GROUP_NUM = t3.GROUP_NUM and t1.LANGUAGE_CODE = t3.LANGUAGE_CODE and t1.ORG_UNIT_CODE =t3.ORG_UNIT_CODE  
join (select max(org_unit_id) org_unit_id from dwi_tmp) t4 on 1=1 
where t2.ORG_UNIT_CODE is null
