--insert overwrite table dwimd.DWI_MD_INTO_TYPE_1001_T 


with SDI_TMP AS (
SELECT
 MANDT	GROUP_NUM
,OTYPE	OBJECT_TYPE
,OBJID	OBJECT_FLAG
,PLVAR	PLAN_VERSION
,RSIGN	RELATIONSHIP_SPECIFICATIONS
,RELAT	RELATIONSHIP_OBJECT
,ISTAT	PLAN_STATUS
,PRIOX	PRIORITY
,VARYF	FILE_PLOG_CHANGE_FIELD
,SEQNR	INFO_TYPE_RECORD_NO
,INFTY	INFO_TYPE
,OTJID	MRGE_OBJ_TYPE_AND_OBJ_FLAG
,SUBTY	SUB_TYPE
,UNAME	USER_NAME
,ITXNR	INFO_TYPE_TEXT_MODULE
,SCLAS	RELATED_OBJECT_TYPE
,SOBID	RELATED_OBJECT_FLAG
,PROZT	WEIGHT_RATE
,FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN AEDTM='00000000' THEN '19000101' ELSE AEDTM END ,'yyyymmdd'),'yyyy-mm-dd')  LAST_MODIFY_DATE   
,'N' DEL_FLAG	
,DW_LAST_UPDATE_DATE
,FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN begda='00000000' THEN '19000101' ELSE begda END ,'yyyymmdd'),'yyyy-mm-dd')  START_DATE   
,FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN endda='99991231' THEN '47121231' ELSE endda END ,'yyyymmdd'),'yyyy-mm-dd')  END_DATE   
FROM SDI.SDI_HRP1001_1001
)
,DWI_TMP AS (
SELECT
 GROUP_NUM
,OBJECT_TYPE
,OBJECT_FLAG
,PLAN_VERSION
,RELATIONSHIP_SPECIFICATIONS
,RELATIONSHIP_OBJECT
,PLAN_STATUS
,PRIORITY
,FILE_PLOG_CHANGE_FIELD
,INFO_TYPE_RECORD_NO
,INFO_TYPE
,MRGE_OBJ_TYPE_AND_OBJ_FLAG
,SUB_TYPE
,USER_NAME
,INFO_TYPE_TEXT_MODULE
,RELATED_OBJECT_TYPE
,RELATED_OBJECT_FLAG
,WEIGHT_RATE
,LAST_MODIFY_DATE   
,DEL_FLAG	
,DW_LAST_UPDATE_DATE
,START_DATE   
,END_DATE   
FROM dwimd.DWI_MD_INTO_TYPE_1001_T 
)


insert overwrite table dwimd.DWI_MD_INTO_TYPE_1001_T 
SELECT 
 T1.GROUP_NUM
,T1.OBJECT_TYPE
,T1.OBJECT_FLAG
,T1.PLAN_VERSION
,T1.RELATIONSHIP_SPECIFICATIONS
,T1.RELATIONSHIP_OBJECT
,T1.PLAN_STATUS
,T1.PRIORITY
,T1.FILE_PLOG_CHANGE_FIELD
,T1.INFO_TYPE_RECORD_NO
,T1.INFO_TYPE
,T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG
,T1.SUB_TYPE
,T1.USER_NAME
,T1.INFO_TYPE_TEXT_MODULE
,T1.RELATED_OBJECT_TYPE
,T1.RELATED_OBJECT_FLAG
,T1.WEIGHT_RATE
,T1.LAST_MODIFY_DATE   
,CASE WHEN T2.MRGE_OBJ_TYPE_AND_OBJ_FLAG IS NULL THEN 'Y' ELSE T1.DEL_FLAG	END DEL_FLAG
,T1.DW_LAST_UPDATE_DATE
,T1.START_DATE   
,T1.END_DATE 
FROM DWI_TMP T1 LEFT JOIN SDI_TMP T2 ON T1.GROUP_NUM = T2.GROUP_NUM AND T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG = T2.MRGE_OBJ_TYPE_AND_OBJ_FLAG AND T1.END_DATE = T2.END_DATE AND T1.START_DATE = T2.START_DATE 
UNION ALL 
SELECT 
 T1.GROUP_NUM
,T1.OBJECT_TYPE
,T1.OBJECT_FLAG
,T1.PLAN_VERSION
,T1.RELATIONSHIP_SPECIFICATIONS
,T1.RELATIONSHIP_OBJECT
,T1.PLAN_STATUS
,T1.PRIORITY
,T1.FILE_PLOG_CHANGE_FIELD
,T1.INFO_TYPE_RECORD_NO
,T1.INFO_TYPE
,T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG
,T1.SUB_TYPE
,T1.USER_NAME
,T1.INFO_TYPE_TEXT_MODULE
,T1.RELATED_OBJECT_TYPE
,T1.RELATED_OBJECT_FLAG
,T1.WEIGHT_RATE
,T1.LAST_MODIFY_DATE   
,T1.DEL_FLAG
,T1.DW_LAST_UPDATE_DATE
,T1.START_DATE   
,T1.END_DATE 
FROM SDI_TMP T1 LEFT JOIN DWI_TMP T2 ON T1.GROUP_NUM = T2.GROUP_NUM AND T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG = T2.MRGE_OBJ_TYPE_AND_OBJ_FLAG AND T1.END_DATE = T2.END_DATE AND T1.START_DATE = T2.START_DATE WHERE T2.MRGE_OBJ_TYPE_AND_OBJ_FLAG IS NULL
