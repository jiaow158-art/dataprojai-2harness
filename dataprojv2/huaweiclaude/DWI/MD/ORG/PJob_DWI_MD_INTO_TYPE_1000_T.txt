WITH SDI_TMP AS (
--INSERT OVERWRITE TABLE dwimd.DWI_MD_INTO_TYPE_1000_T
SELECT
 t1.mandt  GROUP_NUM 
,t1.langu  LANGUAGE_CODE 
,t1.plvar  PLAN_VERSION 
,t1.istat  PLAN_STATUS 
,t1.infty  INFO_TYPE 
,t1.seqnr  INFO_TYPE_RECORD_NO
,t1.itxnr  INFO_TYPE_TEXT_MODULE
,t1.otype  OBJECT_TYPE
,t1.objid  OBJECT_FLAG
,t1.otjid  MRGE_OBJ_TYPE_AND_OBJ_FLAG 
,t1.stext  OBJECT_NAME 
,t1.short  OBJECT_ABBREVIATIONS
,t1.mc_short  MC_OBJECT_ABBREVIATIONS 
,t1.mc_stext  NAME
,t1.mc_seark  SEARCH_TERM 
,FROM_UNIXTIME(UNIX_TIMESTAMP(case when t1.GDATE ='00000000' then '19000101' ELSE t1.GDATE  end ,'yyyymmdd'),'yyyy-mm-dd') DELIMITATION_DATE
,t1.uname  USER_NAME 
,FROM_UNIXTIME(UNIX_TIMESTAMP(t1.aedtm,'yyyymmdd'),'yyyy-mm-dd')  LAST_MODIFY_DATE 
,"N"del_flag
,CURRENT_TIMESTAMP() dw_last_update_date 
,FROM_UNIXTIME(UNIX_TIMESTAMP(t1.begda,'yyyymmdd'),'yyyy-mm-dd') AS START_DATE
,FROM_UNIXTIME(UNIX_TIMESTAMP(case when endda ='99991231' then '47121231' ELSE t1.endda  end ,'yyyymmdd'),'yyyy-mm-dd') as END_DATE 
FROM sdi.sdi_hrp1000_1001 t1  
)
,DWI_TMP AS (
SELECT 
 GROUP_NUM
,LANGUAGE_CODE
,PLAN_VERSION
,PLAN_STATUS
,INFO_TYPE
,INFO_TYPE_RECORD_NO
,INFO_TYPE_TEXT_MODULE
,OBJECT_TYPE
,OBJECT_FLAG
,MRGE_OBJ_TYPE_AND_OBJ_FLAG
,OBJECT_NAME
,OBJECT_ABBREVIATIONS
,MC_OBJECT_ABBREVIATIONS
,NAME
,SEARCH_TERM
,DELIMITATION_DATE
,USER_NAME
,LAST_MODIFY_DATE
,del_flag
,dw_last_update_date
,START_DATE
,END_DATE
FROM dwimd.DWI_MD_INTO_TYPE_1000_T 
)



--INSERT OVERWRITE TABLE dwimd.DWI_MD_INTO_TYPE_1000_T
select 
 T1.GROUP_NUM
,T1.LANGUAGE_CODE
,T1.PLAN_VERSION
,T1.PLAN_STATUS
,T1.INFO_TYPE
,T1.INFO_TYPE_RECORD_NO
,T1.INFO_TYPE_TEXT_MODULE
,T1.OBJECT_TYPE
,T1.OBJECT_FLAG
,T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG
,T1.OBJECT_NAME
,T1.OBJECT_ABBREVIATIONS
,T1.MC_OBJECT_ABBREVIATIONS
,T1.NAME
,T1.SEARCH_TERM
,T1.DELIMITATION_DATE
,T1.USER_NAME
,T1.LAST_MODIFY_DATE
,case when t2.MRGE_OBJ_TYPE_AND_OBJ_FLAG is null  then 'Y' else T1.del_flag END  del_flag
,T1.dw_last_update_date
,T1.START_DATE
,T1.END_DATE
from DWI_TMP t1 left join sdi_tmp t2 on t1.GROUP_NUM = t2.GROUP_NUM and t1.LANGUAGE_CODE = t2.LANGUAGE_CODE and t1.MRGE_OBJ_TYPE_AND_OBJ_FLAG = t2.MRGE_OBJ_TYPE_AND_OBJ_FLAG  
union all 
SELECT 
 T1.GROUP_NUM
,T1.LANGUAGE_CODE
,T1.PLAN_VERSION
,T1.PLAN_STATUS
,T1.INFO_TYPE
,T1.INFO_TYPE_RECORD_NO
,T1.INFO_TYPE_TEXT_MODULE
,T1.OBJECT_TYPE
,T1.OBJECT_FLAG
,T1.MRGE_OBJ_TYPE_AND_OBJ_FLAG
,T1.OBJECT_NAME
,T1.OBJECT_ABBREVIATIONS
,T1.MC_OBJECT_ABBREVIATIONS
,T1.NAME
,T1.SEARCH_TERM
,T1.DELIMITATION_DATE
,T1.USER_NAME
,T1.LAST_MODIFY_DATE
,'N' del_flag
,T1.dw_last_update_date
,T1.START_DATE
,T1.END_DATE
FROM sdi_tmp T1 
LEFT JOIN DWI_TMP T2  ON  t1.GROUP_NUM = t2.GROUP_NUM and t1.LANGUAGE_CODE = t2.LANGUAGE_CODE and t1.MRGE_OBJ_TYPE_AND_OBJ_FLAG = t2.MRGE_OBJ_TYPE_AND_OBJ_FLAG    
where T2.MRGE_OBJ_TYPE_AND_OBJ_FLAG IS NULL
