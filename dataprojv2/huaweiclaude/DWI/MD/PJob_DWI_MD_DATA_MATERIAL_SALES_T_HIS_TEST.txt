insert overwrite table dwimd.dwi_md_data_material_sales_t_his_test
SELECT  							
    MVKE.VKORG AS SALES_ORG_CODE,
    MVKE.MATNR  AS MATERIAL_NUM,
    MVKE.MANDT AS GROUP_NUM,
    MVKE.VTWEG AS DISTRIBUTION_CHANNEL_CODE,
    TVTWT.VTEXT AS DISTRIBUTION_CHNL_NAME,
    TVKOT.VTEXT AS SALES_ORG_NAME,
    MVKE.VMSTA AS SPEC_D_CHAIN_MM_STATUS_CODE,
    MVKE.SKTOF AS CASH_DISCOUNT_FLAG,
	MVKE.MVGR1 AS MATL_GROUP1_CODE,
    MVKE.MVGR2 AS MATL_GROUP2_CODE,
	MVKE.MVGR3 AS MATL_GROUP3_CODE,
	MVKE.MVGR4 AS MATL_GROUP4_CODE,
    MVKE.MVGR5 AS MATL_GROUP5_CODE,
    MVKE.VERSG AS MATL_STATISTICS_GROUP_CODE,
    MVKE.KONDM AS MATL_PRICE_GROUP_CODE,
    MVKE.MTPOS AS MATL_M_FILE_ITEM_C_GROUP_CODE,
	TPTMT.BEZEI AS MATL_M_FILE_ITEM_C_GROUP_NAME,
    MVKE.DWERK AS DELIVERY_FACTORY,
    MVKE.KTGRM AS MATL_SUBJ_SET_GROUP_CODE,
	TVKMT.VTEXT AS MATL_SUBJ_SET_GROUP_NAME,
	MVKE.PRAT1 AS PROD_ATTRIBUTE_ID_1,
	MVKE.PRAT2 AS PROD_ATTRIBUTE_ID_2,
	MVKE.PRAT3 AS PROD_ATTRIBUTE_ID_3,
	MVKE.PRAT4 AS PROD_ATTRIBUTE_ID_4,
	MVKE.PRAT5 AS PROD_ATTRIBUTE_ID_5,
    MVKE.PRAT6 AS PROD_ATTRIBUTE_ID_6,
	MVKE.PRAT7 AS PROD_ATTRIBUTE_ID_7,
	MVKE.PRAT8 AS PROD_ATTRIBUTE_ID_8,
    MVKE.PRAT9 AS PROD_ATTRIBUTE_ID_9,         
    MVKE.PRATA AS PROD_ATTRIBUTE_ID_10,
	'N' AS DEL_FLAG,
    MATLGEN.update_date AS  update_date,
    MATLGEN.update_date AS start_date,
    '4712-12-31' AS end_date,
    TVM1T.BEZEI AS MATL_GROUP1_NAME,
    TVM2T.BEZEI AS MATL_GROUP2_NAME,
	TVM3T.BEZEI AS MATL_GROUP3_NAME,
	TVM4T.BEZEI AS MATL_GROUP4_NAME,
    TVM5T.BEZEI AS MATL_GROUP5_NAME,
	TVSMT.BEZEI20 AS MATL_STATISTICS_GROUP_NAME,
	T178T.VTEXT AS MATL_PRICE_GROUP_NAME,	
	MATLGEN.create_date AS create_date,
	from_unixtime(unix_timestamp() ,'yyyy-MM-dd HH:mm:ss') dw_last_update_date
FROM SDI.SDI_MVKE_TEST_1001 MVKE
LEFT JOIN dwimd.dwi_md_data_material_general_t_his_test MATLGEN
      ON  MVKE.MANDT = MATLGEN.GROUP_NUM
	  AND MVKE.MATNR = MATLGEN.MATERIAL_NUM
LEFT JOIN SDI.SDI_TVTWT_1001 TVTWT
       ON MVKE.VTWEG = TVTWT.VTWEG
	  AND TVTWT.MANDT = '800'
	  AND TVTWT.SPRAS = '1'
LEFT JOIN SDI.SDI_TVKOT_1001 TVKOT	
      ON MVKE.VKORG	= TVKOT.VKORG
	 AND TVKOT.MANDT = '800' 
	 AND TVKOT.SPRAS  = '1'
LEFT JOIN SDI.SDI_TVM1T_1001 TVM1T
       ON MVKE.MVGR1 = TVM1T.MVGR1 
	  AND TVM1T.MANDT = '800'
	  AND TVM1T.SPRAS = '1'	  
LEFT JOIN SDI.SDI_TVM2T_1001 TVM2T
       ON MVKE.MVGR2 = TVM2T.MVGR2 
	  AND TVM2T.MANDT = '800'
	  AND TVM2T.SPRAS = '1'	  
LEFT JOIN SDI.SDI_TVM3T_1001 TVM3T
       ON MVKE.MVGR3 = TVM3T.MVGR3 
	  AND TVM3T.MANDT = '800'
	  AND TVM3T.SPRAS = '1'	  
LEFT JOIN SDI.SDI_TVM4T_1001 TVM4T
       ON MVKE.MVGR4 = TVM4T.MVGR4
	  AND TVM4T.MANDT = '800'
	  AND TVM4T.SPRAS = '1'	  
LEFT JOIN SDI.SDI_TVM5T_1001 TVM5T
       ON MVKE.MVGR5 = TVM5T.MVGR5
	  AND TVM5T.MANDT = '800'
	  AND TVM5T.SPRAS = '1' 
LEFT JOIN SDI.SDI_TPTMT_1001 TPTMT
       ON MVKE.MTPOS = TPTMT.MTPOS
	  AND TPTMT.MANDT = '800'
	  AND TPTMT.SPRAS = '1'
LEFT JOIN SDI.SDI_TVKMT_1001 TVKMT
       ON MVKE.KTGRM = TVKMT.KTGRM
	  AND TVKMT.MANDT = '800'
	  AND TVKMT.SPRAS = '1'
LEFT JOIN SDI.SDI_TVSMT_1001 TVSMT
       ON MVKE.VERSG = TVSMT.STGMA 
	  AND TVSMT.MANDT = '800'
	  AND TVSMT.SPRAS = '1' 
LEFT JOIN SDI.SDI_T178T_1001 T178T
       ON MVKE.KONDM = T178T.KONDM 
	  AND T178T.MANDT = '800'
	  AND T178T.SPRAS = '1'
