INSERT OVERWRITE  TABLE   DWIMD.DWI_MD_MATERIAL_MENU_T (
SELECT
T1.MANDT,
T1.MATNR,
T1.DATAB,
T1.DATBI ,
CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE,
CONCAT(SUBSTR(T1.DATAB,1,4),'-',SUBSTR(T1.DATAB,5,2)) AS S_MOTH,   --有效开始日期
CONCAT(SUBSTR(T1.DATBI,1,4),'-',SUBSTR(T1.DATBI,5,2)) AS E_MOTH,   --有效结束日期
T1.vtweg,      --渠道
CASE WHEN T1.vtweg = '01' THEN  '零售'  
                    when T1.vtweg = '06'  then '工程'
                    when T1.vtweg = '08'  then '整装'
                    end as vtweg_desc,--渠道描述
T1.menu_type,  --类型
case when T1.menu_type = '0' then '主砖' when  T1.menu_type = '1' then '配件'  end  as menu_type_desc,--类型描述
T2.material_name
FROM 
SDI.SDI_ZMMTBI030_1001  T1  
LEFT JOIN  
dwimd.dwi_md_data_material_general_t  T2 
ON  
T1.matnr  = T2.material_num

)
