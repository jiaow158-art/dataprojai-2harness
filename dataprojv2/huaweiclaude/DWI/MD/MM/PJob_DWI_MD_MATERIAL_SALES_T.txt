SELECT  /* +mapjoin(MVKE) */
    NVL(MD.MATERIAL_ID,-999998) AS MATERIAL_ID,

    T3.CONFIG_CODE AS MATL_GROUP2_CODE,
    T4.CONFIG_CODE AS MATL_GROUP1_CODE,
    T5.CONFIG_CODE AS MATL_STATISTICS_GROUP_CODE,
    T6.CONFIG_CODE AS MATL_PRICE_GROUP_CODE,
    FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN MVKE.VMSTD = '00000000'  OR MVKE.VMSTD = '' THEN '19000101' 
			 ELSE MVKE.VMSTD  END ,'yyyymmdd'),'yyyy-mm-dd') AS SPEC_D_C_MATL_S_VALID_DATE,
    MVKE.MTPOS AS MATL_M_FILE_ITEM_C_GROUP_CODE,
    MVKE.DWERK AS DELIVERY_FACTORY,
    MVKE.KTGRM AS MATL_SUBJ_SET_GROUP_CODE,
    MVKE.PRAT6 AS PROD_ATTRIBUTE_ID_6,
    MVKE.PRAT5 AS PROD_ATTRIBUTE_ID_5,
    MVKE.PRAT4 AS PROD_ATTRIBUTE_ID_4,
    MVKE.PRAT3 AS PROD_ATTRIBUTE_ID_3,
    MVKE.PRAT2 AS PROD_ATTRIBUTE_ID_2,
    MVKE.PRATA AS PROD_ATTRIBUTE_ID_10,
    MVKE.PRAT1 AS PROD_ATTRIBUTE_ID_1,
    MVKE.PRAT9 AS PROD_ATTRIBUTE_ID_9,
    MVKE.PRAT8 AS PROD_ATTRIBUTE_ID_8,
    MVKE.PRAT7 AS PROD_ATTRIBUTE_ID_7,
    T7.CONFIG_CODE AS PRODUCT_LEVEL_CODE,
    MVKE.LVORM AS MATL_DEL_FLAG,
    'N'  AS DEL_FLAG,
	MVKE.MATNR AS SRC_MATERIAL_NUM
FROM SDI.SDI_MVKE_1001 MVKE
LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MD
      ON MVKE.MANDT = MD.GROUP_NUM
	  AND MVKE.MATNR = MD.MATERIAL_NUM
	  AND MD.END_DATE = '4712-12-31'
	  AND MD.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_TVTWT_1001 TV
       ON MVKE.VTWEG = TV.VTWEG
	  AND CAST(TV.MANDT AS STRING) = '800'
	  AND CAST(TV.SPRAS AS STRING)  = '1'
LEFT JOIN SDI.SDI_TVKOT_1001 TK	
      ON MVKE.VKORG	= TK.VKORG
	 AND MVKE.MANDT =  CAST(TK.MANDT AS STRING)
	 AND  CAST(TK.SPRAS AS STRING) = '1'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T
       ON MVKE.MVGR5 = T.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T.GROUP_NUM
	  AND T.SOURCE_CODE = 'TVM5T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1
       ON MVKE.MVGR4 = T1.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T1.GROUP_NUM
	  AND T1.SOURCE_CODE = 'TVM4T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T2
       ON MVKE.MVGR3 = T2.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T2.GROUP_NUM
	  AND T2.SOURCE_CODE = 'TVM3T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T3
       ON MVKE.MVGR2 = T3.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T3.GROUP_NUM
	  AND T3.SOURCE_CODE = 'TVM2T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T4
       ON MVKE.MVGR1 = T4.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T4.GROUP_NUM
	  AND T4.SOURCE_CODE = 'TVM1T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T5
       ON MVKE.VERSG = T5.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T5.GROUP_NUM
	  AND T5.SOURCE_CODE = 'TVSMT' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T6
       ON MVKE.KONDM = T6.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T6.GROUP_NUM
	  AND T6.SOURCE_CODE = 'T178T' 
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T7
       ON MVKE.PRODH = T7.SRC_SYS_CONFIG_CODE 
	  AND MVKE.MANDT = T7.GROUP_NUM
	  AND T7.SOURCE_CODE = 'T179T' ;
