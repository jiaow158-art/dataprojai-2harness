INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATL_INVENT_LOC_TMP
SELECT  /* +mapjoin(MARD) */
   NVL(MD.MATERIAL_ID,-999998) AS MATERIAL_ID,
   MARD.MANDT AS GROUP_NUM,
   NVL(MD.MATERIAL_NUM,'NOT EXISTS')  AS MATERIAL_NUM,
   CASE WHEN MARD.WERKS = '' OR MARD.WERKS IS NULL THEN 'SNULL'
		ELSE NVL(FACTORY.FACTORY_CODE,'NOT EXISTS') END AS FACTORY_CODE,
   MARD.LGORT AS INVENT_LOC_CODE,
   MARD.MDRUE AS LAST_MARD_PERIOD_BEFORE,
   FROM_UNIXTIME(UNIX_TIMESTAMP(MARD.ERSDA,'yyyyMMdd'),'yyyy-MM-dd') AS CRT_DATE,
   MARD.LGPBE AS INVENTORY_POSITION,
   MARD.LVORM AS MATL_DEL_FLAG,
   MARD.MDJIN AS CURR_FISCAL_YEAR,
   MARD.PSTAT AS MAINTAIN_STATUS,
   MARD.LFMON AS CURR_PERIOD,
   MARD.LFGJA AS CURR_PERIOD_FISCAL_YEAR,
   CASE WHEN MARD.WERKS = '' OR MARD.WERKS IS NULL  THEN  -999999
        ELSE NVL(FACTORY.FACTORY_ID, -999998) END AS FACTORY_ID,
  CASE WHEN MARD.WERKS = '' OR MARD.WERKS IS NULL THEN '源为空'
       ELSE NVL(FACTORY.FACTORY_NAME,'主数据不存在') END AS FACTORY_NAME,
   TL.LGOBE AS INVENTORY_LOC_NAME,
   'N' AS DEL_FLAG,
    MARD.WERKS AS SRC_FACTORY_CODE,
	MARD.MATNR AS SRC_MATERIAL_NUM
FROM SDI.SDI_MARD_1001 MARD
LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MD
      ON MARD.MANDT = MD.GROUP_NUM
	  AND MARD.MATNR = MD.MATERIAL_NUM
	  AND MD.END_DATE = '4712-12-31'
	  AND MD.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T FACTORY
    ON MARD.MANDT = FACTORY.GROUP_NUM
   AND MARD.WERKS = FACTORY.FACTORY_CODE
   AND FACTORY.END_DATE = '4712-12-31' 
  AND FACTORY.DEL_FLAG = 'N'   
LEFT JOIN SDI.SDI_T001L_1001 TL
   ON MARD.LGORT = TL.LGORT 
  AND MARD.MANDT = TL.MANDT 
   AND MARD.WERKS = TL.WERKS ;
