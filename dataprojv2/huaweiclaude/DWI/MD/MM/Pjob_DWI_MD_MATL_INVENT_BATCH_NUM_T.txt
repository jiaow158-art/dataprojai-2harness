INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATL_INVENT_BATCH_NUM_TMP
SELECT
   NVL(MD.MATERIAL_ID,-999998) AS MATERIAL_ID,
    NVL(MD.MATERIAL_NUM,'NOT EXISTS')  AS MATERIAL_NUM,
   MCHB.CHARG AS BATCH_NUM,
   MCHB.LGORT AS INVENT_LOC_CODE,
   MCHB.MANDT AS GROUP_NUM,
   CASE WHEN MCHB.WERKS = '' OR MCHB.WERKS IS NULL THEN 'SNULL'
       ELSE NVL(FACTORY.FACTORY_CODE,'NOT EXISTS') END AS FACTORY_CODE,
   MCHB.CHRUE AS LAST_MCHB_PERIOD_BEFORE,
   FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN MCHB.LAEDA = '00000000'  OR MCHB.LAEDA IS NULL THEN '19000101' 
			  ELSE MCHB.LAEDA END ,'yyyymmdd'),'yyyy-mm-dd') AS LAST_CHG_DATE,
   MCHB.AENAM AS CHG_OBJECT_NAME,
   CAST(MCHB.CHJIN AS INT) AS CURR_FISCAL_YEAR,
   CAST(MCHB.LFGJA AS INT) AS CURR_PERIOD_ACCT_YEAR,
   MCHB.LFMON AS CURR_PERIOD,
   FROM_UNIXTIME(UNIX_TIMESTAMP(MCHB.ERSDA,'yyyymmdd'),'yyyy-mm-dd') AS CRT_DATE,
   MCHB.ERNAM AS CRT_OBJECT_PERSON_NAME,
   CASE WHEN MCHB.WERKS = '' OR MCHB.WERKS IS NULL THEN  -999999
        ELSE NVL(FACTORY.FACTORY_ID, -999998) END AS FACTORY_ID,
   CASE WHEN MCHB.WERKS = '' OR MCHB.WERKS IS NULL THEN '源为空'
        ELSE NVL(FACTORY.FACTORY_NAME,'主数据不存在') end AS FACTORY_NAME,
   TL.LGOBE AS INVENTORY_LOC_NAME,
   'N'  AS DEL_FLAG,
    MCHB.WERKS AS SRC_FACTORY_CODE,
	MCHB.MATNR AS SRC_MATERIAL_NUM
FROM SDI.SDI_MCHB_1001 MCHB
LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MD
      ON MCHB.MANDT = MD.GROUP_NUM
	  AND MCHB.MATNR = MD.MATERIAL_NUM
	  AND MD.END_DATE = '4712-12-31'
	  AND MD.DEL_FLAG = 'N'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T FACTORY
    ON MCHB.MANDT = FACTORY.GROUP_NUM
   AND MCHB.WERKS = FACTORY.FACTORY_CODE
   AND FACTORY.END_DATE = '4712-12-31'  
   AND FACTORY.DEL_FLAG = 'N'
LEFT JOIN SDI.SDI_T001L_1001 TL
   ON MCHB.LGORT = TL.LGORT 
  AND MCHB.MANDT = TL.MANDT
  AND MCHB.WERKS = TL.WERKS;
