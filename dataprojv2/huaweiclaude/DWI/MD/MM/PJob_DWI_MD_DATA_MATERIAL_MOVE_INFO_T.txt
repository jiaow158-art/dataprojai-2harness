insert OVERWRITE  table dwimd.dwi_md_data_material_move_info_t 
SELECT 
      MATNR AS MATERIAL_NUM,
      MIN(CASE WHEN charg = '' and charg_cid = '' and ((BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411','561','653') and BSTTYP_SG <> 'K') 
                or (BSTTYP_SG = 'K' and BWART ='411')) and EBELN not like '47%' and substr(WERKS,1,2) <> '31' and substr(WERKS,1,2) <> '30'  THEN BUDAT 
               WHEN charg = '' and charg_cid = '' and ((BWART in ('101','911','913','Z01','131','531','561','653') and BSTTYP_SG <> 'K') or (BSTTYP_SG = 'K' and BWART = '101')) 
               THEN BUDAT END) AS EARLIEST_VOUCHER_POST_DATE_NOBATCH, ---批次最早入库日期(BW通用)
      MIN(CASE WHEN charg = '' and charg_cid = '' and BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and WERKS = '7C20' THEN BUDAT 
               WHEN charg = '' and charg_cid = '' and BWART in ('101','911','913','701','Z01','511','982','991','309','131','531','411') and BSTTYP_SG <> 'K' and EBELN not like '47%'
               THEN BUDAT END) AS earliest_voucher_post_date_fc_NOBATCH, --批次最早入库日期(辅材专用) 
      MIN(case when t.stock_qty>0  then budat end) as earliest_stockin_date,
      MIN(case when t.BWART IN('601','602','687','688')  then budat end) as earliest_sales_stockout_date
  FROM SDI.SDI_MATDOC_1001 t
  GROUP BY
  MATNR;
