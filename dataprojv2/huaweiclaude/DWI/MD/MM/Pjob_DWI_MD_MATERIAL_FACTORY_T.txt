      T6.CONFIG_CODE  AS PLAN_STRATEGY_GROUP_CODE,
      R3.DOMAIN_VALUE_CODE AS POST_TO_INSPECT_STOCK,
     FROM_UNIXTIME(UNIX_TIMESTAMP(CASE WHEN MARC.MMSTD  ='00000000' OR MARC.MMSTD IS NULL THEN '19000101' 
			  ELSE MARC.MMSTD END , 'yyyymmdd'),'yyyy-mm-dd') AS MATL_RE_ATV_V_START_DATE,
      T7.CONFIG_CODE  AS FACTORY_SPEC_MM_STATUS_CODE	,
      MARC.FHORI AS FLOAT_P_P_CODE,

LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MD

	  AND MARC.MATNR = MD.MATERIAL_NUM
	  AND MD.END_DATE = '4712-12-31'

	  AND MARC.MANDT = T.GROUP_NUM 
	  AND T.SOURCE_CODE = 'TLGRT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T1

	  AND MARC.MANDT = T1.GROUP_NUM

LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T2 
       ON MARC.MFRGR = T2.SRC_SYS_CONFIG_CODE 
	  AND MARC.MANDT = T2.GROUP_NUM 
	  AND T2.SOURCE_CODE = 'TMFGT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T3 
       ON MARC.DISMM = T3.SRC_SYS_CONFIG_CODE 
	  AND MARC.MANDT = T3.GROUP_NUM 
	  AND T3.SOURCE_CODE = 'T438T'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T4 
       ON MARC.DISLS = T4.SRC_SYS_CONFIG_CODE 
	  AND MARC.MANDT = T4.GROUP_NUM 
	  AND T4.SOURCE_CODE = 'T439T'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T5 
       ON MARC.BWTTY = T5.SRC_SYS_CONFIG_CODE 
	  AND MARC.MANDT = T5.GROUP_NUM 
	  AND T5.SOURCE_CODE = 'T149T'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T6

	  AND MARC.MANDT = T6.GROUP_NUM 
	  AND T6.SOURCE_CODE = 'T461X'

	  AND MARC.MANDT = T7.GROUP_NUM

    ON MARC.MANDT = D.GROUP_NUM
   AND MARC.WERKS = D.FACTORY_CODE
   AND D.END_DATE = '4712-12-31'
