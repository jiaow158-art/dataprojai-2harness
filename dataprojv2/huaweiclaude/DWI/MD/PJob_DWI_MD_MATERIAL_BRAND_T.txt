--物料品牌主数据
INSERT OVERWRITE TABLE DWIMD.DWI_MD_MATERIAL_BRAND_T
SELECT DISTINCT A.WERKS AS FACTORY_WERKS_CODE
,A.MATKL AS MATL_GROUP_CODE,A.EXTWG AS EXTERNAL_MATL_GROUP_CODE
,B.DOMAIN_VALUE_NAME AS BRAND_NAME
,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ZTSDBK02_1001 A
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_CODE_R B
ON A.ZBK = B.SRC_SYS_DOMAIN_VALUE_CODE 
AND B.DOMAIN_NAME = 'ZBK'
AND B.LANG_CODE = '1'
AND B.DEL_FLAG = 'N'
WHERE A.DEL_FLAG = 'N'
;
