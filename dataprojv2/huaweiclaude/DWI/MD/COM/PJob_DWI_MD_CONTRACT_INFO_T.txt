--SDI_CX_CONTAT_INFO_1002 到 DWI_MD_CONTRACT_INFO_T
--工程合同信息
INSERT
  OVERWRITE TABLE DWIMD.DWI_MD_CONTRACT_INFO_T
SELECT
  SDI.ROW_ID AS ROW_ID,
  SDI.UPDATE_ID AS UPDATE_ID,
  SDI.STLMEN_TEMS AS STLMEN_TEMS,
  SDI.SET_TYPE AS SET_TYPE,
  SDI.SET_CYCLE AS SET_CYCLE,
  SDI.QUOTA_ID AS QUOTA_ID,
  SDI.PRO_REP_ID AS PRO_REP_ID,
  SDI.PAY_TYPE AS PAY_TYPE,
  SDI.DB_LAST_UPD_SRC AS DB_LAST_UPD_SRC,
  SDI.CREAT_ID AS CREAT_ID,
  SDI.CONTAT_ORG AS CONTAT_ORG,
  SDI.CHECK_TYPE AS CHECK_TYPE,
  SDI.ACC_BILL AS ACC_BILL,
  SDI.WAR_PER AS WAR_PER,
  SDI.TICK_TIME AS TICK_TIME,
  SDI.SET_START_DATE AS SET_START_DATE,
  SDI.SETTLE_PAY AS SETTLE_PAY,
  SDI.RECON_TIME AS RECON_TIME,
  SDI.PROC_PAY AS PROC_PAY,
  SDI.PAY_TIME AS PAY_TIME,
  CASE
    WHEN SDI.GUATE_DUE IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.GUATE_DUE
  END AS GUATE_DUE,
  CASE
    WHEN SDI.GET_CONTAT_DATE IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.GET_CONTAT_DATE
  END AS GET_CONTAT_DATE,
  SDI.DEPOSIT AS DEPOSIT,
  CASE
    WHEN SDI.DB_LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.DB_LAST_UPD
  END AS DB_LAST_UPD_DATE,
  CASE
    WHEN SDI.COM_DATA IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.COM_DATA
  END AS COM_DATA,
  SDI.CAL_FLAG AS CAL_FLAG,
  SDI.CONFLICT_ID AS CONFLICT_ID,
  SDI.MODIFICATION_NUM AS MODIFICATION_NUM,
  SDI.LAST_UPD_BY AS LAST_UPD_BY,
  CASE
    WHEN SDI.LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.LAST_UPD
  END AS LAST_UPD_DATE,
  SDI.CREATED_BY AS CREATED_BY,
  CASE
    WHEN SDI.CREATED IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.CREATED
  END AS CREATED_DATE,
  SDI.CONFIRM_BACK AS CONFIRM_BACK,
  SDI.X_FINE AS FINE,
  CASE
    WHEN SDI.ACTUAL_DATE IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.ACTUAL_DATE
  END AS ACTUAL_DATE,
  SDI.DP_PRODUCT_INSERT AS DP_PRODUCT_INSERT,
  SDI.X_SUM_FINE AS SUM_FINE,
  SDI.X_REFUND AS REFUND,
  SDI.X_MON_FINE AS MON_FINE,
  SDI.X_HR_RESULT AS HR_RESULT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_CONTAT_INFO_1002 SDI
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.UPDATE_ID,
  DWI.STLMEN_TEMS,
  DWI.SET_TYPE,
  DWI.SET_CYCLE,
  DWI.QUOTA_ID,
  DWI.PRO_REP_ID,
  DWI.PAY_TYPE,
  DWI.DB_LAST_UPD_SRC,
  DWI.CREAT_ID,
  DWI.CONTAT_ORG,
  DWI.CHECK_TYPE,
  DWI.ACC_BILL,
  DWI.WAR_PER,
  DWI.TICK_TIME,
  DWI.SET_START_DATE,
  DWI.SETTLE_PAY,
  DWI.RECON_TIME,
  DWI.PROC_PAY,
  DWI.PAY_TIME,
  DWI.GUATE_DUE,
  DWI.GET_CONTAT_DATE,
  DWI.DEPOSIT,
  DWI.DB_LAST_UPD_DATE,
  DWI.COM_DATA,
  DWI.CAL_FLAG,
  DWI.CONFLICT_ID,
  DWI.MODIFICATION_NUM,
  DWI.LAST_UPD_BY,
  DWI.LAST_UPD_DATE,
  DWI.CREATED_BY,
  DWI.CREATED_DATE,
  DWI.CONFIRM_BACK,
  DWI.FINE,
  DWI.ACTUAL_DATE,
  DWI.DP_PRODUCT_INSERT,
  DWI.SUM_FINE,
  DWI.REFUND,
  DWI.MON_FINE,
  DWI.HR_RESULT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIMD.DWI_MD_CONTRACT_INFO_T DWI
  LEFT JOIN SDI.SDI_CX_CONTAT_INFO_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
