--SDI_CX_COMPANY_CODE_1002 到 DWI_MD_COMPANY_INFO_T
--OSS公司信息表
INSERT
  OVERWRITE TABLE DWIMD.DWI_MD_COMPANY_INFO_T
SELECT
  SDI.ROW_ID AS ROW_ID,
  SDI.DB_LAST_UPD_SRC AS DB_LAST_UPD_SRC,
  SDI.COMPANY_CODE_NAME AS COMPANY_CODE_NAME,
  SDI.COMPANY_CODE AS COMPANY_CODE,
  CASE
    WHEN SDI.DB_LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.DB_LAST_UPD
  END AS DB_LAST_UPD_DATE,
  SDI.CONFLICT_ID AS CONFLICT_ID,
  SDI.MODIFICATION_NUM AS MODIFICATION_NUM,
  SDI.LAST_UPD_BY AS LAST_UPD_BY,
  CASE
    WHEN SDI.LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.LAST_UPD
  END AS LAST_UPD_DATE,
  SDI.CREATED_BY AS CREATED_BY,
  CASE
    WHEN SDI.CREATED IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.CREATED
  END AS CREATED_DATE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_COMPANY_CODE_1002 SDI
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.DB_LAST_UPD_SRC,
  DWI.COMPANY_CODE_NAME,
  DWI.COMPANY_CODE,
  DWI.DB_LAST_UPD_DATE,
  DWI.CONFLICT_ID,
  DWI.MODIFICATION_NUM,
  DWI.LAST_UPD_BY,
  DWI.LAST_UPD_DATE,
  DWI.CREATED_BY,
  DWI.CREATED_DATE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIMD.DWI_MD_COMPANY_INFO_T DWI
  LEFT JOIN SDI.SDI_CX_COMPANY_CODE_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
