WITH Y_FALG_DATA AS(  -- 源表数据删了，打Y标识的数据
	SELECT 
	   T1.GROUP_NUM,	--DWI集团
	   T1.FACTORY_AREA_CODE,	--DWI产区编码
	   T1.FACTORY_WERKS_CODE,	--DWI工厂编码
	   T1.FACTORY_WERKS2_CODE,	--DWI工厂编码
	   T1.SOURCE_MATERIAL_NUM,	--DWI源物料编号
	   'Y' AS DEL_FLAG			--打'Y'标识
	FROM DWIOTD.DWI_PS_MATERIAL_AREA_BIO_T T1	--DWI.SAP物料产区归属表
	LEFT JOIN SDI.SDI_ZTWLCQGSH_1001 T2		--SDI.SAP物料产区归属表
		ON T1.GROUP_NUM = T2.MANDT 		--DWI.集团 = SDI源表字段集团							
		AND T1.FACTORY_AREA_CODE = T2.ZCQBM 	--DWI.产区编码 = SDI源表字段产区编码
		AND T1.FACTORY_WERKS_CODE = T2.WERKS	--DWI.工厂编码 = SDI源表字段工厂编码
		AND T1.FACTORY_WERKS2_CODE = T2.WERKS2	--DWI.工厂编码 = SDI源表字段工厂编码
		AND T1.SOURCE_MATERIAL_NUM = T2.MATNR	--DWI.源物料编号 = SDI源表字段源物料编号
	WHERE T2.MANDT  IS NULL		--SDI源表字段集团为空
		AND T2.ZCQBM  IS NULL	--SDI源表字段产区编码为空
		AND T2.WERKS  IS NULL	--SDI源表字段工厂编码为空		
		AND T2.WERKS2  IS NULL	--SDI源表字段工厂编码为空
		AND T2.MATNR  IS NULL	--SDI源表字段源物料编号为空	
		AND T2.DEL_FLAG = 'N'	--删除标识为'N'
),
SDI_AND_Y_FLAG_DATA AS( -- SDI与打Y标识数据
   SELECT	--查询
	   SZ1.MANDT    AS GROUP_NUM,		--SDI源表字段集团 AS DWI集团
	   SZ1.ZCQBM    AS FACTORY_AREA_CODE,	--SDI源表字段产区编码 AS DWI产区编码
	   SZ1.WERKS    AS FACTORY_WERKS_CODE,	--SDI源表字段工厂编码 AS DWI工厂编码
	   SZ1.WERKS2   AS FACTORY_WERKS2_CODE,	--SDI源表字段工厂编码 AS DWI工厂编码
	   SZ1.MATNR    AS SOURCE_MATERIAL_NUM,	--SDI源表字段源物料编号 AS DWI源物料编号
	   SZ1.DEL_FLAG AS  DEL_FLAG		--删除标识
   FROM SDI.SDI_ZTWLCQGSH_1001 SZ1		--SDI.SAP物料产区归属表
   UNION ALL					--SDI_ZTWLCQGSH_1001SAP物料产区归属表 UNION ALL DWI_PS_MATERIAL_AREA_BIO_T.SAP物料产区归属表
   SELECT		--查询
       T1.GROUP_NUM,	--集团
	   T1.FACTORY_AREA_CODE,	--产区编码
	   T1.FACTORY_WERKS_CODE,	--工厂编码
	   T1.FACTORY_WERKS2_CODE,	--工厂编码
	   T1.SOURCE_MATERIAL_NUM,	--源物料编号
	   T1.DEL_FLAG			--删除标识
   FROM Y_FALG_DATA T1		
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PS_MATERIAL_AREA_BIO_T	--DWI.SAP物料产区归属表
SELECT      --- 关联主数据表
    SZ1.GROUP_NUM        			AS  GROUP_NUM,		--集团
	SZ1.FACTORY_AREA_CODE			AS  FACTORY_AREA_CODE,	--产区编码
	CASE WHEN SZ1.FACTORY_WERKS_CODE IS NULL OR SZ1.FACTORY_WERKS_CODE = ''
	THEN -999999	--关联工厂主数据取工厂ID,判断SDI源表字段为空或者空字符串时赋值-999999
    ELSE NVL(DMF.FACTORY_ID, -999998)	--关联不上主数据赋值-999998
    END 						AS  FACTORY_WERKS_ID, 	--工厂ID
	CASE WHEN SZ1.FACTORY_WERKS_CODE IS NULL OR SZ1.FACTORY_WERKS_CODE = ''
	THEN 'SNULL'	--关联工厂主数据,判断SDI源表字段为空或者空字符串时赋值'SNULL'
	ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
	END 						AS  FACTORY_WERKS_CODE,	--工厂编码
	CASE WHEN SZ1.FACTORY_WERKS2_CODE IS NULL OR SZ1.FACTORY_WERKS2_CODE = ''
	THEN -999999	--关联工厂主数据取工厂ID,判断SDI源表字段为空或者空字符串时赋值-999999
    ELSE NVL(DMF1.FACTORY_ID, -999998)	--关联不上主数据赋值-999998
    END 						AS  FACTORY_WERKS2_ID, 	--工厂ID
	CASE WHEN SZ1.FACTORY_WERKS2_CODE IS NULL OR SZ1.FACTORY_WERKS2_CODE = ''
	THEN 'SNULL'	--关联工厂主数据,判断SDI源表字段为空或者空字符串时赋值'SNULL'
	ELSE NVL(DMF1.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
	END 						AS  FACTORY_WERKS2_CODE,	--工厂编码
	CASE WHEN SZ1.SOURCE_MATERIAL_NUM IS NULL OR SZ1.SOURCE_MATERIAL_NUM = '' 
	THEN -999999	--关联物料主数据取工厂ID,判断SDI源表字段为空或者空字符串时赋值-999999
    ELSE NVL(DMMG.MATERIAL_ID, -999998)	--关联不上主数据赋值-999998
    END 						AS  MATERIAL_ID,    --物料ID
	CASE WHEN SZ1.SOURCE_MATERIAL_NUM IS NULL OR SZ1.SOURCE_MATERIAL_NUM = ''
	THEN 'SNULL'	--关联物料主数据,判断SDI源表字段为空或者空字符串时赋值'SNULL'
	ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
	END 						AS  MATERIAL_NUM,	--物料编号
	SZ1.SOURCE_MATERIAL_NUM					AS  SOURCE_MATERIAL_NUM,	--源物料编号
	SZ1.DEL_FLAG     				AS  DEL_FLAG,	--删除标识
	CURRENT_TIMESTAMP() 		AS  DW_LAST_UPDATE_DATE	--DW最后更新时间
FROM SDI_AND_Y_FLAG_DATA SZ1		--SDI.SAP物料产区归属表
LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF	--关联工厂主数据取工厂ID
   ON  SZ1.GROUP_NUM = DMF.GROUP_NUM	--集团
   AND SZ1.FACTORY_WERKS_CODE = DMF.FACTORY_CODE	--工厂编码
   AND DMF.END_DATE = '4712-12-31'	--结束时间为'4712-12-31'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF1	--关联工厂主数据工厂ID
   ON  SZ1.GROUP_NUM = DMF1.GROUP_NUM	--集团
   AND SZ1.FACTORY_WERKS2_CODE = DMF1.FACTORY_CODE	--工厂编码
   AND DMF1.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据取物料ID
   ON  SZ1.GROUP_NUM = DMMG.GROUP_NUM		--集团
   AND SZ1.SOURCE_MATERIAL_NUM = DMMG.MATERIAL_NUM	--物料编码
   AND DMMG.END_DATE = '4712-12-31'		--结束时间为'4712-12-31'
