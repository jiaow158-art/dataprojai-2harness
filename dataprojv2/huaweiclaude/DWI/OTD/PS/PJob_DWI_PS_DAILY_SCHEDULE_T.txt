--SDI_APS_DAILY_SCHEDULE_1007 到 DWI_PS_DAILY_SCHEDULE_T
--生产日进度
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_DAILY_SCHEDULE_T
SELECT
  SRC.ID AS ID,
  SRC.BATCH AS BATCH,
  SRC.COST_HOUR AS COST_HOUR,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.DATA_SOURCE AS DATA_SOURCE,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.INVENTORY_LOCATION AS INVENTORY_LOCATION,
  SRC.PLAN_TASK_ID AS PLAN_TASK_ID,
  CASE
    WHEN SRC.PLANT IS NULL
    OR SRC.PLANT = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  CASE
    WHEN SRC.PLANT IS NULL
    OR SRC.PLANT = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  SRC.PROCESS_TECH_ORDER_ID AS PROCESS_TECH_ORDER_ID,
  SRC.PRODUCTION_DATE AS PRODUCTION_DATE,
  SRC.QTY AS QTY,
  SRC.SAM AS SAM,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.WIP_RESOURCE_ID AS WIP_RESOURCE_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_APS_DAILY_SCHEDULE_1007 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.PLANT = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.BATCH,
  DWI.COST_HOUR,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.DATA_SOURCE,
  DWI.INV_ORG_ID,
  DWI.INVENTORY_LOCATION,
  DWI.PLAN_TASK_ID,
  DWI.FACTORY_CODE,
  DWI.FACTORY_ID,
  DWI.PROCESS_TECH_ORDER_ID,
  DWI.PRODUCTION_DATE,
  DWI.QTY,
  DWI.SAM,
  DWI.SYNC_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.WIP_RESOURCE_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_DAILY_SCHEDULE_T DWI
  LEFT JOIN SDI.SDI_APS_DAILY_SCHEDULE_1007 SRC ON DWI.ID = SRC.ID
WHERE
  SRC.ID IS NULL;
