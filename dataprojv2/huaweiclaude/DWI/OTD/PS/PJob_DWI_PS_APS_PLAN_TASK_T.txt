--UPDATE FROM HUANGWENBIN 20220414 新增了 CLOSE_TIME,PREPARE_CLOSE_FLAG
--计划任务
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_APS_PLAN_TASK_T
SELECT
  SRC.ID AS ID,
  SRC.ACTUAL_START_TIME AS ACTUAL_START_TIME,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.DURATION AS DURATION,
  SRC.END_TIME AS END_TIME,
  SRC.FRONT_TRANSFER_VALUE AS FRONT_TRANSFER_VALUE,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_ALERT AS IS_ALERT,
  SRC.IS_CANCEL AS IS_CANCEL,
  SRC.IS_COMBINATION AS IS_COMBINATION,
  SRC.IS_LOCKED AS IS_LOCKED,
  SRC.IS_PROGRESS_COMPLETE AS IS_PROGRESS_COMPLETE,
  SRC.IS_SAME_MODE AS IS_SAME_MODE,
  SRC.PLAN_NO AS PLAN_NO,
  SRC.RESOURCE_ID AS RESOURCE_ID,
  SRC.START_TIME AS START_TIME,
  SRC.STATUS AS STATUS,
  SRC.SUB_RES_STRATEGYS_ID AS SUB_RES_STRATEGYS_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.UPLOAD_REVIEW AS UPLOAD_REVIEW,
  SRC.IS_RE_UPLOAD AS IS_RE_UPLOAD,
  SRC.RE_DATE AS RE_DATE,
  SRC.RE_STATE AS RE_STATE,
  SRC.IS_MANUAL_MODIFY AS IS_MANUAL_MODIFY,
  SRC.CLOSE_FLAG AS CLOSE_FLAG,
  SRC.CLOSE_QTY AS CLOSE_QTY,
  SRC.OPEN_FLAG AS OPEN_FLAG,
  SRC.MUST_OUT_MADE AS MUST_OUT_MADE,
  SRC.APPOINT_RESOURCE_ID AS APPOINT_RESOURCE_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  SRC.CLOSE_TIME,
  SRC.PREPARE_CLOSE_FLAG
FROM
  SDI.SDI_APS_PLAN_TASK_1007 SRC 
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.ACTUAL_START_TIME,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.DURATION,
  DWI.END_TIME,
  DWI.FRONT_TRANSFER_VALUE,
  DWI.INV_ORG_ID,
  DWI.IS_ALERT,
  DWI.IS_CANCEL,
  DWI.IS_COMBINATION,
  DWI.IS_LOCKED,
  DWI.IS_PROGRESS_COMPLETE,
  DWI.IS_SAME_MODE,
  DWI.PLAN_NO,
  DWI.RESOURCE_ID,
  DWI.START_TIME,
  DWI.STATUS,
  DWI.SUB_RES_STRATEGYS_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  DWI.UPLOAD_REVIEW,
  DWI.IS_RE_UPLOAD,
  DWI.RE_DATE,
  DWI.RE_STATE,
  DWI.IS_MANUAL_MODIFY,
  DWI.CLOSE_FLAG,
  DWI.CLOSE_QTY,
  DWI.OPEN_FLAG,
  DWI.MUST_OUT_MADE,
  DWI.APPOINT_RESOURCE_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  DWI.CLOSE_TIME,
  DWI.PREPARE_CLOSE_FLAG
FROM
  DWIOTD.DWI_PS_APS_PLAN_TASK_T DWI
  LEFT JOIN SDI.SDI_APS_PLAN_TASK_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
