--SDI_APS_PRO_ORDER_1007 到 DWI_PS_APS_PRO_ORDER_T
--生产订单
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_APS_PRO_ORDER_T
SELECT
  SRC.ID AS ID,
  SRC.AREA AS ACREAGE,
  SRC.CANCEL_QTY AS CANCEL_QTY,
  SRC.CHANGE_TYPE AS CHANGE_TYPE,
  SRC.CODE AS ORDER_NUM,
  SRC.COMPLETE_QTY AS COMPLETE_QTY,
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.CUSTOMER_ID AS SRC_CUST_ID,
  --源系统客户ID
  --取主数据ID,源表关联字段为NULL,赋值-999999,源表关联字段有数据关联不上主数据表,赋值-999998
  CASE
    WHEN SRC.CUSTOMER_ID IS NULL
    OR SRC.CUSTOMER_ID = '' THEN -999999
    ELSE NVL(TMP.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SRC.CUSTOMER_ID IS NULL
    OR SRC.CUSTOMER_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编码
  SRC.DATA_SOURCE AS DATA_SOURCE,
  SRC.FACTORY_ID AS FACTORY_ID,
  SRC.FIRST_FILL AS FIRST_FILL,
  SRC.GROUP_ AS GROUP_NUM,
  SRC.GROUP_WEIGHTED_SEQUENCE AS GROUP_WEIGHTED_SEQUENCE,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_CANCELED AS IS_CANCELED,
  SRC.IS_CUSTOM AS IS_CUSTOM,
  SRC.IS_LOCKED AS IS_LOCKED,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.IS_SUPPLEMENT AS IS_SUPPLEMENT,
  SRC.IS_WHOLE AS IS_WHOLE,
  NVL(SRC.ITEM_ID, -999999) AS SRC_MATERIAL_ID,
  --源系统物料ID
  --取主数据取ID,源表关联字段为NULL,赋值-999999,源表关联字段有数据关联不上主数据表,赋值-999998
  CASE
    WHEN SRC.ITEM_ID IS NULL
    OR SRC.ITEM_ID = '' THEN -999999
    ELSE NVL(TMP1.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.ITEM_ID IS NULL
    OR SRC.ITEM_ID = '' THEN 'SNULL'
    ELSE NVL(TMP1.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.ITEM_NAME AS MATERIAL_NAME,
  SRC.MERGE_PRODUCT_ORDER_ID AS MERGE_PRODUCT_ORDER_ID,
  SRC.OEM_PURCHASE_ORDER_NO AS OEM_PURCHASE_ORDER_NO,
  SRC.ORDER_DATE AS ORDER_DATE,
  SRC.ORDER_STATE AS ORDER_STATE,
  SRC.ORDER_TYPE_SET_ID AS ORDER_TYPE_SET_ID,
  SRC.PLANT_CLASS AS PLANT_CLASS,
  SRC.PRIORITY AS PRIORITY,
  SRC.PROCESS_TECH_ROUTE_ID AS PROCESS_TECH_ROUTE_ID,
  SRC.PROMISE_DELIVERY AS PROMISE_DELIVERY,
  SRC.QTY AS QTY,
  SRC.RE_DATE AS RE_DATE,
  SRC.REAL_STATUS AS REAL_STATUS,
  SRC.RELEVANT_CODE AS RELEVANT_CODE,
  SRC.REQUIRE_DELIVERY AS REQUIRE_DELIVERY,
  SRC.REVIEW_STATUS AS REVIEW_STATUS,
  SRC.SALE_LINE_NO AS SALE_LINE_NO,
  SRC.SALE_NO AS SALE_NO,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.TOP_ORDER_ID AS TOP_ORDER_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  SRC.WEIGHTED AS WEIGHTED,
  SRC.WEIGHTED_SEQUENCE AS WEIGHTED_SEQUENCE,
  SRC.COMPLETE_TIME AS COMPLETE_TIME,
  SRC.GROUP_WEIGHTED AS GROUP_WEIGHTED,
  SRC.REMARK AS REMARK,
  SRC.IS_DELAY AS IS_DELAY,
  SRC.DELAY_DAYS AS DELAY_DAYS,
  SRC.IS_FINISH_CALCULATE AS IS_FINISH_CALCULATE,
  SRC.PLAN_START_TIME AS PLAN_START_TIME,
  SRC.PLAN_END_TIME AS PLAN_END_TIME,
  SRC.PLAN_STATUS AS PLAN_STATUS,
  SRC.REPORT_START_TIME AS REPORT_START_TIME,
  SRC.REPORT_END_TIME AS REPORT_END_TIME,
  SRC.RELEVANT_ORDER_NO AS RELEVANT_ORDER_NO,
  SRC.RE_STATE AS RE_STATE,
  SRC.SPECIAL_REQUIREMENT AS SPECIAL_REQUIREMENT,
  SRC.SPECIAL_PACKAGING AS SPECIAL_PACKAGING,
  SRC.PACKAGING_CODE AS PACKAGING_CODE,
  SRC.SPECIAL_THICKNESS AS SPECIAL_THICKNESS,
  SRC.COLORS AS COLORS,
  SRC.CUSTOMIZED_REQUIREMENTS AS CUSTOMIZED_REQUIREMENTS,
  SRC.IS_SPECIAL_STOCK AS IS_SPECIAL_STOCK,
  SRC.SELF_FOREIGN_FLAG AS SELF_FOREIGN_FLAG,
  SRC.IS_SUITABLE_PRICE AS IS_SUITABLE_PRICE,
  SRC.NOT_MERGE_FLAG AS NOT_MERGE_FLAG,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATEAS,
  SRC.CLOSE_QTY
FROM
  SDI.SDI_APS_PRO_ORDER_1007 SRC --源系统ID LEFT JOIN (编码表和主数据表用编码关联取出ID,编码)
  LEFT JOIN (
    SELECT
      T1.ID,
      T2.CUST_ID,
      --客户ID
      T2.CUST_NUM --客户编码
    FROM
      SDI.SDI_CSM_CUSTOMER_1007 T1 --客户编码表
      INNER JOIN DWIMD.DWI_MD_CUST_GENERAL_T T2 --客户主数据表
      ON T1.CODE = T2.CUST_NUM
      AND T2.END_DATE = '4712-12-31'
      AND T2.DEL_FLAG = 'N'
      AND T1.IS_PHANTOM = 0 --取有效属性数据
      AND T2.GROUP_NUM = '800'
  ) TMP ON SRC.CUSTOMER_ID = TMP.ID
  LEFT JOIN (
    SELECT
      M1.ID,
      M2.MATERIAL_ID,
      --物料ID
      M2.MATERIAL_NUM --物料编码
    FROM
      SDI.SDI_ITEM_1007 M1 --物料编码表
      INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T M2 --物料主数据表
      ON M1.CODE = M2.MATERIAL_NUM
      AND M2.END_DATE = '4712-12-31'
      AND M2.DEL_FLAG = 'N'
      AND M1.IS_PHANTOM = 0 --取有效属性数据
      AND M2.GROUP_NUM = '800'
  ) TMP1 ON SRC.ITEM_ID = TMP1.ID
;
