--SDI_CSM_CUSTOMER_1007 到 DWI_PS_CSM_CUSTOMER_T
--客户
INSERT
  OVERWRITE TABLE DWIOTD.DWI_PS_CSM_CUSTOMER_T
SELECT
  SRC.ID AS ID,
  CASE
    WHEN SRC.CODE IS NULL
    OR SRC.CODE = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SRC.CODE IS NULL
    OR SRC.CODE = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编码
  SRC.CREATE_BY AS CREATE_BY,
  SRC.CREATE_DATE AS CREATE_DATE,
  SRC.CUSTOMER_TYPE AS CUST_TYPE,
  SRC.DESCRIPTION AS DESCRIPTION,
  SRC.INV_ORG_ID AS INV_ORG_ID,
  SRC.IS_PHANTOM AS IS_PHANTOM,
  SRC.NAME AS NAME,
  SRC.SHORT_NAME AS SHORT_NAME,
  SRC.SOURCE_KEY AS SOURCE_KEY,
  SRC.SOURCE_TYPE AS SOURCE_TYPE,
  SRC.STATE AS STATE,
  SRC.SYNC_ID AS SYNC_ID,
  SRC.UPDATE_BY AS UPDATE_BY,
  SRC.UPDATE_DATE AS UPDATE_DATE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CSM_CUSTOMER_1007 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.CODE = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.CREATE_BY,
  DWI.CREATE_DATE,
  DWI.CUST_TYPE,
  DWI.DESCRIPTION,
  DWI.INV_ORG_ID,
  DWI.IS_PHANTOM,
  DWI.NAME,
  DWI.SHORT_NAME,
  DWI.SOURCE_KEY,
  DWI.SOURCE_TYPE,
  DWI.STATE,
  DWI.SYNC_ID,
  DWI.UPDATE_BY,
  DWI.UPDATE_DATE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_PS_CSM_CUSTOMER_T DWI
  LEFT JOIN SDI.SDI_CSM_CUSTOMER_1007 SDI ON DWI.ID = SDI.ID
WHERE
  SDI.ID IS NULL;
