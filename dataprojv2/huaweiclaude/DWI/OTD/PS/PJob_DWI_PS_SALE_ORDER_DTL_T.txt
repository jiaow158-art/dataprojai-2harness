      SDI.SDI_ITEM_1007 M --物料编码表

      AND MAT.END_DATE = '4712-12-31'
      AND MAT.DEL_FLAG = 'N'
      AND MAT.GROUP_NUM = '800'
  ) TMP ON SRC.ITEM_ID = TMP.ID
