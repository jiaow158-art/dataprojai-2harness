--SDI_CX_WH_BATCH_1002 到 DWI_SO_WH_BATCH_OSS_T
--库存批次表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_WH_BATCH_OSS_T
SELECT
  SRC.ROW_ID,
  SRC.ORDER_ITEM_ID,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.CREATED,
  SRC.AVAIL_QUAN,
  SRC.BATCH,
  SRC.ORDER_QTY_NUM,
  SRC.MD_ORDER_ITEM_ID,
  SRC.MODIFICATION_NUM,
  SRC.STATUS,
  SRC.LAST_UPD_BY,
  SRC.CREATED_BY,
  SRC.DB_LAST_UPD_SRC,
  SRC.CONFLICT_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_WH_BATCH_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.ORDER_ITEM_ID,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.CREATE_DATE,
  DWI.AVAIL_QTY,
  DWI.BATCH,
  DWI.ORDER_QTY_NUM,
  DWI.MD_ORDER_ITEM_ID,
  DWI.MODIFY_QTY,
  DWI.STATUS,
  DWI.LAST_UPDATE_BY,
  DWI.CREATED_BY,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.CONFLICT_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_WH_BATCH_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_WH_BATCH_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
