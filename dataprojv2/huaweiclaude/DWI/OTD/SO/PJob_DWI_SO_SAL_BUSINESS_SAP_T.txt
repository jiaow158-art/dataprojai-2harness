--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售凭证业务数据表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SAL_BUSINESS_SAP_T
SELECT
  SV.MANDT AS GROUP_NUM,
  SV.VBELN AS SAL_VCH_NUM,
  SV.POSNR AS SAL_VCH_ITEM_NUM,
  SV.INCO2_L AS INTER_TRADE_CLAUSE_LOC_1,
  SV.KURSK AS DECISION_PRICE_RATE,
  SV.GJAHR AS FISCAL_YEAR,
  SV.VSART AS SHIPPING_TYPE,
  SV.BSTKD_M AS CUST_PO_MATCH_FIELD,
  CASE
    WHEN SV.FKDAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.FKDAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.FKDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS BILL_DATE,
  SV.KONDA AS CUST_PRICE_GROUP,
  CASE
    WHEN SV.KURSK_DAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.KURSK_DAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(
      UNIX_TIMESTAMP(SV.KURSK_DAT, 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
  END AS CONVERSION_DATE1,
  CASE
    WHEN SV.KURRF_DAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.KURRF_DAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(
      UNIX_TIMESTAMP(SV.KURRF_DAT, 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
  END AS CONVERSION_DATE2,
  SV.IHREZ AS YOUR_REFERENCE,
  SV.ZTERM AS PAY_TERM_CODE,
  SV.KDGRP AS CUST_GROUP,
  SV.KURRF AS FIN_ACC_POST_RATE,
  SV.STCUR AS STATISTICS_EX_RATE,
  CASE
    WHEN SV.PRSDT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.PRSDT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.PRSDT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS PRICE_DATE,
  SV.KZAZU AS ORDER_COMB_FLAG,
  SV.PODKZ AS RELATED_POD_PROCESSING,
  CASE
    WHEN SV.FBUDA = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.FBUDA, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.FBUDA, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS PROVIDE_SERVICE_DATE,
  SV.BSTKD AS CUST_REFERENCE_INFO,
  SV.KDKG1 AS CUST_CRITERIA_GROUP_1,
  SV.KDKG2 AS CUST_CRITERIA_GROUP_2,
  SV.KDKG3 AS CUST_CRITERIA_GROUP_3,
  SV.MRNKZ AS MANUAL_INVOICE_FLAG,
  SV.INCO1 AS INTER_TRADE_TERMS_P1,
  SV.INCO2 AS INTER_TRADE_TERMS_P2,
  SV.BSTKD_E AS ARRIVAL_PO_NUM,
  SV.IHREZ_E AS RECEIVER_INFO,
  SV.BZIRK AS SALE_REGION_CODE,
  SV.KTGRD AS CUST_ACCT_ASSIGN_GROUP,
  SV.FPLNR AS ISSUE_INVOICE_PLAN_NUM,
  SV.ABSSC AS PAYMENT_ASSURE_PROCEDURE,
  SV.BSARK AS CUST_PO_TYPE,
  CASE
    WHEN SV.BSTDK = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.BSTDK, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.BSTDK, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS CUST_REFERENCE_DATE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_VBKD_1001 SV 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  DWI.INTER_TRADE_CLAUSE_LOC_1 AS INTER_TRADE_CLAUSE_LOC_1,
  DWI.DECISION_PRICE_RATE AS DECISION_PRICE_RATE,
  DWI.FISCAL_YEAR AS FISCAL_YEAR,
  DWI.SHIPPING_TYPE AS SHIPPING_TYPE,
  DWI.CUST_PO_MATCH_FIELD AS CUST_PO_MATCH_FIELD,
  DWI.BILL_DATE AS BILL_DATE,
  DWI.CUST_PRICE_GROUP AS CUST_PRICE_GROUP,
  DWI.CONVERSION_DATE1 AS CONVERSION_DATE1,
  DWI.CONVERSION_DATE2 AS CONVERSION_DATE2,
  DWI.YOUR_REFERENCE AS YOUR_REFERENCE,
  DWI.PAY_TERM_CODE AS PAY_TERM_CODE,
  DWI.CUST_GROUP AS CUST_GROUP,
  DWI.FIN_ACC_POST_RATE AS FIN_ACC_POST_RATE,
  DWI.STATISTICS_EX_RATE AS STATISTICS_EX_RATE,
  DWI.PRICE_DATE AS PRICE_DATE,
  DWI.ORDER_COMB_FLAG AS ORDER_COMB_FLAG,
  DWI.RELATED_POD_PROCESSING AS RELATED_POD_PROCESSING,
  DWI.PROVIDE_SERVICE_DATE AS PROVIDE_SERVICE_DATE,
  DWI.CUST_REFERENCE_INFO AS CUST_REFERENCE_INFO,
  DWI.CUST_CRITERIA_GROUP_1 AS CUST_CRITERIA_GROUP_1,
  DWI.CUST_CRITERIA_GROUP_2 AS CUST_CRITERIA_GROUP_2,
  DWI.CUST_CRITERIA_GROUP_3 AS CUST_CRITERIA_GROUP_3,
  DWI.MANUAL_INVOICE_FLAG AS MANUAL_INVOICE_FLAG,
  DWI.INTER_TRADE_TERMS_P1 AS INTER_TRADE_TERMS_P1,
  DWI.INTER_TRADE_TERMS_P2 AS INTER_TRADE_TERMS_P2,
  DWI.ARRIVAL_PO_NUM AS ARRIVAL_PO_NUM,
  DWI.RECEIVER_INFO AS RECEIVER_INFO,
  DWI.SALE_REGION_CODE AS SALE_REGION_CODE,
  DWI.CUST_ACCT_ASSIGN_GROUP AS CUST_ACCT_ASSIGN_GROUP,
  DWI.ISSUE_INVOICE_PLAN_NUM AS ISSUE_INVOICE_PLAN_NUM,
  DWI.PAYMENT_ASSURE_PROCEDURE AS PAYMENT_ASSURE_PROCEDURE,
  DWI.CUST_PO_TYPE AS CUST_PO_TYPE,
  DWI.CUST_REFERENCE_DATE AS CUST_REFERENCE_DATE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SAL_BUSINESS_SAP_T DWI
  LEFT JOIN SDI.SDI_VBKD_1001 SV ON DWI.GROUP_NUM = SV.MANDT
  AND DWI.SAL_VCH_NUM = SV.VBELN
  AND DWI.SAL_VCH_ITEM_NUM = SV.POSNR
WHERE
  SV.VBELN IS NULL;
