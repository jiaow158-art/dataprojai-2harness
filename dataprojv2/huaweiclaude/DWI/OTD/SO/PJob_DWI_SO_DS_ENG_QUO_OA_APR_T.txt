--SDI_FORMTABLE_MAIN_519_1004 到 DWI_SO_DS_ENG_QUO_OA_APR_T
--SBC-DCM-A5-004-工程询价报价流程主表(直营子公司)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_DS_ENG_QUO_OA_APR_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.REQUESTID AS REQUEST_ID,
  SRC.LSH AS OA_CODE,
  SRC.BJDH AS SAP_QUOTE_NUM,
  SRC.FJ AS ANNEX_ID,
  SRC.XMMC AS PROJECT_NAME,
  SRC.SXE AS CREDIT_AMT,
  SRC.GCBBH AS PROJ_REP_ID,
  SRC.GCDZ AS PROJ_ADDRESS_NAME,
  SRC.GCMC AS PROJ_NAME,
  SRC.JSFMC AS BUILDING_NAME,
  SRC.FKF AS PAYER_NUM,
  SRC.CGFMC AS PURCHASER_NAME,
  SRC.WBS AS PROJ_WBS,
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编码
  NVL(SRC.KHMC, 'SNULL') AS SRC_CUST_NAME,
  SRC.AMBZDYK AS WHOLE_GAIN_LOSS_AMT,
  SRC.AMBZDMLL AS WHOLE_RATE,
  SRC.BBYXRQ AS REPORT_VALID_DATE,
  SRC.YGYL AS ESTIMATE_QTY,
  SRC.BJDYXQ AS QUOTE_VALID_DATE,
  SRC.SQRQ AS APPRV_DATE,
  SRC.BJXMGK AS QUOTE_PROJ_OVERVIEW,
  SRC.SQR AS APPLY_USER_NAME,
  SRC.DDPJY AS FILING_CLERK_ID,
  SRC.SQRS AS APPLY_USER_ID,
  SRC.JHDD AS DELIVERY_ADDRESS_NAME,
  SRC.FHXX AS RETURN_INFOR,
  SRC.PF AS SCORE,
  SRC.FKFS AS PAY_TYPE,
  SRC.LXR AS CONTACT,
  SRC.LXDH AS CONTACT_PHONE,
  SRC.KHZ AS CUST_GROUP,
  SRC.YSFS AS TRANSPORTATION_MODE,
  SRC.CGXS AS PUR_TYPE,
  SRC.ACT AS PROCESS_NODE,
  SRC.XSZZ AS SALE_ORG_CODE,
  SRC.GCLB1 AS ENGINEERING_CATEGORY,
  SRC.JSFPMHSGFDJ AS INDUSTRY_QUALIFICATION,
  SRC.JSTK AS SETTLE_TERM,
  SRC.QD AS CHANNEL,
  SRC.PDTJ AS JUDGE_CONDITION,
  SRC.BLHKJL AS BAD_RECOARD,
  SRC.XYDJ AS CREDIT_LEVEL,
  SRC.XYPG AS CREDIT_EVALUATION,
  SRC.FWPJ AS SERVICE_EVALUATION,
  SRC.QYLX AS BUSINESS_TYPE,
  SRC.SFQDHT AS IF_SIGN_CONTRACT_FLAG,
  SRC.ZCZBGM AS REGISTER_CAPITAL_SCALE,
  SRC.SFCJKB AS IF_CREATE_CUST_NUM_FLAG,
  SRC.SFCJWBS AS IF_CREATE_PROJ_WBS_FLAG,
  SRC.SFJG AS IF_PROCESS_FLAG,
  SRC.SFKFP AS IF_ISSUE_INVOICE_FLAG,
  SRC.SFQDZLXY AS IF_SIGN_STR_AGR_FLAG,
  SRC.SFWBQ AS IF_500TOP_FLAG,
  SRC.BT AS TITLE_NAME,
  SRC.SX AS PROPERTY,
  SRC.SYZRHPJS AS ACCEPTANCE_BILL_XCHG,
  SRC.YGHTJE AS ESTIMATE_CONTRACT_AMT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_519_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.KHBM = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.REQUEST_ID,
  DWI.OA_CODE,
  DWI.SAP_QUOTE_NUM,
  DWI.ANNEX_ID,
  DWI.PROJECT_NAME,
  DWI.CREDIT_AMT,
  DWI.PROJ_REP_ID,
  DWI.PROJ_ADDRESS_NAME,
  DWI.PROJ_NAME,
  DWI.BUILDING_NAME,
  DWI.PAYER_NUM,
  DWI.PURCHASER_NAME,
  DWI.PROJ_WBS,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.SRC_CUST_NAME,
  DWI.WHOLE_GAIN_LOSS_AMT,
  DWI.WHOLE_RATE,
  DWI.REPORT_VALID_DATE,
  DWI.ESTIMATE_QTY,
  DWI.QUOTE_VALID_DATE,
  DWI.APPRV_DATE,
  DWI.QUOTE_PROJ_OVERVIEW,
  DWI.APPLY_USER_NAME,
  DWI.FILING_CLERK_ID,
  DWI.APPLY_USER_ID,
  DWI.DELIVERY_ADDRESS_NAME,
  DWI.RETURN_INFOR,
  DWI.SCORE,
  DWI.PAY_TYPE,
  DWI.CONTACT,
  DWI.CONTACT_PHONE,
  DWI.CUST_GROUP,
  DWI.TRANSPORTATION_MODE,
  DWI.PUR_TYPE,
  DWI.PROCESS_NODE,
  DWI.SALE_ORG_CODE,
  DWI.ENGINEERING_CATEGORY,
  DWI.INDUSTRY_QUALIFICATION,
  DWI.SETTLE_TERM,
  DWI.CHANNEL,
  DWI.JUDGE_CONDITION,
  DWI.BAD_RECOARD,
  DWI.CREDIT_LEVEL,
  DWI.CREDIT_EVALUATION,
  DWI.SERVICE_EVALUATION,
  DWI.BUSINESS_TYPE,
  DWI.IF_SIGN_CONTRACT_FLAG,
  DWI.REGISTER_CAPITAL_SCALE,
  DWI.IF_CREATE_CUST_NUM_FLAG,
  DWI.IF_CREATE_PROJ_WBS_FLAG,
  DWI.IF_PROCESS_FLAG,
  DWI.IF_ISSUE_INVOICE_FLAG,
  DWI.IF_SIGN_STR_AGR_FLAG,
  DWI.IF_500TOP_FLAG,
  DWI.TITLE_NAME,
  DWI.PROPERTY,
  DWI.ACCEPTANCE_BILL_XCHG,
  DWI.ESTIMATE_CONTRACT_AMT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DS_ENG_QUO_OA_APR_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_519_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
