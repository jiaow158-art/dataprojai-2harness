--SDI_CX_PLAN_O_ITEM_1002 到 DWI_SO_PLAN_ORDER_ITEM_OSS_T
--OSS工程计划单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PLAN_ORDER_ITEM_OSS_T
SELECT
  SRC.ROW_ID AS ROW_ID,
  SRC.ITEM_NUM AS ITEM_NUM,
  SRC.CREATED AS CREATE_DATE,
  SRC.LAST_UPD AS LAST_UPDATE_DATE,
  SRC.DB_LAST_UPD AS DB_LAST_UPDATE_DATE,
  SRC.PLAN_ORDER_ID AS PLAN_ORDER_ID,
  SRC.PURCH_ITEM AS PURCH_ITEM,
  SRC.APPROVAL_QTY AS APPROVAL_QTY,
  SRC.APPLY_QTY AS APPLY_QTY,
  SRC.CREATED_ORDER_QTY AS CREATED_ORDER_QTY,
  SRC.FACTORY_ID AS SRC_FACTORY_ID,
  --源系统工厂ID
  CASE
    WHEN SRC.FACTORY_ID IS NULL
    OR SRC.FACTORY_ID = '' THEN -999999
    ELSE NVL(TMP.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.FACTORY_ID IS NULL
    OR SRC.FACTORY_ID = '' THEN 'SNULL'
    ELSE NVL(TMP.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  NVL(SRC.APPROVAL_FACTORY_ID, -999999) AS SRC_APPROV_FACT_ID,
  --源系统审批工厂ID
  CASE
    WHEN SRC.APPROVAL_FACTORY_ID IS NULL
    OR SRC.APPROVAL_FACTORY_ID = '' THEN -999999
    ELSE NVL(TMP.FACTORY_ID, -999998)
  END AS APPROV_FACT_ID,
  --审批工厂ID
  CASE
    WHEN SRC.APPROVAL_FACTORY_ID IS NULL
    OR SRC.APPROVAL_FACTORY_ID = '' THEN 'SNULL'
    ELSE NVL(TMP1.FACTORY_CODE, 'NOT EXISTS')
  END AS APPROV_FACT_CODE,
  --审批工厂编码
  SRC.AREA AS ACREAGE,
  SRC.UNIT AS UNIT,
  SRC.INTO_REVIEW_TIME AS INTO_REVIEW_TIME,
  SRC.PROD_ID AS PROD_ID,
  SRC.OA_CODE AS OA_CODE,
  SRC.APP_ENTER_DATE AS APP_ENTER_DATE,
  SRC.EXCEPT_DATE AS EXCEPT_DATE,
  SRC.DP_BASE_PRICE AS DP_BASE_PRICE,
  SRC.BATCH AS BATCH,
  SRC.DELIVERY_DATE AS DELIVERY_DATE,
  SRC.NUMBER_SET AS NUMBER_SET,
  SRC.SAP_MIN_RK_DATE AS SAP_MIN_RK_DATE,
  SRC.REVIEW_ID AS REVIEW_ID,
  SRC.DP_REFERENCE_ORDER AS DP_REFERENCE_ORDER,
  SRC.DP_GO_UP AS DP_GO_UP,
  SRC.AVL_AMT AS AVL_AMT,
  SRC.FLOOR_NUM AS FLOOR_NUM,
  SRC.DEFAULT_DELIVERY_TIME AS DEFAULT_DELIVERY_TIME,
  SRC.MARK_BILL_NUM AS MARK_BILL_NUM,
  SRC.HOUSE_TYPE AS HOUSE_TYPE,
  SRC.CANCEL_REASON AS CANCEL_REASON,
  SRC.DELAY_REASON AS DELAY_REASON,
  SRC.DP_BILL_SUBMIT_TIME AS DP_BILL_SUBMIT_TIME,
  SRC.X_ORDER_DESC AS ORDER_DESC,
  SRC.END_TIME_ITEM AS END_TIME_ITEM,
  SRC.MODIFICATION_NUM AS MODIFY_QTY,
  SRC.BUILDING_NUM AS BUILDING_NUM,
  SRC.DP_REVIEW_REMARK AS DP_REVIEW_REMARK,
  SRC.REVIEW_WAREHOUSE AS REVIEW_WAREHOUSE,
  SRC.WARE_ID AS WARE_ID,
  SRC.APPROVAL_INFO AS APPROVAL_INFO,
  SRC.SUBMIT_REMARK AS SUBMIT_REMARK,
  SRC.DP_PLAN_APS_REASON AS DP_PLAN_APS_REASON,
  SRC.DP_BILL_CUT_TIME AS DP_BILL_CUT_TIME,
  SRC.USE_AREA AS USE_AREA,
  SRC.TRANSPORTATION_TIME AS TRANSPORTATION_TIME,
  SRC.DP_REVIEW_PEOPLE AS DP_REVIEW_PEOPLE,
  SRC.STAUTS AS STAUTS,
  SRC.WORK_DAY AS WORK_DAY,
  SRC.DP_SUPPLY_AREA AS DP_SUPPLY_AREA,
  SRC.DP_MOQ_NUMBER AS DP_MOQ_NUMBER,
  SRC.SALES_ORG_ID AS SALES_ORG_ID,
  SRC.CHAN_ID AS CHAN_ID,
  SRC.DP_IS_MOQ AS DP_IS_MOQ,
  SRC.LAST_UPD_BY AS LAST_UPDATE_BY,
  SRC.SHIPPING_TYPE AS SHIPPING_TYPE,
  SRC.REVIEW_MESSAGE AS REVIEW_MESSAGE,
  SRC.CREATED_BY AS CREATED_BY,
  SRC.NEED_PROCESS AS NEED_PROCESS,
  SRC.BEFORE_REVOKE_STATUS AS BEFORE_REVOKE_STATUS,
  SRC.PLAN_PROPERTY AS PLAN_PROPERTY,
  SRC.DP_IS_SPECIAL AS DP_IS_SPECIAL,
  SRC.DP_IS_CUSTOMIZED AS DP_IS_CUSTOMIZED,
  SRC.DP_GO_UP_SCALE AS DP_GO_UP_SCALE,
  SRC.DP_IS_GO_UP AS DP_IS_GO_UP,
  SRC.DP_IS_PRODUCT AS DP_IS_PRODUCT,
  SRC.DP_PLAN_APS_FLAG AS DP_PLAN_APS_FLAG,
  SRC.DP_PLAN_DEPOSIT_RATIO AS DP_PLAN_DEPOSIT_RATIO,
  SRC.DP_IS_BILL AS DP_IS_BILL,
  SRC.DP_CANCEL_REASON AS DP_CANCEL_REASON,
  SRC.X_OPERA_FLAG AS OPERA_FLAG,
  SRC.CONFLICT_ID AS CONFLICT_ID,
  SRC.DB_LAST_UPD_SRC AS DB_LAST_UPDATE_DATE_SRC,
  SRC.OUTPUT_REVIEW_TIME AS OUTPUT_REVIEW_TIME,
  SRC.OUTPUT_REVIEWMAN_ID AS OUTPUT_REVIEWMAN_ID,
  SRC.REVIEW_PRODUCT_TIME AS REVIEW_PRODUCT_TIME,
  SRC.PRODUCT_FACTORY AS PRODUCT_FACTORY,
  SRC.REVIEW_PRODUCT_TIME2 AS REVIEW_PRODUCT_TIME2,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_PLAN_O_ITEM_1002 SRC
  LEFT JOIN (
    SELECT
      M1.ROW_ID,
      FAC.FACTORY_ID,
      --工厂ID
      FAC.FACTORY_CODE --工厂编码
    FROM
      SDI.SDI_CX_FACTORY_1002 M1 --工厂编码表
      INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
      ON M1.FACTORY_CD = FAC.FACTORY_CODE
      AND FAC.END_DATE = '4712-12-31'
      AND FAC.DEL_FLAG = 'N'
      AND FAC.GROUP_NUM = '800'
  ) TMP ON SRC.FACTORY_ID = TMP.ROW_ID
  LEFT JOIN (
    SELECT
      M1.ROW_ID,
      FAC.FACTORY_ID,
      --审批工厂ID
      FAC.FACTORY_CODE --审批工厂编码
    FROM
      SDI.SDI_CX_FACTORY_1002 M1 --工厂编码表
      INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
      ON M1.FACTORY_CD = FAC.FACTORY_CODE
      AND FAC.END_DATE = '4712-12-31'
      AND FAC.DEL_FLAG = 'N'
      AND FAC.GROUP_NUM = '800'
  ) TMP1 ON SRC.APPROVAL_FACTORY_ID = TMP1.ROW_ID
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.ITEM_NUM,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.PLAN_ORDER_ID,
  DWI.PURCH_ITEM,
  DWI.APPROVAL_QTY,
  DWI.APPLY_QTY,
  DWI.CREATED_ORDER_QTY,
  DWI.SRC_FACTORY_ID,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.SRC_APPROV_FACT_ID,
  DWI.APPROV_FACT_ID,
  DWI.APPROV_FACT_CODE,
  DWI.ACREAGE,
  DWI.UNIT,
  DWI.INTO_REVIEW_TIME,
  DWI.PROD_ID,
  DWI.OA_CODE,
  DWI.APP_ENTER_DATE,
  DWI.EXCEPT_DATE,
  DWI.DP_BASE_PRICE,
  DWI.BATCH,
  DWI.DELIVERY_DATE,
  DWI.NUMBER_SET,
  DWI.SAP_MIN_RK_DATE,
  DWI.REVIEW_ID,
  DWI.DP_REFERENCE_ORDER,
  DWI.DP_GO_UP,
  DWI.AVL_AMT,
  DWI.FLOOR_NUM,
  DWI.DEFAULT_DELIVERY_TIME,
  DWI.MARK_BILL_NUM,
  DWI.HOUSE_TYPE,
  DWI.CANCEL_REASON,
  DWI.DELAY_REASON,
  DWI.DP_BILL_SUBMIT_TIME,
  DWI.ORDER_DESC,
  DWI.END_TIME_ITEM,
  DWI.MODIFY_QTY,
  DWI.BUILDING_NUM,
  DWI.DP_REVIEW_REMARK,
  DWI.REVIEW_WAREHOUSE,
  DWI.WARE_ID,
  DWI.APPROVAL_INFO,
  DWI.SUBMIT_REMARK,
  DWI.DP_PLAN_APS_REASON,
  DWI.DP_BILL_CUT_TIME,
  DWI.USE_AREA,
  DWI.TRANSPORTATION_TIME,
  DWI.DP_REVIEW_PEOPLE,
  DWI.STAUTS,
  DWI.WORK_DAY,
  DWI.DP_SUPPLY_AREA,
  DWI.DP_MOQ_NUMBER,
  DWI.SALES_ORG_ID,
  DWI.CHAN_ID,
  DWI.DP_IS_MOQ,
  DWI.LAST_UPDATE_BY,
  DWI.SHIPPING_TYPE,
  DWI.REVIEW_MESSAGE,
  DWI.CREATED_BY,
  DWI.NEED_PROCESS,
  DWI.BEFORE_REVOKE_STATUS,
  DWI.PLAN_PROPERTY,
  DWI.DP_IS_SPECIAL,
  DWI.DP_IS_CUSTOMIZED,
  DWI.DP_GO_UP_SCALE,
  DWI.DP_IS_GO_UP,
  DWI.DP_IS_PRODUCT,
  DWI.DP_PLAN_APS_FLAG,
  DWI.DP_PLAN_DEPOSIT_RATIO,
  DWI.DP_IS_BILL,
  DWI.DP_CANCEL_REASON,
  DWI.OPERA_FLAG,
  DWI.CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.OUTPUT_REVIEW_TIME,
  DWI.OUTPUT_REVIEWMAN_ID,
  DWI.REVIEW_PRODUCT_TIME,
  DWI.PRODUCT_FACTORY,
  DWI.REVIEW_PRODUCT_TIME2,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PLAN_ORDER_ITEM_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_PLAN_O_ITEM_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
