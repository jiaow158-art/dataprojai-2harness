--DSS转储单头
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_TRANS_STORE_HEAD_DSS_T
SELECT
  SBTS.ID AS ID,
  SBTS.CODE AS CODE,
  SBTS.CREATE_TIME AS CREATE_TIME,
  SBTS.UPDATE_TIME AS UPDATE_TIME,
  SBTS.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  SBTS.SAP_CODE AS SAP_CODE,
  SBTS.SUPPLIER AS SUPPLIER_NUM,
  SBTS.SAP_RETURN_INFO AS SAP_RETURN_INFO,
  SBTS.UPDATE_USER AS UPDATE_USER,
  CASE
    WHEN SBTS.COMPANY_CODE IS NULL OR SBTS.COMPANY_CODE = '' THEN -999999
    ELSE NVL(COMP.COMPANY_ID, -999998)
  END AS COMPANY_ID,--公司ID
  CASE
    WHEN SBTS.COMPANY_CODE IS NULL OR SBTS.COMPANY_CODE = '' THEN 'SNULL'
    ELSE NVL(COMP.COMPANY_CODE, 'NOT EXISTS')
  END AS COMPANY_CODE,--公司代码
  SBTS.PURCHASING_ORG AS PURCHASING_ORG,
  SBTS.CREATE_USER AS CREATE_USER,
  SBTS.CREATE_DEPT AS CREATE_DEPT,
  SBTS.STATUS AS STATUS,
  SBTS.TYPE AS TYPE,
  SBTS.PURCHASING_GROUP AS PURCHASING_GROUP,
  SBTS.EDITABLE AS EDITABLE,
  SBTS.TOTAL_AMOUNT AS TOTAL_AMT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_TRANSFER_STORE_1008 SBTS 
  --关联公司主数据表取公司ID、编码
  LEFT JOIN DWIMD.DWI_MD_COMPANY_T COMP ON SBTS.COMPANY_CODE = COMP.COMPANY_CODE
  AND COMP.END_DATE = '4712-12-31'
  AND COMP.DEL_FLAG = 'N'
  AND COMP.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.CODE AS CODE,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.TOTAL_WEIGHT AS TOTAL_WEIGHT,
  DWI.SAP_CODE AS SAP_CODE,
  DWI.SUPPLIER_NUM AS SUPPLIER_NUM,
  DWI.SAP_RETURN_INFO AS SAP_RETURN_INFO,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.COMPANY_ID AS COMPANY_ID,
  DWI.COMPANY_CODE AS COMPANY_CODE,
  DWI.PURCHASING_ORG AS PURCHASING_ORG,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.STATUS AS STATUS,
  DWI.TYPE AS TYPE,
  DWI.PURCHASING_GROUP AS PURCHASING_GROUP,
  DWI.EDITABLE AS EDITABLE,
  DWI.TOTAL_AMT AS TOTAL_AMT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_TRANS_STORE_HEAD_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_TRANSFER_STORE_1008 SBTS ON DWI.ID = SBTS.ID
WHERE
  SBTS.ID IS NULL;
