--SDI_S_USER_1002 到 DWI_SO_USER_OSS_T
--OSS人员表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_USER_OSS_T
SELECT
  SRC.PAR_ROW_ID,
  SRC.ROW_ID,
  SRC.LOGIN,
  SRC.X_WORKCODE,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.X_ADDR_TEXT,
  SRC.CREATED,
  SRC.X_USER_PASSWORD,
  SRC.LAST_LOGIN_TS,
  SRC.LAST_UPD_BY,
  SRC.MODIFICATION_NUM,
  SRC.X_USER_STATUS,
  SRC.DB_LAST_UPD_SRC,
  SRC.CREATED_BY,
  SRC.INTEG_USER_FLG,
  SRC.USER_FLG,
  SRC.CONFLICT_ID,
  SRC.CHALLENGE_ANSWER,
  SRC.TRNG_DL_ADMIN_FLG,
  SRC.CHALLENGE_QUESTION,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_S_USER_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.PAR_ROW_ID,
  DWI.ROW_ID,
  DWI.LOGIN,
  DWI.WORKCODE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.ADDR_TEXT,
  DWI.CREATE_DATE,
  DWI.USER_PASSWORD,
  DWI.LAST_LOGIN_TS,
  DWI.LAST_UPDATE_BY,
  DWI.MODIFY_QTY,
  DWI.USER_STATUS,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.CREATED_BY,
  DWI.INTEG_USER_FLAG,
  DWI.USER_FLAG,
  DWI.CONFLICT_ID,
  DWI.CHALLENGE_ANSWER,
  DWI.TRNG_DL_ADMIN_FLAG,
  DWI.CHALLENGE_QUESTION,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_USER_OSS_T DWI
  LEFT JOIN SDI.SDI_S_USER_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
