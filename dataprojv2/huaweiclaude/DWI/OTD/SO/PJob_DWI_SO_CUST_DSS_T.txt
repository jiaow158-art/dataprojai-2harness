--DSS客户
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_CUST_DSS_T
SELECT
  SBC.ID AS ID,
  SBC.NAME AS NAME,
  SBC.FULL_NAME AS FULL_NAME,
  SBC.TRANSPORT_AREA AS TRANSPORT_AREA,
  SBC.ASCRIPTION AS ASCRIPTION,
  SBC.STATUS AS STATUS,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_CUSTOMER_1008 SBC 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.NAME AS NAME,
  DWI.FULL_NAME AS FULL_NAME,
  DWI.TRANSPORT_AREA AS TRANSPORT_AREA,
  DWI.ASCRIPTION AS ASCRIPTION,
  DWI.STATUS AS STATUS,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_CUST_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_CUSTOMER_1008 SBC ON DWI.ID = SBC.ID
WHERE
  SBC.ID IS NULL;
