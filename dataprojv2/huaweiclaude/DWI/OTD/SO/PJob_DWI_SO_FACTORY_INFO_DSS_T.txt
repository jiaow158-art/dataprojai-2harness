--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--DSS工厂信息
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_FACTORY_INFO_DSS_T
SELECT
  SBFI.ID AS ID,
  SBFI.PRODUCT_ID AS PRODUCT_ID,
  SBFI.EFFECTIVE_START_PERIOD AS EFFECTIVE_START_PERIOD,
  SBFI.WAREHOUSE_CODE AS WAREHOUSE_CODE,
  SBFI.STOCK_LOCATION AS STOCK_LOCATION,
  SBFI.CODE AS CODE,
  SBFI.NAME AS NAME,
  SBFI.MATERIAL_TRANSPORTATION_GROUP AS MATERIAL_TRANS_GROUP,
  SBFI.LOADING_GROUP AS LOADING_GROUP,
  SBFI.UPDATE_TIME AS UPDATE_TIME,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_FACTORY_INFO_1008 SBFI 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.PRODUCT_ID AS PRODUCT_ID,
  DWI.EFFECTIVE_START_PERIOD AS EFFECTIVE_START_PERIOD,
  DWI.WAREHOUSE_CODE AS WAREHOUSE_CODE,
  DWI.STOCK_LOCATION AS STOCK_LOCATION,
  DWI.CODE AS CODE,
  DWI.NAME AS NAME,
  DWI.MATERIAL_TRANS_GROUP AS MATERIAL_TRANS_GROUP,
  DWI.LOADING_GROUP AS LOADING_GROUP,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_FACTORY_INFO_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_FACTORY_INFO_1008 SBFI ON DWI.ID = SBFI.ID
WHERE
  SBFI.ID IS NULL;
