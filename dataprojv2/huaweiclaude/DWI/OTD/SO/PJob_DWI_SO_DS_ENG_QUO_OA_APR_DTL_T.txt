--SDI_FORMTABLE_MAIN_519_DSRC_1004 到 DWI_SO_DS_ENG_QUO_OA_APR_DTL_T
--SBC-DCM-A5-004-工程询价报价流程明细表(直营子公司)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_DS_ENG_QUO_OA_APR_DTL_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.MAINID AS MAIN_ID,
  SRC.DLJMLL AS EP_PRICE_GROSS_RATE,
  SRC.CCBMLL AS REFE_COST_GROSS_RATE,
  SRC.LJSPJG AS APPRV_MATERIAL_PRICE,
  SRC.SQJG AS APPRV_COMPOSITE_PRICE,
  SRC.SQSL AS APPLY_QTY,
  SRC.TPZK AS EP_DISCOUNT,
  SRC.ZHSPJG AS APPROVAL_QUOTATION,
  SRC.BCBMLL AS B_COST_GROSS_RATE,
  SRC.ZHBJ AS APPLICATION_QUOTATION,
  SRC.BDCB AS GUARANTEED_COST,
  SRC.ACBMLL AS BASE_COST_A_RATE,
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  CASE
    WHEN SRC.WLBM IS NULL
    OR SRC.WLBM = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.WLBM IS NULL
    OR SRC.WLBM = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.WLMS AS MATERIAL_DESC,
  SRC.AKCB AS A_COST,
  SRC.ZVRN AS B_COST,
  SRC.CKCB AS REFE_COST,
  SRC.AMBMLL_MX AS AMOEBA_GROSS_RATE,
  SRC.PFS AS SQUARE_QTY,
  SRC.PR00 AS EP_PRICE,
  SRC.YF AS FREIGHT,
  SRC.ZTLSJ AS SHOWROOM_RETAIL_PRICE,
  SRC.FD AS APP_COMPOSITE_SRV_FEE,
  SRC.HYCS AS FREIGHT_CITY,
  SRC.CPCC AS PROD_LEVEL,
  SRC.AMBCB_MX AS AMOEBA_COST_AMT,
  SRC.CWCB AS FINAN_COST,
  SRC.GG AS SPECIFICATIONS,
  SRC.XH AS SEQUENCE,
  SRC.PC_MX AS BATCH,
  SRC.JGF AS PROCESS_COST,
  SRC.KCDD_MX AS INV_LOCATION,
  SRC.SFCZCP AS IS_HISTY_QUOTE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_519_DT1_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.GC = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.WLBM = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.MAIN_ID,
  DWI.EP_PRICE_GROSS_RATE,
  DWI.REFE_COST_GROSS_RATE,
  DWI.APPRV_MATERIAL_PRICE,
  DWI.APPRV_COMPOSITE_PRICE,
  DWI.APPLY_QTY,
  DWI.EP_DISCOUNT,
  DWI.APPROVAL_QUOTATION,
  DWI.B_COST_GROSS_RATE,
  DWI.APPLICATION_QUOTATION,
  DWI.GUARANTEED_COST,
  DWI.BASE_COST_A_RATE,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.MATERIAL_DESC,
  DWI.A_COST,
  DWI.B_COST,
  DWI.REFE_COST,
  DWI.AMOEBA_GROSS_RATE,
  DWI.SQUARE_QTY,
  DWI.EP_PRICE,
  DWI.FREIGHT,
  DWI.SHOWROOM_RETAIL_PRICE,
  DWI.APP_COMPOSITE_SRV_FEE,
  DWI.FREIGHT_CITY,
  DWI.PROD_LEVEL,
  DWI.AMOEBA_COST_AMT,
  DWI.FINAN_COST,
  DWI.SPECIFICATIONS,
  DWI.SEQUENCE,
  DWI.BATCH,
  DWI.PROCESS_COST,
  DWI.INV_LOCATION,
  DWI.IS_HISTY_QUOTE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DS_ENG_QUO_OA_APR_DTL_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_519_DT1_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
