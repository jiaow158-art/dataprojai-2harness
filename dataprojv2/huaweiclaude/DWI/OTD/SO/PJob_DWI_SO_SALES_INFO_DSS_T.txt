set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;

--DSS销售信息
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SALES_INFO_DSS_T
SELECT
  SBSI.ID AS ID,
  SBSI.PRODUCT_ID AS PRODUCT_ID,
  SBSI.UPDATE_TIME AS UPDATE_TIME,
  SBSI.SALES_ORGANIZATION_CODE AS SALES_ORG_CODE,
  SBSI.MATERIAL_GROUP2 AS MATERIAL_GROUP2,
  SBSI.MATERIAL_GROUP1 AS MATERIAL_GROUP1,
  SBSI.MATERIAL_GROUP3 AS MATERIAL_GROUP3,
  SBSI.DISTRIBUTION_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  SBSI.QUARTERLY_POINTS_SCORE AS QUAR_POINTS_SCORE,
  SBSI.ANNUAL_POINTS_SCORE AS ANNUAL_POINTS_SCORE,
  SBSI.MATERIAL_GROUP5 AS MATERIAL_GROUP5,
  SBSI.DECORATION_POINTS_SCORE AS DECORA_POINTS_SCORE,
  SBSI.MONTHLY_POINTS_SCORE AS MONTHLY_POINTS_SCORE,
  SBSI.PRODUCT_ATTRIBUTE10 AS PRODUCT_ATTRIBUTE10,
  SBSI.MATERIAL_GROUP4 AS MATERIAL_GROUP4,
  SBSI.PROMOTIONAL_MATERIAL_POINTS_SCORE AS PROMOL_POINTS_SCORE,
  SBSI.PRODUCT_ATTRIBUTE1 AS PRODUCT_ATTRIBUTE1,
  SBSI.REGIONAL_ADVERTISING_POINTS_SCORE AS REGIONAL_POINTS_SCORE,
  SBSI.BIG_REGIONAL_ADVERTISING_POINTS_SCORE AS BIG_REG_POINTS_SCORE,
  SBSI.DISTRIBUTION_ADVERTISING_POINTS_SCORE AS DISTR_ADV_POINTS_SCORE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_SALES_INFO_1008 SBSI 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.PRODUCT_ID AS PRODUCT_ID,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.SALES_ORG_CODE AS SALES_ORG_CODE,
  DWI.MATERIAL_GROUP2 AS MATERIAL_GROUP2,
  DWI.MATERIAL_GROUP1 AS MATERIAL_GROUP1,
  DWI.MATERIAL_GROUP3 AS MATERIAL_GROUP3,
  DWI.DISTR_CHANNEL_CODE AS DISTR_CHANNEL_CODE,
  DWI.QUAR_POINTS_SCORE AS QUAR_POINTS_SCORE,
  DWI.ANNUAL_POINTS_SCORE AS ANNUAL_POINTS_SCORE,
  DWI.MATERIAL_GROUP5 AS MATERIAL_GROUP5,
  DWI.DECORA_POINTS_SCORE AS DECORA_POINTS_SCORE,
  DWI.MONTHLY_POINTS_SCORE AS MONTHLY_POINTS_SCORE,
  DWI.PRODUCT_ATTRIBUTE10 AS PRODUCT_ATTRIBUTE10,
  DWI.MATERIAL_GROUP4 AS MATERIAL_GROUP4,
  DWI.PROMOL_POINTS_SCORE AS PROMOL_POINTS_SCORE,
  DWI.PRODUCT_ATTRIBUTE1 AS PRODUCT_ATTRIBUTE1,
  DWI.REGIONAL_POINTS_SCORE AS REGIONAL_POINTS_SCORE,
  DWI.BIG_REG_POINTS_SCORE AS BIG_REG_POINTS_SCORE,
  DWI.DISTR_ADV_POINTS_SCORE AS DISTR_ADV_POINTS_SCORE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SALES_INFO_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_SALES_INFO_1008 SBSI ON DWI.ID = SBSI.ID
WHERE
  SBSI.ID IS NULL;
