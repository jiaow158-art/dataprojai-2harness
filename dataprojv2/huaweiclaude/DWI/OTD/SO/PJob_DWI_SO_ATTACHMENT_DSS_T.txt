--DSS附件
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_ATTACHMENT_DSS_T
SELECT
  SBA.ID AS ID,
  SBA.UPDATE_TIME AS UPDATE_TIME,
  SBA.FILENAME AS FILENAME,
  SBA.BLADE_ATTACH_ID AS BLADE_ATTACH_ID,
  SBA.CREATE_TIME AS CREATE_TIME,
  SBA.URL AS URL,
  SBA.SOURCE_ID AS SOURCE_ID,
  SBA.ORIGINAL_NAME AS ORIGINAL_NAME,
  SBA.UPDATE_USER AS UPDATE_USER,
  SBA.CREATE_USER AS CREATE_USER,
  SBA.CREATE_DEPT AS CREATE_DEPT,
  SBA.COMMENTS AS COMMENTS,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_ATTACHMENT_1008 SBA 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.FILENAME AS FILENAME,
  DWI.BLADE_ATTACH_ID AS BLADE_ATTACH_ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.URL AS URL,
  DWI.SOURCE_ID AS SOURCE_ID,
  DWI.ORIGINAL_NAME AS ORIGINAL_NAME,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.COMMENTS AS COMMENTS,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_ATTACHMENT_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_ATTACHMENT_1008 SBA ON DWI.ID = SBA.ID
WHERE
  SBA.ID IS NULL;
