--DSS补货单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_REPAIR_ORDER_ITEM_DSS_T
SELECT
  SBORI.ID AS ID,
  SBORI.ORDER_ITEM_ID AS ORDER_ITEM_ID,
  SBORI.UPDATE_TIME AS UPDATE_TIME,
  SBORI.ORDER_REPLENISH_ID AS ORDER_REPLENISH_ID,
  SBORI.CREATE_TIME AS CREATE_TIME,
  SBORI.ORDER_ID AS ORDER_ID,
  SBORI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  SBORI.BATCH_CODE AS BATCH_CODE,
  SBORI.QUOTATION_ITEM_ID AS QUOTATION_ITEM_ID,
  SBORI.TOTAL_AMOUNT AS TOTAL_AMT,
  SBORI.FINAL_PRICE AS FINAL_PRICE,
  SBORI.PRICE AS PRICE,
  CASE
    WHEN SBORI.FACTORY_CODE IS NULL OR SBORI.FACTORY_CODE = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SBORI.FACTORY_CODE IS NULL OR SBORI.FACTORY_CODE = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  CASE
    WHEN SBORI.PRODUCT_CODE IS NULL OR SBORI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBORI.PRODUCT_CODE IS NULL OR SBORI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBORI.QUANTITY AS QUANTITY,
  SBORI.DISCOUNT_RATE AS DISCOUNT_RATE,
  SBORI.PRODUCT_NAME AS PRODUCT_NAME,
  SBORI.DIRECT_COST AS DIRECT_COST,
  SBORI.DEFAULT_EXPECT_DATE AS DEFAULT_EXPECT_DATE,
  SBORI.EXPECT_DATE AS EXPECT_DATE,
  SBORI.CREATE_USER AS CREATE_USER,
  SBORI.UPDATE_USER AS UPDATE_USER,
  SBORI.WEIGHT AS WEIGHT,
  SBORI.CREATE_DEPT AS CREATE_DEPT,
  SBORI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  SBORI.ITEM_NUMBER AS ITEM_NUMBER,
  SBORI.OPERATE_LOG AS OPERATE_LOG,
  SBORI.REPLENISH_QUANTITY AS REPLENISH_QUANTITY,
  SBORI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  SBORI.INVENTORY_CODE AS INVENTORY_CODE,
  SBORI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  SBORI.PRODUCT_UNIT AS PRODUCT_UNIT,
  SBORI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  SBORI.PROCESSING_UNIT AS PROCESSING_UNIT,
  SBORI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  SBORI.ZXJF_POINT_SCORE AS ZXJF_POINT_SCORE,
  SBORI.NDJF_POINT_SCORE AS NDJF_POINT_SCORE,
  SBORI.YDJF_POINT_SCORE AS YDJF_POINT_SCORE,
  SBORI.GDCP_POINT_SCORE AS GDCP_POINT_SCORE,
  SBORI.JXGG_POINT_SCORE AS JXGG_POINT_SCORE,
  SBORI.QYGG_POINT_SCORE AS QYGG_POINT_SCORE,
  SBORI.JDJF_POINT_SCORE AS JDJF_POINT_SCORE,
  SBORI.DQGG_POINT_FLAG AS DQGG_POINT_FLAG,
  SBORI.QYGG_POINT_FLAG AS QYGG_POINT_FLAG,
  SBORI.XCZL_POINT_FLAG AS XCZL_POINT_FLAG,
  SBORI.JXGGJF_POINT_FLAG AS JXGGJF_POINT_FLAG,
  SBORI.DQGG_POINT_SCORE AS DQGG_POINT_SCORE,
  SBORI.ZXJF_POINT_FLAG AS ZXJF_POINT_FLAG,
  SBORI.YDJF_POINT_FLAG AS YDJF_POINT_FLAG,
  SBORI.XCZL_POINT_SCORE AS XCZL_POINT_SCORE,
  SBORI.GDCP_POINT_FLAG AS GDCP_POINT_FLAG,
  SBORI.STATUS AS STATUS,
  SBORI.JDJF_POINT_FLAG AS JDJF_POINT_FLAG,
  SBORI.NDJF_POINT_FLAG AS NDJF_POINT_FLAG,
  SBORI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  SBORI.COMMISSION_POINTS AS COMMISSION_POINTS,
  SBORI.POINT_DISCOUNT AS POINT_DISCOUNT,
  SBORI.PROJECT_TYPE AS PROJECT_TYPE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_ORDER_REPLENISH_ITEM_1008 SBORI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBORI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800' 
  --关联工厂主数据表DWI_MD_FACTORY_T取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SBORI.FACTORY_CODE = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.ORDER_ITEM_ID AS ORDER_ITEM_ID,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.ORDER_REPLENISH_ID AS ORDER_REPLENISH_ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.ORDER_ID AS ORDER_ID,
  DWI.BATCH_AVAILABLE_QTY AS BATCH_AVAILABLE_QTY,
  DWI.BATCH_CODE AS BATCH_CODE,
  DWI.QUOTATION_ITEM_ID AS QUOTATION_ITEM_ID,
  DWI.TOTAL_AMT AS TOTAL_AMT,
  DWI.FINAL_PRICE AS FINAL_PRICE,
  DWI.PRICE AS PRICE,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.QUANTITY AS QUANTITY,
  DWI.DISCOUNT_RATE AS DISCOUNT_RATE,
  DWI.PRODUCT_NAME AS PRODUCT_NAME,
  DWI.DIRECT_COST AS DIRECT_COST,
  DWI.DEFAULT_EXPECT_DATE AS DEFAULT_EXPECT_DATE,
  DWI.EXPECT_DATE AS EXPECT_DATE,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.WEIGHT AS WEIGHT,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.OPERATE_LOG AS OPERATE_LOG,
  DWI.REPLENISH_QUANTITY AS REPLENISH_QUANTITY,
  DWI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,
  DWI.PROCESSING_UNIT_PRICE AS PROCESSING_UNIT_PRICE,
  DWI.PRODUCT_UNIT AS PRODUCT_UNIT,
  DWI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  DWI.PROCESSING_UNIT AS PROCESSING_UNIT,
  DWI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  DWI.ZXJF_POINT_SCORE AS ZXJF_POINT_SCORE,
  DWI.NDJF_POINT_SCORE AS NDJF_POINT_SCORE,
  DWI.YDJF_POINT_SCORE AS YDJF_POINT_SCORE,
  DWI.GDCP_POINT_SCORE AS GDCP_POINT_SCORE,
  DWI.JXGG_POINT_SCORE AS JXGG_POINT_SCORE,
  DWI.QYGG_POINT_SCORE AS QYGG_POINT_SCORE,
  DWI.JDJF_POINT_SCORE AS JDJF_POINT_SCORE,
  DWI.DQGG_POINT_FLAG AS DQGG_POINT_FLAG,
  DWI.QYGG_POINT_FLAG AS QYGG_POINT_FLAG,
  DWI.XCZL_POINT_FLAG AS XCZL_POINT_FLAG,
  DWI.JXGGJF_POINT_FLAG AS JXGGJF_POINT_FLAG,
  DWI.DQGG_POINT_SCORE AS DQGG_POINT_SCORE,
  DWI.ZXJF_POINT_FLAG AS ZXJF_POINT_FLAG,
  DWI.YDJF_POINT_FLAG AS YDJF_POINT_FLAG,
  DWI.XCZL_POINT_SCORE AS XCZL_POINT_SCORE,
  DWI.GDCP_POINT_FLAG AS GDCP_POINT_FLAG,
  DWI.STATUS AS STATUS,
  DWI.JDJF_POINT_FLAG AS JDJF_POINT_FLAG,
  DWI.NDJF_POINT_FLAG AS NDJF_POINT_FLAG,
  DWI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  DWI.COMMISSION_POINTS AS COMMISSION_POINTS,
  DWI.POINT_DISCOUNT AS POINT_DISCOUNT,
  DWI.PROJECT_TYPE AS PROJECT_TYPE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_REPAIR_ORDER_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_ORDER_REPLENISH_ITEM_1008 SBORI ON DWI.ID = SBORI.ID
WHERE
  SBORI.ID IS NULL;
