  END AS CUST_ID,

    WHEN SRC.KHBM IS NULL

  SRC.JSTK AS SETTLEMENT_CLAUSE,
  SRC.PDTJ AS JUDGE_CONDITIONS,
  SRC.QYLX AS ENTERPRISE_TYPE,
  SRC.ZCZBGM AS REGISTERED_CAPITAL,
  SRC.CGXS AS PURCHASING_PATTERN,
  SRC.QD AS CHANNEL,
  SRC.XSBM AS SALES_DEPT,
  SRC.XSZZ AS SALE_ORG_CODE,
  SRC.AMBZDMLL AS AMOEBA_ABUNDANT_RATE,
  SRC.AMBZDYK AS WHOLE_GAIN_LOSS_AMT,
  SRC.FWPJ AS SERVICE_EVALUATION,
  SRC.SFBQKH AS IF_STRATEGY_CUST_FLAG,
  SRC.SFWBQ AS WHETHER_TOP_500,
  SRC.SX AS ATTRIBUTE,
  SRC.SXE AS CREDIT_AMOUNT,
  SRC.ZBJ AS WARRANTY_AMT,
  SRC.BLHKJL AS POOR_COLLECTION,
  SRC.BQKH AS STRATEGY_CUST_NAME,
  SRC.FKFS AS PAYMENT_METHOD,
  SRC.SFQC AS CLEARANCE_FLAG,
  SRC.SFQDZLXY AS SIGN_AGREEMENT_FLAG,
  SRC.XMWHR1 AS PROJ_MAINTAINER1,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_908_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.KHBM = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.REQUEST_ID,
  DWI.OA_CODE,
  DWI.WHOLE_GROSS_PROFIT_AMT,
  DWI.ACCUMULATED_PROFIT_AMT,
  DWI.WHOLE_GROSS_PROFIT_RATE,
  DWI.ANNEX_ID,
  DWI.SAP_QUOTE_NUM,
  DWI.ASSOCIATED_ATTACHMENT,
  DWI.PROJECT_NAME,
  DWI.APPLICATION_DATE,
  DWI.SUM_GROSS_PROFIT_AMT,
  DWI.CONTRACT_SIGNING_DATE,
  DWI.WBS_ELEMENT,
  DWI.PAYER_NUM,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.PROJ_NAME,
  DWI.PROJ_REP_ID,
  DWI.REPORT_EFFECTIVE_DATE,
  DWI.PROJECT_ADDRESS,
  DWI.PURCHASER_NAME,
  DWI.BUILDER_NAME,
  DWI.PROJECT_OVERVIEW,
  DWI.VALIDITY_CONTRACT,
  DWI.ESTIMATED_AMOUNT,
  DWI.SETTLEMENT_TERMS,
  DWI.VALIDITY_OFFER,
  DWI.SCORE,
  DWI.APPLICANT_NAME,
  DWI.APPLICANT_ID,
  DWI.FILING_CLERK_ID,
  DWI.PROJECT_DEVELOPER,
  DWI.PHONE_NUMBER,
  DWI.PROJ_TYPE,
  DWI.QUALIFICATION_LEVEL,
  DWI.PROCESS_NODE,
  DWI.WHETHER_SIGN_CONTRACT,
  DWI.CREDIT_RATING,
  DWI.CREDIT_EVALUATION,
  DWI.RETURN_INFO,

  DWI.PURCHASING_PATTERN,

  DWIOTD.DWI_SO_DC_ENG_QUO_OA_APR_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_908_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
