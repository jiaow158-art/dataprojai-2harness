--SDI_FORMTABLE_MAIN_829_DSRC_1004 到 DWI_SO_SANIT_QUO_OA_APR_DTL_T
--DEA-A1-005-洁具询价报价OA审批流程明细表(工程及家装)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SANIT_QUO_OA_APR_DTL_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.MAINID AS MAIN_ID,
  SRC.MX1_AMLE AS A_GROSS_PROFIT_AMT,
  SRC.MX1_XJJE AS SUBTOTAL_AMT,
  SRC.MX1_AMLL AS A_GROSS_PROFIT_RATE,
  SRC.MX1_BMLE AS B_GROSS_PROFIT_AMT,
  SRC.MX1_BMLL AS B_GROSS_PROFIT_RATE,
  SRC.MX1_SQZK AS APPLY_DISCOUNT,
  SRC.MX1_SPZK AS APPROVE_DISCOUNT,
  SRC.MX1_SPLJ AS APPROVE_BARE_PRICE,
  SRC.MX1_SPJ AS APPROVE_PRICE,
  SRC.MX1_SQJ AS APPLY_PRICE,
  SRC.MX1_WLJC AS ITEM_ABBREVIATION,
  SRC.MX1_ACB AS A_COST,
  CASE
    WHEN SRC.MX1_GC IS NULL
    OR SRC.MX1_GC = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.MX1_GC IS NULL
    OR SRC.MX1_GC = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  SRC.MX1_BCB AS B_COST,
  --B成本
  CASE
    WHEN SRC.MX1_WLBM IS NULL
    OR SRC.MX1_WLBM = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.MX1_WLBM IS NULL
    OR SRC.MX1_WLBM = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.MX1_SAPBZJSJ AS SAP_SETTLEMENT_PRICE,
  SRC.MX1_XSSL AS SALE_QTY,
  SRC.MX1_FWF AS COMPR_SERVICE_RATE,
  SRC.MX1_FWS AS SERVICE_EVALUATE,
  SRC.MX1_YF AS FREIGHT,
  SRC.MX1_ZJS AS MIDDLEMEN,
  SRC.MX1_WLZ AS MATERIAL_GROUP,
  SRC.MX1_JLDW AS PLAN_GROUP,
  SRC.MX1_DLS AS AGEN,
  SRC.MX1_SFLS AS PROD_CHANNELG101_RETAIL,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_829_DT1_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.MX1_GC = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.MX1_WLBM = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.MAIN_ID,
  DWI.A_GROSS_PROFIT_AMT,
  DWI.SUBTOTAL_AMT,
  DWI.A_GROSS_PROFIT_RATE,
  DWI.B_GROSS_PROFIT_AMT,
  DWI.B_GROSS_PROFIT_RATE,
  DWI.APPLY_DISCOUNT,
  DWI.APPROVE_DISCOUNT,
  DWI.APPROVE_BARE_PRICE,
  DWI.APPROVE_PRICE,
  DWI.APPLY_PRICE,
  DWI.ITEM_ABBREVIATION,
  DWI.A_COST,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.B_COST,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.SAP_SETTLEMENT_PRICE,
  DWI.SALE_QTY,
  DWI.COMPR_SERVICE_RATE,
  DWI.SERVICE_EVALUATE,
  DWI.FREIGHT,
  DWI.MIDDLEMEN,
  DWI.MATERIAL_GROUP,
  DWI.MEASUREMENT_UNIT,
  DWI.AGEN,
  DWI.PROD_CHANNELG101_RETAIL,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SANIT_QUO_OA_APR_DTL_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_829_DT1_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
