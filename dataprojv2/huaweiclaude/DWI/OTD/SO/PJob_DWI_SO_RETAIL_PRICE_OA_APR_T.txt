--SDI_FORMTABLE_MAIN_72_1004 到 DWI_SO_RETAIL_PRICE_OA_APR_T
--SBC-DCM-A5-001-直营价格审批申请流程主表(零售)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RETAIL_PRICE_OA_APR_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.REQUESTID AS REQUEST_ID,
  SRC.LSH AS OA_CODE,
  SRC.BJDH AS SAP_QUOTE_NUM,
  SRC.ZJJE AS TOTAL_AMT,
  SRC.KHSHDZ AS CUST_DELIVERY_ADDRESS,
  SRC.HTQDDW AS CONTACT_SIGN_UNIT,
  SRC.YKZJE AS TOTAL_GAIN_LOSS_AMT,
  SRC.SPQX AS APPRV_CERAMIC_TILE_AUTH,
  SRC.FHXX AS RETURN_INFO,
  SRC.XSZMJ AS TOTAL_SALE_AREA,
  SRC.SJS AS DESIGNER,
  SRC.KHZ AS CUST_GROUP,
  SRC.JZRQ AS QUOTE_DUE_DATE,
  SRC.AMBZDYK AS WHOLE_GAIN_LOSS_AMT,
  SRC.SQRQ AS APPLICATION_DATE,
  SRC.SJR2 AS PROCESS_RECIPIEN,
  SRC.AMBZDMLL AS AMOEBA_WHOLE_FENG_RATE,
  SRC.BBH AS REPORT_NO,
  SRC.KHYJSYSJ AS CUS_ESTIMATED_USAGE_TIME,
  SRC.FKF AS PAYER,
  SRC.KHJC AS CUST_REFERRED,
  SRC.SPQX0 AS APP_WARE_PERMISSION,
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN -999999
    ELSE NVL(CUST.CUST_ID, -999998)
  END AS CUST_ID,
  --客户ID
  CASE
    WHEN SRC.KHBM IS NULL
    OR SRC.KHBM = '' THEN 'SNULL'
    ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')
  END AS CUST_NUM,
  --客户编码
  SRC.BBYXRQ AS INSURANCE_DATE,
  SRC.GCMC AS PROJ_NAME,
  SRC.SJDW AS DESIGN_UNIT,
  SRC.GCDZ AS PROJ_ADDRESS,
  SRC.SQR AS APPLY_USER,
  SRC.ZZSPR AS FINAL_APPROVER,
  SRC.ZJDW AS MEDIATION_UNIT,
  SRC.ZJR AS INTERMEDIARIE,
  SRC.SHR AS REVIEWER,
  SRC.SQRSSBM AS APPLICANT_DEPT,
  SRC.FKFXSBMBM AS CODE_PAYER_SALE_DEPT,
  SRC.XSZZ AS SALE_ORGANIZATION,
  SRC.ACT AS PROCESS_NODE,
  SRC.QD AS CHANNEL,
  SRC.XFQD AS SEGMENTED_CHANNEL,
  SRC.SFXR AS WHETHER_WRITE_FLAG,
  SRC.JFCGF AS PARTY_BUYER,
  SRC.JGGY AS PROCESS_TECHNIC_FLAG,
  SRC.SFHFWF AS IF_SERVICE_CHARGE,
  SRC.XSQY AS SALES_AREA,
  SRC.BT AS TITLE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_72_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST --客户主数据表
  ON SRC.KHBM = CUST.CUST_NUM
  AND CUST.END_DATE = '4712-12-31'
  AND CUST.DEL_FLAG = 'N'
  AND CUST.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.REQUEST_ID,
  DWI.OA_CODE,
  DWI.SAP_QUOTE_NUM,
  DWI.TOTAL_AMT,
  DWI.CUST_DELIVERY_ADDRESS,
  DWI.CONTACT_SIGN_UNIT,
  DWI.TOTAL_GAIN_LOSS_AMT,
  DWI.APPRV_CERAMIC_TILE_AUTH,
  DWI.RETURN_INFO,
  DWI.TOTAL_SALE_AREA,
  DWI.DESIGNER,
  DWI.CUST_GROUP,
  DWI.QUOTE_DUE_DATE,
  DWI.WHOLE_GAIN_LOSS_AMT,
  DWI.APPLICATION_DATE,
  DWI.PROCESS_RECIPIEN,
  DWI.AMOEBA_WHOLE_FENG_RATE,
  DWI.REPORT_NO,
  DWI.CUS_ESTIMATED_USAGE_TIME,
  DWI.PAYER,
  DWI.CUST_REFERRED,
  DWI.APP_WARE_PERMISSION,
  DWI.CUST_ID,
  DWI.CUST_NUM,
  DWI.INSURANCE_DATE,
  DWI.PROJ_NAME,
  DWI.DESIGN_UNIT,
  DWI.PROJ_ADDRESS,
  DWI.APPLY_USER,
  DWI.FINAL_APPROVER,
  DWI.MEDIATION_UNIT,
  DWI.INTERMEDIARIE,
  DWI.REVIEWER,
  DWI.APPLICANT_DEPT,
  DWI.CODE_PAYER_SALE_DEPT,
  DWI.SALE_ORGANIZATION,
  DWI.PROCESS_NODE,
  DWI.CHANNEL,
  DWI.SEGMENTED_CHANNEL,
  DWI.WHETHER_WRITE_FLAG,
  DWI.PARTY_BUYER,
  DWI.PROCESS_TECHNIC_FLAG,
  DWI.IF_SERVICE_CHARGE,
  DWI.SALES_AREA,
  DWI.TITLE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RETAIL_PRICE_OA_APR_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_72_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
