--SDI_CX_SALES_ORG_1002 到 DWI_SO_SALES_ORG_OSS_T
--OSS销售组织
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SALES_ORG_OSS_T
SELECT
  SRC.ROW_ID,
  SRC.SALES_ORG,
  SRC.SALES_ORG_CD,
  SRC.CREATED,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.SALES_ORG_TYPE,
  SRC.MODIFICATION_NUM,
  SRC.SUB_FLAG,
  SRC.LAST_UPD_BY,
  SRC.CONFLICT_ID,
  SRC.DB_LAST_UPD_SRC,
  SRC.CREATED_BY,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_SALES_ORG_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.SALES_ORG,
  DWI.SALES_ORG_CD,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.SALES_ORG_TYPE,
  DWI.MODIFY_QTY,
  DWI.SUB_FLAG,
  DWI.LAST_UPDATE_BY,
  DWI.CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.CREATED_BY,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SALES_ORG_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_SALES_ORG_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
