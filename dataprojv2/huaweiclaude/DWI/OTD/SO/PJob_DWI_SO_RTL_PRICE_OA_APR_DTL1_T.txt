--SDI_FORMTABLE_MAIN_72_DSRC_1004 到 DWI_SO_RTL_PRICE_OA_APR_DTL1_T
--SBC-DCM-A5-001-直营价格审批申请流程明细表1(零售)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL1_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.MAINID AS MAIN_ID,
  SRC.XJJE2 AS SUBTOTAL_AMT,
  SRC.YKJE AS GAIN_LOSS_AMT,
  SRC.SPZK2 AS APP_EXAMINE_DISCOUNT,
  SRC.SQZK2 AS APP_EXAMINE_DISCOUNT1,
  SRC.XJJE3 AS LITTLE_AMT,
  SRC.SPJ2 AS APP_PRICE,
  SRC.SQJ2 AS APPLY_PRICE,
  SRC.SAPJG AS SAP_SETTLEMENT_PRICE,
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  CASE
    WHEN SRC.WLBM_1 IS NULL
    OR SRC.WLBM_1 = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.WLBM_1 IS NULL
    OR SRC.WLBM_1 = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.CKCBMLL AS REFER_COST_GROSS_RATE,
  SRC.WLJC_1 AS MATERIAL_ABBREVIATION,
  SRC.BCBMLL AS B_GROSS_MARGIN_RATE,
  SRC.CKCB2 AS REFER_COST,
  SRC.XSSL AS SALE_QTY,
  SRC.BCB AS B_COST,
  SRC.XSMJ AS SALE_AREA,
  SRC.AMBMLL_MX AS GROSS_MARGIN_RATE,
  SRC.AKCB AS A_COST,
  SRC.CPCC AS PROD_LEVEL,
  SRC.AMBCB_MX AS AMOEBA_COST,
  SRC.PC_MX AS BATCH_NUM,
  SRC.FWF AS SERVICE_FEE,
  SRC.HXM AS ITEM_ID,
  SRC.DWMJ AS UNIT_AREA,
  SRC.JLDW AS MEASURING_UNIT,
  SRC.WLZ2 AS MATERIAL_GROUP1,
  SRC.GGBT AS ADVERT_POINTS_FLAG,
  SRC.NDFL AS YEARS_POINTS_FLAG,
  SRC.YDFL AS MONTHLY_INTEGRAL,
  SRC.ZXBT AS RENOVATION_POINTS_FLAG,
  SRC.SFCZCP AS CERAMIC_TILE_DIVISION_FLAG,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_72_DT1_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.GC = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.WLBM_1 = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.MAIN_ID,
  DWI.SUBTOTAL_AMT,
  DWI.GAIN_LOSS_AMT,
  DWI.APP_EXAMINE_DISCOUNT,
  DWI.APP_EXAMINE_DISCOUNT1,
  DWI.SUBTOTAL_APP_AMT,
  DWI.APP_PRICE,
  DWI.APPLY_PRICE,
  DWI.SAP_SETTLEMENT_PRICE,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.REFER_COST_GROSS_RATE,
  DWI.MATERIAL_ABBREVIATION,
  DWI.B_GROSS_MARGIN_RATE,
  DWI.REFER_COST,
  DWI.SALE_QTY,
  DWI.B_COST,
  DWI.SALE_AREA,
  DWI.GROSS_MARGIN_RATE,
  DWI.A_COST,
  DWI.PROD_LEVEL,
  DWI.AMOEBA_COST,
  DWI.BATCH_NUM,
  DWI.SERVICE_FEE,
  DWI.ITEM_ID,
  DWI.UNIT_AREA,
  DWI.MEASURING_UNIT,
  DWI.MATERIAL_GROUP1,
  DWI.ADVERT_POINTS_FLAG,
  DWI.YEARS_POINTS_FLAG,
  DWI.MONTHLY_INTEGRAL,
  DWI.RENOVATION_POINTS_FLAG,
  DWI.CERAMIC_TILE_DIVISION_FLAG,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL1_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_72_DT1_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
