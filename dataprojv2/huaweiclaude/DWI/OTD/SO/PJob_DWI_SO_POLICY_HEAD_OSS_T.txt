INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_POLICY_HEAD_OSS_T
SELECT
  SRC.ROW_ID,
  SRC.CREATED,
  SRC.LAST_UPD,
  SRC.DB_LAST_UPD,
  SRC.POLICY_NUM,
  SRC.NAME,
  SRC.START_DATE,
  SRC.QUOTA_END_DATE,
  SRC.END_DATE,
  SRC.MODIFICATION_NUM,
  SRC.INTG_ID,
  SRC.SALES_ORG_ID,
  SRC.ACCNT_GROUP3,
  SRC.LAST_UPD_BY,
  SRC.CREATED_BY,
  SRC.PRI_ORG_ID,
  SRC.CHANNEL_ID,
  SRC.ACCNT_GROUP6,
  SRC.STATUS,
  SRC.APPLY_TYPE,
  SRC.CON_GROUP4,
  SRC.DB_LAST_UPD_SRC,
  SRC.GEN_SAP_SUB_TYPE,
  SRC.APPLY_ALL_SALES_ORG,
  SRC.INTER_RATIO,
  SRC.DECO_ENTER_ID,
  SRC.TYPE,
  SRC.CROSS_RATIO,
  SRC.CONFLICT_ID,
  SRC.POT_FLAG,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_PRI_POY_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.POLICY_NUM,
  DWI.NAME,
  DWI.START_DATE,
  DWI.QUOTA_END_DATE,
  DWI.END_DATE,
  DWI.MODIFY_QTY,
  DWI.INTG_ID,
  DWI.SALES_ORG_ID,
  DWI.ACCNT_GROUP3,
  DWI.LAST_UPDATE_BY,
  DWI.CREATED_BY,
  DWI.PRI_ORG_ID,
  DWI.CHANNEL_ID,
  DWI.ACCNT_GROUP6,
  DWI.STATUS,
  DWI.APPLY_TYPE,
  DWI.CON_GROUP4,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.GEN_SAP_SUB_TYPE,
  DWI.APPLY_ALL_SALES_ORG,
  DWI.INTER_RATIO,
  DWI.DECO_ENTER_ID,
  DWI.TYPE,
  DWI.CROSS_RATIO,
  DWI.CONFLICT_ID,
  DWI.POT_FLAG,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_POLICY_HEAD_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_PRI_POY_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
