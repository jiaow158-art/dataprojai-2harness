--SDI_S_POSTN_1002 到 DWI_SO_POSTN_OSS_T
--OSS组织表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_POSTN_OSS_T
SELECT
  SRC.PAR_ROW_ID,
  SRC.ROW_ID,
  SRC.NAME,
  SRC.PR_EMP_ID,
  SRC.DB_LAST_UPD,
  SRC.OU_ID,
  SRC.BU_ID,
  SRC.CREATED,
  SRC.LAST_UPD,
  SRC.X_SALES_ORG_ID,
  SRC.POSTN_TYPE_CD,
  SRC.MODIFICATION_NUM,
  SRC.DESC_TEXT,
  SRC.X_POSTN_TYPE_CD_OSS,
  SRC.DB_LAST_UPD_SRC,
  SRC.LAST_UPD_BY,
  SRC.CREATED_BY,
  SRC.PAR_POSTN_ID,
  SRC.CMPNS_CURCY_CD,
  SRC.COMPENSATABLE_FLG,
  SRC.PR_POSTN_ADDR_ID,
  SRC.PR_TERR_ID,
  SRC.IN_IC_RETRORUN_FLG,
  SRC.CONFLICT_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_S_POSTN_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.PAR_ROW_ID,
  DWI.ROW_ID,
  DWI.NAME,
  DWI.PR_EMP_ID,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.OU_ID,
  DWI.BU_ID,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.SALES_ORG_ID,
  DWI.POSTN_TYPE_CD,
  DWI.MODIFY_QTY,
  DWI.DESC_TEXT,
  DWI.POSTN_TYPE_CD_OSS,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.LAST_UPDATE_BY,
  DWI.CREATED_BY,
  DWI.PAR_POSTN_ID,
  DWI.CMPNS_CURCY_CD,
  DWI.COMPENSATABLE_FLAG,
  DWI.PR_POSTN_ADDR_ID,
  DWI.PR_TERR_ID,
  DWI.IN_IC_RETRORUN_FLAG,
  DWI.CONFLICT_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_POSTN_OSS_T DWI
  LEFT JOIN SDI.SDI_S_POSTN_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
