--SDI_CX_PURCH_ITM_1002 到 DWI_SO_PURCH_ITEM_OSS_T
--要货单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PURCH_ITEM_OSS_T
SELECT
  SRC.ROW_ID AS ROW_ID,
  SRC.CREATED AS CREATE_DATE,
  SRC.LAST_UPD AS LAST_UPDATE_DATE,
  SRC.DB_LAST_UPD AS DB_LAST_UPDATE_DATE,
  SRC.HEAD_ID AS HEAD_ID,
  SRC.DP_REQUITE_ORDE_ITEMR_ID AS DP_REQUITE_ORDE_ITEMR_ID,
  SRC.PLAN_NUM AS PLAN_NUM,
  SRC.DP_ORDER_ITEM_ID AS DP_ORDER_ITEM_ID,
  SRC.YI_PICK_AMT AS YI_PICK_AMT,
  SRC.PROD_NUM AS PROD_NUM,
  SRC.TRAN_NUM AS TRAN_NUM,
  SRC.DP_ORDER_ID AS DP_ORDER_ID,
  SRC.DP_ACTUAL_QUANTITY AS DP_ACTUAL_QUANTITY,
  SRC.YI_PICK AS YI_PICK,
  SRC.DP_PLAN_FACTORY AS SRC_FACTORY_ID,
  --源系统工厂ID
  CASE
    WHEN SRC.DP_PLAN_FACTORY IS NULL
    OR SRC.DP_PLAN_FACTORY = '' THEN -999999
    ELSE NVL(tmp.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.DP_PLAN_FACTORY IS NULL
    OR SRC.DP_PLAN_FACTORY = '' THEN 'SNULL'
    ELSE NVL(TMP.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  SRC.PROD_ID AS PROD_ID,
  SRC.DP_REMARK AS DP_REMARK,
  SRC.CANCEL_NUM AS CANCEL_NUM,
  SRC.ARIVAL_TIME AS ARIVAL_TIME,
  SRC.UPDATE_ID AS UPDATE_ID,
  SRC.CREAT_ID AS CREAT_ID,
  SRC.YI_RETURN_PICK AS YI_RETURN_PICK,
  SRC.YI_CHECK AS YI_CHECK,
  SRC.MODIFICATION_NUM AS MODIFY_QTY,
  SRC.DP_PLAN_WAREHOUSE AS DP_PLAN_WAREHOUSE,
  SRC.UNIT_PRI AS UNIT_PRI,
  SRC.YI_RETURN_CHECK AS YI_RETURN_CHECK,
  SRC.LAST_UPD_BY AS LAST_UPDATE_BY,
  SRC.CREATED_BY AS CREATED_BY,
  SRC.CONFLICT_ID AS CONFLICT_ID,
  SRC.DB_LAST_UPD_SRC AS DB_LAST_UPDATE_DATE_SRC,
  SRC.STLE_STATUS AS STLE_STATUS,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_PURCH_ITM_1002 SRC
  LEFT JOIN (
    SELECT
      M1.ROW_ID,
      FAC.FACTORY_ID,
      --工厂ID
      FAC.FACTORY_CODE --工厂编码
    FROM
      SDI.SDI_CX_FACTORY_1002 M1 --工厂编码表
      INNER JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
      ON M1.FACTORY_CD = FAC.FACTORY_CODE
      AND FAC.END_DATE = '4712-12-31'
      AND FAC.DEL_FLAG = 'N'
      AND FAC.GROUP_NUM = '800'
  ) TMP ON SRC.DP_PLAN_FACTORY = TMP.ROW_ID
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.CREATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.HEAD_ID,
  DWI.DP_REQUITE_ORDE_ITEMR_ID,
  DWI.PLAN_NUM,
  DWI.DP_ORDER_ITEM_ID,
  DWI.YI_PICK_AMT,
  DWI.PROD_NUM,
  DWI.TRAN_NUM,
  DWI.DP_ORDER_ID,
  DWI.DP_ACTUAL_QUANTITY,
  DWI.YI_PICK,
  DWI.SRC_FACTORY_ID,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.PROD_ID,
  DWI.DP_REMARK,
  DWI.CANCEL_NUM,
  DWI.ARIVAL_TIME,
  DWI.UPDATE_ID,
  DWI.CREAT_ID,
  DWI.YI_RETURN_PICK,
  DWI.YI_CHECK,
  DWI.MODIFY_QTY,
  DWI.DP_PLAN_WAREHOUSE,
  DWI.UNIT_PRI,
  DWI.YI_RETURN_CHECK,
  DWI.LAST_UPDATE_BY,
  DWI.CREATED_BY,
  DWI.CONFLICT_ID,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.STLE_STATUS,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PURCH_ITEM_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_PURCH_ITM_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
