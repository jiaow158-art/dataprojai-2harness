--DSS折扣政策行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_POLICY_ITEM_DSS_T
SELECT
  SBBI.ID AS ID,
  SBBI.DISCOUNT_PERCENT AS DISCOUNT_PERCENT,
  CASE
    WHEN SBBI.PRODUCT_CODE IS NULL OR SBBI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBBI.PRODUCT_CODE IS NULL OR SBBI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBBI.PRODUCT_NAME AS PRODUCT_NAME,
  SBBI.UPDATE_TIME AS UPDATE_TIME,
  SBBI.CREATE_TIME AS CREATE_TIME,
  SBBI.START_TIME AS START_TIME,
  SBBI.POLICY_ID AS POLICY_ID,
  SBBI.END_TIME AS END_TIME,
  SBBI.STATUS AS STATUS,
  SBBI.CREATE_DEPT AS CREATE_DEPT,
  SBBI.CREATE_USER AS CREATE_USER,
  SBBI.UPDATE_USER AS UPDATE_USER,
  SBBI.MAX_QUANTITY AS MAX_QUANTITY,
  SBBI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  SBBI.MIN_QUANTITY AS MIN_QUANTITY,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_POLICY_ITEM_1008 SBBI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBBI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,
  DWI.DISCOUNT_PERCENT AS DISCOUNT_PERCENT,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.PRODUCT_NAME AS PRODUCT_NAME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.START_TIME AS START_TIME,
  DWI.POLICY_ID AS POLICY_ID,
  DWI.END_TIME AS END_TIME,
  DWI.STATUS AS STATUS,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.MAX_QUANTITY AS MAX_QUANTITY,
  DWI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  DWI.MIN_QUANTITY AS MIN_QUANTITY,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_POLICY_ITEM_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_POLICY_ITEM_1008 SBBI ON DWI.ID = SBBI.ID
WHERE
  SBBI.ID IS NULL;
