--DSS零售产品
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RETAIL_PROD_DSS_T
SELECT
  SRP.ID AS ID,
  SRP.PRODUCT_MODEL AS PRODUCT_MODEL,
  SRP.NAME AS NAME,
  SRP.PRICE AS PRICE,
  SRP.DISCOUNTED_PRICES AS DISCOUNTED_PRICES,
  SRP.UPDATE_TIME AS UPDATE_TIME,
  SRP.SPECIFICATION AS SPECIFICATION,
  SRP.CREATE_TIME AS CREATE_TIME,
  SRP.REMARK AS REMARK,
  SRP.END_DATE AS END_DATE,
  SRP.START_DATE AS START_DATE,
  SRP.SERIES AS SERIES,
  SRP.UPDATE_USER AS UPDATE_USER,
  SRP.CREATE_DEPT AS CREATE_DEPT,
  SRP.CREATE_USER AS CREATE_USER,
  SRP.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_RETAIL_PRODUCT_1008 SRP 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.PRODUCT_MODEL AS PRODUCT_MODEL,
  DWI.NAME AS NAME,
  DWI.PRICE AS PRICE,
  DWI.DISCOUNTED_PRICES AS DISCOUNTED_PRICES,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.SPECIFICATION AS SPECIFICATION,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.REMARK AS REMARK,
  DWI.END_DATE AS END_DATE,
  DWI.START_DATE AS START_DATE,
  DWI.SERIES AS SERIES,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RETAIL_PROD_DSS_T DWI
  LEFT JOIN SDI.SDI_RETAIL_PRODUCT_1008 SRP ON DWI.ID = SRP.ID
WHERE
  SRP.ID IS NULL;
