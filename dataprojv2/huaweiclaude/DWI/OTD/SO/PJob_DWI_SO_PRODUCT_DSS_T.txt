--DSS产品
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PRODUCT_DSS_T
SELECT
  SBP.ID AS ID,
  SBP.MATERIAL_DESCRIPTION AS MATERIAL_DESC,
  SBP.QR_CODE_URL AS QR_CODE_URL,
  SBP.GROSS_WEIGHT AS GROSS_WEIGHT,
  SBP.NET_WEIGHT AS NET_WEIGHT,
  SBP.LEVEL AS LEVEL,
  SBP.BUSINESS_VOLUME AS BUSINESS_VOLUME,
  SBP.MAGNITUDE AS MAGNITUDE,
  SBP.LENGTH AS LENGTH,
  SBP.WIDTH AS WIDTH,
  SBP.ACREAGE AS ACREAGE,
  SBP.HEIGHT AS HEIGHT,
  SBP.BASIC_UNIT AS BASIC_UNIT,
  SBP.UPDATE_TIME AS UPDATE_TIME,
  SBP.CHANNEL AS CHANNEL,
  SBP.MATERIAL_GROUP AS MATERIAL_GROUP,
  SBP.EXTERNAL_MATERIAL_GROUP AS EXTER_MATERIAL_GROUP,
  SBP.GENERAL_PROJECT_CATEGORY_GROUP AS GEN_PROJ_TYPE_GROUP,
  SBP.RESERVED_FIELD5 AS RESERVED_FIELD5,
  SBP.VOLUME_UNIT AS VOLUME_UNIT,
  SBP.CLASSIFICATION AS CLASSIFICATION,
  SBP.UNIT_OF_WEIGHT AS UNIT_OF_WEIGHT,
  SBP.MEASUREMENT_UNIT AS MEASUREMENT_UNIT,
  SBP.GRADE AS GRADE,
  SBP.BATCH_MANAGEMENT AS BATCH_MANAGEMENT,
  SBP.PROFIT_ANALYSIS AS PROFIT_ANALYSIS,
  SBP.ACREAGE_UNIT AS ACREAGE_UNIT,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_PRODUCT_1008 SBP 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.MATERIAL_DESC AS MATERIAL_DESC,
  DWI.QR_CODE_URL AS QR_CODE_URL,
  DWI.GROSS_WEIGHT AS GROSS_WEIGHT,
  DWI.NET_WEIGHT AS NET_WEIGHT,
  DWI.LEVEL AS LEVEL,
  DWI.BUSINESS_VOLUME AS BUSINESS_VOLUME,
  DWI.MAGNITUDE AS MAGNITUDE,
  DWI.LENGTH AS LENGTH,
  DWI.WIDTH AS WIDTH,
  DWI.ACREAGE AS ACREAGE,
  DWI.HEIGHT AS HEIGHT,
  DWI.BASIC_UNIT AS BASIC_UNIT,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.CHANNEL AS CHANNEL,
  DWI.MATERIAL_GROUP AS MATERIAL_GROUP,
  DWI.EXTER_MATERIAL_GROUP AS EXTER_MATERIAL_GROUP,
  DWI.GEN_PROJ_TYPE_GROUP AS GEN_PROJ_TYPE_GROUP,
  DWI.RESERVED_FIELD5 AS RESERVED_FIELD5,
  DWI.VOLUME_UNIT AS VOLUME_UNIT,
  DWI.CLASSIFICATION AS CLASSIFICATION,
  DWI.UNIT_OF_WEIGHT AS UNIT_OF_WEIGHT,
  DWI.MEASUREMENT_UNIT AS MEASUREMENT_UNIT,
  DWI.GRADE AS GRADE,
  DWI.BATCH_MANAGEMENT AS BATCH_MANAGEMENT,
  DWI.PROFIT_ANALYSIS AS PROFIT_ANALYSIS,
  DWI.ACREAGE_UNIT AS ACREAGE_UNIT,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PRODUCT_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_PRODUCT_1008 SBP ON DWI.ID = SBP.ID
WHERE
  SBP.ID IS NULL;
