--SDI_FORMTABLE_MAIN_72_DT2_1004 到 DWI_SO_RTL_PRICE_OA_APR_DTL2_T
--SBC-DCM-A5-001-直营价格审批申请流程明细表2(零售)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL2_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.XJJE AS SUBTOTAL_AMT,
  SRC.MAINID AS MAIN_ID,
  SRC.SQZK AS APPLY_DISCOUNT,
  SRC.SPZK AS APPROVE_DISCOUNT,
  SRC.SPJ AS APP_PRICE,
  SRC.SQJ AS APPLY_PRICE,
  SRC.XSMLL AS SALE_GROSS_PROFIT_MARGIN,
  CASE
    WHEN SRC.WLBM_2 IS NULL
    OR SRC.WLBM_2 = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.WLBM_2 IS NULL
    OR SRC.WLBM_2 = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.CKCB AS REFER_COST,
  SRC.WLJC_2 AS MATERIAL_ABBREVIATION,
  SRC.FWFL AS SERVICE_FEE,
  SRC.HXM AS ITEM_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_72_DT2_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.WLBM_2 = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.SUBTOTAL_AMT,
  DWI.MAIN_ID,
  DWI.APPLY_DISCOUNT,
  DWI.APPROVE_DISCOUNT,
  DWI.APP_PRICE,
  DWI.APPLY_PRICE,
  DWI.SALE_GROSS_PROFIT_MARGIN,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.REFER_COST,
  DWI.MATERIAL_ABBREVIATION,
  DWI.SERVICE_FEE,
  DWI.ITEM_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_RTL_PRICE_OA_APR_DTL2_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_72_DT2_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
