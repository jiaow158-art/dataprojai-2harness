--SDI_FORMTABLE_MAIN_908_DSRC_1004 到 DWI_SO_DC_ENG_QUO_OA_APR_DTL_T
--DCM-A5-027-工程询价报价OA审批流程明细表(经销渠道)(新)
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_DC_ENG_QUO_OA_APR_DTL_T
SELECT
  SRC.ID AS ROW_ID,
  SRC.MAINID AS MAIN_ID,
  SRC.BDMLJEMX AS SINGLE_GUARANTEED_AMOUNT,
  SRC.BZCBMLL_MX AS STANDARD_COST_GROSS,
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,
  --工厂ID
  CASE
    WHEN SRC.GC IS NULL
    OR SRC.GC = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,
  --工厂编码
  CASE
    WHEN SRC.WLBM IS NULL
    OR SRC.WLBM = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,
  --物料ID
  CASE
    WHEN SRC.WLBM IS NULL
    OR SRC.WLBM = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,
  --物料编码
  SRC.BDJMLL_MX AS GROSS_MARGIN_RATE,
  SRC.TPZK AS EP_DISCOUNT,
  SRC.SQJG AS APPROVAL_BARE_PRICE,
  SRC.WLMS AS MATERIAL_DESC,
  SRC.BZCB_MX AS STANDARD_COST,
  SRC.CLLJ_MX AS MATERIAL_BARE_PRICE,
  SRC.ZHSPJG AS APPROVAL_QUOTATION,
  SRC.ZHBJ AS APPLICATION_QUOTATION,
  SRC.AKCB AS A_COST,
  SRC.PR00 AS EP_PRICE,
  SRC.BDJ_MX AS RESERVE_PRICE,
  SRC.SQSL AS APPLICATIONS_QTY,
  SRC.BCB AS B_COST,
  SRC.FD AS COMPREHENSIVE_SERVICE_FEE,
  SRC.CLLJFWF_MX AS MATERIAL_SERVICE_FEE,
  SRC.CKCB AS REFERENCE_COST,
  SRC.XH AS SERIAL_ID,
  SRC.YF AS FREIGHT_PIECE,
  SRC.HYCS AS FREIGHT_CITY,
  SRC.GG AS SPECIFICATIONS,
  SRC.JGF AS PROCESSING_FEE,
  SRC.ZTLSJ AS SHOWROOM_RETAIL_PRICE,
  SRC.CPQD_MX AS PRODUCT_CHANNEL,
  SRC.PC_MX AS BATCH_NUM,
  SRC.KCDD_MX AS INV_LOCATION,
  SRC.QCYY AS CLEAR_REASON,
  SRC.AMBCB_MX AS COST_AMT,
  SRC.AMBMLL_MX AS AMBA_GROSS_MARGIN_RATE,
  SRC.LABST6_MX AS INVENTORY,
  SRC.CWCB AS FINAN_COST,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_FORMTABLE_MAIN_908_DT1_1004 SRC
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC --工厂主数据表
  ON SRC.GC = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT --物料主数据表
  ON SRC.WLBM = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.MAIN_ID,
  DWI.SINGLE_GUARANTEED_AMOUNT,
  DWI.STANDARD_COST_GROSS,
  DWI.FACTORY_ID,
  DWI.FACTORY_CODE,
  DWI.MATERIAL_ID,
  DWI.MATERIAL_NUM,
  DWI.GROSS_MARGIN_RATE,
  DWI.EP_DISCOUNT,
  DWI.APPROVAL_BARE_PRICE,
  DWI.MATERIAL_DESC,
  DWI.STANDARD_COST,
  DWI.MATERIAL_BARE_PRICE,
  DWI.APPROVAL_QUOTATION,
  DWI.APPLICATION_QUOTATION,
  DWI.A_COST,
  DWI.EP_PRICE,
  DWI.RESERVE_PRICE,
  DWI.APPLICATIONS_QTY,
  DWI.B_COST,
  DWI.COMPREHENSIVE_SERVICE_FEE,
  DWI.MATERIAL_SERVICE_FEE,
  DWI.REFERENCE_COST,
  DWI.SERIAL_ID,
  DWI.FREIGHT_PIECE,
  DWI.FREIGHT_CITY,
  DWI.SPECIFICATIONS,
  DWI.PROCESSING_FEE,
  DWI.SHOWROOM_RETAIL_PRICE,
  DWI.PRODUCT_CHANNEL,
  DWI.BATCH_NUM,
  DWI.INV_LOCATION,
  DWI.CLEAR_REASON,
  DWI.COST_AMT,
  DWI.AMBA_GROSS_MARGIN_RATE,
  DWI.INVENTORY,
  DWI.FINAN_COST,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_DC_ENG_QUO_OA_APR_DTL_T DWI
  LEFT JOIN SDI.SDI_FORMTABLE_MAIN_908_DT1_1004 SDI ON DWI.ROW_ID = SDI.ID
WHERE
  SDI.ID IS NULL;
