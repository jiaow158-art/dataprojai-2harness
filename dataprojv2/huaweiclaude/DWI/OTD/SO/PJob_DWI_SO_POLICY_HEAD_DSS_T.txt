--DSS折扣政策头
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_POLICY_HEAD_DSS_T
SELECT
  SBP.ID AS ID,
  SBP.CREATE_TIME AS CREATE_TIME,
  SBP.UPDATE_TIME AS UPDATE_TIME,
  SBP.UUID AS UUID,
  SBP.POLICY_NAME AS POLICY_NAME,
  SBP.CREATE_USER AS CREATE_USER,
  SBP.CREATE_DEPT AS CREATE_DEPT,
  SBP.UPDATE_USER AS UPDATE_USER,
  SBP.SALES_ORG_CODE AS SALES_ORG_CODE,
  SBP.CHANNEL_CODE AS CHANNEL_CODE,
  SBP.STATUS AS STATUS,
  SBP.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  SBP.USE_SCOPE AS USE_SCOPE,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_POLICY_1008 SBP 
--源系统硬删除的数据打'Y' 标识
UNION ALL
SELECT
  DWI.ID AS ID,
  DWI.CREATE_TIME AS CREATE_TIME,
  DWI.UPDATE_TIME AS UPDATE_TIME,
  DWI.UUID AS UUID,
  DWI.POLICY_NAME AS POLICY_NAME,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.SALES_ORG_CODE AS SALES_ORG_CODE,
  DWI.CHANNEL_CODE AS CHANNEL_CODE,
  DWI.STATUS AS STATUS,
  DWI.SUB_COMPANY_ID AS SUB_COMPANY_ID,
  DWI.USE_SCOPE AS USE_SCOPE,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_POLICY_HEAD_DSS_T DWI
  LEFT JOIN SDI.SDI_BASE_POLICY_1008 SBP ON DWI.ID = SBP.ID
WHERE
  SBP.ID IS NULL;
