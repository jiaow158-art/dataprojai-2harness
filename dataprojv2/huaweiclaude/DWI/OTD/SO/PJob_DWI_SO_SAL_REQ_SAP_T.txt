--优化性能SQL
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
--SAP销售需求表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_SAL_REQ_SAP_T
SELECT
  SV.MANDT AS GROUP_NUM,
  SV.VBELN AS SAL_VCH_NUM,
  SV.POSNR AS SAL_VCH_ITEM_NUM,
  SV.ETENR AS SAL_PLAN_ITEM_NUM,
  R1.CONFIG_CODE AS SAL_VCH_TYPE,
  SV.AWAHR AS PROJ_SO_PROBABILITY,
  SV.BDART AS REQ_TYPE,
  SV.BERID AS MRP_SCOPE,
  SV.CHARG AS BATCH_NUM,
  SV.FSH_RALLOC_QTY AS REQ_DISTR_QTY,
  R2.CONFIG_CODE AS SUBJECT_DISTR_TYPE,
  SV.KUNNR AS CUST_NUM,
  M1.DOMAIN_VALUE_CODE AS GIVEN_INV_REVALUED,
  M2.DOMAIN_VALUE_CODE AS CONSUMPTION_ENTRY,
  CASE
    WHEN SV.WERKS IS NULL OR SV.WERKS = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SV.WERKS IS NULL OR SV.WERKS = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  SV.LGORT AS INV_LOCATION,
  CASE
    WHEN SV.MATNR IS NULL OR SV.MATNR = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SV.MATNR IS NULL OR SV.MATNR = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  CASE
    WHEN SV.MBDAT = '00000000' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    WHEN SUBSTR(SV.MBDAT, 1, 4) = '9999' THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),
      'yyyy-MM-dd'
    )
    ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SV.MBDAT, 'yyyyMMdd'), 'yyyy-MM-dd')
  END AS AVAIL_DATE,
  SV.MEINS AS BASE_MEASURE_UNIT,
  SV.NODIS AS MRP_NC_REQ_RECORD,
  SV.OMENG AS OPEN_QTY,
  SV.PLART AS PLAN_TYPE,
  SV.POSNE AS BUSINESS_PROJ_NUM,
  SV.PSPEL AS WBS_ELEMENT,
  SV.PZMNG AS REQ_PLAN_DISTR_QTY,
  R3.CONFIG_CODE AS SPECIAL_INV_FLAG,
  SV.UMREF AS CONVER_QTY,
  SV.VBELE AS BUSINESS_VCH_NUM,
  M3.DOMAIN_VALUE_CODE AS SALES_DIS_VCH_TYPE,
  SV.VMENG AS SKU_CONFIRM_QTY,
  SV.VPZUO AS DISTRIBUTION_FLAG,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_VBBE_1001 SV 
  --关联通用配置表，将SDI表相关字段进行转码处理(取通用配置表CONFIG_CODE字段)
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1 ON SV.MANDT = R1.GROUP_NUM
  AND SV.AUART = R1.SRC_SYS_CONFIG_CODE
  AND UPPER(R1.SOURCE_CODE) = 'TVAKT' --销售凭证类型
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2 ON SV.MANDT = R2.GROUP_NUM
  AND SV.KNTTP = R2.SRC_SYS_CONFIG_CODE
  AND UPPER(R2.SOURCE_CODE) = 'T163I' --帐户分配类型
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3 ON SV.MANDT = R3.GROUP_NUM
  AND SV.SOBKZ = R3.SRC_SYS_CONFIG_CODE --特殊库存描述
  AND UPPER(R3.SOURCE_CODE) = 'T148T' 
  --关联DD07T配置表，将SDI表相关字段进行转码处理(取DD07T配置表DOMAIN_VALUE_CODE字段)
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R M1 ON SV.KZBWS = M1.SRC_SYS_DOMAIN_VALUE_CODE
  AND UPPER(M1.DOMAIN_NAME) = 'KZBWS' --特定库存的评估
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R M2 ON SV.KZVBR = M2.SRC_SYS_DOMAIN_VALUE_CODE
  AND UPPER(M2.DOMAIN_NAME) = 'KZVBR' --消耗记帐
  LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R M3 ON SV.VBTYP = M3.SRC_SYS_DOMAIN_VALUE_CODE
  AND UPPER(M3.DOMAIN_NAME) = 'VBTYPL' --销售和分销凭证类别                         --分配标识
  --关联工厂主数据表取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SV.MANDT = FAC.GROUP_NUM
  AND SV.WERKS = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N' 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SV.MANDT = MAT.GROUP_NUM
  AND SV.MATNR = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.GROUP_NUM AS GROUP_NUM,
  DWI.SAL_VCH_NUM AS SAL_VCH_NUM,
  DWI.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,
  DWI.SAL_PLAN_ITEM_NUM AS SAL_PLAN_ITEM_NUM,
  DWI.SAL_VCH_TYPE AS SAL_VCH_TYPE,
  DWI.PROJ_SO_PROBABILITY AS PROJ_SO_PROBABILITY,
  DWI.REQ_TYPE AS REQ_TYPE,
  DWI.MRP_SCOPE AS MRP_SCOPE,
  DWI.BATCH_NUM AS BATCH_NUM,
  DWI.REQ_DISTR_QTY AS REQ_DISTR_QTY,
  DWI.SUBJECT_DISTR_TYPE AS SUBJECT_DISTR_TYPE,
  DWI.CUST_NUM AS CUST_NUM,
  DWI.GIVEN_INV_REVALUED AS GIVEN_INV_REVALUED,
  DWI.CONSUMPTION_ENTRY AS CONSUMPTION_ENTRY,
  DWI.FACTORY_ID AS FACTORY_ID,
  DWI.FACTORY_CODE AS FACTORY_CODE,
  DWI.INV_LOCATION AS INV_LOCATION,
  DWI.MATERIAL_ID AS MATERIAL_ID,
  DWI.MATERIAL_NUM AS MATERIAL_NUM,
  DWI.AVAIL_DATE AS AVAIL_DATE,
  DWI.BASE_MEASURE_UNIT AS BASE_MEASURE_UNIT,
  DWI.MRP_NC_REQ_RECORD AS MRP_NC_REQ_RECORD,
  DWI.OPEN_QTY AS OPEN_QTY,
  DWI.PLAN_TYPE AS PLAN_TYPE,
  DWI.BUSINESS_PROJ_NUM AS BUSINESS_PROJ_NUM,
  DWI.WBS_ELEMENT AS WBS_ELEMENT,
  DWI.REQ_PLAN_DISTR_QTY AS REQ_PLAN_DISTR_QTY,
  DWI.SPECIAL_INV_FLAG AS SPECIAL_INV_FLAG,
  DWI.CONVER_QTY AS CONVER_QTY,
  DWI.BUSINESS_VCH_NUM AS BUSINESS_VCH_NUM,
  DWI.SALES_DIS_VCH_TYPE AS SALES_DIS_VCH_TYPE,
  DWI.SKU_CONFIRM_QTY AS SKU_CONFIRM_QTY,
  DWI.DISTRIBUTION_FLAG AS DISTRIBUTION_FLAG,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_SAL_REQ_SAP_T DWI
  LEFT JOIN SDI.SDI_VBBE_1001 SV ON DWI.GROUP_NUM = SV.MANDT
  AND DWI.SAL_VCH_NUM = SV.VBELN
  AND DWI.SAL_VCH_ITEM_NUM = SV.POSNR
  AND DWI.SAL_PLAN_ITEM_NUM = SV.ETENR
WHERE
  SV.VBELN IS NULL;
