--DSS修改订单行
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_ITEM_MODIFY_DSS_T
SELECT
  SBOMI.ID AS ID,
  SBOMI.UPDATE_TIME AS UPDATE_TIME,
  SBOMI.CREATE_TIME AS CREATE_TIME,
  SBOMI.ORDER_MODIFY_ID AS ORDER_MODIFY_ID,
  SBOMI.ORDER_ITEM_ID AS ORDER_ITEM_ID,

  SBOMI.DISCOUNT_RATE AS DISCOUNT_RATE,
  SBOMI.FINAL_PRICE AS FINAL_PRICE,
  SBOMI.PRICE AS PRICE,
  CASE
    WHEN SBOMI.FACTORY_CODE IS NULL OR SBOMI.FACTORY_CODE = '' THEN -999999
    ELSE NVL(FAC.FACTORY_ID, -999998)
  END AS FACTORY_ID,--工厂ID
  CASE
    WHEN SBOMI.FACTORY_CODE IS NULL OR SBOMI.FACTORY_CODE = '' THEN 'SNULL'
    ELSE NVL(FAC.FACTORY_CODE, 'NOT EXISTS')
  END AS FACTORY_CODE,--工厂编码
  CASE
    WHEN SBOMI.PRODUCT_CODE IS NULL OR SBOMI.PRODUCT_CODE = '' THEN -999999
    ELSE NVL(MAT.MATERIAL_ID, -999998)
  END AS MATERIAL_ID,--物料ID
  CASE
    WHEN SBOMI.PRODUCT_CODE IS NULL OR SBOMI.PRODUCT_CODE = '' THEN 'SNULL'
    ELSE NVL(MAT.MATERIAL_NUM, 'NOT EXISTS')
  END AS MATERIAL_NUM,--物料编码
  SBOMI.PRODUCT_NAME AS PRODUCT_NAME,
  SBOMI.DIRECT_COST AS DIRECT_COST,
  SBOMI.OPERATE_LOG AS OPERATE_LOG,
  SBOMI.QUANTITY AS QUANTITY,
  SBOMI.MODIFY_QUANTITY AS MODIFY_QUANTITY,
  SBOMI.BRICK_AREA AS BRICK_AREA,

  SBOMI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  SBOMI.CREATE_DEPT AS CREATE_DEPT,
  SBOMI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  SBOMI.ITEM_NUMBER AS ITEM_NUMBER,
  SBOMI.INVENTORY_CODE AS INVENTORY_CODE,
  SBOMI.DESIGNER_FD_POINTS AS DESIGNER_FD_POINTS,
  SBOMI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  SBOMI.REMARK AS REMARK,
  SBOMI.PRODUCT_UNIT AS PRODUCT_UNIT,
  SBOMI.COLLECT_FREIGHT AS COLLECT_FREIGHT,
  SBOMI.YDJF_POINT_SCORE AS YDJF_POINT_SCORE,
  SBOMI.NDJF_POINT_SCORE AS NDJF_POINT_SCORE,
  SBOMI.PROCESSING_TYPE_ID AS PROCESSING_TYPE_ID,
  SBOMI.PROCESSING_UNIT AS PROCESSING_UNIT,
  SBOMI.FLAG AS FLAG,
  SBOMI.GDCP_POINT_SCORE AS GDCP_POINT_SCORE,
  SBOMI.JDJF_POINT_SCORE AS JDJF_POINT_SCORE,
  SBOMI.JXGG_POINT_SCORE AS JXGG_POINT_SCORE,
  SBOMI.QYGG_POINT_SCORE AS QYGG_POINT_SCORE,
  SBOMI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,
  SBOMI.STATUS AS STATUS,
  SBOMI.GDCP_POINT_FLAG AS GDCP_POINT_FLAG,
  SBOMI.NDJF_POINT_FLAG AS NDJF_POINT_FLAG,
  SBOMI.YDJF_POINT_FLAG AS YDJF_POINT_FLAG,
  SBOMI.JDJF_POINT_FLAG AS JDJF_POINT_FLAG,
  SBOMI.XCZL_POINT_SCORE AS XCZL_POINT_SCORE,
  SBOMI.JXGGJF_POINT_FLAG AS JXGGJF_POINT_FLAG,
  SBOMI.ZXJF_POINT_FLAG AS ZXJF_POINT_FLAG,
  SBOMI.ZXJF_POINT_SCORE AS ZXJF_POINT_SCORE,
  SBOMI.XCZL_POINT_FLAG AS XCZL_POINT_FLAG,
  SBOMI.QYGG_POINT_FLAG AS QYGG_POINT_FLAG,
  SBOMI.DQGG_POINT_FLAG AS DQGG_POINT_FLAG,
  SBOMI.PROJECT_TYPE AS PROJECT_TYPE,
  SBOMI.POINT_DISCOUNT AS POINT_DISCOUNT,
  SBOMI.DQGG_POINT_SCORE AS DQGG_POINT_SCORE,
  SBOMI.COMMISSION_POINTS AS COMMISSION_POINTS,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_BASE_ORDER_MODIFY_ITEM_1008 SBOMI 
  --关联物料主数据表取物料ID、编码
  LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ON SBOMI.PRODUCT_CODE = MAT.MATERIAL_NUM
  AND MAT.END_DATE = '4712-12-31'
  AND MAT.DEL_FLAG = 'N'
  AND MAT.GROUP_NUM = '800' 
  --关联工厂主数据表DWI_MD_FACTORY_T取工厂ID、编码
  LEFT JOIN DWIMD.DWI_MD_FACTORY_T FAC ON SBOMI.FACTORY_CODE = FAC.FACTORY_CODE
  AND FAC.END_DATE = '4712-12-31'
  AND FAC.DEL_FLAG = 'N'
  AND FAC.GROUP_NUM = '800'
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ID AS ID,

  DWI.WEIGHT AS WEIGHT,
  DWI.DEFAULT_EXPECT_DATE AS DEFAULT_EXPECT_DATE,
  DWI.EXPECT_DATE AS EXPECT_DATE,
  DWI.REFUSAL_CAUSE AS REFUSAL_CAUSE,
  DWI.PROCESSING_QUANTITY AS PROCESSING_QUANTITY,
  DWI.CREATE_USER AS CREATE_USER,
  DWI.UPDATE_USER AS UPDATE_USER,
  DWI.STAY_GOODS_ITEM_CODE AS STAY_GOODS_ITEM_CODE,
  DWI.PROCESSING_DESCRIPTION AS PROCESSING_DESCRIPTION,
  DWI.CREATE_DEPT AS CREATE_DEPT,
  DWI.SAP_ITEM_CODE AS SAP_ITEM_CODE,
  DWI.ITEM_NUMBER AS ITEM_NUMBER,
  DWI.INVENTORY_CODE AS INVENTORY_CODE,

  DWI.STAY_GOODS_CODE AS STAY_GOODS_CODE,
  DWI.REMARK AS REMARK,

  DWI.YDJF_POINT_SCORE AS YDJF_POINT_SCORE,

  DWI.DESIGNER_FD_COEFFICIENT AS DESIGNER_FD_COEFFICIENT,

  DWI.NDJF_POINT_FLAG AS NDJF_POINT_FLAG,
  DWI.YDJF_POINT_FLAG AS YDJF_POINT_FLAG,
  DWI.JDJF_POINT_FLAG AS JDJF_POINT_FLAG,
  DWI.XCZL_POINT_SCORE AS XCZL_POINT_SCORE,
  DWI.JXGGJF_POINT_FLAG AS JXGGJF_POINT_FLAG,
  DWI.ZXJF_POINT_FLAG AS ZXJF_POINT_FLAG,
  DWI.ZXJF_POINT_SCORE AS ZXJF_POINT_SCORE,
  DWI.XCZL_POINT_FLAG AS XCZL_POINT_FLAG,
  DWI.QYGG_POINT_FLAG AS QYGG_POINT_FLAG,
  DWI.DQGG_POINT_FLAG AS DQGG_POINT_FLAG,
  DWI.PROJECT_TYPE AS PROJECT_TYPE,
  DWI.POINT_DISCOUNT AS POINT_DISCOUNT,
  DWI.DQGG_POINT_SCORE AS DQGG_POINT_SCORE,
  DWI.COMMISSION_POINTS AS COMMISSION_POINTS,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_ITEM_MODIFY_DSS_T DWI
