--SDI_CX_QUATO_PLAN_1002 到 DWI_SO_QUATO_PLAN_OSS_T
--报价单与计划单中间表
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_QUATO_PLAN_OSS_T
SELECT
  SRC.ROW_ID,
  SRC.CREATED,
  SRC.DB_LAST_UPD,
  SRC.LAST_UPD,
  SRC.PLAN_ITEM_ID,
  SRC.QUATO_ITEM_ID,
  SRC.APPLY_QUANTITY,
  SRC.COMMENTS,
  SRC.STATUS,
  SRC.MODIFICATION_NUM,
  SRC.DB_LAST_UPD_SRC,
  SRC.LAST_UPD_BY,
  SRC.CONFLICT_ID,
  SRC.CREATED_BY,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_QUATO_PLAN_1002 SRC
UNION ALL
  --更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
SELECT
  DWI.ROW_ID,
  DWI.CREATE_DATE,
  DWI.DB_LAST_UPDATE_DATE,
  DWI.LAST_UPDATE_DATE,
  DWI.PLAN_ITEM_ID,
  DWI.QUATO_ITEM_ID,
  DWI.APPLY_QUANTITY,
  DWI.COMMENTS,
  DWI.STATUS,
  DWI.MODIFY_QTY,
  DWI.DB_LAST_UPDATE_DATE_SRC,
  DWI.LAST_UPDATE_BY,
  DWI.CONFLICT_ID,
  DWI.CREATED_BY,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_QUATO_PLAN_OSS_T DWI
  LEFT JOIN SDI.SDI_CX_QUATO_PLAN_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
