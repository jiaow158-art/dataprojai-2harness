--SDI_CX_PAY_ORDER_1002 到 DWI_SO_PAY_ORDER_T
--付款单
INSERT
  OVERWRITE TABLE DWIOTD.DWI_SO_PAY_ORDER_T
SELECT
  SDI.ROW_ID AS ROW_ID,
  SDI.ACT_UPDATED_BY AS ACT_UPDATED_BY,
  SDI.ACT_CREATED_BY AS ACT_CREATED_BY,
  SDI.ACCOUNT_ID AS ACCOUNT_ID,
  CASE
    WHEN SDI.EXCHANGE_DATE IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.EXCHANGE_DATE
  END AS EXCHANGE_DATE,
  SDI.AMOUNT AS AMOUNT,
  SDI.CONFLICT_ID AS CONFLICT_ID,
  SDI.MODIFICATION_NUM AS MODIFICATION_NUM,
  SDI.LAST_UPD_BY AS LAST_UPD_BY,
  CASE
    WHEN SDI.LAST_UPD IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.LAST_UPD
  END AS LAST_UPD_DATE,
  SDI.CREATED_BY AS CREATED_BY,
  CASE
    WHEN SDI.CREATED IS NULL THEN FROM_UNIXTIME(
      UNIX_TIMESTAMP('19000101000000', 'yyyyMMddhhmmss'),
      'yyyy-MM-dd hh:mm:ss'
    )
    ELSE SDI.CREATED
  END AS CREATED_DATE,
  SDI.STLMEN_TEMS AS STLMEN_TEMS,
  SDI.STATUS AS STATUS,
  SDI.RECEVING_NAME AS RECEVING_NAME,
  SDI.RECEVING_ACCOUNT AS RECEVING_ACCOUNT,
  SDI.PAY_FOLLOW_ID AS PAY_FOLLOW_ID,
  SDI.PAY_FOLLOW AS PAY_FOLLOW,
  SDI.PAYMENT_TYPE AS PAYMENT_TYPE,
  SDI.PAYER_NAME AS PAYER_NAME,
  SDI.PAYER_ACCOUNT AS PAYER_ACCOUNT,
  SDI.COMPANY_NAME AS COMPANY_NAME,
  SDI.COMPANY_ID AS COMPANY_ID,
  SDI.BANK_ORDER_NUMBER AS BANK_ORDER_NUMBER,
  SDI.BANK_ORDER_ID AS BANK_ORDER_ID,
  'N' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  SDI.SDI_CX_PAY_ORDER_1002 SDI
UNION ALL
  --源系统硬删除的数据打'Y' 标识
SELECT
  DWI.ROW_ID,
  DWI.ACT_UPDATED_BY,
  DWI.ACT_CREATED_BY,
  DWI.ACCOUNT_ID,
  DWI.EXCHANGE_DATE,
  DWI.AMOUNT,
  DWI.CONFLICT_ID,
  DWI.MODIFICATION_NUM,
  DWI.LAST_UPD_BY,
  DWI.LAST_UPD_DATE,
  DWI.CREATED_BY,
  DWI.CREATED_DATE,
  DWI.STLMEN_TEMS,
  DWI.STATUS,
  DWI.RECEVING_NAME,
  DWI.RECEVING_ACCOUNT,
  DWI.PAY_FOLLOW_ID,
  DWI.PAY_FOLLOW,
  DWI.PAYMENT_TYPE,
  DWI.PAYER_NAME,
  DWI.PAYER_ACCOUNT,
  DWI.COMPANY_NAME,
  DWI.COMPANY_ID,
  DWI.BANK_ORDER_NUMBER,
  DWI.BANK_ORDER_ID,
  'Y' AS DEL_FLAG,
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM
  DWIOTD.DWI_SO_PAY_ORDER_T DWI
  LEFT JOIN SDI.SDI_CX_PAY_ORDER_1002 SDI ON DWI.ROW_ID = SDI.ROW_ID
WHERE
  SDI.ROW_ID IS NULL;
