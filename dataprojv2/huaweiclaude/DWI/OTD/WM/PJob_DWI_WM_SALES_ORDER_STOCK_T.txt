set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
WITH Y_FALG_DATA AS(  -- 源表SDI.SAP销售订单库存数据删了，打Y标识的数据
	SELECT 
	DWSOS.GROUP_NUM,
	DWSOS.MATERIAL_NUM,
	DWSOS.FACTORY_CODE,
	DWSOS.STOCK_PLACE,
	DWSOS.BATCH_NUM,
	DWSOS.SPECIAL_STOCK_FLAG,
	DWSOS.RETAIL_VOUCHER_NO,
	DWSOS.RETAIL_VOUCHER_PROJECT_NO,
	DWSOS.CURRENT_STOCK_FIN_YEAR,
	DWSOS.CREATE_DATE,
	DWSOS.CURRENT_PERIOD,
	DWSOS.CURRENT_PERIOD_FISCAL_YEAR,
	DWSOS.FROZEN_STOCK,
	DWSOS.LAST_FROZEN_STOCK,
	DWSOS.DSTRT_STOCK_QTY,
	DWSOS.CHECK_STOCK,
	DWSOS.LAST_QUALITY_CHECK_STOCK,
	DWSOS.LAST_PASS_CHECK_DATE,
	DWSOS.LAST_LIMIT_USE_STOCK,
	'Y' AS DEL_FLAG
FROM DWIOTD.DWI_WM_SALES_ORDER_STOCK_T DWSOS		--DWI.SAP销售订单库存
LEFT JOIN SDI.SDI_MSKA_1001 SM1			--SDI.SAP销售订单库存
	 ON DWSOS.GROUP_NUM = SM1.MANDT 			--集团					
     AND DWSOS.MATERIAL_NUM = SM1.MATNR 		--物料编号
	 AND DWSOS.FACTORY_CODE = SM1.WERKS			--工厂编码
	 AND DWSOS.STOCK_PLACE = SM1.LGORT			--库存地点
	 AND DWSOS.BATCH_NUM = SM1.CHARG			--批号
	 AND DWSOS.SPECIAL_STOCK_FLAG = SM1.SOBKZ	--特殊库存标识
	 AND DWSOS.RETAIL_VOUCHER_NO = SM1.VBELN	--销售和分销凭证号
	 AND DWSOS.RETAIL_VOUCHER_PROJECT_NO = SM1.POSNR	--销售和分销凭证的项目号
	 WHERE SM1.MANDT  IS NULL
     AND SM1.MATNR  IS NULL
	 AND SM1.WERKS  IS NULL		
	 AND SM1.LGORT  IS NULL	
	 AND SM1.CHARG  IS NULL	
	 AND SM1.SOBKZ  IS NULL
	 AND SM1.VBELN  IS NULL
	 AND SM1.POSNR  IS NULL
	 AND SM1.DEL_FLAG = 'N'),
SDI_AND_Y_FLAG_DATA AS( -- SDI与打Y标识数据
   SELECT
   SM1.MANDT			AS GROUP_NUM,
   SM1.MATNR            AS MATERIAL_NUM,
   SM1.WERKS            AS FACTORY_CODE,
   SM1.LGORT            AS STOCK_PLACE,
   SM1.CHARG            AS BATCH_NUM,
   SM1.SOBKZ            AS SPECIAL_STOCK_FLAG,
   SM1.VBELN            AS RETAIL_VOUCHER_NO,
   SM1.POSNR            AS RETAIL_VOUCHER_PROJECT_NO,
   SM1.KAJIN            AS CURRENT_STOCK_FIN_YEAR,
   SM1.ERSDA            AS CREATE_DATE,
   SM1.LFMON            AS CURRENT_PERIOD,
   SM1.LFGJA            AS CURRENT_PERIOD_FISCAL_YEAR,
   SM1.KASPE            AS FROZEN_STOCK,
   SM1.KAVSP            AS LAST_FROZEN_STOCK,
   SM1.FSH_SALLOC_QTY   AS DSTRT_STOCK_QTY,
   SM1.KAINS            AS CHECK_STOCK,
   SM1.KAVIN            AS LAST_QUALITY_CHECK_STOCK,
   SM1.KADLL            AS LAST_PASS_CHECK_DATE,
   SM1.KAVEI            AS LAST_LIMIT_USE_STOCK,
   SM1.DEL_FLAG 		AS DEL_FLAG
FROM SDI.SDI_MSKA_1001 SM1		--SDI.SAP销售订单库存
UNION ALL		--SDI_MSKA_1001.SAP销售订单库存 UNION ALL 拼接 DWI_WM_SALES_ORDER_STOCK_T.SAP销售订单库存
SELECT 
DWSOS.GROUP_NUM,
DWSOS.MATERIAL_NUM,
DWSOS.FACTORY_CODE,
DWSOS.STOCK_PLACE,
DWSOS.BATCH_NUM,
DWSOS.SPECIAL_STOCK_FLAG,
DWSOS.RETAIL_VOUCHER_NO,
DWSOS.RETAIL_VOUCHER_PROJECT_NO,
DWSOS.CURRENT_STOCK_FIN_YEAR,
DWSOS.CREATE_DATE,
DWSOS.CURRENT_PERIOD,
DWSOS.CURRENT_PERIOD_FISCAL_YEAR,
DWSOS.FROZEN_STOCK,
DWSOS.LAST_FROZEN_STOCK,
DWSOS.DSTRT_STOCK_QTY,
DWSOS.CHECK_STOCK,
DWSOS.LAST_QUALITY_CHECK_STOCK,
DWSOS.LAST_PASS_CHECK_DATE,
DWSOS.LAST_LIMIT_USE_STOCK,
DWSOS.DEL_FLAG
FROM Y_FALG_DATA DWSOS)
INSERT OVERWRITE TABLE DWIOTD.DWI_WM_SALES_ORDER_STOCK_T		--DWI.SAP销售订单库存
SELECT --- 关联主数据表
		SM1.GROUP_NUM					AS	GROUP_NUM,
		CASE WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''		--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)		--关联不上主数据时赋值-999998
        END 						AS  MATERIAL_ID, 	--物料ID
		CASE
		WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')		--关联不上主数据时赋值'NOT EXISTS'
		END 						AS MATERIAL_NUM,	--物料编号
		CASE WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''		--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)	--关联不上主数据时赋值-999998
        END 						AS  FACTORY_ID, 	--工厂ID
		CASE
		WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 						AS 	FACTORY_CODE,	--工厂ID
		SM1.STOCK_PLACE                   AS	STOCK_PLACE,
		SM1.BATCH_NUM                   AS	BATCH_NUM,
		T.CONFIG_CODE           	AS	SPECIAL_STOCK_FLAG,
		SM1.RETAIL_VOUCHER_NO                   AS	RETAIL_VOUCHER_NO,
		SM1.RETAIL_VOUCHER_PROJECT_NO                   AS	RETAIL_VOUCHER_PROJECT_NO,
		SM1.CURRENT_STOCK_FIN_YEAR                   AS	CURRENT_STOCK_FIN_YEAR,
		CASE WHEN SM1.CREATE_DATE = '00000000' 	--判断SDI源表字段创建日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SM1.CREATE_DATE, 1, 4) = '9999' --SDI源表字段创建日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SM1.CREATE_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 						AS  CREATE_DATE,
		SM1.CURRENT_PERIOD                   AS	CURRENT_PERIOD,
		SM1.CURRENT_PERIOD_FISCAL_YEAR                   AS	CURRENT_PERIOD_FISCAL_YEAR,
		SM1.FROZEN_STOCK                   AS	FROZEN_STOCK,
		SM1.LAST_FROZEN_STOCK                   AS	LAST_FROZEN_STOCK,
		SM1.DSTRT_STOCK_QTY          AS	DSTRT_STOCK_QTY,
		SM1.CHECK_STOCK                   AS	CHECK_STOCK,
		SM1.LAST_QUALITY_CHECK_STOCK                   AS	LAST_QUALITY_CHECK_STOCK,
		CASE WHEN SM1.LAST_PASS_CHECK_DATE = '00000000' 	--判断SDI源表字段上次过账的盘点日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SM1.LAST_PASS_CHECK_DATE, 1, 4) = '9999' --SDI源表字段上次过账的盘点日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SM1.LAST_PASS_CHECK_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END 						AS  LAST_PASS_CHECK_DATE,
		SM1.LAST_LIMIT_USE_STOCK                   AS	LAST_LIMIT_USE_STOCK,
		SM1.DEL_FLAG 			AS DEL_FLAG,
		CURRENT_TIMESTAMP() 		AS  DW_LAST_UPDATE_DATE
		FROM SDI_AND_Y_FLAG_DATA SM1			--SDI.SAP销售订单库存
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据取物料ID
			 ON  SM1.GROUP_NUM = DMMG.GROUP_NUM			--集团
			 AND SM1.MATERIAL_NUM = DMMG.MATERIAL_NUM	--物料编号
			 AND DMMG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
			 AND DMMG.DEL_FLAG = 'N'					--删除标识为'N'
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF			--关联工厂主数据取工厂ID
			 ON  SM1.GROUP_NUM = DMF.GROUP_NUM			--集团
			 AND SM1.FACTORY_CODE = DMF.FACTORY_CODE	--工厂编码
			 AND DMF.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
			 AND DMF.DEL_FLAG = 'N'						--删除标识为'N'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T		--关联通用配置表
			 ON T.SRC_SYS_CONFIG_CODE = SM1.SPECIAL_STOCK_FLAG	--取特殊库存描述中文名称
			 AND T.GROUP_NUM = SM1.GROUP_NUM			--集团
			 AND T.SOURCE_CODE = 'T148T'
