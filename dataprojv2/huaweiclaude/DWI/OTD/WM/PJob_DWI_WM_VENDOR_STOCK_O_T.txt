set hive.cbo.enable=true;		--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
WITH Y_FALG_DATA AS(  -- 源表SDI.SAP供应商特殊库存(O)数据删了，打Y标识的数据
	SELECT
	DWVSO.GROUP_NUM,
	DWVSO.MATERIAL_NUM,
	DWVSO.FACTORY_CODE,
	DWVSO.BATCH_NUM,
	DWVSO.SPECIAL_STOCK_FLAG,
	DWVSO.SUPPLIER_OR_CREDITOR_NUM,
	DWVSO.CURRENT_STOCK_FIN_YEAR,
	DWVSO.CREATE_DATE,
	DWVSO.CURRENT_PERIOD,
	DWVSO.CURRENT_PERIOD_FISCAL_YEAR,
	DWVSO.LIMIT_BATCH_TOTOAL_STOCK,
	DWVSO.CHECK_STOCK,
	DWVSO.WX_WORK_TRANSFER_STOCK,
	DWVSO.LAST_QUALITY_CHECK_STOCK,
	DWVSO.LAST_PASS_CHECK_DATE,
	DWVSO.LAST_LIMIT_USE_STOCK,
	'Y' AS DEL_FLAG
FROM DWIOTD.DWI_WM_VENDOR_STOCK_O_T DWVSO	--DWI.SAP供应商特殊库存(O)
LEFT JOIN SDI.SDI_MSLB_1001 SM1		--SDI.SAP供应商特殊库存(O)
 ON DWVSO.GROUP_NUM = SM1.MANDT 	--集团							
 AND DWVSO.MATERIAL_NUM = SM1.MATNR 	--物料编号
 AND DWVSO.FACTORY_CODE = SM1.WERKS		--工厂编码
 AND DWVSO.BATCH_NUM = SM1.CHARG		--批号
 AND DWVSO.SPECIAL_STOCK_FLAG = SM1.SOBKZ	--特殊库存标识
 AND DWVSO.SUPPLIER_OR_CREDITOR_NUM = SM1.LIFNR	--供应商或债权人的帐号
 WHERE SM1.MANDT  IS NULL
 AND SM1.MATNR  IS NULL
 AND SM1.WERKS  IS NULL		
 AND SM1.CHARG  IS NULL	
 AND SM1.SOBKZ  IS NULL	
 AND SM1.LIFNR  IS NULL
 AND SM1.DEL_FLAG = 'N'),
 SDI_AND_Y_FLAG_DATA AS( -- SDI与打Y标识数据
   SELECT
	SM1.MANDT		AS	GROUP_NUM,							
	SM1.MATNR       AS	MATERIAL_NUM,                       
	SM1.WERKS       AS	FACTORY_CODE,                       
	SM1.CHARG       AS	BATCH_NUM,                          
	SM1.SOBKZ       AS	SPECIAL_STOCK_FLAG,                 
	SM1.LIFNR       AS	SUPPLIER_OR_CREDITOR_NUM,           
	SM1.LBJIN       AS	CURRENT_STOCK_FIN_YEAR,             
	SM1.ERSDA       AS	CREATE_DATE,                        
	SM1.LFMON       AS	CURRENT_PERIOD,                     
	SM1.LFGJA       AS	CURRENT_PERIOD_FISCAL_YEAR,         
	SM1.LBEIN       AS	LIMIT_BATCH_TOTOAL_STOCK,           
	SM1.LBINS       AS	CHECK_STOCK,                        
	SM1.LBUML       AS	WX_WORK_TRANSFER_STOCK,             
	SM1.LBVIN       AS	LAST_QUALITY_CHECK_STOCK,           
	SM1.LBDLL       AS	LAST_PASS_CHECK_DATE,               
	SM1.LBVEI       AS	LAST_LIMIT_USE_STOCK,
	SM1.DEL_FLAG    AS  DEL_FLAG
FROM SDI.SDI_MSLB_1001 SM1		--SDI.SAP供应商特殊库存(O)
UNION ALL		--SDI_MSLB_1001.SAP供应商特殊库存(O) UNION ALL 拼接 DWI_WM_VENDOR_STOCK_O_T.SAP供应商特殊库存(O)
SELECT
	DWVSO.GROUP_NUM,								
	DWVSO.MATERIAL_NUM,                             
	DWVSO.FACTORY_CODE,                             
	DWVSO.BATCH_NUM,                                
	DWVSO.SPECIAL_STOCK_FLAG,                       
	DWVSO.SUPPLIER_OR_CREDITOR_NUM,                 
	DWVSO.CURRENT_STOCK_FIN_YEAR,                   
	DWVSO.CREATE_DATE,                              
	DWVSO.CURRENT_PERIOD,                           
	DWVSO.CURRENT_PERIOD_FISCAL_YEAR,               
	DWVSO.LIMIT_BATCH_TOTOAL_STOCK,                 
	DWVSO.CHECK_STOCK,                              
	DWVSO.WX_WORK_TRANSFER_STOCK,                   
	DWVSO.LAST_QUALITY_CHECK_STOCK,                 
	DWVSO.LAST_PASS_CHECK_DATE,                     
	DWVSO.LAST_LIMIT_USE_STOCK,                     
	DWVSO.DEL_FLAG
FROM Y_FALG_DATA DWVSO)
INSERT OVERWRITE TABLE DWIOTD.DWI_WM_VENDOR_STOCK_O_T		--DWI.SAP供应商特殊库存(O)
SELECT 
		SM1.GROUP_NUM			AS		GROUP_NUM,
		CASE WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''	----关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMMG.MATERIAL_ID, -999998)		----关联不上主数据时赋值-999998
        END    			AS      MATERIAL_ID,	--物料ID
		CASE
		WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(SM1.MATERIAL_NUM, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 			AS MATERIAL_NUM,	--物料编号
		CASE WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''	--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)	--关联不上主数据时赋值-999998
        END 			AS 		FACTORY_ID, 	--工厂ID
		CASE
		WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(SM1.FACTORY_CODE, 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 			 AS		FACTORY_CODE,	--工厂编码
		SM1.BATCH_NUM           AS		BATCH_NUM,
		T.CONFIG_CODE   AS		SPECIAL_STOCK_FLAG,
		CASE WHEN SM1.SUPPLIER_OR_CREDITOR_NUM IS NULL OR SM1.SUPPLIER_OR_CREDITOR_NUM = ''	--关联供应商主数据取工厂ID,判断SDI源表字段供应商或债权人的帐号为空或者空字符串时赋值-999999
		THEN -999999
        ELSE NVL(DMSG.SUPPLIER_OR_CREDITOR_ID, -999998)	--关联不上主数据时赋值-999998
		END 			AS 		SUPPLIER_OR_CREDITOR_ID,	--供应商ID
		CASE
		WHEN SM1.SUPPLIER_OR_CREDITOR_NUM IS NULL OR SM1.SUPPLIER_OR_CREDITOR_NUM = ''	--关联工厂主数据,判断SDI源表字段供应商或债权人的帐号为空或者空字符串时赋值'SNULL'
		THEN 'SNULL'
		ELSE NVL(SM1.SUPPLIER_OR_CREDITOR_NUM , 'NOT EXISTS')	--关联不上主数据时赋值'NOT EXISTS'
		END 			AS		SUPPLIER_OR_CREDITOR_NUM,	--供应商编号
		SM1.CURRENT_STOCK_FIN_YEAR           AS		CURRENT_STOCK_FIN_YEAR,
		CASE WHEN SM1.CREATE_DATE = '00000000' 	--判断SDI源表字段创建日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SM1.CREATE_DATE, 1, 4) = '9999' --SDI源表字段创建日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SM1.CREATE_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END             AS 		CREATE_DATE,
		SM1.CURRENT_PERIOD           AS		CURRENT_PERIOD,
		SM1.CURRENT_PERIOD_FISCAL_YEAR           AS		CURRENT_PERIOD_FISCAL_YEAR,
		SM1.LIMIT_BATCH_TOTOAL_STOCK           AS		LIMIT_BATCH_TOTOAL_STOCK,
		SM1.CHECK_STOCK           AS		CHECK_STOCK,
		SM1.WX_WORK_TRANSFER_STOCK           AS		WX_WORK_TRANSFER_STOCK,
		SM1.LAST_QUALITY_CHECK_STOCK           AS		LAST_QUALITY_CHECK_STOCK,
		CASE WHEN SM1.LAST_PASS_CHECK_DATE = '00000000' --判断SDI源表字段上次过账的盘点日期为00000000
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('19000101', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为19000101,1900-01-01
		WHEN SUBSTR(SM1.LAST_PASS_CHECK_DATE, 1, 4) = '9999' --SDI源表字段上次过账的盘点日期为9999
		THEN FROM_UNIXTIME(UNIX_TIMESTAMP('47121231', 'yyyyMMdd'),'yyyy-MM-dd')	--转换日期为47121231,4712-12-31
		ELSE FROM_UNIXTIME(UNIX_TIMESTAMP(SM1.LAST_PASS_CHECK_DATE, 'yyyyMMdd'), 'yyyy-MM-dd')
		END             AS 		LAST_PASS_CHECK_DATE,
		SM1.LAST_LIMIT_USE_STOCK           AS		LAST_LIMIT_USE_STOCK,
		SM1.DEL_FLAG			    AS 		DEL_FLAG,
		CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE
FROM SDI_AND_Y_FLAG_DATA SM1		--SDI.SAP供应商特殊库存(O)
INNER JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG	--关联物料主数据取物料ID
	 ON  SM1.GROUP_NUM = DMMG.GROUP_NUM			--集团
	 AND SM1.MATERIAL_NUM = DMMG.MATERIAL_NUM	--物料编号
	 AND DMMG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
INNER JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T DMSG	--关联供应商主数据取供应商ID
	 ON  SM1.GROUP_NUM = DMSG.GROUP_NUM			--集团
	 AND SM1.SUPPLIER_OR_CREDITOR_NUM = DMSG.SUPPLIER_OR_CREDITOR_NUM	--供应商编号
	 AND DMSG.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
INNER JOIN DWIMD.DWI_MD_FACTORY_T DMF			--关联工厂主数据取工厂ID
	 ON  SM1.GROUP_NUM = DMF.GROUP_NUM			--集团
	 AND SM1.FACTORY_CODE = DMF.FACTORY_CODE	--工厂编码
	 AND DMF.END_DATE = '4712-12-31'			--结束时间为'4712-12-31'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T	--关联通用配置表
	 ON T.SRC_SYS_CONFIG_CODE = SM1.SPECIAL_STOCK_FLAG	--取特殊库存描述中文名称
	 AND T.GROUP_NUM = SM1.GROUP_NUM			--集团
	 AND T.SOURCE_CODE = 'T148T'
