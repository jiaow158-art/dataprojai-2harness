set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
WITH Y_FALG_DATA AS(  -- 源表数据删了，打Y标识的数据
SELECT 
 B.GROUP_NUM
,B.MATERIAL_NUM
,B.FACTORY_CODE
,B.STOCK_PLACE
,B.BATCH_NUM
,B.SPECIAL_STOCK_FLAG
,B.WBS_ELEMENT
,B.LAST_MXXX_PERIOD
,B.CURRENT_STOCK_FIN_YEAR
,B.CREATE_DATE
,B.CURRENT_PERIOD
,B.CURRENT_PERIOD_FISCAL_YEAR
,B.FROZEN_STOCK
,B.LAST_FROZEN_STOCK
,B.LIMIT_BATCH_TOTOAL_STOCK
,B.CHECK_STOCK
,B.LAST_QUALITY_CHECK_STOCK
,B.UNRESTRICTED_EVALUATED_STOCK
,B.LAST_PASS_CHECK_DATE
,B.LAST_LIMIT_USE_STOCK
,'Y' AS DEL_FLAG
FROM DWIOTD.DWI_WM_PROJECT_STOCK_T B		--DWI.SAP项目库存表
LEFT JOIN SDI.SDI_MSPR_1001 B1				--SDI.SAP项目库存表
ON B.GROUP_NUM = B1.MANDT					--集团
AND B.MATERIAL_NUM = B1.MATNR				--物料编号
AND B.FACTORY_CODE = B1.WERKS				--工厂编码
AND B.STOCK_PLACE = B1.LGORT				--库存地点
AND B.BATCH_NUM = B1.CHARG					--批号
AND B.SPECIAL_STOCK_FLAG = B1.SOBKZ			--特殊库存标识
AND B.WBS_ELEMENT = B1.PSPNR				--工作分解结构元素 (WBS 元素)
WHERE B1.MANDT  IS NULL
AND B1.MATNR  IS NULL
AND B1.WERKS  IS NULL
AND B1.LGORT  IS NULL
AND B1.CHARG  IS NULL
AND B1.SOBKZ  IS NULL
AND B1.PSPNR  IS NULL
AND B1.DEL_FLAG = 'N'),
SDI_AND_Y_FLAG_DATA AS( -- SDI与打Y标识数据
   SELECT
SM1.MANDT  AS GROUP_NUM,
SM1.MATNR  AS MATERIAL_NUM,
SM1.WERKS  AS FACTORY_CODE,
SM1.LGORT  AS STOCK_PLACE,
SM1.CHARG  AS BATCH_NUM,
SM1.SOBKZ  AS SPECIAL_STOCK_FLAG,
SM1.PSPNR  AS WBS_ELEMENT,
SM1.PRRUE       AS	LAST_MXXX_PERIOD,
SM1.PRJIN       AS	CURRENT_STOCK_FIN_YEAR,
SM1.ERSDA       AS	CREATE_DATE,
SM1.LFMON       AS	CURRENT_PERIOD,
SM1.LFGJA       AS	CURRENT_PERIOD_FISCAL_YEAR,
SM1.PRSPE       AS	FROZEN_STOCK,
SM1.PRVSP       AS	LAST_FROZEN_STOCK,
SM1.PREIN       AS	LIMIT_BATCH_TOTOAL_STOCK,
SM1.PRINS       AS	CHECK_STOCK,
SM1.PRVIN       AS	LAST_QUALITY_CHECK_STOCK,
SM1.PRLAB		AS  UNRESTRICTED_EVALUATED_STOCK,
SM1.PRDLL       AS	LAST_PASS_CHECK_DATE,
SM1.PRVEI       AS	LAST_LIMIT_USE_STOCK,
SM1.DEL_FLAG AS DEL_FLAG
FROM SDI.SDI_MSPR_1001 SM1			--SDI.SAP项目库存表
UNION ALL			--SDI_MSPR_1001.SAP项目库存表 UNION ALL 拼接 DWI_WM_PROJECT_STOCK_T.SAP项目库存表
SELECT 
 B.GROUP_NUM
,B.MATERIAL_NUM
,B.FACTORY_CODE
,B.STOCK_PLACE
,B.BATCH_NUM
,B.SPECIAL_STOCK_FLAG
,B.WBS_ELEMENT
,B.LAST_MXXX_PERIOD
,B.CURRENT_STOCK_FIN_YEAR
,B.CREATE_DATE
,B.CURRENT_PERIOD
,B.CURRENT_PERIOD_FISCAL_YEAR
,B.FROZEN_STOCK
,B.LAST_FROZEN_STOCK
,B.LIMIT_BATCH_TOTOAL_STOCK
,B.CHECK_STOCK
,B.LAST_QUALITY_CHECK_STOCK
,B.UNRESTRICTED_EVALUATED_STOCK
,B.LAST_PASS_CHECK_DATE
,B.LAST_LIMIT_USE_STOCK
,B.DEL_FLAG
FROM Y_FALG_DATA B)
INSERT OVERWRITE TABLE DWIOTD.DWI_WM_PROJECT_STOCK_T		--DWI.SAP项目库存表
SELECT
SM1.GROUP_NUM		AS	GROUP_NUM,
CASE WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''	--关联物料主数据取物料ID,判断SDI源表字段物料编号为空或者空字符串时赋值-999999
THEN -999999
ELSE NVL(DMMG.MATERIAL_ID, -999998)			--关联不上主数据时赋值-999998
END			AS  MATERIAL_ID, 		--物料ID
CASE
WHEN SM1.MATERIAL_NUM IS NULL OR SM1.MATERIAL_NUM = ''	--关联物料主数据,判断SDI源表字段物料编号为空或者空字符串时赋值'SNULL'
THEN 'SNULL'
ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')		--关联不上主数据时赋值'NOT EXISTS'
END         	AS	MATERIAL_NUM,		--物料编号
CASE WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''		--关联工厂主数据取工厂ID,判断SDI源表字段工厂编码为空或者空字符串时赋值-999999
THEN -999999
ELSE NVL(DMF.FACTORY_ID, -999998)	----关联不上主数据时赋值-999998
END 		AS  FACTORY_ID, 	--工厂ID
CASE	
WHEN SM1.FACTORY_CODE IS NULL OR SM1.FACTORY_CODE = ''	--关联工厂主数据,判断SDI源表字段工厂编码为空或者空字符串时赋值'SNULL'
THEN 'SNULL'
ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')	----关联不上主数据时赋值'NOT EXISTS'
END 			AS	FACTORY_CODE,		--工厂编码
SM1.STOCK_PLACE       AS	STOCK_PLACE,
SM1.BATCH_NUM       AS	BATCH_NUM,
T.CONFIG_CODE       AS	SPECIAL_STOCK_FLAG,
SM1.WBS_ELEMENT       AS	WBS_ELEMENT,
SM1.LAST_MXXX_PERIOD       AS	LAST_MXXX_PERIOD,
SM1.CURRENT_STOCK_FIN_YEAR       AS	CURRENT_STOCK_FIN_YEAR,
SM1.CREATE_DATE       AS	CREATE_DATE,
SM1.CURRENT_PERIOD       AS	CURRENT_PERIOD,
SM1.CURRENT_PERIOD_FISCAL_YEAR       AS	CURRENT_PERIOD_FISCAL_YEAR,
SM1.FROZEN_STOCK       AS	FROZEN_STOCK,
SM1.LAST_FROZEN_STOCK       AS	LAST_FROZEN_STOCK,
SM1.LIMIT_BATCH_TOTOAL_STOCK       AS	LIMIT_BATCH_TOTOAL_STOCK,
SM1.CHECK_STOCK       AS	CHECK_STOCK,
SM1.LAST_QUALITY_CHECK_STOCK       AS	LAST_QUALITY_CHECK_STOCK,
SM1.UNRESTRICTED_EVALUATED_STOCK		AS  UNRESTRICTED_EVALUATED_STOCK,
SM1.LAST_PASS_CHECK_DATE       AS	LAST_PASS_CHECK_DATE,
SM1.LAST_LIMIT_USE_STOCK       AS	LAST_LIMIT_USE_STOCK,
SM1.DEL_FLAG 	    AS  DEL_FLAG,
CURRENT_TIMESTAMP() 	AS      DW_LAST_UPDATE_DATE
FROM SDI_AND_Y_FLAG_DATA SM1		--SDI.SAP项目库存表
LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG		--关联物料主数据取物料ID
	 ON  SM1.GROUP_NUM = DMMG.GROUP_NUM				--集团
	 AND SM1.MATERIAL_NUM = DMMG.MATERIAL_NUM		--物料编号
	 AND DMMG.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
	 AND DMMG.DEL_FLAG = 'N'						--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF				--关联工厂主数据取工厂ID
	 ON  SM1.GROUP_NUM = DMF.GROUP_NUM				--集团
	 AND SM1.FACTORY_CODE = DMF.FACTORY_CODE		--工厂编码
	 AND DMF.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
	 AND DMF.DEL_FLAG = 'N'							--删除标识为'N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T		--关联通用配置表
	 ON T.SRC_SYS_CONFIG_CODE = SM1.SPECIAL_STOCK_FLAG	--取特殊库存描述中文名称
	 AND T.GROUP_NUM = SM1.GROUP_NUM				--集团
	 AND T.SOURCE_CODE = 'T148T'
