	   T1.doc_type , -- 件数类别
	   T1.locations , --仓位

	   T1.customer_code, -- 客户编码

	JOIN dwimd.dwi_md_data_material_general_t T2
	ON CASE WHEN SUBSTRING(T1.material_num,-2) NOT IN ('_A') THEN CONCAT(T1.material_num,'_A') ELSE T1.material_num END = T2.material_num
	join sdi.sdi_ct_customer_data_1037 T3
	ON T1.dealer_code = T3.customer_code
	WHERE T3.data_source_falg = '万维'
