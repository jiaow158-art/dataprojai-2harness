set hive.cbo.enable=true;				--性能优化SQL
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIOTD.DWI_WM_ALL_TYPE_STOCK_T		--DWI-SAP所有库存类型总库存表
SELECT
		SZT1.CLIENT				AS	GROUP_NUM,		--集团
		CASE WHEN SZT1.MATERIAL IS NULL OR SZT1.MATERIAL = ''	--关联物料主数据取MATERIAL_ID,判断MATERIAL（SDI.物料）字段为空或者空字符串的时候赋值-999999
		THEN -999999											
        ELSE NVL(DMMG.MATERIAL_ID, -999998)						--关联不上主数据赋值-999998
        END 				AS  MATERIAL_ID, 		--物料ID
		CASE WHEN SZT1.MATERIAL IS NULL OR SZT1.MATERIAL = ''	--判断SDI.物料字段为空或者空字符串的时候赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMMG.MATERIAL_NUM, 'NOT EXISTS')				--关联不上主数据赋值'NOT EXISTS'
		END 				AS	MATERIAL_NUM,		--物料编号
		CASE WHEN SZT1.PLANT IS NULL OR SZT1.PLANT = ''			--关联工厂主数据取FACTORY_ID,判断PLANT(SDI.工厂源字段)为空或者空字符串的时候赋值-999999
		THEN -999999
        ELSE NVL(DMF.FACTORY_ID, -999998)						--关联不上主数据赋值-999998
        END 				AS  FACTORY_ID, 		--工厂ID
		CASE WHEN SZT1.PLANT IS NULL OR SZT1.PLANT = ''			--判断SDI.工厂源字段为空或者空字符串的时候赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMF.FACTORY_CODE, 'NOT EXISTS')				--关联不上主数据赋值'NOT EXISTS'
		END       			AS	FACTORY_CODE,		--工厂编码
		SZT1.STOR_LOC            AS	STOCK_PLACE,	--库存地点
		SZT1.BATCH               AS	BATCH_NUM,		--批号
		CASE WHEN SZT1.COMP_CODE IS NULL OR SZT1.COMP_CODE = ''	--关联公司主数据取COMPANY_ID,判断COMP_CODE（SDI.公司源字段）字段为空或者空字符串的时候赋值-999999
		THEN -999999
        ELSE NVL(DMC.COMPANY_ID, -999998)						--关联不上主数据赋值-999998
        END 				AS  COMPANY_ID, 		--公司ID
		CASE WHEN SZT1.COMP_CODE IS NULL OR SZT1.COMP_CODE = ''	--判断SDI.公司源字段为空或者空字符串的时候赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMC.COMPANY_CODE, 'NOT EXISTS')				--关联不上主数据赋值'NOT EXISTS'
		END 				AS	COMPANY_CODE,		--公司编号
		T.CONFIG_CODE       AS	SPECIAL_STOCK_FLAG,	--特殊库存标识
		SZT1.SO_NO               AS	RETAIL_VOUCHER_NO,	--销售和分销凭证号
		SZT1.SO_ITEM             AS	RETAIL_VOUCHER_PROJECT_NO,	--销售和分销凭证的项目号
		CASE WHEN SZT1.VENDOR IS NULL OR SZT1.VENDOR = ''		--关联供应商主数据取SUPPLIER_OR_CREDITOR_ID,判断VENDOR(SDI.供应商源字段）字段为空或者空字符串的时候赋值-999999
		THEN -999999
        ELSE NVL(DMSG.SUPPLIER_OR_CREDITOR_ID, -999998)			--关联不上主数据赋值-999998
        END 				AS  SUPPLIER_OR_CREDITOR_ID, 		--供应商帐户ID
		CASE WHEN SZT1.VENDOR IS NULL OR SZT1.VENDOR = ''		--判断SDI.供应商源字段为空或者空字符串的时候赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(DMSG.SUPPLIER_OR_CREDITOR_NUM, 'NOT EXISTS')	--关联不上主数据赋值'NOT EXISTS'
		END 				AS	SUPPLIER_OR_CREDITOR_NUM,		--供应商编码
		CASE WHEN SZT1.CUSTOMER IS NULL OR SZT1.CUSTOMER = ''	--关联客户主数据取CUST_ID,判断CUSTOMER(SDI.客户源字段)字段为空或者空字符串的时候赋值-999999
		THEN -999999
        ELSE NVL(CUST.CUST_ID, -999998)							--关联不上主数据赋值-999998
        END					AS  CUST_ID, 			--客户ID
		CASE WHEN SZT1.CUSTOMER IS NULL OR SZT1.CUSTOMER = ''	--判断SDI.客户源字段为空或者空字符串的时候赋值'SNULL'
		THEN 'SNULL'
	     ELSE NVL(CUST.CUST_NUM, 'NOT EXISTS')					--关联不上主数据赋值'NOT EXISTS'
		END 				AS	CUST_NUM,			--客户编号
		SZT1.WBS                 AS	WBS_ELEMENT,	--WBS要素
		SZT1.STOCK_TYPE          AS	STOCK_TYPE,		--库存类型
		SZT1.STOCK_QTY           AS	STOCK_QTY,		--库存数量
		'N'						AS	DEL_FLAG,		--删除标识
		CURRENT_TIMESTAMP() 	AS	DW_LAST_UPDATE_DATE		--DW最后更新时间
		FROM SDI.SDI_ZCDSV_TOTALSTCK_1001 SZT1		--SDI.SAP所有库存类型总库存表
		LEFT JOIN DWIMD.DWI_MD_MATERIAL_GENERAL_T DMMG		--关联物料主数据,取MATERIAL_ID
			 ON  SZT1.CLIENT = DMMG.GROUP_NUM				--集团
			 AND SZT1.MATERIAL = DMMG.MATERIAL_NUM			--物料编号
			 AND DMMG.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
			 AND DMMG.DEL_FLAG = 'N'						--删除标识='N'
		LEFT JOIN DWIMD.DWI_MD_SUPPLIER_GENERAL_T DMSG		--关联供应商主数据,取SUPPLIER_OR_CREDITOR_ID
			 ON  SZT1.CLIENT = DMSG.GROUP_NUM				--集团
			 AND SZT1.VENDOR = DMSG.SUPPLIER_OR_CREDITOR_NUM		----供应商编码
			 AND DMSG.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
			 AND DMSG.DEL_FLAG = 'N'						--删除标识='N'
		LEFT JOIN DWIMD.DWI_MD_CUST_GENERAL_T CUST			--关联客户主数据,取CUST_ID
			 ON  SZT1.CLIENT = CUST.GROUP_NUM				--集团
			 AND SZT1.CUSTOMER = CUST.CUST_NUM				--客户编号
			 AND CUST.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
			 AND CUST.DEL_FLAG = 'N'						--删除标识='N'
		LEFT JOIN DWIMD.DWI_MD_FACTORY_T DMF				--关联工厂主数据,取FACTORY_ID
			 ON  SZT1.CLIENT = DMF.GROUP_NUM				--集团
			 AND SZT1.PLANT = DMF.FACTORY_CODE				--工厂编码
			 AND DMF.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
			 AND DMF.DEL_FLAG = 'N'							--删除标识='N'
		LEFT JOIN DWIMD.DWI_MD_COMPANY_T DMC				--关联公司主数据,取COMPANY_ID
			 ON  SZT1.CLIENT = DMC.GROUP_NUM				--集团
			 AND SZT1.COMP_CODE = DMC.COMPANY_CODE			--公司编码
			 AND DMC.END_DATE = '4712-12-31'				--结束时间为'4712-12-31'
			 AND DMC.DEL_FLAG = 'N'							--删除标识='N'
		LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R T		--关联通用配置表
			 ON T.SRC_SYS_CONFIG_CODE = SZT1.SPEC_TYPE		--特殊库存标识
			 AND T.GROUP_NUM = SZT1.CLIENT					--集团
			 AND T.SOURCE_CODE = 'T148T'					--取'特殊库存描述'中文名称
