INSERT OVERWRITE TABLE DWIOTD.DWI_WM_B1_OBTQ_T
SELECT
  'SW' AS CUSTOMER_TYPE, -- - 客户编码
  CUSTCODE AS cust_code,  -- 客户编码
  itemcode,   -- 物料代码
  sysnumber,  -- 系统编号
  whscode,  -- 仓库编号
  quantity,  -- 数量
  commitqty,  -- 已占库数量
  countqty,  -- 已盘点数量
  absentry,  -- abs entry
  mdabsentry,  -- MD Abs Entry
  trackingnt,  -- 货物报关跟踪注释
  trackintln,  -- 货物报关跟踪注释行
  ccdquant,  -- 货物报关质量
  'N' AS del_flag,  -- 删除标识
  CURRENT_TIMESTAMP() AS dw_last_update_date -- 最后更新时间
FROM SDI.SDI_SW_OBTQ_1010;
