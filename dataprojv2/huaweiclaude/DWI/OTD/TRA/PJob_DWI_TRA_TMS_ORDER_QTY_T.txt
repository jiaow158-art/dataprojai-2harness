INSERT OVERWRITE TABLE DWIOTD.DWI_TRA_TMS_ORDER_QTY_T
SELECT
  A.ORDER_COUNT, 
  A.YEAR_MONTH, 
  A.YEAR, 
  A.MONTH, 
  A.DEL_FLAG, 
  CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
  B.ORDER_CNT
FROM SDI.SDI_V_SJZT_MONTHLY_ORDER_1005 A
LEFT JOIN (SELECT SUBSTR(B.ENTRYDT,1,7) AS STAT_MONTH
                 ,COUNT(DISTINCT A.EXTERNAL_ORDER_CODE) AS ORDER_CNT
             FROM SDI.SDI_TMS_ORDERDETAIL_1005 A
			INNER JOIN SDI.SDI_TMS_ORDER_1005 B
			   ON A.ORDER_CODE = B.CODE
			  AND A.DEL_FLAG = 'N'
			  AND B.DEL_FLAG = 'N'
			  AND COALESCE(A.EXTERNAL_ORDER_CODE,'') <> ''
			GROUP BY SUBSTR(B.ENTRYDT,1,7)
          ) B
  ON A.YEAR_MONTH = B.STAT_MONTH
WHERE A.DEL_FLAG = 'N'
