with BOM_DETAIL_DATA AS(   -- BOM数据
	select factory_code, 
		   material_num, 
		   small_material_num,
		   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
	from(
		select t1.werks as factory_code, -- 工厂编码
			   t1.MATNR as material_num,  -- 物料编码
			   t2.IDNRK as small_material_num, -- 子项物料编码
			   t3.material_name 
		from(
			select MATNR,
				   STLNR,
				   werks,
				   row_number() over(partition by werks,MATNR order by ANDAT desc ) as RN
			from SDI.SDI_MAST_1001 
			WHERE STLAN = '1' -- 生产BOM
			AND werks LIKE '3%'
		) t1
		join sdi.sdi_STPO_1001 t2 on t1.STLNR = t2.STLNR
	    join DWIMD.dwi_md_data_material_general_t t3
	    on t2.IDNRK = t3.material_num
	    and t3.matl_type_name = '半成品（东鹏）'
		where t1.RN = 1
	  )t
	group by factory_code, 
			 material_num, 
			 small_material_num
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T
select *
from BOM_DETAIL_DATA t;
