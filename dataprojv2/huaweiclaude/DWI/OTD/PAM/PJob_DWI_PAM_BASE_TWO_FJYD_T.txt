--CREATE TABLE DWIOTD.DWI_PAM_BASE_TWO_FJYD_T AS -- 二窑分级优等明细数据(江门&江西)
with jm_sy_jj_data as( -- 江门升降级数据
	select t1.prod_date, 
		   t1.factory_code, 
		   --t1.material_num,
		   t2.material_num,
		   0 as ysyd_qty, -- 一烧优等
		   0 as csyd_qty,   -- 重烧优等
		   sum(t1.sy_jj_qty) as sy_jj_qty  -- 升级降级
	from(
		-- 打磨分级
		select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date, -- 分级时间
			   a.jm_factory_code as factory_code,
		       a.jm_gc_code as material_num,   -- 光瓷型号编码
		       sum(case when a.jm_fj_grade=1    then 1 else 0 end ) sy_jj_qty -- 优等品  -- JM_YD_NUM
		  from SDI.SDI_jm_t_fj_dm_1038 a -- 打磨分级
		  where a.jm_is_del=0   
		  AND substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		  group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
		   	   a.jm_factory_code,
			   a.jm_gc_code
		union all
		-- 喷补分级	      
		select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date,  -- 分级时间
		       a.jm_factory_code ,
			   a.jm_gc_code,   -- 光瓷型号编码
			   sum(case when a.jm_fj_grade=1    then 1 else 0 end ) sy_jj_qty -- 优等品  -- JM_YD_NUM
		from sdi.sdi_jm_t_fj_pb_1038 a -- 喷补分级
		where a.jm_is_del=0    
		AND substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
		       a.jm_factory_code ,
			   a.jm_gc_code
		union all
		-- 漏气分级		  
		select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date, -- 分级时间
		       a.jm_factory_code as factory_code, 
			   a.jm_gc_code as material_num,    -- 光瓷型号编码
		       sum(case when a.jm_fj_grade=1 then 1 else 0 end ) sy_jj_qty  -- 优等品  -- JM_YD_NUM
		from sdi.sdi_jm_t_fj_lq_1038 a    -- 漏气分级
		where a.jm_is_del=0   
		and substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
				 a.jm_factory_code,
				 a.jm_gc_code
		union all
		 -- 升降级日报表 升级数据		 
		select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date,  -- 分级时间
		       a.jm_factory_code as factory_code,
		       a.jm_gc_code as material_num,    -- 光瓷型号编码
			   SUM(CASE WHEN A.JM_UUID IS NOT NULL THEN 1 ELSE 0 END ) sy_jj_qty  -- JM_UUID 唯一标识  数量
		   from SDI.SDI_jm_t_fj_sy_1038 a   -- 升优分级
		  WHERE A.JM_IS_DEL=0  
		  and substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		  group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
		  		   a.jm_factory_code,
		  		   a.jm_gc_code
		union all  		  
		-- 升降级日报表 降级分级  		   
		select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS PROD_DATE,  -- 分级时间
			   a.jm_factory_code as factory_code, --工厂编码
			   a.jm_gc_code as material_num,     -- 光瓷型号编码
		       - SUM(CASE WHEN A.JM_UUID IS NOT NULL THEN 1 ELSE 0 END ) sy_jj_qty  -- 数量
		from SDI.SDI_jm_t_fj_jj_1038 a   -- 降级分级
		WHERE A.JM_IS_DEL=0  
		AND substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
		group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
				 a.jm_factory_code,
				 a.jm_gc_code
		)t1
		left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t2 -- 生产BOM关系数据
		on t1.factory_code = t2.factory_code
		and t1.material_num = t2.small_material_num
		group by t1.prod_date, 
				 t1.factory_code, 
				 --t1.material_num
				 t2.material_num
),
jm_two_yds_data as(  -- 江门二窑分级优等数
select t1.prod_date,   -- 生产日期
       t1.factory_code,  -- 工厂编码
       t1.material_num, -- 物料编码
       sum(t1.ysfj_qty) as ysfj_qty, -- 一烧分级数
       sum(t1.ysyd_qty) as  ysyd_qty, -- 一烧优等数
       sum(t1.csyd_qty) as csyd_qty,   -- 重烧优等数
       sum(t1.sy_jj_qty) as sy_jj_qty  -- 升优降级数
from(
    select t1.prod_date,   -- 生产日期
           t1.factory_code,  -- 工厂编码
           --t1.material_num, -- 物料编码
           t2.material_num, -- 物料编码
           sum(t1.ysfj_qty) as ysfj_qty, -- 一烧分级数
           sum(t1.ysyd_qty) as  ysyd_qty, -- 一烧优等数
           sum(t1.csyd_qty) as csyd_qty,   -- 重烧优等数
           sum(t1.sy_jj_qty) as sy_jj_qty  -- 升优降级数
    from(
    	-- 一烧优等数
    	select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date,  -- 分级时间
    	       a.jm_factory_code as factory_code, -- 工厂编码
    		   a.jm_gc_code as material_num,  -- 光瓷型号编码
    		   --a.jm_np_code,  -- 泥坯编码
    		   sum(case when a.jm_uuid is not null then 1 else 0 end ) ysfj_qty, -- 分级数  -- JM_FJ_NUM
    		   sum(case when a.jm_fj_grade_frist=1  then 1 else 0 end ) ysyd_qty, -- 一烧优等数  -- JM_YD_NUM  jm_fj_grade_frist 一烧分级等级
    		   0 as csyd_qty,   -- 重烧优等
    		   0 as sy_jj_qty   -- 升降级
    	   from SDI.SDI_jm_t_fj_first_1038 a  -- 一烧分级
    	   where a.jm_is_del=0 
    	   and a.JM_IS_BL_COMPLEMENT=0 
    	   and a.JM_IS_BM=0   
    	   and substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传当天
    	   group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
    	   			a.jm_factory_code,
    	            a.jm_gc_code
    	   union all
           --重烧优等
    	   select substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7) AS prod_date, -- 返烧分级时间 jm_fs_fj_date
    		       a.jm_factory_code as factory_code, -- 工厂编码
    			   a.jm_gc_code as material_num,  -- 光瓷型号编码
    			   0 as ysfj_qty,
    			   0 as ysyd_qty, -- 一烧优等数
    			   sum(case when a.jm_fj_grade=1 then 1 else 0 end ) csyd_qty,     -- 优等品  -- JM_YD_NUM
    			   0 as sy_jj_qty   -- 升降级
    		from SDI.SDI_jm_t_fj_fs_1038 a    -- 返烧分级
    		left join sdi.sdi_jm_t_wy_base_gc_1038 b  -- 光瓷信息
    		on a.jm_gc_id=b.jm_uuid
    		where a.jm_is_del=0   
    		and substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,10) between '2022-06-01' and '${PERIOD_ID_D}'  -- 传昨天
    		group by substring(from_unixtime(unix_timestamp(a.jm_fj_date) - 60*60*7.5,'yyyy-MM-dd HH:mm:ss'),1,7),
    				 a.jm_factory_code,
    		  		 a.jm_gc_code
    	)t1
    	left join DWIOTD.DWI_PAM_BOM_DETAIL_DATA_T t2 -- 生产BOM关系数据
    	on t1.factory_code = t2.factory_code
    	and t1.material_num = t2.small_material_num
    	group by t1.prod_date, 
             	 t1.factory_code, 
             	 --t1.material_num
             	 t2.material_num
    	union all
    	select t1.prod_date, 
    		   t1.factory_code, 
    		   t1.material_num,
    		   0 as ysfj_qty,
    		   0 as ysyd_qty, -- 一烧优等
    		   0 as csyd_qty,   -- 重烧优等
    		   t1.sy_jj_qty  -- 升级降级
    	from jm_sy_jj_data t1
	)t1
	group by t1.prod_date, 
         	 t1.factory_code, 
         	 t1.material_num
),
jx_two_yds_data as(  -- 江西二窑分级优等数
	select t.prod_date,   -- 月份
	       T.factory_code,  -- 工厂编码
		   T.material_num,  -- 物理编码
		   SUM(T.ysfj_qty) AS ysfj_qty, -- 一烧分级数
		   SUM(T.ysyd_qty) AS ysyd_qty, -- 一烧优等数
		   SUM(T.csyd_qty) AS csyd_qty,   -- 重烧优等数
		   SUM(T.sy_jj_qty) as sy_jj_qty  -- 升优降级数
	FROM(
		SELECT T.grade_date as prod_date,   -- 月份
			   T.factory_code,  -- 工厂编码
			   T.material_num,  -- 物理编码
			   T.ysfj_qty AS ysfj_qty, -- 一烧分级数
			   T.ysyd_qty AS ysyd_qty, -- 一烧优等数
			   0 AS csyd_qty,   -- 重烧优等数
			   0 as sy_jj_qty  -- 升优降级数
		FROM DWIOTD.DWI_PAM_BASE_YSFJYD_JX_T T   -- 一烧分级数据(江西)
		union all
		SELECT T.grand_date as prod_date,   -- 月份
			   T.factory_code,  -- 工厂编码
			   T.material_num,  -- 物理编码
			   0 as ysfj_qty, -- 一烧分级数
			   0 as ysyd_qty, -- 一烧优等数
			   t.csyd_qty as csyd_qty,   -- 重烧优等数
			   0 as sy_jj_qty  -- 升优降级数
		FROM DWIOTD.DWI_PAM_BASE_CSFJ_JX_T T   -- 重烧分级数据(江西)
	   )T
	   GROUP BY t.prod_date,   -- 月份
		        T.factory_code,  -- 工厂编码
			    T.material_num  -- 物理编码
)
INSERT OVERWRITE TABLE DWIOTD.DWI_PAM_BASE_TWO_FJYD_T
select T.prod_date,   -- 月份
       T.factory_code,  -- 工厂编码
	   T.material_num,  -- 物理编码
	   SUM(T.ysfj_qty) AS ysfj_qty, -- 一烧分级数
	   SUM(T.ysyd_qty) AS ysyd_qty, -- 一烧优等数
	   SUM(T.csyd_qty) AS csyd_qty,   -- 重烧优等数
	   SUM(T.sy_jj_qty) as sy_jj_qty,  -- 升优降级数
	   CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE -- 数据同步时间
FROM(
	select t1.* from jm_two_yds_data t1
	union all
	select t1.* from jx_two_yds_data t1
)T
GROUP BY T.prod_date,   -- 月份
         T.factory_code,  -- 工厂编码
	     T.material_num  -- 物理编码
