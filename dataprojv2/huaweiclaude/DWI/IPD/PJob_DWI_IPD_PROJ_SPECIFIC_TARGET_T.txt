insert overwrite table  DWIIPD.DWI_IPD_PROJ_SPECIFIC_TARGET_T
SELECT 
TAB.NAME            AS TABLE_NAME,
PRJ.CODE            AS PROJ_NUM,--项目编号
PRJ.NAME            AS PROJ_NAME,--项目名称
COL.NAME            AS ITEM_NAME,--指标名称
RO.SN               AS ITEM_COLUMN_NUM,---指标字段序号
VAL.VAL             AS ITEM_VALUE,--指标值
'N'                 AS DEL_FLAG,
current_timestamp() as DW_LAST_UPDATE_DATE
FROM SDI.sdi_t_pim_pim_1003 prj,--项目基础信息
SDI.SDI_EXT_OBJ_TABLEVALUE_1003 VAL,--指标值
SDI.SDI_EXT_OBJ_TABLECOL_1003 COL,--指标字段
SDI.SDI_EXT_OBJ_TABLE_1003 TAB,--指标表
SDI.SDI_EXT_OBJ_TABLEROW_1003 RO--指标项目关系
WHERE 1=1
--AND prj.NAME ='零售结晶大理石项目'
AND PRJ.ID = RO.OBJECTID  
AND VAL.TABLECOLID  = COL.ID
AND VAL.TABLEROWID =RO.ID 
AND COL.TABLEID  = TAB.ID 
AND TAB.NAME ='项目具体目标';
