INSERT OVERWRITE TABLE DWIPI.DWI_WF_ZSD_ZMKB_T
SELECT T.MANDT,
       T.KUNNR,
       T.ZZKUNNR,
       T.NAME1,
       T.NAME2,
       T.VKORG,
       T.VTWEG,
       T.SPART,
       T.KDGRP,
       T.BZIRK,
       'N' AS DEL_FLAG,
       CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE,
       T.freeze,
       A.BZTXT AS BZIRK_NAME,
       T.vkgrp
  FROM SDI.SDI_ZSD_ZMKB_1001 T
  LEFT JOIN SDI.sdi_t171t_1001 A ON A.BZIRK = T.BZIRK 
        AND A.del_flag = 'N'    AND A.spras = '1';
