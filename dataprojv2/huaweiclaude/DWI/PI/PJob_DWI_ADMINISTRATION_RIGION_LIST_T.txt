INSERT OVERWRITE TABLE DWIPI.DWI_ADMINISTRATION_RIGION_LIST_T
  SELECT A.HTBH, /*合同编号*/
         A.OALSH, /*流水号*/
         A.KHBM, /*客户编码*/
         A.SSSYB, /*所属事业部,0是瓷砖事业部*/
         A.KHSX, /*客户属性*/
         A.KHDJ, /*客户等级*/
         C.SELECTNAME, /*客户等级名称*/
         A.HTZT, /*合同状态*/
         B.SHENG, /*省*/
         B.SHI, /*市*/
         B.QX, /*区县*/
         'N' AS DEL_FLAG,
         CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
    FROM DWIPI.DWI_WF_UF_HTGL_T A /*获取合同台账基本记录*/
   INNER JOIN DWIPI.DWI_WF_UF_HTGL_DT1_T B
      ON A.ID = B.MAINID /*获取合同台账行政区域记录表*/
    LEFT JOIN DWIPI.DWI_WF_WORKFLOW_SELECTITEM_T C
      ON A.KHDJ = C.SELECTVALUE
     AND C.FIELDID = 244017 /*获取合同台账行政区域记录表*/
