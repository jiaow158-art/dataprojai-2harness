--SQL性能优化设置
set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;

INSERT OVERWRITE TABLE DWICS.DWI_WT_USER_BEH_ANALY_T
SELECT CAST(ID AS NUMERIC) AS ID
      ,CAST(DEALER_ID AS NUMERIC) AS DEALER_ID --经销商ID
      ,CAST(STORE_ID AS NUMERIC) AS STORE_ID --店铺ID
      ,CAST(TYPE AS NUMERIC) AS BEH_TYPE --类型 0:浏览 1:收藏
      ,CAST(USER_ID AS NUMERIC) AS USER_ID --用户ID
      ,CAST(USER_SOURCE AS NUMERIC) AS USER_SOURCE --客户来源
      ,CAST(ITEM_NO AS STRING) AS ITEM_NO --商品唯一码
      ,CREATE_TIME --创建时间
      ,UPDATE_TIME --更新时间
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_ITEM_COLLECTION_1030 T1
WHERE T1.IS_DELETED = 0
AND T1.EP_ID = 3044
AND T1.APP_ID = 1831
AND T1.DEL_FLAG = 'N'
;
