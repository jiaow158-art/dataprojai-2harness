insert overwrite table dwifin.DWI_CRE_INTEGRAL_CHANGE_T partition(period_id_m ='${start_date}')  
SELECT 
T1.MANDT         as GROUP_NUM,
CG.CUST_ID       as CUST_ID,
T1.KUNNR         as CUST_NUM,
T1.ZJFGGBH       as INTEGRAL_CHANGE_NUM,
T1.XREF1         as BUSINESS_PARTNER_REF_CODE1,
T1.XREF2         as BUSINESS_PARTNER_REF_CODE2,
T1.KWMENG        as SALE_ACCUMULATE_QTY,
T1.ZJFBGE_I      as INPUT_CHANGE_AMT,
T1.KURSK         as DECISION_PRICE_RATE,
from_unixtime(unix_timestamp(T1.BUDAT,'yyyyMMdd'),'yyyy-MM-dd') as VOUCHER_POST_DATE,
T1.NETWR         as VC_NET_WORTH_AMT,
T1.ERNAM         as CREATE_BY,
T1.VBELN_VF      as BILL_VOUCHER_NUM,
T1.POSNR_VF      as BILL_VOUCHER_ITEM_NUM,
T1.ZSGDRZ        as MANUAL_INPUT_NUM,
(case 
when T1.ZQZRQ < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.ZQZRQ > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.ZQZRQ,'yyyyMMdd'),'yyyy-MM-dd') end) as SQUARE_UP_DATE,
T1.ZQZBS         as SQUARE_UP_FLAG,
T1.ZQZBH         as SQUARE_UP_NUM,
MG.MATERIAL_ID   as MATERIAL_ID,
T1.MATNR         as MATERIAL_NUM,
T1.ZJFBXED       as INTEGRAL_REIMB_AMT,
T1.ZJFLX         as INTEGRAL_TYPE,
from_unixtime(unix_timestamp(T1.ERDAT,'yyyyMMdd'),'yyyy-MM-dd') as CREATE_DATE,
T1.ERZET         as INPUT_TIME,
T1.VRKME         as SALE_UNIT,
T1.VBELN         as SALE_VOUCHER_NUM,
T1.POSNR         as ACC_VOUCHER_LINE_ITEM_NUM,
T1.WAERK         as VOUCHER_CURRENCY_CODE,
T1.WBSID         as WBS_ELEMENT,
T1.ZCXBS         as OFFSET_FLAG,
T1.ZJFFP         as INTEGRAL_DISTRIBUTE_DESC,
T1.ZJFNF         as INTEGRAL_YEAR,
T1.ZJFBZ         as TEXT,
T1.ZJFYF         as INTEGRAL_MONTH,
T1.ZJFYEN        as NEW_INTEGRAL_BAL,
T1.ZJFYEO        as OLD_INTEGRAL_BAL,
T1.ZJFBGLX       as INTEGRAL_CHANGE_TYPE,
T1.ZJFBGE        as INTEGRAL_CHANGE_AMT,
T1.ZJFFW         as INTEGRAL_SCPOE,
T1.ZJFGGIT       as LINE_ITEM,
T1.DEL_FLAG,                 
T1.DW_LAST_UPDATE_DATE,
T1.vtweg         as dis_channel_code
FROM SDI.SDI_ZJFTAB2_1001 T1 
LEFT JOIN  DWIMD.DWI_MD_CUST_GENERAL_T CG 
ON  T1.KUNNR  = CG.cust_num  
AND T1.MANDT = CG.GROUP_NUM
AND CG.del_flag  = 'N'
AND CG.end_date ='4712-12-31'
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MG 
ON  T1.MATNR = MG.MATERIAL_NUM
AND T1.MANDT = MG.GROUP_NUM
AND MG.del_flag  = 'N'
AND MG.end_date ='4712-12-31'
where T1.period_id_m ='${start_date}';
