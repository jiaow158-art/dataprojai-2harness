--成本要素 (数据与成本控制范围相关)

--建临时表
WITH DWI_COST_ELEMENTS_CTRL_SCOPE_T_TMP AS(
--直接插入SDI的数据，删除标识置为N
SELECT 
   0 AS SEQ_ID
  ,T1.DATBI AS END_EFFECT_DATE
  ,T1.MANDT AS GROUP_NUM
  ,T1.KOKRS AS CONTROL_SCOPE
  ,from_unixtime(unix_timestamp(T1.DATAB,'yyyymmdd'),'yyyy-mm-dd') AS START_EFFECT_DATE
  ,T1.KOSTL AS COST_CENTER
  ,from_unixtime(unix_timestamp(T1.DATBI,'yyyymmdd'),'yyyy-mm-dd') AS END_EFFECT_DATE2
  ,T1.KSTAR AS COST_ELEMENTS
  ,T1.KATYP AS COST_ELEMENTS_TYPE
  ,from_unixtime(unix_timestamp(T1.ERSDA,'yyyymmdd'),'yyyy-mm-dd') AS CREATE_DATE
  ,T1.USNAM AS INPUT_BY
  ,CASE WHEN T1.LAST_CHANGED_TS = '0' THEN '1900-01-01 00:00:00'
        ELSE CONCAT(SUBSTR(T1.LAST_CHANGED_TS,1,4),'-',SUBSTR(T1.LAST_CHANGED_TS,5,2),'-',SUBSTR(T1.LAST_CHANGED_TS,7,2),' '
		            ,SUBSTR(T1.LAST_CHANGED_TS,9,2),':',SUBSTR(T1.LAST_CHANGED_TS,11,2),':',SUBSTR(T1.LAST_CHANGED_TS,13,2))
		 END AS LAST_UPDATE_TIME
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_CSKB_1001 T1
)

INSERT OVERWRITE TABLE DWIFIN.DWI_COST_ELEMENTS_CTRL_SCOPE_T
--保留DWI存在而SDI不存在的数据，删除标识置为Y
SELECT
   T1.SEQ_ID
  ,T1.END_EFFECT_DATE
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,T1.START_EFFECT_DATE
  ,T1.COST_CENTER
  ,T1.END_EFFECT_DATE2
  ,T1.COST_ELEMENTS
  ,T1.COST_ELEMENTS_TYPE
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,T1.LAST_UPDATE_TIME
  ,'Y' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_ELEMENTS_CTRL_SCOPE_T T1
LEFT JOIN DWI_COST_ELEMENTS_CTRL_SCOPE_T_TMP T2
ON T2.END_EFFECT_DATE = T1.END_EFFECT_DATE
AND T2.GROUP_NUM = T1.GROUP_NUM
AND T2.CONTROL_SCOPE = T1.CONTROL_SCOPE
AND T2.COST_ELEMENTS = T1.COST_ELEMENTS
WHERE T2.END_EFFECT_DATE IS NULL

UNION ALL

--保留DWI存在且SDI存在的数据，删除标识置为N
SELECT
   T1.SEQ_ID
  ,T1.END_EFFECT_DATE
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,T1.START_EFFECT_DATE
  ,T1.COST_CENTER
  ,T1.END_EFFECT_DATE2
  ,T1.COST_ELEMENTS
  ,T1.COST_ELEMENTS_TYPE
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,T1.LAST_UPDATE_TIME
  ,'N' AS DEL_FLAG
  ,T1.DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_COST_ELEMENTS_CTRL_SCOPE_T T1
LEFT JOIN DWI_COST_ELEMENTS_CTRL_SCOPE_T_TMP T2
ON T2.END_EFFECT_DATE = T1.END_EFFECT_DATE
AND T2.GROUP_NUM = T1.GROUP_NUM
AND T2.CONTROL_SCOPE = T1.CONTROL_SCOPE
AND T2.COST_ELEMENTS = T1.COST_ELEMENTS
WHERE T2.END_EFFECT_DATE IS NOT NULL

UNION ALL

--直接插入SDI存在而DWI不存在的数据，删除标识置为N，SDI的SEQ_ID加上DWI的最大SEQ_ID
SELECT 
   T1.SEQ_ID + NVL(T2.SEQ_ID,0) AS SEQ_ID
  ,T1.END_EFFECT_DATE
  ,T1.GROUP_NUM
  ,T1.CONTROL_SCOPE
  ,T1.START_EFFECT_DATE
  ,T1.COST_CENTER
  ,T1.END_EFFECT_DATE2
  ,T1.COST_ELEMENTS
  ,T1.COST_ELEMENTS_TYPE
  ,T1.CREATE_DATE
  ,T1.INPUT_BY
  ,T1.LAST_UPDATE_TIME
  ,'N' AS DEL_FLAG
  ,CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM(SELECT
       ROW_NUMBER() OVER() AS SEQ_ID
      ,T1.END_EFFECT_DATE
      ,T1.GROUP_NUM
      ,T1.CONTROL_SCOPE
      ,T1.START_EFFECT_DATE
      ,T1.COST_CENTER
      ,T1.END_EFFECT_DATE2
      ,T1.COST_ELEMENTS
      ,T1.COST_ELEMENTS_TYPE
      ,T1.CREATE_DATE
      ,T1.INPUT_BY
      ,T1.LAST_UPDATE_TIME  
     FROM DWI_COST_ELEMENTS_CTRL_SCOPE_T_TMP T1
     LEFT JOIN DWIFIN.DWI_COST_ELEMENTS_CTRL_SCOPE_T T2
     ON T2.END_EFFECT_DATE = T1.END_EFFECT_DATE
     AND T2.GROUP_NUM = T1.GROUP_NUM
     AND T2.CONTROL_SCOPE = T1.CONTROL_SCOPE
     AND T2.COST_ELEMENTS = T1.COST_ELEMENTS
     WHERE T2.END_EFFECT_DATE IS NULL) T1
LEFT JOIN (SELECT MAX(SEQ_ID) AS SEQ_ID FROM DWIFIN.DWI_COST_ELEMENTS_CTRL_SCOPE_T T1) AS T2
ON 1 = 1
;
