--建临时表
WITH DWI_COST_CENTER_MAIN_T_TMP
AS(
SELECT
  0 AS SEQ_ID,
  T1.KOSTL COST_CENTER,
  T1.DATBI SRC_END_EFFECT_DATE,
  T1.MANDT GROUP_NUM,
  T1.KOKRS CONTROL_SCOPE,
  T1.WAERS CURRENCY_CODE,
  from_unixtime(unix_timestamp(T1.DATAB,'yyyymmdd'),'yyyy-mm-dd') AS START_EFFECT_DATE,
  T1.VERAK LEADER,
  T1.BUKRS COMPANY_CODE,
  T1.MGEFL REC_CONS_QUA_FLAG,
  T1.SPRAS LANG_CODE,
  from_unixtime(unix_timestamp(CASE
                               WHEN T1.DATBI = '00000000' THEN '19000101'
							   WHEN T1.DATBI = '99991231' THEN '47121231'
                               ELSE T1.DATBI
                                END,'yyyymmdd'),'yyyy-mm-dd') END_EFFECT_DATE,
  T1.KOMPL COST_CENTER_MSTRREC_INT_FLAG,
  T1.GSBER BUSINESS_SCOPE,
  T1.KHINR STANDARD_HIERARCHY_AREA,
  T1.VMETH ALLOWED_ALLOCTION_METHOD_FLAG,
  T1.PRCTR PROFIT_CENTER,
  T1.FUNC_AREA FUNC_SCOPE,
  T1.OBJNR OBJECT_NO,
  T1.KOSAR COST_CENTER_TYPE,
  T1.BKZER REALINCOME_CHECKIN_FREEZE_FLAG,
  T1.BKZKS REAL_SUBCOST_FREEZE_FLAG,
  T1.PKZER PLAN_INCOME_FREEZE_FLAG,
  T1.PKZKP PLAN_ETARY_COST_FREEZE_FLAG,
  T1.PKZKS PLAN_SUB_COST_FREEZE_FLAG,
  T1.BKZOB OPEN_ITEMS_FREEZE_FLAG,
  T1.BKZKP REAL_ETARY_COST_FREEZE_FLAG,
  from_unixtime(unix_timestamp(T1.ERSDA,'yyyymmdd'),'yyyy-mm-dd') AS CREATE_DATE,
  T1.USNAM INPUT_BY,
  'N' DEL_FLAG,
  CURRENT_TIMESTAMP() DW_LAST_UPDATE_DATE
FROM SDI.SDI_CSKS_1001 T1
)

--成本中心主数据
INSERT OVERWRITE TABLE DWIFIN.DWI_COST_CENTER_MAIN_T
--保留DWI存在而SDI不存在的数据，删除标识置为Y 
SELECT
  T1.SEQ_ID,
  T1.COST_CENTER,
  T1.SRC_END_EFFECT_DATE,
  T1.GROUP_NUM,
  T1.CONTROL_SCOPE,
  T1.CURRENCY_CODE,
  T1.START_EFFECT_DATE,
  T1.LEADER,
  T1.COMPANY_CODE,
  T1.REC_CONS_QUA_FLAG,
  T1.LANG_CODE,
  T1.END_EFFECT_DATE,
  T1.COST_CENTER_MSTRREC_INT_FLAG,
  T1.BUSINESS_SCOPE,
  T1.STANDARD_HIERARCHY_AREA,
  T1.ALLOWED_ALLOCTION_METHOD_FLAG,
  T1.PROFIT_CENTER,
  T1.FUNC_SCOPE,
  T1.OBJECT_NO,
  T1.COST_CENTER_TYPE,
  T1.REALINCOME_CHECKIN_FREEZE_FLAG,
  T1.REAL_SUBCOST_FREEZE_FLAG,
  T1.PLAN_INCOME_FREEZE_FLAG,
  T1.PLAN_ETARY_COST_FREEZE_FLAG,
  T1.PLAN_SUB_COST_FREEZE_FLAG,
  T1.OPEN_ITEMS_FREEZE_FLAG,
  T1.REAL_ETARY_COST_FREEZE_FLAG,
  T1.CREATE_DATE,
  T1.INPUT_BY,
  'N' DEL_FLAG,
  T1.DW_LAST_UPDATE_DATE,
  t2.KTEXT as cost_center_type_desc
FROM DWI_COST_CENTER_MAIN_T_TMP T1
left join sdi.sdi_tkt05_1001 t2 on t1.cost_center_type = t2.KOSAR and t2.spras = 1 and t2.mandt = 800
;
