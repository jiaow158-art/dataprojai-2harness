SET hive.exec.dynamic.partition=true;  
SET hive.exec.dynamic.partition.mode=nonstrict; 
SET hive.exec.max.dynamic.partitions.pernode = 1000;
SET hive.exec.max.dynamic.partitions=1000;
insert  overwrite table  DWIFIN.DWI_GL_INVOICE_HEAD_T partition(period_id_m)  
select 
T1.MANDT       AS GROUP_NUM,
T1.VBELN       AS BILL_VOUCHER_NUM,
T1.VF_TODO     AS SD_BILL_VOUCHER_PROBLEM,
T1.VF_STATUS   AS SD_BILL_VOUCHER_STATUS,
T1.KIDNO       AS PAYMENT_REFERENCE_ID,
T1.KUNRG       AS PAYER_ID,
T1.ZTERM       AS PAY_TERM_CODE,
T1.MWSBK       AS VC_TAX_AMT,
T1.BELNR       AS ACCOUNT_VOUCHER_NUM,
T1.RFBSK       AS ACCOUNT_CHECK_STATUS,
T1.KKBER       AS CREDIT_CONTROL_SCOPE,
T1.CMWAE       AS CREDIT_CONTROL_SCOPE_CURRENCY,
COM.COMPANY_ID AS COMPANY_ID,
T1.BUKRS       AS COMPANY_CODE,
T1.KNUMV       AS VCH_TERM_NUM,
T1.GBSTK       AS WHOLE_VCH_DEAL_STATUS,
T1.NETWR       AS VC_NET_WORTH_AMT,
T1.FKSTO       AS BILL_CANCEL_FLAG,
T1.FKTYP       AS ISSUE_INVOICE_TYPE,
T1.ZUONR       AS DISTRIBUTION_NUM,
T1.VTWEG       AS DISTRIBUTION_CHANNEL_CODE,
T1.ERNAM       AS CREATE_BY,
T1.XBLNR       AS REFERENCE_VOUCHER_NUM,
T1.FKART_RL    AS BILL_LIST_TYPE,
T1.LANDTX      AS START_SHIPMENT_COUNTRY_CODE,
T1.KUNAG       AS CUST_SAL,
T1.STCEG_L     AS TAX_ARRIVE_COUNTRY,
T1.INCO2_L     AS INTER_TRADE_CLAUSE_LOC_1,
T1.INCO1       AS INTER_TRADE_TERMS_P1,
T1.INCO2       AS INTER_TRADE_TERMS_P2,
T1.REGIO       AS REGION_CODE,
T1.CMKUF       AS CREDIT_EX_RATE,
T1.ZZ_TAX      AS VAT_INVOICE_FLAG,
T1.HITYP_PR    AS PRICE_LEVEL_TYPE,
T1.KONDA       AS CUST_PRICE_GROUP,
T1.ZZ_UVK01    AS CUST_RESERVE_STATUS,
T1.TAXK1       AS ALTERNATIVE_TAX_TYPE,
T1.KDGRP       AS CUST_GROUP,
T1.UVPRS       AS VCH_NOWHOLE_STATUS,
T1.SFAKN       AS CANCEL_BILL_VOUCHER_NUM,
T1.KNKLI       AS CREDIT_REF_CUST_ACCOUNT_NUM,
T1.ZUKRI       AS BILL_VOUCHER_CONDITION_NUM,
T1.BUCHK       AS BILL_VOUCHER_POST_STATUS,
from_unixtime(unix_timestamp(T1.FKDAT,'yyyyMMdd'),'yyyy-MM-dd') as BILL_DATE,
T1.FKART       AS BILL_TYPE,
T1.MRNKZ       AS MANUAL_INVOICE_FLAG,
from_unixtime(unix_timestamp(T1.KURRF_DAT,'yyyyMMdd'),'yyyy-MM-dd') as CONVERSION_DATE2,
(case 
when T1.AEDAT < '19000101' THEN from_unixtime(unix_timestamp('19000101','yyyyMMdd'),'yyyy-MM-dd')
when T1.AEDAT > '47121231' THEN from_unixtime(unix_timestamp('47121231','yyyyMMdd'),'yyyy-MM-dd')
else from_unixtime(unix_timestamp(T1.AEDAT,'yyyyMMdd'),'yyyy-MM-dd') end) as LAST_UPDATE_DATE,
T1.CPKUR       AS EX_RATE_SET_FLAG,
T1.KTGRD       AS CUST_ACCT_ASSIGN_GROUP,
T1.LAND1       AS CNTRY_REGION_CODE,
from_unixtime(unix_timestamp(T1.ERDAT,'yyyyMMdd'),'yyyy-MM-dd') as CREATE_DATE,
T1.KURRF       AS FIN_ACC_POST_RATE,
T1.GJAHR       AS FISCAL_YEAR,
T1.ERZET       AS INPUT_TIME,
T1.KALSM       AS PRICING_PROCEDURE,
T1.VBTYP       AS SALE_DIS_VCH_TYPE,
T1.WAERK       AS VOUCHER_CURRENCY_CODE,
T1.BZIRK       AS SALE_REGION_CODE,
T1.VKORG       AS SALE_ORG_CODE,
T1.STCEG_H     AS OUTPUT_TAX_SOURCE_CODE,
T1.CHANGED_ON  AS UTC_TIME,
T1.DEL_FLAG,
T1.DW_LAST_UPDATE_DATE,
T1.ZLSCH AS PAY_TYPE,
T1.BLART AS VOUCHER_TYPE,
T1.SPART AS product_group,
SUBSTR(T1.FKDAT,1,6) AS period_id_m
from SDI.SDI_VBRK_1001  t1 
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM 
WHERE T1.BUKRS = COM.COMPANY_CODE 
AND COM.DEL_FLAG  = 'N' 
AND COM.END_DATE  = '4712-12-31';
