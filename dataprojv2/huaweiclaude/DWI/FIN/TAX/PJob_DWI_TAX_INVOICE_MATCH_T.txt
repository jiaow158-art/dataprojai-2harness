INSERT OVERWRITE TABLE DWIFIN.DWI_TAX_INVOICE_MATCH_T    --税票对账
SELECT
MANDT          AS     GROUP_NUM,
CASE
    WHEN S.BUKRS IS NULL OR S.BUKRS = '' THEN -999999
    ELSE NVL(COM.COMPANY_ID, -999998)
END AS COMPANY_ID,
CASE                                        
   WHEN S.BUKRS IS NULL OR S.BUKRS = '' THEN 'SNULL'            
   ELSE NVL(COM.COMPANY_CODE,'NOT EXISTS') 
END AS COMPANY_CODE,
POSNR2         AS     SALE_VOUCHER_ITEM_NUM,
POSNR          AS     BILL_ITEM_NUM,
VBELN          AS     BILL_VOUCHER_NUM,
TAX_NO         AS     TAX_NUM,
TAX_CX         AS     ORDER_NUM,
TAX_RATE       AS     TAX_RATE,
FKIMG          AS     ACT_BILLED_QTY,
GROES          AS     DIMENSION_SIZE,
CASE WHEN S.KUNNR IS NULL OR S.KUNNR = '' THEN -999999             
ELSE NVL(CUS.CUST_ID,-999998) 
END AS CUST_ID,
CASE                                        
   WHEN S.KUNNR IS NULL OR S.KUNNR = '' THEN 'SNULL'            
   ELSE NVL(CUS.CUST_NUM,'NOT EXISTS') 
END AS CUST_NUM,
MAKTX          AS     MATERIAL_DESC,
CASE WHEN S.MATNR IS NULL OR S.MATNR = '' THEN -999999             
ELSE NVL(MAT.MATERIAL_ID,-999998) 
END AS MATERIAL_ID,
CASE                                        
   WHEN S.MATNR IS NULL OR S.MATNR = '' THEN 'SNULL'            
   ELSE NVL(MAT.MATERIAL_NUM,'NOT EXISTS') 
END AS MATERIAL_NUM,
MEINS2         AS     BASE_MEASURE_UNIT2,
MEINS          AS     BASE_MEASURE_UNIT,
MWSBP          AS     DOC_CURRENCY_TAX_AMT,
NAME12         AS     NAME12,
NAME1          AS     NAME1,
from_unixtime(unix_timestamp(DATE_D,'yyyyMMdd'),'yyyy-MM-dd')  AS PRINTING_DATE, --8位字符转换为日期格式
DEMO           AS     DEMO_DESC,
NOTE           AS     NOTE_DESC,
NAME           AS     INPUT_NAME,
TAX_MONEY      AS     NET_VALUE_AMT,
TAX_MONEY2     AS     NET_VALUE_AMT2,
TEXT_NO        AS     TAX_INVOICE_TEXT_NUM,
TIME_T         AS     PRINTING_TIME,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE 
FROM SDI.SDI_ZINVOICE4MATCH_1001 S
LEFT JOIN  DWIMD.DWI_MD_CUST_GENERAL_T CUS  ----与客户主数据关联，取CUST_ID
ON CUS.CUST_NUM  = S.KUNNR 
AND CUS.GROUP_NUM = S.MANDT 
AND CUS.END_DATE ='4712-12-31'
AND CUS.DEL_FLAG='N'
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT ------与物料主数据关联，取MATERIAL_ID
ON MAT.MATERIAL_NUM  = S.MATNR 
AND MAT.GROUP_NUM = S.MANDT 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG='N'
LEFT JOIN DWIMD.DWI_MD_COMPANY_T COM    ------与公司主数据关联，取COMPANY_ID 
ON COM.COMPANY_CODE  = S.BUKRS
AND COM.group_num=S.MANDT
AND COM.DEL_FLAG  = 'N'
AND COM.END_DATE  = '4712-12-31' 
UNION ALL
SELECT
DWI.GROUP_NUM,
DWI.COMPANY_ID,
DWI.COMPANY_CODE,
DWI.SALE_VOUCHER_ITEM_NUM,
DWI.BILL_ITEM_NUM,
DWI.BILL_VOUCHER_NUM,
DWI.TAX_NUM,
DWI.ORDER_NUM,
DWI.TAX_RATE,
DWI.ACT_BILLED_QTY,
DWI.DIMENSION_SIZE,
DWI.CUST_ID,
DWI.CUST_NUM,
DWI.MATERIAL_DESC,
DWI.MATERIAL_ID,
DWI.MATERIAL_NUM,
DWI.BASE_MEASURE_UNIT2,
DWI.BASE_MEASURE_UNIT,
DWI.DOC_CURRENCY_TAX_AMT,
DWI.NAME12,
DWI.NAME1,
DWI.PRINTING_DATE,
DWI.DEMO_DESC,
DWI.NOTE_DESC,
DWI.INPUT_NAME,
DWI.NET_VALUE_AMT,
DWI.NET_VALUE_AMT2,
DWI.TAX_INVOICE_TEXT_NUM,
DWI.PRINTING_TIME,
'Y' AS DEL_FLAG,      ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_TAX_INVOICE_MATCH_T DWI
LEFT JOIN SDI.SDI_ZINVOICE4MATCH_1001 SDI
ON  SDI.MANDT=DWI.GROUP_NUM 
AND SDI.POSNR2=DWI.SALE_VOUCHER_ITEM_NUM
AND SDI.POSNR=DWI.BILL_ITEM_NUM
AND SDI.VBELN=DWI.BILL_VOUCHER_NUM
AND SDI.TAX_NO=DWI.TAX_NUM
WHERE  SDI.MANDT IS NULL
AND SDI.POSNR2 IS NULL
AND SDI.POSNR IS NULL
AND SDI.VBELN IS NULL
AND SDI.TAX_NO IS NULL
