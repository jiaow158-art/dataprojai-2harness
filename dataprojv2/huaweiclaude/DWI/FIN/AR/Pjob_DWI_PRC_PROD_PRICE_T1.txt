set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PROD_PRICE_T PARTITION(SOURCE_CODE='A004')   ---取SOURCE_CODE字段=对应表表名分区
SELECT 
S6.MANDT AS GROUP_NUM,               
S6.KNUMH AS CONDITION_RECORD_NUM,   
S6.DATAB AS SRC_SYS_START_DATE,      
FROM_UNIXTIME(UNIX_TIMESTAMP(S6.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE,    ---8位字符转换为日期格式
CASE WHEN SUBSTR(S6.DATBI,1,4) ='9999'          
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S6.DATBI,SUBSTR(S6.DATBI,1,4),'4712'),'yyyyMMdd'),'yyyy-MM-dd')
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(S6.DATBI,'yyyyMMdd'),'yyyy-MM-dd')
END AS END_DATE,                   ---8位字符，转换为日期格式，99991231转换为4712-12-31
NULL AS APPLICATION_PROGRAM_CODE,   
NULL AS CONDITION_TYPE,            
NULL AS APPROVAL_STATUS_CODE,       
NULL AS PRICE_TREATMENT_STATUS,
NULL AS SALE_OFFICE_CODE,
R.CONFIG_CODE AS SALE_ORG_CODE,      
R1.CONFIG_CODE AS DIS_CHANNEL_CODE,  
'-999999' AS FACTORY_ID,
NULL AS FACTORY_CODE,
NULL AS PAYER_ID,
'-999999' AS CUST_ID,               
NULL AS CUST_NUM,
CASE WHEN S6.MATNR IS NULL OR S6.MATNR = '' THEN -999999             
ELSE NVL(MAT.MATERIAL_ID,-999998) 
END AS MATERIAL_ID,  
CASE                                        
   WHEN S6.MATNR IS NULL OR S6.MATNR = '' THEN 'SNULL'            
   ELSE NVL(MAT.MATERIAL_NUM,'NOT EXISTS') 
END AS MATERIAL_NUM,
NULL AS CUST_PRICE_GROUP,
'SDI_A004_1001' AS SOURCE_TYPE,
'销售机构/分销渠道/物料' AS SOURCE_DESC,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE   
FROM SDI.SDI_A004_1001 S6
LEFT JOIN  DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R   ---机构单位：销售机构：文本
ON S6.VKORG=R.Src_sys_config_code
AND S6.MANDT=R.GROUP_NUM
AND R.SOURCE_CODE='TVKOT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1   ---机构单位：分销渠道：文本
ON S6.VTWEG=R1.Src_sys_config_code
AND S6.MANDT=R1.GROUP_NUM
AND R1.SOURCE_CODE='TVTWT'
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT   ---与物料主数据关联，取MATERIAL_ID,源字段非空处理
ON MAT.MATERIAL_NUM  = S6.MATNR 
AND MAT.GROUP_NUM = S6.MANDT 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG='N'
UNION ALL
SELECT
DWI.GROUP_NUM,               
DWI.CONDITION_RECORD_NUM,    
DWI.SRC_SYS_START_DATE,      
DWI.START_DATE,              
DWI.END_DATE,                
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_TYPE,          
DWI.APPROVAL_STATUS_CODE,    
DWI.PRICE_TREATMENT_STATUS,  
DWI.SALE_OFFICE_CODE,       
DWI.SALE_ORG_CODE,          
DWI.DIS_CHANNEL_CODE,        
DWI.FACTORY_ID,              
DWI.FACTORY_CODE,            
DWI.PAYER_ID,                
DWI.CUST_ID,                 
DWI.CUST_NUM,                
DWI.MATERIAL_ID,             
DWI.MATERIAL_NUM,            
DWI.CUST_PRICE_GROUP,        
DWI.SOURCE_TYPE,             
DWI.SOURCE_DESC,                         
'Y' AS DEL_FLAG,            ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T DWI
LEFT JOIN SDI.SDI_A004_1001 S6
ON NVL(S6.MANDT,'SNULL')=NVL(DWI.GROUP_NUM,'SNULL')
AND NVL(S6.KNUMH,'SNULL')=NVL(DWI.CONDITION_RECORD_NUM,'SNULL')
AND NVL(S6.DATAB,'SNULL')=NVL(DWI.SRC_SYS_START_DATE,'SNULL')
WHERE S6.MANDT  IS NULL
AND S6.KNUMH  IS NULL
AND S6.DATAB  IS NULL
AND source_type='SDI_A004_1001'
