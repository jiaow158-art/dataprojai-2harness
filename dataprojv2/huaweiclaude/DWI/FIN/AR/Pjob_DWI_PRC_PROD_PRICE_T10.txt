set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PROD_PRICE_T PARTITION(SOURCE_CODE='A963')
SELECT 
S5.MANDT AS GROUP_NUM,
S5.KNUMH AS CONDITION_RECORD_NUM,
S5.DATAB AS SRC_SYS_START_DATE,
FROM_UNIXTIME(UNIX_TIMESTAMP(S5.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE,   ---8位字符转换为日期格式
CASE WHEN SUBSTR(S5.DATBI,1,4) ='9999' 
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S5.DATBI,SUBSTR(S5.DATBI,1,4),'4712'),'yyyyMMdd'),'yyyy-MM-dd')
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(S5.DATBI,'yyyyMMdd'),'yyyy-MM-dd')
END AS END_DATE,                                    ---8位字符，转换为日期格式，99991231转换为4712-12-31
R1.CONFIG_CODE AS APPLICATION_PROGRAM_CODE,
S5.KSCHL AS CONDITION_TYPE,
NULL AS APPROVAL_STATUS_CODE,
NULL AS PRICE_TREATMENT_STATUS,
NULL AS SALE_OFFICE_CODE,
NULL AS SALE_ORG_CODE,
NULL AS DIS_CHANNEL_CODE,
'-999999' AS FACTORY_ID,
NULL AS FACTORY_CODE,
NULL AS PAYER_ID,
'-999999' AS CUST_ID,
NULL AS CUST_NUM,
CASE WHEN S5.MATNR IS NULL OR S5.MATNR = '' THEN -999999             
ELSE NVL(MAT.MATERIAL_ID,-999998) 
END AS MATERIAL_ID,
CASE                                        
   WHEN S5.MATNR IS NULL OR S5.MATNR = '' THEN 'SNULL'            
   ELSE NVL(MAT.MATERIAL_NUM,'NOT EXISTS') 
END AS MATERIAL_NUM,
R.CONFIG_CODE AS CUST_PRICE_GROUP,
'SDI_A963_1001' AS SOURCE_TYPE,
'价格组/物料' AS SOURCE_DESC,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_A963_1001 S5
LEFT JOIN  DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R      ---条件： 客户等级组： 文本
ON S5.KONDA=R.Src_sys_config_code
AND S5.MANDT=R.GROUP_NUM
AND R.SOURCE_CODE='T188T' 
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT    ---与物料主数据关联，取MATERIAL_ID,源字段非空处理
ON MAT.MATERIAL_NUM  = S5.MATNR 
AND MAT.GROUP_NUM = S5.MANDT 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG='N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1    ---条件类型： 文本
ON S5.KAPPL=R1.Src_sys_config_code
AND S5.MANDT=R1.GROUP_NUM
AND R1.SOURCE_CODE='T685T' 
UNION ALL
SELECT
DWI.GROUP_NUM,               
DWI.CONDITION_RECORD_NUM,    
DWI.SRC_SYS_START_DATE,      
DWI.START_DATE,              
DWI.END_DATE,                
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_TYPE,          
DWI.APPROVAL_STATUS_CODE,    
DWI.PRICE_TREATMENT_STATUS,  
DWI.SALE_OFFICE_CODE,       
DWI.SALE_ORG_CODE,          
DWI.DIS_CHANNEL_CODE,        
DWI.FACTORY_ID,              
DWI.FACTORY_CODE,            
DWI.PAYER_ID,                
DWI.CUST_ID,                 
DWI.CUST_NUM,                
DWI.MATERIAL_ID,             
DWI.MATERIAL_NUM,            
DWI.CUST_PRICE_GROUP,        
DWI.SOURCE_TYPE,             
DWI.SOURCE_DESC,                         
'Y' AS DEL_FLAG,          ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T DWI
LEFT JOIN SDI.SDI_A963_1001 S5
ON NVL(S5.MANDT,'SNULL')=NVL(DWI.GROUP_NUM,'SNULL')
AND NVL(S5.KNUMH,'SNULL')=NVL(DWI.CONDITION_RECORD_NUM,'SNULL')
AND NVL(S5.DATAB,'SNULL')=NVL(DWI.SRC_SYS_START_DATE,'SNULL')
WHERE S5.MANDT  IS NULL
AND S5.KNUMH  IS NULL
AND S5.DATAB  IS NULL
AND source_type='SDI_A963_1001'
