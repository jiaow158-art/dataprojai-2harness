INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PRICE_CONDITION_T     --内部结算价格
SELECT 
S1.MANDT AS GROUP_NUM,
S1.KNUMH AS CONDITION_RECORD_NUM,
S1.KOPOS AS CONDITION_SERIAL_NUM,
S1.KBETR AS CONDITION_AMT,
S1.KUMNE AS CONVERSION_DENOMINATOR,	
S1.KUMZA AS CONVERSION_MOLECULE,          ---转换为8位日期，00000000转换为1900-01-01
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S1.VALDT,SUBSTR(S1.VALDT,1,8),'19000101'),'yyyyMMdd'),'yyyy-MM-dd') AS FIXED_START_DATE,
S1.MEINS AS BASE_MEASURE_UNIT,
R1.CONFIG_CODE AS APPLICATION_PROGRAM_CODE,
S1.KMEIN AS CONDITION_UNIT,
S1.KONWA AS CONDITION_UNIT_TYPE,
S1.KPEIN AS CONDITION_PRICE_UNIT,
S1.KZNEP AS CONDITION_EXCLUSION_FLAG,
S1.MXWRT AS CONDITION_LOWER_LIMIT,
T.domain_value_code AS CONDITION_CALC_TYPE,
S1.KONMS AS CONDITION_LEVEL_MEASURE_UNIT,
S1.KSCHL AS CONDITION_TYPE,
S1.KWAEH AS CONDITION_CURRENCY_CODE,
S1.LOEVM_KO AS CONDITION_ITEM_DEL_FLAG,
S1.ZAEHK_IND AS CONDITION_ITEM_INDEX,
S1.MDFLG AS MATRIX_MAINTENANCE_FALG,
S1.KZBZG AS BASIC_LEVEL_FALG,
S1.STFKZ AS LEVEL_TYPE,
S1.MWSK1 AS SALES_PURCHASE_TAX_CODE,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_KONP_1001 S1
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1    --条件类型： 文本
ON S1.KAPPL=R1.Src_sys_config_code
AND S1.MANDT=R1.GROUP_NUM
AND R1.SOURCE_CODE='T685T'
LEFT JOIN DWIMD.DWI_MD_DOMAIN_CONFIG_REL_R T    --与配置表关联，转换为域值编码，配置字段KRECH
ON  T.domain_name  = 'KRECH'
AND T.src_sys_domain_value_code =S1.KRECH
UNION ALL
SELECT
DWI.GROUP_NUM,
DWI.CONDITION_RECORD_NUM,
DWI.CONDITION_SERIAL_NUM,
DWI.CONDITION_AMT,
DWI.CONVERSION_DENOMINATOR,
DWI.CONVERSION_MOLECULE,
DWI.FIXED_START_DATE,
DWI.BASE_MEASURE_UNIT,
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_UNIT,
DWI.CONDITION_UNIT_TYPE,
DWI.CONDITION_PRICE_UNIT,
DWI.CONDITION_EXCLUSION_FLAG,
DWI.CONDITION_LOWER_LIMIT,
DWI.CONDITION_CALC_TYPE,
DWI.CONDITION_LEVEL_MEASURE_UNIT,
DWI.CONDITION_TYPE,
DWI.CONDITION_CURRENCY_CODE,
DWI.CONDITION_ITEM_DEL_FLAG,
DWI.CONDITION_ITEM_INDEX,
DWI.MATRIX_MAINTENANCE_FALG,
DWI.BASIC_LEVEL_FALG,
DWI.LEVEL_TYPE,
DWI.SALES_PURCHASE_TAX_CODE,
'Y' AS DEL_FLAG,                   ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PRICE_CONDITION_T DWI
LEFT JOIN SDI.SDI_KONP_1001 SDI
ON SDI.MANDT=DWI.GROUP_NUM
AND SDI.KNUMH=DWI.CONDITION_RECORD_NUM
AND SDI.KOPOS=DWI.CONDITION_SERIAL_NUM
WHERE SDI.MANDT IS NULL
AND SDI.KNUMH IS NULL
AND SDI.KOPOS IS NULL
