set hive.cbo.enable=true;
set hive.mapjoin.smalltable.filesize=100000000;
set hive.auto.convert.join.noconditionaltask.size=200000000;
set hive.mapred.local.mem=4096;
INSERT OVERWRITE TABLE DWIFIN.DWI_PRC_PROD_PRICE_T PARTITION(SOURCE_CODE='A930')
SELECT 
S1.MANDT AS GROUP_NUM,
S1.KNUMH AS CONDITION_RECORD_NUM,
S1.DATAB AS SRC_SYS_START_DATE,
FROM_UNIXTIME(UNIX_TIMESTAMP(S1.DATAB,'yyyyMMdd'),'yyyy-MM-dd') AS START_DATE,   ---8位字符转换为日期格式
CASE WHEN SUBSTR(S1.DATBI,1,4) ='9999' 
THEN 
FROM_UNIXTIME(UNIX_TIMESTAMP(REPLACE(S1.DATBI,SUBSTR(S1.DATBI,1,4),'4712'),'yyyyMMdd'),'yyyy-MM-dd')
ELSE 
FROM_UNIXTIME(UNIX_TIMESTAMP(S1.DATBI,'yyyyMMdd'),'yyyy-MM-dd')
END AS END_DATE,                                     ---8位字符，转换为日期格式，99991231转换为4712-12-31
R4.CONFIG_CODE AS APPLICATION_PROGRAM_CODE,
S1.KSCHL AS CONDITION_TYPE,
NULL AS APPROVAL_STATUS_CODE,
NULL AS PRICE_TREATMENT_STATUS,
R3.CONFIG_CODE AS SALE_OFFICE_CODE,
R.CONFIG_CODE AS SALE_ORG_CODE,
R1.CONFIG_CODE AS DIS_CHANNEL_CODE,
'-999999' AS FACTORY_ID,
NULL AS FACTORY_CODE,
NULL AS PAYER_ID,
'-999999' AS CUST_ID,
NULL AS CUST_NUM,
CASE WHEN S1.MATNR IS NULL OR S1.MATNR = '' THEN -999999             
ELSE NVL(MAT.MATERIAL_ID,-999998) 
END AS MATERIAL_ID,
CASE                                        
   WHEN S1.MATNR IS NULL OR S1.MATNR = '' THEN 'SNULL'            
   ELSE NVL(MAT.MATERIAL_NUM,'NOT EXISTS') 
END AS MATERIAL_NUM,
R2.CONFIG_CODE AS CUST_PRICE_GROUP,
'SDI_A930_1001' AS SOURCE_TYPE,
'销售组织/分销渠道/销售部门/价格组/物料' AS SOURCE_DESC,
'N' AS DEL_FLAG,
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM SDI.SDI_A930_1001 S1
LEFT JOIN  DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R       ---机构单位：销售机构：文本
ON S1.VKORG=R.Src_sys_config_code
AND S1.MANDT=R.GROUP_NUM
AND R.SOURCE_CODE='TVKOT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R1       ---机构单位：分销渠道：文本
ON S1.VTWEG=R1.Src_sys_config_code
AND S1.MANDT=R1.GROUP_NUM
AND R1.SOURCE_CODE='TVTWT'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R2       ---条件： 客户等级组： 文本
ON S1.KONDA=R2.Src_sys_config_code
AND S1.MANDT=R2.GROUP_NUM
AND R2.SOURCE_CODE='T188T'   
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R3      ---机构单位：销售办公室：文本
ON S1.VKBUR=R3.Src_sys_config_code
AND S1.MANDT=R3.GROUP_NUM
AND R3.SOURCE_CODE='TVKBT' 
LEFT JOIN  DWIMD.DWI_MD_MATERIAL_GENERAL_T MAT       ---与物料主数据关联，取MATERIAL_ID,源字段非空处理
ON MAT.MATERIAL_NUM  = S1.MATNR 
AND MAT.GROUP_NUM = S1.MANDT 
AND MAT.END_DATE ='4712-12-31'
AND MAT.DEL_FLAG='N'
LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_REL_R R4     ---条件类型： 文本
ON S1.KAPPL=R4.Src_sys_config_code
AND S1.MANDT=R4.GROUP_NUM
AND R4.SOURCE_CODE='T685T' 
UNION ALL
SELECT
DWI.GROUP_NUM,               
DWI.CONDITION_RECORD_NUM,    
DWI.SRC_SYS_START_DATE,      
DWI.START_DATE,              
DWI.END_DATE,                
DWI.APPLICATION_PROGRAM_CODE,
DWI.CONDITION_TYPE,          
DWI.APPROVAL_STATUS_CODE,    
DWI.PRICE_TREATMENT_STATUS,  
DWI.SALE_OFFICE_CODE,       
DWI.SALE_ORG_CODE,          
DWI.DIS_CHANNEL_CODE,        
DWI.FACTORY_ID,              
DWI.FACTORY_CODE,            
DWI.PAYER_ID,                
DWI.CUST_ID,                 
DWI.CUST_NUM,                
DWI.MATERIAL_ID,             
DWI.MATERIAL_NUM,            
DWI.CUST_PRICE_GROUP,        
DWI.SOURCE_TYPE,             
DWI.SOURCE_DESC,                         
'Y' AS DEL_FLAG,            ---删除标识判断
CURRENT_TIMESTAMP() AS DW_LAST_UPDATE_DATE
FROM DWIFIN.DWI_PRC_PROD_PRICE_T DWI
LEFT JOIN SDI.SDI_A930_1001 S1
ON NVL(S1.MANDT,'SNULL')=NVL(DWI.GROUP_NUM,'SNULL')
AND NVL(S1.KNUMH,'SNULL')=NVL(DWI.CONDITION_RECORD_NUM,'SNULL')
AND NVL(S1.DATAB,'SNULL')=NVL(DWI.SRC_SYS_START_DATE,'SNULL')
WHERE S1.MANDT  IS NULL
AND S1.KNUMH  IS NULL
AND S1.DATAB  IS NULL
AND source_type='SDI_A930_1001'
