insert overwrite table DWIBS .DWI_APR_RECORD_LOG_T
select 
ID
,REQUESTID AS REQUEST_ID
,NODEID  AS NODE_ID
,ISREMARK AS OPERAT_STATUS
,OPERATORID AS OPERATOR_ID
,OPERATORTYPE AS OPERATOR_TYPE
,FROM_UNIXTIME(UNIX_TIMESTAMP(OPERATEDATE,'yyyymmdd'),'yyyy-mm-dd') AS OPERAT_DATE
,OPERATETIME AS OPERAT_TIME
,OPERATETYPE AS OPERAT_TYPE
,OPERATENAME AS OPERAT_TYPE_CN_NAME
,OPERATECODE AS OPERAT_TYPE_CODE
,ISINVALID AS FORCED_WITHDRAW
,INVALIDID AS FORCED_WITHDRAW_USER_ID
,FROM_UNIXTIME(UNIX_TIMESTAMP(INVALIDDATE,'yyyymmdd'),'yyyy-mm-dd') AS FORCED_WITHDRAW_DATE
,INVALIDTIME AS FORCED_WITHDRAW_TIME
,DETAILINFO AS DETAIL_JSON_DATA
,RETYPE  AS OPERAT_TYPE_1
,REOPERATORID AS PRINCIPAL_ID
,'N' AS DEL_FLAG
,current_timestamp() as DW_LAST_UPDATE_DATE
from SDI.SDI_WORKFLOW_REQUESTOPERATELOG_1004
UNION 
--更新删除标识:DWI表外关联SDI表，DWI表有SDI表没有的数据。DEL_FLAG打上Y，DW_LAST_UPDATE_DATE 取系统时间，其余字段取DWI表数据
select 
 T1.ID
,T1.REQUEST_ID
,T1.NODE_ID
,T1.OPERAT_STATUS
,T1.OPERATOR_ID
,T1.OPERATOR_TYPE
,T1.OPERAT_DATE
,T1.OPERAT_TIME
,T1.OPERAT_TYPE
,T1.OPERAT_TYPE_CN_NAME
,T1.OPERAT_TYPE_CODE
,T1.FORCED_WITHDRAW
,T1.FORCED_WITHDRAW_USER_ID
,T1.FORCED_WITHDRAW_DATE
,T1.FORCED_WITHDRAW_TIME
,T1.DETAIL_JSON_DATA
,T1.OPERAT_TYPE_1
,T1.PRINCIPAL_ID
,'Y' AS DEL_FLAG
,current_timestamp() AS DW_LAST_UPDATE_DATE
from DWIBS .DWI_APR_RECORD_LOG_T T1
left join sdi.sdi_workflow_requestoperatelog_1004 t2 
on T1.ID = T2.ID
where T2.ID is null;
