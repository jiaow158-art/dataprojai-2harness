DELETE FROM DM.DM_WM_DEFECTIVE_PRODUCT_STOCK_DETAIL  WHERE stat_month = '${PERIOD_ID_M}';

INSERT INTO DM.DM_WM_DEFECTIVE_PRODUCT_STOCK_DETAIL
with mat as (
select t1.material_num,
       t1.material_name,
	   t1.product_level2_name  as cat,
	   t1.dimension,
       t1.def_prod_channel  as prod_channel ,
       t1.width * t1.length_1/1000/1000 as area
  from dwimd.dwi_md_data_material_general_t t1 
  where matl_group_code ='1000010'
	and external_matl_group_code in('10010','10030','10070')
    --and t1.product_level2_code  like '1%'
    --and t1.product_level2_code  not in('1011','1012','1013','1014','1016')
	and matl_type_code='FERT'
	and material_num not like '%_E'
),
material_group2_data as(
	SELECT MATERIAL_NUM
                ,MATL_GROUP1_CODE --物料组1编码
                ,MATL_GROUP1_NAME --物料组1名称
                ,MATL_GROUP2_CODE --物料组2编码
                ,MATL_GROUP2_NAME --物料组2名称
                ,MATL_GROUP3_CODE --物料组3编码
                ,MATL_GROUP3_NAME --物料组3名称
                ,MATL_GROUP4_CODE --物料组4编码
                ,MATL_GROUP4_NAME --物料组4名称
                ,MATL_GROUP5_CODE --物料组5编码
                ,MATL_GROUP5_NAME --物料组5名称
				,SALES_ORG_CODE --销售组织
                ,ROW_NUMBER() OVER(PARTITION BY MATERIAL_NUM ORDER BY SALES_ORG_CODE ASC) AS RN
                FROM DWIMD.DWI_MD_DATA_MATERIAL_SALES_T
                WHERE DEL_FLAG = 'N'
                AND END_DATE = '4712-12-31'
                AND GROUP_NUM = 800
)
select 1,
       '${PERIOD_ID_M}' as stat_month,
       a.material_num ,
	   a.batch_num,
	   t1.cat,
	   t1.dimension ,
	   a.factory_werks_code,
	   a.inv_location ,
       t.BATCH_AMB_IN_TIME,
       CASE WHEN  (substr(t.batch_in_time,1,1) = '0' OR t.batch_in_time IS NULL) THEN '2年以上'
            WHEN  date_month_cz.months_between(last_day(to_date(a.month_date ,'yyyyMMdd'))+1,to_date(t.batch_in_time,'yyyy-MM-dd'))<=6  then '半年以内'
            when  date_month_cz.months_between(last_day(to_date(a.month_date ,'yyyyMMdd'))+1,to_date(t.batch_in_time,'yyyy-MM-dd'))<=12 then '7-12个月'
            when  date_month_cz.months_between(last_day(to_date(a.month_date ,'yyyyMMdd'))+1,to_date(t.batch_in_time,'yyyy-MM-dd'))<=24 then '1-2年'
            else '2年以上'  end as stock_age_seg,
		case when A.factory_werks_code  LIKE '%80' and t.clearance_reason_code='QC19' then  '清仓小堆头' 
		     when (C1.PROMOTION_REASON_CODE ='AC05'  and (a.special_stock_flag_2='Q_T148T' or a.inv_location in('GCCA','GCTA','GCSA','BJ01','GCQA','GCPA','GCSD')) ) then '超一年定制(战略保存部分)'
		     else C1.REJECTS_TYPE end AS REJECTS_TYPE, --不良品类型  
		--,T.FIRST_PROMOTE_DATE --首次维护促销原因日期
		sum( a.stock_qty*t1.area  ) as stock_area,
		sum( a.stock_qty ) as stock_qty,
		current_timestamp() as dw_last_update_date,
		t1.material_name,     -- 物料名称
		a.inv_location_name,   -- 库存地点名称
		t4.matl_group2_code,   -- 物料组编码
		t4.matl_group2_name,    -- 物料组名称
		t.batch_in_time as   batch_in_time
 from dwrotd.dwr_wm_all_type_stock_month_t  a 
 join mat t1 on a.material_num  = t1.material_num 
LEFT JOIN DWIMD.DWI_MD_MATL_BATCH_INFO_T T
  ON A.MATERIAL_NUM = T.MATERIAL_NUM
 AND A.BATCH_NUM = T.BATCH_NUM     
LEFT JOIN UPLOAD.UPLOAD_REJECTS_DEFINITION_T C1
  ON T.PROMOTE_REASON_CODE = C1.PROMOTION_REASON_CODE
LEFT JOIN material_group2_data T4
ON a.material_num = T4.material_num and T4.RN = 1
where a.month_date = '${PERIOD_ID_M}'
     and a.goods_move_stock_type_sid IN('01','02','07')
     and CASE WHEN A.batch_num  LIKE '%888888' or A.batch_num LIKE '%999999' THEN 0-- 虚拟入库批次不计算
      WHEN A.factory_werks_code LIKE '%80' THEN 1   -- 80全算
      --WHEN (C1.PROMOTION_REASON_CODE ='AC05'  and (a.special_stock_flag_2='Q_T148T' or a.inv_location in('GCCA','GCTA','GCSA','BJ01','GCQA','GCPA','GCSD')) ) THEN 0 --AC05的项目库存不计算不良品，配送仓也不算
      WHEN (C1.PROMOTION_REASON_CODE IS NOT NULL and  (month_date>=replace(substr(T.FIRST_PROMOTE_DATE,1,7),'-','') OR T.FIRST_PROMOTE_DATE is null))  THEN 1
	  ELSE 0
	   END =1
	group by a.month_date,
			 a.material_num ,
			 a.batch_num,
			 t1.cat,
			 t1.dimension ,
			 a.factory_werks_code,
			 a.inv_location ,
			 t.BATCH_AMB_IN_TIME,
			case when A.factory_werks_code  LIKE '%80' and t.clearance_reason_code='QC19' then  '清仓小堆头' 
					 when (C1.PROMOTION_REASON_CODE ='AC05'  and (a.special_stock_flag_2='Q_T148T' or a.inv_location in('GCCA','GCTA','GCSA','BJ01','GCQA','GCPA','GCSD')) ) then '超一年定制(战略保存部分)'
					 else C1.REJECTS_TYPE end,
			stock_age_seg,
			t1.material_name,
			a.inv_location_name,
			t4.matl_group2_code,
			t4.matl_group2_name,
			t.batch_in_time
			having sum(a.stock_qty*t1.area)>0;
