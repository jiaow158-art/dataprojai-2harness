--清空临时表
TRUNCATE TABLE DWROTD.DWR_OTD_SALES_ORDER_SAP_TMP;
--对关联字段去重，防止发散
WITH PRCD_ELEMENTS AS (
  SELECT
    PE.CONDITION_ITEM_NUM, --条件项目号
    PE.VCH_TERM_NUM,       --凭证条件号
    PE.CONDITION_TYPE,     --条件类型
    MAX(PE.CONDITION_PRICE_UNIT) AS CONDITION_PRICE_UNIT, --条件定价单位
    MAX(PE.CONDITION_AMT) AS CONDITION_AMT --费率（金额或百分比）
  FROM
    DWRFIN.DWR_PRC_CONDITIONAL_PRICE_F PE  --定价要素表
  WHERE 
    1=1
    AND PE.CONDITION_ACTIVE_FLAG IS NULL     --限制取有效(激活)数据
    AND (UPPER(PE.CONDITION_TYPE) = 'MWSI'   --限制条件类型为MWSI，取税率
     OR UPPER(PE.CONDITION_TYPE) = 'ZDJ0'    --限制条件类型为ZDJ0，取折扣率
     OR UPPER(PE.CONDITION_TYPE) = 'PR00')   --限制条件类型为PR00，取含税单价
  GROUP BY
    PE.CONDITION_ITEM_NUM,
    PE.VCH_TERM_NUM,
    PE.CONDITION_TYPE
)
--向临时表插入数据
INSERT INTO
  DWROTD.DWR_OTD_SALES_ORDER_SAP_TMP
SELECT
  DSHS.GROUP_NUM AS GROUP_NUM,--集团
  DSHS.SAL_VCH_NUM AS SAL_VCH_NUM,--销售凭证
  DSIS.SAL_VCH_ITEM_NUM AS SAL_VCH_ITEM_NUM,--销售凭证项目
  R.SRC_SYS_CONFIG_CODE AS SRC_DIS_CHANNEL_CODE,--分销渠道
  DSHS.DIS_CHANNEL AS DIS_CHANNEL,--分销渠道(合成)
  DSHS.DIS_CHANNEL_NAME AS DIS_CHANNEL_NAME,--分销渠道中文名称
  DSHS.SO_NUM AS SO_NUM,--销售订单编码
  DSHS.ENGNR_REPORT_NUM AS ENGNR_REPORT_NUM,--工程报表单号(商机/机会点)
  DSHS.SAL_ORG_ID AS SAL_ORG_ID,--销售组织
  DSHS.SAL_ORG_ID_NAME AS SAL_ORG_ID_NAME,--销售组织中文名称
  DSHS.CUST_PO_TYPE AS CUST_PO_TYPE,--客户采购订单类型
  R1.SRC_SYS_CONFIG_CODE AS SRC_PROD_GROUP_CODE,--产品组
  DSIS.PROD_GROUP AS PROD_GROUP,--产品组(合成)
  DSIS.PROD_GROUP_NAME AS PROD_GROUP_NAME,--产品组中文名称
  DSHS.CUST_ID AS CUST_ID,--客户ID
  DSHS.CUST_NUM AS CUST_NUM,--客户编码
  DSHS.CUST_NAME AS CUST_NAME,--客户名称
  DSIS.MATERIAL_ID AS MATERIAL_ID,--物料ID
  DSIS.MATERIAL_NUM AS MATERIAL_NUM,--物料编码
  DSIS.MATERIAL_NAME AS MATERIAL_NAME,--物料名称
  DSIS.BASE_MEASURE_UNIT AS BASE_MEASURE_UNIT,--基本计量单位
  DSIS.SAL_TARGET_QTY AS SAL_TARGET_QTY,--目标数量（采用销售单位）
  DSIS.SAL_ACMLT_CONFIRM_QTY AS SAL_ACMLT_CONFIRM_QTY,--以销售单位表示的累计确认数量
  DSIS.ACMLT_BASE_CONFIRM_QTY AS ACMLT_BASE_CONFIRM_QTY,--以基本单位表示的累计确认数量
  DSHS.CREATE_DATE AS LAST_UPD_TIMESTAMP,--最后更改的日期
  CASE
    WHEN T3.CONDITION_PRICE_UNIT = 0
    OR T3.CONDITION_PRICE_UNIT IS NULL THEN 0
    ELSE T3.CONDITION_AMT / T3.CONDITION_PRICE_UNIT
  END AS INCLUDE_TAX_PRICE,--含税单价
  T4.CONDITION_AMT AS TAX_RATE,--税率
  T5.CONDITION_AMT AS DISCOUNT_RATE,--折扣率
  NULL AS DELIVERY_STATUS,--交货状态
  DSIS.ITEM_NETWORTH AS ITEM_NETWORTH,--不含税总净值
  DSIS.DOC_CURRENCY_TAX AS DOC_CURRENCY_TAX,--总税额
  DSIS.SAL_ACCUMULATE_QTY AS SAL_ACCUMULATE_QTY,--销售数量
  DSIS.SUBTOTAL_PRICE_PROCESS1 AS SUBTOTAL_PRICE_PROCESS1,--折扣后，未使用积分前的不含税价格
  DSIS.SUBTOTAL_PRICE_PROCESS2 AS SUBTOTAL_PRICE_PROCESS2,--折扣后，使用积分后的不含税价格
  'N' AS DEL_FLAG,--删除标识
  SYSDATE AS DW_LAST_UPDATE_DATE --DW最后更新时间
FROM
  DWROTD.DWR_SO_HEAD_SAP_F DSHS                  --SAP销售订单头表
  INNER JOIN DWROTD.DWR_SO_ITEM_SAP_F DSIS       --SAP销售订单行表
  ON DSHS.SAL_VCH_NUM = DSIS.SAL_VCH_NUM
  LEFT JOIN PRCD_ELEMENTS T3                     --定价要素表(取含税单价)
  ON DSIS.SAL_VCH_ITEM_NUM = T3.CONDITION_ITEM_NUM
  AND DSHS.VCH_TERM_NO = T3.VCH_TERM_NUM
  AND UPPER(T3.CONDITION_TYPE) = 'PR00'
  LEFT JOIN PRCD_ELEMENTS T4                     --定价要素表(税率)
  ON DSIS.SAL_VCH_ITEM_NUM = T4.CONDITION_ITEM_NUM
  AND DSHS.VCH_TERM_NO = T4.VCH_TERM_NUM
  AND UPPER(T4.CONDITION_TYPE) = 'MWSI'
  LEFT JOIN PRCD_ELEMENTS T5                     --定价要素表(折扣率)
  ON DSIS.SAL_VCH_ITEM_NUM = T5.CONDITION_ITEM_NUM
  AND DSHS.VCH_TERM_NO = T5.VCH_TERM_NUM
  AND UPPER(T5.CONDITION_TYPE) = 'ZDJ0'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R --通用配置表
  ON DSHS.GROUP_NUM = R.GROUP_NUM
  AND DSHS.DIS_CHANNEL = R.CONFIG_CODE   --分销渠道
  AND R.LANG_CODE = '1'
  LEFT JOIN DWIMD.DWI_MD_GENERAL_CONFIG_CODE_R R1 --通用配置表
  ON DSIS.GROUP_NUM = R1.GROUP_NUM
  AND DSIS.PROD_GROUP = R1.CONFIG_CODE   --产品组
  AND R1.LANG_CODE = '1'
;
--将临时表转换成正式表
ALTER TABLE
  DWROTD.DWR_OTD_SALES_ORDER_SAP_API RENAME TO DWR_OTD_SALES_ORDER_SAP_API_01;
ALTER TABLE
  DWROTD.DWR_OTD_SALES_ORDER_SAP_TMP RENAME TO DWR_OTD_SALES_ORDER_SAP_API;
ALTER TABLE
  DWROTD.DWR_OTD_SALES_ORDER_SAP_API_01 RENAME TO DWR_OTD_SALES_ORDER_SAP_TMP;
--重新建立索引
DROP INDEX IF EXISTS DWROTD.PN_START_TIME_TMP_INDEX;
DROP INDEX IF EXISTS DWROTD.PN_START_TIME_INDEX;
CREATE INDEX PN_START_TIME_TMP_INDEX ON DWROTD.DWR_OTD_SALES_ORDER_SAP_TMP(LAST_UPD_TIMESTAMP, SRC_DIS_CHANNEL_CODE);
CREATE INDEX PN_START_TIME_INDEX ON DWROTD.DWR_OTD_SALES_ORDER_SAP_API(LAST_UPD_TIMESTAMP, SRC_DIS_CHANNEL_CODE);
