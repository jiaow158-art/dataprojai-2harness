do
$$
    declare
        P_DATE varchar(500) ;
    begin
        for P_DATE in (select day_id
                       from dm.dm_dim_date_d
                       where day_id >= to_char(to_date('${PERIOD_ID_D1}'), 'yyyyMMdd')
                         and day_id <= to_char(to_date('${PERIOD_ID_D2}'), 'yyyyMMdd')
                       order by day_id)
            loop
                DELETE
                FROM DM.DM_SA_CT_Integrated_head_manual_2022_T
                WHERE CALDAY = to_char(to_date(P_DATE), 'yyyyMMdd');
                insert into DM.DM_SA_CT_Integrated_head_manual_2022_T

                with last_year_month_days as (
                    select date_part('days',
                                     date_trunc('months', to_date(P_DATE) - interval '1 years') + interval '1 months' -
                                     interval '1 days') ::int days
                ),
                     ct_2022_GD02 as (
                         select sum(case
                                        when stat_month = to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM') then
                                                target / b.days *
                                                to_char(to_date(P_DATE) - interval '1 years', 'dd'):: int
                                        else 0 end)   ct_2022_GD02_zxsyj_qntq,--去年同期业绩
                                sum(case
                                        when stat_month = to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM') then
                                            target
                                        else 0 end)   ct_2022_GD02_zxsyj_qnzy,--去年同期整月
                                sum(case
                                        when to_char(to_date(stat_month || '01'), 'yyyyQ') =
                                             to_char(to_date(P_DATE) - interval '1 years', 'yyyyQ')
                                            and stat_month < to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                            then target
                                        else 0 end
                                    + case
                                          when stat_month = to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM') then
                                                  target / b.days *
                                                  to_char(to_date(P_DATE) - interval '1 years', 'dd'):: int
                                          else 0 end) ct_2022_GD02_zxsyj_jd,--去年季度
                                sum(case
                                        when (stat_month between '202201' and to_char(
                                                to_date(P_DATE) - interval '1 years 1 months',
                                                'yyyyMM')) then target
                                        else 0 end
                                    + case
                                          when stat_month = to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM') then
                                                  target / b.days *
                                                  to_char(to_date(P_DATE) - interval '1 years', 'dd'):: int
                                          else 0 end) ct_2022_GD02_zxsyj_qnlj,--去年累计
                                sum(case
                                        when (stat_month between '202201' and to_char(
                                                to_date(P_DATE) - interval '1 years',
                                                'yyyy12')) then target
                                        else 0 end)
                                                      ct_2022_GD02_zxsyj_qnzn,--去年全年
                                department_name
                         from (
                                  SELECT unnest(string_to_array((
                                                                                                                                                        COALESCE(january_value, '0') || ',' ||
                                                                                                                                                        COALESCE(february_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(march_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(april_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(may_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(june_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(july_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(august_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(september_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(october_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(november_value, '0') ||
                                                                                                                                                        ',' ||
                                                                                                                                                        COALESCE(december_value, '0')),
                                                                ',')) ::numeric(38, 10)                      target,
                                         '2022' ||
                                         unnest(string_to_array('01,02,03,04,05,06,07,08,09,10,11,12', ',')) stat_month,
                                         all_year_value,
                                         department_name
                                  FROM upload.upload_ct_2022_integral_head_t a
                                  where department_name != '总计'
                              ) a,
                              last_year_month_days b
                         where stat_month between '202201' and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                         group by department_name
                     )
                select level1,
                       channel,
                       channel_name,
                       level2,
                       level3,
                       sum(last_year_month_achievement)     last_year_month_achievement,
                       sum(last_year_all_month_achievement) last_year_all_month_achievement,
                       sum(last_year_quarter_achievement)   last_year_quarter_achievement,
                       sum(last_year_achievement)           last_year_achievement,
                       sum(last_all_year_achievement)       last_all_year_achievement,
                       P_DATE
                from (
                         select b.node_desc3                                                                  level1,
                                case
                                    when integrate_channel = 'GD03' then 'GD03'
                                    else case
                                             when coalesce(integrate_channel2, 'GD01') = 'GD04' then 'GD04'
                                             else case
                                                      when integrate_channel = 'GD01' then 'GD01'
                                                      when integrate_channel = 'GD02' then 'GD02' end end end channel,
                                case
                                    when integrate_channel = 'GD03' then '工程渠道'
                                    else case
                                             when coalesce(integrate_channel2, 'GD01') = 'GD04' then '特惠品'
                                             else case
                                                      when integrate_channel = 'GD01' then '大零售渠道'
                                                      when integrate_channel = 'GD02'
                                                          then '整装头部' end end end                             channel_name,
                                b.node_desc4                                                                  level2,
                                b.node_desc5                                                                  level3,
                                sum(case
                                        when channel != 'GD02' and
                                             replace(calmonth, '-', '') =
                                             to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                            and calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                            then coalesce(ambperformance, 0)
                                        else 0 end) /
                                10000                                                                         last_year_month_achievement,
                                sum(case
                                        when channel != 'GD02' and
                                             replace(calmonth, '-', '') =
                                             to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                            then coalesce(ambperformance, 0)
                                        else 0 end) /
                                10000                                                                         last_year_all_month_achievement,
                                sum(case
                                        when channel != 'GD02' and
                                             calquarter = to_char(to_date(P_DATE) - interval '1 years', 'yyyyQ')
                                            and calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                            then coalesce(ambperformance, 0)
                                        else 0 end) /
                                10000                                                                         last_year_quarter_achievement,
                                sum(case
                                        when channel != 'GD02' and
                                             replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                            and calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                            then coalesce(ambperformance, 0)
                                        else 0 end) /
                                10000                                                                         last_year_achievement,
                                sum(case
                                        when channel != 'GD02' and
                                             replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                            then coalesce(ambperformance, 0)
                                        else 0 end) /
                                10000                                                                         last_all_year_achievement
                         from dm.dm_ambv2_zzt_xssr_t a
                                  left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
                         where b.node_name4 in ('H04230685', 'H04230672', 'H04230655', 'H04230600')
                           and b.node_desc4 != b.node_desc5
                           and channel != 'GD01'
                           and replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyy12')
                         group by level1, level2, level3, channel, channel_name
--大零售去年逻辑：
                         union all
                         select level1,
                                channel,
                                channel_name,
                                level2,
                                level3,
                                sum(last_year_month_achievement) - b.ct_2022_GD02_zxsyj_qntq     last_year_month_achievement,
                                sum(last_year_all_month_achievement) - b.ct_2022_GD02_zxsyj_qnzy last_year_all_month_achievement,
                                sum(last_year_quarter_achievement) - b.ct_2022_GD02_zxsyj_jd     last_year_quarter_achievement,
                                sum(last_year_achievement) - b.ct_2022_GD02_zxsyj_qnlj           last_year_achievement,
                                sum(last_all_year_achievement) - b.ct_2022_GD02_zxsyj_qnzn       last_all_year_achievement
                         from (select b.node_desc3 level1,
                                      'GD01'       channel,
                                      '大零售渠道'      channel_name,
                                      b.node_desc4 level2,
                                      b.node_desc5 level3,
                                      sum(case
                                              when channel != 'GD02' and
                                                   replace(calmonth, '-', '') =
                                                   to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                                  and
                                                   calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                                  then coalesce(ambperformance, 0)
                                              else 0 end) /
                                      10000        last_year_month_achievement,
                                      sum(case
                                              when channel != 'GD02' and
                                                   replace(calmonth, '-', '') =
                                                   to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                                  then coalesce(ambperformance, 0)
                                              else 0 end) /
                                      10000        last_year_all_month_achievement,
                                      sum(case
                                              when channel != 'GD02' and
                                                   calquarter = to_char(to_date(P_DATE) - interval '1 years', 'yyyyQ')
                                                  and
                                                   calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                                  then coalesce(ambperformance, 0)
                                              else 0 end) /
                                      10000        last_year_quarter_achievement,
                                      sum(case
                                              when channel != 'GD02' and
                                                   replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                                  and
                                                   calday <= to_char(to_date(P_DATE) - interval '1 years', 'yyyyMMdd')
                                                  then coalesce(ambperformance, 0)
                                              else 0 end) /
                                      10000        last_year_achievement,
                                      sum(case
                                              when channel != 'GD02' and
                                                   replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                                                  then coalesce(ambperformance, 0)
                                              else 0 end) /
                                      10000        last_all_year_achievement
                               from dm.dm_ambv2_zzt_xssr_t a
                                        left join dm.dm_rpt_sales_group_t b on a.sales_grp = b.node_name9
                               where b.node_name4 in ('H04230685', 'H04230672', 'H04230655', 'H04230600')
                                 and b.node_desc4 != b.node_desc5
                                 and coalesce(zqdzh,'G101') in ('G101', 'G103', 'G105')
                                 and coalesce(integrate_channel2, 'GD01') != 'GD04'
                                 and replace(calmonth, '-', '') between to_char(to_date(P_DATE) - interval '1 years', 'yyyy01') and to_char(to_date(P_DATE) - interval '1 years', 'yyyyMM')
                               group by level1, level2, level3, channel, channel_name
                              ) a
                                  left join ct_2022_GD02 b
                                            on a.level3 = b.department_name
                         group by level1, level2, level3, channel, channel_name, b.ct_2022_GD02_zxsyj_qntq,
                                  b.ct_2022_GD02_zxsyj_qnzy, b.ct_2022_GD02_zxsyj_jd, b.ct_2022_GD02_zxsyj_qnlj,
                                  b.ct_2022_GD02_zxsyj_qnzn
                                  ----整装头部去年
                         union all
                         select b.node_desc3                   level1,
                                'GD02'                         channel,
                                '整装头部'                         channel_name,
                                b.node_desc4                   level2,
                                b.node_desc5                   level3,
                                sum(ct_2022_GD02_zxsyj_qntq)   last_year_month_achievement,
                                sum(ct_2022_GD02_zxsyj_qnzy)   last_year_all_month_achievement,
                                sum(ct_2022_GD02_zxsyj_jd)     last_year_quarter_achievement,
                                sum(a.ct_2022_GD02_zxsyj_qnlj) last_year_achievement,
                                sum(ct_2022_GD02_zxsyj_qnzn)   last_all_year_achievement
                         from ct_2022_GD02 a
                                  left join (select distinct node_desc3, node_desc4, node_desc5
                                             from dm.dm_rpt_sales_group_t
                         ) b
                                            on a.department_name = b.node_desc5
                         group by level1, level2, level3, channel, channel_name) aa
                group by level1, level2, level3, channel, channel_name;
            end loop;
    end ;
$$
