    (SELECT

-- 计算资金成本：若（期初余额+期末余额）为0，则资金成本为0，否则为,[（期初余额+期末余额）/2-2020年12月余额]*4%/12
STOCK_CAPITAL_COST3 AS (
SELECT
    T.BUSINESS_DEPARTMENT									-- 事业部
    ,T.BUSINESS_DEPARTMENT_DESC			                    -- 事业部描述
    ,T.MONTH												-- 日历月
    ,T.COMP_CODE											-- 公司代码
    ,T.FACTORY_CODE											-- 工厂
    ,T.MATERIAL_CODE										-- 物料编码
    ,T.MATERIAL_NAME										-- 物料描述
    ,T.BATCH												-- 批次
    ,T.STOCK_LOCATION										-- 库存地点
    ,T.STOCK_TYPE											-- 库存类别
    ,T.ZDPZGSDQ
    ,T.CLOSING_BALANCE_202012								-- 2020年12月余额
    ,T.CLOSING_BALANCE_LAST_MONTH							-- 期初余额
    ,T.CLOSING_BALANCE										-- 期末余额
/*
    ,CASE WHEN NVL(T.CLOSING_BALANCE_LAST_MONTH_J,0) + NVL(T.CLOSING_BALANCE_J,0) = 0 THEN 0
        ELSE ((NVL(T.CLOSING_BALANCE_LAST_MONTH,0) + NVL(T.CLOSING_BALANCE,0))/2 - T.CLOSING_BALANCE_202012) * 0.04 / 12 
    END                 AS CAPITAL_COST						-- 资金成本	
*/
    ,((NVL(T.CLOSING_BALANCE_LAST_MONTH,0) + NVL(T.CLOSING_BALANCE,0))/2 - T.CLOSING_BALANCE_202012) * 0.04 / 12
                        AS CAPITAL_COST						-- 资金成本
FROM 
    STOCK_CAPITAL_COST2 T
)
,
STOCK_CAPITAL_COST4 AS (
SELECT 
    T.BUSINESS_DEPARTMENT									-- 事业部
    ,T.BUSINESS_DEPARTMENT_DESC			                    -- 事业部描述
    ,T.MONTH												-- 日历月
    ,T.COMP_CODE											-- 公司代码
    ,T.FACTORY_CODE											-- 工厂
    ,T.MATERIAL_CODE										-- 物料编码
    ,T.MATERIAL_NAME										-- 物料描述
    ,T.BATCH												-- 批次
    ,T.STOCK_LOCATION										-- 库存地点
    ,T.STOCK_TYPE											-- 库存类别
    ,T.ZDPZGSDQ
    ,T.CLOSING_BALANCE_202012								-- 2020年12月余额
    ,T.CLOSING_BALANCE_LAST_MONTH							-- 期初余额
    ,T.CLOSING_BALANCE										-- 期末余额
    ,T.CAPITAL_COST											-- 资金成本	
    ,SUM(NVL(T.CAPITAL_COST,0))OVER(PARTITION BY T.BUSINESS_DEPARTMENT,T.BUSINESS_DEPARTMENT_DESC,T.COMP_CODE,T.FACTORY_CODE,T.MATERIAL_CODE,T.MATERIAL_NAME,T.BATCH,T.STOCK_LOCATION,T.STOCK_TYPE,T.ZDPZGSDQ ORDER BY T.MONTH) 
						    AS CAPITAL_COST_SUM				-- 累计资金成本
FROM 
    STOCK_CAPITAL_COST3 T
WHERE T.MONTH <= (to_number('${PERIOD_ID_Y}') - 1)||'12' AND T.MONTH >= (to_number('${PERIOD_ID_Y}') - 1)||'01'

UNION ALL

SELECT 
    T.BUSINESS_DEPARTMENT									-- 事业部
    ,T.BUSINESS_DEPARTMENT_DESC			                    -- 事业部描述
    ,T.MONTH												-- 日历月
    ,T.COMP_CODE											-- 公司代码
    ,T.FACTORY_CODE											-- 工厂
    ,T.MATERIAL_CODE										-- 物料编码
    ,T.MATERIAL_NAME										-- 物料描述
    ,T.BATCH												-- 批次
    ,T.STOCK_LOCATION										-- 库存地点
    ,T.STOCK_TYPE											-- 库存类别
    ,T.ZDPZGSDQ
    ,T.CLOSING_BALANCE_202012								-- 2020年12月余额
    ,T.CLOSING_BALANCE_LAST_MONTH							-- 期初余额
    ,T.CLOSING_BALANCE										-- 期末余额
    ,T.CAPITAL_COST											-- 资金成本	
    ,SUM(NVL(T.CAPITAL_COST,0))OVER(PARTITION BY T.BUSINESS_DEPARTMENT,T.BUSINESS_DEPARTMENT_DESC,T.COMP_CODE,T.FACTORY_CODE,T.MATERIAL_CODE,T.MATERIAL_NAME,T.BATCH,T.STOCK_LOCATION,T.STOCK_TYPE,T.ZDPZGSDQ ORDER BY T.MONTH) 
						    AS CAPITAL_COST_SUM				-- 累计资金成本
FROM 
    STOCK_CAPITAL_COST3 T
WHERE T.MONTH <= '${PERIOD_ID_M}' AND T.MONTH >= '${PERIOD_ID_Y}'||'01'
)

    ,T.MATERIAL___T	 	AS MATERIAL_NAME					-- 物料描述
    ,T.BATCH		 	AS BATCH							-- 批次
    ,T.STOR_LOC		 	AS STOCK_LOCATION					-- 库存地点
    ,T.STOCKCAT		 	AS STOCK_TYPE						-- 库存类别
    ,T.ZDPZGSDQ                                             -- 子公司大区
FROM 
    DM.DM_FIN_STOCK_DETAIL_ACCAGE_T_2023 T
WHERE (T.STOCKCAT IS NULL OR T.STOCKCAT <> 'K')
AND T.CALMONTH <= '${PERIOD_ID_M}' AND T.CALMONTH >= (to_number('${PERIOD_ID_Y}') - 1)||'01'
) D
ON NVL(T.BUSINESS_DEPARTMENT,'NULL') = NVL(D.BUSINESS_DEPARTMENT,'NULL')
AND NVL(T.MONTH,'NULL') = NVL(D.MONTH,'NULL')
AND NVL(T.COMP_CODE,'NULL') = NVL(D.COMP_CODE,'NULL')
AND NVL(T.FACTORY_CODE,'NULL') = NVL(D.FACTORY_CODE,'NULL')
AND NVL(T.MATERIAL_CODE,'NULL') = NVL(D.MATERIAL_CODE,'NULL')
AND NVL(T.BATCH,'NULL') = NVL(D.BATCH,'NULL')
AND NVL(T.STOCK_LOCATION,'NULL') = NVL(D.STOCK_LOCATION,'NULL')
AND NVL(T.STOCK_TYPE,'NULL') = NVL(D.STOCK_TYPE,'NULL')
*/
GROUP BY
    T.BUSINESS_DEPARTMENT									-- 事业部
    ,T.MONTH												-- 日历月
    ,T.COMP_CODE											-- 公司代码
    ,T.FACTORY_CODE											-- 工厂
    ,T.MATERIAL_CODE										-- 物料编码
    ,T.MATERIAL_NAME										-- 物料描述
    ,T.BATCH												-- 批次
    ,T.STOCK_LOCATION										-- 库存地点
    ,T.STOCK_TYPE											-- 库存类别
    ,NVL(T.CLOSING_BALANCE_202012,0)						-- 2020年12月余额
    ,NVL(T.CLOSING_BALANCE_LAST_MONTH,0)					-- 期初余额
    ,NVL(T.CLOSING_BALANCE,0)								-- 期末余额
    ,NVL(T.CAPITAL_COST,0)									-- 资金成本	
    ,NVL(T.CAPITAL_COST_SUM,0)				                -- 累计资金成本
	,T.BUSINESS_DEPARTMENT_DESC			                    -- 事业部描述
	,CASE WHEN T.COMP_CODE IS NOT NULL AND U.COMP_TYPE = '非全资' THEN U.SALES_GRP

	              WHEN T.BUSINESS_DEPARTMENT = '11000003' THEN CASE WHEN T.ZDPZGSDQ = '33000014' THEN 'RR5'

	                                                                WHEN T.ZDPZGSDQ = '33000016'  THEN 'P76'
	                                                                WHEN T.ZDPZGSDQ = '33000019'  THEN 'R6D'
	                                                               ELSE ''
	                                                            END
	              WHEN T.BUSINESS_DEPARTMENT = '11000002' THEN CASE WHEN U.SALES_GRP IS NULL THEN 'R2T' ELSE U.SALES_GRP END
	              WHEN T.BUSINESS_DEPARTMENT = '11000001' THEN CASE WHEN U.SALES_GRP IS NULL THEN 'R40' ELSE U.SALES_GRP END
	            ELSE 'NULL'
	           END
