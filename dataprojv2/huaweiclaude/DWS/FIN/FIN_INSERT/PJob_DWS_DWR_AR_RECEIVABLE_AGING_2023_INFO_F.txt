DELETE FROM DWRFIN.DWR_AR_RECEIVABLE_AGING_2023_INFO_F 
        WHERE query_date in (SELECT CASE WHEN substr(current_date,1,7) = substr('${PERIOD_ID_D}',1,7) AND SUBSTR('${PERIOD_ID_D}',9,2) IN('01') 
                      THEN SUBSTR(last_day(add_months('${PERIOD_ID_D}',-1)),1,10) END     
                            -- 当月月结2-5日传上月月底
                            UNION ALL
                            SELECT '${PERIOD_ID_D}'     -- 昨天
                            UNION ALL
                            SELECT substr(to_date('${PERIOD_ID_D}','yyyy-MM-dd') - 1,1,10)  -- 前天
     )
AND query_year = '${PERIOD_ID_Y}'
;
