TRUNCATE TABLE DM.DM_DMT_DEALER_JCH_T;

WITH T1 AS
(
SELECT CALMONTH AS STAT_MONTH
      ,CALDAY AS STAT_DATE
      ,MATERIAL AS MATERIAL_NUM
      ,MATERIAL___T AS MATERIAL_NAME
      ,SOLD_TO AS DEALER_CODE
      ,SOLD_TO___T AS DEALER_NAME
      ,B.DIMENSION
      ,C.LVE2_CODE AS CATEGORY_NAME
      ,SUM(ZXSMJ) AS JH_AREA --进货面积
      ,SUM(COALESCE(ZXSYJ_KJY,0)+ZXSYJ_ZD) AS JH_AMOUNT --进货金额
FROM DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T A
LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T B
ON A.MATERIAL = B.MATERIAL_NUM
LEFT JOIN DM.DM_RPT_PROD_LEVEL_T C
ON B.PRODUCT_LEVEL3_CODE = C.PROD_CODE3 
WHERE CALDAY <= CURRENT_DATE - 1
AND CALDAY >= '20220101'
AND CALDAY != '00000000'--调整数部分
AND A.MATERIAL IS NOT NULL
AND B.MATL_GROUP_CODE = '1000010'
AND B.EXTERNAL_MATL_GROUP_CODE = '10010'
GROUP BY CALMONTH
      ,CALDAY
      ,MATERIAL
      ,MATERIAL___T
      ,SOLD_TO
      ,SOLD_TO___T
      ,B.DIMENSION
      ,C.LVE2_CODE
),T2 AS(
SELECT 
 REPLACE(SUBSTR(DOC_DATE,1,7),'-','') AS STAT_MONTH
,REPLACE(SUBSTR(DOC_DATE,1,10),'-','') AS STAT_DATE
,MATERIAL_NUM
,MATERIAL_NAME
,CATEGORY_NAME
,DIMENSION
,LPAD(DEALER_CODE,10,'0') AS DEALER_CODE
,DEALER_NAME
,SUM(LENGTH_1 * WIDTH * MATERIAL_QTY) / 1000000 CH_AREA --出货面积
,SUM(COALESCE(CAST(TOTAL_SALES_AMT AS NUMERIC(38,10)),0)) AS CH_AMOUNT --出货金额
FROM DM.DM_B1_SALES_OUTGOODS_T
WHERE DOC_DATE <= CURRENT_DATE - 1
AND SUBSTR(DOC_DATE,1,10) >= '2022-01-01'
AND CUSTOMER_TYPE = 'CT'
AND MATL_GROUP_CODE = '1000010'
AND EXTERNAL_MATL_GROUP_CODE = '10010'
GROUP BY 
 REPLACE(SUBSTR(DOC_DATE,1,7),'-','')
,REPLACE(SUBSTR(DOC_DATE,1,10),'-','')
,MATERIAL_NUM
,MATERIAL_NAME
,CATEGORY_NAME
,DIMENSION
,LPAD(DEALER_CODE,10,'0')
,DEALER_NAME
),T3 AS(
SELECT STAT_MONTH
      ,STAT_DATE
      ,MATERIAL_NUM
      ,MATERIAL_NAME
      ,DEALER_CODE
      ,DEALER_NAME
      ,DIMENSION
      ,CATEGORY_NAME
      ,JH_AREA --进货面积
      ,JH_AMOUNT --进货金额
	  ,0 AS CH_AREA --出货面积
	  ,0 AS CH_AMOUNT --出货金额
  FROM T1
 
 UNION ALL
 
SELECT STAT_MONTH
      ,STAT_DATE
      ,MATERIAL_NUM
      ,MATERIAL_NAME
      ,DEALER_CODE
      ,DEALER_NAME
      ,DIMENSION
      ,CATEGORY_NAME
      ,0 AS JH_AREA --进货面积
      ,0 AS JH_AMOUNT --进货金额
	  ,CH_AREA --出货面积
	  ,CH_AMOUNT --出货金额
  FROM T2
)
INSERT INTO DM.DM_DMT_DEALER_JCH_T
SELECT STAT_MONTH
      ,STAT_DATE
      ,MATERIAL_NUM
      ,MATERIAL_NAME
      ,DEALER_CODE
      ,DEALER_NAME
      ,DIMENSION
      ,CATEGORY_NAME
      ,SUM(JH_AREA) AS JH_AREA --进货面积
      ,SUM(JH_AMOUNT) AS JH_AMOUNT --进货金额
	  ,SUM(CH_AREA) AS CH_AREA --出货面积
	  ,SUM(CH_AMOUNT) AS CH_AMOUNT --出货金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,CASE WHEN B.CUSTOMER_CODE IS NOT NULL THEN 1
	        ELSE 0
			 END AS IS_B1_CUSTOMER --是否上B1经销商
  FROM T3
  LEFT JOIN (SELECT DISTINCT LPAD(CUSTOMER_CODE,10,'0') AS CUSTOMER_CODE
               FROM UPLOAD.UPLOAD_MD_B1_CUSTOMER_T
			  WHERE CUSTOMER_TYPE = '瓷砖' 
			 ) B
	ON DEALER_CODE = B.CUSTOMER_CODE
 GROUP BY
       STAT_MONTH
      ,STAT_DATE
      ,MATERIAL_NUM
      ,MATERIAL_NAME
      ,DEALER_CODE
      ,DEALER_NAME
      ,DIMENSION
      ,CATEGORY_NAME
	  ,CASE WHEN B.CUSTOMER_CODE IS NOT NULL THEN 1
	        ELSE 0
			 END
;
