TRUNCATE TABLE DM.DM_B1_PUR_INTOWH_PRE_T;

INSERT INTO DM.DM_B1_PUR_INTOWH_PRE_T
SELECT A.DELIVERY_DATE
	  ,A.DELIVERY_BILL_NUM
	  ,A.RECEIPT_DATE
	  ,A.RECEIPT_BILL_NUM
	  ,A.SUPPLIER_CODE
	  ,A.SUPPLIER_NAME
	  ,A.MATERIAL_CODE
	  ,A.MATERIAL_DESC
	  ,A.QUANTITY
	  ,A.PRICE
	  ,A.LINE_ACCOUNT
	  ,A.WAREHOUSE_NUM
	  ,A.BATCH_NUM
	  ,A.LINE_NUM
	  ,A.DELIVERY_STATUS
	  ,A.SAP_DELIVERY_NUM
	  ,A.SAP_DELIVERY_LINE_NUM
	  ,A.MATERIAL_NUM
	  ,A.MATERIAL_NAME
	  ,A.ACTUAL_DELIVERY_QTY
	  ,A.INV_LOCATION_NAME
	  ,A.SAP_DELIVERY_DATE
	  ,A.CUST_CODE
	  ,A.CUST_NAME
	  ,A.RECEIVING_ADDR
	  ,COALESCE(A.CUSTOMER_CODE,A.CUST_CODE) AS CUSTOMER_CODE
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,coalesce(C.abbreviation,D.abbreviation) AS cust_byname
  FROM DM.DM_B1_PUR_INTOWH_T A
  LEFT JOIN UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T B
    ON A.MATERIAL_NUM = B.MATERIAL_NUM
  LEFT JOIN UPLOAD.UPLOAD_B1_CUSTOMER_DATA_T C
    ON A.CUST_CODE = LPAD(C.CUST_CODE,10,'0')
  LEFT JOIN UPLOAD.UPLOAD_B1_CUSTOMER_DATA_T D
    ON A.CUST_CODE = D.CUST_CODE  
 WHERE B.MATERIAL_TYPE NOT IN('非东鹏','其它')
   AND (C.CUST_CODE IS NOT NULL OR D.CUST_CODE IS NOT NULL)
   AND A.ACTUAL_DELIVERY_QTY > 0
;
