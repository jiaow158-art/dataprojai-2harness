,TC AS(

			ELSE '零售'

LEFT JOIN DM.DM_RPT_BIC_PZAMBCPBM_T AS B ON A."MATERIAL" = B."/BIC/ZAMBCPBM"

AND (SUBSTR("STOR_LOC",1,2) = 'BE' OR "STOR_LOC" IN('BJ01','YX03'))
AND B."/BIC/ZCPSYB" = '11000001'
AND "STOCKCAT" = 'K'
GROUP BY "CALMONTH"
        ,CASE WHEN C."/BIC/ZYL01" IN('G102','G115') THEN '工程'
	          WHEN C."/BIC/ZYL01" = 'G103' THEN '大包'
			  WHEN C."/BIC/ZYL01" = 'G105' THEN '设计师'
			  ELSE '零售'
