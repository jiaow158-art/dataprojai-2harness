DELETE FROM dm.dm_rpt_wm_cxc_day_sum
where stat_date = replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','');

INSERT INTO DM.DM_RPT_WM_CXC_DAY_SUM
with batch_area  as(
	select batch_start,
		   max(case when batch_start='0Y' then '华盛昌基地'
		            when factory_type <>'外协' or factory_type is null then belong_area_name else 'OEM' end) as belong_area_name
	from DWRDIM.DWR_DIM_PRODUCT_AREA_D
	group by batch_start  
)
, mat as (
-- 产区
	select t1.material_num,t1.product_level2_name  as cat,t1.dimension,
        t1.def_prod_channel  as prod_channel ,
        t1.width * t1.length_1/1000/1000 as area
  from dwimd.dwi_md_data_material_general_t t1 
  where matl_group_code ='1000010'
	and external_matl_group_code in('10010','10080')
    and t1.product_level2_code  like '1%'
    and t1.product_level2_code  not in('1011','1012','1013','1014','1016')
	and matl_type_code='FERT'
),
stock_chg_det as(
-- 出入库
select  t1.VOUCHER_POST_DATE,
		t1.material_num,  -- 关联品类、渠道
		t1.batch_num ,    -- 关联基地
		t2.cat,
		t2.dimension,
		t2.prod_channel,
		sum(case when MOVE_TYPE IN(
						 '601','602','643','644','653','654') then (case when DR_CR_ID='S_SHKZG' then -QUANTITY else  QUANTITY end) end*t2.area) as sales_stock_out_area,
		sum(case when MOVE_TYPE IN(
						 '101','102','911','912','913','914','131','132','531','532' ) then (case when DR_CR_ID='S_SHKZG' then QUANTITY else  -QUANTITY end*t2.area) end) as stock_in_area
from  dwrotd.DWR_WM_PRODSTOCK_OUT_IN_F t1
join mat t2 
on t1.material_num  = t2.material_num 
where VOUCHER_POST_DATE>= date_trunc('year',date '${PERIOD_ID_D}') 
and  VOUCHER_POST_DATE< date'${PERIOD_ID_D}'+1
	and MOVE_TYPE IN(
					 '601','602','643','644','653','654',  -- 销售出库
					 '911','912','913','914','131','132','531','532','101','102'   -- 生产入库&采购入库
					 )
    group by t1.VOUCHER_POST_DATE ,t1.material_num,t1.batch_num,t2.cat,t2.dimension,t2.prod_channel
 ),
 stock_chg_sum as
 (
 select coalesce(t2.belong_area_name,t3.belong_area_name,'广东产区') as belong_area_name,
        t1.cat,t1.dimension,t1.prod_channel,
        sum(sales_stock_out_area) sales_stock_out_area_year,
        sum(stock_in_area) stock_in_area_year,
        sum(case when VOUCHER_POST_DATE>=date_trunc('month',date'${PERIOD_ID_D}') then sales_stock_out_area else 0 end) sales_stock_out_area_month,
        sum(case when VOUCHER_POST_DATE>=date_trunc('month',date'${PERIOD_ID_D}') then stock_in_area else 0 end ) stock_in_area_month,
        sum(case when VOUCHER_POST_DATE>=date'${PERIOD_ID_D}' then sales_stock_out_area else 0 end) sales_stock_out_area_day,
        sum(case when VOUCHER_POST_DATE>=date'${PERIOD_ID_D}' then stock_in_area else 0 end ) stock_in_area_day
 from stock_chg_det t1
 left join batch_area t2  on substr(t1.batch_num,1,3) = t2.batch_start
 left join batch_area t3  on substr(t1.batch_num,1,2) = t3.batch_start
 group by coalesce(t2.belong_area_name,t3.belong_area_name,'广东产区'),cat,dimension,prod_channel
 ),
 stock_det as (
 select coalesce(t3.belong_area_name,t4.belong_area_name,'广东产区') as belong_area_name,cat,dimension,prod_channel,
        sum(case when stock_flg='本月库存' then stock_qty end*t2.area ) stock_area_month_end,
        sum(case when stock_flg='月初库存' then stock_qty end*t2.area ) stock_area_month_start,
        sum(case when stock_flg='年初库存' then stock_qty end*t2.area ) stock_area_year_start
 from 
 (
 -- 月初库存、年初库存、本月库存
select batch_num,material_num,sum(stock_qty) as stock_qty, '本月库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = date_format(date_trunc('month',date'${PERIOD_ID_D}'),'%Y%m')
group by  batch_num,material_num
 union all 
select batch_num,material_num,sum(stock_qty) as stock_qty,'月初库存' stock_flg from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = date_format(date_trunc('month',date'${PERIOD_ID_D}')-1,'%Y%m')
group by  batch_num,material_num
union all
 select batch_num,material_num,sum(stock_qty) as stock_qty,'年初库存' stock_flg  from dwrotd.dwr_wm_all_type_stock_month_t
 where  month_date = date_format(date_trunc('year',date'${PERIOD_ID_D}')-1,'%Y%m')
group by  batch_num,material_num
) t1
join mat t2 on t1.material_num  = t2.material_num 
left join batch_area t3  on substr(t1.batch_num,1,3) = t3.batch_start
left join batch_area t4  on substr(t1.batch_num,1,2) = t4.batch_start
group by coalesce(t3.belong_area_name,t4.belong_area_name,'广东产区'),cat,dimension,prod_channel
),
mid as (
select  belong_area_name,cat,dimension,prod_channel,
         stock_area_month_end,
        stock_area_month_start,
        stock_area_year_start,
        0.0 as sales_stock_out_area_year,0.0 as stock_in_area_year,
        0.0 as sales_stock_out_area_month,0.0 as stock_in_area_month,
        0.0 as sales_stock_out_area_day,0.0 as stock_in_area_day
        from stock_det
union all 
select  belong_area_name,cat,dimension,prod_channel,
        0,0,0,
        sales_stock_out_area_year,stock_in_area_year,
        sales_stock_out_area_month,stock_in_area_month,
        sales_stock_out_area_day,stock_in_area_day
        from stock_chg_sum
        )
 select replace(substr(to_date('${PERIOD_ID_D}'),1,10),'-','') as stat_date,
        belong_area_name,
		cat,
		dimension,
		prod_channel,
        sum(stock_area_month_end) as stock_area_month_end,
        sum(stock_area_month_start) as stock_area_month_start,
        sum(stock_area_year_start) as stock_area_year_start,
        sum(sales_stock_out_area_year) as sales_stock_out_area_year, 
		sum(stock_in_area_year) as stock_in_area_year,
        sum(sales_stock_out_area_month) as sales_stock_out_area_month, 
		sum(stock_in_area_month) as stock_in_area_month,
        sum(sales_stock_out_area_day) as sales_stock_out_area_day, 
		sum(stock_in_area_day) as stock_in_area_day,
		current_timestamp() as dw_last_update_date
 from mid 
 group by belong_area_name,cat,dimension,prod_channel;
