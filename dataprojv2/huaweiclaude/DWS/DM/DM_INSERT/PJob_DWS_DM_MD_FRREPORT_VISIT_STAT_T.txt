TRUNCATE TABLE DM.DM_MD_FRREPORT_VISIT_STAT_T;

WITH T1 AS(
SELECT REPLACE(SPLIT_PART(A.USERNAME,'(',2),')','') AS EMP_NUM
      ,CASE WHEN SUBSTR(A.TNAME,1,3) = 'PC端' THEN REPLACE(REPLACE(REPLACE(SPLIT_PART(A.TNAME,'/',LENGTH(A.TNAME) - LENGTH(REPLACE(A.TNAME,'/',''))+1),'.cptx',''),'.frm',''),'.cpt','')
            ELSE COALESCE(REPLACE(REPLACE(REPLACE(SPLIT_PART(C.NEW_REPORT_PATH,'/',LENGTH(C.NEW_REPORT_PATH) - LENGTH(REPLACE(C.NEW_REPORT_PATH,'/',''))+1),'.cptx',''),'.frm',''),'.cpt','')
                         ,SPLIT_PART(A.DISPLAYNAME,'/',1+LENGTH(A.DISPLAYNAME)-LENGTH(REPLACE(A.DISPLAYNAME,'/',''))))
			 END AS REPORT_NAME
	  ,B.LEV2_DEPT_NAME AS DEPT_NAME
	  ,A.BEGIN_TIME
	  ,COALESCE(C.NEW_REPORT_PATH,A.TNAME) AS TNAME
	  ,A.TYPE AS VISIT_TYPE
	  ,CASE WHEN A.TYPE = '0' THEN '分页预览'
	        WHEN A.TYPE = '1' THEN '在线分析'
			WHEN A.TYPE = '2' THEN '填报预览'
			WHEN A.TYPE = '3' THEN '决策报表预览'
			WHEN A.TYPE = '6' THEN '新填报'
			WHEN A.TYPE = '7' THEN '新引擎预览'
			WHEN A.TYPE = '10' THEN 'Excel 分页导出'
			WHEN A.TYPE = '11' THEN 'Excel 原样导出'
			WHEN A.TYPE = '12' THEN 'Excel 整页导出'
			WHEN A.TYPE = '13' THEN 'Excel 分页分 sheet 导出'
			WHEN A.TYPE = '14' THEN 'PDF 导出'
			WHEN A.TYPE = '15' THEN 'Word 导出'
			WHEN A.TYPE = '20' THEN 'PNG 图片导出'
			WHEN A.TYPE = '33' THEN '零客户端打印'
             END AS VISIT_SUB_TYPE_NAME	
      ,CASE WHEN A.TYPE IN('2','6') THEN '填报'
            ELSE '预览'
             END AS VISIT_TYPE_NAME
  FROM DWIMD.DWI_MD_FR_FINE_RECORD_EXECUTE_T A
  LEFT JOIN DM.DM_WF_EMP_INFO_T B
    ON REPLACE(SPLIT_PART(A.USERNAME,'(',2),')','') = B.EMP_NUM
  LEFT JOIN UPLOAD.UPLOAD_MD_FR_REPORT_PATH_RELEVANT_T C
    ON C.OLD_REPORT_PATH = A.TNAME
 WHERE A.DISPLAYNAME IS NOT NULL
   AND SUBSTR(A.DISPLAYNAME,1,2) <> '测试'
)
,T2 AS(
SELECT SUBSTR(BEGIN_TIME,1,7) AS STAT_MONTH
      ,REPORT_NAME
	  ,TNAME
	  ,VISIT_TYPE
	  ,VISIT_SUB_TYPE_NAME
	  ,VISIT_TYPE_NAME
	  ,SUBSTR(BEGIN_TIME - INTERVAL '1 MONTH',1,7) AS LAST_MONTH
	  ,COUNT(1) AS VISIT_CNT
  FROM T1
 GROUP BY
       SUBSTR(BEGIN_TIME,1,7)
      ,REPORT_NAME
	  ,TNAME
	  ,VISIT_TYPE
	  ,VISIT_SUB_TYPE_NAME
	  ,VISIT_TYPE_NAME 
	  ,SUBSTR(BEGIN_TIME - INTERVAL '1 MONTH',1,7)
)
,T3 AS(SELECT  STAT_MONTH
              ,LAST_MONTH
              ,REPORT_NAME
			  ,TNAME
			  ,SUM(VISIT_CNT) AS VISIT_CNT			  
         FROM T2
		GROUP BY
		       STAT_MONTH
              ,LAST_MONTH			   
              ,REPORT_NAME
			  ,TNAME 
	   )
INSERT INTO DM.DM_MD_FRREPORT_VISIT_STAT_T
(
       STAT_MONTH
      ,REPORT_NAME
	  ,TNAME
	  ,VISIT_TYPE
	  ,VISIT_SUB_TYPE_NAME
	  ,VISIT_TYPE_NAME
	  ,VISIT_CNT
	  ,L1M_VISIT_CNT
      ,DW_LAST_UPDATE_DATE
)
SELECT A.STAT_MONTH
      ,CASE WHEN SUBSTR(A.REPORT_NAME,6,1) IN('-','_') THEN A.REPORT_NAME
 	        ELSE CONCAT(CONCAT(SUBSTR(A.REPORT_NAME,1,5),'-'),SUBSTR(A.REPORT_NAME,6,LENGTH(A.REPORT_NAME)-5))
			 END AS REPORT_NAME
	  ,A.TNAME
	  ,C.VISIT_TYPE
	  ,C.VISIT_SUB_TYPE_NAME
	  ,C.VISIT_TYPE_NAME
	  ,A.VISIT_CNT
      ,COALESCE(B.VISIT_CNT,0) AS L1M_VISIT_CNT
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T3 A
  LEFT JOIN T3 B		   
    ON A.TNAME = B.TNAME
   AND A.REPORT_NAME = B.REPORT_NAME
   AND A.LAST_MONTH = B.STAT_MONTH
  LEFT JOIN(SELECT TNAME
                  ,REPORT_NAME
	              ,VISIT_TYPE
	              ,VISIT_SUB_TYPE_NAME
	              ,VISIT_TYPE_NAME				  
				  ,ROW_NUMBER() OVER(PARTITION BY TNAME,REPORT_NAME ORDER BY BEGIN_TIME DESC) AS RN
              FROM T1
           ) C
	ON A.TNAME = C.TNAME
   AND A.REPORT_NAME = C.REPORT_NAME
   AND C.RN = 1
 WHERE SUBSTR(A.TNAME,1,6) <> '财经共享平台'  
;
