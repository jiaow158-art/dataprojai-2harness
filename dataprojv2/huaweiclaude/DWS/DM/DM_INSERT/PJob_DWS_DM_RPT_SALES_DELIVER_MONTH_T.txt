delete from dm.dm_rpt_sales_deliver_month_t  where stat_month= REPLACE(SUBSTRING(date '${PERIOD_ID_D}' + interval '-1' day,1,7),'-',''); -- 昨天所在的月份

insert into dm.dm_rpt_sales_deliver_month_t 
with tmp as(
select stat_month,
    case when sub_name='运营部' then '工程市场中心' else sub_name end sub_name,
    case when sub_grp_name like '%运营部%' then '战略工程' else sub_grp_name end  sub_grp_name , 
    sum(zxsmj_dy) as stockout_area,
    sum(yc_area) as predict_area,
    sum(zxsyj_dy) as stockout_amount,
    sum(yc_amount) as predict_amount,
    0.0 as delivernosend_area_ls ,
	   0.0 as delivernosend_area_gc ,
	   0.0 as delivernosend_area_db ,
	   0.0 as delivernosend_area_sjs ,
	   0.0 as nodeliver_area_ondue_ls ,
	   0.0 as nodeliver_area_ondue_gc ,
	   0.0 as nodeliver_area_ondue_db ,
	   0.0 as nodeliver_area_ondue_sjs ,
	   0.0 as nodeliver_area_overdue_ls ,
	   0.0 as nodeliver_area_overdue_gc ,
	   0.0 as nodeliver_area_overdue_db ,
	   0.0 as nodeliver_area_overdue_sjs
from  DM.DM_RPT_SALE_FORECAST_2023_T
where flag='X'
and stat_month=REPLACE(SUBSTRING(date '${PERIOD_ID_D}' + interval '-1' day,1,7),'-','')
group by stat_month,
		 sub_name ,
		 case when sub_grp_name like '%运营部%' then '战略工程' else sub_grp_name end 
union all 
select stat_month,
       case when dept_lv4_code  = '工程市场中心' then '工程市场中心'
            when dept_lv4_code  = '直营经营管理板块' then '子公司'
            when dept_lv4_code  = '区域运营中心' then '运营中心'
            when dept_lv3_code  = '国际营销' then '国贸'
            when dept_lv3_code  = '梦之家事业部' then '梦之家'
                 else dept_lv3_name end,
       case when dept_lv3_code = '国际营销' then '国贸'
             when dept_lv4_code = '家装大包部' then dept_lv4_code
	         when dept_lv3_code = '梦之家事业部' then '梦之家'
	         when dept_lv5_code = 'Z031' then '幕墙'
       		 when dept_lv5_code = '国内工程' then '战略工程'
	         when dept_lv4_code in('直营经营管理板块','区域运营中心') then dept_lv5_code
	         else dept_lv5_name
			 end ,
			 0 as stockout_area,0 as predict_area,
			 0,0,
	   sum(case when channel_name ='零售' then delivernosend_area else 0  end)/10000 as delivernosend_area_ls ,
	   sum(case when channel_name ='工程' then delivernosend_area else 0 end)/10000 as delivernosend_area_gc ,
	   sum(case when channel_name ='大包' then delivernosend_area else 0 end)/10000 as delivernosend_area_db ,
	   sum(case when channel_name ='设计师' then delivernosend_area else 0 end)/10000 as delivernosend_area_sjs ,
	   sum(case when channel_name ='零售' and expect_date_aps>current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_ondue_ls ,
	   sum(case when channel_name ='工程' and expect_date_aps>current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_ondue_gc ,
	   sum(case when channel_name ='大包' and expect_date_aps>current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_ondue_db ,
	   sum(case when channel_name ='设计师' and expect_date_aps>current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_ondue_sjs,
	   sum(case when channel_name ='零售' and expect_date_aps<=current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_overdue_ls ,
	   sum(case when channel_name ='工程' and expect_date_aps<=current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_overdue_gc ,
	   sum(case when channel_name ='大包' and expect_date_aps<=current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_overdue_db ,
	   sum(case when channel_name ='设计师' and expect_date_aps<=current_date -1 then nodeliver_area_aps else 0 end)/10000 as nodeliver_area_overdue_sjs
from  DM.DM_RPT_SALES_DELIVER_T
where stat_month =REPLACE(SUBSTRING(date '${PERIOD_ID_D}' + interval '-1' day,1,7),'-','')
group by stat_month,
       case when dept_lv4_code  = '工程市场中心' then '工程市场中心'
            when dept_lv4_code  = '直营经营管理板块' then '子公司'
            when dept_lv4_code  = '区域运营中心' then '运营中心'
            when dept_lv3_code  = '国际营销' then '国贸'
            when dept_lv3_code  = '梦之家事业部' then '梦之家'
                 else dept_lv3_name end,
       case when dept_lv3_code = '国际营销' then '国贸'
             when dept_lv4_code = '家装大包部' then dept_lv4_code
	         when dept_lv3_code = '梦之家事业部' then '梦之家'
	         when dept_lv5_code = 'Z031' then '幕墙'
       		 when dept_lv5_code = '国内工程' then '战略工程'
	         when dept_lv4_code in('直营经营管理板块','区域运营中心') then dept_lv5_code
	         else dept_lv5_name
			 end
union all
select stat_month,'工程市场中心','战略专属仓',
    0.0  as stockout_area,
    0.0 as predict_area,
    0.0  as stockout_amount,
    0.0  as predict_amount,
    0.0 as delivernosend_area_ls ,
	  sum(available_inventory_area)/10000 as delivernosend_area_gc ,
	   0.0 as delivernosend_area_db ,
	   0.0 as delivernosend_area_sjs ,
	   0.0 as nodeliver_area_ondue_ls ,
	   0.0 as nodeliver_area_ondue_gc ,
	   0.0 as nodeliver_area_ondue_db ,
	   0.0 as nodeliver_area_ondue_sjs ,
	   0.0 as nodeliver_area_overdue_ls ,
	   0.0 as nodeliver_area_overdue_gc ,
	   0.0 as nodeliver_area_overdue_db ,
	   0.0 as nodeliver_area_overdue_sjs
from DM.DM_WM_STRATEGY_USER_WAREHOUSE_T
where stat_month = REPLACE(SUBSTRING(date '${PERIOD_ID_D}' + interval '-1' day,1,7),'-','')
group by stat_month
			 )
select stat_month,
		sub_name,
		sub_grp_name,
		sum(stockout_area) as stockout_area,
		sum(predict_area) as predict_area,
		sum(stockout_amount) as stockout_amount,
		sum(predict_amount) as predict_amount,
		sum(delivernosend_area_ls) as  delivernosend_area_ls,
		sum(delivernosend_area_gc) as delivernosend_area_gc ,
		sum(delivernosend_area_db) as delivernosend_area_db ,
		sum(delivernosend_area_sjs) as delivernosend_area_sjs ,
		sum(nodeliver_area_ondue_ls) as nodeliver_area_ondue_ls ,
		sum(nodeliver_area_ondue_gc) as nodeliver_area_ondue_gc ,
		sum(nodeliver_area_ondue_db) as nodeliver_area_ondue_db ,
		sum(nodeliver_area_ondue_sjs) as nodeliver_area_ondue_sjs ,
		sum(nodeliver_area_overdue_ls) as nodeliver_area_overdue_ls ,
		sum(nodeliver_area_overdue_gc) as nodeliver_area_overdue_gc ,
		sum(nodeliver_area_overdue_db) as nodeliver_area_overdue_db ,
		sum(nodeliver_area_overdue_sjs) as nodeliver_area_overdue_sjs,
		current_timestamp() as dw_last_update_date
 from tmp 
 group by stat_month,
		  sub_name,
		  sub_grp_name
