truncate table DM.DM_FIN_GL_PROFIT_T;
insert into DM.DM_FIN_GL_PROFIT_T
select
calmonth 
,sales_grp 
,node_NAME5  as sales_are                --销售地区
,sum(zysr)    revenue                      --主营业务收入
,sum(gsjsr)   revenue_inter                --主营业务收入-公司间
,sum(zysrqt)  revenue_other                --主营业务收入-其他
,(revenue + revenue_inter + revenue_other)  as revenue_s  --主营业务收入-合计
,sum(zycb)    cost_of_sales                --主营业务成本
,sum(gsjcb)   cost_of_sales_inter          --主营业务成本-公司间
,sum(zycbqt)  cost_of_sales_other          --主营业务成本-其他
,sum(actual_cost) as   cost_of_actual_cost --实际成本
,SUM(FD)  as FD_s                          --FD
,(cost_of_actual_cost + cost_of_sales_other) as cost_of_sales_s    --主营业务成本-合计
,sum(syjf) as  used_points                 --使用积分
,sum(ytjf) as  Accrued_points              --预提积分
,(revenue_s  - cost_of_sales_s  - used_points - Accrued_points + fd_s) as profit --毛利
,case when (revenue_s - used_points - Accrued_points +fd_s = 0) then 0 else profit/(revenue_s - used_points - Accrued_points + fd_S) end as gross_profit_margin --毛利率
,sum(qtsr) as other_operating_income       --其他业务收入-订单             
,sum(gsjqtsr) other_operating_income_inter --其他业务收入-公司间
,sum(qtsrqt)  other_operating_income_other --其他业务收入-其他
,(other_operating_income + other_operating_income_inter + other_operating_income_other) as other_operating_income_s --其他业务收入合计
,sum(qtcb)   other_operating_expense       --其他业务成本-订单
,sum(gsjqtcb)  other_operating_expense_inter       --其他业务成本-公司间
,sum(qtcbqt)   other_operating_expense_other       --其他业务成本-其他
,(other_operating_expense + other_operating_expense_other )   as other_operating_expense_s --其他业务成本合计
,(other_operating_income_s - other_operating_expense_s) as other_operating_profit         --其他业务利润
--,sum(sjjfj)  sales_tax                --税金及附加
,case when  profit > 0 then profit*0.13*0.2  else 0 end as sales_tax   --税金及附加
,sum(zcjzss) assets_impairment_loss   --资产减值损失
,sum(xyjzss) Credit_impairment_loss   --信用减值损失
,sum(jzss)   impairment_loss          --资产/信用减值损失
,sum(qtsy)   investment_income        --其他收益/投资收益
,sum(zcsy)   Asset_profit_and_loss    --资产处理损益
,sum(sds)    income_tax               --所得税
,sum(yywsr)  non_operating_income     --营业外收入
,sum(yywzc)  non_operating_expense    --营业外支出
,sum(xsfy) selling_expenses           --销售费用
,sum(glfy) general_and_administrative_expenses--管理费用
,sum(yffy) research_and_development_costs     --研发费用
,sum(cwfy) finaneial_expense                  --财务费用
,sum(lx)   Financial_interest                 --财务利息
,(selling_expenses  + general_and_administrative_expenses + research_and_development_costs + finaneial_expense ) as expense_s --费用合计
,(profit - sales_tax + other_operating_profit - expense_s - impairment_loss -  investment_income - Asset_profit_and_loss) as profit_from_operation --营业利润
,(profit_from_operation - non_operating_income - non_operating_expense ) as profit_before_tax --利润总额
,(profit_before_tax - income_tax) as net_profit--净利润
,case when (revenue_s - used_points - Accrued_points = 0) then 0 else selling_expenses/(revenue_s - used_points - Accrued_points) end as selling_expenses_margin --销售费用率%
,case when (revenue_s - used_points - Accrued_points = 0) then 0 else general_and_administrative_expenses/(revenue_s - used_points - Accrued_points) end as general_and_administrative_expenses_margin--管理费用率%
,case when (revenue_s - used_points - Accrued_points = 0) then 0 else research_and_development_costs/(revenue_s - used_points - Accrued_points) end as research_and_development_costs_margin --研发费用率%
,case when (revenue_s - used_points - Accrued_points = 0) then 0 else finaneial_expense/(revenue_s - used_points - Accrued_points) end as finaneial_expense_margin--财务费用率%
,case when (revenue_s - used_points - Accrued_points = 0) then 0 else profit_from_operation/(revenue_s - used_points - Accrued_points) end as profit_margin--利润率%
,current_timestamp() as dw_last_update_date -- 数据同步时间
from 
(select 
calmonth 
,sales_grp 
,sum(nvl(zzyywsr_gn,0)  + nvl(zyywsr_wm,0) + nvl(zhwjzjgtz,0) + nvl(zzxcyf,0) + nvl(zyywsr_zyck,0) + nvl(zyywsr_jtj,0) ) zysr --主营业务收入
,sum(nvl(zyywsr_gsj,0) ) GSJSR    --主营业务收入-公司间
,sum(nvl(zzyywsr_qt,0) ) ZYSRQT   --主营业务收入-其他 
,sum(nvl(zyywcb,0) + nvl(zyywcb_dsf,0) + nvl(zzyywcb_clcbcy,0) + nvl(zzyywcb_cbcbcy,0) + nvl(zzyywcb_cpjjss,0) + nvl(zyywcb_yf,0) + nvl(zyywcb_zxf,0) ) ZYCB --主营业务成本
,SUM(nvl(zyywcb_gsj,0)) GSJCB     --主营业务成本-公司间
,SUM(nvl(zzyywsr_qt,0)) ZYCBQT    --主营业务成本-其他
,SUM(nvl(zsyjf,0)) SYJF           --使用积分
,SUM(nvl(zytjf_sg,0) + nvl(zytjf_xt,0)) YTJF  --预提积分
,sum(case when t4.external_matl_group_code in ('10050') then t6.price /1.13 *  zxsmj + nvl(zcbtzz,0)
                  when t4.matl_group_code in('3000010','4000010') then t6.price1 /1.13 *  zxssl + nvl(zcbtzz,0)
                  when t4.matl_group_code  = '1000010' then nvl(zyywcb,0) + nvl(zcbtzz,0) 
                  when t4.external_matl_group_code in('10030','10080') then nvl(zyywcb,0) + nvl(zcbtzz,0) 
                  when t4.external_matl_group_code in('10020','10021','10022','10060','10090') and plant in ('1200','1220','1260','1280','1300','1310','1320','1350','1360','1380','1500'
,'1510','1520','1560','1580','1600','1610','1620','1650','1660','1680','1700','1710','1720','1750','1780','1800','1820','1850','1860','1880','1900','1920','1960','1980')
                  then nvl(zyywcb,0)
                  when t4.external_matl_group_code in('10020','10021','10022','10060','10090')  then nvl(zyywsr_gsj,0)
                  else nvl(zyywcb,0)* 1.18 + nvl(zcbtzz,0) * 1.18
                  end
) actual_cost                      --实际成本（跨事业部）
,sum(nvl(zfdje,0) ) FD              --FD
,0 as qtsr
,0 as GSJQTSR
,0 as QTSRQT
,0 as QTCB
,0 as GSJQTCB
,0 as QTCBQT
,0 as sjjfj
,0 as zcjzss
,0 as xyjzss
,0 as jzss
,0 as qtsy
,0 as zcsy
,0 as sds
,0 as yywsr
,0 as yywzc
,0 as xsfy
,0 as glfy
,0 as yffy
,0 as cwfy
,0 as lx
FROM DM.DM_ZZT_XSSR_T t1 
left join dwimd.dwi_md_data_material_general_t t4 on  t1.material = t4.material_num
left join dwifin.DWI_PRC_ZTAMB_COST_T t6 on t6.matnr = t4.material_num  and t6.datef <= t1.calday and t6.dated >= t1.calday  and   zsyb_code = '01'
left join DM.DM_RPT_PROD_LEVEL_T t3 on coalesce(t4.product_level3_code,t4.product_level2_code,'其他杂类'  )  = coalesce(t3.prod_code3,t3.lve4_code,'其他杂类') 
group by  (calmonth ,sales_grp )

select

,sales_grp

,0 as ZYCB

,0 as SYJF
,0 as YTJF
,0 as actual_cost
,0 as FD
,sum(nvl(zqtywsr_jgf_dd,0)  + nvl(zqtywsr_ddy,0) + nvl(zzyywsr_pdsr,0) + nvl(zqtywsr_jtj_ddy,0) + nvl(zzyywsr_jgf_fdd,0)  ) qtsr --其他业务收入-订单
,sum(nvl(zqtywsr_gsj,0) ) GSJQTSR   --其他业务收入-公司间
,sum(nvl(zzyywsr_qt,0) ) QTSRQT     --其他业务收入-其他
,sum(nvl(zqtywzc_jgf_dd,0) + nvl(zqtywzc_jgf_fdd,0) + nvl(zqtywzc_ddy,0) + nvl(zqtywzc_jtj,0) ) QTCB   --其他业务成本-订单

,0 as sjjfj

,sum(case when gl_account between '0056010100' and '0056019900' then zje else 0 end ) as yywzc --营业外支出
