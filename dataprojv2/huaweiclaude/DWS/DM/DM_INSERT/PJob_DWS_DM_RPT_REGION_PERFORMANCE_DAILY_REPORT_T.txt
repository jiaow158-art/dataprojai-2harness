                       on a.三级 = b.营销部
         where 三级 like '%运营中心%'
         group by a.三级
         union all
         select a.三级,
                row_number()
                over (order by case nvl(sum(三级月目标), 0) when 0 then 0 else sum(月业绩) / sum(三级月目标) end desc) 三级排名
         from (select sum(月业绩) 月业绩, 三级 from 业绩 WHERE 五级 IS NOT NULL group by 三级) a
                  join 三级月目标 b
                       on a.三级 = b.营销部
         where 三级 like '%子公司%'
         group by a.三级
     ),
     渠道排名 as (
         select a.四级,
                row_number()
                over (partition by 渠道 order by case nvl(sum(四级月目标), 0) when 0 then 0 else sum(月业绩) / sum(四级月目标) end desc) 排名
         from (select sum(月业绩) 月业绩, 四级, 三级, 渠道 from 业绩 where 五级 is not null group by 四级, 三级, 渠道) a
                  join 四级月目标 b
                       on a.四级 = b.营销部
         where 三级 like '%运营中心%'
         group by a.四级, 渠道
         union all
         select a.四级,
                row_number()
                over (partition by 渠道 order by case nvl(sum(四级月目标), 0) when 0 then 0 else sum(月业绩) / sum(四级月目标) end desc) 排名
         from (select sum(月业绩) 月业绩, 四级, 三级, 渠道 from 业绩 where 五级 is not null group by 四级, 三级, 渠道) a
                  join 四级月目标 b
                       on a.四级 = b.营销部
         where 三级 like '%子公司%'
         group by a.四级, 渠道
     ),
     季目标 as (
         select 营销部     四级
              , 线组      五级
              , sum(目标) 季目标
         from 目标
         where 1 = 1
           and EXTRACT(quarter FROM 月份 || '-01') = EXTRACT(quarter FROM current_date - 1)
           and left(月份
                   , 4) = to_char(current_date - 1
                   , 'yyyy')
         group by 营销部, 线组
     ),
     年目标 as (
         select 营销部     四级
              , 线组      五级
              , sum(目标) 年目标
         from 目标
         where 1 = 1
           and 月份 between to_char(current_date - 1
             , 'yyyy-01')
             and to_char(current_date - 1
                 , 'yyyy-mm')
         group by 营销部, 线组
     )
select 一级,
       a.二级,
       a.三级,
       a.渠道,
       a.四级,
       a.五级,
       日业绩,
       coalesce(月业绩, 0)                         月业绩,
       coalesce(毛利额, 0)                         毛利额,
       coalesce(不含税销售收入, 0)                     不含税销售收入,
       去年同期业绩,
       coalesce(历史季业绩, 0)                       季业绩,
       季目标,
       coalesce(历史年业绩, 0)                       年业绩,
       年累计毛利额                                   年累计毛利额,
       年累计不含税销售收入,
       去年业绩,
       年目标,
       a.六级编码,
       a.六级,
       s.排名,
       t.三级排名,
       z.顺序,
       to_char(current_date - 1, 'yyyyMMdd') AS CALDAY,
       CURRENT_TIMESTAMP()                   AS dw_last_update_date
from 转换线组 a
         left join 业绩 b on a.三级 = b.三级 and a.四级 = b.四级 and a.五级 = b.五级 and a.渠道 = b.渠道 and a.六级编码 = b.六级编码
         left join 季目标 q on a.四级 = q.四级 and
                            a.五级 = q.五级
         left join 年目标 r on a.四级 = r.四级 and
                            a.五级 = r.五级
         left join 渠道排名 s on a.四级 = s.四级
         left join 三级排名 t on a.三级 = t.三级
         left join upload.upload_ctseq z on a.三级 = z.ct运营中心
     -- left join 新毛利 zz on a.二级=zz.二级 and a.三级 = zz.三级 and a.五级 = zz.五级 and a.四级 = zz.四级 and a.六级编码 = zz.customer
where 1 = 1
  AND a.五级 is not null
order by a.二级编码, z.顺序, a.三级编码,
         (case a.渠道 when '零售' then 1 when '工程' then 2 when '大包' then 3 else 4 end), a.五级;
