WITH T_ZCJE AS

	"CO_AREA",/*成本控制域*/

	WHEN "GL_ACCOUNT" = '0016030900' THEN 'ZA_CQZC'
	WHEN "GL_ACCOUNT" IN ('0019010100','0019010200','0019010300','0019010400','0019019900') THEN 'ZA_CQDTFY'

	WHEN "GL_ACCOUNT" IN ('0018010100','0018010200','0018010300','0018019900','0018020100','0018020200','0018020300',
						'0018029900') THEN 'ZA_WXZC'
	END AS "ZCJE_TYPE",
	SUM("HHSL") AS ZA_ZCJE
	FROM
	DM.DM_RPT_BIC_AZMFI_O0072_T
	WHERE
	"CO_AREA" = 'DP00'
	AND "CALMONTH2" NOT IN ('00')
	AND nvl("ANBWA",'SNULL') NOT IN ('Z01','Z02')   /*2023.04.14 guzhengfeng 增加对字段"ANBWA"的空值判断*/
	AND "CALMONTH" <= '${start_date}'
	GROUP BY
	"COMP_CODE",
	"ANLN1",
	"ANLN2",
	"ANBWA",
	"AKTIV",
	"CALMONTH",
	"GL_ACCOUNT",
	"COSTCENTER",
	"CO_AREA",
	"WBS_ELEMT",/*WBS元素*/
	CASE WHEN "GL_ACCOUNT" = '0016030100' THEN 'ZA_ZJGC'
	WHEN "GL_ACCOUNT" = '0016030900' THEN 'ZA_CQZC'
	WHEN "GL_ACCOUNT" IN ('0019010100','0019010200','0019010300','0019010400','0019019900') THEN 'ZA_CQDTFY'
	WHEN "GL_ACCOUNT" IN ('0060510900','0060510200','0060510100','0060510300') THEN 'ZA_ZCSYF'

(
	SELECT 
	"COMP_CODE",
	"VENDOR",
	"CALMONTH",
	"GL_ACCOUNT",
	"AC_DOC_NR",
	"DCINDIC",
	"WBS_ELEMT",/*WBS元素*/
	SUM(-1 * ZA_GYSXF_H - ZA_GYSXF_S) * 0.04 / 12 * 6 AS ZJCB_GYSXF ,
	SUM(ZA_GYSXF_H) AS ZJCB_GYSXF_H,--供应商现付贷方
	SUM(ZA_GYSXF_S) AS ZJCB_GYSXF_S--供应商现付借方
	FROM
	(
		SELECT --供应商现付-贷方
		"COMP_CODE",
		"VENDOR",
		"CALMONTH",
		"GL_ACCOUNT",
		"AC_DOC_NR",
		"DCINDIC",
		"WBS_ELEMT",/*WBS元素*/
		SUM(ZA_GYSXF) AS ZA_GYSXF_H,
		0 AS ZA_GYSXF_S
		FROM T_GYSXF
		WHERE 
		"CALMONTH" = '${start_date}'
		AND "DCINDIC" = 'H'
		GROUP BY
		"COMP_CODE",
		"VENDOR",
		"CALMONTH",
		"GL_ACCOUNT",
		"AC_DOC_NR",
		"DCINDIC",
		"WBS_ELEMT"/*WBS元素*/
	UNION ALL
		SELECT --供应商现付-借方
		"COMP_CODE",
		"VENDOR",
		"CALMONTH",
		"GL_ACCOUNT",
		"AC_DOC_NR",
		"DCINDIC",
		"WBS_ELEMT",/*WBS元素*/
		0 AS ZA_GYSXF_H,
		SUM(ZA_GYSXF) AS ZA_GYSXF_S
		FROM T_GYSXF
		WHERE 
		"CALMONTH" = '${start_date}'
		AND "DCINDIC" = 'S'
		GROUP BY
		"COMP_CODE",
		"VENDOR",
		"CALMONTH",
		"GL_ACCOUNT",
		"AC_DOC_NR",
		"DCINDIC",
		"WBS_ELEMT"/*WBS元素*/
	)
	GROUP BY
	"COMP_CODE",
	"VENDOR",
	"CALMONTH",
	"GL_ACCOUNT",
	"AC_DOC_NR",
	"DCINDIC",
	"WBS_ELEMT"/*WBS元素*/
),
