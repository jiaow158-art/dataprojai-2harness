DELETE FROM DM.DM_FIN_BUDGET_CONDITION_T;
INSERT INTO DM.DM_FIN_BUDGET_CONDITION_T
with TMP_A as (
	select 
		A."COSTCENTER" AS "成本中心",
		A."CO_AREA" AS "成本控制域",
		A."GL_ACCOUNT" AS "总账科目",
		A."CHRT_ACCTS" AS "会计科目表",
		A."FUNC_AREA" AS "功能范围",
		A."CALMONTH" AS "日历年/月",
		A."FISCPER" AS "会计年度",
		A."FISCPER3" AS "记账期间",
		A."/BIC/ZDATAINDX" AS "数据标识",
		SUM("SALES") AS "预算金额"
	FROM dm.DM_RPT_BIC_AZFISGD022_T A 
		where 1=1  
		and A."GL_ACCOUNT" not in ('0065040000','0065010000')
		and "FUNC_AREA" IN ('5501','5504','5502','5503','4110','6100','6200')
	--	and "CALMONTH" = '202207' --改成查询月
		and "FISCVARNT" = 'K4'
		and "/BIC/ZDATAINDX" = 'Y'
		group by 
		A."COSTCENTER" 
		,A."CO_AREA" 
		,A."GL_ACCOUNT" 
		,A."CHRT_ACCTS"
		,A."FUNC_AREA" 
		,A."CALMONTH" 
		,A."FISCPER" 
		,A."FISCVARNT" 
		,A."FISCPER3" 
		,A."/BIC/ZDATAINDX" 	
),
TMP_B AS(
	select
		T_A."日历年/月"  月份
		,'3' "层次"
		,T2."cost_center_name_3" "层次名称"
		, sum("预算金额" ) "预算金额" 
		, SUM(sum("预算金额")) over(partition by T2.cost_center_name_3,substr(T_A."日历年/月" ,1,4) order by T_A."日历年/月" ) "预算金额年至今" 
		,0 "实际费用"
		,0 "实际费用年至今"
		,0 "滚动预算"
		,0 "滚动预算年至今"
	from TMP_A T_A 
	join DWIMD.DWI_MD_COST_CENTER_INFO_T T2 
	on T2.zcost_center = concat(T_A."成本控制域",T_A."成本中心")
	where T2.cost_center_name_2 ='卫浴事业部'
	group by T2.cost_center_name_3,
			 T_A."日历年/月"
	union all 
	select 
		T_A."日历年/月"  月份
		 ,'4' "层次"
		,T2."cost_center_name_4" "层次名称"
		, sum("预算金额" ) "预算金额" 
		, SUM(sum("预算金额")) over(partition by T2.cost_center_name_4,substr(T_A."日历年/月" ,1,4) order by T_A."日历年/月" ) "预算金额年至今" 
		,0 "实际费用"
		,0"实际费用年至今"
		,0 "滚动预算"
		,0 "滚动预算年至今"
	from TMP_A T_A 
	join DWIMD.DWI_MD_COST_CENTER_INFO_T T2 
	on T2.zcost_center = concat(T_A."成本控制域",T_A."成本中心")
	where T2.cost_center_name_2 ='卫浴事业部'  
	and "cost_center_name_3" = '制造中心' 
	group by T2."cost_center_name_4",
			 T_A."日历年/月"
	union all
	select 
		 calmonth "月份"
		,'3' "层次"
		,cost_center_name3   "层次名称"
		,0  "预算金额" 
		,0  "预算金额年至今" 
		,sum(cost_current_month ) "实际费用"
		,sum(cost_year) "实际费用年至今"
		,0 "滚动预算"
		,0 "滚动预算年至今"
	from DM.DM_FIN_COST_BATHROOM_T  
	where COST_CENTER_CODE4 <> 'GW36FCD' and new_cost_category is not null 
	and gl_subject_name not in
		  ('CNDP/15010500', 'CNDP/15010100', 'CNDP/15010300', 'CNDP/15010200', 'CNDP/15010600', 'CNDP/15010400',
	   'CNDP/15020500', 'CNDP/15020600', 'CNDP/15020300', 'CNDP/15020200', 'CNDP/15020400', 'CNDP/18019900',
	   'CNDP/18029900', 'CNDP/16030100', 'CNDP/19010100')
	 and gl_subject_name not like 'CNDP/1%'
	group by calmonth,
			 cost_center_name3
	union all
	select 
		calmonth 月份
		, '4' "层次"
		,cost_center_name4   "层次名称"
		,0  "预算金额" 
		,0  "预算金额年至今" 
		,sum(cost_current_month) "实际费用"
		,sum(cost_year) "实际费用年至今"
		,0 "滚动预算"
		,0 "滚动预算年至今"
	from DM.DM_FIN_COST_BATHROOM_T 
	where cost_center_name3='制造中心' and new_cost_category is not null
	group by cost_center_name4,
			 calmonth
	union all
	select 
		 months
		, level_code 层次
		,budget_dept "层次名称"
		,0  "预算金额" 
		,0  "预算金额年至今" 
		,0  "实际费用"
		,0  "实际费用年至今"
		,sum(budget_amt ) "滚动预算"
		,sum(sum(budget_amt)) over(partition by level_code,substr(months,1,4),budget_dept order by months ) "滚动预算年至今"
	from DM.DM_RPT_ROLL_BUDGET_T 
	group by level_code,
			budget_dept,
			months
)
select 
	"月份" as months  -- 月份
	,"层次" as hierarchys  -- 层次
	,"层次名称" as hierarchys_name  -- 层次名称
	,sum("预算金额") budget_amt   -- 预算金额
	,sum("预算金额年至今") budget_amt_ytd  -- 预算金额年至今
	,sum("实际费用")  actual_cost   -- 实际费用
	,sum("实际费用年至今") actual_cost_ytd   -- 实际费用年至今
	,sum("滚动预算") rolling_budget_amt    -- 滚动预算
	,sum("滚动预算年至今") rolling_budget_amt_ytd  -- 滚动预算年至今
	,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE  -- 数据同步时间
from TMP_B
group by "层次"
		,"层次名称",
		"月份"
