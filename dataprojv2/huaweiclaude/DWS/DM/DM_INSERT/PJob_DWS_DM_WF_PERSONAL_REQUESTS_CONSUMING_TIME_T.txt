DELETE FROM DM.DM_WF_PERSONAL_REQUESTS_CONSUMING_TIME_T
WHERE STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
;

WITH T1 AS(
SELECT 
STAT_MONTH 
,DEPT_NAME 
,LASTNAME 
,SAPID  
,EMP_POST_NAME  
,CURRENT_NODE_TYPE 
,WF_TYPE_NAME 
,WF_NAME 
,NODE_NAME 
,SUM(GZ_HOUR) GZ_HOUR
,SUM(NODE_COUNT) NODE_COUNT
,SUM(CASE WHEN STAT_MONTH < 202209 AND GZ_HOUR > 6 THEN 1  
          WHEN STAT_MONTH >= 202209 AND STAT_MONTH < 202303 AND GZ_HOUR > 4 THEN 1
          WHEN STAT_MONTH >= 202303 AND GZ_HOUR > 3.5 THEN 1		  
          ELSE 0 END ) TIMEOUT_TIME
,LEV1_DEPT_NAME
,LEV2_DEPT_NAME
,LEV3_DEPT_NAME
,LEV4_DEPT_NAME
,LEV5_DEPT_NAME
,LEV6_DEPT_NAME
,LEV7_DEPT_NAME
,LEV8_DEPT_NAME
,LEV2_DEPT_NAME||LEV3_DEPT_NAME||LEV4_DEPT_NAME||LEV5_DEPT_NAME||LEV6_DEPT_NAME||LEV7_DEPT_NAME LEV_DEPT_NAME
FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_PRE_T
WHERE 1 = 1
AND STAT_MONTH = SUBSTR(REPLACE(RECEIVE_DATE,'-',''),1,6)
AND STAT_MONTH >= TO_CHAR(TO_DATE('${PERIOD_ID_D}','yyyy-MM-dd') + INTERVAL '-1 MONTH','YYYYMM')
GROUP BY STAT_MONTH
,DEPT_NAME 
,LASTNAME
,SAPID  
,STAT_MONTH 
,EMP_POST_NAME
,CURRENT_NODE_TYPE
,WF_TYPE_NAME
,WF_NAME 
,NODE_NAME 
,LEV1_DEPT_NAME
,LEV2_DEPT_NAME
,LEV3_DEPT_NAME
,LEV4_DEPT_NAME
,LEV5_DEPT_NAME
,LEV6_DEPT_NAME
,LEV7_DEPT_NAME
,LEV8_DEPT_NAME
,LEV2_DEPT_NAME||LEV3_DEPT_NAME||LEV4_DEPT_NAME||LEV5_DEPT_NAME||LEV6_DEPT_NAME||LEV7_DEPT_NAME
)
INSERT INTO DM.DM_WF_PERSONAL_REQUESTS_CONSUMING_TIME_T
(
     STAT_MONTH
    ,DEPT_NAME
    ,EMP_NAME
    ,EMP_ID
    ,EMP_POST_NAME
    ,CURRENT_NODE_TYPE
    ,WF_TYPE_NAME
    ,WF_NAME
    ,NODE_NAME
    ,GZ_HOUR
    ,NODE_COUNT
    ,TIMEOUT_TIME
    ,LAST_GZ_HOUR
    ,LAST_NODE_COUNT
    ,LEV1_DEPT_NAME
    ,LEV2_DEPT_NAME
    ,LEV3_DEPT_NAME
    ,LEV4_DEPT_NAME
    ,LEV5_DEPT_NAME
    ,LEV6_DEPT_NAME
    ,LEV7_DEPT_NAME
    ,LEV8_DEPT_NAME
    ,LEV_DEPT_NAME
    ,DW_LAST_UPDATE_DATE
)
SELECT 
        A.STAT_MONTH 
       ,A.DEPT_NAME 
       ,A.LASTNAME AS EMP_NAME
       ,A.SAPID AS EMP_ID  
       ,A.EMP_POST_NAME  
       ,A.CURRENT_NODE_TYPE 
       ,A.WF_TYPE_NAME
       ,A.WF_NAME 
       ,A.NODE_NAME 
       ,A.GZ_HOUR
       ,A.NODE_COUNT
       ,A.TIMEOUT_TIME
       ,B.GZ_HOUR AS LAST_GZ_HOUR
       ,B.NODE_COUNT AS LAST_NODE_COUNT
       ,A.LEV1_DEPT_NAME
       ,A.LEV2_DEPT_NAME
       ,A.LEV3_DEPT_NAME
       ,A.LEV4_DEPT_NAME
       ,A.LEV5_DEPT_NAME
       ,A.LEV6_DEPT_NAME
       ,A.LEV7_DEPT_NAME
       ,A.LEV8_DEPT_NAME
       ,A.LEV_DEPT_NAME
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T1 AS A
  LEFT JOIN T1 AS B
    ON A.STAT_MONTH = TO_CHAR(TO_DATE(SUBSTR(B.STAT_MONTH,1,4)||'-'||SUBSTR(B.STAT_MONTH,5,2) ||'-01') + INTERVAL '1 MONTH','YYYYMM')
   AND A.LASTNAME = B.LASTNAME
   AND A.SAPID = B.SAPID 
   AND A.EMP_POST_NAME = B.EMP_POST_NAME
   AND A.CURRENT_NODE_TYPE = B.CURRENT_NODE_TYPE
   AND A.WF_TYPE_NAME = B.WF_TYPE_NAME
   AND A.WF_NAME = B.WF_NAME
   AND A.NODE_NAME = B.NODE_NAME
 WHERE A.STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
;
