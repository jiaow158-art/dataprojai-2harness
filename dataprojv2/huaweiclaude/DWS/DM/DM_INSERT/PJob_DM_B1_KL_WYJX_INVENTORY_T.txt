/*
DROP TABLE IF EXISTS DM.DM_B1_KL_WYJX_INVENTORY_T_TMP01;

CREATE TABLE DM.DM_B1_KL_WYJX_INVENTORY_T_TMP01
AS */   /* 2023/04/04 guzhengfeng DROP TABLE~/CREATE TABLE~ 修改为 TRUNCATE/INSERT INTO */
TRUNCATE TABLE DM.DM_B1_KL_WYJX_INVENTORY_T_TMP01;
INSERT INTO DM.DM_B1_KL_WYJX_INVENTORY_T_TMP01
SELECT
       A.STAT_MONTH --日期
	  ,A.CUST_BYNAME --客户别名
	  ,A.DEALER_CODE --经销商编码
	  ,A.MATERIAL_NUM --物料编码
	  --,SUM(A."本期入库") AS BQ_INTOWARE_QUANTITY
	  ,SUM(CASE WHEN A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY <= 0 AND A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY >= 0
				THEN A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY
	            WHEN A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY >= 0 AND A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY > 0
	            THEN A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY + A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY
				WHEN A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY < 0 AND A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY > 0
				THEN A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY
				WHEN A.PUR_INTO_QUANTITY - A.PUR_OUT_QUANTITY <= 0 AND A.OTHER_INTO_QUANTITY - A.OTHER_OUT_QUANTITY <= 0
				THEN 0
				 END) AS BQ_INTOWARE_QUANTITY
  FROM (SELECT SUBSTR(A."日期",1,7) AS STAT_MONTH --日期
	          ,A."客户别名" AS CUST_BYNAME --客户别名
	          ,D.CUST_CODE AS DEALER_CODE --经销商编码
	          ,A."物料编码" AS MATERIAL_NUM --物料编码
			  ,SUM(A."期间采购收货数量") AS PUR_INTO_QUANTITY
			  ,SUM(A."期间采购退货数量") AS PUR_OUT_QUANTITY
			  ,SUM(A."期间其他入库数量") AS OTHER_INTO_QUANTITY
			  ,SUM(A."期间其他出库数量") AS OTHER_OUT_QUANTITY
          FROM DM.DM_WM_B1_INVENTORY_T_D A
		  LEFT JOIN UPLOAD.UPLOAD_B1_CUSTOMER_DATA_T D
            ON A."客户别名" = D.ABBREVIATION
           AND D.IS_MAIN = '是'	
         GROUP BY
                 SUBSTR(A."日期",1,7)
	            ,A."客户别名"
	            ,D.CUST_CODE
	            ,A."物料编码"
       ) A
  GROUP BY  A.STAT_MONTH --日期
	  ,A.CUST_BYNAME --客户别名
	  ,A.DEALER_CODE --经销商编码
	  ,A.MATERIAL_NUM --物料编码  
;
/*
DROP TABLE IF EXISTS DM.DM_B1_KL_WYJX_INVENTORY_T_TMP02;

CREATE TABLE DM.DM_B1_KL_WYJX_INVENTORY_T_TMP02
AS */   /* 2023/04/04 guzhengfeng DROP TABLE~/CREATE TABLE~ 修改为 TRUNCATE/INSERT INTO */
TRUNCATE TABLE DM.DM_B1_KL_WYJX_INVENTORY_T_TMP02;
INSERT INTO DM.DM_B1_KL_WYJX_INVENTORY_T_TMP02
SELECT A.STAT_MONTH
      ,B.CUST_BYNAME
	  ,B.DEALER_CODE
	  ,B.MATERIAL_NUM
	  ,COALESCE(C.BQ_INTOWARE_QUANTITY,0) AS BQ_INTOWARE_QUANTITY
  FROM(SELECT DISTINCT SUBSTR(DAY_DT,1,7) AS STAT_MONTH 
         FROM DM.DM_DIM_DATE_D
      ) A
  LEFT JOIN(SELECT DISTINCT
                   A."客户别名" AS CUST_BYNAME --客户别名
	              ,D.CUST_CODE AS DEALER_CODE --经销商编码
	              ,A."物料编码" AS MATERIAL_NUM --物料编码
			  FROM DM.DM_WM_B1_INVENTORY_T_D A	  
              LEFT JOIN (SELECT * FROM UPLOAD.UPLOAD_B1_CUSTOMER_DATA_T
                          WHERE IS_MAIN = '是'
		                ) D
                ON A."客户别名" = D.ABBREVIATION 
           ) B
    ON 1 = 1
  LEFT JOIN DM.DM_B1_KL_WYJX_INVENTORY_T_TMP01 C
    ON A.STAT_MONTH = C.STAT_MONTH
   AND B.CUST_BYNAME = C.CUST_BYNAME
   AND B.DEALER_CODE = C.DEALER_CODE
   AND B.MATERIAL_NUM = C.MATERIAL_NUM
;
/*
DROP TABLE IF EXISTS DM.DM_B1_KL_WYJX_INVENTORY_T_TMP03;

CREATE TABLE DM.DM_B1_KL_WYJX_INVENTORY_T_TMP03

      ,SUM(BQ_INTOWARE_QUANTITY) OVER(PARTITION BY A.CUST_BYNAME,A.DEALER_CODE,A.MATERIAL_NUM
	                                  ORDER BY A.STAT_MONTH ASC ROWS BETWEEN 11 PRECEDING AND CURRENT ROW) AS KL_QUANTITY2 --半年到一年本期入库量
	  ,SUM(BQ_INTOWARE_QUANTITY) OVER(PARTITION BY A.CUST_BYNAME,A.DEALER_CODE,A.MATERIAL_NUM
	                                  ORDER BY A.STAT_MONTH ASC ROWS BETWEEN 23 PRECEDING AND CURRENT ROW) AS KL_QUANTITY3 --一年到两年本期入库量								  									  
  FROM DM.DM_B1_KL_WYJX_INVENTORY_T_TMP02 A
;

	   ,IS_SW --是否卫浴物料
       )
WITH TMP AS(
SELECT A."日期",A."客户别名",A."物料编码",A."物料描述",
       SUM(A."期末数量") AS "期末数量",
       SUM(A."期末金额") AS "期末金额"
FROM 
DM.DM_WM_B1_INVENTORY_T_D A
WHERE SUBSTR((CAST(A."日期" AS DATE) + INTERVAL '1 MONTH'),1,7) = SUBSTR((CAST(A."日期" AS DATE) + INTERVAL '1 DAY'),1,7)
   AND SUBSTR(A."日期",1,7) <= SUBSTR('${PERIOD_ID_D}',1,7)
   AND SUBSTR(A."日期",1,7) >= SUBSTR((TO_DATE('${PERIOD_ID_D}','yyyy-MM-dd') + INTERVAL '-1 MONTH'),1,7)
GROUP BY A."日期",A."客户别名",A."物料编码",A."物料描述"
)
SELECT 
       SUBSTR(A."日期",1,7) AS STAT_MONTH --日期
	  ,A."客户别名" AS CUST_BYNAME --客户别名
	  ,D.CUST_CODE AS DEALER_CODE --经销商编码
	  ,D.COMPANY_NAME AS DEALER_NAME --经销商编码
	  ,A."物料编码" AS MATERIAL_NUM --物料编码
	  ,A."物料描述" AS MATERIAL_NAME --物料描述
	  ,F.MATERIAL_TYPE --分类
	  ,F.PRODUCT_CLASS --品类
	  ,F.ANALYZE_PRODUCT_CLASS --分析品类
	  ,CASE WHEN A."期末数量" <= COALESCE(C.KL_QUANTITY1,0)
	        THEN A."期末数量" 
	        WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0)
			THEN COALESCE(C.KL_QUANTITY1,0)
			ELSE 0 END AS KL_QUANTITY1 --半年内库存量	
	  ,CASE WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0) AND A."期末数量" <= COALESCE(C.KL_QUANTITY2,0) 
	        THEN A."期末数量" - COALESCE(C.KL_QUANTITY1,0)
			WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0) AND A."期末数量" > COALESCE(C.KL_QUANTITY2,0)
			THEN COALESCE(C.KL_QUANTITY2,0) - COALESCE(C.KL_QUANTITY1,0)
			ELSE 0 END AS KL_QUANTITY2 --半年到一年库存量	
	  ,CASE WHEN A."期末数量" > COALESCE(C.KL_QUANTITY2,0) AND A."期末数量" <= COALESCE(C.KL_QUANTITY3,0)           	  
	        THEN A."期末数量" - COALESCE(C.KL_QUANTITY2,0)
			WHEN A."期末数量" > COALESCE(C.KL_QUANTITY2,0) AND A."期末数量" > COALESCE(C.KL_QUANTITY3,0)
			THEN COALESCE(C.KL_QUANTITY3,0) - COALESCE(C.KL_QUANTITY2,0)
			ELSE 0 END AS KL_QUANTITY3 --一年到两年库存量	
      ,CASE WHEN A."期末数量" > COALESCE(C.KL_QUANTITY3,0) --AND COALESCE(C.KL_QUANTITY3,0) > COALESCE(C.KL_QUANTITY2,0)
			THEN A."期末数量" - COALESCE(C.KL_QUANTITY3,0)
			ELSE 0 END AS KL_QUANTITY4 --两年以上库存量	
      ,CASE WHEN B.INVENTORY_QUANTITY = 0 THEN 0
	        WHEN A."期末数量" <= COALESCE(C.KL_QUANTITY1,0)
			THEN CAST(A."期末数量"  * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
	        WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0)
			THEN CAST(COALESCE(C.KL_QUANTITY1,0) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			ELSE 0 END AS KL_AMOUNT1 --半年内库存金额	
	  ,CASE WHEN B.INVENTORY_QUANTITY = 0 THEN 0
	        WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0) AND A."期末数量" <= COALESCE(C.KL_QUANTITY2,0) 
	        THEN CAST((A."期末数量" - COALESCE(C.KL_QUANTITY1,0)) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			WHEN A."期末数量" > COALESCE(C.KL_QUANTITY1,0) AND A."期末数量" > COALESCE(C.KL_QUANTITY2,0)
			THEN CAST((COALESCE(C.KL_QUANTITY2,0) - COALESCE(C.KL_QUANTITY1,0)) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			ELSE 0 END AS KL_AMOUNT2 --半年到一年库存金额	
	  ,CASE WHEN B.INVENTORY_QUANTITY = 0 THEN 0
	        WHEN A."期末数量" > COALESCE(C.KL_QUANTITY2,0) AND A."期末数量" <= COALESCE(C.KL_QUANTITY3,0)           	  
	        THEN CAST((A."期末数量" - COALESCE(C.KL_QUANTITY2,0)) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			WHEN A."期末数量" > COALESCE(C.KL_QUANTITY2,0) AND A."期末数量" > COALESCE(C.KL_QUANTITY3,0)
			THEN CAST((COALESCE(C.KL_QUANTITY3,0) - COALESCE(C.KL_QUANTITY2,0)) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			ELSE 0 END AS KL_AMOUNT3 --一年到两年库存金额
      ,CASE WHEN B.INVENTORY_QUANTITY = 0 THEN 0
	        WHEN A."期末数量" > COALESCE(C.KL_QUANTITY3,0) --AND COALESCE(C.KL_QUANTITY3,0) > COALESCE(C.KL_QUANTITY2,0)
	        THEN CAST((A."期末数量" - COALESCE(C.KL_QUANTITY3,0)) * B.INVENTORY_AMOUNT / B.INVENTORY_QUANTITY AS NUMERIC(38,10))
			ELSE 0 END AS KL_AMOUNT4 --两年以上库存金额
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
      ,A."期末数量" AS QM_QUANTITY --期末数量
      ,A."期末金额" AS QM_AMOUNT --期末金额
	  ,E.MATERIAL_ID AS CUR_SAP_MATERIAL_ID --当前物料ID
      ,E.MATERIAL_NAME AS CUR_SAP_MATERIAL_NAME --当前SAP物料名称
      ,E.MATL_TYPE_CODE AS CUR_SAP_MATL_TYPE_CODE --当前SAP物料类型
      ,E.MATL_TYPE_NAME AS CUR_SAP_MATL_TYPE_NAME --当前SAP物料类型名称
      ,E.PRODUCT_GROUP_CODE AS CUR_SAP_PRODUCT_GROUP_CODE --当前SAP产品组
      ,E.PRODUCT_GROUP_NAME AS CUR_SAP_PRODUCT_GROUP_NAME --当前SAP产品组名称
      ,E.PRODUCT_CLASSIFY_CODE AS CUR_SAP_PRODUCT_CLASSIFY_CODE --当前SAP产品分级
      ,E.PRODUCT_CLASSIFY_NAME AS CUR_SAP_PRODUCT_CLASSIFY_NAME --当前SAP产品分级名称
      ,E.PRODUCT_LEVEL_CODE AS CUR_SAP_PRODUCT_LEVEL_CODE --当前SAP层次代码
      ,E.PRODUCT_LEVEL_NAME AS CUR_SAP_PRODUCT_LEVEL_NAME --当前SAP层次名称
      ,E.PRODUCT_LEVEL1_CODE AS CUR_SAP_PRODUCT_LEVEL1_CODE --当前SAP层次1
      ,E.PRODUCT_LEVEL1_NAME AS CUR_SAP_PRODUCT_LEVEL1_NAME --当前SAP层次1名称
      ,E.PRODUCT_LEVEL2_CODE AS CUR_SAP_PRODUCT_LEVEL2_CODE --当前SAP层次2
      ,E.PRODUCT_LEVEL2_NAME AS CUR_SAP_PRODUCT_LEVEL2_NAME --当前SAP层次2名称
      ,E.PRODUCT_LEVEL3_CODE AS CUR_SAP_PRODUCT_LEVEL3_CODE --当前SAP层次3
      ,E.PRODUCT_LEVEL3_NAME AS CUR_SAP_PRODUCT_LEVEL3_NAME --当前SAP层次3名称
      ,E.PRODUCT_LEVEL4_CODE AS CUR_SAP_PRODUCT_LEVEL4_CODE --当前SAP层次4
      ,E.PRODUCT_LEVEL4_NAME AS CUR_SAP_PRODUCT_LEVEL4_NAME --当前SAP层次4名称	  
      ,E.MATL_GROUP_CODE AS CUR_SAP_MATL_GROUP_CODE --当前SAP物料组编码
      ,E.MATL_GROUP_NAME AS CUR_SAP_MATL_GROUP_NAME --当前SAP物料组名称
      ,E.EXTERNAL_MATL_GROUP_CODE AS CUR_SAP_EXTERNAL_MATL_GROUP_CODE --当前SAP外部物料组编码
      ,E.EXTERNAL_MATL_GROUP_NAME AS CUR_SAP_EXTERNAL_MATL_GROUP_NAME --当前SAP外部物料组名称
	  ,CASE WHEN E.MATL_GROUP_CODE IN('3000010','4000010') THEN 1
	        WHEN E.MATL_GROUP_CODE = '9000010' AND E.PRODUCT_LEVEL2_CODE = '9006' THEN 1
			WHEN F.MATERIAL_TYPE <> '非东鹏' THEN 1
			ELSE 0
			 END IS_SW --是否卫浴物料
  FROM TMP A
  LEFT JOIN (SELECT * FROM UPLOAD.UPLOAD_B1_CUSTOMER_DATA_T
              WHERE IS_MAIN = '是'
		    ) D
    ON A."客户别名" = D.ABBREVIATION
  LEFT JOIN DM.DM_B1_KL_WYJX_INVENTORY_T_TMP03 C
    ON SUBSTR(A."日期",1,7) = C.STAT_MONTH
   AND A."客户别名" = C.CUST_BYNAME
   AND D.CUST_CODE = C.DEALER_CODE
   AND A."物料编码" = C.MATERIAL_NUM
  LEFT JOIN (SELECT SUBSTR("日期",1,7) AS STAT_MONTH
                   ,"物料编码" AS MATERIAL_NUM
				   ,"客户别名" AS CUST_BYNAME
                   ,SUM("期末金额") AS INVENTORY_AMOUNT
                   ,SUM("期末数量") AS INVENTORY_QUANTITY
			   FROM DM.DM_WM_B1_INVENTORY_T_D A
              WHERE SUBSTR("日期",1,10) <= SUBSTR(TO_DATE('${PERIOD_ID_D}','yyyy-MM-dd'),1,10)                
				AND SUBSTR((CAST(A."日期" AS DATE) + INTERVAL '1 MONTH'),1,7) = SUBSTR((CAST(A."日期" AS DATE) + INTERVAL '1 DAY'),1,7)
			  GROUP BY SUBSTR("日期",1,7)
                   ,"物料编码"
				   ,"客户别名"
            ) B
    ON A."物料编码" = B.MATERIAL_NUM
   AND A."客户别名" = B.CUST_BYNAME
   AND SUBSTR(A."日期",1,7) = B.STAT_MONTH	
  LEFT JOIN DWIMD.DWI_MD_DATA_MATERIAL_GENERAL_T E
    ON A."物料编码" = E.MATERIAL_NUM   
  LEFT JOIN UPLOAD.UPLOAD_VM_B1_INV_CATE_MATCH_T F
    ON A."物料编码" = F.MATERIAL_NUM
