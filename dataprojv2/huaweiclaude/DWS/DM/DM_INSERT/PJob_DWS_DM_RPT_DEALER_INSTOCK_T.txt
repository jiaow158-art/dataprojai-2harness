TRUNCATE TABLE DM.DM_RPT_DEALER_INSTOCK_T;
insert into DM.DM_RPT_DEALER_INSTOCK_T ("区域", "客户名称", "客户编码", calday, calmonth, "零售签约", "零售业绩", "工程签约", "工程业绩", "大包签约",
                                        "大包业绩", "设计师签约", "设计师业绩", "目标")
with 层次 as (
    select distinct case
                        when subcompany_big_area_name = '大包市场部' then '大包市场中心'
                        else subcompany_big_area_name end              三级,
                    subcompany_big_area                                三级编码,
                    case
                        when zqdzh = 'G103' then '大包'
                        when zqdzh = 'G102' and zyl01 != 'G105' then '工程'
                        when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105')) then '设计师'
                        else zqdzh___t end                             渠道,
                    case
                        when system in ('22000001', '22000004') and
                             subcompany_big_area not in ('33000030', '33000046')
                            then
                            case
                                when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105')) then replace(
                                        replace(subcompany_big_area_name, '子公司', '设计师业务'),
                                        '运营中心', '设计师业务')
                                else replace(
                                             replace(subcompany_big_area_name, '子公司', ''),
                                             '运营中心', '') || 渠道 ||
                                     (case when subcompany_big_area_name like '%运营中心%' then '营销部' else '渠道' end) end
                        else case
                                 when subcompany_big_area = '33000046' then '工程市场中心'
                                 when subcompany_big_area = '33000030' then '大包市场中心'
                                 else subcompany_big_area_name end end 四级,
                    case
                        when subcompany_big_area = '33000046' then '工程市场中心'
                        when subcompany_big_area = '33000030' then '大包市场中心'
                        when system in ('22000001') and subcompany_big_area not in ('33000030', '33000046') then
                            case
                                when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105')) then replace(
                                        replace(subcompany_big_area_name, '子公司', '设计师业务'),
                                        '运营中心', '设计师业务')
                                else case
                                         when sales_group_name in ('嘉兴组', '湖州组') then '嘉湖组'
                                         when sales_group_name = '海南工程组' then '海南组'
                                         else sales_group_name end end
                        else 四级 end                                    五级,
                    sold_to                                            六级编码,
                    c.txtmd                                            六级
    from DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T a
             join dwrfin.dwr_gl_sales_group_v b
                  on a.sales_grp = b.sales_group
             join dm.dm_rpt_bi0_tcustomer_t c on a.sold_to = c.customer
    where ((system in ('22000001', '22000004') and subcompany_big_area not in ('00000000', '33000046')) or
           subcompany_big_area in ('33000030', '33000046'))
      and sales_group not like 'R%'
),
     转换线组 as (
         select distinct 三级,
                         三级编码,
                         渠道,
                         四级,
                         case a.渠道
                             when '零售' then (case b.零售线组
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(b.零售线组, '子公司', '') end)
                             when '工程' then (case b.工程线组
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(b.工程线组, '子公司', '') end)
                             when '大包' then (case b.大包线组
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(b.大包线组, '子公司', '') end)
                             when '设计师' then (case b.设计师线组
                                                  when '大包市场部' then '大包市场中心'
                                                  else replace(b.设计师线组, '子公司设计师渠道', '设计师业务') end)
                             else a.四级 end 五级,
                         六级编码,
                         六级
         from 层次 a
                  join upload.upload_achievement_relation b on a.六级编码 = lpad(b.客编, 10, '0') and
                                                               a.三级 = (case b.运营中心 when '大包市场部' then '大包市场中心' else b.运营中心 end)
     /*     union all
         select distinct case
                             when subcompany_big_area_name = '大包市场部' then '大包市场中心'
                             else subcompany_big_area_name end              三级,
                         subcompany_big_area                                三级编码,
                         case
                             when zqdzh = 'G103' then '大包'
                             when zqdzh = 'G102' and zyl01 != 'G105' then '工程'
                             when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105')) then '设计师'
                             else zqdzh___t end                             渠道,
                         case
                             when system in ('22000001', '22000004') and
                                  subcompany_big_area not in ('33000030', '33000046')
                                 then
                                 case
                                     when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105')) then replace(
                                             replace(subcompany_big_area_name, '子公司', '设计师业务'),
                                             '运营中心', '设计师业务')
                                     else replace(
                                                  replace(subcompany_big_area_name, '子公司', ''),
                                                  '运营中心', '') || 渠道 ||
                                          (case when subcompany_big_area_name like '%运营中心%' then '营销部' else '渠道' end) end
                             else case
                                      when subcompany_big_area = '33000046' then '工程市场中心'
                                      when subcompany_big_area = '33000030' then '大包市场中心'
                                      else subcompany_big_area_name end end 四级,
                         case 渠道
                             when '零售' then (case a.sales_grp___t
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(a.sales_grp___t, '子公司', '') end)
                             when '工程' then (case a.sales_grp___t
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(a.sales_grp___t, '子公司', '') end)
                             when '大包' then (case a.sales_grp___t
                                                 when '大包市场部' then '大包市场中心'
                                                 else replace(a.sales_grp___t, '子公司', '') end)
                             when '设计师' then (case a.sales_grp___t
                                                  when '大包市场部' then '大包市场中心'
                                                  else replace(a.sales_grp___t, '子公司设计师渠道', '设计师业务') end)
                             else 四级 end                                    五级,
                         sold_to                                            六级编码,
                         txtmd                                              六级
         from DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T a
                  join dwrfin.dwr_gl_sales_group_v b
                       on a.sales_grp = b.sales_group
                  join dm.dm_rpt_bi0_tcustomer_t c on a.sold_to = c.customer
                  left join upload.upload_achievement_relation z on a.sold_to = lpad(z.客编, 10, '0') and
                                                                    case
                                                                        when subcompany_big_area_name = '大包市场部'
                                                                            then '大包市场中心'
                                                                        else subcompany_big_area_name end =
                                                                    (case z.运营中心 when '大包市场部' then '大包市场中心' else z.运营中心 end)
         where ((system in ('22000001', '22000004') and subcompany_big_area not in ('00000000', '33000046')) or
                subcompany_big_area in ('33000030', '33000046'))
           and sales_group not like 'R%'
           and z.客编 is null
           and calmonth >= '202208' */
     ),
     业绩 as (
         select sum(coalesce(zxsyj_kjy, 0) + zxsyj_zd) / 10000 业绩,
                c.三级,
                c.渠道,
                c.四级,
                c.五级,
                c.六级编码,
                c.六级,
                calday,
                calmonth
         from DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T a
                  join dwrfin.dwr_gl_sales_group_v b on a.sales_grp = b.sales_group
                  join 转换线组 c on a.sold_to = c.六级编码 and (case b.subcompany_big_area_name
                                                             when '大包市场部' then '大包市场中心'
                                                             else b.subcompany_big_area_name end) = c.三级 and (case
                                                                                                                  when zqdzh = 'G103'
                                                                                                                      then '大包'
                                                                                                                  when zqdzh = 'G102' and zyl01 != 'G105'
                                                                                                                      then '工程'
                                                                                                                  when (zqdzh = 'G105' or (zqdzh = 'G102' and zyl01 = 'G105'))
                                                                                                                      then '设计师'
                                                                                                                  else zqdzh___t end) =
                                                                                                             c.渠道
         where 1 = 1
         group by c.三级, c.渠道, c.四级, c.五级, c.六级编码, c.六级, calday, calmonth
         union all
         select sum(coalesce(zxsyj_kjy, 0) + zxsyj_zd) / 10000 日业绩,
                b.subcompany_big_area_name                     三级,
                a.zqdzh___t,
                '',
                '调整',
                '调整',
                '调整',
                calday,
                calmonth
         from DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T a
                  join dwrfin.dwr_gl_sales_group_v b on a.sales_grp = b.sales_group
         where 1 = 1
           and sold_to is null
         group by b.subcompany_big_area_name, zqdzh___t, calday, calmonth
     ),
     目标 as (select 销售地区                                                                                  三级,
                   lpad(客户编码, 10, '0')                                                                   六级编码,
                   case
                       when 渠道 = '设计师' then replace(
                               replace(销售地区, '营销部', '设计师业务'),
                               '运营中心', '设计师业务')
                       else replace(
                                    replace(销售地区, '子公司', ''),
                                    '运营中心', '') || 渠道 ||
                            (case when 销售地区 like '%运营中心%' then '营销部' else '渠道' end) end                  四级,
                   unnest(string_to_array((
                                                                                                                                  COALESCE(一月, '0') || ',' ||
                                                                                                                                  COALESCE(二月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(三月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(四月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(五月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(六月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(七月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(八月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(九月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(十月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(十一月, '0') ||
                                                                                                                                  ',' ||
                                                                                                                                  COALESCE(十二月, '0')),
                                          ','))                                                          目标,
                   left(now(), 4) || unnest(string_to_array('01,02,03,04,05,06,07,08,09,10,11,12', ',')) 月份
            FROM upload.upload_achievementcustomer),
     目标汇总 as (
         select sum(目标) 目标,
                三级,
                p.六级编码,
                月份
         from 目标 p
         where 1 = 1
         group by 三级, 六级编码, 月份
     )
select a.三级                                                                                          区域
     , a.六级                                                                                          客户名称
     , a.六级编码                                                                                        客户编码
     , b.calday
     , b.calmonth
     , case when coalesce(sum(case a.渠道 when '零售' then 业绩 else 0 end), 0) > 0 then '是' else '否' end  零售签约
     , coalesce(sum(case a.渠道 when '零售' then 业绩 else 0 end), 0)                                      零售业绩
     , case when coalesce(sum(case a.渠道 when '工程' then 业绩 else 0 end), 0) > 0 then '是' else '否' end  工程签约
     , coalesce(sum(case a.渠道 when '工程' then 业绩 else 0 end), 0)                                      工程业绩
     , case when coalesce(sum(case a.渠道 when '大包' then 业绩 else 0 end), 0) > 0 then '是' else '否' end  大包签约
     , coalesce(sum(case a.渠道 when '大包' then 业绩 else 0 end), 0)                                      大包业绩
     , case when coalesce(sum(case a.渠道 when '设计师' then 业绩 else 0 end), 0) > 0 then '是' else '否' end 设计师签约
     , coalesce(sum(case a.渠道 when '设计师' then 业绩 else 0 end), 0)                                     设计师业绩
     , sum(目标)                                                                                       目标
from 转换线组 a
         left join 业绩 b on a.三级 = b.三级 and a.四级 = b.四级 and a.五级 = b.五级 and a.渠道 = b.渠道 and a.六级编码 = b.六级编码
         left join 目标汇总 c on a.六级编码 = c.六级编码 and a.三级 = c.三级 and b.calmonth = c.月份
where 1 = 1
  AND a.五级 is not null
group by a.三级, a.六级, a.六级编码, b.calday, b.calmonth;
