TRUNCATE TABLE DM.DM_CS_EMP_ORDER_STAT_T;

INSERT INTO DM.DM_CS_EMP_ORDER_STAT_T(
       EMP_NAME --销售员工
      ,SALE_AMT --签单额（元）
	  ,ORDER_CNT --签单数
	  ,OVER_ORDER_CNT --完工数
	  ,AVG_EVALU_SCORE --满意度平均
	  ,OVER_ORDER_RATE --完工率
      ,CNSTR_AREA --施工面积	  
	  ,DATA_TYPE --数据类型	
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
      ,PRVDR_NAME --服务商名称
	  ,QD_ORDER_RATE --签单率
      ,AVG_SALE_AMT --平均客单价
      ,SG_ORDER_CNT --施工数	  
)
SELECT EMP_NAME --销售员工
      ,SUM(SALE_AMT) AS SALE_AMT --签单额（元）
	  ,COUNT(DISTINCT ORDER_NUM) AS ORDER_CNT --签单数
	  ,0 AS OVER_ORDER_CNT --完工数
	  ,0 AS AVG_EVALU_SCORE --满意度平均
	  ,0 AS OVER_ORDER_RATE --完工率
      ,0 AS CNSTR_AREA --施工面积	  
	  ,CAST('销售' AS VARCHAR) AS DATA_TYPE --数据类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,PRVDR_NAME --服务商名称
	  ,0 AS QD_ORDER_RATE --签单率
      ,0 AS AVG_SALE_AMT --平均客单价
      ,0 AS SG_ORDER_CNT --施工数	  
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE A.STAGE_NAME NOT IN('废弃订单','合同取消')
   AND CNTC_CODE IS NOT NULL
 GROUP BY EMP_NAME
         ,PRVDR_NAME

 UNION ALL

SELECT DECOR_NAME AS EMP_NAME --装修师傅
      ,0 AS SALE_AMT --签单额（元）
	  ,COUNT(DISTINCT ORDER_NUM) AS ORDER_CNT --订单数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_NAME IN('待验收','已完成') 
	                       THEN ORDER_ID
						    END
			) AS OVER_ORDER_CNT --完工数
	  ,0 AS AVG_EVALU_SCORE --满意度平均
	  ,0 AS OVER_ORDER_RATE --完工率
      ,0 AS CNSTR_AREA --施工面积	  
	  ,CAST('装修' AS VARCHAR) AS DATA_TYPE --数据类型		
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
      ,PRVDR_NAME --服务商名称
	  ,0 AS QD_ORDER_RATE --签单率
      ,0 AS AVG_SALE_AMT --平均客单价
      ,COUNT(DISTINCT CASE WHEN A.STAGE_NAME IN('施工中') 
	                       THEN ORDER_ID
						    END
			) AS SG_ORDER_CNT --施工数	  
  FROM DWRCS.DWR_CS_CNSTR_INFO_F A
 WHERE A.STAGE_NAME NOT IN('废弃订单','合同取消')
   AND A.DECOR_NAME IS NOT NULL
   AND CNTC_CODE IS NOT NULL
   AND A.EMP_TYPE = 2
 GROUP BY DECOR_NAME
         ,PRVDR_NAME --服务商名称 
 
 UNION ALL
 
SELECT PRVDR_NAME --服务商名称
      ,SUM(SALE_AMT) AS SALE_AMT --销售额（元）
	  ,COUNT(DISTINCT ORDER_NUM) AS ORDER_CNT --签单数
	  ,0 AS OVER_ORDER_CNT --完工数
	  ,SUM(EVALU_SCORE) / SUM(EVALU_CNT) / COUNT(DISTINCT ID) AS AVG_EVALU_SCORE --满意度平均分
	  ,0 AS OVER_ORDER_RATE --完工率
      ,0 AS CNSTR_AREA --施工面积
	  ,CAST('服务商' AS VARCHAR) AS DATA_TYPE --数据类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,PRVDR_NAME --服务商名称
	  ,0 AS QD_ORDER_RATE --签单率
	  ,0 AS AVG_SALE_AMT --平均客单价
	  ,0 AS SG_ORDER_CNT --施工数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE A.CNTC_CODE IS NOT NULL
   AND A.PRVDR_NAME IS NOT NULL
   AND A.STAGE_NAME NOT IN('废弃订单','合同取消')
 GROUP BY PRVDR_NAME
 
 UNION ALL
 
SELECT CITY_SHORT_NAME --城市
      ,SUM(CASE WHEN A.CNTC_CODE IS NOT NULL THEN SALE_AMT ELSE 0 END) AS SALE_AMT --签单额（元）
	  ,COUNT(DISTINCT CASE WHEN A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END) AS ORDER_CNT --签单数
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_NAME IN('待验收','已完成') AND A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END) AS OVER_ORDER_CNT --完工数
	  ,0 AS AVG_EVALU_SCORE --满意度平均
	  ,COUNT(DISTINCT CASE WHEN A.STAGE_NAME IN('待验收','已完成') AND A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END)
	  /COUNT(DISTINCT ORDER_NUM) AS OVER_ORDER_RATE --完工率
	  ,SUM(CASE WHEN A.CNTC_CODE IS NOT NULL THEN CNSTR_AREA ELSE 0 END) AS CNSTR_AREA --施工面积
	  ,CAST('城市' AS VARCHAR) AS DATA_TYPE --数据类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,PRVDR_NAME --服务商名称
	  ,COUNT(DISTINCT CASE WHEN A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END)
	  /COUNT(DISTINCT ORDER_NUM) AS QD_ORDER_RATE --签单率
	  ,CASE WHEN COUNT(DISTINCT CASE WHEN A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END) = 0 then 0
	        ELSE SUM(CASE WHEN A.CNTC_CODE IS NOT NULL THEN SALE_AMT ELSE 0 END)
	            /COUNT(DISTINCT CASE WHEN A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END)
	         END AS AVG_SALE_AMT --平均客单价
	  ,0 AS SG_ORDER_CNT --施工数		 
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE A.STAGE_NAME NOT IN('废弃订单','合同取消')
   --AND A.CNTC_CODE IS NOT NULL 
 GROUP BY CITY_SHORT_NAME
         ,PRVDR_NAME
HAVING SUM(CASE WHEN A.CNTC_CODE IS NOT NULL THEN SALE_AMT ELSE 0 END) > 0		 
 
 UNION ALL
 
SELECT PROV_SHORT_NAME AS PROV_NAME --省份
      ,SUM(SALE_AMT) AS SALE_AMT --签单额（元）
	  ,COUNT(DISTINCT A.ORDER_NUM) AS ORDER_CNT --签单数
	  ,0 AS OVER_ORDER_CNT --完工数
	  ,0 AS AVG_EVALU_SCORE --满意度平均
	  ,0 AS OVER_ORDER_RATE --完工率
	  ,0 AS CNSTR_AREA --施工面积
	  ,CAST('省份' AS VARCHAR) AS DATA_TYPE --数据类型
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,NULL AS PRVDR_NAME --服务商名称
	  ,0 AS QD_ORDER_RATE --签单率
	  ,SUM(CASE WHEN A.CNTC_CODE IS NOT NULL THEN SALE_AMT ELSE 0 END)
	  /COUNT(DISTINCT CASE WHEN A.CNTC_CODE IS NOT NULL THEN ORDER_NUM END) AS AVG_SALE_AMT --平均客单价
      ,0 AS SG_ORDER_CNT --施工数
  FROM DWRCS.DWR_CS_ZDJ_ORDER_INFO_F A
 WHERE A.STAGE_NAME NOT IN('废弃订单','合同取消')
   AND A.CNTC_CODE IS NOT NULL 
 GROUP BY PROV_SHORT_NAME
