DELETE FROM DM.DM_WF_SNGL_REQUESTS_CONSUMING_TIME_T
WHERE STAT_MONTH = '${PERIOD_ID_M}'
;
INSERT INTO DM.DM_WF_SNGL_REQUESTS_CONSUMING_TIME_T(
         STAT_MONTH
        ,REQUEST_CODE
        ,WF_TYPE_NAME
        ,WF_NAME
        ,REQUEST_EXPLAIN
        ,NODE_CNT
        ,EMP_CNT
        ,REQUEST_STATUS
        ,REQUEST_EMP_NAME
        ,REQUEST_DEPT_NAME
        ,REQUEST_DAY
        ,GD_DAY
        ,WORK_HS
        ,WORK_HS_HOUR
        ,HS_24HOUR
        ,HS_24HOR_TIME
        ,REQUEST_LEV1_DEPT_NAME
        ,REQUEST_LEV2_DEPT_NAME
        ,REQUEST_LEV3_DEPT_NAME
        ,REQUEST_LEV4_DEPT_NAME
        ,REQUEST_LEV5_DEPT_NAME
        ,REQUEST_LEV6_DEPT_NAME
        ,REQUEST_LEV7_DEPT_NAME
        ,REQUEST_LEV8_DEPT_NAME
        ,IS_OVERTIEM
        ,DW_LAST_UPDATE_DATE
)
WITH T1 AS(
SELECT CONCAT(CONCAT(SUBSTR(A.STAT_MONTH,1,4),'-'),SUBSTR(A.STAT_MONTH,5,2)) AS STAT_MONTH --统计月
      ,A.REQUEST_CODE --请求ID
	  ,A.WF_TYPE_NAME --L1流程
	  ,A.WF_NAME --流程模板
	  ,A.REQUEST_EXPLAIN --流水号
	  ,SUM(A.NODE_COUNT) AS NODE_CNT --流程节点数
	  ,COUNT(SAPID) AS EMP_CNT --流程处理人次
	  ,A.REQUEST_STATUS --流程状态
	  ,A.REQUEST_EMP_NAME --发起人名称
	  ,A.REQUEST_DEPT_NAME --所属部门
      ,A.REQUEST_DAY --启动时间	
	  ,A.GD_DAY --结束时间
      ,SUM(A.WORK_HS) AS WORK_HS --耗时（工作时间：小时）
	  ,SUM(A.HS_24HOUR) AS HS_24HOUR --耗时24小时制
	  ,A.REQUEST_LEV1_DEPT_NAME --发起人所在一级部门
	  ,A.REQUEST_LEV2_DEPT_NAME --发起人所在二级部门
	  ,A.REQUEST_LEV3_DEPT_NAME --发起人所在三级部门
	  ,A.REQUEST_LEV4_DEPT_NAME --发起人所在四级部门
	  ,A.REQUEST_LEV5_DEPT_NAME --发起人所在五级部门
	  ,A.REQUEST_LEV6_DEPT_NAME --发起人所在六级部门
	  ,A.REQUEST_LEV7_DEPT_NAME --发起人所在七级部门
	  ,A.REQUEST_LEV8_DEPT_NAME --发起人所在八级部门
	  ,'' AS IS_OVERTIEM --是否超时
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_T A
 WHERE A.STAT_MONTH = REPLACE('${PERIOD_ID_M}','-','')
   AND SUBSTR(A.REQUEST_DAY,1,10) >= '2023-01-01' 
 GROUP BY
       CONCAT(CONCAT(SUBSTR(A.STAT_MONTH,1,4),'-'),SUBSTR(A.STAT_MONTH,5,2)) --统计月
      ,A.REQUEST_CODE --请求ID
	  ,A.WF_TYPE_NAME --L1流程
	  ,A.WF_NAME --流程模板
	  ,A.REQUEST_EXPLAIN --流水号
	  ,A.REQUEST_STATUS --流程状态
	  ,A.REQUEST_EMP_NAME --发起人名称
	  ,A.REQUEST_DEPT_NAME --所属部门
      ,A.REQUEST_DAY --启动时间	
	  ,A.GD_DAY --结束时间
	  ,A.REQUEST_LEV1_DEPT_NAME --发起人所在一级部门
	  ,A.REQUEST_LEV2_DEPT_NAME --发起人所在二级部门
	  ,A.REQUEST_LEV3_DEPT_NAME --发起人所在三级部门
	  ,A.REQUEST_LEV4_DEPT_NAME --发起人所在四级部门
	  ,A.REQUEST_LEV5_DEPT_NAME --发起人所在五级部门
	  ,A.REQUEST_LEV6_DEPT_NAME --发起人所在六级部门
	  ,A.REQUEST_LEV7_DEPT_NAME --发起人所在七级部门
	  ,A.REQUEST_LEV8_DEPT_NAME --发起人所在八级部门
)
SELECT STAT_MONTH --统计月
      ,REQUEST_CODE --请求ID
	  ,WF_TYPE_NAME --L1流程
	  ,WF_NAME --流程模板
	  ,REQUEST_EXPLAIN --流水号
	  ,NODE_CNT --流程节点数
	  ,EMP_CNT --流程处理人次
	  ,REQUEST_STATUS --流程状态
	  ,REQUEST_EMP_NAME --发起人名称
	  ,REQUEST_DEPT_NAME --所属部门
      ,REQUEST_DAY --启动时间	
	  ,GD_DAY --结束时间
      ,WORK_HS --耗时（工作时间：秒）
	  ,ROUND(WORK_HS / 3600,2) AS WORK_HS_HOUR --耗时（工作时间：小时）
	  ,HS_24HOUR --耗时24小时制（秒）
	  ,REPLACE(CASE WHEN HS_24HOUR >= 86400 
	                THEN TRUNC(HS_24HOUR/86400) || '天' || TRUNC(MOD(HS_24HOUR,86400)/3600) || '小时' || TRUNC(MOD(MOD(HS_24HOUR,86400),3600)/60) || '分钟'
			        WHEN HS_24HOUR < 86400 AND HS_24HOUR >= 3600
			        THEN '0天'||TRUNC(HS_24HOUR/3600) || '小时' || TRUNC(MOD(HS_24HOUR,3600)/60) || '分钟'
			        WHEN HS_24HOUR < 3600
			        THEN '0天0小时'||TRUNC(HS_24HOUR/60)||'分钟'
			         END,'.0','') AS HS_24HOR_TIME --耗时24小时制
	  ,REQUEST_LEV1_DEPT_NAME --发起人所在一级部门
	  ,REQUEST_LEV2_DEPT_NAME --发起人所在二级部门
	  ,REQUEST_LEV3_DEPT_NAME --发起人所在三级部门
	  ,REQUEST_LEV4_DEPT_NAME --发起人所在四级部门
	  ,REQUEST_LEV5_DEPT_NAME --发起人所在五级部门
	  ,REQUEST_LEV6_DEPT_NAME --发起人所在六级部门
	  ,REQUEST_LEV7_DEPT_NAME --发起人所在七级部门
	  ,REQUEST_LEV8_DEPT_NAME --发起人所在八级部门
	  ,IS_OVERTIEM --是否超时
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
  FROM T1
