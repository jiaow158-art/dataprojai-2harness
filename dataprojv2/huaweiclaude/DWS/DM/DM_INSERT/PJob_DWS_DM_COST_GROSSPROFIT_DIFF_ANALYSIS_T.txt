with tmp as (

		,sales_region 
		,operation_center

	,sales_region 
	,operation_center
	,months              -- 月份
	,days
	,prod_hierarchys  -- 产品层次
	,integration_channel

,tmp1 as (
	select 
		 node_desc2
		,node_desc3
		,sales_region 
		,operation_center
		,months              -- 月份
		,prod_hierarchys  -- 产品层次
		,integration_channel 
		,sum(nvl(budget_amt,0)) budget_amt
		,sum(nvl(budget_notax_sales_net_amt,0)) budget_notax_sales_net_amt
		,sum(nvl(budget_cost_a,0)) budget_cost_a
		,sum(budget_sales_area) budget_sales_area
		,sum(sales_amt) sales_amt
		,sum(notax_sales_net_amt) notax_sales_net_amt
		,sum(actual_cost) actual_cost
		,sum(nvl(sales_area,0)) sales_area
		,sum(gross_profit) gross_profit 
		,sum(sum(nvl(sales_area,0))) over(partition by months)  all_area
		,sum(sum(nvl(sales_area,0))) over(partition by months,node_desc2,node_desc3) node_desc3_area
		,sum(sum(nvl(sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region) sales_region_area
		,sum(sum(nvl(sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center)operation_center_area
		,sum(sum(nvl(sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel) integration_channel_area
		,sum(sum(nvl(sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel,prod_hierarchys) prod_hierarchys_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months)  budget_all_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months,node_desc2,node_desc3) budget_node_desc3_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region) budget_sales_region_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center) budget_operation_center_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel) budget_integration_channel_area
		,sum(sum(nvl(budget_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel,prod_hierarchys) budget_prod_hierarchys_area
		,sum(nvl(last_year_sales_area,0)) last_year_sales_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months)  last_year_all_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months,node_desc2,node_desc3) last_year_node_desc3_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region) last_year_sales_region_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center) last_year_operation_center_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel) last_year_integration_channel_area
		,sum(sum(nvl(last_year_sales_area,0))) over(partition by months,node_desc2,node_desc3,sales_region,operation_center,integration_channel,prod_hierarchys) last_year_prod_hierarchys_area
	from tmp 
	group by 
		 node_desc2
		,node_desc3
		,sales_region 
		,operation_center
		,months              -- 月份
		,prod_hierarchys     -- 产品层次
		,integration_channel 
)
,tmp2 as (
	select 
		 months
		,node_desc2    
		,node_desc3   
		,sales_region   	
		,operation_center   	
		,integration_channel   
		,prod_hierarchys    
		,sum(sales_area) sales_area
		,sum(notax_sales_net_amt) notax_sales_net_amt
		,sum(budget_sales_area) budget_sales_area
		,sum(budget_notax_sales_net_amt) budget_notax_sales_net_amt
		,sum(actual_cost) actual_cost
		,sum(budget_cost_a) budget_cost_a
		,sum(gross_profit) gross_profit 
		,sum(sum(sales_area)) over() total_area 
		,case  when sum(sales_area)  <> 0 then sum(notax_sales_net_amt) / sum(sales_area) else 0 end actual_unit_price 
		,case  when sum(sales_area)  <> 0 then sum(actual_cost) / sum(sales_area) else 0 end actual_unit_cost
		,case when  sum(sales_area)  <> 0 then sum(sum(sales_area)) over(partition by node_desc2,node_desc3) / sum(sum(sales_area)) over()else 0 end org_perc
		,case when sum(sum(sales_area)) over(partition by node_desc2,node_desc3) <> 0 then  sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region) / sum(sum(sales_area)) over(partition by node_desc2,node_desc3)
		else 0 end  region_perc
		, case when sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region) <> 0 then  sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center) /sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region) else 0 end  operation_center_perc

		, case when  sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center)  <> 0 
		then sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel) / sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center) 
		else 0 end  integration_channel_perc
		, case when sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel)  <> 0 
		then sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel,prod_hierarchys) / sum(sum(sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel)  
		else 0 end  prod_hierarchys_perc
		,case  when sum(budget_sales_area)  <> 0 then sum(budget_notax_sales_net_amt) / sum(budget_sales_area) else 0 end budget_unit_price
		,case  when sum(budget_sales_area)  <> 0 then sum(budget_cost_a) / sum(budget_sales_area) else 0 end budget_unit_cost
		--, sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3) / sum(sum(budget_sales_area)) over()  budget_org_perc  update from chenzhiyong 20230103
		,case when sum(sum(budget_sales_area)) over() <> 0 
		   then sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3) / sum(sum(budget_sales_area)) over() 
		 else 0 end as budget_org_perc
		, case when sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3)  <> 0 then sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region) /sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3) 
		else 0 end  budget_region_perc
		, case when sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region) <> 0 then sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center) /sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region)
		else 0 end  budget_operation_center_perc
		, case when sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center) <> 0  then sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel) / sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center)
		else 0 end  budget_integration_channel_perc
		, case when sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel)  <> 0 
		then sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel,prod_hierarchys) / sum(sum(budget_sales_area)) over(partition by node_desc2,node_desc3,sales_region,operation_center,integration_channel)  
		else 0 end  budget_prod_hierarchys_perc
	from tmp1
	group by 
		 node_desc2    
		,node_desc3   
		,sales_region   	
		,operation_center   	
		,integration_channel   
		,prod_hierarchys
		,months
)
select 
	 months
	,node_desc2    
	,node_desc3
	,sales_region
	,operation_center
	,integration_channel
	,prod_hierarchys
	,actual_unit_price
	,actual_unit_cost
	,budget_cost_a
	,budget_unit_price         
	,budget_unit_cost      
	,sales_area
	,budget_sales_area
	,total_area      
	,round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * budget_prod_hierarchys_perc,2)  product_restore_structure
	,round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * prod_hierarchys_perc,2)         channel_restore_structure
	,round((actual_unit_price - budget_unit_price) * sales_area,2) price_deviation
	,round((actual_unit_cost - budget_unit_cost) * sales_area * -1 ,2) cost_deviation
	,(budget_unit_price-budget_unit_cost) *  (round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * budget_prod_hierarchys_perc,2) - budget_sales_area)  volume_deviation   
	,(budget_unit_price-budget_unit_cost) *  (sales_area - round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * prod_hierarchys_perc,2) ) channel_structure_deviation 
	,(budget_unit_price-budget_unit_cost) * (round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * prod_hierarchys_perc,2)   - 
	round(total_area * budget_org_perc * budget_region_perc * budget_operation_center_perc * budget_integration_channel_perc * budget_prod_hierarchys_perc,2) ) product_structure_deviation 
	,(budget_notax_sales_net_amt-budget_cost_a)  budget_gross_profit
	,(notax_sales_net_amt-actual_cost)  gross_profit
	,current_timestamp() as dw_last_update_date
from tmp2
where sales_area <> 0;
