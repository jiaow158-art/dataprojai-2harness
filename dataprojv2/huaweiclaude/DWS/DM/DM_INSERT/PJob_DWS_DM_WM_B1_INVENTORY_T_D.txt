,tmp1 as (
		  select 
		       dt.day_dt,
			   case when t0.docdate< dt.day_dt then 'qc'

				case when nvl(t1.transvalue,0.0)>0.0 then t1.transvalue 
					 else 0.0 
				end as inlinebalance,       
				case when nvl(t1.transvalue,0.0)<0.0 then -t1.transvalue 
					 else 0.0 
				end as outlinebalance,	       
				case when (t0.transtype=18 or t0.transtype=20) then t0.inqty 
					 else 0.0 
				end as puch_inqty,        
				case when (t0.transtype=14 or t0.transtype=16) then t0.inqty 
					else 0.0
				end as sale_inqty,       
				case when nvl(t1.transvalue,0.0)>=0.0 and t0.transtype<>18 and t0.transtype<>20 and t0.transtype<>14 and t0.transtype<>16 then t0.inqty 
					else 0.0 
				end as other_inqty,        
				case when (t0.transtype=19 or t0.transtype=21) then t0.outqty 
					else 0.0 
				end as puch_outqty,        
				case when (t0.transtype=13 or t0.transtype=15) then t0.outqty 
					else 0.0 
				end as sale_outqty,       
				case when nvl(t1.transvalue,0.0)<=0.0 and t0.transtype<>19 and t0.transtype<>21 and t0.transtype<>13 and t0.transtype<>15 then t0.outqty 
					else 0.0 
				end as other_outqty,     
				t1.transvalue as linebalance,
				t0.loccode as loccode ,
				t0.customer_short_name      
		from oivl t0  
		inner join ivl1 t1 on t0.transseq = t1.transseq  and t0.customer_short_name = t1.customer_short_name         
		inner join oitm t2  on t0.itemcode = t2.itemcode  and t0.customer_short_name = t2.customer_short_name      
		inner join oitb t3  on t2.itmsgrpcod=t3.itmsgrpcod  and t2.customer_short_name = t3.customer_short_name
		inner join omrc t4  on t2.firmcode=t4.firmcode    and t2.customer_short_name = t4.customer_short_name
		left join oilm t5 on nvl(t0.messageid,'') = nvl(t5.messageid,'') and nvl(t5.effectqty,0.0)<>0.0      and t0.customer_short_name = t5.customer_short_name
		left join ocrd t6 on nvl(t5.bpcardcode,'') = nvl(t6.cardcode,'')    and t5.customer_short_name = t6.customer_short_name   
		left join ocrd t7 on nvl(t2.cardcode,'')=nvl(t7.cardcode,'')        and t2.customer_short_name = t7.customer_short_name

    ,sum(case when t.dttype='qm' then nvl(t.linebalance,0.0) else 0.0 end) as qmlinebalance
    ,sum(case when t.dttype='qj' then nvl(t.puch_inqty,0.0) else 0.0 end) as qjpuch_inqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_inqty,0.0) else 0.0 end) as qjsale_inqty
    ,sum(case when t.dttype='qj' then nvl(t.other_inqty,0.0) else 0.0 end) as qjother_inqty
    ,sum(case when t.dttype='qj' then nvl(t.puch_outqty,0.0) else 0.0 end)as qjpuch_outqty
    ,sum(case when t.dttype='qj' then nvl(t.sale_outqty,0.0) else 0.0 end) as qjsale_outqty
    ,sum(case when t.dttype='qj' then nvl(t.other_outqty,0.0) else 0.0 end) as qjother_outqty
      from tmp1 t

          t9.material_name ,
