delete
from DM.DM_RPT_CT_DEPARTMENT_SALES_PERFORMANCE_ENGINEERINGDESIGN_T ----瓷砖销售业绩工程设计师双计
where calday = to_char(current_date - 1, 'yyyyMMdd');
insert into DM.DM_RPT_CT_DEPARTMENT_SALES_PERFORMANCE_ENGINEERINGDESIGN_T (一级,
                                                                           二级,
                                                                           三级,
                                                                           渠道,
                                                                           四级,
                                                                           五级,
                                                                           本日业绩,
                                                                           月业绩,
                                                                           季业绩,
                                                                           年业绩,
                                                                           去年同期业绩,
                                                                           季度同期业绩,
                                                                           去年业绩,
                                                                           六级编码,
                                                                           六级,
                                                                           calday)
with 层次 as (
    select distinct '国内营销系统'                                           一级,
                    case
                        when system = '22000001' and subcompany_big_area != '33000046' then '区域运营中心'
                        when subcompany_big_area = '33000046' then '工程市场中心'
                        when subcompany_big_area = '33000030' then '大包市场中心'
                        else system_name end                           二级,
                    case
                        when system = '22000001' and subcompany_big_area != '33000046' then '22000001'
                        when subcompany_big_area = '33000046' then '33000046'
                        when subcompany_big_area = '33000030' then '33000030'
                        else system end                                二级编码,
                    case
                        when subcompany_big_area_name = '大包市场部' then '大包市场中心'
                        else subcompany_big_area_name end              三级,
                    subcompany_big_area                                三级编码,
                    '工程'                                               渠道,
                    case
                        when system in ('22000001', '22000004') and
                             subcompany_big_area not in ('33000030', '33000046')
                            then
                                replace(
                                        replace(subcompany_big_area_name, '子公司', ''),
                                        '运营中心', '') || '工程' ||
                                (case when subcompany_big_area_name like '%运营中心%' then '营销部' else '渠道' end)
                        else case
                                 when subcompany_big_area = '33000046' then '工程市场中心'
                                 when subcompany_big_area = '33000030' then '大包市场中心'
                                 else subcompany_big_area_name end end 四级,
                    case
                        when subcompany_big_area = '33000046' then '工程市场中心'
                        when subcompany_big_area = '33000030' then '大包市场中心'
                        when system in ('22000001') and subcompany_big_area not in ('33000030', '33000046') then
                            case
                                when (distr_chan = '06' and zyl01 = 'G105') then replace(
                                        replace(subcompany_big_area_name, '子公司', '设计师业务'),
                                        '运营中心', '设计师业务')
                                else case
                                         when sales_group_name in ('嘉兴组', '湖州组') then '嘉湖组'
                                         when sales_group_name = '海南工程组' then '海南组'
                                         else sales_group_name end end
                        else 四级 end                                    五级,
                    sold_to                                            六级编码,
                    c.txtmd                                            六级
    from DM.DM_RPT_GROUP_ACHIEVEMENT_AMB_T a
             join dwrfin.dwr_gl_sales_group_v b
                  on a.sales_grp = b.sales_group
             join dm.dm_rpt_bi0_tcustomer_t c on a.sold_to = c.customer
    where ((system in ('22000001', '22000004') and subcompany_big_area not in ('00000000', '33000046')) or
           subcompany_big_area in ('33000030', '33000046'))
      --and sales_group not like 'R%'
      --and (distr_chan = '06' and zyl01 = 'G105')
    union all
    select '国内营销系统'   一级,
           '区域运营中心'   二级,
           '22000001' 二级编码,
           '黑吉辽运营中心'  三级,
           '33000069' 三级编码,
           '工程'       渠道,
           '黑吉辽工程营销部' 四级,
           '辽宁组'      五级,
           '0000321042',
           '内蒙古锡林郭勒盟嘉旭'
    from upload.upload_achievement_history
    where 客编 = '0000321042'
),
     转换线组 as (
         select distinct 一级,
                         二级,
                         二级编码,
                         三级,
                         三级编码,
                         渠道,
                         四级,
                         case b.运营中心
                             when '工程市场中心' then '工程市场中心'
                             else
                                 case a.渠道
                                     when '零售' then (case b.零售线组
                                                         when '大包市场部' then '大包市场中心'
                                                         else replace(b.零售线组, '子公司', '') end)
                                     when '工程' then (case b.工程线组
                                                         when '大包市场部' then '大包市场中心'
                                                         else replace(b.工程线组, '子公司', '') end)
                                     when '大包' then (case b.大包线组
                                                         when '大包市场部' then '大包市场中心'
                                                         else replace(b.大包线组, '子公司', '') end)
                                     when '设计师' then (case b.设计师业务
                                                          when '大包市场部' then '大包市场中心'
                                                          else replace(b.设计师业务, '子公司设计师渠道', '设计师业务') end)

                                     else a.四级 end end 五级,
                         六级编码,
                         六级
         from 层次 a
                  join upload.upload_achievement_history b on a.六级编码 = lpad(b.客编, 10, '0') and
                                                              a.三级 = (case b.运营中心 when '大包市场部' then '大包市场中心' else b.运营中心 end)
         union all
         select distinct '国内营销系统'                                                                    一级,
                         case
                             when system = '22000001' and subcompany_big_area != '33000046' then '区域运营中心'
                             when subcompany_big_area = '33000046' then '工程市场中心'
                             when subcompany_big_area = '33000030' then '大包市场中心'
                             else system_name end                                                    二级,
                         case
                             when system = '22000001' and subcompany_big_area != '33000046' then '22000001'
                             when subcompany_big_area = '33000046' then '33000046'
                             when subcompany_big_area = '33000030' then '33000030'
                             else system end                                                         二级编码,
                         case
                             when subcompany_big_area_name = '大包市场部' then '大包市场中心'
                             else subcompany_big_area_name end                                       三级,
                         subcompany_big_area                                                         三级编码,
                         '工程'                                                                        渠道,
                         replace(
                                 replace(subcompany_big_area_name, '子公司', ''),
                                 '运营中心', '') || 渠道 ||
                         (case when subcompany_big_area_name like '%运营中心%' then '营销部' else '渠道' end) 四级,
                         replace(b.sales_group_name, '子公司', '')                                      五级,
                         son_cust_num                                                                六级编码,
                         txtmd                                                                       六级
         from DM.dm_rpt_sales_achievement_send_rpt_t a
                  join dwrfin.dwr_gl_sales_group_v b
                       on a.sales_grp_code = b.sales_group
                  join dm.dm_rpt_bi0_tcustomer_t c on a.son_cust_num = c.customer
                  left join upload.upload_achievement_history z on a.son_cust_num = lpad(z.客编, 10, '0') and
                                                                   case
                                                                       when subcompany_big_area_name = '大包市场部'
                                                                           then '大包市场中心'
                                                                       else subcompany_big_area_name end =
                                                                   (case z.运营中心 when '大包市场部' then '大包市场中心' else z.运营中心 end)
         where ((system in ('22000001', '22000004') and subcompany_big_area not in ('00000000', '33000046')) or
                subcompany_big_area in ('33000030', '33000046'))
           and sales_group not like 'R%'
           and z.客编 is null
           and calmonth = to_char(current_date - 1, 'yyyyMM')
           and zxsyj_dn != 0
    and (distr_chan_code = '06' and product_channel_code = 'G105')
    ),
    业绩 as (
select sum (zxsyj_dr) / 10000 本日业绩,
    sum (zxsyj_dy) / 10000 本月业绩合计,
    sum (zxsyj_dy_a000) / 10000 本月工程差价积分,
    sum (zxsyj_dy_g000) / 10000 本月共享仓业绩,
    sum (zxsyj_dy_qt) / 10000 本月其他业绩调整,
    sum (zxsmj_dy_715) / 10000 本月715销售面积,
    sum (zxsmj_dy_cdb) / 10000 本月超大板销售面积,
    sum (zxsyj_dy_gz) / 10000 本月高值,
    sum (zxqwchje_dy) / 10000 本月需求未出货金额,
    c.三级,
    c.渠道,
    c.四级,
    c.五级,
    c.六级编码,
    c.六级
from DM.DM_RPT_SALES_ACHIEVEMENT_SEND_RPT_T a
    join dwrfin.dwr_gl_sales_group_v b
on a.sales_grp_code = b.sales_group
    join 转换线组 c on a.son_cust_num = c.六级编码 and (case b.subcompany_big_area_name
    when '大包市场部' then '大包市场中心'
    else b.subcompany_big_area_name end) = c.三级
where 1 = 1
  and calday = to_char(current_date - 1
    , 'yyyymmdd')
  and (distr_chan_code = '06'
  and product_channel_code = 'G105')
group by c.三级, c.渠道, c.四级, c.五级, c.六级编码, c.六级
union all
select sum (zxsyj_dr) / 10000 本日业绩,
    sum (zxsyj_dy) / 10000 本月业绩合计,
    sum (zxsyj_dy_a000) / 10000 本月工程差价积分,
    sum (zxsyj_dy_g000) / 10000 本月共享仓业绩,
    sum (zxsyj_dy_qt) / 10000 本月其他业绩调整,
    sum (zxsmj_dy_715) / 10000 本月715销售面积,
    sum (zxsmj_dy_cdb) / 10000 本月超大板销售面积,
    sum (zxsyj_dy_gz) / 10000 本月高值,
    sum (zxqwchje_dy) / 10000 本月需求未出货金额,
    b.subcompany_big_area_name 三级,
    '',
    '',
    '调整',
    '',
    ''
from DM.DM_RPT_SALES_ACHIEVEMENT_SEND_RPT_T a
    join dwrfin.dwr_gl_sales_group_v b
on a.sales_grp_code = b.sales_group
where 1 = 1
  and calday = to_char(current_date - 1
    , 'yyyymmdd')
  and ((system in ('22000001'
    , '22000004')
  and subcompany_big_area not in ('00000000'
    , '33000046'))
   or
    subcompany_big_area in ('33000030'
    , '33000046'))
  and a.son_cust_num is null
  and (distr_chan_code = '06'
  and product_channel_code = 'G105')
group by b.subcompany_big_area_name
    ),
    年历史 as (
select 运营中心 三级,
    '工程' 渠道,
    replace(工程线组, '子公司', '') 五级,
    lpad(客编, 10, '0') 六级编码,
    sum (设计师工程季度) / 10000 历史季业绩,
    sum (设计师工程年累计) / 10000 历史年业绩
from upload.upload_achievement_history_07before
where 工程线组 is not null
group by 运营中心, 工程线组, lpad(客编, 10, '0')
    ),
    去年历史 as (
select 运营中心 三级,
    '工程' 渠道,
    replace(工程线组, '子公司', '') 五级,
    sum (设计师工程去年同期) / 10000 去年同期业绩,
    sum (设计师工程季度同期) / 10000 季度同期业绩,
    sum (设计师工程去年累计) / 10000 去年业绩,
    lpad(客编, 10, '0') 六级编码
from upload.upload_achievement_history
where 工程线组 is not null
group by 运营中心, 工程线组, lpad(客编, 10, '0')
    )
select 一级,
       二级,
       a.三级,
       a.渠道,
       a.四级,
       a.五级,
       b.本日业绩,
       coalesce(b.本月业绩合计, 0) 月业绩,
       case
           when right(to_char(current_date - 1, 'MM'), 2) in ('01', '04', '07', '10') then 0
    else coalesce (历史季业绩, 0)
end + coalesce(b.本月业绩合计, 0) 季业绩,
       coalesce(历史年业绩, 0) + coalesce(b.本月业绩合计, 0)              年业绩,
       coalesce(去年同期业绩, 0)                                     去年同期业绩,
       coalesce(季度同期业绩, 0)                                     季度同期业绩,
       去年业绩,
       a.六级编码,
       a.六级,
       to_char(current_date - 1, 'yyyyMMdd')                   calday
from 转换线组 a
         left join 业绩 b on a.三级 = b.三级 and a.四级 = b.四级 and a.五级 = b.五级 and a.渠道 = b.渠道 and a.六级编码 = b.六级编码
         left join 去年历史 f on a.三级 = f.三级 and a.五级 = f.五级 and a.渠道 = f.渠道 and a.六级编码 = f.六级编码
         left join 年历史 h on a.三级 = h.三级 and a.五级 = h.五级 and a.渠道 = h.渠道 and a.六级编码 = h.六级编码
         left join upload.upload_ctseq z on a.三级 = z.ct运营中心
where 1 = 1
  AND a.五级 is not null
order by z.顺序, a.二级编码, a.三级编码, (case a.渠道 when '零售' then 1 when '工程' then 2 when '大包' then 3 else 4 end), a.五级
