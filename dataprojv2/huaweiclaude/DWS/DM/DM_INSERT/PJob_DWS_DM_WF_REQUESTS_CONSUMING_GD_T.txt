DELETE FROM DM.DM_WF_REQUESTS_CONSUMING_GD_T
WHERE STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
;

INSERT INTO DM.DM_WF_REQUESTS_CONSUMING_GD_T
SELECT 
 A.STAT_MONTH --年月
,A.WF_TYPE_NAME --L1流程分类
,A.WF_NAME --流程名
--,NODE_NAME 节点名
,A.CURRENT_NODE_TYPE --审批操作分类
,SUM(A.GZ_HOUR) AS GZ_HOUR --总耗时
,COUNT(1) TOTAL_NOCE_COUNT --总节点
,COUNT(DISTINCT A.WF_NAME) AS WF_COUNT --流程数
,COUNT(DISTINCT A.NODE_NAME) AS NOCE_COUNT --节点数
,COUNT(DISTINCT A.REQUEST_EXPLAIN) AS REQUEST_COUNT --请求流程数
--,归档判断
--,归档时间
,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
FROM DM.DM_WF_REQUESTS_CONSUMING_TIME_PRE_T A
INNER JOIN DWRPI.DWR_WF_REQUEST_BASIC_INTO_F B
ON A.REQUEST_CODE = B.REQUEST_CODE
AND B.CURRENT_NODE_TYPE = 3
WHERE A.STAT_MONTH = REPLACE(SUBSTR('${PERIOD_ID_D}',1,7),'-','')
AND REPLACE(SUBSTR(B.LAST_OPERATE_DATE,1,7),'-','') = A.STAT_MONTH
GROUP  BY A.WF_TYPE_NAME 
,A.WF_NAME 
--,NODE_NAME 节点名
,A.CURRENT_NODE_TYPE
,A.STAT_MONTH
;
