with recursive tmp  as (
select  
org_short_name_cn::varchar(1024) as org_short_name_cn_all ,
org_code ::varchar(1024) as org_code_all,
org_name_cn::varchar(1024) org_name_cn_all,

org_name_cn
FROM  DM.DM_ORG_STRUCTURE_ENTWECHAT_D_TEST  H

and data_flag = 0

select
(t2.org_short_name_cn_all||'>,'||t1.org_short_name_cn )::varchar(1024) org_short_name_cn_all,

(t2.org_name_cn_all||'>,'||t1.org_name_cn)::varchar(1024) org_name_cn_all,
t1.org_code ,
t1.up_org_code,
t1.org_short_name_cn,
t1.org_name_cn
from  
 DM.DM_ORG_STRUCTURE_ENTWECHAT_D_TEST  t1 
 join tmp t2   on t1.up_org_code = t2.org_code
 where  data_flag = 0
)
