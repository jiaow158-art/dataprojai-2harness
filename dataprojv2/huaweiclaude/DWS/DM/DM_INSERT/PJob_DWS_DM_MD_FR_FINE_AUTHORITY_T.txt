DELETE FROM DM.DM_MD_FR_FINE_AUTHORITY_T
 WHERE STAT_MONTH = '${PERIOD_ID_M}'
;

WITH T1 AS(
SELECT DISTINCT
	   COALESCE(E.USERNAME,D.USERNAME) AS USERNAME
	  ,COALESCE(E.REALNAME,D.REALNAME) AS REALNAME
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_ID,F.REPORT_ID),G.REPORT_ID),H.REPORT_ID),I.REPORT_ID),J.REPORT_ID),K.REPORT_ID),N.REPORT_ID),M.REPORT_ID),O.REPORT_ID) AS REPORT_ID
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_NUM,F.REPORT_NUM),G.REPORT_NUM),H.REPORT_NUM),I.REPORT_NUM),J.REPORT_NUM),K.REPORT_NUM),N.REPORT_NUM),M.REPORT_NUM),O.REPORT_NUM) AS REPORT_NUM
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_NAME,F.REPORT_NAME),G.REPORT_NAME),H.REPORT_NAME),I.REPORT_NAME),J.REPORT_NAME),K.REPORT_NAME),N.REPORT_NAME),M.REPORT_NAME),O.REPORT_NAME) AS REPORT_NAME
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_PATH,F.REPORT_PATH),G.REPORT_PATH),H.REPORT_PATH),I.REPORT_PATH),J.REPORT_PATH),K.REPORT_PATH),N.REPORT_PATH),M.REPORT_PATH),O.REPORT_PATH) AS REPORT_PATH
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_FULL_NAME,F.REPORT_FULL_NAME),G.REPORT_FULL_NAME),H.REPORT_FULL_NAME),I.REPORT_FULL_NAME),J.REPORT_FULL_NAME),K.REPORT_FULL_NAME),N.REPORT_FULL_NAME),M.REPORT_FULL_NAME),O.REPORT_FULL_NAME) AS REPORT_FULL_NAME
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.DEPT_NAME,F.DEPT_NAME),G.DEPT_NAME),H.DEPT_NAME),I.DEPT_NAME),J.DEPT_NAME),K.DEPT_NAME),N.DEPT_NAME),M.DEPT_NAME),O.DEPT_NAME) AS DEPT_NAME
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,A.ROLETYPE AS ROLE_TYPE_CODE --角色类型编码
	  ,CASE WHEN A.ROLETYPE = 1 THEN '部门'
	        WHEN A.ROLETYPE = 2 THEN '自定义角色'
			WHEN A.ROLETYPE = 3 THEN '用户'
			 END AS ROLE_TYPE_NAME --角色类型描述
	  ,CASE WHEN A.ROLETYPE = 2 THEN A.ROLEID END AS ROLE_ID --角色ID
	  ,CASE WHEN A.ROLETYPE = 2 THEN C.ROLE_NAME END AS ROLE_NAME --角色名称
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.EXPANDID,F.EXPANDID),G.EXPANDID),H.EXPANDID),I.EXPANDID),J.EXPANDID),K.EXPANDID),N.EXPANDID),M.EXPANDID),O.EXPANDID) AS EXPANDID --扩展ID
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.SHOW_TYPE_CODE,F.SHOW_TYPE_CODE),G.SHOW_TYPE_CODE),H.SHOW_TYPE_CODE),I.SHOW_TYPE_CODE),J.SHOW_TYPE_CODE),K.SHOW_TYPE_CODE),N.SHOW_TYPE_CODE),M.SHOW_TYPE_CODE),O.SHOW_TYPE_CODE) AS SHOW_TYPE_CODE --预览类型编码
	  ,COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.SHOW_TYPE_NAME,F.SHOW_TYPE_NAME),G.SHOW_TYPE_NAME),H.SHOW_TYPE_NAME),I.SHOW_TYPE_NAME),J.SHOW_TYPE_NAME),K.SHOW_TYPE_NAME),N.SHOW_TYPE_NAME),M.SHOW_TYPE_NAME),O.SHOW_TYPE_NAME) AS SHOW_TYPE_NAME --预览类型描述
  FROM DWIMD.DWI_MD_FR_FINE_AUTHORITY_T A
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T B
    ON A.AUTHORITYENTITYID = B.REPORT_ID
   AND B.REPORT_PATH IS NOT NULL   
  LEFT JOIN DWIMD.DWI_MD_FR_USER_ROLE_MIDDLE_T C
    ON A.ROLEID = C.ROLEID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_USER_T D
    ON C.USERID = D.ID
  LEFT JOIN DWIMD.DWI_MD_FR_FINE_USER_T E
    ON A.ROLEID = E.ID
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T F
    ON A.AUTHORITYENTITYID = F.ROOT_ID
   AND F.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T G
    ON A.AUTHORITYENTITYID = G.ROOT_ID1
   AND G.REPORT_PATH IS NOT NULL   
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T H
    ON A.AUTHORITYENTITYID = H.ROOT_ID2
   AND H.REPORT_PATH IS NOT NULL   
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T I
    ON A.AUTHORITYENTITYID = I.ROOT_ID3
   AND I.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T J
    ON A.AUTHORITYENTITYID = J.ROOT_ID4
   AND J.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T K
    ON A.AUTHORITYENTITYID = K.ROOT_ID5
   AND K.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T N
    ON A.AUTHORITYENTITYID = N.ROOT_ID6
   AND N.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T M
    ON A.AUTHORITYENTITYID = M.ROOT_ID7
   AND M.REPORT_PATH IS NOT NULL
  LEFT JOIN DM.DM_MD_FR_REPORT_INFO_T O
    ON A.AUTHORITYENTITYID = O.ROOT_ID8
   AND O.REPORT_PATH IS NOT NULL     
 WHERE A.AUTHORITY = 2
   AND A.AUTHORITYENTITYTYPE = 0
   AND A.AUTHORITYTYPE = 1
   AND COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(COALESCE(B.REPORT_NUM,F.REPORT_NUM),G.REPORT_NUM),H.REPORT_NUM),I.REPORT_NUM),J.REPORT_NUM),K.REPORT_NUM),N.REPORT_NUM),M.REPORT_NUM),O.REPORT_NUM) IS NOT NULL
)
INSERT INTO DM.DM_MD_FR_FINE_AUTHORITY_T(
       USERNAME
	  ,REALNAME
	  ,REPORT_ID
	  ,REPORT_NUM
	  ,REPORT_NAME
	  ,REPORT_PATH
	  ,REPORT_FULL_NAME
	  ,DEPT_NAME
	  ,DW_LAST_UPDATE_DATE
	  ,ROLE_TYPE_CODE --角色类型编码
	  ,ROLE_TYPE_NAME --角色类型描述
	  ,ROLE_ID --角色ID
	  ,ROLE_NAME --角色名称
	  ,TEMPL_CODE --模板编码
	  ,STAT_MONTH --统计月份
	  ,EXPANDID
	  ,SHOW_TYPE_CODE
	  ,SHOW_TYPE_NAME
)
SELECT 
	   T1.USERNAME
	  ,T1.REALNAME
	  ,T1.REPORT_ID
	  ,T1.REPORT_NUM
	  ,T1.REPORT_NAME
	  ,T1.REPORT_PATH
	  ,T1.REPORT_FULL_NAME
	  ,T1.DEPT_NAME
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
	  ,T1.ROLE_TYPE_CODE --角色类型编码
	  ,T1.ROLE_TYPE_NAME --角色类型描述
	  ,T1.ROLE_ID --角色ID
	  ,T1.ROLE_NAME --角色名称
	  ,CASE WHEN SUBSTR(SPLIT_PART(T1.REPORT_PATH,'/',LENGTH(T1.REPORT_PATH) - LENGTH(REPLACE(T1.REPORT_PATH,'/',''))+1),1,1) = 'M'
	        THEN SUBSTR(SPLIT_PART(T1.REPORT_PATH,'/',LENGTH(T1.REPORT_PATH) - LENGTH(REPLACE(T1.REPORT_PATH,'/',''))+1),1,6)
            WHEN SUBSTR(SPLIT_PART(T1.REPORT_PATH,'/',LENGTH(T1.REPORT_PATH) - LENGTH(REPLACE(T1.REPORT_PATH,'/',''))+1),1,1) IN('F','S','P','H')
	        THEN SUBSTR(SPLIT_PART(T1.REPORT_PATH,'/',LENGTH(T1.REPORT_PATH) - LENGTH(REPLACE(T1.REPORT_PATH,'/',''))+1),1,5)
	         END AS TEMPL_CODE --模板编码
	  ,'${PERIOD_ID_M}' AS STAT_MONTH --统计月份
	  ,T1.EXPANDID --扩展ID
	  ,T1.SHOW_TYPE_CODE --预览类型编码
	  ,T1.SHOW_TYPE_NAME --预览类型描述
  FROM T1
  LEFT JOIN DM.DM_WF_EMP_INFO_T T
    ON T1.USERNAME = T.EMP_NUM
 WHERE T.STATUS IN('0','1','2','3')
;
