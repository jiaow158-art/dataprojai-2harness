DELETE FROM DM_SW.DM_WM_INTRANSIT_RISK_STOCK_T WHERE stat_month = '${PERIOD_ID_M}';

INSERT INTO DM_SW.DM_WM_INTRANSIT_RISK_STOCK_T
with tmp as(
select substr("BUDAT" ,1,6) as 月份, "MATNR" as 物料编码,"MBLNR",c.purchase_voucher_type,
       sum(case when "BWART" in('101','911','162','123') and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) and (substr("LIFNR",1,6) <> '000000' or "LIFNR" is null )--排除内部供应商
       then  "MENGE" when "BWART" in('102','912','161','122') and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) and (substr("LIFNR",1,6) <> '000000' or "LIFNR" is null ) then  -"MENGE"  else 0 end ) as 入库,--排除库存转移
       sum(case when "BWART" in('101','911','162','123') and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) and (substr("LIFNR",1,6) <> '000000' or "LIFNR" is null ) 
       then  "DMBTR" when "BWART" in('102','912','161','122') and ( c.purchase_voucher_type <> 'ZSUB' or c.purchase_voucher_type is null ) and (substr("LIFNR",1,6) <> '000000' or "LIFNR" is null ) then  -"DMBTR"  else 0 end ) as 入库金额,
       sum(case when "BWART" in('601','654') then "MENGE" when "BWART" in('602','653') then  -"MENGE"  else 0 end ) as 销售出库,
       sum(case when "BWART" in('601','654') then  "DMBTR" when "BWART" in('602','653') then  -"DMBTR"  else 0 end ) as 销售出库金额,
       sum(case when "BWART" IN('933','951') then  "MENGE" when "BWART" IN('934','952') then  -"MENGE"  else 0 end ) as 其他出库,
       sum(case when "BWART" IN('933','951') then  "DMBTR" when "BWART" IN('934','952') then  -"DMBTR"  else 0 end ) as 其他出库金额,
       sum(case when "BWART" not IN('101','911','102','912', '601','602','933','951','934','952' ) then case when "SHKZG" = 'S' then "MENGE"  else  -"MENGE"  end else 0 end) 变动
from dm.dm_rpt_bic_azemm_o0032_t a
join dm.dm_rpt_bathroom_risk_inventory_list b  on  substr("BUDAT" ,1,6) = b.calmonth  and a."MATNR" = b.material_num and (a."LGORT"  =b.store_loc  or  b.store_loc = 'all') 
left join (select distinct purchase_voucher_num,purchase_voucher_type from DWRPUR.DWR_PTP_PURCHASE_VCH_HEADER_F) c on  a."EBELN" = c.purchase_voucher_num 
where "WERKS"  like '3%'
and "WERKS" not like '3A%'
and "WERKS" not like '39%'
and substr("BUDAT" ,1,6)='${PERIOD_ID_M}' 
and "LGORT" is not null
and ("KUNNR" not in ('0000322217') or "KUNNR" is null) and "MATNR" = 'W114105TYQ'
group by substr("BUDAT" ,1,6), "MBLNR", "MATNR" ,c.purchase_voucher_type
),
mat as (
select b."MATERIAL" as material_num,
case when substr(b."/BIC/ZPRODH3",1,1) in ('9','7')  then '配件' 
	 when substr(b."/BIC/ZPRODH3",1,2) in ('31','41')  then '配件' 
	 when substr(b."/BIC/ZPRODH3",1,4) = '4011' then '整装'
	 else k.prod_type end as category_name
from dm.dm_rpt_bi0_pmaterial_t as b 
left join upload.upload_wm_sw_sap_prod_grp_t as k on b."/BIC/ZPRODH3" = k.prod_code3 
),
ZT as (
select 
 material_num as 物料编码
--,stat_month
,category_name as 品类
,transit_num as 在途数量
,transit_amount as 在途金额
from dm.dm_transit_inventory_t 
where 1 = 1 and stat_month = '${PERIOD_ID_M}'
)
,

tmp0 as (
select c.prod_code as 物料编码,m.category_name as 品类, 
	   sum(c.sales_cost)/10000 as 销售出库金额
from dm.dm_rpt_bathroom_prodsales_s as c 
left join mat m on c.prod_code = m.material_num
join dm.dm_rpt_bathroom_risk_inventory_list b  on c.calendar_year_month = b.calmonth  and c.prod_code = b.material_num 
where c.calendar_year_month = '${PERIOD_ID_M}' and c.seller_to_party <> '0000322217' and ( substr(seller_to_party,1,6) <> '000000' or seller_to_party is null )
group by c.prod_code,m.category_name 
),
tmp1 as (  
select c.prod_code as 物料编码,m.category_name as 品类, 
	   sum( no_sto_sales_qty + sto_qty ) as 捐赠数量,
	   sum((c.no_sto_sales_qty + c.sto_qty)*c.unit_cost) as 捐赠金额
from dm.dm_rpt_bathroom_prodsales_s as c 
left join mat m on c.prod_code = m.material_num
join dm.dm_rpt_bathroom_risk_inventory_list b  on c.calendar_year_month = b.calmonth  and c.prod_code = b.material_num 
where c.calendar_year_month = '${PERIOD_ID_M}' and c.seller_to_party = '0000322217' 
group by c.prod_code,m.category_name
),
tmp2 as(
select coalesce(t1.物料编码, t2.物料编码) as 物料编码,
       coalesce(t1.品类, t2.品类) as 品类,
       coalesce(t2.月末库存数量,0) as 期初库存数量,
       t2.月末库存金额/10000 as 期初库存金额,
       coalesce(t1.月末库存数量,0) as 期末库存数量,
       t1.月末库存金额/10000 as 期末库存金额,
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.入库 else 0 end as 入库, 
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.销售出库 else 0 end as 销售出库, 
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.其他出库 else 0 end as 其他出库,
--       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.捐赠数量 else 0 end as 捐赠数量,
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.入库金额/10000 else 0 end as 入库金额, 
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.销售出库金额/10000 else 0 end as 销售出库金额, 
       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.其他出库金额/10000 else 0 end as 其他出库金额,
--       case when t1.物料编码 is not null and t2.物料编码 is not null then  t3.捐赠金额/10000 else 0 end as 捐赠金额,
       case when t1.物料编码 is null  then t2.月末库存数量 else 0 end as 型号剔除数量,
       case when t1.物料编码 is null  then t2.月末库存金额 else 0 end as 型号剔除金额,
       case when t2.物料编码 is null  then t1.月末库存数量 else 0 end as 型号新增数量,
       case when t2.物料编码 is null  then t1.月末库存金额 else 0 end as 型号新增金额,
       t2.月末库存金额/10000 as 上月库存金额,
       t1.月末库存金额/10000 as 本月库存金额
  from (
 SELECT b.material_num 物料编码,m.category_name as 品类,sum(月末库存金额+月末外协寄售金额) 月末库存金额,sum(月末库存数量+月末外协寄售数量) 月末库存数量 
 FROM   dm.dm_rpt_bathroom_risk_inventory_list b 
 left join mat m on b.material_num = m.material_num
  left join dm.dm_own_inventory_t a 
 on a.库存月份 = b.calmonth  and a.物料编码 = b.material_num and (a.库存地点 = b.store_loc  or  b.store_loc = 'all') and a."工厂" not like '39%'
 where b.calmonth='${PERIOD_ID_M}'  -- 所选月份
 group by   b.material_num ,m.category_name 
 ) t1 full outer  join 
  (  SELECT b.material_num 物料编码,m.category_name as 品类,sum(月末库存金额+月末外协寄售金额) 月末库存金额,sum(月末库存数量+月末外协寄售数量) 月末库存数量 
 FROM   dm.dm_rpt_bathroom_risk_inventory_list b 
 left join mat m on b.material_num = m.material_num
  left join dm.dm_own_inventory_t a 
 on a.库存月份 = b.calmonth  and a.物料编码 = b.material_num and (a.库存地点 = b.store_loc  or  b.store_loc = 'all') and a."工厂" not like '39%'
 where b.calmonth=replace(substring(add_months(to_date('${PERIOD_ID_M}','YYYYMM'),-1) ,1,7),'-','')  -- 所选月份上一月份
 group by   b.material_num ,m.category_name
  
 ) t2 on t1.物料编码  = t2.物料编码
left join (
 select 月份, 物料编码,
       sum(入库) as 入库,
       sum(销售出库 ) as 销售出库,
       sum(其他出库 ) as 其他出库,
       sum(入库金额) as 入库金额,
       sum(销售出库金额 ) as 销售出库金额,
       sum(其他出库金额 ) as 其他出库金额
from tmp
group by 月份,物料编码
 ) t3 on coalesce (t1.物料编码,t2.物料编码) = t3.物料编码
 )
 ,
 tmp3 as(
 select 
 a.物料编码,a.品类,期初库存数量,期初库存金额 ,期末库存数量,期末库存金额,coalesce(b.捐赠数量,0) as 捐赠数量,coalesce(b.捐赠金额,0) as 捐赠金额,
      入库,销售出库, 其他出库,case when 期末库存数量 <> 0 then 入库*(期末库存金额/期末库存数量) else 入库金额 end as 入库金额
      ,c.销售出库金额, case when 期末库存数量 <> 0 then 其他出库*(期末库存金额/期末库存数量) else 其他出库金额 end as 其他出库金额
      ,型号剔除数量,型号新增数量,上月库存金额,本月库存金额,型号剔除金额,型号新增金额,d.在途数量,d.在途金额  
 from tmp2 a  
 left join tmp1 b on a.物料编码 = b.物料编码 
 left join tmp0 c on a.物料编码 = c.物料编码 
 left join ZT d on a.物料编码 = d.物料编码 

 ) 
 ,
 tmp4 as (
  select 物料编码,品类,期初库存数量,期初库存金额 ,期末库存数量,期末库存金额,捐赠数量,捐赠金额,在途数量,在途金额,
      入库,销售出库, 其他出库,入库金额,销售出库金额,其他出库金额,型号剔除数量,型号新增数量,上月库存金额,本月库存金额,
      (期末库存数量-期初库存数量)-(入库+型号新增数量-型号剔除数量-销售出库-其他出库-捐赠数量) as 调节,
      (期末库存金额-期初库存金额)-(入库金额+型号新增金额-型号剔除金额-销售出库金额-其他出库金额-捐赠金额) as 金额调节  
 from tmp3
 )
 select '${PERIOD_ID_M}' as stat_month, -- 月份
		物料编码 as material_num, -- 物料编码
		品类 as category_name, -- 品类
		期初库存数量 as begin_period_stock_qty, -- 期初库存数量
		期初库存金额 as begin_period_stock_amt, -- 期初库存金额
		期末库存数量 as end_period_stock_qty, -- 期末库存数量
		期末库存金额 as end_period_stock_amt, -- 期末库存金额
		捐赠数量 as donation_qty, -- 捐赠数量
		捐赠金额 as donation_amt, -- 捐赠金额
		调节 as adjust_qty, -- 调节数量
		金额调节 as adjust_amt, -- 调节金额
		在途数量 as intransit_qty, -- 在途数量
		在途金额 as intransit_amt, -- 在途金额
        入库+case when 调节>0 then 调节 else 0 end as IN_STOCK_QTY, -- 入库
		销售出库 as sales_out_stock_qty, --  销售出库
		其他出库+case when 调节<0 then -调节 else 0 end as other_out_stock_qty, -- 其他出库
		型号剔除数量 as model_delete_qty, -- 型号剔除数量
		型号新增数量 as model_add_qty, -- 型号新增数量
		上月库存金额 as af_stock_amt, -- 上月库存金额
		本月库存金额 as curr_stock_amt, -- 本月库存金额
        入库金额+case when 金额调节>0 then 金额调节 else 0 end as IN_STOCK_AMT, -- 入库金额
        销售出库金额 as sales_out_stock_amt,  -- 销售出库金额
        其他出库金额+case when 金额调节<0 then -金额调节 else 0 end as other_out_stock_amt, -- 其他出库金额
        current_timestamp() as dw_last_update_date --  数据同步时间
      from tmp4;
