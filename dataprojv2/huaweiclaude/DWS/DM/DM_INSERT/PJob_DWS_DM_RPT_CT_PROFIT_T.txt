DELETE FROM DM.DM_RPT_CT_PROFIT_T WHERE MONTHS = substr('${PERIOD_ID_D}',1,7);
with dim_date as (
 select
 '本月' months_type
,substr(day_id,1,6) months 
from dm.dm_dim_date_d where '${PERIOD_ID_D}' = day_dt
group by substr(day_id,1,6)
union all 
 select
 '上月' months_type
, substr(day_id_mend_mago,1,6) months 
from dm.dm_dim_date_d where '${PERIOD_ID_D}' = day_dt
group by substr(day_id_mend_mago,1,6)
union all 
 select
 '去年' months_type
, substr(day_id_mend_yago,1,6) months 
from dm.dm_dim_date_d where '${PERIOD_ID_D}'  = day_dt
group by substr(day_id_mend_yago,1,6)
)

select
'毛利率A-含清仓' index_type 
,case when this_month_notax_sales_net_amt = 0 then 0
    else this_month_gross_profit / this_month_notax_sales_net_amt 
 end this_month
, 0 budget
,case when last_month_notax_sales_net_amt = 0 then 0
    else last_month_gross_profit / last_month_notax_sales_net_amt
 end last_month
,case when last_year_notax_sales_net_amt = 0 then 0
    else last_year_gross_profit / last_year_notax_sales_net_amt
 end last_year
FROM
(
select
sum(this_month_gross_profit) this_month_gross_profit
,sum(this_month_notax_sales_net_amt) this_month_notax_sales_net_amt
,sum(last_month_gross_profit)  last_month_gross_profit
,sum(last_month_notax_sales_net_amt) last_month_notax_sales_net_amt
,sum(last_year_gross_profit) last_year_gross_profit
,sum(last_year_notax_sales_net_amt) last_year_notax_sales_net_amt
from  (
	select 
   sum(case when months_type = '本月' then gross_profit else 0 end )              this_month_gross_profit
  ,sum(case when months_type = '上月' then gross_profit else 0 end )       		 last_month_gross_profit
  ,sum(case when months_type = '本月' then last_year_gross_profit else 0 end )    last_year_gross_profit
  ,sum(case when months_type = '本月' then notax_sales_net_amt else 0 end )              this_month_notax_sales_net_amt
  ,sum(case when months_type = '上月' then notax_sales_net_amt else 0 end )       		 last_month_notax_sales_net_amt
  ,sum(case when months_type = '本月' then last_year_notax_sales_net_amt else 0 end )    last_year_notax_sales_net_amt
from DWRFIN.DWRFIN_COST_SALES_GROSS_PROFIT_D t1 
join dim_date t3 on t1.months  = t3.months 
where 1=1
union all 
select 
  sum(case when months_type = '本月' and trim(index_type) ='毛利A-含清仓' then indicator_value else 0 end  )    this_month_gross_profit
 ,sum(case when months_type = '上月' and trim(index_type) ='毛利A-含清仓' then indicator_value  else 0 end  )    last_month_gross_profit
 ,sum(case when months_type = '去年' and trim(index_type) ='毛利A-含清仓' then indicator_value else 0 end  )    last_year_gross_profit
 ,sum(case when months_type = '本月' and trim(index_type) ='收入净额' then indicator_value else 0 end  )    this_month_notax_sales_net_amt
 ,sum(case when months_type = '上月' and trim(index_type) ='收入净额' then indicator_value  else 0 end  )    last_month_notax_sales_net_amt
 ,sum(case when months_type = '去年' and trim(index_type) ='收入净额' then indicator_value else 0 end  )    last_year_notax_sales_net_amt
from 
upload.upload_ct_internal_transaction t1 join dim_date t3 on t1.months  = t3.months 

)
)
