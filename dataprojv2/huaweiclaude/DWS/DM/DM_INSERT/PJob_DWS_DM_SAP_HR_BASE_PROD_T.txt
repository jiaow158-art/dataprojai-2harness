TRUNCATE TABLE DM.DM_SAP_HR_BASE_PROD_T;

WITH T1 AS(
SELECT STAT_MONTH --统计月份
      ,DEPT_NAME --部门名称
	  ,CAPACITY AS PROD_QTY --产量
FROM UPLOAD.UPLOAD_EHR_MONTHLY_PER_CAPACITY_T
),T2 AS(
SELECT A.STAT_MONTH --统计月份
	  ,CASE WHEN A.LVL5_DEPT_CODE = '00008165' or A.LVL6_DEPT_CODE = '00008165' THEN '山西基地'
	        WHEN A.LVL5_DEPT_CODE = '00002000' or A.LVL6_DEPT_CODE = '00002000' THEN '清远基地'
			WHEN A.LVL5_DEPT_CODE = '00005019' or A.LVL6_DEPT_CODE = '00005019' THEN '江西基地'
			WHEN A.LVL5_DEPT_CODE = '00006963' or A.LVL6_DEPT_CODE = '00006963' THEN '重庆基地'
			WHEN A.LVL5_DEPT_CODE = '00005021' or A.LVL6_DEPT_CODE = '00005021' THEN '湖南基地'
			WHEN A.LVL5_DEPT_CODE = '00004500' or A.LVL6_DEPT_CODE = '00004500' THEN '高明基地'
			WHEN A.LVL5_DEPT_CODE = '00004200' or A.LVL6_DEPT_CODE = '00004200' THEN '江西基地'
			WHEN A.LVL5_DEPT_CODE = '00007032' or A.LVL6_DEPT_CODE = '00007032' THEN '江门基地'
			END AS DEPT_NAME --部门名称
      ,COUNT(DISTINCT EMP_NUM) AS EMP_CNT --员工人数
	  ,A.LVL3_DEPT_NAME AS UP_DEPT_NAME --上级部门名称
  FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F A
 WHERE COALESCE(A.FX_PAY_AMT,0) + COALESCE(A.FLT_PAY_AMT,0) + COALESCE(A.BNS_AMT,0)
     + COALESCE(A.HR_WELFARE_AMT,0) + COALESCE(A.PAYROLL_AMT,0) + COALESCE(A.SVRNC_AMT,0) > 0
   AND A.LVL5_DEPT_CODE IN('00008165','00002000','00005019','00006963','00005021','00004500','00004200','00007032') or 
       A.LVL6_DEPT_CODE IN('00008165','00002000','00005019','00006963','00005021','00004500','00004200','00007032')
 GROUP BY
       A.STAT_MONTH --统计月份
      ,CASE WHEN A.LVL5_DEPT_CODE = '00008165' or A.LVL6_DEPT_CODE = '00008165' THEN '山西基地'
	        WHEN A.LVL5_DEPT_CODE = '00002000' or A.LVL6_DEPT_CODE = '00002000' THEN '清远基地'
			WHEN A.LVL5_DEPT_CODE = '00005019' or A.LVL6_DEPT_CODE = '00005019' THEN '江西基地'
			WHEN A.LVL5_DEPT_CODE = '00006963' or A.LVL6_DEPT_CODE = '00006963' THEN '重庆基地'
			WHEN A.LVL5_DEPT_CODE = '00005021' or A.LVL6_DEPT_CODE = '00005021' THEN '湖南基地'
			WHEN A.LVL5_DEPT_CODE = '00004500' or A.LVL6_DEPT_CODE = '00004500' THEN '高明基地'
			WHEN A.LVL5_DEPT_CODE = '00004200' or A.LVL6_DEPT_CODE = '00004200' THEN '江西基地'
			WHEN A.LVL5_DEPT_CODE = '00007032' or A.LVL6_DEPT_CODE = '00007032' THEN '江门基地'
			END --部门名称
	  ,A.LVL3_DEPT_NAME --上级部门名称		

 UNION ALL

SELECT A.STAT_MONTH --统计月份
      ,CAST(CONCAT(A.LVL3_DEPT_NAME,'制造基地') AS VARCHAR) AS DEPT_NAME --部门名称
      ,COUNT(DISTINCT EMP_NUM) AS EMP_CNT --员工人数
      ,CAST('东鹏集团' AS VARCHAR) AS UP_DEPT_NAME --上级部门名称
  FROM DWRDIM.DWR_DIM_SAP_EMP_COST_F A
 WHERE COALESCE(A.FX_PAY_AMT,0) + COALESCE(A.FLT_PAY_AMT,0) + COALESCE(A.BNS_AMT,0)
     + COALESCE(A.HR_WELFARE_AMT,0) + COALESCE(A.PAYROLL_AMT,0) + COALESCE(A.SVRNC_AMT,0) > 0
   AND A.LVL5_DEPT_CODE IN('00008165','00002000','00005019','00006963','00005021','00004500','00004200','00007032') or 
       A.LVL6_DEPT_CODE IN('00008165','00002000','00005019','00006963','00005021','00004500','00004200','00007032')
 GROUP BY
       A.STAT_MONTH --统计月份
	  ,A.LVL3_DEPT_NAME
),T3 AS(
SELECT STAT_MONTH --统计月份
      ,BASE_NAME AS DEPT_NAME --部门名称
	  ,DEPT_NAME AS UP_DEPT_NAME --上级部门名称
	  ,PROD_AMT --产值（万元）
	  ,EMP_CNT --员工人数
  FROM UPLOAD.UPLOAD_EHR_BASE_PROD_T
)

-- (
--       STAT_MONTH --统计月份
-- 	  ,DEPT_NAME --部门名称
-- 	  ,PROD_QTY --产量
--       ,EMP_CNT --员工人数
-- 	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
-- 	  ,UP_DEPT_NAME --上级部门名称
-- 	  ,DATA_TYPE --统计类型
-- )
,T4 AS (
SELECT A.STAT_MONTH --统计月份
	  ,A.DEPT_NAME --部门名称
	  ,B.PROD_QTY --产量
      ,A.EMP_CNT --员工人数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.UP_DEPT_NAME --上级部门名称
	  ,CAST('产能' AS VARCHAR) AS DATA_TYPE --统计类型
  FROM T2 A
  INNER JOIN T1 B
    ON A.STAT_MONTH = B.STAT_MONTH
   AND A.DEPT_NAME = B.DEPT_NAME
 WHERE A.UP_DEPT_NAME = '瓷砖事业部'  
   
 UNION ALL

SELECT A.STAT_MONTH --统计月份
	  ,A.DEPT_NAME --部门名称
	  ,A.PROD_AMT --产值（万元）
      ,COALESCE(B.EMP_CNT,A.EMP_CNT) AS EMP_CNT --员工人数
	  ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,A.UP_DEPT_NAME --上级部门名称
	  ,CAST('产值' AS VARCHAR) AS DATA_TYPE --统计类型
  FROM T3 A
  LEFT JOIN T2 B
    ON A.STAT_MONTH = B.STAT_MONTH
   AND A.DEPT_NAME = B.DEPT_NAME
   AND A.UP_DEPT_NAME = B.UP_DEPT_NAME
   AND B.STAT_MONTH >= '2023-01'
)
INSERT INTO DM.DM_SAP_HR_BASE_PROD_T
SELECT STAT_MONTH --统计月份
	  ,DEPT_NAME --部门名称
	  ,PROD_QTY --产值（万元）
      ,EMP_CNT --员工人数
	  ,DW_LAST_UPDATE_DATE --DW数据同步时间
	  ,UP_DEPT_NAME --上级部门名称
	  ,DATA_TYPE --统计类型
	  ,CASE WHEN DEPT_NAME = '山西基地' THEN '1900'
	        WHEN DEPT_NAME = '清远基地' THEN '1300'
	        WHEN DEPT_NAME = '江西基地' AND UP_DEPT_NAME = '瓷砖事业部' THEN '1500'
	        WHEN DEPT_NAME = '江西基地' AND UP_DEPT_NAME = '卫浴事业部' THEN '3200'
	        WHEN DEPT_NAME = '重庆基地' THEN '1800'
	        WHEN DEPT_NAME = '湖南基地' THEN '1600'
	        WHEN DEPT_NAME = '高明基地' THEN '3700'
	        WHEN DEPT_NAME = '江门基地' THEN '3500'
	        ELSE '' END DEPT_CODE
  FROM T4
