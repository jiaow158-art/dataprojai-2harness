DELETE FROM DM.DM_B1_WYJX_INVENTORY_TAT_T
WHERE SUBSTR(STAT_DATE,1,10) >= SUBSTR(CAST('${PERIOD_ID_D}' AS DATE)  - INTERVAL '100 DAY',1,10)
;

WITH T1 AS(
SELECT
        B.DAY_DT AS STAT_DATE
       ,SUBSTR(B.DAY_DT,1,7) AS STAT_MONTH
	   ,A.CUST_BYNAME
	   ,A.MATERIAL_TYPE
	   ,A.DEALER_CODE
	   ,A.DEALER_NAME
	   ,REPLACE(A.PRODUCT_STATUS,'SNULL','') AS PRODUCT_STATUS
	   ,A.MATERIAL_NUM
       ,SUM(COALESCE(C.COST_TOTAL_AMT,0)) OVER(PARTITION BY SUBSTR(B.DAY_DT,1,7),A.CUST_BYNAME
	                                           ,A.MATERIAL_TYPE,A.DEALER_NAME,A.DEALER_CODE
											   ,A.PRODUCT_STATUS,A.MATERIAL_NUM
									   ORDER BY B.DAY_DT ASC
	                              ) AS COST_TOTAL_AMT_CUR_M --当月销售成本
       ,SUM(COALESCE(C.COST_TOTAL_AMT,0)) OVER(PARTITION BY SUBSTR(B.DAY_DT,1,4),A.CUST_BYNAME
	                                           ,A.MATERIAL_TYPE,A.DEALER_NAME,A.DEALER_CODE
											   ,A.PRODUCT_STATUS,A.MATERIAL_NUM
									   ORDER BY B.DAY_DT ASC
	                              ) AS COST_TOTAL_AMT_CUR_Y	--当年销售成本						  
  FROM(SELECT DAY_DT
         FROM DM.DM_DIM_DATE_D
        WHERE YEAR_ID >= SUBSTR(CAST('${PERIOD_ID_D}' AS DATE)  - INTERVAL '100 DAY',1,4)
          AND DAY_DT <= TO_CHAR(CURRENT_DATE - INTERVAL '1 DAY','YYYY-MM-DD')) B
  LEFT JOIN(SELECT DISTINCT
                    CUST_BYNAME
                   ,MATERIAL_TYPE
                   ,DEALER_CODE
                   ,DEALER_NAME
                   ,CASE WHEN PRODUCT_STATUS IS NOT NULL THEN PRODUCT_STATUS
				         ELSE 'SNULL'
						  END AS PRODUCT_STATUS
                   ,MATERIAL_NUM
              FROM DM.DM_WM_B1_SALES_ORDER_T A
             WHERE A.IS_SW = 1
               AND DOC_DATE <= CURRENT_DATE - INTERVAL '1 DAY'
           ) A
    ON 1 = 1
  LEFT JOIN (SELECT SUBSTR(A.DOC_DATE,1,10) AS DOC_DATE
	               ,A.CUST_BYNAME
	               ,A.DEALER_CODE
	               ,A.MATERIAL_NUM
				   ,A.DEALER_NAME
				   ,CASE WHEN PRODUCT_STATUS IS NOT NULL THEN PRODUCT_STATUS
				         ELSE 'SNULL'
						  END AS PRODUCT_STATUS
			       ,SUM(A.COST_TOTAL_AMT) AS COST_TOTAL_AMT
              FROM DM.DM_WM_B1_SALES_ORDER_T A
             WHERE A.IS_SW = 1
		     GROUP BY
		            SUBSTR(A.DOC_DATE,1,10)
	               ,A.CUST_BYNAME
	               ,A.DEALER_CODE
	               ,A.MATERIAL_NUM
				   ,A.DEALER_NAME
				   ,CASE WHEN PRODUCT_STATUS IS NOT NULL THEN PRODUCT_STATUS
				         ELSE 'SNULL'
						  END 
	        ) C
	ON SUBSTR(B.DAY_DT,1,10) = SUBSTR(C.DOC_DATE,1,10)
   AND A.CUST_BYNAME = C.CUST_BYNAME
   AND A.DEALER_CODE = C.DEALER_CODE
   AND A.MATERIAL_NUM = C.MATERIAL_NUM
   AND A.DEALER_NAME = C.DEALER_NAME
   AND A.PRODUCT_STATUS = C.PRODUCT_STATUS
),T2 AS(
SELECT  
       A.STAT_DATE
	  ,SUBSTR(A.STAT_DATE,1,7) AS STAT_MONTH
	  ,TO_DATE(SUBSTR(A.STAT_DATE,1,8)||'01','YYYY-MM-DD') - INTERVAL '1 DAY' AS LAST_DATE
	  ,SUBSTR(TO_DATE(SUBSTR(A.STAT_DATE,1,8)||'01','YYYY-MM-DD') - INTERVAL '1 DAY',1,7) AS LAST_MONTH
	  ,A.CUST_BYNAME
	  ,A.MATERIAL_TYPE
	  ,A.DEALER_CODE
	  ,A.DEALER_NAME
	  ,CASE WHEN A.PRODUCT_STATUS IS NULL THEN 'SNULL' ELSE A.PRODUCT_STATUS END AS PRODUCT_STATUS
	  ,A.MATERIAL_NUM
      ,SUM(A.INVENTORY_AMOUNT) QM_AMOUNT --期末金额
  FROM DM.DM_WM_B1_WYJX_INVENTORY_T A
 WHERE A.IS_SW = 1
   AND SUBSTR(A.STAT_DATE,1,10) >= CONCAT(SUBSTR(CAST('${PERIOD_ID_D}' AS DATE)  - INTERVAL '100 DAY',1,4),'-12-31')
 GROUP BY
       A.STAT_DATE
	  ,SUBSTR(A.STAT_DATE,1,7)
	  ,TO_DATE(SUBSTR(A.STAT_DATE,1,8)||'01','YYYY-MM-DD') - INTERVAL '1 DAY'
	  ,SUBSTR(TO_DATE(SUBSTR(A.STAT_DATE,1,8)||'01','YYYY-MM-DD') - INTERVAL '1 DAY',1,7)
	  ,A.CUST_BYNAME
	  ,A.MATERIAL_TYPE
	  ,A.DEALER_CODE
	  ,A.DEALER_NAME
	  ,CASE WHEN A.PRODUCT_STATUS IS NULL THEN 'SNULL' ELSE A.PRODUCT_STATUS END
	  ,A.MATERIAL_NUM)
,T3 AS(
SELECT  
       A.STAT_DATE
	  ,A.STAT_MONTH
	  ,CASE WHEN SUBSTR(A.STAT_DATE,6,5) = '12-31' THEN SUBSTR(A.STAT_DATE + INTERVAL '1 YEAR',1,4)
	        ELSE SUBSTR(A.STAT_DATE,1,4)
			 END AS STAT_YEAR
	  ,A.LAST_DATE
	  ,A.LAST_MONTH
	  ,A.CUST_BYNAME
	  ,A.MATERIAL_TYPE
	  ,A.DEALER_CODE
	  ,A.DEALER_NAME
	  ,A.PRODUCT_STATUS
	  ,A.MATERIAL_NUM
      ,A.QM_AMOUNT AS NC_AMOUNT --年期初金额
  FROM T2 A
 INNER JOIN(SELECT CUST_BYNAME
                  ,DEALER_CODE
				  ,DEALER_NAME
				  ,MIN(STAT_DATE) AS MIN_STAT_DATE
              FROM T2
			  WHERE STAT_DATE = LAST_DAY(STAT_DATE)
			 GROUP BY
			       CUST_BYNAME
                  ,DEALER_CODE
				  ,DEALER_NAME
           ) B
	ON A.CUST_BYNAME = B.CUST_BYNAME
   AND A.DEALER_CODE = B.DEALER_CODE
   AND A.DEALER_NAME = B.DEALER_NAME
   AND A.STAT_DATE = B.MIN_STAT_DATE
),T4 AS(
SELECT  SUBSTR(A.STAT_DATE,1,10) AS STAT_DATE
       ,A.STAT_MONTH
	   ,A.CUST_BYNAME
	   ,A.MATERIAL_TYPE
	   ,A.DEALER_CODE
	   ,A.DEALER_NAME
	   ,A.PRODUCT_STATUS
	   ,A.MATERIAL_NUM
	   ,A.COST_TOTAL_AMT_CUR_M --当月销售成本
	   ,A.COST_TOTAL_AMT_CUR_Y --当年销售成本
	   ,0 AS QM_AMOUNT --期末金额
	   ,0 AS YC_AMOUNT --月初金额
	   ,0 AS NC_AMOUNT --年期初金额
  FROM T1 A
  
 UNION ALL

SELECT  SUBSTR(A.STAT_DATE,1,10) AS STAT_DATE
       ,A.STAT_MONTH
	   ,A.CUST_BYNAME
	   ,A.MATERIAL_TYPE
	   ,A.DEALER_CODE
	   ,A.DEALER_NAME
	   ,REPLACE(A.PRODUCT_STATUS,'SNULL','') AS PRODUCT_STATUS
	   ,A.MATERIAL_NUM
	   ,0 AS COST_TOTAL_AMT_CUR_M --当月销售成本
	   ,0 AS COST_TOTAL_AMT_CUR_Y --当年销售成本
	   ,A.QM_AMOUNT --期末金额
	   ,B.QM_AMOUNT AS YC_AMOUNT --月初金额
	   ,0 AS NC_AMOUNT --年期初金额
  FROM T2 A
  LEFT JOIN T2 B
    ON A.CUST_BYNAME = B.CUST_BYNAME
   AND A.DEALER_CODE = B.DEALER_CODE
   AND A.MATERIAL_NUM = B.MATERIAL_NUM
   AND A.DEALER_NAME = B.DEALER_NAME
   AND A.PRODUCT_STATUS = B.PRODUCT_STATUS
   AND SUBSTR(A.LAST_DATE,1,10) = SUBSTR(B.STAT_DATE,1,10)
   
 UNION ALL

SELECT  SUBSTR(A.STAT_DATE,1,10) AS STAT_DATE
       ,A.STAT_MONTH
	   ,B.CUST_BYNAME
	   ,B.MATERIAL_TYPE
	   ,B.DEALER_CODE
	   ,B.DEALER_NAME
	   ,REPLACE(B.PRODUCT_STATUS,'SNULL','') AS PRODUCT_STATUS
	   ,B.MATERIAL_NUM
	   ,0 AS COST_TOTAL_AMT_CUR_M --当月销售成本
	   ,0 AS COST_TOTAL_AMT_CUR_Y --当年销售成本
	   ,0 AS QM_AMOUNT --期末金额
	   ,0 AS YC_AMOUNT --月初金额
	   ,NC_AMOUNT --年期初金额
  FROM (SELECT DISTINCT
               SUBSTR(A.STAT_DATE,1,10) AS STAT_DATE
              ,A.STAT_MONTH
          FROM T2 A) A
 INNER JOIN T3 B
    ON SUBSTR(A.STAT_DATE,1,4) = B.STAT_YEAR
)
INSERT INTO DM.DM_B1_WYJX_INVENTORY_TAT_T
(
        STAT_DATE
       ,STAT_MONTH
	   ,CUST_BYNAME
	   ,MATERIAL_TYPE
	   ,DEALER_CODE
	   ,DEALER_NAME
	   ,PRODUCT_STATUS
	   ,MATERIAL_NUM
	   ,COST_TOTAL_AMT_CUR_M --当月销售成本
	   ,COST_TOTAL_AMT_CUR_Y --当年销售成本
	   ,QM_AMOUNT --期末金额
	   ,YC_AMOUNT --月初金额
	   ,NC_AMOUNT --年期初金额
	   ,DW_LAST_UPDATE_DATE
)
SELECT  A.STAT_DATE
       ,A.STAT_MONTH
	   ,A.CUST_BYNAME
	   ,A.MATERIAL_TYPE
	   ,A.DEALER_CODE
	   ,A.DEALER_NAME
	   ,A.PRODUCT_STATUS
	   ,A.MATERIAL_NUM
	   ,SUM(A.COST_TOTAL_AMT_CUR_M) AS COST_TOTAL_AMT_CUR_M --当月销售成本
	   ,SUM(A.COST_TOTAL_AMT_CUR_Y) AS COST_TOTAL_AMT_CUR_Y --当年销售成本
	   ,SUM(A.QM_AMOUNT) AS QM_AMOUNT --期末金额
	   ,SUM(A.YC_AMOUNT) AS YC_AMOUNT --月初金额
	   ,SUM(A.NC_AMOUNT) AS NC_AMOUNT --年期初金额
	   ,CURRENT_TIMESTAMP AS DW_LAST_UPDATE_DATE
  FROM T4 A
 WHERE SUBSTR(A.STAT_DATE,1,10) >= SUBSTR(CAST('${PERIOD_ID_D}' AS DATE)  - INTERVAL '100 DAY',1,10) 
 GROUP BY
        A.STAT_DATE
       ,A.STAT_MONTH
	   ,A.CUST_BYNAME
	   ,A.MATERIAL_TYPE
	   ,A.DEALER_CODE
	   ,A.DEALER_NAME
	   ,A.PRODUCT_STATUS
	   ,A.MATERIAL_NUM
