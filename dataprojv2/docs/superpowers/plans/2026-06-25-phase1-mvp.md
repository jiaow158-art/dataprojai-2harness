"""Append-only JSONL audit logger. One file per day."""
from __future__ import annotations
import json
import os
import datetime
from pathlib import Path
from typing import Any


class AuditLogger:
    """Writes one JSON record per line to {log_dir}/{YYYY-MM-DD}.jsonl.

    Thread-safe via append mode. No rotation in P1 (manual cleanup).
    """

    def __init__(self, log_dir: str | Path):
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)

    def log(
        self,
        session_id: str,
        question: str,
        domain: str,
        sql: str,
        row_count: int,
        review_passed: bool,
        latency_ms: int,
        tldr: str,
        model: str,
        **extra: Any,
    ) -> str:
        """Write a single audit record. Returns the file path written to."""
        record = {
            "timestamp": datetime.datetime.now().isoformat(timespec="seconds"),
            "session_id": session_id,
            "question": question,
            "domain": domain,
            "sql": sql[:2000],  # truncate huge SQL
            "row_count": row_count,
            "review_passed": review_passed,
            "latency_ms": latency_ms,
            "tldr": tldr,
            "model": model,
            **extra,
        }
        filename = f"{datetime.date.today().isoformat()}.jsonl"
        filepath = self.log_dir / filename
        with filepath.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
        return str(filepath)
```

```python
"""COUNT(*) preflight — root-cause fix for ME2 「伪 1 行 NULL」.

Extracts FROM ... WHERE ... from the original SQL, runs a base-table COUNT,
returns the row count. Caller raises NoDataError when count == 0.

Returns -1 for unparseable queries (skip preflight, don't block).
"""
from __future__ import annotations
import re


class NoDataError(Exception):
    """Raised when preflight COUNT returns 0 — filter matches nothing."""


_FROM_WHERE_RE = re.compile(
    r"\bFROM\s+(schema\.\w+|\w+\.\w+|\w+)(?:\s+(?:AS\s+)?\w+)?\s+WHERE\s+(.*?)"
    r"(?:\s+(?:GROUP\s+BY|ORDER\s+BY|LIMIT|HAVING|UNION|INTERSECT|EXCEPT|;)\b|$)",
    re.IGNORECASE | re.DOTALL,
)


def make_count_sql(sql: str) -> str | None:
    """Build a COUNT(*) SQL from a simple single-table SELECT.

    Returns None for queries we can't safely transform (CTEs, JOINs, subqueries in FROM).
    """
    if re.search(r"\bWITH\b", sql, re.IGNORECASE):
        return None
    if re.search(r"\bJOIN\b", sql, re.IGNORECASE):
        return None
    m = _FROM_WHERE_RE.search(sql)
    if not m:
        return None
    table, where = m.group(1), m.group(2).strip().rstrip(";").strip()
    if not table or not where:
        return None
    return f"SELECT COUNT(*) AS cnt FROM {table} WHERE {where}"


async def count_preflight(mcp, sql: str) -> int:
    """Returns row count from base table matching the same WHERE.

    - count > 0: safe to run original SQL
    - count == 0: caller should raise NoDataError
    - count == -1: extraction failed, skip preflight
    """
    count_sql = make_count_sql(sql)
    if count_sql is None:
        return -1
    rows, _ = await mcp.run_query(count_sql)
    if not rows:
        return 0
    return int(rows[0].get("cnt", 0))
```

```python
"""KPI card structure for C-Level answer.

Output of step5 (INTERPRET_PROMPT). Defines a strict JSON schema the LLM
must produce, with safe parsing fallbacks.
"""
from __future__ import annotations
import json
import re
from dataclasses import dataclass, field


@dataclass
class KpiCard:
    tldr: list[str] = field(default_factory=list)
    kpis: list[dict] = field(default_factory=list)  # each dict has: label, value, unit?, yoy?, mom?, target?, rank?, direction?

    def to_dict(self) -> dict:
        return {"tldr": self.tldr, "kpis": self.kpis}

    def format_tldr_markdown(self) -> str:
        if not self.tldr:
            return ""
        bullets = "\n".join(f"- {s}" for s in self.tldr)
        return f"**TL;DR**\n\n{bullets}"


_REQUIRED_KPI_FIELDS = ("label", "value")
_OPTIONAL_KPI_FIELDS = ("unit", "yoy", "mom", "target", "rank", "direction")


def parse_llm_kpi_output(raw: str) -> KpiCard:
    """Parse LLM output into KpiCard. Tolerates ```json fences and bad JSON."""
    text = _strip_code_fence(raw).strip()
    try:
        data = json.loads(text)
    except json.JSONDecodeError:
        return KpiCard()

    if not isinstance(data, dict):
        return KpiCard()

    tldr = data.get("tldr", []) or []
    if not isinstance(tldr, list):
        tldr = [str(tldr)]

    raw_kpis = data.get("kpis", []) or []
    kpis = []
    for k in raw_kpis:
        if not isinstance(k, dict):
            continue
        if "label" not in k or "value" not in k:
            continue
        normalized = {f: k.get(f) for f in _REQUIRED_KPI_FIELDS}
        for f in _OPTIONAL_KPI_FIELDS:
            normalized[f] = k.get(f)
        kpis.append(normalized)

    return KpiCard(tldr=tldr, kpis=kpis)


def _strip_code_fence(text: str) -> str:
    if "```" not in text:
        return text
    parts = text.split("```")
    if len(parts) < 2:
        return text
    inner = parts[1]
    if inner.startswith("json"):
        inner = inner[4:]
    return inner
```

```python
"""Per-domain magnitude sanity check.

Identifies KPI values that fall outside business-common-sense ranges.
Returns warnings (not errors) — caller decides whether to block or annotate.
"""
from __future__ import annotations
import re
from webapp.orchestrator.kpi_card import KpiCard


# Each rule: regex on KPI label, then min/max for the value, plus a label for the warning.
# Sales-performance rules calibrated against metrics.md business ranges.
RULES: dict[str, list[dict]] = {
    "sales-performance": [
        {
            "label_regex": r"销售额|业绩|达成额|ambperformance|achievement|actual",
            "value_key": "value",
            "min": 1_000_000,       # < 100万 = 异常（瓷砖事业部月业绩正常 3-5亿）
            "max": 1e11,            # > 1000亿 = 异常
            "warning_label": "业绩金额",
        },
        {
            "label_regex": r"销售额|业绩|达成额|ambperformance|achievement|actual",
            "value_key": "yoy",
            "min": -0.8,
            "max": 2.0,
            "warning_label": "业绩同比",
        },
        {
            "label_regex": r"达成率|rate|pct",
            "value_key": "value",
            "min": 0.0,
            "max": 1.5,             # > 150% = 异常（目标分解滞后或字段错）
            "warning_label": "达成率",
        },
        {
            "label_regex": r"占比|share|proportion",
            "value_key": "value",
            "min": 0.0,
            "max": 1.0,
            "warning_label": "占比",
        },
    ],
}


def check_magnitude(domain: str, card: KpiCard) -> list[str]:
    """Return list of warning strings. Empty list = all values within common-sense range."""
    rules = RULES.get(domain, [])
    if not rules:
        return []

    warnings: list[str] = []
    for kpi in card.kpis:
        label = kpi.get("label", "")
        for rule in rules:
            if not re.search(rule["label_regex"], label, re.IGNORECASE):
                continue
            value = kpi.get(rule["value_key"])
            if not isinstance(value, (int, float)) or isinstance(value, bool):
                continue
            if value < rule["min"] or value > rule["max"]:
                field_name = rule["warning_label"]
                key_name = rule["value_key"]
                if key_name != "value":
                    field_name += f"（{key_name}）"
                warnings.append(
                    f"⚠️ KPI 「{label}」{key_name}={value} 超出常识范围 "
                    f"[{rule['min']}, {rule['max']}]（{field_name}），建议复核字段口径"
                )
    return warnings
```

    async def step4_execute(self, sql: str) -> tuple[list[dict], str]:
        # Preflight: COUNT(*) on base table with same WHERE. Catches ME2 「伪 1 行 NULL」.
        base_count = await count_preflight(self.mcp, sql)

            raise NoDataError(
                f"COUNT preflight returned 0 — filter matches no rows. "
                f"Check time range and WHERE conditions. SQL preview: {sql[:200]}"
            )
        # base_count == -1 means extraction failed (CTE/JOIN); skip preflight, proceed.

        rows, status = await self.mcp.run_query(sql)
        if "ERROR" in status.upper():
            # Gather schemas for any tables mentioned in the error or SQL
            extra_schemas = ""
            mentioned = set(re.findall(r'[\w]+\.[\w]+', f"{sql}\n{status}"))
            for t in mentioned:
                try:
                    cols = await self.mcp.describe_table(t)
                    if cols:
                        names = [(c["column_name"], c["data_type"]) for c in cols[:60]]
                        extra_schemas += f"\n表 {t}：{json.dumps(names, ensure_ascii=False)}"
                except Exception:
                    pass

            fixed = await self.llm.chat(
                system="修复以下SQL错误。只用列出的表，参考给定的列清单。不要JOIN。直接输出修复后的SQL，不要额外文字。" + extra_schemas,
                user=f"SQL：{sql}\n错误：{status}",
                temperature=0,
            )
            return await self.mcp.run_query(_extract_sql(fixed))

    session.add_message("assistant", result.analysis)

    # Audit
    latency = int((time.time() - start_time) * 1000)
    tldr = result.kpi_card.tldr[0] if result.kpi_card and result.kpi_card.tldr else ""
    app.state.audit.log(
        session_id=session.session_id, question=req.question, domain=route.domain,
        sql=result.sql, row_count=len(result.data),
        review_passed=result.review.passed, latency_ms=latency,
        tldr=tldr, model="deepseek-v4-pro",
        magnitude_warnings=result.magnitude_warnings,
    )

    return AskResponse(
        session_id=session.session_id,
        domain=route.domain,
        confidence=route.confidence,
        markdown=result.analysis,
        sql=result.sql,
        review_passed=result.review.passed,
        kpi_card=result.kpi_card.to_dict() if result.kpi_card else None,
        magnitude_warnings=result.magnitude_warnings,
        data_freshness=data_freshness,
        report_url=report_url,
        trace=result.trace,
    )


async def _get_data_freshness(mcp, sql: str) -> dict | None:
    """Extract main table from SQL, query its dw_last_update_date."""
    import re
    m = re.search(r'\bFROM\s+([\w\.]+)', sql, re.IGNORECASE)
    if not m:
        return None
    table = m.group(1)
    try:
        rows, _ = await mcp.run_query(
            f"SELECT dw_last_update_date as last_updated FROM {table} LIMIT 1"
        )
        if not rows:
            return None
        last_updated = rows[0].get("last_updated")
        if not last_updated:
            return None
        # Compute lag in days
        import datetime
        if isinstance(last_updated, str):
            try:
                # Try common formats
                for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d", "%Y%m%d"):
                    try:
                        parsed = datetime.datetime.strptime(last_updated[:19], fmt)
                        break
                    except ValueError:
                        continue
                else:
                    parsed = None
            except Exception:
                parsed = None
        elif isinstance(last_updated, datetime.datetime):
            parsed = last_updated
        else:
            parsed = None
        lag_days = (datetime.datetime.now() - parsed).days if parsed else None
        return {"table": table, "last_updated": str(last_updated), "lag_days": lag_days}
    except Exception:
        return None

        for c in CANDIDATES:
            rows, status = await mcp.run_query(c["sql"])
            entry = {
                **c,
                "row_count": len(rows),
                "elapsed_ms": 0,
                "data": rows[:5],
                "checksum": f"rows={len(rows)}",
            }
            new_entries.append(entry)
            print(f"✓ {c['pattern']}: {len(rows)} rows")
    finally:
        await mcp.stop()

    # Append to eval_dataset.json
    with open(EVAL_FILE, "r", encoding="utf-8") as f:
        data = json.load(f)
    data["questions"].extend(new_entries)
    data["meta"]["total"] = len(data["questions"])
    with open(EVAL_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
    print(f"Wrote {len(new_entries)} entries → {EVAL_FILE}")


if __name__ == "__main__":
    asyncio.run(main())
```
