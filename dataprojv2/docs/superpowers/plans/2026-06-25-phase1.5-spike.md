    # -- Step 1+2 (merged): Understand + Table Select --
    async def step1_understand_and_select(
        self, question: str, domain: str, history: list[dict] | None = None
    ) -> tuple[dict, dict, list[str]]:
        """Combined step1 (parse) + step2 (table choice) in one LLM call.

        Returns (parse_dict, table_dict, used_context_list).
        Saves one ~12s serial LLM call vs running step1 then step2.
        """
        skills = self.loader.get(domain)
        used = ["SKILL.md", "metrics.md"]
        system = skills.knowledge_md + "\n\n" + UNDERSTAND_AND_SELECT_PROMPT.format(metrics=skills.get_metrics())
        system += (
            "\n\n多轮对话规则：若当前问题是追问/下钻（如'按产品维度下钻'、'换个事业部看看'、'去年同期呢'），"
            "必须从上文继承未明确说出的 scope/time_range/metric。继承后在 ambiguities 里注明'继承自上文'。"
        )
        user_msg = _build_user_msg_with_history(question, history)
        raw = await self.llm.chat(system=system, user=user_msg, temperature=0.1)
        parsed = _parse_json(raw) or {
            "metric": question, "dimensions": [], "time_range": "", "scope": "",
            "table": "", "reason": "",
        }
        # Split flat dict into parse + table for downstream compat
        parse = {k: parsed.get(k, "") for k in ("metric", "dimensions", "time_range", "scope", "key_entities", "ambiguities")}
        table = {"table": parsed.get("table", ""), "reason": parsed.get("reason", "")}
        if history:
            used.append("history")
        return parse, table, used
```

import urllib.error

BASE = "http://127.0.0.1:8001"

QUESTIONS = [
    "本月集团达成情况？同比如何？",
    "瓷砖事业部本月业绩 vs 目标？",
    "5 大事业部本月排名？",
    "工程渠道本月同比变化？",
    "卫浴事业部 TOP10 客户？",
    "国际营销中心本月按渠道分布？",
    "丽适岩板本月达成率？",
    "瓷砖事业部高值产品占比？",
    "各事业部预算执行情况？",
    "瓷砖事业部 1-5 月趋势？",
]


def ask(question: str) -> tuple[dict, int, str]:
    body = json.dumps({"question": question}, ensure_ascii=False).encode("utf-8")
    req = urllib.request.Request(
        f"{BASE}/ask", data=body, headers={"Content-Type": "application/json"}
    )
    start = time.time()
    try:
        with urllib.request.urlopen(req, timeout=180) as r:
            data = json.loads(r.read().decode("utf-8"))
            return data, int((time.time() - start) * 1000), ""
    except urllib.error.HTTPError as e:
        return {}, int((time.time() - start) * 1000), f"HTTP {e.code}"
    except Exception as e:
        return {}, int((time.time() - start) * 1000), f"{type(e).__name__}"


def main():
    results = []
    for i, q in enumerate(QUESTIONS, 1):
        resp, latency, err = ask(q)
        is_nodata = (
            not err
            and resp.get("kpi_card") is None
            and any(t.get("name") == "execute" and t.get("error") == "no_data"
                    for t in resp.get("trace", []))
        )
        s = {
            "question": q,
            "latency_ms": latency,
            "no_data_error": is_nodata,
            "has_kpi_card": resp.get("kpi_card") is not None,
            "domain": resp.get("domain"),
            "trace_steps": [t.get("name") for t in resp.get("trace", [])],
            "error": err,
        }
        results.append(s)
        with open("D:/dataproj/webapp/tests/_clvl_p15_results.json", "w", encoding="utf-8") as f:
            json.dump(results, f, ensure_ascii=False, indent=2)
        print(f"[{i}/10] {q[:30]} → latency={latency}ms, no_data={is_nodata}", flush=True)

    n = len(results)
    nodata_count = sum(1 for r in results if r["no_data_error"])
    avg_lat = sum(r["latency_ms"] for r in results) // n if n else 0
    summary = {
        "total": n,
        "no_data_count": nodata_count,
        "no_data_rate": f"{nodata_count}/{n}",
        "avg_latency_ms": avg_lat,
        "max_latency_ms": max((r["latency_ms"] for r in results), default=0),
        "kpi_card_returned": f"{sum(1 for r in results if r['has_kpi_card'])}/{n}",
    }
    with open("D:/dataproj/webapp/tests/_clvl_p15_summary.json", "w", encoding="utf-8") as f:
        json.dump(summary, f, ensure_ascii=False, indent=2)
    print("\n=== Summary ===", flush=True)
    print(json.dumps(summary, ensure_ascii=False, indent=2), flush=True)


if __name__ == "__main__":
    main()
```
